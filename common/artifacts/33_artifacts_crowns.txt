# The Crown of Sigmar: gold and ivory, a gift of the Dwarfs, which the
# Ar-Ulric set on Sigmar's head at Reikdorf a year after Black Fire Pass.
# Sigmar left it on the table when he walked away from his throne. Its
# powers wake only for a crowned Emperor. Made at the first coronation of an
# Emperor (coronation_regalia_effect).
crown_of_sigmar = {
	picture = "GFX_crown_hre"
	slot = crown
	quality = 5
	stacking = no
	indestructible = yes

	active = {
		has_landed_title = e_the_empire
		custom_tooltip = { text = crown_of_sigmar_crowned_TT coronation_crowned_trigger = yes }
	}
	allowed_gift = { always = no }

	monthly_character_prestige = 1.0
	monthly_character_piety = 0.5
	diplomacy = 3
	vassal_opinion = 5
	church_opinion = 5

	flags = { crown unique jewelry crown_jewel empire }
}
