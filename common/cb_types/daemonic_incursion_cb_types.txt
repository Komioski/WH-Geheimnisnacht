# Daemonic incursion: a Chaos realm of the warp attacks a mortal ruler
# through a rift (events/daemonic_incursion_events.txt).
# ROOT = the incursion leader (warp governor), FROM = the mortal ruler.
daemonic_incursion = {
	name = CB_NAME_DAEMONIC_INCURSION
	war_name = WAR_NAME_DAEMONIC_INCURSION
	sprite = 30
	truce_days = 1825
	hostile_against_others = yes
	can_ask_to_join_war = no
	display_on_map = no
	allowed_to_target_tributaries = yes

	defender_unoccupied_warscore = yes

	can_use = {
		ROOT = {
			has_character_flag = daemonic_incursion_leader
			NOT = { same_realm = FROM }
		}
	}

	is_valid = {
		always = yes
	}

	on_success = { # The daemons ravage the realm
		FROM = {
			prestige = -200
			piety = -100
			add_character_modifier = {
				name = daemonic_incursion_ravaged
				years = 10
			}
		}
		ROOT = {
			clr_character_flag = daemonic_incursion_leader
			disband_event_forces = daemonic_incursion_host
		}
		add_doom_progress_medium_effect = yes
	}

	on_fail = { # The rift closes on its own
		FROM = {
			prestige = 100
		}
		ROOT = {
			clr_character_flag = daemonic_incursion_leader
			disband_event_forces = daemonic_incursion_host
		}
	}

	on_reverse_demand = { # The daemons are banished
		FROM = {
			piety = 300
			participation_scaled_prestige = 1000
			add_character_modifier = {
				name = daemonic_incursion_banished
				years = 20
			}
		}
		any_defender = {
			limit = { NOT = { character = FROM } }
			hidden_tooltip = {
				participation_scaled_piety = 150
				participation_scaled_prestige = 500
			}
		}
		any_defender = { # Emperor's triumph (LoR.30)
			limit = {
				has_dlc = "Legacy of Rome"
				has_landed_title = e_the_empire
				religion_group = old_world_group
			}
			hidden_tooltip = { character_event = { id = LoR.30 days = 12 } }
		}
		ROOT = {
			clr_character_flag = daemonic_incursion_leader
			disband_event_forces = daemonic_incursion_host
		}
		reduce_doom_progress_medium_effect = yes
	}

	on_attacker_leader_death = {
		end_war = invalid
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}
}
