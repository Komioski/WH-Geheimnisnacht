###New CB's###

#Reclaim Staff of Jade
reclaiming_the_staff_of_jade_cb = {
    name = CB_NAME_STAFF_OF_JADE
    war_name = WAR_NAME_STAFF_OF_JADE
    sprite = 8
	truce_days = 3650
    hostile_against_others = yes
	is_permanent = yes
	is_holy_war = yes
	allow_whitepeace = no
    check_all_titles = yes
	can_ask_to_join_war = yes
    defender_unoccupied_warscore = yes
    max_attacker_battle_score = 25


    can_use = {
        ROOT = {
            culture_group = lizardman_group
			primary_title = { has_title_flag = reclaimation_instigator }
            NOT = { same_realm = FROM }
            NOT = { is_liege_or_above = FROM }
        }
        FROM = {
			has_artifact = staff_of_jade
        }
    }

    can_use_title = {
        NAND = {
			location = { region = world_albion }
			ROOT = {
				has_global_flag = mists_of_albion
				capital_scope = { NOT = { region = world_albion } }
			}
		}
    }

	is_valid = {
        # The defender is not in any black crusades, otherwise the joining effect for defenders will join the wrong war
		FROM = {
			NOT = {
				any_war = {
					OR = {
						using_cb = new_crusade
						using_cb = greenskin_migration
						using_cb = waaagh_war
						using_cb = vlad_subjugation
					}
				}
			}
		}
        ROOT = {
            independent = yes
            religion = old_ones
            NOT = { same_realm = FROM }
        }
	}

    on_add = {
		FROM = { character_event = { id = antoch.2 } }
	}

    on_success = {
        ROOT = {
            piety = 1000
			add_artifact = staff_of_jade
        }
        FROM = {
            piety = -1000
            prestige = -1000
			unsafe_destroy_artifact = staff_of_jade
			primary_title = {
				if = {
					limit = {
						NOT = { has_landed_title = k_knights_of_antoch }
					}
					unsafe_destroy_landed_title = d_knights_of_antoch
					activate_title = { title = d_knights_of_antoch status = no }
				}
				else_if = {
					limit = {
						is_title_active = k_knights_of_antoch
					}
					unsafe_destroy_landed_title = k_knights_of_antoch
					activate_title = { title = k_knights_of_antoch status = no }
				}
			}
		}
		clr_global_flag = reclaiming_the_staff_of_jade
    }

    on_reverse_demand = {
        ROOT = {
            piety = -1000
			prestige = -1000
			primary_title = {
				if = {
					limit = {
						higher_tier_than = KING
					}
					destroy_landed_title = THIS
				}
			}
        }
        FROM = {
            piety = 1000
			prestige = 1000
        }
		clr_global_flag = reclaiming_the_staff_of_jade
    }

    attacker_ai_victory_worth = {
        factor = 200
    }

    attacker_ai_defeat_worth = {
        factor = 100
    }

    defender_ai_victory_worth = {
        factor = -1 # always accept
    }

    defender_ai_defeat_worth = {
        factor = 100
    }
}

#Southern Crusade
southern_crusade_war = {
	name = CB_NAME_SOUTHERN_CRUSADE
	war_name = WAR_NAME_SOUTHERN_CRUSADE
	sprite = 38
	truce_days = 1825
	is_permanent = yes
	is_holy_war = yes
	check_de_jure_tier = DUKE
	allow_whitepeace = no
	can_ask_to_join_war = yes

	defender_unoccupied_warscore = yes

	allowed_to_target_tributaries = yes

	can_use_gui = {
		ROOT = {
			prisoner = no
			is_adult = yes
		}
	}

	can_use = {
		ROOT = {
			religion = grail
			crusade_south_order_trigger = yes
		}
		FROM = {
			NOT = {
				religion_group = ROOT
			}
		}
	}
	can_use_title = {
		kingdom = {
			has_title_flag = northern_crusade_active_target
		}
	}
	is_valid = {
		FROM = {
			NOT = {
				religion = ROOT
			}
		}
	}
	on_success = {
		any_attacker = {
			limit = {
				NOR = {
					has_landed_title = d_knights_of_antoch
					has_landed_title = k_knights_of_antoch
				}
			}
			piety = 150
		}
		any_defender = {
			piety = -100
		}
		ROOT = { crusade_south_knights_errant_effect = yes }
	}
	on_success_title = {
		save_event_target_as = northern_crusade_title

		ROOT = {
			vassalize_or_take_under_title = {
				title = PREV
				enemy = FROM
				same_religion = yes # Only vassalize rulers of my religion
				is_religious = yes
				type = holy_war
			}

			hidden_tooltip = {
				# If the Knights of Antoch have no base to work out from, get one
				if = {
					limit = {
						capital_scope = {
							duchy = {
								title = PREVPREVPREV
							}
						}
					}
					teutonic_order_upgrade_capital_effect = yes
					holy_order_distribute_titles_effect = yes
					character_event = { id = HFP.41090 days = 1 }
					character_event = { id = HFP.41090 days = 3 }
				}

				# Become the Kingdom of Antoch, if they are not already
				if = {
					limit = {
						NOT = {
							has_landed_title = k_knights_of_antoch
						}
					}
					k_knights_of_antoch = {
						copy_title_history = d_knights_of_antoch
						copy_title_laws = d_knights_of_antoch
					}
					grant_title_no_opinion = k_knights_of_antoch
					unsafe_destroy_landed_title = d_knights_of_antoch
					activate_title = { title = d_knights_of_antoch status = no }
				}

				## Rewards
				# Check who to give artifacts
				any_attacker = {
					limit = {
						NOT = {
							character = ROOT
						}
						any_war = {
							using_cb = southern_crusade_war
							war_participation = {
								who = PREV
								score > 0.4
							}
						}
					}
					set_character_flag = northern_crusade_artifact_reward
				}
				# Check who to give money
				any_attacker = {
					limit = {
						NOT = {
							character = ROOT
						}
						any_war = {
							using_cb = southern_crusade_war
							war_participation = {
								who = PREV
								score > 0.2
							}
						}
					}
					set_character_flag = northern_crusade_gold_reward
				}
				# Check who to give artifacts
				any_attacker = {
					limit = {
						NOT = {
							character = ROOT
						}
						any_war = {
							using_cb = southern_crusade_war
							war_participation = {
								who = PREV
								score > 0.01
							}
						}
					}
					letter_event = { id = crusade.22 days = 1 }
				}
				# Checking for Bloodline
				most_participating_attacker = {
					if = {
						limit = {
							NOT = {
								character = ROOT
							}
						}

						if = {
							limit = {
								check_variable = {
									which = northern_crusade_ally
									value == 1
								}
							}
							set_variable = {
								which = northern_crusade_ally
								value = 2
							}
						}
						else_if = {
							limit = {
								check_variable = {
									which = northern_crusade_ally
									value = 2
								}
							}
							narrative_event = { id = crusade.24 }
						}
						else = {
							set_variable = {
								which = northern_crusade_ally
								value = 1
							}
						}
					}
				}

				## Misc
				# The kingdom is won when every county in it is held by the faithful
				if = {
					limit = {
						PREV = {
							kingdom = { crusade_south_kingdom_won_trigger = yes }
						}
					}
					PREV = {
						kingdom = { save_event_target_as = northern_crusade_kingdom }
					}
					crusade_south_victory_effect = yes
				}
			}
		}
	}
	on_reverse_demand = {

		ROOT = {
			any_attacker = {
				piety = -100
			}
		}

		FROM = {
			any_defender = {
				piety = 150
			}
		}

		hidden_tooltip = {
			ROOT = {
				if = {
					limit = {
						check_variable = {
							which = northernCrusadeValue
							value == 1
						}
					}
					set_variable = {
						which = northernCrusadeValue
						value = 2
					}
				}
				else_if = {
					limit = {
						check_variable = {
							which = northernCrusadeValue
							value = 2
						}
					}
					crusade_south_failure_effect = yes
				}
				else = {
					set_variable = {
						which = northernCrusadeValue
						value = 1
					}
				}
			}
		}
	}

	attacker_ai_victory_worth = {
		factor = -1
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1
	}

	defender_ai_defeat_worth = {
		factor = 100
	}
}
