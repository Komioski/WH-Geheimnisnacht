# Death texts: the line under a dead ruler's name in the succession screen.
#
# The game weighs every entry with weight_multiplier and picks one; the text
# is the localisation key of the same name. A modifier with factor = 0 rules
# an entry out.
#
# 00_death_text.txt is vanilla's file name and is kept empty so that it
# overrides vanilla's copy; everything is here.
#
# Sections: deeds, shame, lifestyle traits, body and mind, sins, virtues,
# wonders, children.
#
# "Holy warrior" (death_text_crusader) covers the mod's holy warrior and
# Solkan templar traits besides the vanilla holy war traits. "Sold their
# soul" (death_text_satanist) is the leader of any Chaos cult.
# death_text_empire_restored is for whoever restored the Empire or the
# Eternal Empire.

death_text_evil_wonder = {
	weight_multiplier = {
		factor = 10

		modifier = {
			factor = 0
			OR = {
				is_wicked = no
				is_virtous = yes
			}
		}
		modifier = {
			factor = 0
			NOT = {
				any_realm_wonder = {
					is_active = yes
					has_wonder_flag = burial_ground
				}
			}
		}
	}
}

			NOT = { has_ambition = obj_create_wonder }
		}
	}

death_text_architect_wonder = {
	weight_multiplier = {
		factor = 5

		modifier = {
			factor = 0
			NOT = {
				calc_true_if = {
					amount = 2
					trait = architect
					has_character_modifier = master_builder
					has_character_flag = created_wonder
					num_of_wonders = 3
				}
			}
		}
	}
}
