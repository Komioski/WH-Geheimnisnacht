###############################################################################
# WHG_disease.txt: epidemics
#
# Each entry is a disease that can break out in provinces and spread between
# them; "trait" is the illness a character gets. "rip = yes" means the entry is
# only used with The Reaper's Due (RIP), "rip = no" only without it, and no
# "rip" both. With RIP the game infects characters only through
# on_character_infection (rolled every character_infection_chances months);
# without RIP it gives the trait itself. The same key may be defined once for
# each case. Outbreaks happen in their timeperiod blocks only (the game starts
# in 2010).
#
# Sections:
#   The Reaper's Due diseases (7): rip = yes, with symptoms, treatment and the
#     minor_epidemics game rule
#   Plagues of the Old World (3): Nurgle's Rot, Rattlelung, Creeping Buboes,
#     in both versions of the game
#   Without The Reaper's Due (5): simple versions of the RIP diseases for
#     players without the DLC
#
# The vanilla-name files 00_disease.txt and 01_disease_old.txt in this folder
# are empty on purpose: they stop vanilla's real-world diseases from loading.
###############################################################################

###############################################################################
# The Reaper's Due diseases
###############################################################################

tuberculosis = { # Weevil Cough
	rip = yes
	contagiousness = 0.25
	outbreak_chance = 0.016
	effect = {
		local_tax_modifier = -0.25
		supply_limit = -2

	yearly_province_pulse = {
		random_list = {
			92 = {
				modifier = {
					factor = 1.1
					hospital_level = 1
				}
				modifier = {
					factor = 1.1
					hospital_level = 2
				}

					factor = 1.1

					factor = 1.1
					hospital_level = 6
				}
			}

				increase_depopulation_effect = yes
			}
		}
	}
	icon = 1
	tooltip = TUBERCULOSIS_INFO
	months = 30
	trait = has_tuberculosis
	character_infection_chances = {
		months = 3
	}
	on_character_infection = {
		if = {
			limit = {
				NOR = {
					has_character_flag = immune_tuberculosis
					has_character_flag = got_tuberculosis
					trait = has_tuberculosis
					trait = horse
					culture_group = daemonic_group
					culture_group = forest_group
					culture_group = goblin_group
					culture_group = orc_group
					culture_group = lizardman_group
					culture_group = monstrous_group
					culture_group = nehekharan_group
					is_vampire_trigger = yes
					has_character_flag = got_bubonic_plague
					has_character_flag = got_aztec_disease
					has_character_flag = got_smallpox
					has_character_flag = got_measles
					has_character_flag = got_typhus
					has_character_flag = got_typhoid_fever
					has_character_flag = got_syphilis
					has_character_flag = got_pneumonia
					has_character_flag = got_cancer
					has_character_flag = got_rabies
					has_character_flag = got_gout
					has_character_flag = got_flu
					has_character_flag = got_dysentery
					has_character_flag = got_food_poisoning
				}
			}
			random_list = {
				10 = { # Get the disease
					modifier = {
						factor = 0
						has_character_modifier = in_seclusion
					}
					modifier = {
						factor = 0.05
						OR = {
							religion = nurgle
							society_member_of = cult_of_nurgle
							trait = plaguelord
						}
					}
					modifier = {
						factor = 0.1
						is_nomadic = yes
						NOT = { is_unimportant_trigger = yes }
					}
					modifier = {
						factor = 0.5
						is_tribal = yes
						NOT = { is_unimportant_trigger = yes }
					}
					modifier = {
						factor = 50
						is_unimportant_trigger = yes
					}
					clr_character_flag = disease_not_serious
					set_character_flag = got_tuberculosis
					set_character_flag = developing_illness
					character_event = { id = symptom.16 }
				}
				90 = { # Nothing happens
				}
			}
		}
	}
	on_province_infection = {
		if = {
			limit = {
				OR = {
					culture = nehekharan
					culture_group = forest_group
					culture_group = monstrous_group
				}
				NOT = { religion = nurgle }
			}
			reduce_disease = 1
		}
		province_event = {
			id = prosper.4 # Prosperity ended by epidemic
		}
	}
	always_get_message = no
	color = { 84 171 176 }

	timeperiod = { # Minor Outbreak

		}

					}
					add_trait = has_creeping_buboes
					nurgle_blessing_check_effect = yes # Nurgle's faithful rejoice (WHG_chaos_effects)

					culture_group = monstrous_group
				}
				NOT = { religion = nurgle }
			}
			reduce_disease = 1
		}
		province_event = { id = prosper.4 } # Prosperity ended by epidemic
		province_event = { id = plague.6 } # Warn the players next door
		province_event = { id = plague.50 } # Strip it from immune races
	}

		start_date = 2010.7.1
		end_date = 3021.7.1

		max_total_provinces = 15

		one_only = no
	}
