nurgles_rot = {
	rip = no
	contagiousness = 0.7
	outbreak_chance = 0.005
	effect = {
	city_tax_modifier = -0.8
		supply_limit = -4
       	max_attrition = 0.05
        }
	# Great Plague: empties the land (Part 3 of WHG_disease_events tells the world)
	yearly_province_pulse = {
		random_list = {
			85 = {
				modifier = {
					factor = 1.1
					hospital_level = 1
				}
				modifier = {
					factor = 1.1
					hospital_level = 2
				}
				modifier = {
					factor = 1.1
					hospital_level = 3
				}
				modifier = {
					factor = 1.1
					hospital_level = 4
				}
				modifier = {
					factor = 1.1
					hospital_level = 5
				}
				modifier = {
					factor = 1.1
					hospital_level = 6
				}
			}
			15 = {
				modifier = {
					factor = 0
					religion = nurgle
				}
				increase_depopulation_effect = yes
			}
		}
	}
	on_province_infection = {
		if = {
			limit = {
				OR = {
					culture = nehekharan
					culture_group = forest_group
					culture_group = monstrous_group
				}
				NOT = { religion = nurgle }
			}
			reduce_disease = 1
		}
		province_event = { id = prosper.4 } # Prosperity ended by epidemic
		province_event = { id = plague.6 } # Warn the players next door
		province_event = { id = plague.50 } # Strip it from immune races
	}
	icon = 1 #TODO
	tooltip = NURGLES_ROT_INFO #TODO
	months = 7
        trait = has_nurgles_rot
	always_get_message = yes
	color = { 120 155 45 }

	timeperiod = {
		start_date = 2010.7.1
		end_date = 3021.7.1

		one_only = no
		
		max_total_provinces = 25
	}
}
