###############################################################################
# WHG_ambitions.txt - every ambition in the mod
#
# Merges the vanilla ambition files (00_ambitions, 00_zeus_ambitions,
# 00_HF_ambitions, 00_JD_ambitions, 00_LT_ambitions, left as empty stubs so
# the vanilla copies stay off) and wh_ambitions. Sorted by tier: an ambition
# is for that rank and above unless its note says "up to".
#
# Cheap rule: the AI weighs potential and allow for every character an
# ambition could suit, so they hold only cheap tests (age, tier, race, faith,
# culture, a scope or two). Searches and counts go in success, which is only
# checked for those holding the ambition. Ambitions about an action (a war, a
# battle, a murder, a build, a loot) read a flag or variable set by the hidden
# hook events ambition.200-219 (events/WHG_ambition_events.txt).
#
# Shared triggers: common/scripted_triggers/WHG_legend_triggers.txt
# Shared effects: common/scripted_effects/WHG_legend_effects.txt
# Loc: localisation/00_WHG_ambitions.csv
###############################################################################

# Master a skill (martial)
# Offered below martial 13; done at 15. The skill's chain is in
# WHG_ambition_events (namespace skill).
obj_improve_martial = {
	type = character

	potential = {
		age = 16
		is_playable = yes

	allow = {
		WHG_ambition_able_trigger = yes
		NOT = { martial = 13 }

		factor = 100

		modifier = { factor = 1.25 trait = misguided_warrior }
		modifier = { factor = 1.5 trait = ambitious }

			trait = incapable

		}

		clr_character_flag = toy_army
		clr_character_flag = street_ruffians
		clr_character_flag = warrior_mentor
		clr_character_flag = warrior_mentor_search
	}

		prestige = 100

		clr_character_flag = street_ruffians

# Have a son
# Married parents of peoples with women of their own; not the undead or
# daemons (WHG_ambition_can_have_children_trigger).
obj_have_a_son = {
	type = character

	fertility = 0.25

	allow = {
		WHG_ambition_able_trigger = yes
		WHG_ambition_can_have_children_trigger = yes
		is_marriage_adult = yes
		OR = {
			is_married = yes
			is_consort = yes
		}
		NOT = {
			any_child = {
				is_female = no
			}
		}
		OR = {
			ai = no
			is_female = no
			has_religion_feature = religion_patriarchal
			# Have at least two daughters
			any_child = {
				is_female = yes
				ROOT = {
					any_child = {
						is_female = yes
						NOT = { character = PREVPREV }
					}
				}
			}
		}
		NOT = {
			AND = {
				age = 45
				is_female = yes
			}
		}
		NOT = { age = 75 }
		NOT = { trait = celibate }
		NOT = { trait = incapable }
	}
	chance = {
		factor = 50

		modifier = { factor = 0.1 has_religion_feature = religion_matriarchal }
		modifier = { factor = 10.0 has_religion_feature = religion_patriarchal }
		modifier = { factor = 1.5 num_of_children = 3 }
		modifier = { factor = 3.0 num_of_children = 4 }
		modifier = { factor = 10.0 num_of_children = 5 }
	}
	success = {
		any_child = {
			is_female = no
		}
	}
	abort = {
		OR = {
			AND = {
				age = 45
				is_female = yes
			}
			age = 75
			trait = celibate
			trait = incapable
		}
	}
	abort_effect = {
	}
	effect = {
		piety = 10
	}
}

# Become a paragon of virtue
# Piety 2000. The nickname follows the faith: the Dark Gods, Nagash and the
# Horned Rat call their saints damned.
obj_become_paragon_of_virtue = {
	type = character

	potential = {
		is_adult = yes
		is_playable = yes
	}

	allow = {
		WHG_ambition_able_trigger = yes
		is_adult = yes
		NOT = { piety = 1000 }
		OR = {
			has_education_learning_trigger = yes
			has_pious_trait_trigger = yes
		}
		has_impious_trait_trigger = no
		is_playable = yes
		is_landed = yes
		NOT = { trait = incapable }
	}

	chance = {
		factor = 50

		modifier = { factor = 1.25 trait = patient }
		modifier = {
			factor = 0.01
			NOR = {
				tier = king
				tier = emperor
				is_theocracy = yes
			}
		}
		modifier = { factor = 10.0 has_education_learning_trigger = yes }
		modifier = { factor = 10.0 trait = zealous }
	}
	success = {
		piety = 2000
	}
	abort = {
		OR = {
			has_impious_trait_trigger = yes
			trait = incapable
		}
	}
	abort_effect = {
	}
	effect = {
		change_learning = 1
		piety = 100
		if = {
			limit = {
				has_nickname = no
				OR = {
					religion_group = chaos_gods_group
					religion_group = necromantic_group
					religion_group = horned_rat_god_group
				}
			}
			give_nickname = nick_the_damned
		}
		else_if = {
			limit = { has_nickname = no }
			give_nickname = nick_the_holy
		}
		religion_authority = {
			modifier = paragon_of_virtue
			years = 20
		}
		religion_head = {
			opinion = {
				who = ROOT
				modifier = opinion_paragon
				years = 100
			}
		}
		any_vassal = {
			limit = { is_theocracy = yes religion = ROOT }
			opinion = {
				who = ROOT
				modifier = opinion_paragon
				years = 100
			}
		}
		hidden_tooltip = {
			if = {
				limit = { ai = no }
				set_character_flag = achievement_become_paragon_of_virtue
			}
		}
	}
}

		modifier = { factor = 0.25 NOT = { martial = 5 } }

		modifier = { factor = 1.25 martial = 16 }
		modifier = { factor = 1.25 martial = 19 }
		modifier = { factor = 1.25 martial = 22 }
	}
	success = {
		custom_tooltip = {
			text = DR_obj_win_war
			hidden_tooltip = { has_character_flag = obj_win_war_won_war }
		}
	}
	abort = {
		OR = {
			is_ruler = no
			prisoner = yes

		}

		if = {
			limit = { NOT = { has_character_flag = obj_win_war } }

		clr_character_flag = obj_win_war_won_war
	}

# Crush a rival
# A rival dies or ends up in your dungeon. Up to Duke; kings take Bring a
# rival king to heel. The flag is set by ambition.202 (on_death). A servant
# may bring a secret about the rival (ambition.400).
obj_crush_a_rival = {
	type = character

	creation_effect = {
		character_event = { id = ambition.400 days = 180 random = 365 }
	}

	potential = {
		is_adult = yes
		lower_tier_than = KING
		num_of_rivals = 1
	}

	allow = {
		WHG_ambition_able_trigger = yes
		NOT = { trait = content }
	}

	chance = {
		factor = 30
		modifier = { factor = 2 trait = wroth }
		modifier = { factor = 2 trait = ambitious }
		modifier = { factor = 1.5 trait = envious }
		modifier = { factor = 1.5 intrigue = 14 }
		modifier = { factor = 0.2 trait = kind }
		modifier = { factor = 0.2 trait = craven }
	}

	success = {
		custom_tooltip = {
			text = obj_crush_a_rival_tt
			hidden_tooltip = {
				OR = {
					has_character_flag = amb_rival_crushed
					any_rival = {
						prisoner = yes
						host = { character = ROOT }
					}
				}
			}
		}
	}

	abort = {
		trait = incapable
	}

	abort_effect = {
		clr_character_flag = amb_rival_crushed
	}

	effect = {
		clr_character_flag = amb_rival_crushed
		prestige = 100
		if = {
			limit = { intrigue = 10 }
			change_intrigue = 1
		}
		else = {
			change_martial = 1
		}
	}
}

# Strengthen the faith
# Emperors raise the authority of their faith.
obj_strengthen_religion = {
	type = character

	creation_effect = {
		change_variable = { which = strengthen_religion value = 0 }
		random_list = {
			150 = {
				set_character_flag = had_ambition_opportunity_chance
			}
			50 = {
				character_event = { id = warcb.10 days = 14 random = 14 }
				set_character_flag = had_ambition_opportunity_chance
				modifier = {
					factor = 0
					NOT = {
						is_nomadic = no
						mercenary = no
						NOT = { pacifist = yes }
						top_liege = {
							religion = ROOT
						}
						any_realm_province = {
							any_neighbor_province = {
								holder_scope = {
									NOT = { same_realm = ROOT }
									top_liege = {
										NOT = { religion = ROOT }
										any_realm_province = {
											religion = ROOT
											kingdom = {
												any_de_jure_vassal_title = {
													tier = COUNT
													holder_scope = { same_realm = PREVPREVPREVPREV }
													location = {
														any_neighbor_province = {
															holder_scope = {
																top_liege = {
																	character = ROOT
																}
															}
														}
													}
												}
											}
										}
									}
								}
							}
						}
					}
				}
				modifier = { trait = zealous factor = 5 }
				modifier = { ai = no factor = 3 }
				modifier = { has_focus = focus_theology factor = 2 }
				modifier = { trait = cynical factor = 0 }
				modifier = { has_character_flag = had_liberate_opportunity factor = 0 }
				modifier = { has_character_flag = had_ambition_opportunity_chance factor = 0 }
			}
		}
	}

	potential = {
		has_dlc = "Jade Dragon"
		NOT = { religion_authority = 0.3 }
		is_playable = yes
		is_landed = yes
		higher_tier_than = KING
		is_nomadic = no
		liege = { is_nomadic = no }
		NOT = { has_character_flag = strengthened_religion }
	}

	allow = {
		WHG_ambition_able_trigger = yes
		independent = yes
		is_adult = yes
		NOR = {
			trait = incapable
			has_secret_religion = yes
			is_devil_worshiper_trigger = yes
			is_inaccessible_trigger = yes
		}
	}

	chance = {
		factor = 50

		modifier = { factor = 0.1 NOT = { tier = king } NOT = { tier = emperor } }
		modifier = {
			factor = 10.0
			OR = {
				trait = detached_priest
				trait = martial_cleric
				trait = scholarly_theologian
				trait = mastermind_theologian
			}
		}
		modifier = { factor = 10.0 trait = zealous }
		modifier = { factor = 0.01 trait = cynical }
	}

	success = {
		custom_tooltip = {
			text = tooltip_obj_strengthen_religion
			hidden_tooltip = {
				check_variable = { which = strengthen_religion value = 3 }
			}
		}
	}

	abort = {
		religion_authority = 0.50
	}

	abort_effect = {

	}

	effect = {
		religion_authority = {
			modifier = strengthened_religion
			years = 50
		}
		piety = 100
		add_character_modifier = {
			modifier = strengthen_religion_ambition
			years = 30
		}
		set_character_flag = strengthened_religion
	}
}
