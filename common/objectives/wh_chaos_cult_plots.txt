###############################################################################
# Chaos cult sabotage plots
# A cultist of one of the four gods gathers backers to bring their god's gift
# to a ruler's lands: Nurgle's plague, a Khornate beastmen frenzy, Tzeentch's
# warpdust or a Slaaneshi revel. Members of the Undivided cult may choose any.
# The strike itself is an event of the plotter, chaoscult.430-437 in
# events/WHG_chaos_cult_events.txt, which sets chaos_sabotage_done to end the
# plot. Restored in the 2026 events refactor (the plots had been lost and the
# old events in BR_cultist_plot_events.txt commented out).
# Scopes: potential has the plotter as ROOT; allow, success and abort have
# the target as ROOT and the plotter as FROM.
###############################################################################

plot_province_disease = {
	type = realm_characters_plus
	intrigue_plot = yes

	potential = {
		prisoner = no
		age = 16
		NOT = { trait = incapable }
		OR = {
			society_member_of = cult_of_nurgle
			society_member_of = cult_of_undivided
		}
	}

	allow = {
		is_ruler = yes
		is_landed = yes
		NOT = { character = FROM }
		is_within_diplo_range = FROM
		NOT = { religion = nurgle }
		NOT = { society_member_of = cult_of_nurgle }
		any_demesne_province = {
			NOT = { has_province_modifier = nurgle_province_disease }
		}
	}

	chance = {
		factor = 1
		modifier = {
			factor = 5
			FROM = { is_rival = ROOT }
		}
		modifier = {
			factor = 2
			NOT = { reverse_opinion = { who = FROM value = 0 } }
		}
		modifier = {
			factor = 2
			FROM = { trait = zealous }
		}
		modifier = {
			factor = 0.2
			FROM = {
				OR = {
					trait = kind
					trait = craven
					trait = content
				}
			}
		}
		modifier = {
			factor = 0
			FROM = { is_friend = ROOT }
		}
	}

	creation_effect = {
		FROM = { set_character_flag = chaos_sabotage_plotting }
	}

	success = {
		FROM = { has_character_flag = chaos_sabotage_done }
	}

	abort = {
		OR = {
			is_alive = no
			is_landed = no
			FROM = {
				NOR = {
					society_member_of = cult_of_nurgle
					society_member_of = cult_of_undivided
				}
			}
		}
	}

	abort_effect = {
		FROM = {
			clr_character_flag = chaos_sabotage_plotting
			clr_character_flag = chaos_sabotage_done
		}
	}

	effect = {
		FROM = {
			clr_character_flag = chaos_sabotage_plotting
			clr_character_flag = chaos_sabotage_done
		}
	}
}
