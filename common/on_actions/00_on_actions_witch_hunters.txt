# Witch hunter on_actions. Each event sits on the slowest pulse that fits how
# often it should fire, so the yearly pulse stays light.

on_yearly_pulse = {
	events = {
		whunter.1 #Time to Put Witch Hunters to work (Non Society)
	}
}
on_bi_yearly_pulse = {
	events = {
		whunter.45 #No witch hunters in the realm: hire some (was a 1-month MTTH, then yearly)
		whunter.46 #Unlanded heretics leave Old World realms (was yearly)
	}
	random_events = {
		whunter_realm_heretics = {	# Idle witch hunters report open vampires or Chaos worshippers in the realm
			100 = 0
			70 = whunter.35
			70 = whunter.40
		}
	}
}
on_five_year_pulse = {
	random_events = {
		whunter_order_holding = {	# The Order asks a ruler in debt for a castle site (about 1 in 17 years; was MTTH 5 years)
			100 = 0
			40 = whunter.105
		}
		whunter_order_site = {	# The Order asks any ruler for a castle site (about 1 in 22 years; was MTTH 20 years)
			100 = 0
			29 = whunter.110
		}
	}
}
on_decade_pulse = {
	random_events = {
		whunter_order_son = {	# The Order asks a ruler in debt for a son (about 1 in 15 years, 15-year cooldown; was MTTH 10 years)
			100 = 0
			200 = whunter.100
		}
		whunter_volunteers = {	# A ruler's son volunteers for the Order (was MTTH 100 years: 1 in 10 per decade; one per court in 15 years)
			100 = 0
			11 = whunter.160
		}
	}
}
on_character_ask_to_join_society = {
	events = {
		whunter.165 #Join Witch Hunters
	}
}
# ROOT is the character, FROM is the title, FROMFROM is the old holder
on_death = {
	events = {
		whunter.50 #Witch Hunters Elective Succession
	}
}
# Fires every two years for characters who are members of any society
on_society_bi_yearly_pulse = {
	events = {
		whunter.170 # Inquisitorial societies: quest tombola
	}
}
