# Building and wonders: on_actions (WHG_building_on_actions.txt)
#
# Holdings, forts and hospitals built, and the wonders: owners, stages,
# upgrades, restoration.
#
# How on_actions work: the game fires each on_action at a fixed moment and
# ROOT gets the listed events. Every event in `events = { }` is checked;
# `random_events = { }` picks one valid event by weight, and each named
# sub-list (`name = { delay = N  w = ns.id }`) picks one of its own, so it
# never changes the odds of the others. A `0` entry is the chance of no
# event. The same on_action may appear in several files and the lists add
# up, but in this mod every on_action is written exactly once, so one
# search finds everything that fires on it. The file index is in
# 00_on_actions.txt.
#
# Each line is an event id with a comment: its title from the event's doc
# comment, then the event file, e.g. "Warpstone from the mines
# (WHG_skaven_events)".
#
# On_actions here, in order:
#   on_settlement_construction_start on_settlement_construction_completed
#   on_fort_construction_completed on_hospital_construction_completed
#   on_holding_building_start on_wonder_owner_change
#   on_wonder_construction_start on_wonder_stage_finish
#   on_wonder_upgrade_finish on_wonder_upgrade_destroyed
#   on_wonder_restore_finish

# Triggers when a character becomes the owner of a wonder.
# ROOT = new owner
# FROM = wonder
# FROMFROM = previous owner (this person might be dead or whatever, but still needed)
# FROMFROMFROM = province wonder is in
on_wonder_owner_change = {
	events = {
		underway.6 # A hold changes hands (WHG_dwarf_events)
	}
	random_events = {
		1000 = 0
		delayed_upgrade_finish_effects = {
			delay = 14
			50 = wonder.58 # The traps of the old owner (WHG_wonder_events)
		}
	}
}

# Triggers when a wonder begins construction of any stage.
# ROOT = owner
# FROM = wonder
# FROMFROM = province (location of wonder)
on_wonder_construction_start = {
	random_events = {
		1000 = 0
		100 = wonder.120 # Digging begins (start) (WHG_wonder_events)
	}
}

# Triggers when a wonder finishes building a stage.
# ROOT = owner
# FROM = wonder
# FROMFROM = province (location of wonder)
on_wonder_stage_finish = {
	events = {
		wonder.1 # A stage is finished (WHG_wonder_events)
	}
}

# Triggers when construction of an upgrade finishes in a wonder.
# ROOT = owner
# FROM = upgrade
# FROMFROM = wonder
# FROMFROMFROM = province (location of wonder)
on_wonder_upgrade_finish = {
	events = {
		wonder.2 # An upgrade is finished (WHG_wonder_events)
		wonder.190 # Walls for the townships (WHG_wonder_events)
		wonder.4 # A statue of a god is raised (WHG_wonder_events)
		underway.1 # An upgrade is finished (WHG_dwarf_events)
	}
	random_events = {
		1000 = 0
		delayed_upgrade_finish_effects = {
			delay = 50
			100 = wonder.50 # A town springs up (WHG_wonder_events)
		}
	}
}

# Triggers when an upgrade is destroyed in a wonder.

# FROMFROMFROM = province (location of wonder)
on_wonder_upgrade_destroyed = {

}

# Triggers when restoration of a wonder completed
