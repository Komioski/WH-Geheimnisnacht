# Deeds of legend: hooks for the chronicle deeds (events/WHG_chronicle_events.txt,
# chronicle.100 - chronicle.150). Every deed is for players only; each event
# checks ai = no (or a global flag) first, so the cost is a trigger check.

# A title is created (chronicle.100: Vala-Azril, Ulthuan, Araby, Nehekhara,
# the Empire under a vampire). ROOT is the creator, FROM the title.
on_create_title = {
	events = {
		chronicle.100
	}
}

# A title changes hands (chronicle.100). ROOT is the new holder, FROM the
# title, FROMFROM the old holder.
on_new_holder = {
	events = {
		chronicle.100
	}
}
on_new_holder_inheritance = {
	events = {
		chronicle.100
	}
}
on_new_holder_usurpation = {
	events = {
		chronicle.100
	}
}

# A holding is taken (chronicle.118: a great city for the Skaven, Marienburg
# for Norsca). ROOT is the commander, FROM the holding.
on_siege_won_leader = {
	events = {
		chronicle.118
	}
}

# Battles won in person (chronicle.127: da Biggest). ROOT is the commander,
# FROM the enemy commander.
on_major_battle_won = {
	events = {
		chronicle.127
	}
}
on_major_battle_won_leader = {
	events = {
		chronicle.127
	}
}

# A settlement is looted (chronicle.129: the Despoiler). ROOT is the looter.
on_loot_settlement = {
	events = {
		chronicle.129
	}
}

# A war ends. FROM is the attacker, ROOT the defender.
# The attacker won (chronicle.134: a Black Ark repelled).
on_war_ended_victory = {
	events = {
		chronicle.134
	}
}
# The attacker lost (chronicle.125: the Bulwark holds; chronicle.135: a
# Black Ark repelled).
on_war_ended_defeat = {
	events = {
		chronicle.125
		chronicle.135
	}
}
# White peace (chronicle.136: the Black Ark marks are cleared).
on_war_ended_whitepeace = {
	events = {
		chronicle.136
	}
}

# A holy war is won (chronicle.139: the Spear of Myrmidia). ROOT is the
# title, FROM the taker, new_character the enemy.
on_crusade_success = {
	events = {
		chronicle.139
	}
}
