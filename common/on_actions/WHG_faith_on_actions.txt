# Faith and holy war: on_actions (WHG_faith_on_actions.txt)
#
# Conversion of characters and counties, reformation, heresy taking over,
# excommunication, and the holy wars (crusades): preparation, pledges,
# launch, monthly pulse and the outcome.
#
# How on_actions work: the game fires each on_action at a fixed moment and
# ROOT gets the listed events. Every event in `events = { }` is checked;
# `random_events = { }` picks one valid event by weight, and each named
# sub-list (`name = { delay = N  w = ns.id }`) picks one of its own, so it
# never changes the odds of the others. A `0` entry is the chance of no
# event. The same on_action may appear in several files and the lists add
# up, but in this mod every on_action is written exactly once, so one
# search finds everything that fires on it. The file index is in
# 00_on_actions.txt.
#
# Each line is an event id with a comment: its title from the event's doc
# comment, then the event file, e.g. "Warpstone from the mines
# (WHG_skaven_events)".
#
# On_actions here, in order:
#   on_crusade_creation on_holy_order_leave on_crusade_success
#   on_crusade_failure on_crusade_invalid on_reform_religion
#   on_county_religion_change on_vassal_accepts_religious_conversion
#   on_heresy_takeover on_character_convert_religion
#   on_crusade_preparation_starts on_crusade_launches on_crusade_canceled
#   on_crusade_monthly on_crusade_target_changes
#   on_pledge_crusade_participation on_unpledge_crusade_participation
#   on_excommunicate_interaction

on_crusade_creation = {
	events = {
		holywar.1 # A holy war is called. ROOT is the head of (WHG_on_action_events)
		chronicle.16 # Holy war declared (WHG_chronicle_events)
	}
}

on_holy_order_leave = {
	events = {
		debt.34 #  (WHG_realm_events)
	}
}

# When a mission succeeds
on_crusade_success = {
	events = {
		holywar.10 # A holy war is won (WHG_on_action_events)
		chronicle.17 # Holy war won (WHG_chronicle_events)
		# A holy war is won (ambition.211)
		ambition.211 # A holy war is won (WHG_ambition_events)
		# A holy war is won (chronicle.139: the Spear of Myrmidia). ROOT is the
		# title, FROM the taker, new_character the enemy.
		chronicle.139 # A holy war won against Araby (hidden) (WHG_chronicle_events)
	}
}

# When a mission fails
on_crusade_failure = {
	events = {
		holywar.11 # A holy war fails. FROM is the head of the faith, new_character the enemy (WHG_on_action_events)
		chronicle.18 # Holy war lost (WHG_chronicle_events)
		# A holy war fails (ambition.212)
		ambition.212 # A holy war fails (WHG_ambition_events)
	}
}

# When a mission invalidates
on_crusade_invalid = {
	events = {
		holywar.12 # A holy war comes to nothing, because the enemy converted or lost the land (WHG_on_action_events)
	}
}

	}

# When the religion changes in a county, ROOT is province. FROM has the old religion.

	}
}

# When a character accepts religious conversion (the diplomatic action)
# FROMFROM is the demander. FROM is the vassal. ROOT is the character. Fires for the vassal and each of his courtiers and vassals.

	events = {
		faith.700 # A vassal accepts a demand to convert (WHG_faith_events)
	}
}

# Character converts religion, for whatever reason. ROOT is the character after conversion and the FROM scope has the old religion.

	events = {
		chronicle.25 # Converted faith (WHG_chronicle_events)
		upkeep.30 # A rival who shares the faith may react to a conversion (WHG_on_action_events)
		patron.2 # A convert loses the patron of the old faith (WHG_patron_god_events)
		patron_god.21 # A convert loses the Old World patron (WHG_patron_god_events)
		# A convert out of the southern faiths loses the sacred-pet modifiers
		holypet.6 # A ruler leaves the southern faiths (WHG_araby_events)
		# Baptism and mass conversion
		baptism.12 # On conversion of sponsor during baptism (WHG_baptism_events)
		baptism.18 # On conversion of converter after baptism (WHG_baptism_events)
		baptism.20 # On conversion of sponsor after baptism (WHG_baptism_events)
		baptism.27 # On baptism of deceitful player, maintain previous faith (WHG_baptism_events)
		coronation.408 # Ruler converted when in the middle of a coronation, cancel everything (WHG_coronation_events)
		coronation.416 # Clean up uncrowned modifier (WHG_coronation_events)
		doctrine.33 # Acquire Pagan religious branch when converting to a religion that has No Head (WHG_doctrine_events)
		doctrine.34 # Acquire Gender law when converting to Matriarchal/Equal/Patriarchal religion (WHG_doctrine_events)
		doctrine.458 # Host of Blood Games converts to non-Bloodthirsty religion in the middle of the tournament (WHG_doctrine_events)
		# Crusades
		bcrusade.11 # The chosen lord turns away (WHG_black_crusade_events)
		# The Reconquest of the South
		reconquest.3 # The reconquistador leaves the faith (WHG_crusade_events)
		reconquest.7 # A promised ally leaves the faith (WHG_crusade_events)
		# Faith-bound titles and edicts
		empire.67 # The giver turns from the Wolf and the Huntsman: the Edict ends (WHG_empire_events)
		califate.2 # Sayyid and Mirza after conversion (WHG_araby_events)
		cultpath.1 # A cultist no longer (WHG_chaos_cult_events)
		amazon.166 # The Kalith line is lost (WHG_amazon_events)
		aiclean.32 # Wrong faith (WHG_maintenance_events)
	}
}
