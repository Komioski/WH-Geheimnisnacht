# Societies and quests: on_actions (WHG_society_on_actions.txt)
#
# Joining, leaving and rising in societies, a society lost, its progress
# bar full, and society quests won or failed.
#
# How on_actions work: the game fires each on_action at a fixed moment and
# ROOT gets the listed events. Every event in `events = { }` is checked;
# `random_events = { }` picks one valid event by weight, and each named
# sub-list (`name = { delay = N  w = ns.id }`) picks one of its own, so it
# never changes the odds of the others. A `0` entry is the chance of no
# event. The same on_action may appear in several files and the lists add
# up, but in this mod every on_action is written exactly once, so one
# search finds everything that fires on it. The file index is in
# 00_on_actions.txt.
#
# Each line is an event id with a comment: its title from the event's doc
# comment, then the event file, e.g. "Warpstone from the mines
# (WHG_skaven_events)".
#
# On_actions here, in order:
#   on_character_switch_society_interest on_character_society_rank_up
#   on_character_join_society on_character_leave_society
#   on_character_ask_to_join_society on_character_kicked_from_society
#   on_character_stop_showing_interest on_society_destroyed on_quest_success
#   on_quest_failure on_society_progress_full

# A character has switched their society interest
# Root = Character
# From = Society
on_character_switch_society_interest = {
	events = {
		society.10 # Interest is shown (WHG_society_events)
		society.91 # An empty hall (WHG_society_events)
	}
}

# A character ranks up in their society, or becomes grandmaster
# Root = Character
# From = Society
on_character_society_rank_up = {
	events = {
		society.95 # The new grandmaster (WHG_society_events)
		pleasure.50 # Rising in the cult (WHG_slaanesh_events)
		seas.160 # The sea traits come with rank (hidden) (WHG_sea_events)
		venture.125 # A tale of the Venturers (WHG_venture_events)
	}
}

# A character joins a society. The has_game_started trigger can be useful to avoid firing events from auto-assignment of characters at game start
# Root = Character
# From = Society
on_character_join_society = {
	events = {
		# Root = Character, From = Society
		engineer.7 # A new member's branch (WHG_dwarf_guild_events)
	}
}

# A character leaves or is forced to leave a society
# Root = Character
# From = Society
on_character_leave_society = {
	events = {
		society.90 # Leaving a society (WHG_society_events)
		verena.184 # The apprentices are released (WHG_verena_events)
		warband.9 # The Call to Glory ends (hidden) (WHG_warrior_society_events)
		# Root = Character, From = Society
		engineer.9 # A member leaves the guild (WHG_dwarf_guild_events)
	}
}

}

# The last member of a society leaves the society and is not replaced
# Root = Character
# From = Society
on_society_destroyed = {
	events = {
		lodge.53 # A fallen lodge rises again (WHG_warrior_society_events)
		warband.63 # A fallen warband rises again (WHG_warrior_society_events)
	}
}

# A character succeeds in a quest. This happens just before the quest disappears, so you can access the quest itself. Only invoked when a quest is cleared via script. E.G., does not happen on death
# Root = Character
on_quest_success = {
	events = {
		society.70 # A mission done (WHG_society_events)
	}
}

# A character fails a quest. This happens just before the quest disappears, so you can access the quest itself. Only invoked when a quest is cleared via script. E.G., does not happen on death
# Root = Character
on_quest_failure = {
	events = {
		society.71 # A mission failed (WHG_society_events)
	}
}

# Triggers when the society progress is increased / set to the value 100
# Root = The affected society grandmaster
# From = The affected society
on_society_progress_full = {
	events = {
		lodge.700 # The legend is full (WHG_warrior_society_events)
		seas.100 # The legend is full (hidden) (WHG_sea_events)
	}
}
