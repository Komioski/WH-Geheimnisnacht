###############################################################################
# WHG_religious_titles.txt - titles a head of faith hands out
#
# A religious title is held next to a character's other titles and, with
# can_elect_head, makes its holders the electors of the head of faith. The
# mod has one: the Lectors of Sigmar, who elect the Grand Theogonist. This
# folder has replace_path, so vanilla's cardinals are not loaded.
#
# Keys: religion owns the title; grant_limit caps the holders; realm_in_name
# adds the realm to the name; allowed_to_grant/allow say who may give and
# hold it; gain_effect/lose_effect fire when it is gained or lost; message
# tells the holder. Every other key is a character modifier.
###############################################################################

# The Lectors of Sigmar: nine theocrats of the Cult of Sigmar who elect the
# Grand Theogonist. Nobody grants it by hand (allowed_to_grant always = no):
# the title passes to Sigmarite theocrats within reach of the Cult.
title_lector = {
	religion = sigmarite
	can_elect_head = yes
	revoke_allowed = no

	grant_limit = 9

	realm_in_name = no
	show_as_title = yes

	opinion_effect = 20
	dignity = 0.9

	monthly_salary = 0.1
	monthly_prestige = 0.5

	allowed_to_grant = {
		always = no
	}
	allow = {
		# cheap checks first
		religion = sigmarite
		is_theocracy = yes
		trait = creature_human
		NOR = { trait = witch_hunter trait = inquisitor trait = solkan_templar }
		k_cult_of_sigmar = {
			holder_scope = {
				is_within_diplo_range = ROOT
			}
		}
		NOT = {
			top_liege = {
				any_vassal = {
					any_claim = { title = k_cult_of_sigmar }
				}
			}
		}
	}

	gain_effect = {
		character_event = { id = empire.21 } # a new Lector may take an empty Arch-Lectory
	}

	message = yes
}
