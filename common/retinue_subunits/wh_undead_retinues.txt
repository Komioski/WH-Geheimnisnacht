# Undead retinues for masters of the dead, from Komi's DoW-PoD submod, beside
# the older ones in 00_retinue_subunits.txt (RETTYPE_V_*) and the bloodline

# Damned Knights: fallen knights of Bretonnia who ride for the dead lords of
# Mousillon.

	first_type = 4
	first_amount = 75
	hire_cost = 120
	maintenance_multiplier = 0.45

	potential = {
