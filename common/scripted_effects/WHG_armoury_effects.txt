# A new smith comes to court: a steward of 45 who will not marry. A woman one
# time in seven, more often where the law lets women hold office, never in a
# race without women.
# Scope: ROOT is the employer. Saves event_target:smith.
armoury_create_smith_effect = {
	random_list = {
		15 = { # A woman
			modifier = { factor = 0 is_no_females_race = yes }
			modifier = { factor = 2 primary_title = { has_law = status_of_women_2 } }
			modifier = { factor = 2 primary_title = { has_law = status_of_women_3 } }
			modifier = { factor = 2 primary_title = { has_law = status_of_women_4 } }
			create_random_steward = {
				age = 45
				random_traits = yes
				female = yes
				dynasty = none
				attributes = { stewardship = 3 }
				historical = yes
			}
			new_character = {
				character_event = { id = aiclean.31 } # racial trait
				save_event_target_as = smith
				set_character_flag = ai_flag_refuse_concubinage
			}
		}
		85 = { # A man
			create_random_steward = {
				age = 45
				random_traits = yes
				female = no
				dynasty = none
				attributes = { stewardship = 3 }
				historical = yes
			}
			new_character = {
				character_event = { id = aiclean.31 } # racial trait
				save_event_target_as = smith
			}
		}
	}
	event_target:smith = {
		health = 3
		set_character_flag = ai_flag_refuse_marriage
		set_character_flag = cannot_marry
		set_character_flag = no_court_invites
		will_not_marry_effect = yes
	}
}

# The smith's craft: stewardship, a temperament that decides whether the
# smith is tempted to steal, and sometimes a better education.
# Scope: the smith.
armoury_smith_talent_effect = {
	random_list = {
		15 = { change_stewardship = 1 }
		25 = { change_stewardship = 3 }
		20 = {
			artisan_craftsmen_bloodline_increase_score = yes
			change_stewardship = 5
		}
		10 = {
			artisan_craftsmen_bloodline_increase_score = yes
			change_stewardship = 7
		}
	}
	random_list = {
		1 = { add_trait = greedy }
		1 = { add_trait = envious }
		1 = {
			artisan_craftsmen_bloodline_increase_score = yes
			add_trait = charitable
		}
		1 = {
			artisan_craftsmen_bloodline_increase_score = yes
			add_trait = diligent
		}
		1 = { add_trait = deceitful }
	}
	random = {
		chance = 30
		modifier = {
			factor = 2
			trait = indulgent_wastrel
		}
		upgrade_education_effect = yes
	}
}

# The order: the price is paid, the work begins. Sets the tier flag on the
# employer and the smith and rolls what will come of it: the work done
# (better with a skilled, careful smith or a realm Armoury), trouble with
# the materials, or a theft (never for a ruler of the Artisan bloodline).
# A goldsmith uses the goldsmith_ outcome flags. The smith reports in 50 to
# 150 days (armoury.10, goldsmith armoury.61); something may happen in the
# smithy before (armoury.70).
# Scope: ROOT is the employer, event_target:smith the smith.
armoury_order_expensive_effect = {
	if = {
		limit = {
			OR = {
				has_character_flag = weaponsmith
				has_character_flag = armorsmith
			}
		}
		event_target:smith = { set_character_flag = expensive_weapons }
		set_character_flag = expensive_weapons
		random_list = {
			55 = { # Done
				modifier = { factor = 0.7 event_target:smith = { stewardship <= 2 } }
				modifier = { factor = 0.7 event_target:smith = { stewardship <= 4 } }
				modifier = { factor = 0.7 event_target:smith = { stewardship <= 6 } }
				modifier = { factor = 0.7 event_target:smith = { stewardship <= 8 } }
				modifier = { factor = 0.7 event_target:smith = { stewardship <= 10 } }
				modifier = { factor = 1.1 event_target:smith = { stewardship >= 10 } }
				modifier = { factor = 1.1 event_target:smith = { stewardship >= 12 } }
				modifier = { factor = 1.1 event_target:smith = { stewardship >= 14 } }
				modifier = { factor = 1.1 event_target:smith = { stewardship >= 16 } }
				modifier = { factor = 1.1 event_target:smith = { stewardship >= 18 } }
				modifier = { factor = 1.1 event_target:smith = { stewardship >= 20 } }
				modifier = { factor = 1.3 event_target:smith = { trait = greedy } }
				modifier = { factor = 1.3 event_target:smith = { trait = diligent } }
				modifier = { factor = 1.3 event_target:smith = { trait = patient } }
				modifier = { factor = 1.3 event_target:smith = { is_smart_trigger = yes } }
				modifier = { factor = 1.5 event_target:the_armory = { always = yes } }
				set_character_flag = smith_success
			}
			45 = { # Trouble with the materials
				set_character_flag = smith_fail
			}
			10 = { # Theft
				trigger = {
					NOT = { any_owned_bloodline = { has_bloodline_flag = bloodline_artisan_craftsmen } }
				}
				modifier = { factor = 1.3 event_target:smith = { trait = envious } }
				modifier = { factor = 0.2 event_target:smith = { trait = charitable } }
				modifier = { factor = 0.2 event_target:smith = { trait = content } }
				modifier = { factor = 1.5 event_target:smith = { trait = ambitious } }
				modifier = { factor = 1.3 event_target:smith = { trait = deceitful } }
				set_character_flag = smith_steal
			}
		}
		character_event = { id = armoury.10 days = 50 random = 100 }
	}
	else = {
		event_target:smith = { set_character_flag = expensive_jewels }
		set_character_flag = expensive_jewels
		random_list = {
			40 = { # Done
				modifier = { factor = 0.7 event_target:smith = { stewardship <= 2 } }
				modifier = { factor = 0.7 event_target:smith = { stewardship <= 4 } }
				modifier = { factor = 0.7 event_target:smith = { stewardship <= 6 } }
				modifier = { factor = 0.7 event_target:smith = { stewardship <= 8 } }
				modifier = { factor = 0.7 event_target:smith = { stewardship <= 10 } }
				modifier = { factor = 1.1 event_target:smith = { stewardship >= 10 } }
				modifier = { factor = 1.1 event_target:smith = { stewardship >= 12 } }
				modifier = { factor = 1.1 event_target:smith = { stewardship >= 14 } }
				modifier = { factor = 1.1 event_target:smith = { stewardship >= 16 } }
				modifier = { factor = 1.1 event_target:smith = { stewardship >= 18 } }
				modifier = { factor = 1.1 event_target:smith = { stewardship >= 20 } }
				modifier = { factor = 1.3 event_target:smith = { trait = greedy } }
				modifier = { factor = 1.3 event_target:smith = { trait = diligent } }
				modifier = { factor = 1.3 event_target:smith = { trait = patient } }
				modifier = { factor = 1.3 event_target:smith = { is_smart_trigger = yes } }
				set_character_flag = goldsmith_success
			}
			60 = { # Trouble with the materials
				set_character_flag = goldsmith_fail
			}
			10 = { # Theft
				trigger = {
					NOT = { any_owned_bloodline = { has_bloodline_flag = bloodline_artisan_craftsmen } }
				}
				modifier = { factor = 1.3 event_target:smith = { trait = envious } }
				modifier = { factor = 0.2 event_target:smith = { trait = charitable } }

# A greenskin smith's work, by the weapon ordered and the price paid: a
# choppa, an 'ard axe, a club, a stabba or 'ard armour (a gnasha helm
# for one who already has his armour).
# Scope: whoever gets it. Reads the flags of event_target:smith.
armoury_greenskin_weapon_effect = {
	if = {
		limit = { event_target:smith = { has_character_flag = weapon_choppa } }
		if = {
			limit = { event_target:smith = { has_character_flag = expensive_weapons } }
			add_artifact = giant_choppa
		}
		else_if = {
			limit = { event_target:smith = { has_character_flag = good_weapons } }
			add_artifact = orc_big_choppa
		}
		else = { add_artifact = orc_choppa }
	}
	else_if = {
		limit = { event_target:smith = { has_character_flag = weapon_ax } }
		if = {
			limit = { event_target:smith = { has_character_flag = expensive_weapons } }
			add_artifact = giant_ax
		}
		else = { add_artifact = greenskin_ax }
	}
	else_if = {
		limit = { event_target:smith = { has_character_flag = weapon_club } }
		if = {
			limit = { event_target:smith = { has_character_flag = expensive_weapons } }
			add_artifact = giant_club
		}
		else_if = {
			limit = { event_target:smith = { has_character_flag = good_weapons } }
			add_artifact = big_club
		}
		else = { add_artifact = club }
	}
	else_if = {
		limit = { event_target:smith = { has_character_flag = weapon_stabba } }
		if = {
			limit = { event_target:smith = { has_character_flag = expensive_weapons } }
			add_artifact = back_stabba_blade
		}
		else_if = {
			limit = { event_target:smith = { has_character_flag = good_weapons } }
			add_artifact = goblin_stabba
		}
		else = { add_artifact = dirk }
	}
	else_if = {
		limit = {
			event_target:smith = { has_character_flag = ard_armour }
			NOR = {
				has_artifact = greenskin_ard_armour
				has_artifact = greenskin_ard_armour_2
				has_artifact = greenskin_ard_armour_3
			}
			event_target:smith = {
				OR = {
					has_character_flag = expensive_weapons
					has_character_flag = good_weapons
				}
			}
		}
		if = {
			limit = { event_target:smith = { has_character_flag = expensive_weapons } }
			random_list = {
				1 = { add_artifact = greenskin_ard_armour_2 }
				1 = { add_artifact = greenskin_ard_armour_3 }
			}
		}
		else = { add_artifact = greenskin_ard_armour }
	}
	else = { add_artifact = greenskin_gnashas }
}

# The price of a gromril axe or hammer (armoury.41): more for kings and
# emperors, prestige too for dukes and counts. A runesmith who does the
# work himself gets part of it back.
# Scope: ROOT is the ruler, FROM the runesmith.
armoury_gromril_cost_effect = {
	if = {
		limit = { tier = EMPEROR }
		scaled_wealth = { value = -4.5 min = -740 max = -2000 }
	}
	else_if = {
		limit = { tier = KING }
		scaled_wealth = { value = -3.5 min = -680 max = -2000 }
	}
	else = {
		prestige = -300
		scaled_wealth = { value = -2.5 min = -550 max = -1800 }
	}
	if = {
		limit = { FROM = { character = ROOT } }
		scaled_wealth = { value = 1.5 min = 250 max = 800 }
	}
}

			modifier = { factor = 1.3 event_target:smith = { is_smart_trigger = yes } }

# The smith is kept at the work (armoury.72, armoury.73): some prestige,
# and one time in ten (one in twenty for a good order) the work rises a
# grade. Elven smiths' work is already a mastered art: only a cheap order
# can still rise. Work of the best grade only brings more prestige.
# Scope: ROOT is the employer, FROM the smith.
armoury_smith_focus_effect = {
	if = {
		limit = {
			OR = {
				AND = {
					OR = {
						has_character_flag = weaponsmith
						has_character_flag = armorsmith
					}
					OR = {
						NOR = {
							culture_group = asur_group
							culture_group = druchii_group
							culture_group = asrai_group
						}
						has_character_flag = cheap_weapons
					}
					NOR = {
						has_character_flag = expensive_weapons
						has_character_flag = amazing_weapons
					}
				}
				AND = {
					NOR = {
						has_character_flag = weaponsmith
						has_character_flag = armorsmith
					}
					NOR = {
						has_character_flag = expensive_jewels
						has_character_flag = amazing_jewels
					}
				}
			}
		}
		prestige = 50
		random = {
			chance = 10
			modifier = {
				factor = 0.5
				OR = {
					has_character_flag = good_weapons
					has_character_flag = good_jewels
				}
			}
			custom_tooltip = {
				text = smith_improves_quality_tt
				if = {
					limit = { has_character_flag = cheap_weapons }
					clr_character_flag = cheap_weapons
					set_character_flag = good_weapons
					FROM = {
						clr_character_flag = cheap_weapons
						set_character_flag = good_weapons
					}
				}
				else_if = {
					limit = { has_character_flag = good_weapons }
					clr_character_flag = good_weapons
					set_character_flag = expensive_weapons
					FROM = {
						clr_character_flag = good_weapons
						set_character_flag = expensive_weapons
					}
				}
				else_if = {
					limit = { has_character_flag = cheap_jewels }
					clr_character_flag = cheap_jewels
					set_character_flag = good_jewels
					FROM = {
						clr_character_flag = cheap_jewels
						set_character_flag = good_jewels
					}
				}
				else_if = {
					limit = { has_character_flag = good_jewels }
					clr_character_flag = good_jewels
					set_character_flag = expensive_jewels
					FROM = {
						clr_character_flag = good_jewels
						set_character_flag = expensive_jewels
					}
				}
			}
		}
	}
	else = { prestige = 75 }
}
