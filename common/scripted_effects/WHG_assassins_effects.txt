# Effects for events/WHG_assassins_events.txt (the Hashishin of Ormazd and
# Clan Eshin of the Skaven)

# Scope: a member. Saves the Grand Master of his order as grand_master.
assassins_save_grand_master_effect = {
	if = {
		limit = { society_member_of = eshin_society }
		random_society_member = {
			limit = {
				society_member_of = eshin_society
				is_society_grandmaster = yes
			}
			save_event_target_as = grand_master
		}
	}
	else_if = {
		limit = { society_member_of = the_assassins }
		random_society_member = {
			limit = {
				society_member_of = the_assassins
				is_society_grandmaster = yes
			}
			save_event_target_as = grand_master
		}
	}
}

# Scope: a debtor. Pays the order back what he owes. The gold goes to the
# Grand Master of the lending order if he is the AI lord of its seat, who
# lent it from his own purse (assassins.2).
assassins_pay_debt_effect = {
	if = {
		limit = { has_character_flag = assassins_debt_eshin }
		random_society_member = {
			limit = {
				society_member_of = eshin_society
				is_society_grandmaster = yes
			}
			save_event_target_as = assassins_lender
		}
	}
	else = {
		random_society_member = {
			limit = {
				society_member_of = the_assassins
				is_society_grandmaster = yes
			}
			save_event_target_as = assassins_lender
		}
	}
	if = {
		limit = { is_variable_equal = { which = assassins_debt value = 50 } }
		wealth = -50
		hidden_effect = {
			event_target:assassins_lender = {
				if = {
					limit = { assassins_order_bank_trigger = yes }
					wealth = 50
				}
			}
		}
	}
	else_if = {
		limit = { is_variable_equal = { which = assassins_debt value = 250 } }
		wealth = -250
		hidden_effect = {
			event_target:assassins_lender = {
				if = {
					limit = { assassins_order_bank_trigger = yes }
					wealth = 250
				}
			}
		}
	}
	else_if = {
		limit = { is_variable_equal = { which = assassins_debt value = 300 } }
		wealth = -300
		hidden_effect = {
			event_target:assassins_lender = {
				if = {
					limit = { assassins_order_bank_trigger = yes }
					wealth = 300
				}
			}
		}
	}
	else_if = {
		limit = { is_variable_equal = { which = assassins_debt value = 500 } }
		wealth = -500
		hidden_effect = {
			event_target:assassins_lender = {
				if = {
					limit = { assassins_order_bank_trigger = yes }
					wealth = 500
				}
			}
		}
	}
}

# Scope: a debtor. The debt is settled: no more demands, no more collectors.
assassins_clear_debt_effect = {
	if = {
		limit = { has_character_modifier = assassins_debt }
		remove_character_modifier = assassins_debt
		hidden_effect = {
			set_variable = { which = assassins_debt value = 0 }
			clr_character_flag = assassins_target_debt
			clr_character_flag = assassins_debt_eshin
			clr_character_flag = assassins_refused_to_pay_debt
			clear_delayed_event = { id = assassins.10 }
		}
	}
}

# Scope: the recruiter (assassins.26) or the recruit (assassins.31). Saves a
# Propagandist of the order as assassins_inductor. Clan Eshin if the scope
# is an Eshin member or the recruit is joining Clan Eshin, else the
# Hashishin. An order without one has its Grand Master raise one.
assassins_find_inductor_effect = {
	if = {
		limit = {
			OR = {
				society_member_of = eshin_society
				has_character_flag = assassins_joining_eshin
			}
		}
		if = {
			limit = {
				NOT = {
					any_society_member = {
						society_member_of = eshin_society
						society_rank == 3
					}
				}
			}
			random_society_member = {
				limit = {
					society_member_of = eshin_society
					is_society_grandmaster = yes
				}
				create_character = {
					age = 30
					female = no
					random_traits = no
					dynasty = random
					culture = eshin
					trait = elusive_shadow
					trait = patient
					trait = zealous
					trait = scholar
					trait = temperate
				}
				new_character = {
					character_event = { id = aiclean.31 } # Racial trait
					join_society = eshin_society
					society_rank_up = { society = eshin_society rank = 2 }
				}
			}
		}
		random_society_member = {
			limit = {
				society_member_of = eshin_society
				society_rank == 3
			}
			save_event_target_as = assassins_inductor
		}
	}
	else = {
		if = {
			limit = {
				NOT = {
					any_society_member = {
						society_member_of = the_assassins
						society_rank == 3
					}
				}
			}
			random_society_member = {
				limit = {
					society_member_of = the_assassins
					is_society_grandmaster = yes
				}
				create_character = {
					age = 30
					female = no
					random_traits = no
					dynasty = random
					religion = ormazd
					trait = elusive_shadow
					trait = patient
					trait = zealous
					trait = scholar
					trait = chaste
					trait = temperate
				}
				new_character = {
					character_event = { id = aiclean.31 } # Racial trait
					join_society = the_assassins
					society_rank_up = { society = the_assassins rank = 2 }
				}
			}
		}
		random_society_member = {
			limit = {
				society_member_of = the_assassins
				society_rank == 3
			}
			save_event_target_as = assassins_inductor
		}
	}
}

# Scope: the recruit, FROM the recruiter's messenger. The recruit said no:
# the recruiter is told he failed and the Grand Master hears of it.
assassins_recruitment_failed_effect = {
	FROM = { character_event = { id = assassins.39 days = 3 } }
	if = {
		limit = { event_target:assassins_recruiter = { always = yes } }
		event_target:assassins_recruiter = { set_character_flag = recruitment_fail }
		event_target:grand_master = { character_event = { id = assassins.40 } }
	}
}

# Scope: a recruit who fled the induction, FROM the inductor. On the way
# out he struck the inductor down: wounded, maimed or dead.
assassins_escape_blow_effect = {
	random_list = {
		33 = {
			FROM = { add_trait_wounded_effect = yes }
			set_character_flag = assassins_target_assaulted_member
		}
		33 = {
			FROM = { add_maimed_trait_effect = yes }
			set_character_flag = assassins_target_assaulted_member
		}
		33 = {
			FROM = { death = { death_reason = death_murder_unknown killer = ROOT } }
			set_character_flag = assassins_target_killed_member
		}
	}
}

# Scope: the assassin, event_target:plot_target the dead. If the kill was
# his mission, the mission is done and the order pays by the rank of the
# dead.
assassins_kill_reward_effect = {
	if = {
		limit = {
			has_quest = quest_the_assassins_assassination
			quest_target = { character = event_target:plot_target }
		}
		clr_quest = quest_the_assassins_assassination
		sound_effect = assassins_assassinate_character
		if = {
			limit = { event_target:plot_target = { higher_tier_than = DUKE } }
			add_society_currency_major_effect = yes
		}
		else_if = {
			limit = { event_target:plot_target = { tier = DUKE } }
			add_society_currency_medium_effect = yes
		}
		else = {
			add_society_currency_minor_effect = yes
		}
	}
}

# Scope: any. If event_target:plot_target was marked for death, the Grand
# Master of event_target:assassin_plot_owner's order hears it is done
# (assassins.344).
assassins_mark_fulfilled_effect = {
	if = {
		limit = {
			event_target:plot_target = {
				OR = {
					has_character_flag = assassins_target_marked_for_death
					has_character_flag = eshin_target_marked_for_death
				}
			}
		}
		if = {
			limit = { event_target:assassin_plot_owner = { society_member_of = eshin_society } }
			random_society_member = {
				limit = {
					society_member_of = eshin_society
					is_society_grandmaster = yes
				}
				letter_event = { id = assassins.344 }
			}
		}
		else = {
			random_society_member = {
				limit = {
					society_member_of = the_assassins
					is_society_grandmaster = yes
				}
				letter_event = { id = assassins.344 }
			}
		}
	}
}

# Scope: the victim. Killed by an assassin no one saw;
# event_target:assassin_plot_owner gets the blame he cannot be shown.
assassins_murder_unknown_effect = {
	if = {
		limit = { event_target:assassin_plot_owner = { society_member_of = eshin_society } }
		death = { death_reason = death_murder_unknown_eshin killer = event_target:assassin_plot_owner }
	}
	else = {
		death = { death_reason = death_murder_unknown_hashashin killer = event_target:assassin_plot_owner }
	}
}

# Scope: the victim. Killed by event_target:assassin, who was seen.
assassins_murder_known_effect = {
	if = {
		limit = { event_target:assassin = { society_member_of = eshin_society } }
		death = { death_reason = death_murder_eshin killer = event_target:assassin }
	}
	else = {
		death = { death_reason = death_murder_hashashin killer = event_target:assassin }
	}
}

# Scope: an offered mission target. Clears the reasons the order picked him
# for this mission (assassins.201 - 202). Lasting reasons (a struck member,
# a debt, the mark of death) stay.
assassins_clear_target_flags_effect = {
	clr_character_flag = assassins_target_persecutor
	clr_character_flag = assassins_target_upstart
	clr_character_flag = assassins_target_chaos_cultist
	clr_character_flag = assassins_target_fallback
	clr_character_flag = assassins_target_war_attacker
	clr_character_flag = assassins_target_war_attacker_vassal
	clr_character_flag = assassins_target_war_attacker_commander
}
