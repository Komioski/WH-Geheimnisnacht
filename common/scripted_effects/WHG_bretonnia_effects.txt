# Scope: the loser of a tilt. How the fall goes: most are merely unhorsed,
# some wounded, a few crippled and one in a hundred killed. A rider who took
# a purse falls on purpose and comes to no harm. The host and the players at
# the tourney hear of a crippling or a death (tourney.29).
# Used by: WHG_bretonnia_effects
tourney_fall_effect = {
	if = {
		limit = { has_character_flag = tourney_bribed }
		set_character_flag = tourney_fate_threw
	}
	else = {
		random_list = {
			88 = {
				set_character_flag = tourney_fate_unhorsed
			}
			8 = {
				set_character_flag = tourney_fate_wounded
				if = {
					limit = { NOT = { trait = wounded } }
					add_trait = wounded
				}
			}
			3 = {
				set_character_flag = tourney_fate_crippled
				random_list = {
					25 = {
						modifier = { factor = 0 trait = one_eyed }
						add_trait = one_eyed
					}
					25 = {
						modifier = { factor = 0 trait = one_legged }
						add_trait = one_legged
					}
					25 = {
						modifier = { factor = 0 trait = disfigured }
						add_trait = disfigured
					}
					25 = {
						modifier = { factor = 0 trait = maimed }
						add_trait = maimed
					}
				}
				if = {
					limit = { NOT = { trait = wounded } }
					add_trait = wounded
				}
			}
			1 = {
				set_character_flag = tourney_fate_killed
			}
		}
	}
	if = {
		limit = {
			OR = {
				has_character_flag = tourney_fate_crippled
				has_character_flag = tourney_fate_killed
			}
		}
		event_target:tourney_host = {
			if = {
				limit = { NOT = { has_character_flag = tourney_no_msg_fallen } }
				character_event = { id = tourney.29 }
			}
			any_opinion_modifier_target = {
				limit = {
					ai = no
					NOT = { has_character_flag = tourney_no_msg_fallen }
					NOT = { character = event_target:tourney_tilt_winner }
					NOT = { character = event_target:tourney_tilt_loser }
					OR = {
						has_opinion_modifier = { who = PREV modifier = opinion_attending_tournament }
						has_opinion_modifier = { who = PREV modifier = opinion_spectating_tournament }
					}
				}
				character_event = { id = tourney.29 }
			}
		}
	}
	if = {
		limit = { has_character_flag = tourney_fate_killed }
		if = {
			limit = { is_invincible_trigger = yes }
			antideath_effect = yes
		}
		else = {
			death = {
				death_reason = death_duel
				killer = event_target:tourney_tilt_winner
			}
		}
	}
}

# Scope: a character. ROOT is the host. Sends a letter to ride (tourney.2) to
# a fit rider good enough for this tourney's size, or else, to a lord, a
# letter to watch (tourney.3).
# Used by: WHG_bretonnia_effects
tourney_send_invitation_effect = {
	if = {
		limit = {
			NOT = { character = ROOT }
			tourney_can_ride_trigger = yes
			OR = {
				AND = {
					ROOT = { has_character_flag = tourney_local }
					tourney_local_rider_trigger = yes
				}
				tourney_worthy_rider_trigger = yes
			}
		}
		letter_event = { id = tourney.2 days = 2 random = 5 }
	}
	else_if = {
		limit = { tourney_can_watch_trigger = yes }
		letter_event = { id = tourney.3 days = 7 random = 5 }
	}
}

# The man the Damsel must hunt (bretonmagic.3): from her Bretonnian liege's
# realm first, else a neighbouring Bretonnian realm. Prefers men she
# dislikes.
# Scope: ROOT is the Damsel. Saves event_target:male_target.
# Used by: WHG_bretonnia_events
damsel_find_male_mage_effect = {
	if = {
		limit = {
			top_liege = {
				damsel_realm_trigger = yes
				any_realm_character = { kislev_male_mage_trigger = yes }
			}
		}
		top_liege = {
			random_realm_character = {
				limit = { kislev_male_mage_trigger = yes }
				score_value = { value = 10 potential_rival_score = yes }
				score_value = { value = 10 disliked_target_selection_score = yes }
				save_event_target_as = male_target
			}
		}
	}
	else = {
		random_neighbor_independent_ruler = {
			limit = {
				damsel_realm_trigger = yes
				any_realm_character = {
					kislev_male_mage_trigger = yes
					ROOT = { capital_scope = { distance = { where = PREV distance < 1000 } } }
				}
			}
			random_realm_character = {
				limit = {
					kislev_male_mage_trigger = yes
					ROOT = { capital_scope = { distance = { where = PREV distance < 1000 } } }
				}
				score_value = { value = 10 potential_rival_score = yes }
				score_value = { value = 10 disliked_target_selection_score = yes }
				save_event_target_as = male_target
			}
		}
	}
}

# The girl the Damsel must recruit (bretonmagic.8), in order: a player, her
# own daughter, a woman of her realm, someone of her Bretonnian liege's
# realm, of a neighbouring Bretonnian realm. Prefers young girls with strong
# gifts (ice_witch_recruit_score).
# Scope: ROOT is the Damsel. Saves event_target:recruit_to_be.
# Used by: WHG_bretonnia_events
damsel_find_recruit_effect = {
	if = {
		limit = { any_player = { damsel_recruit_trigger = yes } }
		random_player = {
			limit = { damsel_recruit_trigger = yes }
			save_event_target_as = recruit_to_be
		}
	}
	else_if = {
		limit = { any_child = { damsel_recruit_trigger = yes } }
		random_child = {
			limit = { damsel_recruit_trigger = yes }
			score_value = { value = 10 ice_witch_recruit_score = yes }
			save_event_target_as = recruit_to_be
		}
	}
	else_if = {
		limit = { any_realm_lord = { damsel_recruit_trigger = yes } }
		random_realm_lord = {
			limit = { damsel_recruit_trigger = yes }
			score_value = { value = 10 ice_witch_recruit_score = yes }
			save_event_target_as = recruit_to_be
		}
	}
	else_if = {
		limit = {
			top_liege = {
				damsel_realm_trigger = yes
				any_realm_character = { damsel_recruit_trigger = yes }
			}
		}
		top_liege = {
			random_realm_character = {
				limit = { damsel_recruit_trigger = yes }
				score_value = { value = 10 ice_witch_recruit_score = yes }
				save_event_target_as = recruit_to_be
			}
		}
	}
	else = {
		random_neighbor_independent_ruler = {
			limit = {
				damsel_realm_trigger = yes
				any_realm_character = { damsel_recruit_trigger = yes }
			}
			random_realm_character = {
				limit = { damsel_recruit_trigger = yes }
				score_value = { value = 10 ice_witch_recruit_score = yes }
				save_event_target_as = recruit_to_be
			}
		}
	}
}

# Scope: a rider. Leaves the lists for good (unhorsed, hurt, thrown out or
# yielded). At the semi-finals or the final, earns third or second place.
# Used by: WHG_tournament_events, WHG_bretonnia_effects
tourney_leave_lists_effect = {
	set_character_flag = tourney_out
	clr_character_flag = tourney_tilting
	if = {
		limit = { event_target:tourney_host = { has_character_flag = tourney_final_round } }
		set_character_flag = tourney_second
	}
	else_if = {
		limit = { event_target:tourney_host = { has_character_flag = tourney_semi_round } }
		set_character_flag = tourney_third
	}
}

# Scope: a rider (ROOT). event_target:tourney_foe is the other. The two ride
# at each other until one is unhorsed. Skill at arms decides most tilts; a
# rider who took a purse to lose always falls, and the Lady's blessing or a
# brave heart help. The loser may be hurt, crippled or, one time in a
# hundred, killed. Players are told the result (tourney.27, tourney.28).
# Used by: WHG_tournament_events
tourney_tilt_effect = {
	random_list = {
		50 = {
			modifier = {
				factor = 0
				has_character_flag = tourney_bribed
			}
			modifier = {
				factor = 1000
				event_target:tourney_foe = { has_character_flag = tourney_bribed }
				NOT = { has_character_flag = tourney_bribed }
			}
			modifier = {
				factor = 1.5
				combat_rating_diff = { who = event_target:tourney_foe value = 10 }
			}
			modifier = {
				factor = 1.5
				combat_rating_diff = { who = event_target:tourney_foe value = 25 }
			}
			modifier = {
				factor = 2
				combat_rating_diff = { who = event_target:tourney_foe value = 50 }
			}
			modifier = {
				factor = 2
				combat_rating_diff = { who = event_target:tourney_foe value = 100 }
			}
			modifier = {
				factor = 1.5
				has_character_flag = tourney_blessed
			}
			modifier = {
				factor = 1.2
				trait = brave
			}
			modifier = {
				factor = 0.8
				trait = craven
			}
			modifier = {
				factor = 0.7
				OR = {
					trait = wounded
					is_ill = yes
				}
			}
			save_event_target_as = tourney_tilt_winner
			event_target:tourney_foe = { save_event_target_as = tourney_tilt_loser }
		}
		50 = {
			modifier = {
				factor = 0
				event_target:tourney_foe = { has_character_flag = tourney_bribed }
			}
			modifier = {
				factor = 1000
				has_character_flag = tourney_bribed
				event_target:tourney_foe = { NOT = { has_character_flag = tourney_bribed } }
			}
			modifier = {
				factor = 1.5
				event_target:tourney_foe = { combat_rating_diff = { who = ROOT value = 10 } }
			}
			modifier = {
				factor = 1.5
				event_target:tourney_foe = { combat_rating_diff = { who = ROOT value = 25 } }
			}
			modifier = {
				factor = 2
				event_target:tourney_foe = { combat_rating_diff = { who = ROOT value = 50 } }
			}
			modifier = {
				factor = 2
				event_target:tourney_foe = { combat_rating_diff = { who = ROOT value = 100 } }
			}
			modifier = {
				factor = 1.5
				event_target:tourney_foe = { has_character_flag = tourney_blessed }
			}
			modifier = {
				factor = 1.2
				event_target:tourney_foe = { trait = brave }
			}
			modifier = {
				factor = 0.8
				event_target:tourney_foe = { trait = craven }
			}
			modifier = {
				factor = 0.7
				event_target:tourney_foe = {
					OR = {
						trait = wounded
						is_ill = yes
					}
				}
			}
			event_target:tourney_foe = { save_event_target_as = tourney_tilt_winner }
			save_event_target_as = tourney_tilt_loser
		}
	}
	event_target:tourney_tilt_winner = {
		clr_character_flag = tourney_tilting
		clr_character_flag = tourney_blessed
		if = {
			limit = { event_target:tourney_tilt_loser = { higher_tier_than = COUNT } }
			prestige = 10
		}
		else = {

# Scope: the host. Invites the right people for the tourney's size:
# - local: the host's own realm and court, the host's liege and the liege's
#   other vassals;
# - regional: the realm of the host's king-tier liege (a Bretonnian
#   dukedom), or the host's own realm if the host is of that rank or higher;
# - grand: the whole realm of the top liege.
# Used by: WHG_tournament_events
tourney_invite_effect = {
	if = {
		limit = { has_character_flag = tourney_grand }
		top_liege = {
			if = {
				limit = { NOT = { character = ROOT } }
				tourney_send_invitation_effect = yes
			}
			any_realm_character = { tourney_send_invitation_effect = yes }
		}
	}
	else_if = {
		limit = {
			has_character_flag = tourney_regional
			lower_tier_than = KING
			any_liege = { tier = KING }
		}
		any_liege = {
			limit = { tier = KING }
			tourney_send_invitation_effect = yes
			any_realm_character = { tourney_send_invitation_effect = yes }
		}
	}
	else = {
		any_realm_character = { tourney_send_invitation_effect = yes }
		liege = {
			if = {
				limit = { NOT = { character = ROOT } }
				tourney_send_invitation_effect = yes
				any_vassal = {
					limit = { NOT = { character = ROOT } }
					tourney_send_invitation_effect = yes
				}
			}
		}
	}
}

# Cursed Mousillon follows its ruler: every Mousillon province the character
# holds takes the curse under a vampire and loses it under anyone else.
# Scope: the holder of one or more Mousillon counties.
# Used by: WHG_mousillon_events
mousillon_cursed_update_effect = {
	if = {
		limit = { is_vampire_trigger = yes }
		any_demesne_province = {
			limit = {
				mousillon_province_trigger = yes
				NOT = { has_province_modifier = cursed_mousillon }
			}
			add_province_modifier = { name = cursed_mousillon duration = -1 }
		}
	}
	else = {
		any_demesne_province = {
			limit = {
				mousillon_province_trigger = yes
				has_province_modifier = cursed_mousillon
			}
			remove_province_modifier = cursed_mousillon
		}
	}
}

# Counts a Mousillon tale lived through. The third earns "the Damned".
# Scope: the character.
# Used by: WHG_mousillon_events
mousillon_survived_effect = {
	change_variable = { which = mousillon_tales value = 1 }
	if = {
		limit = {
			check_variable = { which = mousillon_tales value >= 3 }
			NOT = { has_nickname = yes }
		}
		give_nickname = nick_the_damned
	}
}

# Saves event_target:mousillon_province: a Mousillon province the character
# holds, else the one they are in.
# Scope: the character.
# Used by: WHG_mousillon_events
mousillon_save_province_effect = {
	if = {
		limit = { any_demesne_province = { mousillon_province_trigger = yes } }
		random_demesne_province = {
			limit = { mousillon_province_trigger = yes }
			save_event_target_as = mousillon_province
		}
	}
	else = {
		location = { save_event_target_as = mousillon_province }
	}
}
