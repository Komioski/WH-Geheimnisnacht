###############################################################################

# The envoy goes home to Cathay and leaves the map; an undying envoy is
# spared instead.
# Scope: the envoy.
cathayart_envoy_home_effect = {
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = { death = { death_reason = death_went_back_to_cathay } }
}

# A keepsake of Cathay a prince or princess might have loved: art, a
# statue, a robe or a tapestry. Saved as event_target:cathayan_artifact.
# Scope: the dead prince or princess.
cathayart_keepsake_effect = {
	random_list = {
		20 = { add_artifact = cathayan_artwork }
		20 = { add_artifact = cathayan_calligraphy }
		20 = { add_artifact = cathayan_sculpture }
		15 = { add_artifact = cathayan_ceremonial_robes }
		10 = { add_artifact = cathayan_tapestry }
		10 = { add_artifact = cathayan_bronze_sculpture }
		5 = { add_artifact = silk_shroud }
	}
	new_artifact = { save_event_target_as = cathayan_artifact }
}
