# Curbs the merchants of one of the vassal's cities for ten years and tells
# the vassal (celestial.33).
# Scope: the vassal; ROOT is the liege.
celestial_reprimand_vassal_effect = {
	random_realm_province = {
		limit = { celestial_reprimand_province_trigger = yes }
		add_province_modifier = {
			name = reprimanded_merchants
			duration = 3650
		}
	}
	character_event = { id = celestial.33 days = 14 random = 30 }
}
