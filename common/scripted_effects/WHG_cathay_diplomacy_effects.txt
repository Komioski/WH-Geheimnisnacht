# Calls one more lord of the contest's rank to the Emperor's trade contest
# (celestial.70).
# Scope: the Jade Viceroy, with the rank's celestial_contest_tier_* flag.
celestial_pick_contestant_effect = {
	random_playable_ruler = {
		limit = {
			celestial_trade_contestant_trigger = yes
			OR = {
				AND = { ROOT = { has_character_flag = celestial_contest_tier_count } tier = COUNT }
				AND = { ROOT = { has_character_flag = celestial_contest_tier_duke } tier = DUKE }
				AND = { ROOT = { has_character_flag = celestial_contest_tier_king } tier = KING }
				AND = {
					ROOT = { has_character_flag = celestial_contest_tier_emperor }
					tier = EMPEROR
				}
			}
		}
		set_character_flag = trade_competition_participant
	}
}

# Stores the renown each contestant has won since the contest began
# (celestial.78, celestial.80).
# Scope: any.
celestial_contest_measure_effect = {
	any_playable_ruler = {
		limit = { has_character_flag = trade_competition_accumulate_prestige }
		export_to_variable = { which = celestial_contest_gain value = prestige }
		subtract_variable = { which = celestial_contest_gain which = celestial_contest_start }
	}
}

# Picks the winner's prize: an Imperial trade contract (with 750 grace) if
# none is held, else a Cathayan treasure (with 750 grace) once per life and
# only if none is held, else 1,500 grace (celestial.75, celestial.81).
# Scope: the winner.
celestial_contest_prize_setup_effect = {
	if = {
		limit = { NOT = { has_character_modifier = cathayan_imperial_trade_contract } }
		set_character_flag = trade_competition_winner_modifier
	}
	else_if = {
		limit = { celestial_gift_allowed_trigger = yes }
		set_character_flag = trade_competition_winner_artifact
		set_character_flag = celestial_gift_received
		celestial_viceroy_gift_effect = yes
	}
	else = { set_character_flag = trade_competition_winner_grace }
}

# Grants the prize picked by celestial_contest_prize_setup_effect.
# Scope: the winner; FROM is the Jade Viceroy.
celestial_contest_prize_effect = {
	if = {
		limit = { has_character_flag = trade_competition_winner_modifier }
		add_character_modifier = { name = cathayan_imperial_trade_contract years = 15 }
		add_offmap_currency = { offmap = offmap_cathay value = 750 }
	}
	else_if = {
		limit = { has_character_flag = trade_competition_winner_artifact }
		tooltip = {
			event_target:cathayan_artifact = {
				show_scope_change = no
				transfer_artifact = { from = FROM to = ROOT }
			}
		}
		add_offmap_currency = { offmap = offmap_cathay value = 750 }
	}
	else = { add_offmap_currency = { offmap = offmap_cathay value = 1500 } }
	clr_character_flag = trade_competition_winner_modifier
	clr_character_flag = trade_competition_winner_artifact
	clr_character_flag = trade_competition_winner_grace
}

# Turns a fresh Cathayan into a trained expert: skills, education, gifts,
# flaws and one of four temperaments (conservative, supportive, decisive,
# sociable), each with fitting lifestyle and personality traits. The
# profession is the flag the caller set first: is_cathayan_administrator,
# is_cathayan_master_engineer, is_cathayan_strategist or is_cathayan_doctor.
# The physician has seven temperaments of their own
# (celestial_expert_physician_effect). A golden age in Cathay sends better
# trained experts.
# Used by: celestial.95.
# Scope: the expert.
celestial_expert_setup_effect = {
	set_character_flag = originated_from_cathayan_court
	set_character_flag = no_court_invites
	set_character_flag = ignore_refund
	# Skills, the job's modifier and schooling
	if = {
		limit = { has_character_flag = is_cathayan_administrator }
		change_diplomacy = 2
		change_stewardship = 4
		change_learning = 3
		add_character_modifier = { name = jd_administrator duration = -1 }
		random_list = {
			30 = { add_trait = thrifty_clerk }
			15 = { add_trait = fortune_builder }
			5 = { add_trait = midas_touched }
			30 = { add_trait = martial_cleric }
			15 = { add_trait = scholarly_theologian }
			5 = { add_trait = mastermind_theologian }
		}
	}
	else_if = {
		limit = { has_character_flag = is_cathayan_master_engineer }
		change_martial = 3
		change_stewardship = 4
		change_intrigue = 1
		change_learning = 1
		add_character_modifier = { name = jd_master_engineer duration = -1 }
		random_list = {
			30 = { add_trait = thrifty_clerk }
			15 = { add_trait = fortune_builder }
			5 = { add_trait = midas_touched }
			30 = { add_trait = tough_soldier }
			15 = { add_trait = skilled_tactician }
			5 = { add_trait = brilliant_strategist }
		}
	}
	else_if = {
		limit = { has_character_flag = is_cathayan_strategist }
		change_martial = 4
		change_stewardship = 1
		change_intrigue = 2
		add_character_modifier = { name = jd_strategist duration = -1 }
		random_list = {
			80 = { add_trait = skilled_tactician }
			20 = { add_trait = brilliant_strategist }
		}
	}
	else = {
		change_diplomacy = 3
		change_stewardship = 1
		change_intrigue = 1
		change_learning = 4
		random_list = {
			67 = { add_trait = scholarly_theologian }
			33 = { add_trait = mastermind_theologian }
		}
		add_trait = physician
	}
	# Gifts of mind and body
	if = {
		limit = { has_character_flag = is_cathayan_strategist }
		random_list = {
			20 = { add_trait = brave }
			20 = { add_trait = strong }
			20 = { add_trait = shrewd }
			40 = { add_trait = robust }
		}
	}
	else = {
		random_list = {
			20 = { }
			30 = { add_trait = erudite }
			25 = { add_trait = shrewd }
			15 = { add_trait = quick }
			5 = { add_trait = genius }
			5 = {
				trigger = { NOT = { has_character_flag = is_cathayan_doctor } }
				add_trait = fair
			}
			10 = {
				trigger = { has_character_flag = is_cathayan_master_engineer }
				add_trait = robust
			}
		}
	}
	# Flaws: a soldier's scars, or a scholar's frailties
	if = {
		limit = { has_character_flag = is_cathayan_strategist }
		random_list = {
			71 = { }
			20 = { add_scarred_tiered_effect = yes }
			5 = {
				trigger = { has_dlc = "Reapers" }
				add_trait = one_eyed
				change_martial = 1
			}
			2 = {
				trigger = { has_dlc = "Reapers" }
				add_trait = one_handed
				change_martial = 1
			}
			2 = { add_trait = ugly }
		}
	}
	else = {
		random_list = {
			85 = { }
			1 = { add_trait = clubfooted }
			2 = { add_trait = harelip }
			2 = { add_trait = hunchback }
			3 = { add_trait = lisp }
			3 = { add_trait = stutter }
			2 = {
				add_trait = ugly
				remove_trait = fair
			}
			1 = { add_trait = dwarf }
			1 = {
				add_trait = weak
				remove_trait = robust
			}
		}
	}
	random = {
		chance = 10
		add_trait = homosexual
	}
	# Temperament
	if = {
		limit = { has_character_flag = is_cathayan_doctor }
		celestial_expert_physician_effect = yes
	}
	else = {
		celestial_expert_temperament_effect = yes
	}
}

# Sends the traveller back to the road after a road scene: one road scene
# fewer to go, then the next roll (kowtow.3).
# Scope: any; acts on event_target:kow_tow_traveler.
kowtow_return_to_road_effect = {
	hidden_effect = {
		event_target:kow_tow_traveler = {
			subtract_variable = { which = cathay_kow_tow_travel_events value = 1 }
			character_event = { id = kowtow.3 days = 90 random = 60 }
		}
	}
}

# Sends the traveller on after a scene in Cathay: one scene fewer to go,
# then the next roll (kowtow.51).
# Scope: any; acts on event_target:kow_tow_traveler.
kowtow_return_to_cathay_effect = {
	hidden_effect = {
		event_target:kow_tow_traveler = {
			subtract_variable = { which = cathay_kow_tow_travel_events value = 1 }
			character_event = { id = kowtow.51 days = 70 random = 30 }
		}
	}
}

# Ends the journey: the travel trait, the do-not-disturb flag, the scene
# flags and the Emperor's impression.
# Scope: the traveller.
kowtow_clean_up_effect = {
	remove_trait = kow_tow_travels
	clr_character_flag = do_not_disturb
	clr_character_flag = read_the_guanzi
	clr_character_flag = missed_loved_one
	clr_character_flag = met_mercenaries
	clr_character_flag = met_traveling_merchant
	clr_character_flag = visited_market
	clr_character_flag = kow_tow_invited_by_local_lord
	clr_character_flag = kow_tow_heavy_rain
	clr_character_flag = kow_tow_played_go
	clr_character_flag = kow_tow_hide_abort_notification
	clr_character_flag = kow_tow_visited_temple
	set_variable = { which = cathay_kow_tow_travel_events value = 0 }
	set_variable = { which = kow_tow_emperor_impression_evaluation value = 0 }
}

# Calls the journey off (war, a closed Cathay): tells the traveller unless
# hidden, stops the pending scenes and lets them try again later.
# Scope: the traveller.
kowtow_abort_effect = {
	if = {
		limit = { NOT = { has_character_flag = kow_tow_hide_abort_notification } }
		character_event = { id = kowtow.71 }
	}
	clear_delayed_event = { id = kowtow.3 }
	clear_delayed_event = { id = kowtow.21 }
	clear_delayed_event = { id = kowtow.23 }
	clear_delayed_event = { id = kowtow.50 }
	clear_delayed_event = { id = kowtow.51 }
	clear_delayed_event = { id = kowtow.60 }
	clr_character_flag = went_on_kow_tow_pilgrimage
	kowtow_clean_up_effect = yes
}

# The price in pride of bowing to the Dragon Emperor: by rank (a nomad pays
# as a duke). The Asur pay far more: 2,500 for a city, 25,000 for the
# Phoenix King, 5,000 for other lords.
# Scope: the traveller.
kowtow_prestige_cost_effect = {
	if = {
		limit = { trait = creature_elf }
		if = {
			limit = { is_republic = yes }
			prestige = -2500
		}
		else_if = {
			limit = { has_landed_title = e_ulthuan }
			prestige = -25000
		}
		else = { prestige = -5000 }
	}
	else_if = {
		limit = { is_nomadic = yes }
		prestige = -250
	}
	else = {
		trigger_switch = {
			on_trigger = real_tier
			DUKE = { prestige = -250 }
			KING = { prestige = -500 }
			EMPEROR = { prestige = -1000 }
		}
	}
}
