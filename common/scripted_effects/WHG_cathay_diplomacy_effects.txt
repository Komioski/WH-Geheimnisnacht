# Curbs the merchants of one of the vassal's cities for ten years and tells
# the vassal (celestial.33).
# Scope: the vassal; ROOT is the liege.
celestial_reprimand_vassal_effect = {
	random_realm_province = {
		limit = { celestial_reprimand_province_trigger = yes }
		add_province_modifier = { name = reprimanded_merchants duration = 3650 }
	}
	character_event = { id = celestial.33 days = 14 random = 30 }
}

# Sends a traveller to the Jade Viceroy's court, showing whether Cathay likes
# them (celestial.52).
# Scope: the traveller.
celestial_send_traveller_effect = {
	check_likes_or_dislikes_THIS_effect = yes
	add_character_modifier = { name = out_traveling duration = 365 }
}

# Calls one more lord of the contest's rank to the Emperor's trade contest
# (celestial.70).
# Scope: the Jade Viceroy, with the rank's celestial_contest_tier_* flag.
celestial_pick_contestant_effect = {
	random_playable_ruler = {
		limit = {
			celestial_trade_contestant_trigger = yes
			OR = {
				AND = { ROOT = { has_character_flag = celestial_contest_tier_count } tier = COUNT }
				AND = { ROOT = { has_character_flag = celestial_contest_tier_duke } tier = DUKE }
				AND = { ROOT = { has_character_flag = celestial_contest_tier_king } tier = KING }
				AND = {
					ROOT = { has_character_flag = celestial_contest_tier_emperor }
					tier = EMPEROR
				}
			}
		}
		set_character_flag = trade_competition_participant
	}
}

# Stores the renown each contestant has won since the contest began
# (celestial.78, celestial.80).
# Scope: any.
celestial_contest_measure_effect = {
	any_playable_ruler = {
		limit = { has_character_flag = trade_competition_accumulate_prestige }
		export_to_variable = { which = celestial_contest_gain value = prestige }
		subtract_variable = { which = celestial_contest_gain which = celestial_contest_start }
	}
}

# Picks the winner's prize: an Imperial trade contract (with 750 grace) if
# none is held, else a Cathayan treasure (with 750 grace) once per life and
# only if none is held, else 1,500 grace (celestial.75, celestial.81).
# Scope: the winner.
celestial_contest_prize_setup_effect = {
	if = {
		limit = { NOT = { has_character_modifier = cathayan_imperial_trade_contract } }
		set_character_flag = trade_competition_winner_modifier
	}
	else_if = {
		limit = { celestial_gift_allowed_trigger = yes }
		set_character_flag = trade_competition_winner_artifact
		set_character_flag = celestial_gift_received
		celestial_viceroy_gift_effect = yes
	}
	else = { set_character_flag = trade_competition_winner_grace }
}

# Grants the prize picked by celestial_contest_prize_setup_effect.
# Scope: the winner; FROM is the Jade Viceroy.
celestial_contest_prize_effect = {
	if = {
		limit = { has_character_flag = trade_competition_winner_modifier }
		add_character_modifier = { name = cathayan_imperial_trade_contract years = 15 }
		add_offmap_currency = { offmap = offmap_cathay value = 750 }
	}
	else_if = {
		limit = { has_character_flag = trade_competition_winner_artifact }
		tooltip = {
			event_target:cathayan_artifact = {
				show_scope_change = no
				transfer_artifact = { from = FROM to = ROOT }
			}
		}
		add_offmap_currency = { offmap = offmap_cathay value = 750 }
	}
	else = { add_offmap_currency = { offmap = offmap_cathay value = 1500 } }
	clr_character_flag = trade_competition_winner_modifier
	clr_character_flag = trade_competition_winner_artifact
	clr_character_flag = trade_competition_winner_grace
}

# Clears a contestant's contest flags and records.
# Scope: a contestant.
celestial_contest_clear_effect = {
	clr_character_flag = trade_competition_participant
	clr_character_flag = trade_competition_build_city
	clr_character_flag = trade_competition_accumulate_prestige
	clr_character_flag = trade_competition_city_winner
	clr_character_flag = trade_competition_winner
	clr_character_flag = started_building_holding
	set_variable = { which = celestial_contest_start value = 0 }
	set_variable = { which = celestial_contest_gain value = 0 }
}
