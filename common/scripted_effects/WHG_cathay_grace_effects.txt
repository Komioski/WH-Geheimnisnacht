# Gives this Cathayan a commander's seat: a free one, else that of the
# weakest commander of ROOT's realm who was not Cathayan-trained (a
# landless one first).
# Scope: the Cathayan; ROOT is the lord.
cathaygrace_seat_commander_effect = {
	if = {
		limit = { can_hold_title = title_commander ROOT = { can_grant_title = title_commander } }
		give_minor_title = title_commander
	}
	else_if = {
		limit = {
			NOT = { has_minor_title = title_commander }
			ROOT = { any_realm_character = { cathaygrace_replaceable_commander_trigger = yes } }
		}
		ROOT = {
			random_realm_character = {
				limit = { cathaygrace_replaceable_commander_trigger = yes }
				preferred_limit = { NOT = { higher_tier_than = BARON } NOT = { martial = 10 } }
				preferred_limit = { NOT = { higher_tier_than = BARON } }
				remove_title = title_commander
			}
		}
		give_minor_title = title_commander
	}
}
