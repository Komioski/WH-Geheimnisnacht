# Saves one of the road ends at the eastern edge of the map as
# event_target:cathayinv_road_end: the four great roads east most often,

# Where the Viceroy's armies muster: his capital when he holds land, a
# road end otherwise. Saved as
# event_target:cathayan_invasion_spawn_point_province (the war CBs read it).
# Scope: the Jade Viceroy.
cathayinv_spawn_point_effect = {
	if = {
		limit = { has_offmap_tmp_flag = WP_is_on_map }
		capital_scope = { save_event_target_as = cathayan_invasion_spawn_point_province }
	}
	else = {
		cathayinv_road_end_effect = yes
		event_target:cathayinv_road_end = {
			save_event_target_as = cathayan_invasion_spawn_point_province
		}
	}
}

# Picks a small realm to ask for tribute, saved as
# event_target:cathayan_invasion_target. Rulers reached through Cathay's
# tributaries come first (no border gore), then those the Emperor likes,
# then raiders of Cathay; lords who sell the Viceroy horses come last.
# Scope: the Jade Viceroy.
cathayinv_pick_small_tributary_effect = {
	random_independent_ruler = {
		limit = {
			cathayinv_in_reach_trigger = yes
			OR = { cathayinv_small_realm_trigger = yes cathayinv_raider_trigger = yes }
		}
		preferred_limit = {
			valid_offmap_cathay_target_location_tributary_chain = yes
			cathayinv_small_realm_trigger = yes
			cathayinv_liked_trigger = yes
			cathayinv_raider_trigger = yes
		}
		preferred_limit = {
			valid_offmap_cathay_target_location_tributary_chain = yes
			cathayinv_small_realm_trigger = yes
			cathayinv_liked_trigger = yes
			NOT = { has_character_modifier = cathay_supply_horses }
		}
		preferred_limit = {
			valid_offmap_cathay_target_location_tributary_chain = yes
			cathayinv_small_realm_trigger = yes
			cathayinv_liked_trigger = yes
		}
		preferred_limit = {
			valid_offmap_cathay_target_location_tributary_chain = yes
			cathayinv_small_realm_trigger = yes
			cathayinv_raider_trigger = yes
		}
		preferred_limit = {
			valid_offmap_cathay_target_location_tributary_chain = yes
			cathayinv_small_realm_trigger = yes
			NOT = { has_character_modifier = cathay_supply_horses }
		}
		preferred_limit = {
			valid_offmap_cathay_target_location_tributary_chain = yes
			cathayinv_small_realm_trigger = yes
		}
		preferred_limit = {
			valid_offmap_cathay_target_location_tributary_chain = yes
			cathayinv_raider_trigger = yes
		}
		preferred_limit = {
			valid_offmap_cathay_target_location = yes
			cathayinv_small_realm_trigger = yes
			cathayinv_liked_trigger = yes
			cathayinv_raider_trigger = yes
		}
		preferred_limit = {
			valid_offmap_cathay_target_location = yes
			cathayinv_small_realm_trigger = yes
			cathayinv_liked_trigger = yes
			NOT = { has_character_modifier = cathay_supply_horses }
		}
		preferred_limit = {
			valid_offmap_cathay_target_location = yes
			cathayinv_small_realm_trigger = yes
			cathayinv_liked_trigger = yes
		}
		preferred_limit = {
			valid_offmap_cathay_target_location = yes
			cathayinv_small_realm_trigger = yes
			cathayinv_raider_trigger = yes
		}
		preferred_limit = {
			valid_offmap_cathay_target_location = yes
			cathayinv_small_realm_trigger = yes
			NOT = { has_character_modifier = cathay_supply_horses }
		}
		preferred_limit = {
			valid_offmap_cathay_target_location = yes
			cathayinv_small_realm_trigger = yes
		}
		preferred_limit = {
			valid_offmap_cathay_target_location = yes
			cathayinv_raider_trigger = yes
		}
		save_event_target_as = cathayan_invasion_target
	}
}

# Gives a displaced prince a parent of the fallen house: the last Emperor,
# his father, another Emperor or Empress of the house, or any man of it.
# Scope: the prince; event_target:previousEmperor is the last Emperor.
cathayinv_prince_parent_effect = {
	if = {
		limit = {
			event_target:previousEmperor = { could_be_parent_of = event_target:displaced_prince }
		}
		set_father = event_target:previousEmperor
	}
	else_if = {
		limit = {
			event_target:previousEmperor = {
				father_even_if_dead = { could_be_parent_of = event_target:displaced_prince }
			}
		}
		event_target:previousEmperor = {
			father_even_if_dead = { event_target:displaced_prince = { set_father = PREV } }
		}
	}
	else_if = {
		limit = {
			event_target:previousEmperor = {
				any_dynasty_member_even_if_dead = {
					is_or_was_offmap_power_ruler = yes
					is_female = no
					could_be_parent_of = event_target:displaced_prince
				}
			}
		}
		event_target:previousEmperor = {
			random_dynasty_member_even_if_dead = {
				limit = {
					is_or_was_offmap_power_ruler = yes
					is_female = no
					could_be_parent_of = event_target:displaced_prince
				}
				event_target:displaced_prince = { set_father = PREV }
			}
		}
	}
	else_if = {
		limit = {
			event_target:previousEmperor = {
				any_dynasty_member_even_if_dead = {
					is_or_was_offmap_power_ruler = yes
					is_woman_trigger = yes
					could_be_parent_of = event_target:displaced_prince
				}
			}
		}
		event_target:previousEmperor = {
			random_dynasty_member_even_if_dead = {
				limit = {
					is_or_was_offmap_power_ruler = yes
					is_woman_trigger = yes
					could_be_parent_of = event_target:displaced_prince
				}
				event_target:displaced_prince = { set_mother = PREV }
			}
		}
	}
	else_if = {
		limit = {
			event_target:previousEmperor = {
				any_dynasty_member_even_if_dead = {
					is_female = no
					could_be_parent_of = event_target:displaced_prince
				}
			}
		}
		event_target:previousEmperor = {
			random_dynasty_member_even_if_dead = {
				limit = { is_female = no could_be_parent_of = event_target:displaced_prince }
				event_target:displaced_prince = { set_father = PREV }
			}
		}
	}
}
