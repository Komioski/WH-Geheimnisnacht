# Starts a raid: the raid modifier and, a little later, the Jade
# Viceroy's warning (cathayraid.5).
# Scope: the raider.
cathayraid_start_effect = {
	add_character_modifier = { name = cathay_raid_active duration = -1 }
	offmap_cathay = { governor = { save_event_target_as = cathayraid_viceroy } }
	letter_event = { id = cathayraid.5 days = 15 random = 15 }
}

# Ends a raid. No new raid for five years.
# Scope: the raider.
cathayraid_end_effect = {
	remove_character_modifier = cathay_raid_active
	add_character_modifier = { name = cathay_raid_grace_cd hidden = yes years = 5 }
}

# Picks this year's outcome and sets its flag. A Cathay in famine, plague,
# civil war or unrest is easy prey; a stable or golden Cathay guards its
# border. Treasures come once per lifetime, pandas once.
# Scope: the raider.
cathayraid_pick_outcome_effect = {
	random_list = {
		400 = { set_character_flag = cathayraid_loot }
		30 = {
			modifier = {
				factor = 1.5
				offmap_cathay = {
					OR = {
						has_status = cathay_famine
						has_status = cathay_plague
						has_status = cathay_civil_war
						has_status = cathay_unrest
					}
				}
			}
			modifier = { factor = 0.5 offmap_cathay = { has_status = cathay_golden_age } }
			set_character_flag = cathayraid_rich
		}
		5 = {
			modifier = {
				factor = 1.5
				offmap_cathay = {
					OR = {
						has_status = cathay_famine
						has_status = cathay_plague
						has_status = cathay_civil_war
						has_status = cathay_unrest
					}
				}
			}
			modifier = { factor = 0.5 offmap_cathay = { has_status = cathay_golden_age } }
			set_character_flag = cathayraid_ransom
		}
		30 = {
			modifier = {
				factor = 2
				offmap_cathay = {
					OR = { has_status = cathay_golden_age has_status = cathay_stable }
				}
			}
			modifier = {
				factor = 0.5
				offmap_cathay = {
					OR = {
						has_status = cathay_famine
						has_status = cathay_plague
						has_status = cathay_civil_war
						has_status = cathay_unrest
					}
				}
			}
			set_character_flag = cathayraid_bad
		}
		30 = { trigger = { is_adult = yes } set_character_flag = cathayraid_captive }
		30 = { set_character_flag = cathayraid_engineer }
		15 = {
			trigger = { NOT = { has_character_flag = cathayraid_treasure_taken } }
			set_character_flag = cathayraid_treasure
		}
		5 = {
			trigger = { NOT = { has_character_modifier = panda_pet } }
			set_character_flag = cathayraid_panda
		}
	}
}

# Clears this year's outcome flag.
# Scope: the raider.
cathayraid_clear_outcome_effect = {
	clr_character_flag = cathayraid_loot
	clr_character_flag = cathayraid_rich
	clr_character_flag = cathayraid_ransom
	clr_character_flag = cathayraid_bad
	clr_character_flag = cathayraid_captive
	clr_character_flag = cathayraid_engineer
	clr_character_flag = cathayraid_treasure
	clr_character_flag = cathayraid_panda
}

# The gold of this year's outcome. Hordes get more and also bring people
# home.
# Scope: the raider.
cathayraid_spoils_effect = {
	if = {
		limit = { has_character_flag = cathayraid_loot }
		if = { limit = { is_nomadic = no } scaled_wealth = { value = 0.25 min = 15 max = 75 } }
		else = { scaled_wealth = { value = 0.25 min = 30 max = 100 } population = 150 }
	}
	else_if = {
		limit = { has_character_flag = cathayraid_rich }
		if = { limit = { is_nomadic = no } scaled_wealth = { value = 1.5 min = 30 max = 200 } }
		else = { scaled_wealth = { value = 1.5 min = 60 max = 300 } population = 300 }
	}
	else_if = {
		limit = { has_character_flag = cathayraid_ransom }
		if = { limit = { is_nomadic = no } scaled_wealth = { value = 2.5 min = 60 max = 300 } }
		else = { scaled_wealth = { value = 2.5 min = 120 max = 400 } population = 600 }
	}
	else_if = {
		limit = { has_character_flag = cathayraid_treasure }
		if = { limit = { is_nomadic = no } scaled_wealth = { value = 0.75 min = 30 max = 125 } }
		else = { scaled_wealth = { value = 0.75 min = 70 max = 175 } population = 150 }
	}
	else_if = {
		limit = { has_character_flag = cathayraid_panda }
		if = { limit = { is_nomadic = no } scaled_wealth = { value = 1 min = 35 max = 150 } }
		else = { scaled_wealth = { value = 1 min = 70 max = 225 } population = 200 }
	}
	else = { # a captive or an engineer
		if = { limit = { is_nomadic = no } scaled_wealth = { value = 0.6 min = 30 max = 125 } }
		else = { scaled_wealth = { value = 0.6 min = 70 max = 175 } population = 150 }
	}
}

# A young Cathayan of a mandarin's house, carried off by the raiders. A
# woman for a lord who could take a concubine, a son held for ransom
# otherwise. Saved as event_target:cathayraid_captive.
# Scope: the raider.
cathayraid_captive_effect = {
	if = {
		limit = { is_woman_trigger = no is_no_females_race = no }
		create_character = {
			age = 20
			dynasty = actually_culture
			religion = draconite
			culture = cathayan
			female = yes
			health = 6
			random_traits = yes
			trait = creature_human
		}
	}
	else = {
		create_character = {
			age = 20
			dynasty = actually_culture
			religion = draconite
			culture = cathayan
			female = no
			health = 6
			random_traits = yes
			trait = creature_human
		}
	}
	new_character = {
		remove_trait = clubfooted
		remove_trait = harelip
		remove_trait = hunchback
		remove_trait = lisp
		remove_trait = stutter
		remove_trait = ugly
		remove_trait = dwarf
		remove_trait = weak
		add_trait = fair
		random = { chance = 35 add_trait = strong }
		random = { chance = 35 add_trait = quick }
		save_event_target_as = cathayraid_captive
	}
}

# An apprentice of a Cathayan siege workshop, dragged west. A sapper with
# high martial, better in a golden age. Saved as
# event_target:cathayraid_engineer.
# Scope: the raider.
cathayraid_engineer_effect = {
	create_character = {
		age = 32
		dynasty = actually_culture
		religion = draconite
		culture = cathayan
		female = no
		health = 6
		random_traits = yes
		trait = creature_human
		attributes = { diplomacy = 2 learning = 2 stewardship = 2 intrigue = 2 martial = 13 }
	}
	new_character = {
		set_character_flag = originated_from_cathayan_court
		set_character_flag = no_court_invites
		set_character_flag = taught_cathayan_strategy
		set_character_flag = is_cathayan_siege_engineer
		set_character_flag = ignore_refund
		remove_trait = weak
		remove_trait = slow
		remove_trait = imbecile
		remove_trait = dull
		remove_trait = craven
		remove_trait = mystic
		remove_trait = socializer
		remove_trait = gamer
		remove_trait = hedonist
		remove_trait = duelist
		remove_trait = hunter
		remove_trait = strategist
		remove_trait = administrator
		remove_trait = architect
		remove_trait = gardener
		remove_trait = schemer
		remove_trait = impaler
		remove_trait = seducer
		remove_trait = seductress
		remove_trait = theologian
		remove_trait = scholar
		add_trait = sapper
		if = { limit = { offmap_cathay = { has_status = cathay_golden_age } } change_martial = 5 }
		random_list = {
			15 = { change_martial = 2 }
			10 = { add_trait = robust }
			10 = { add_trait = duelist }
			10 = { add_trait = architect }
			5 = { add_trait = strategist }
			5 = { change_martial = 4 }
			5 = {
				if = { limit = { has_dlc = "Reapers" } add_trait = one_eyed }
				else = { change_martial = 1 }
			}
			5 = { add_trait = shrewd }
		}
		save_event_target_as = cathayraid_engineer
	}
}

# Makes the captured engineer a commander. If no seat is free, the weakest
# commander who is not Cathayan-trained is dismissed.
# Scope: the raider.
cathayraid_engineer_commander_effect = {
	if = {
		limit = {
			event_target:cathayraid_engineer = { can_hold_title = title_commander }
			can_grant_title = title_commander
		}
		event_target:cathayraid_engineer = { give_minor_title = title_commander }
	}
	else = {
		random_realm_character = {
			limit = {
				has_minor_title = title_commander
				liege = { character = ROOT }
				NOT = { higher_tier_than = BARON }
			}
			preferred_limit = {
				NOR = {
					has_character_flag = originated_from_cathayan_court
					has_character_flag = taught_cathayan_strategy
				}
				NOT = { martial = 10 }
			}
			preferred_limit = {
				NOR = {
					has_character_flag = originated_from_cathayan_court
					has_character_flag = taught_cathayan_strategy
				}
			}
			remove_title = title_commander
		}
		event_target:cathayraid_engineer = { give_minor_title = title_commander }
	}
}
