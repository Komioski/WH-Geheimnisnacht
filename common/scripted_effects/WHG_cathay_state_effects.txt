# Cathay beats off an invasion: peace returns and the memory of past
# defeats fades by one step (cathay_badly_beaten_1 - 3).
# Scope: offmap_cathay.
cathaystate_invasion_beaten_effect = {
	set_offmap_flag = no_status_news
	set_status = cathay_stable
	clr_offmap_flag = no_status_news
	if = {
		limit = { has_offmap_flag = cathay_badly_beaten_1 }
		clr_offmap_flag = cathay_badly_beaten_1
	}
	if = {
		limit = { has_offmap_flag = cathay_badly_beaten_2 }
		clr_offmap_flag = cathay_badly_beaten_2
		set_offmap_flag = cathay_badly_beaten_1
	}
	if = {
		limit = { has_offmap_flag = cathay_badly_beaten_3 }
		clr_offmap_flag = cathay_badly_beaten_3
		set_offmap_flag = cathay_badly_beaten_2
	}
}

# The west beats Cathay in an invasion war (invasion_of_cathay_cb
# on_success): the defeat is remembered (cathay_badly_beaten_1 - 3, which
# make an Expansionist turn less likely, cathaystate.10) and the court
# leans towards closing the borders (cathay_isolationist_trend).
# Scope: offmap_cathay.
cathaystate_beaten_by_west_effect = {
	if = {
		limit = { has_offmap_flag = cathay_badly_beaten_2 }
		clr_offmap_flag = cathay_badly_beaten_2
		set_offmap_flag = cathay_badly_beaten_3
	}
	else_if = {
		limit = { has_offmap_flag = cathay_badly_beaten_1 }
		clr_offmap_flag = cathay_badly_beaten_1
		set_offmap_flag = cathay_badly_beaten_2
	}
	else_if = {
		limit = { NOT = { has_offmap_flag = cathay_badly_beaten_3 } }
		set_offmap_flag = cathay_badly_beaten_1
	}
	set_offmap_flag = cathay_isolationist_trend
}

# The invaders win: unrest follows, the defeats of the west are forgotten,
# and each victory of the steppe makes the next invasion less likely
# (cathay_mongols_won_1 - 3).
# Scope: offmap_cathay.
cathaystate_invasion_won_effect = {
	set_offmap_flag = no_status_news
	set_status = cathay_unrest
	set_offmap_flag = cathay_had_unrest
	clr_offmap_flag = no_status_news
	clr_offmap_flag = cathay_badly_beaten_3
	clr_offmap_flag = cathay_badly_beaten_2
	clr_offmap_flag = cathay_badly_beaten_1
	if = {
		limit = { has_offmap_flag = cathay_mongols_won_2 }
		clr_offmap_flag = cathay_mongols_won_2
		set_offmap_flag = cathay_mongols_won_3
	}
	else_if = {
		limit = { has_offmap_flag = cathay_mongols_won_1 }
		clr_offmap_flag = cathay_mongols_won_1
		set_offmap_flag = cathay_mongols_won_2
	}
	else_if = {
		limit = { NOT = { has_offmap_flag = cathay_mongols_won_3 } }
		set_offmap_flag = cathay_mongols_won_1
	}
}

# Crowns a new Emperor once the Dragon Throne has fallen. The kind of house
# is set by an offmap flag before the call and cleared after:
# cathaystate_heir_hobgoblin (the kin of event_target:cathaystate_khan, or
# a new khan), cathaystate_heir_manchu (a Hung khagan) or none (a Cathayan
# general). The fallout follows from on_offmap_ruler_changed
# (cathaystate.1 and 10).
# Scope: the Jade Viceroy (the new Emperor is created at his court first).
cathaystate_new_emperor_effect = {
	if = {
		limit = {
			offmap_cathay = { has_offmap_flag = cathaystate_heir_hobgoblin }
			event_target:cathaystate_khan = { is_alive = yes }
		}
		create_character = {
			age = 25
			random_traits = yes
			female = no
			religion = event_target:cathaystate_khan
			culture = event_target:cathaystate_khan
			dynasty = event_target:cathaystate_khan
			trait = creature_hobgoblin
			trait = brilliant_strategist
		}
	}
	else_if = {
		limit = { offmap_cathay = { has_offmap_flag = cathaystate_heir_hobgoblin } }
		create_character = {
			age = 30
			random_traits = yes
			female = no
			religion = greenskin_gods
			culture = hobgoblin
			dynasty = actually_culture
			trait = creature_hobgoblin
			trait = brilliant_strategist
		}
	}
	else_if = {
		limit = { offmap_cathay = { has_offmap_flag = cathaystate_heir_manchu } }
		create_character = {
			age = 30
			random_traits = yes
			female = no
			religion = chaos
			culture = man_chu
			dynasty = actually_culture
			trait = creature_human
			trait = brilliant_strategist
		}
	}
	else = {
		random_list = {
			50 = {
				create_character = {
					age = 35
					random_traits = yes
					female = yes
					religion = draconite
					culture = cathayan
					dynasty = actually_culture
					trait = creature_human
					trait = brilliant_strategist
				}
			}
			50 = {
				create_character = {
					age = 35
					random_traits = yes
					female = no
					religion = draconite
					culture = cathayan
					dynasty = actually_culture
					trait = creature_human
					trait = brilliant_strategist
				}
			}
		}
	}
	new_character = { change_martial = 4 offmap_cathay = { set_offmap_holder = PREV } }
	offmap_cathay = {
		clr_offmap_flag = cathaystate_heir_hobgoblin
		clr_offmap_flag = cathaystate_heir_manchu
	}
}

# Sets the worth of the four roads east (the Ivory Road, the Old Silk Road,
# the Khyprian Road and the Spice Route) from Cathay's status and policy:
# doubled in a golden age, cut by unrest, war, famine and plague, nothing
# while Cathay is closed.
# Scope: offmap_cathay.
cathaystate_road_values_effect = {
	if = { limit = { has_policy = cathay_isolationist } cathaystate_set_road_value_0_effect = yes }
	else_if = {
		limit = { has_status = cathay_plague }
		set_trade_route_value_multiplier = { which = ivory_road value = 0.1 }
		set_trade_route_value_multiplier = { which = old_silk_road value = 0.1 }
		set_trade_route_value_multiplier = { which = khyprian_road value = 0.1 }
		set_trade_route_value_multiplier = { which = spice_route value = 0.1 }
	}
	else_if = {
		limit = { has_status = cathay_famine }
		set_trade_route_value_multiplier = { which = ivory_road value = 0.25 }
		set_trade_route_value_multiplier = { which = old_silk_road value = 0.25 }
		set_trade_route_value_multiplier = { which = khyprian_road value = 0.25 }
		set_trade_route_value_multiplier = { which = spice_route value = 0.25 }
	}
	else_if = {
		limit = { cathaystate_at_peace_trigger = no }
		set_trade_route_value_multiplier = { which = ivory_road value = 0.5 }
		set_trade_route_value_multiplier = { which = old_silk_road value = 0.5 }
		set_trade_route_value_multiplier = { which = khyprian_road value = 0.5 }
		set_trade_route_value_multiplier = { which = spice_route value = 0.5 }
	}
	else_if = {
		limit = { has_status = cathay_unrest }
		set_trade_route_value_multiplier = { which = ivory_road value = 0.75 }
		set_trade_route_value_multiplier = { which = old_silk_road value = 0.75 }
		set_trade_route_value_multiplier = { which = khyprian_road value = 0.75 }
		set_trade_route_value_multiplier = { which = spice_route value = 0.75 }
	}
	else_if = {
		limit = { has_status = cathay_golden_age }
		set_trade_route_value_multiplier = { which = ivory_road value = 2 }
		set_trade_route_value_multiplier = { which = old_silk_road value = 2 }
		set_trade_route_value_multiplier = { which = khyprian_road value = 2 }
		set_trade_route_value_multiplier = { which = spice_route value = 2 }
	}
	else = {
		set_trade_route_value_multiplier = { which = ivory_road value = 1 }
		set_trade_route_value_multiplier = { which = old_silk_road value = 1 }
		set_trade_route_value_multiplier = { which = khyprian_road value = 1 }
		set_trade_route_value_multiplier = { which = spice_route value = 1 }
	}
}

# The roads east carry nothing while Cathay is closed.
# Scope: offmap_cathay.
cathaystate_set_road_value_0_effect = {
	set_trade_route_value_multiplier = { which = ivory_road value = 0 }
	set_trade_route_value_multiplier = { which = old_silk_road value = 0 }
	set_trade_route_value_multiplier = { which = khyprian_road value = 0 }
	set_trade_route_value_multiplier = { which = spice_route value = 0 }
}

# The reward of a hobgoblin khan whose hordes beat Cathay: gold, prestige,
# grace, a peace deal and two Cathayan siege engineers dragged west. Half
# of it (and no engineers) when the Dragon Emperor pays the hordes off.
# Scope: the khan.
cathaystate_khan_reward_effect = {
	if = {
		limit = { dragon_emperor_reigns_trigger = yes }
		wealth = 500
		prestige = 500
		add_offmap_currency = { offmap = offmap_cathay value = 2500 }
		add_character_modifier = { name = peace_deal_with_cathay years = 25 inherit = yes }
	}
	else = {
		wealth = 1000
		prestige = 1000
		add_offmap_currency = { offmap = offmap_cathay value = 5000 }
		add_character_modifier = { name = peace_deal_with_cathay years = 50 inherit = yes }
		cathayraid_engineer_effect = yes
		cathayraid_engineer_effect = yes
	}
}

# A tributary of the old house reads that it is free (cathaystate.1 already
# ended the tribute; this shows it and clears the note).
# Scope: the former tributary.
cathaystate_old_tributary_effect = {
	if = {
		limit = { has_character_flag = was_tributary_of_cathay }
		tooltip = {
			offmap_cathay = {
				show_scope_change = no
				governor = { ROOT = { remove_tributary = PREV } }
			}
		}
		clr_character_flag = was_tributary_of_cathay
	}
}

# After a lost war or a fallen house, exiles head west: a displaced prince
# of the old house (only once the Dragon Throne has fallen) or a Man-chu
# warlord with his riders (jd_chinese_invasion_events). Not with the
# cathayan_invasions game rule set to none.
# Scope: the Jade Viceroy.
cathaystate_exiles_flee_effect = {
	if = {
		limit = { NOT = { has_game_rule = { name = cathayan_invasions value = none } } }
		random_list = {
			50 = {
				trigger = { dragon_emperor_reigns_trigger = no }
				character_event = { id = cathayinv.23 days = 15 } # a displaced prince
				set_character_flag = spawning_a_displaced_royal
			}
			50 = {
				character_event = { id = cathayinv.22 days = 15 } # a Man-chu warlord
				set_character_flag = spawning_a_jurchen_invader
			}
		}
	}
}
