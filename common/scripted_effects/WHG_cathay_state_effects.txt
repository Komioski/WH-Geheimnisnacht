# Gives Cathay a new name for the house that now holds the Dragon Throne.
# Never keeps the current name. Cathayan houses favour the old names of the
# south (Lotus, Bamboo, Tiger), conquerors name the empire after their own
# kind: Wolf for hobgoblins, Steppe for the Kurgan, Mountain for ogres.

# invasion wars use the same names (common/cb_types/01_cb_types.txt).
# Scope: offmap_cathay; ROOT is the new ruler.
cathaystate_new_empire_name_effect = {
	random_list = {
