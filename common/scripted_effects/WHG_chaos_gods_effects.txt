# Effects for events/WHG_chaos_gods_events.txt (the Dark Gods test their own)

###############################################################################
# Khorne's trial
###############################################################################

# Scope: the challenger (ROOT). Raises the foe of the current round of
# Khorne's trial (variable khorne_trial: 1 a young warrior of the
# challenger's people, 2 a Khornate berserker, 3 a Bloodletter) at the
# challenger's court and saves it as khorne_trial_foe.
khorne_trial_foe_effect = {
	if = {
		limit = { check_variable = { which = khorne_trial value = 3 } }
		create_character = {
			random_traits = yes
			religion = khorne
			culture = bloodletter_khorne
			female = no
			age = 300
			flag = ai_flag_refuse_conversion
			attributes = { martial = 18 }
		}
		new_character = {
			add_trait = khorne_demon
			add_trait = khorne_bloodletter
			add_trait = wroth
			add_trait = brave
			save_event_target_as = khorne_trial_foe
		}
	}
	else_if = {
		limit = { check_variable = { which = khorne_trial value = 2 } }
		create_random_soldier = {
			random_traits = yes
			dynasty = none
			religion = khorne
			culture = ROOT
			female = no
			age = 30
		}
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			add_trait = brave
			add_trait = wroth
			add_trait = strong
			add_trait = khorne_berserker
			change_martial = 2
			save_event_target_as = khorne_trial_foe
		}
	}
	else = {
		create_random_soldier = {
			random_traits = yes
			dynasty = none
			religion = khorne
			culture = ROOT
			female = no
			age = 16
		}
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			add_trait = brave
			change_martial = 1
			save_event_target_as = khorne_trial_foe
		}
	}
}

# Scope: the challenger. Ends the trial for good.
khorne_trial_end_effect = {
	set_character_flag = tested_by_Khorne
	set_variable = { which = khorne_trial value = 0 }
	event_target:khorne_trial_foe = {
		if = {
			limit = {
				is_alive = yes
				trait = khorne_bloodletter
			}
			death = { death_reason = death_vanished } # Back to the Realm of Chaos
		}
	}
}

###############################################################################
# Nurgle's embrace
###############################################################################

# Scope: the sufferer. Gains a trait of despair unless already held, and
# remembers which ones the despair gave (flag nurgle_gave_<trait>).
nurgle_despair_craven_effect = {
	if = {
		limit = { NOT = { trait = craven } }
		add_trait = craven
		set_character_flag = nurgle_gave_craven
	}
}
nurgle_despair_wroth_effect = {
	if = {
		limit = { NOT = { trait = wroth } }
		add_trait = wroth
		set_character_flag = nurgle_gave_wroth
	}
}
nurgle_despair_charitable_effect = {
	if = {
		limit = { NOT = { trait = charitable } }
		add_trait = charitable
		set_character_flag = nurgle_gave_charitable
	}
}
nurgle_despair_depressed_effect = {
	if = {
		limit = { NOT = { trait = depressed } }
		add_trait = depressed
		set_character_flag = nurgle_gave_depressed
	}
}

# Scope: the sufferer. Takes back the traits the despair gave.
nurgle_despair_lift_effect = {
	if = {
		limit = { has_character_flag = nurgle_gave_craven }
		remove_trait = craven
		clr_character_flag = nurgle_gave_craven
	}
	if = {
		limit = { has_character_flag = nurgle_gave_wroth }
		remove_trait = wroth
		clr_character_flag = nurgle_gave_wroth
	}
	if = {
		limit = { has_character_flag = nurgle_gave_charitable }
		remove_trait = charitable
		clr_character_flag = nurgle_gave_charitable
	}
	if = {
		limit = { has_character_flag = nurgle_gave_depressed }
		remove_trait = depressed
		clr_character_flag = nurgle_gave_depressed
	}
}

###############################################################################
# Tzeentch's visions
###############################################################################

# Scope: the dreamer. Loses the dull wits the visions burn away.
tzeentch_clear_dull_wits_effect = {
	remove_trait = dim
	remove_trait = dumb
	remove_trait = slow
	remove_trait = idiot
	remove_trait = imbecile
}

# Scope: the dreamer. Loses every gift of the mind.
tzeentch_lose_wits_effect = {
	remove_trait = genius
	remove_trait = brilliant
	remove_trait = quick
	remove_trait = smart
	remove_trait = bright
}
