# Scope: the chancellor (ROOT). A kingdom first (become-king ambition only,
# 2% at diplomacy below 15, 5% at 15, 10% at 18, 20% at 22), then a duchy
# (diplomacy 15: 10%, 18: 20%, 20: 30%), else the county.
chancellor_forge_claim_effect = {
	if = {
		limit = {
			liege = {
				has_ambition = obj_become_king
				NOT = { has_character_flag = kingdom_claimed_with_ambition }
				ROOT = {
					location = {
						kingdom = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { NOT = { character = ROOT } owns = PREVPREV }
						}
					}
				}
			}
		}
		random = {
			chance = 2
			modifier = { factor = 2.5 diplomacy = 15 }
			modifier = { factor = 2 diplomacy = 18 }
			modifier = { factor = 2 diplomacy = 22 }
			location = {
				kingdom = {
					save_event_target_as = target_title
					holder_scope = { character_event = { id = chancellor.11 days = 1 } }
				}
			}
			liege = { set_character_flag = kingdom_claimed_with_ambition }
			set_character_flag = chancellor_claim_done
		}
	}
	if = {
		limit = {
			NOT = { has_character_flag = chancellor_claim_done }
			diplomacy = 15
			liege = {
				ROOT = {
					location = {
						duchy = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { NOR = { is_liege_of = ROOT character = ROOT } owns = PREVPREV }
						}
					}
				}
			}
		}
		random = {
			chance = 10
			modifier = { factor = 2 diplomacy = 18 }
			modifier = { factor = 1.5 diplomacy = 20 }
			location = {
				duchy = {
					save_event_target_as = target_title
					holder_scope = { character_event = { id = chancellor.11 days = 1 } }
				}
			}
			set_character_flag = chancellor_claim_done
		}
	}
	if = {
		limit = {
			NOT = { has_character_flag = chancellor_claim_done }
			liege = {
				ROOT = {
					location = {
						county = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { NOR = { is_liege_of = ROOT character = ROOT } owns = PREVPREV }
						}
					}
				}
			}
		}
		location = {
			county = {
				save_event_target_as = target_title
				holder_scope = { character_event = { id = chancellor.11 days = 1 } }
			}
		}
	}
	clr_character_flag = chancellor_claim_done
}

# A lord gives in to the liege's letter (spymaster.22 - 25).
# Scope: the lord; FROM is the liege. Leaves the faction (flag
# faction_member) or joins the liege's (flag faction_recruit).
spymaster_faction_yield_effect = {
	if = {
		limit = { has_character_flag = faction_member }
		custom_tooltip = {
			text = EVTTOOLTIPspymaster26
			hidden_effect = {
				FROM = { letter_event = { id = spymaster.26 days = 7 } }
			}
		}
	}
	else = {
		custom_tooltip = {
			text = EVTTOOLTIPspymaster35
			hidden_effect = {
				FROM = { letter_event = { id = spymaster.35 days = 7 } }
			}
		}
	}
}

# A lord refuses the liege's letter (spymaster.22 - 25).

		}

}

# A lord sends men after the spymaster (spymaster.52, 53, 62).
# Scope: the lord; the spymaster is the event target spy_agent. Half the
# time the spymaster gets away, half the time not.
spymaster_strike_back_effect = {
	random_list = {
		50 = {
			event_target:spy_agent = {
				character_event = { id = spymaster.54 days = 1 tooltip = EVTTOOLTIPspymaster54 }
			}
		}
		50 = {
			modifier = { factor = 0.67 event_target:spy_agent = { martial = 10 } }
			event_target:spy_agent = {
				character_event = { id = spymaster.55 days = 1 tooltip = EVTTOOLTIPspymaster55 }
			}
		}
	}
}
