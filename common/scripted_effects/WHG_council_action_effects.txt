# Scope: the chancellor (ROOT). A kingdom first (become-king ambition only,
# 2% at diplomacy below 15, 5% at 15, 10% at 18, 20% at 22), then a duchy
# (diplomacy 15: 10%, 18: 20%, 20: 30%), else the county.
chancellor_forge_claim_effect = {
	if = {
		limit = {
			liege = {
				has_ambition = obj_become_king
				NOT = { has_character_flag = kingdom_claimed_with_ambition }
				ROOT = {
					location = {
						kingdom = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { NOT = { character = ROOT } owns = PREVPREV }
						}
					}
				}
			}
		}
		random = {
			chance = 2
			modifier = { factor = 2.5 diplomacy = 15 }
			modifier = { factor = 2 diplomacy = 18 }
			modifier = { factor = 2 diplomacy = 22 }
			location = {
				kingdom = {
					save_event_target_as = target_title
					holder_scope = { character_event = { id = chancellor.11 days = 1 } }
				}
			}
			liege = { set_character_flag = kingdom_claimed_with_ambition }
			set_character_flag = chancellor_claim_done
		}
	}
	if = {
		limit = {
			NOT = { has_character_flag = chancellor_claim_done }
			diplomacy = 15
			liege = {
				ROOT = {
					location = {
						duchy = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = {
								NOR = { is_liege_of = ROOT character = ROOT }
								owns = PREVPREV
							}
						}
					}
				}
			}
		}
		random = {
			chance = 10
			modifier = { factor = 2 diplomacy = 18 }
			modifier = { factor = 1.5 diplomacy = 20 }
			location = {
				duchy = {
					save_event_target_as = target_title
					holder_scope = { character_event = { id = chancellor.11 days = 1 } }
				}
			}
			set_character_flag = chancellor_claim_done
		}
	}
	if = {
		limit = {
			NOT = { has_character_flag = chancellor_claim_done }
			liege = {
				ROOT = {
					location = {
						county = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = {
								NOR = { is_liege_of = ROOT character = ROOT }
								owns = PREVPREV
							}
						}
					}
				}
			}
		}
		location = {
			county = {
				save_event_target_as = target_title
				holder_scope = { character_event = { id = chancellor.11 days = 1 } }
			}
		}
	}
	clr_character_flag = chancellor_claim_done
}

# A lord gives in to the liege's letter (spymaster.22 - 25).
# Scope: the lord; FROM is the liege. Leaves the faction (flag
# faction_member) or joins the liege's (flag faction_recruit).
spymaster_faction_yield_effect = {
	if = {
		limit = { has_character_flag = faction_member }
		custom_tooltip = {
			text = EVTTOOLTIPspymaster26
			hidden_effect = { FROM = { letter_event = { id = spymaster.26 days = 7 } } }
		}
	}
	else = {
		custom_tooltip = {
			text = EVTTOOLTIPspymaster35
			hidden_effect = { FROM = { letter_event = { id = spymaster.35 days = 7 } } }
		}
	}
}

# A lord sends men after the spymaster (spymaster.52, 53, 62).
# Scope: the lord; the spymaster is the event target spy_agent. Half the
# time the spymaster gets away, half the time not.
spymaster_strike_back_effect = {
	random_list = {
		50 = {
			event_target:spy_agent = {
				character_event = { id = spymaster.54 days = 1 tooltip = EVTTOOLTIPspymaster54 }
			}
		}
		50 = {
			modifier = { factor = 0.67 event_target:spy_agent = { martial = 10 } }
			event_target:spy_agent = {
				character_event = { id = spymaster.55 days = 1 tooltip = EVTTOOLTIPspymaster55 }
			}
		}
	}
}

# A new soldier of the liege's people comes to court, martial 10, stripped
# of the traits that do not fit a commander. Saved as found_commander.
# Scope: the liege. Women only where the liege's faith or law allows them
# to command, and never for peoples without women.
marshal_new_commander_effect = {
	if = {
		limit = { is_no_females_race = no gender_equality_trigger = yes }
		create_random_soldier = {
			random_traits = yes
			dynasty = none
			female = 50
			attributes = { martial = 10 }
		}
	}
	else_if = {
		limit = { is_no_females_race = no has_religion_feature = religion_matriarchal }
		create_random_soldier = {
			random_traits = yes
			dynasty = none
			female = 90
			attributes = { martial = 10 }
		}
	}
	else = {
		create_random_soldier = {
			random_traits = yes
			dynasty = none
			female = no
			attributes = { martial = 10 }
		}
	}
	new_character = {
		character_event = { id = aiclean.31 } # racial trait
		remove_trait = weak
		remove_trait = slow
		remove_trait = imbecile
		remove_trait = dull
		remove_trait = craven
		if = {
			limit = { trait = misguided_warrior }
			remove_trait = misguided_warrior
			add_trait = tough_soldier
		}
		remove_trait = mystic
		remove_trait = socializer
		remove_trait = gamer
		remove_trait = hedonist
		remove_trait = duelist
		remove_trait = hunter
		remove_trait = strategist
		remove_trait = administrator
		remove_trait = architect
		remove_trait = gardener
		remove_trait = schemer
		remove_trait = impaler
		remove_trait = seducer
		remove_trait = seductress
		remove_trait = theologian
		remove_trait = scholar
		random_list = {
			15 = { change_martial = 2 }
			10 = { add_trait = robust }
			10 = { add_trait = duelist }
			10 = { add_trait = hunter }
			5 = { add_trait = strategist }
			5 = { change_martial = 4 }
			5 = { if = { limit = { has_dlc = "Reapers" } add_trait = one_eyed change_martial = 1 } }
			5 = { add_trait = shrewd }
		}
		save_event_target_as = found_commander
	}
}

# The accused is marked: a known devil worshipper if there was suspicion,
# else an accused apostate. A follower of the liege's Old World cult who is
# not its head is cast out of it. Scope: the accused; ROOT is the liege.
spiritual_apostate_mark_effect = {
	if = {
		limit = {
			is_devil_worshiper_trigger = yes
			OR = {
				has_character_modifier = under_suspicion
				has_character_modifier = highly_suspect
			}
		}
		add_character_modifier = { name = known_satanist duration = -1 }
	}
	else = { add_character_modifier = { name = accused_apostate duration = -1 } }
	if = {
		limit = {
			religion = ROOT
			controls_religion = no
			OR = { religion = sigmarite religion = myrmidian }
		}
		add_trait = excommunicated
	}
}

# Players of the accused's society who are of the realm, kin or friends
# hear of it (spiritual.107). Scope: the accused.
spiritual_apostate_tell_brethren_effect = {
	any_player = {
		limit = {
			OR = { same_realm = ROOT is_close_relative = ROOT is_friend = ROOT }
			same_society_as = ROOT
			NOT = { character = ROOT }
		}
		character_event = { id = spiritual.107 }
	}
}

# The accused dies: burned at the stake, or put to the sword in Araby,
# which holds no burnings. Scope: the accused; FROM is the liege.
spiritual_apostate_execute_effect = {
	hidden_effect = { spiritual_apostate_tell_brethren_effect = yes }
	if = {
		limit = { FROM = { council_may_burn_trigger = yes } }
		add_trait = staked
		death = { death_reason = death_execution_burning killer = FROM }
	}
	else = { death = { death_reason = death_execution killer = FROM } }
}
