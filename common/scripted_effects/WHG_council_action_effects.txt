# Scope: the chancellor (ROOT). A kingdom first (become-king ambition only,
# 2% at diplomacy below 15, 5% at 15, 10% at 18, 20% at 22), then a duchy
# (diplomacy 15: 10%, 18: 20%, 20: 30%), else the county.
chancellor_forge_claim_effect = {
	if = {
		limit = {
			liege = {
				has_ambition = obj_become_king
				NOT = { has_character_flag = kingdom_claimed_with_ambition }
				ROOT = {
					location = {
						kingdom = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { NOT = { character = ROOT } owns = PREVPREV }
						}
					}
				}
			}
		}
		random = {
			chance = 2
			modifier = { factor = 2.5 diplomacy = 15 }
			modifier = { factor = 2 diplomacy = 18 }
			modifier = { factor = 2 diplomacy = 22 }
			location = {
				kingdom = {
					save_event_target_as = target_title
					holder_scope = { character_event = { id = chancellor.11 days = 1 } }
				}
			}
			liege = { set_character_flag = kingdom_claimed_with_ambition }
			set_character_flag = chancellor_claim_done
		}
	}
	if = {
		limit = {
			NOT = { has_character_flag = chancellor_claim_done }
			diplomacy = 15
			liege = {
				ROOT = {
					location = {
						duchy = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { NOR = { is_liege_of = ROOT character = ROOT } owns = PREVPREV }
						}
					}
				}
			}
		}
		random = {
			chance = 10
			modifier = { factor = 2 diplomacy = 18 }
			modifier = { factor = 1.5 diplomacy = 20 }
			location = {
				duchy = {
					save_event_target_as = target_title
					holder_scope = { character_event = { id = chancellor.11 days = 1 } }
				}
			}
			set_character_flag = chancellor_claim_done
		}
	}
	if = {
		limit = {
			NOT = { has_character_flag = chancellor_claim_done }
			liege = {
				ROOT = {
					location = {
						county = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { NOR = { is_liege_of = ROOT character = ROOT } owns = PREVPREV }
						}
					}
				}
			}
		}
		location = {
			county = {
				save_event_target_as = target_title
				holder_scope = { character_event = { id = chancellor.11 days = 1 } }
			}
		}
	}
	clr_character_flag = chancellor_claim_done
}
