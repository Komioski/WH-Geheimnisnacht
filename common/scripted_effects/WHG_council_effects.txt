# Clears the traits that do not fit a newly made councillor; the event then
# adds one for her office.
# Scope: the new woman.
council_talent_clear_traits_effect = {
	remove_trait = content
	remove_trait = slow
	remove_trait = imbecile
	remove_trait = dull
	remove_trait = socializer
	remove_trait = gamer
	remove_trait = hedonist
	remove_trait = duelist
	remove_trait = hunter
	remove_trait = strategist
	remove_trait = administrator
	remove_trait = architect
	remove_trait = gardener
	remove_trait = schemer
	remove_trait = impaler
	remove_trait = seductress
	remove_trait = scholar
	remove_trait = theologian
	remove_trait = mystic
}

# The woman is glad of her seat, and the court may be scandalised, the more
# so where the law keeps women from that seat (Conclave). Without Conclave
# the scandal is always the worst.
# Scope: the ruler. FROM: the woman, now in her seat.
council_talent_honour_effect = {
	if = {
		limit = { FROM = { has_job_title = job_spymaster } }
		if = {
			limit = { FROM = { is_close_relative = ROOT } }
			reverse_opinion = { modifier = opinion_given_council_position_2 who = FROM years = 5 }
		}
		else = {
			reverse_opinion = { modifier = opinion_given_council_position_1 who = FROM years = 5 }
		}
	}
	else_if = {
		limit = { NOT = { has_dlc = "Conclave" } }
		female_councilor_effect_3 = yes
	}
	else_if = {
		limit = { FROM = { has_job_title = job_marshal } }
		if = {
			limit = { primary_title = { has_law = status_of_women_2 } }
			female_councilor_effect_1 = yes
		}
		else_if = {
			limit = { primary_title = { has_law = status_of_women_1 } }
			female_councilor_effect_2 = yes
		}
		else_if = {
			limit = { primary_title = { has_law = status_of_women_0 } }
			female_councilor_effect_3 = yes
		}
	}
	else = { # Chancellor or steward
		if = {
			limit = { primary_title = { has_law = status_of_women_1 } }
			female_councilor_effect_1 = yes
		}
		else_if = {
			limit = { primary_title = { has_law = status_of_women_0 } }
			female_councilor_effect_2 = yes
		}
	}
}

# The stranger is wed to an unimportant man at court, or to a new man of
# her office's kind, so that she stays.
# Scope: the ruler. FROM: the woman, now in her seat.
council_talent_marry_effect = {
	if = {
		limit = { any_courtier = { council_talent_husband_trigger = yes } }
		random_courtier = {
			limit = { council_talent_husband_trigger = yes }
			add_spouse_matrilineal = FROM
		}
	}
	else_if = {
		limit = { FROM = { has_job_title = job_marshal } }
		create_random_soldier = { random_traits = yes dynasty = NONE female = no age = 37 }
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			add_spouse_matrilineal = FROM
		}
	}
	else_if = {
		limit = { FROM = { has_job_title = job_treasurer } }
		create_random_steward = { random_traits = yes dynasty = NONE female = no age = 42 }
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			add_spouse_matrilineal = FROM
		}
	}
	else_if = {
		limit = { FROM = { has_job_title = job_spymaster } }
		create_random_intriguer = { random_traits = yes dynasty = NONE female = no age = 29 }
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			add_spouse_matrilineal = FROM
		}
	}
	else = {
		create_random_diplomat = { random_traits = yes dynasty = NONE female = no age = 37 }
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			add_spouse_matrilineal = FROM
		}
	}
}

# The capital prospers when a councillor's work pays off (Reapers prosperity).
# Scope: the ruler.
council_capital_prospers_effect = {
	if = {
		limit = { has_dlc = "Reapers" }
		custom_tooltip = {
			text = capital_prospers_custom_tooltip
			hidden_tooltip = {
				capital_scope = {
					change_variable = { which = prosperity_value value = 6 }
				}
			}
		}
	}
}

# A province prospers (Reapers prosperity). Scope: the province.
council_province_prospers_effect = {
	if = {
		limit = { has_dlc = "Reapers" }
		custom_tooltip = {
			text = increase_prosperity_custom_tooltip
			hidden_tooltip = {
				change_variable = { which = prosperity_value value = 6 }
			}
		}
	}
}
