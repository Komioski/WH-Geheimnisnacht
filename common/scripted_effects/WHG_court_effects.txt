# Scope: the chancellor (ROOT). A kingdom first (become-king ambition only,
# 2% at diplomacy below 15, 5% at 15, 10% at 18, 20% at 22), then a duchy
# (diplomacy 15: 10%, 18: 20%, 20: 30%), else the county.
# Used by: WHG_council_action_events
chancellor_forge_claim_effect = {
	if = {
		limit = {
			liege = {
				has_ambition = obj_become_king
				NOT = { has_character_flag = kingdom_claimed_with_ambition }
				ROOT = {
					location = {
						kingdom = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { NOT = { character = ROOT } owns = PREVPREV }
						}
					}
				}
			}
		}
		random = {
			chance = 2
			modifier = { factor = 2.5 diplomacy = 15 }
			modifier = { factor = 2 diplomacy = 18 }
			modifier = { factor = 2 diplomacy = 22 }
			location = {
				kingdom = {
					save_event_target_as = target_title
					holder_scope = { character_event = { id = chancellor.11 days = 1 } }
				}
			}
			liege = { set_character_flag = kingdom_claimed_with_ambition }
			set_character_flag = chancellor_claim_done
		}
	}
	if = {
		limit = {
			NOT = { has_character_flag = chancellor_claim_done }
			diplomacy = 15
			liege = {
				ROOT = {
					location = {
						duchy = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = {
								NOR = { is_liege_of = ROOT character = ROOT }
								owns = PREVPREV
							}
						}
					}
				}
			}
		}
		random = {
			chance = 10
			modifier = { factor = 2 diplomacy = 18 }
			modifier = { factor = 1.5 diplomacy = 20 }
			location = {
				duchy = {
					save_event_target_as = target_title
					holder_scope = { character_event = { id = chancellor.11 days = 1 } }
				}
			}
			set_character_flag = chancellor_claim_done
		}
	}
	if = {
		limit = {
			NOT = { has_character_flag = chancellor_claim_done }
			liege = {
				ROOT = {
					location = {
						county = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = {
								NOR = { is_liege_of = ROOT character = ROOT }
								owns = PREVPREV
							}
						}
					}
				}
			}
		}
		location = {
			county = {
				save_event_target_as = target_title
				holder_scope = { character_event = { id = chancellor.11 days = 1 } }
			}
		}
	}
	clr_character_flag = chancellor_claim_done
}

# A new soldier of the liege's people comes to court, martial 10, stripped
# of the traits that do not fit a commander. Saved as found_commander.
# Scope: the liege. Women only where the liege's faith or law allows them
# to command, and never for peoples without women.
# Used by: WHG_council_action_events
marshal_new_commander_effect = {
	if = {
		limit = { is_no_females_race = no gender_equality_trigger = yes }
		create_random_soldier = {
			random_traits = yes
			dynasty = none
			female = 50
			attributes = { martial = 10 }
		}
	}
	else_if = {
		limit = { is_no_females_race = no has_religion_feature = religion_matriarchal }
		create_random_soldier = {
			random_traits = yes
			dynasty = none
			female = 90
			attributes = { martial = 10 }
		}
	}
	else = {
		create_random_soldier = {
			random_traits = yes
			dynasty = none
			female = no
			attributes = { martial = 10 }
		}
	}
	new_character = {
		character_event = { id = aiclean.31 } # racial trait
		remove_trait = weak
		remove_trait = slow
		remove_trait = imbecile
		remove_trait = dull
		remove_trait = craven
		if = {
			limit = { trait = misguided_warrior }
			remove_trait = misguided_warrior
			add_trait = tough_soldier
		}
		remove_trait = mystic
		remove_trait = socializer
		remove_trait = gamer
		remove_trait = hedonist
		remove_trait = duelist
		remove_trait = hunter
		remove_trait = strategist
		remove_trait = administrator
		remove_trait = architect
		remove_trait = gardener
		remove_trait = schemer
		remove_trait = impaler
		remove_trait = seducer
		remove_trait = seductress
		remove_trait = theologian
		remove_trait = scholar
		random_list = {
			15 = { change_martial = 2 }
			10 = { add_trait = robust }
			10 = { add_trait = duelist }
			10 = { add_trait = hunter }
			5 = { add_trait = strategist }
			5 = { change_martial = 4 }
			5 = { if = { limit = { has_dlc = "Reapers" } add_trait = one_eyed change_martial = 1 } }
			5 = { add_trait = shrewd }
		}
		save_event_target_as = found_commander
	}
}

				has_character_modifier = highly_suspect
			}
		}
		add_character_modifier = { name = known_satanist duration = -1 }
	}
	else = { add_character_modifier = { name = accused_apostate duration = -1 } }

			controls_religion = no
			OR = { religion = sigmarite religion = myrmidian }
		}
		add_trait = excommunicated
	}
}

# The woman is glad of her seat, and the court may be scandalised, the more
# so where the law keeps women from that seat (Conclave). Without Conclave
# the scandal is always the worst.
# Scope: the ruler. FROM: the woman, now in her seat.
# Used by: WHG_council_events
council_talent_honour_effect = {
	if = {
		limit = { FROM = { has_job_title = job_spymaster } }
		if = {
			limit = { FROM = { is_close_relative = ROOT } }
			reverse_opinion = { modifier = opinion_given_council_position_2 who = FROM years = 5 }
		}
		else = {
			reverse_opinion = { modifier = opinion_given_council_position_1 who = FROM years = 5 }
		}
	}
	else_if = {
		limit = { NOT = { has_dlc = "Conclave" } }
		female_councilor_effect_3 = yes
	}
	else_if = {
		limit = { FROM = { has_job_title = job_marshal } }
		if = {
			limit = { primary_title = { has_law = status_of_women_2 } }
			female_councilor_effect_1 = yes
		}
		else_if = {
			limit = { primary_title = { has_law = status_of_women_1 } }
			female_councilor_effect_2 = yes
		}
		else_if = {
			limit = { primary_title = { has_law = status_of_women_0 } }
			female_councilor_effect_3 = yes
		}
	}
	else = { # Chancellor or steward
		if = {
			limit = { primary_title = { has_law = status_of_women_1 } }
			female_councilor_effect_1 = yes
		}
		else_if = {
			limit = { primary_title = { has_law = status_of_women_0 } }
			female_councilor_effect_2 = yes
		}
	}
}

# Scope: the host whose cup or dish is poisoned. ROOT is the Cupbearer or
# Food Taster who did it (taster.1, taster.10). A paranoid host is more
# likely to spot it. Depending on the outcome the host gets one of the flags
# court_poison_failed, court_poison_ill or court_poison_mad, which taster.2
# reads; a host who is only sickened or driven mad finds out only with good
# intrigue or a sharp spymaster.
# Used by: WHG_court_office_events
court_poison_host_effect = {
	random_list = {
		50 = {
			modifier = {
				factor = 1.4
				trait = paranoid
			}
			set_character_flag = court_poison_failed
			character_event = { id = taster.2 tooltip = EVTTOOLTIPtaster1_failed }
		}
		25 = {
			modifier = {
				factor = 0.6
				trait = paranoid
			}
			if = {
				limit = { is_disease_immune_race = no }
				if = {
					limit = { has_dlc = "Reapers" }
					set_character_flag = disease_not_serious
					add_trait = vomiting
					character_event = { id = treatment.6 days = 10 random = 30 }
				}
				else = {
					add_trait = ill
				}
				hidden_effect = { character_event = { id = traitnote.290 } } # Notify Ill
			}
			if = {
				limit = { court_poison_noticed_trigger = yes }
				set_character_flag = court_poison_ill
				character_event = { id = taster.2 days = 3 random = 3 }
			}
		}
		20 = {
			modifier = {
				factor = 0.5
				trait = paranoid
			}
			if = {
				limit = { NOT = { trait = lunatic_1 } }
				add_trait = lunatic_1
				set_variable = { which = "lunatic_counter" value = 5 }
				hidden_effect = { character_event = { id = traitnote.304 } } # Notify Lunatic
			}
			if = {
				limit = { court_poison_noticed_trigger = yes }
				set_character_flag = court_poison_mad
				character_event = { id = taster.2 days = 3 random = 3 }
			}
		}
		5 = {
			death = {
				death_reason = death_murder_unknown_poison
				killer = ROOT
			}
		}
	}
}
