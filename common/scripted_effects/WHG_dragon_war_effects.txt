# Scope: the governor. Saves spawn_province, where the Dragon's host
# arrives. The roads east end far from the western protectorate: Skrapland,
# the nearest, is some 900 from Taharq, across the Dark Lands. So the host
# comes by sea, in the war junks of the Sea Dragon's fleet out of Fu-Chow,
# and lands at a port of the governor's realm (the one nearest the enemy's
# capital, event_target:cathay_war_front). It marches down a road only when
# the enemy's capital lies within 700 of a road end, or the governor holds
# no port. Sets the character flag dragonwar_host_by_sea for the texts.
dragonwar_pick_spawn_effect = {
	clr_character_flag = dragonwar_host_by_sea
	if = {
		limit = {
			NOT = { event_target:cathay_war_front = { dragonwar_near_road_end_trigger = yes } }
			any_realm_province = { port = yes }
		}
		set_character_flag = dragonwar_host_by_sea
		random_realm_province = {
			limit = { port = yes }
			preferred_limit = { distance = { where = event_target:cathay_war_front distance < 150 } }
			preferred_limit = { distance = { where = event_target:cathay_war_front distance < 300 } }
			preferred_limit = { distance = { where = event_target:cathay_war_front distance < 600 } }
			save_event_target_as = spawn_province
		}
	}
	else = {
		dragonwar_pick_road_effect = yes
	}
}

# Scope: any. Saves spawn_province: the end of the road east that

# Scope: the governor or the enemy (dragonwar.11, 12). Sends the war's end
# (dragonwar.20 if Cathay lost, 21 if it won) to players who follow Cathay's
# news and are in range of the governor.
dragonwar_news_effect = {
	if = {
		limit = {
			offmap_cathay = {
				OR = {
					has_offmap_tmp_flag = cathay_lost_offensive_war
					has_offmap_tmp_flag = cathay_lost_defensive_war
				}
			}
		}
		any_player = {
			limit = {
				has_offmap_news_enabled = offmap_cathay
				is_within_diplo_range = event_target:the_protector_general
			}
			narrative_event = { id = dragonwar.20 days = 1 }
		}
	}
	else = {
		any_player = {
			limit = {
				has_offmap_news_enabled = offmap_cathay
				is_within_diplo_range = event_target:the_protector_general
			}
			narrative_event = { id = dragonwar.21 days = 1 }
		}
	}
}
