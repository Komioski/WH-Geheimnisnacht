# Used by: WHG_ancestor_offering_events, WHG_black_crusade_events +1 more
increase_prosperity_effect = {
	if = {
		limit = {
			NOR = {
				has_province_modifier = prosperity_modifier_1
				has_province_modifier = prosperity_modifier_2
				has_province_modifier = prosperity_modifier_3
				has_province_modifier = depopulated_1
				has_province_modifier = depopulated_2
				has_province_modifier = depopulated_3
			}
		}

		add_province_modifier = { name = prosperity_modifier_1 duration = -1 }
		hidden_tooltip = { set_variable = { which = prosperity_value value = 100 } }
		break = yes
	}

	if = {
		limit = { has_province_modifier = prosperity_modifier_1 }

		hidden_tooltip = { remove_province_modifier = prosperity_modifier_1 }
		add_province_modifier = { name = prosperity_modifier_2 duration = -1 }
		hidden_tooltip = { set_variable = { which = prosperity_value value = 250 } }
		break = yes
	}

	if = {
		limit = { has_province_modifier = prosperity_modifier_2 }

		hidden_tooltip = { remove_province_modifier = prosperity_modifier_2 }
		add_province_modifier = { name = prosperity_modifier_3 duration = -1 }
		hidden_tooltip = { set_variable = { which = prosperity_value value = 500 } }
		break = yes
	}
}

# Adds one level of slave factory to the capital holding (up to level 4).
# Scope: a ruler.
# Used by: WHG_dwarf_effects, WHG_chaos_dwarf_events, WHG_slavery_decisions

	capital_holding = {
		if = {
			limit = { holding_type = castle }

# Drags up to five courtiers of a raided court off in chains. Each has a 1 in 4
# chance to slip away.
# Scope: the raided ruler. ROOT is the slaver.
# Used by: WHG_chaos_dwarf_events
dawizharr_take_captives_effect = {
	while = {
		count = 5
		random_courtier = {
			limit = { dawizharr_captive_trigger = yes }
			random_list = {
				75 = { imprison = ROOT move_character = ROOT }
				25 = { set_character_flag = dawizharr_hid }
			}
		}
	}
	any_courtier = {
		limit = { has_character_flag = dawizharr_hid }
		clr_character_flag = dawizharr_hid
	}
}

# Throws up to five courtiers of a raided court into Hashut's furnaces.
# Scope: the raided ruler. ROOT is the slaver.
# Used by: WHG_chaos_dwarf_events
dawizharr_burn_captives_effect = {
	while = {
		count = 5
		random_courtier = {
			limit = { dawizharr_captive_trigger = yes }
			random_list = {
				75 = {
					if = {
						limit = { is_invincible_trigger = yes }
						antideath_effect = yes
						set_character_flag = dawizharr_hid
					}
					else = { death = { death_reason = death_sacrificed killer = ROOT } }
				}
				25 = { set_character_flag = dawizharr_hid }
			}
		}
	}
	any_courtier = {
		limit = { has_character_flag = dawizharr_hid }
		clr_character_flag = dawizharr_hid
	}
}

# The furnaces have been fed: lifts Hashut's demand and restarts the four-year
# clock (dawizharr_sacrifice_due_trigger).
# Scope: a ruler.
# Used by: WHG_chaos_dwarf_events
dawizharr_sacrifice_made_effect = {
	if = {
		limit = { has_character_modifier = must_sacrifice_hashut }
		remove_character_modifier = must_sacrifice_hashut
	}
	clr_character_flag = flag_sacrifice_made_hashut
	set_character_flag = flag_sacrifice_made_hashut
}
