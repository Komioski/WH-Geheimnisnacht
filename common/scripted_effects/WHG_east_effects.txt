# Used by: WHG_nomad_events, 00_cb_types, WHG_revolt_events
set_appropriate_nomad_gov = {
	if = {
		limit = {
			religion_group = chaos_gods_group
			NOT = { culture_group = beastman_group }
		}
		set_government_type = chaos_horde_government
	}
	else_if = {
		limit = {
			culture_group = beastman_group
		}
		set_government_type = beastman_government
	}
	else = {
		set_government_type = nomadic_government
	}
}

	add_character_modifier = {
		name = invigorated
		years = 1
	}

		limit = { trait = severely_injured }
		remove_trait = severely_injured
	}
	if = { #Most Pressing Wounds first
		limit = { trait = maimed }
		remove_trait = maimed
		break = yes
	}
	else_if = { #Harmful for Nurglites
		limit = {
			true_religion = nurgle
			trait = food_poisoning
		}
		remove_trait = food_poisoning
	}
	else_if = { #Incureables
		limit = { trait = has_bubonic_plague }

# Turns a fresh Cathayan into a trained expert: skills, education, gifts,

# is_cathayan_master_engineer, is_cathayan_strategist or is_cathayan_doctor.

# Cathay's might over the years, a wave from the start of the game: at its
# height in 2010, falling to a low ebb around 2070 - 2089, rising again from
# about 2100, then declining from 2200 to the standard baseline in 2250. Adds
# regiments (of about 500), the share of the enemy's levies matched, and the
# troop quality. Used by dragonwar.3 and the invasion of Cathay.
# Used by: WHG_cathay_invasion_events
cathay_war_modify_variables_by_century_effect = {
	if = { # 2010 - 2029: the height of the Dragon's might
		limit = { NOT = { year = 2030 } }
		change_variable = { which = cathay_war_troop_quantity value = 8 }
		change_variable = { which = cathay_war_target_match_mult value = 0.05 }
		change_variable = { which = cathay_war_troop_quality value = 1 }
	}
	else_if = { # 2030 - 2049: the first decline
		limit = { NOT = { year = 2050 } }
		change_variable = { which = cathay_war_troop_quantity value = 5 }
		change_variable = { which = cathay_war_target_match_mult value = 0.05 }
	}
	else_if = { # 2050 - 2069
		limit = { NOT = { year = 2070 } }
		change_variable = { which = cathay_war_troop_quantity value = 3 }
	}
	else_if = { # 2070 - 2089: the low ebb
		limit = { NOT = { year = 2090 } }
		# no change
	}
	else_if = { # 2090 - 2109: Cathay rises again
		limit = { NOT = { year = 2110 } }
		change_variable = { which = cathay_war_troop_quantity value = 4 }
	}
	else_if = { # 2110 - 2149: the second height
		limit = { NOT = { year = 2150 } }
		change_variable = { which = cathay_war_troop_quantity value = 7 }
		change_variable = { which = cathay_war_target_match_mult value = 0.05 }
		change_variable = { which = cathay_war_troop_quality value = 1 }
	}
	else_if = { # 2150 - 2199
		limit = { NOT = { year = 2200 } }
		change_variable = { which = cathay_war_troop_quantity value = 5 }
		change_variable = { which = cathay_war_target_match_mult value = 0.05 }
	}
	else_if = { # 2200 - 2249: the slow decline
		limit = { NOT = { year = 2250 } }
		change_variable = { which = cathay_war_troop_quantity value = 2 }
	}
	# 2250 on: the baseline, no change
}

# Gives a displaced prince a parent of the fallen house: the last Emperor,
# his father, another Emperor or Empress of the house, or any man of it.
# Scope: the prince; event_target:previousEmperor is the last Emperor.
# Used by: WHG_cathay_invasion_events
cathayinv_prince_parent_effect = {
	if = {
		limit = {
			event_target:previousEmperor = { could_be_parent_of = event_target:displaced_prince }
		}
		set_father = event_target:previousEmperor
	}
	else_if = {
		limit = {
			event_target:previousEmperor = {
				father_even_if_dead = { could_be_parent_of = event_target:displaced_prince }
			}
		}
		event_target:previousEmperor = {
			father_even_if_dead = { event_target:displaced_prince = { set_father = PREV } }
		}
	}
	else_if = {
		limit = {
			event_target:previousEmperor = {
				any_dynasty_member_even_if_dead = {
					is_or_was_offmap_power_ruler = yes
					is_female = no
					could_be_parent_of = event_target:displaced_prince
				}
			}
		}
		event_target:previousEmperor = {
			random_dynasty_member_even_if_dead = {
				limit = {
					is_or_was_offmap_power_ruler = yes
					is_female = no
					could_be_parent_of = event_target:displaced_prince
				}
				event_target:displaced_prince = { set_father = PREV }
			}
		}
	}
	else_if = {
		limit = {
			event_target:previousEmperor = {
				any_dynasty_member_even_if_dead = {
					is_or_was_offmap_power_ruler = yes
					is_woman_trigger = yes
					could_be_parent_of = event_target:displaced_prince
				}
			}
		}
		event_target:previousEmperor = {
			random_dynasty_member_even_if_dead = {
				limit = {
					is_or_was_offmap_power_ruler = yes
					is_woman_trigger = yes
					could_be_parent_of = event_target:displaced_prince
				}
				event_target:displaced_prince = { set_mother = PREV }
			}
		}
	}
	else_if = {
		limit = {
			event_target:previousEmperor = {
				any_dynasty_member_even_if_dead = {
					is_female = no
					could_be_parent_of = event_target:displaced_prince
				}
			}
		}
		event_target:previousEmperor = {
			random_dynasty_member_even_if_dead = {
				limit = { is_female = no could_be_parent_of = event_target:displaced_prince }
				event_target:displaced_prince = { set_father = PREV }
			}
		}
	}
}

# Picks this year's outcome and sets its flag. A Cathay in famine, plague,
# civil war or unrest is easy prey; a stable or golden Cathay guards its
# border. Treasures come once per lifetime, pandas once.
# Scope: the raider.
# Used by: WHG_cathay_invasion_events
cathayraid_pick_outcome_effect = {
	random_list = {
		400 = { set_character_flag = cathayraid_loot }
		30 = {
			modifier = {
				factor = 1.5
				offmap_cathay = {
					OR = {
						has_status = cathay_famine
						has_status = cathay_plague
						has_status = cathay_civil_war
						has_status = cathay_unrest
					}
				}
			}
			modifier = { factor = 0.5 offmap_cathay = { has_status = cathay_golden_age } }
			set_character_flag = cathayraid_rich
		}
		5 = {
			modifier = {
				factor = 1.5
				offmap_cathay = {
					OR = {
						has_status = cathay_famine
						has_status = cathay_plague
						has_status = cathay_civil_war
						has_status = cathay_unrest
					}
				}
			}
			modifier = { factor = 0.5 offmap_cathay = { has_status = cathay_golden_age } }
			set_character_flag = cathayraid_ransom
		}
		30 = {
			modifier = {
				factor = 2
				offmap_cathay = {
					OR = { has_status = cathay_golden_age has_status = cathay_stable }
				}
			}
			modifier = {
				factor = 0.5
				offmap_cathay = {
					OR = {
						has_status = cathay_famine
						has_status = cathay_plague
						has_status = cathay_civil_war
						has_status = cathay_unrest
					}
				}
			}
			set_character_flag = cathayraid_bad

enhanced_beauty_effect = {
	if = {
		limit = {
			OR = {
				trait = plain
				trait = ugly
				trait = unsightly
				trait = disfigured
				trait = uncouth
				trait = clubfooted
				trait = harelip
				trait = hunchback
				trait = lisp
				trait = stutter
			}
		}
		hidden_tooltip = {
			remove_trait = plain
			remove_trait = ugly
			remove_trait = unsightly
			remove_trait = disfigured
			remove_trait = uncouth
			remove_trait = clubfooted
			remove_trait = harelip
			remove_trait = hunchback
			remove_trait = lisp
			remove_trait = stutter
		}
		add_trait = fair
		add_trait = groomed
	}
	else_if = {
		limit = {
			NOR = {
				trait = beautiful
				trait = pretty
				trait = fair
				trait = groomed
			}
		}
		add_trait = fair
		add_trait = groomed
	}
	else_if = {
		limit = { trait = fair }
		remove_trait = fair
		add_trait = pretty
	}
	else_if = {
		limit = { trait = pretty }
		remove_trait = pretty
		add_trait = beautiful
	}
}

	}

			NOT = { culture_group = beastman_group }

		add_offmap_currency = { offmap = slaanesh_warp value = 600 }
