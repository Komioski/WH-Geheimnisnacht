# The Wild Hunt runs down one more courtier of a scourged holding (three in
# four die). Call it several times; asrai_wild_hunt_cleanup_effect clears
# the marks after.
# Scope: the holder of the scourged holding. ROOT: the hunt's leader.
# Used by: WHG_asrai_events
asrai_wild_hunt_slay_effect = {
	random_courtier = {
		limit = {
			is_abroad = no
			NOT = { trait = creature_elf }
			NOT = { has_character_flag = asrai_hunted }
		}
		set_character_flag = asrai_hunted
		random = {
			chance = 75
			if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
			else = { death = { death_reason = death_murder killer = ROOT } }
		}
	}
}

# The Wild Hunt carries off one more courtier of a scourged holding into the
# mists (three in four are taken). Call it several times;
# asrai_wild_hunt_cleanup_effect clears the marks after.
# Scope: the holder of the scourged holding. ROOT: the hunt's leader.
# Used by: WHG_asrai_events
asrai_wild_hunt_take_effect = {
	random_courtier = {
		limit = {
			is_abroad = no
			is_ruler = no
			NOT = { trait = creature_elf }
			NOT = { trait = creature_forestspirit }
			NOT = { trait = creature_treekin }
			NOT = { has_character_flag = asrai_hunted }
		}
		set_character_flag = asrai_hunted
		random = { chance = 75 imprison = ROOT move_character = ROOT }
	}
}

# Scope: the holder of the scourged holding.
# Used by: WHG_asrai_events
asrai_wild_hunt_cleanup_effect = {
	any_courtier = {
		limit = { has_character_flag = asrai_hunted }
		clr_character_flag = asrai_hunted
	}
	ROOT = {
		any_courtier = {
			limit = { has_character_flag = asrai_hunted }
			clr_character_flag = asrai_hunted
		}
		any_prisoner = {
			limit = { has_character_flag = asrai_hunted }
			clr_character_flag = asrai_hunted
		}
	}
}

# The guest grants what the host asked for at a private meeting: gold, a
# favour, the release of the host's debt, a divorce or a marriage.
# Scope: any. Needs the event targets asurmeet_host and asurmeet_guest.
# Used by: WHG_asur_events
asurmeet_grant_ask_effect = {
	event_target:asurmeet_host = {
		if = {
			limit = { has_character_flag = asurmeet_ask_gold_500 }
			wealth = 500
			event_target:asurmeet_guest = { wealth = -500 }
		}
		else_if = {
			limit = { has_character_flag = asurmeet_ask_gold_250 }
			wealth = 250
			event_target:asurmeet_guest = { wealth = -250 }
		}
		else_if = {
			limit = { has_character_flag = asurmeet_ask_gold_100 }
			wealth = 100
			event_target:asurmeet_guest = { wealth = -100 }
		}
		else_if = {
			limit = { has_character_flag = asurmeet_ask_favor }
			add_favor = event_target:asurmeet_guest
		}
		else_if = {
			limit = { has_character_flag = asurmeet_ask_clear }
			event_target:asurmeet_guest = { remove_favor = event_target:asurmeet_host }
		}
		else_if = {
			limit = { has_character_flag = asurmeet_ask_divorce }
			event_target:asurmeet_guest = {
				prestige = -100
				piety = -100
				spouse = {
					save_event_target_as = asurmeet_old_spouse
					remove_spouse = event_target:asurmeet_guest
				}
			}
			event_target:asurmeet_old_spouse = { character_event = { id = asurdebate.18 } }
		}
		else_if = {
			limit = { has_character_flag = asurmeet_ask_marriage }
			asurmeet_wedding_effect = yes
		}
	}
}

# The host pays the guest's price: 200 gold, a favour (or the guest's own
# debt to the host is forgiven), or marriage.
# Scope: any. Needs the event targets asurmeet_host and asurmeet_guest.
# Used by: WHG_asur_events
asurmeet_grant_counter_effect = {
	event_target:asurmeet_guest = {
		if = {
			limit = { has_character_flag = asurmeet_counter_gold }
			wealth = 200
			event_target:asurmeet_host = { wealth = -200 }
		}
		else_if = {
			limit = {
				has_character_flag = asurmeet_counter_favor
				owes_favor_to = event_target:asurmeet_host
			}
			event_target:asurmeet_host = { remove_favor = event_target:asurmeet_guest }
		}
		else_if = {
			limit = { has_character_flag = asurmeet_counter_favor }
			add_favor = event_target:asurmeet_host
		}
		else_if = {
			limit = { has_character_flag = asurmeet_counter_marriage }
			asurmeet_wedding_effect = yes
		}
	}
}

# The courtship is over, whatever the outcome.
# Scope: the suitor or the beloved.
# Used by: WHG_elven_marriage_events, WHG_elves_effects
elfwed_end_courtship_effect = {
	remove_trait = courting
	remove_character_modifier = elfwed_courtship_alive
	set_variable = { which = "courting_score" value = 0 }
	clr_character_flag = asking_for_matrilineal
	clr_character_flag = landless_fiancee_flag
	elfwed_clear_gift_effect = yes
}

# The suitor proposes. A ruler answers for themself (elfwed.9); a courtier
# of the suitor's own court is simply wed; any other courtier hears the
# proposal first (elfwed.7) and then their liege decides.
# Scope: the suitor, FROM the beloved (elfwed.2).
# Used by: WHG_elven_marriage_events
elfwed_propose_effect = {
	if = { limit = { FROM = { is_ruler = yes } } FROM = { character_event = { id = elfwed.9 } } }
	else_if = {
		limit = { FROM = { liege = { character = ROOT } } }
		elfwed_wed_effect = yes
		character_event = { id = elfwed.10 }
	}
	else = { FROM = { character_event = { id = elfwed.7 } } }
}

# The wedding. The suitor's flag says whether it is matrilineal.
# Uses the event targets elfwed_suitor and elfwed_beloved.
# Scope: anyone in the courtship chain.
# Used by: WHG_elven_marriage_events, WHG_elves_effects
elfwed_wed_effect = {
	# The courting trait forbids marriage, so it goes first
	event_target:elfwed_suitor = { remove_trait = courting }
	event_target:elfwed_beloved = { remove_trait = courting }
	event_target:elfwed_beloved = {
		if = {
			limit = { event_target:elfwed_suitor = { has_character_flag = asking_for_matrilineal } }
			add_spouse_matrilineal = event_target:elfwed_suitor
		}
		else = { add_spouse = event_target:elfwed_suitor }
	}
	event_target:elfwed_suitor = {
		elfwed_end_courtship_effect = yes
		remove_trait = elven_maiden
		remove_trait = elven_scion
		character_event = { id = elfwed.70 days = 1 } # Remember a matrilineal marriage
		elfwed_wedding_rite_effect = yes
	}
	event_target:elfwed_beloved = {
		elfwed_end_courtship_effect = yes
		remove_trait = elven_maiden
		remove_trait = elven_scion
		character_event = { id = elfwed.70 days = 1 }
		elfwed_wedding_rite_effect = yes
	}
}

# A divorce sours what was left between the two.
# Scope: the divorced spouse, FROM the one who divorced them.
# Used by: WHG_elven_marriage_events
elfwed_divorce_sour_effect = {
	if = { limit = { is_lover = FROM } random = { chance = 50 remove_lover = FROM } }
	if = { limit = { is_friend = FROM } random = { chance = 25 remove_friend = FROM } }
	if = {
		limit = { NOT = { is_friend = FROM } NOT = { is_lover = FROM } NOT = { num_of_rivals = 3 } }
		random = { chance = 15 add_rival = FROM }
	}
}

# Forget the old marriage once the divorce is settled.
# Scope: either of the two.
# Used by: WHG_elven_marriage_events
elfwed_divorce_settled_effect = {
	clr_character_flag = was_matrimarried_flag
	clr_character_flag = had_matriheritage
}

# The two of them marry. A ruling lady keeps the children in her line.
# Scope: any. Needs the event targets asurmeet_host and asurmeet_guest.
# Used by: WHG_elves_effects
asurmeet_wedding_effect = {
	event_target:asurmeet_host = {
		if = { limit = { is_female = yes } add_spouse_matrilineal = event_target:asurmeet_guest }
		else = { add_spouse = event_target:asurmeet_guest }
	}
}

# The wedding rite of the character's own people.
# Scope: a newly wed elf.
# Used by: WHG_elves_effects
elfwed_wedding_rite_effect = {
	if = {
		limit = { culture_group = asur_group }
		long_character_event = { id = elfwed.12 days = 2 }
	}
	else_if = {
		limit = { culture_group = druchii_group }
		long_character_event = { id = elfwed.13 days = 2 }
	}
	else_if = {
		limit = { culture_group = asrai_group }
		long_character_event = { id = elfwed.14 days = 2 }
	}
}
