# Used by: WHG_witch_hunter_events, WHG_chaos_cult_events +5 more
increase_visibility_minor_effect = {
	hidden_effect = {
		change_variable = { which = visibility_value value = 2 }
		if = {
			limit = {
				check_variable = { which = visibility_value value = 15 }
				NOR = {
					has_character_modifier = under_suspicion
					has_character_modifier = highly_suspect
				}
			}
			add_character_modifier = {
				name = under_suspicion
				duration = -1
			}
			character_event = { id = society.61 days = 1 }
		}
		if = {
			limit = {
				check_variable = { which = visibility_value value = 25 }
				NOT = {
					has_character_modifier = highly_suspect
				}
			}
			remove_character_modifier = under_suspicion
			add_character_modifier = {
				name = highly_suspect duration = -1
			}
			character_event = { id = society.62 days = 1 }
			liege = {
				set_character_flag = secret_societies_interference
				character_event = { id = society.60 days = 1095 random = 1095 }
			}
		}
		if = { # Too visible? You auto-reveal yourself if you are independent
			limit = {
				independent = yes
				is_ruler = yes
				check_variable = { which = visibility_value value = 40 }
				NOT = {
					has_character_flag = self_reveal
				}
			}
			set_character_flag = self_reveal
			character_event = { id = society.65 days = 1 } # Found out: brands and excommunicates
		}
	}
}

	any_demesne_title = {
		limit = {
			NOT = {
				holding_type = castle
			}
			lower_tier_than = DUKE
		}
		if = {
			limit = {
				holding_type = city
			}
			create_random_steward = {
				random_traits = yes
				dynasty = random
			}
		}
		else_if = {
			limit = {
				holding_type = temple
			}
			create_random_priest = {
				random_traits = yes
				dynasty = random
			}
		}
		else = {
			create_random_soldier = {
				random_traits = yes
				dynasty = random
			}
		}
		grant_title_no_opinion = new_character
		new_character = {
			# Set Racial Trait
			character_event = {
				id = aiclean.31
			}
		}
	}
}

# The Runefang of Solland: grudge_settler.
# Used by: WHG_empire_events, WHG_empire_effects, WHG_realm_effects
empire_runefang_solland_effect = {
	if = {
		limit = { k_solland = { holder_scope = { empire_runefang_bearer_trigger = yes } } }
		k_solland = { holder_scope = { save_event_target_as = runefang_bearer } }
	}
	else_if = {
		limit = {
			e_the_empire = {
				holder_scope = {
					any_realm_character = {
						has_minor_title = title_reiksmarshal
						empire_runefang_bearer_trigger = yes
					}
				}
			}
		}
		e_the_empire = {
			holder_scope = {
				random_realm_character = {
					limit = {
						has_minor_title = title_reiksmarshal
						empire_runefang_bearer_trigger = yes
					}
					save_event_target_as = runefang_bearer
				}
			}
		}
	}
	else_if = {
		limit = { e_the_empire = { holder_scope = { empire_runefang_bearer_trigger = yes } } }
		e_the_empire = { holder_scope = { save_event_target_as = runefang_bearer } }
	}
	if = {
		limit = { event_target:runefang_bearer = { NOT = { has_artifact = grudge_settler } } }
		if = {
			limit = { any_character = { has_artifact = grudge_settler } }
			random_character = {
				limit = { has_artifact = grudge_settler }
				random_artifact = {
					limit = { artifact_type = grudge_settler }
					transfer_artifact = { from = PREV to = event_target:runefang_bearer }
				}
			}
		}
		else = { event_target:runefang_bearer = { add_artifact = grudge_settler } }
	}
	clear_event_target = runefang_bearer
}

# The scoped witch hunter's liege gives them the court title that fits the
# liege's lands.
# Scope: the witch hunter (a courtier).
# Used by: WHG_witch_hunter_events, employment_decisions +1 more
whunter_give_court_title_effect = {
	liege = {
		if = {
			limit = { whunter_chekist_lands_trigger = yes }
			PREV = { give_minor_title = title_court_chekist }
		}
		else_if = {
			limit = { whunter_inquisition_lands_trigger = yes }
			PREV = { give_minor_title = title_court_inquisitor }
		}
		else_if = {
			limit = { whunter_solkan_lands_trigger = yes }
			PREV = { give_minor_title = title_court_templar }
		}
		else = {
			PREV = { give_minor_title = title_court_witchhunter }
		}
	}
}

				}
			}
		}
		else = { event_target:runefang_bearer = { add_artifact = mothers_ruin } }
	}
	clear_event_target = runefang_bearer
}

# Ghal Maraz, the hammer of Sigmar, belongs to the Emperor. It goes to a
# worthy Emperor from whoever holds it, and is forged anew if it was lost.
# While the Empire is held by one unworthy of it, the Grand Theogonist
# keeps it in Sigmar's name. Before the first coronation no one has it.
# Callers: empire.2 (coronation), empire.41 (the Empire changes hands),
# reclaim_runefangs. Scope: any. Uses the event target ghal_maraz_bearer.
# Used by: WHG_empire_events, wh_empire_decisions
empire_ghal_maraz_effect = {
	if = {
		limit = { e_the_empire = { holder_scope = { empire_runefang_bearer_trigger = yes } } }
		e_the_empire = { holder_scope = { save_event_target_as = ghal_maraz_bearer } }
	}
	else_if = {
		limit = {
			e_the_empire = { has_holder = yes }
			k_cult_of_sigmar = { holder_scope = { is_alive = yes religion = sigmarite } }
		}
		k_cult_of_sigmar = { holder_scope = { save_event_target_as = ghal_maraz_bearer } }
	}
	if = {
		limit = { event_target:ghal_maraz_bearer = { NOT = { has_artifact = ghal_maraz } } }
		if = {
			limit = { any_character = { has_artifact = ghal_maraz } }
			random_character = {
				limit = { has_artifact = ghal_maraz }
				random_artifact = {
					limit = { artifact_type = ghal_maraz }
					transfer_artifact = { from = PREV to = event_target:ghal_maraz_bearer }
				}
			}
		}
		else_if = {
			limit = { event_target:ghal_maraz_bearer = { has_landed_title = e_the_empire } }
			event_target:ghal_maraz_bearer = { add_artifact = ghal_maraz }
		}
	}
	clear_event_target = ghal_maraz_bearer
}

# Gives the scoped hunter the trait and culture of the chapter they serve:
# Inquisitors become inquisitor / inquisitor_culture, Templars of Solkan
# solkan_templar / templar_culture, Order hunters keep witch_hunter. Hunters
# of no chapter (Ulric's, Taal's) keep what they have.
# Scope: the witch hunter.
# Used by: WHG_witch_hunter_events, WHG_empire_effects
whunter_chapter_trait_effect = {
	if = {
		limit = { society_member_of = kislev_chekist } # keeps their own culture
		if = { limit = { trait = witch_hunter } remove_trait = witch_hunter }
		if = { limit = { trait = inquisitor } remove_trait = inquisitor }
		if = { limit = { trait = solkan_templar } remove_trait = solkan_templar }
		if = { limit = { NOT = { trait = chekist } } add_trait = chekist }
	}
	else_if = {
		limit = { society_member_of = holy_inquisition_myrmidia }
		if = { limit = { trait = witch_hunter } remove_trait = witch_hunter }
		if = { limit = { trait = solkan_templar } remove_trait = solkan_templar }
		if = { limit = { NOT = { trait = inquisitor } } add_trait = inquisitor }
		if = {
			limit = { is_ruler = no NOT = { culture = inquisitor_culture } }
			culture = inquisitor_culture
			set_graphical_culture = inquisitor_culture # hunters wear the chapter's look
		}
	}
	else_if = {
		limit = { society_member_of = templars_of_solkan }
		if = { limit = { trait = witch_hunter } remove_trait = witch_hunter }
		if = { limit = { trait = inquisitor } remove_trait = inquisitor }
		if = { limit = { NOT = { trait = solkan_templar } } add_trait = solkan_templar }
		if = {
			limit = { is_ruler = no NOT = { culture = templar_culture } }
			culture = templar_culture
			set_graphical_culture = templar_culture # hunters wear the chapter's look
		}
	}
	else_if = {
		limit = { society_member_of = sigmar_witch_hunters }
		if = { limit = { trait = inquisitor } remove_trait = inquisitor }
		if = { limit = { trait = solkan_templar } remove_trait = solkan_templar }
		if = { limit = { trait = chekist } remove_trait = chekist }
		if = { limit = { NOT = { trait = witch_hunter } } add_trait = witch_hunter }
	}
}

# Restored from vanilla CK2 3.3.5.1 and adapted to the mod's holy orders.
# Run in a title scope after a holy war (ROOT = winner, FROM = winner).
# - Knights Mariner (Manann) ask for islands won in the Old World's seas (was
# Knights Hospitaller).
# - Knights of the White Wolf (Ulric) ask for a duchy won in the north (was the
# Teutonic Order).
# Used by: 00_cb_types, WHG_war_effects

	hidden_tooltip = {
		save_event_target_as = holy_order_potential_title

		if = { # An island the Knights Mariner could be interested in
			limit = {
				is_title_active = d_knights_mariner
				d_knights_mariner = {
					owner = {
						religion_group = ROOT
					}
				}
				ROOT = {
					is_heretic = no
					NOT = {
						has_character_flag = refused_to_give_island_to_knights
					}
				}
				tier = DUKE
				wh_mariner_island_title_trigger = yes
				any_de_jure_vassal_title = {
					tier = COUNT
					owner = {
						OR = {
							character = FROM
							any_liege = {
								character = FROM
							}
						}
					}
				}
			}
			save_event_target_as = potential_title
			ROOT = { save_event_target_as = title_owner }
			d_knights_mariner = {
				owner = {
					character_event = { id = holyorder.1 days = 1 }
				}
			}
		}
		else_if = { # A county the Knights Mariner could be interested in
			limit = {
				tier = COUNT
				is_title_active = d_knights_mariner
				d_knights_mariner = {
					owner = {
						religion_group = ROOT
					}
				}
				ROOT = {
					is_heretic = no
					NOT = {
						has_character_flag = refused_to_give_island_to_knights
					}
				}
				wh_mariner_island_title_trigger = yes
			}
			save_event_target_as = potential_title
			ROOT = { save_event_target_as = title_owner }
			d_knights_mariner = {
				owner = {
					character_event = { id = holyorder.1 days = 1 }
				}
			}
		}

		if = { # A Duchy the Knights of the White Wolf could be interested in
			limit = {
				# Checking that ROOT has the whole Duchy
				NOT = {
					any_direct_de_jure_vassal_title = {
						NOT = {
							owner = {
								OR = {
									character = ROOT
									any_liege = {
										character = ROOT
									}
								}
							}
						}
					}
				}
				# Checking that the White Wolves are active
				is_title_active = d_knights_of_the_white_wolf
				# Checking that Root shares the White Wolves' faith
				d_knights_of_the_white_wolf = {
					owner = {
						religion_group = ROOT
					}
				}
				# Checking that Root don't already own the associated Kingdom of that title
				kingdom = {
					OR = {
						owner = {
							NOT = {
								character = ROOT
							}
						}
						NOT = {
							owner = {
								always = yes
							}
						}
					}
				}
				# Checking that the White Wolves don't already have land
				d_knights_of_the_white_wolf = {
					owner = {
						NOT = {
							any_demesne_title = {
								tier = COUNT
							}
						}
					}
				}
				# Checking that the White Wolves aren't already on a northern crusade
				d_knights_of_the_white_wolf = {
					owner = {
						NOT = {
							any_war = {
								using_cb = northern_crusade_war
							}
						}
					}
				}
				# Checking the ROOT
				ROOT = {
					is_heretic = no
					NOT = {
						has_character_flag = refused_to_give_land_to_white_wolves
					}
				}
				# Duke tier
				tier = DUKE
				# Somewhere in the north: Norsca, northern Kislev or the Chaos Wastes
				OR = {
					region = world_norsca
					region = world_old_world_kislev_northern
					region = world_chaos_wastes
				}
				OR = {
					# Is by water
					any_direct_de_jure_vassal_title = {
						location = {
							port = yes
						}
						owner = {
							OR = {
								character = FROM
								any_liege = {
									character = FROM
								}
							}
						}
					}
					# Is the neighbor to another realm of the same faith
					any_direct_de_jure_vassal_title = {
						location = {
							any_neighbor_province = {
								OR = {
									religion_group = ROOT
									owner = {
										religion_group = ROOT
									}
									owner = {
										any_liege = {
											religion_group = ROOT
										}
									}
								}
							}
						}
					}
				}
				# Is adjacent to enemies of the faith
				any_direct_de_jure_vassal_title = {
					location = {
						any_neighbor_province = {
							owner = {
								OR = {
									NOT = {
										religion_group = ROOT
									}
									any_liege = {
										NOT = {
											religion_group = ROOT
										}
									}
								}
							}
						}
					}
				}
			}
			d_knights_of_the_white_wolf = {
				owner = {
					character_event = { id = holyorder.5 days = 1 }
				}
			}
		}
	}

# A ruler of the Empire crowned by the head of his cult (the coronation
# decision, coronation.366) is endorsed by the cult: a strong claim on the
# Empire, and a flag naming the cult for one faith triumphant (empire.10).
# Scope: the newly crowned ruler. event_target:coronation_priest: the
# priest who crowned him.
# Used by: WHG_coronation_events
empire_coronation_endorsement_effect = {
	if = {
		limit = {
			has_character_flag = flag_crowned_by_pope
			culture_group = imperial_group
			higher_tier_than = DUKE
			NOT = { has_global_flag = united_empire }
			e_the_empire = { has_holder = no }
			event_target:coronation_priest = {
				controls_religion = yes
				OR = { religion = sigmarite religion = ulrican religion = taalite }
			}
		}
		if = { limit = { NOT = { has_strong_claim = e_the_empire } } add_claim = e_the_empire }
		clr_character_flag = empire_crowned_by_sigmarite
		clr_character_flag = empire_crowned_by_ulrican
		clr_character_flag = empire_crowned_by_taalite
		if = {
			limit = { event_target:coronation_priest = { religion = ulrican } }
			set_character_flag = empire_crowned_by_ulrican
		}
		else_if = {
			limit = { event_target:coronation_priest = { religion = taalite } }
			set_character_flag = empire_crowned_by_taalite
		}
		else = { set_character_flag = empire_crowned_by_sigmarite }
		custom_tooltip = { text = empire_coronation_claim_tt }
	}
}
