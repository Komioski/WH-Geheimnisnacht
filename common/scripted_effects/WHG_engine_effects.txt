# buttons, portrait triggers read by interface/portraits

# A scripted effect is a named list of commands, used anywhere an effect can go
# as "name = yes". It runs in the scope it is called from. Names are global
# across the folder and must be unique; the load order of files does not
# matter.
#
# The game calls these by name (vanilla marks them hard-coded or referenced by
# a button), or interface/portraits reads them: never rename or remove one.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Other helpers used by one or two files (1)
#   Not used by any live script yet: kept for content to come (9)

# Referenced by the "Join Crusade" button in the Crusade view for Crusades
# using the new Crusade system
# Used by: wh_chaos_decisions, WHG_black_crusade_events
pledge_crusade_effect = {
	pledge_crusade_participation = yes
	piety = 100
	add_character_modifier = {
		name = recently_pledged_to_crusade_cooldown
		hidden = yes
		duration = 45
	}

	trigger_switch = {
		on_trigger = tier
		EMPEROR = {
			if = {
				limit = {
					is_nomadic = yes
					NOT = { realm_size = 80 }
				}
				add_to_crusade_prestige_pot = 250 add_to_crusade_piety_pot = 100
			}
			else = {
				add_to_crusade_prestige_pot = 2000 add_to_crusade_piety_pot = 800
			}
		}
		KING = { add_to_crusade_prestige_pot = 1000 add_to_crusade_piety_pot = 400 }
		DUKE = { add_to_crusade_prestige_pot = 500 add_to_crusade_piety_pot = 200 }
		COUNT = { add_to_crusade_prestige_pot = 250 add_to_crusade_piety_pot = 100 }
		BARON = { add_to_crusade_prestige_pot = 125 add_to_crusade_piety_pot = 100 }
	}

	random = {
		chance = 50
		add_to_crusade_artifact_pot = 1
	}

	if = {
		limit = {
			trait = excommunicated
		}
		rightful_religious_head_scope = {
			character_event = { id = bcrusade.66 }
		}
	}

	remove_character_modifier = not_pledged_to_crusade
}

unpledge_crusade_effect = {
	pledge_crusade_participation = no
	piety = -300
	add_character_modifier = {
		name = recently_unpledged_to_crusade_cooldown
		hidden = yes
		duration = 365
	}

	trigger_switch = {
		on_trigger = tier
		 EMPEROR = {
			if = {
				limit = {
					is_nomadic = yes
					NOT = { realm_size = 80 }
				}
				add_to_crusade_prestige_pot = -250 add_to_crusade_piety_pot = -100
			}
			else = {
				add_to_crusade_prestige_pot = -2000 add_to_crusade_piety_pot = -800
			}
		}
		KING = { add_to_crusade_prestige_pot = -1000 add_to_crusade_piety_pot = -400 }
		DUKE = { add_to_crusade_prestige_pot = -500 add_to_crusade_piety_pot = -200 }
		COUNT = { add_to_crusade_prestige_pot = -250 add_to_crusade_piety_pot = -100 }
		BARON = { add_to_crusade_prestige_pot = -125 add_to_crusade_piety_pot = -100 }
	}

	random = {
		chance = 50
		add_to_crusade_artifact_pot = -1
	}

	add_character_modifier = {
		name = not_pledged_to_crusade
		years = 3
	}

	set_character_flag = asked_to_pledge
}
