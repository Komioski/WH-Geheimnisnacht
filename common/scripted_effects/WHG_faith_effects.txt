# Sympathy for the faith a character is about to leave: the sympathy trait of
# his current religion group. Call it before the religion changes.
# Groups without a sympathy trait (Chaos, greenskins, the Horned Rat...) get
# nothing. Scope: the convert. Used by faith.601 and faith.700.
faith_old_faith_sympathy_effect = {
	if = {
		limit = { religion_group = old_world_group }
		add_trait = sympathy_old_world_group
	}
	else_if = {
		limit = { religion_group = ormazd_gods_group }
		add_trait = sympathy_ormazd_gods_group
	}
	else_if = {
		limit = { religion_group = cadai_group }
		add_trait = sympathy_cadai_group
	}
	else_if = {
		limit = { religion_group = cytharai_group }
		add_trait = sympathy_cytharai_group
	}
	else_if = {
		limit = { religion_group = dwarf_gods_group }
		add_trait = sympathy_dwarf_gods_group
	}
	else_if = {
		limit = { religion_group = halfling_gods_group }
		add_trait = sympathy_halfling_gods_group
	}
	else_if = {
		limit = { religion_group = oriental_gods_group }
		add_trait = sympathy_oriental_gods_group
	}
}
