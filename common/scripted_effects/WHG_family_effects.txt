# The kinsman gives up on the liege's word and flees abroad to raise an army
# (kinreq.10 in 45 days).
# Scope: the relative. FROM: the liege.
kinreq_flee_abroad_effect = {
	opinion = {
		who = FROM
		modifier = opinion_broken_title_promise
		months = 120
	}
	clr_character_flag = flag_promised_title
	clr_character_flag = flag_later_title
	remove_character_modifier = promised_a_title
	set_character_flag = flag_denied_title
	FROM = { save_event_target_as = kinreq_old_liege }
	character_event = { id = kinreq.10 days = 45 }
}

# Four captains join the pretender's host: women in a matriarchal faith
# whose people have women, men otherwise.
# Scope: the pretender.
kinreq_pretender_captains_effect = {
	if = {
		limit = {
			has_religion_feature = religion_matriarchal
			is_no_females_race = no
		}
		create_random_soldier = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = yes age = 30 }
		create_random_soldier = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = yes age = 25 }
		create_random_soldier = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = yes age = 33 }
		create_random_soldier = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = yes age = 29 }
	}
	else = {
		create_random_soldier = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = no age = 30 }
		create_random_soldier = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = no age = 25 }
		create_random_soldier = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = no age = 33 }
		create_random_soldier = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = no age = 29 }
	}
}
