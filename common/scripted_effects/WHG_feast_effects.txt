# What a boar can do to a hunter: mostly nothing, sometimes a lesson in
# martial skill, sometimes a gored leg, rarely worse.
# Scope: the hunter.
feast_boar_hunt_risk_effect = {
	random_list = {
		80 = { }
		10 = { change_martial = 1 }
		7 = {
			add_trait = wounded
			hidden_effect = { character_event = { id = traitnote.280 } } # Notify wounded
		}
		2 = { add_maimed_trait_effect = yes }
		1 = {
			if = {
				limit = { is_invincible_trigger = yes }
				antideath_effect = yes
			}
			else = {
				death = { death_reason = death_accident_boar }
			}
		}
	}
}

# A good show rubs off on the host: a chance to grow fonder of company and
# more open-handed, unless their character is already set in its ways.
# Scope: the host.
feast_show_effect = {
	prestige = 10
	if = {
		limit = {
			NOT = { trait = gregarious }
			NOT = { trait = shy }
			NOT = { num_of_traits = 5 }
		}
		random = {
			chance = 25
			add_trait = gregarious
			hidden_effect = { character_event = { id = traitnote.265 } } # Notify gregarious
		}
	}
	if = {
		limit = {
			NOT = { trait = charitable }
			NOT = { trait = greedy }
			NOT = { num_of_traits = 5 }
		}
		random = {
			chance = 25
			add_trait = charitable
			hidden_effect = { character_event = { id = traitnote.273 } } # Notify charitable
		}
	}
}

# Kills a duellist, unless something keeps them from dying. FROM, the other
# duellist, is named as the killer.
# Scope: the duellist.
feast_duel_death_effect = {
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		death = { death_reason = death_duel killer = FROM }
	}
}
