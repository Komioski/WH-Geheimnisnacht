# Effects for the Way of Life focus events (events/WHG_<focus>_events.txt:
# theology, scholarship, rulership, family, carousing, hunting, business,
# intrigue, war, and seduction with the lovers). One section per focus.

###############################################################################
# Theology focus
###############################################################################

# One step on the theologian's road. The first study gives Religious
# Studies; after five years of it, the next study turns it into Religious
# Philosopher, which opens the theologian lifestyle (theology.70).
# Scope: the student.
theology_study_step_effect = {
	if = {
		limit = {
			NOT = { has_character_modifier = religious_studies }
			NOT = { has_character_modifier = religious_philosopher }
			NOT = { trait = theologian }
		}
		add_character_modifier = { name = religious_studies duration = -1 }
		set_character_flag = religious_studies_timer
	}
	else_if = {
		limit = {
			has_character_modifier = religious_studies
			had_character_flag = { flag = religious_studies_timer days = 1800 }
		}
		clr_character_flag = religious_studies_timer
		remove_character_modifier = religious_studies
		add_character_modifier = { name = religious_philosopher duration = -1 }
	}
}

# A small chance to keep a token of the shrine: holy water for the Old World
# faiths, a stone of the hold for the Dawi (the pilgrimage tokens of
# 38_artifacts_pilgrimage.txt). Only once per lifetime, and only for a
# visitor who holds no token yet.
# Scope: the visitor.
theology_shrine_token_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = pilgrim_got_token }
			NOT = { any_artifact = { has_artifact_flag = pilgrim_token } }
			OR = {
				religion_group = old_world_group
				religion_group = dwarf_gods_group
			}
		}
		random = {
			chance = 10
			pilgrim_token_effect = yes
		}
	}
}

# The gains of a stay in seclusion that every outcome shares: perhaps
# temperance (with a little weight lost) and perhaps diligence.
# Scope: the hermit.
theology_seclusion_gains_effect = {
	if = {
		limit = { NOR = { trait = temperate trait = gluttonous } }
		random = {
			chance = 20
			add_trait = temperate
			lose_weight_small_effect = yes
			weight_trait_check_effect = yes
			character_event = { id = traitnote.276 } # Notify temperate
		}
	}
	if = {
		limit = { NOR = { trait = diligent trait = slothful } }
		random = {
			chance = 10
			add_trait = diligent
			character_event = { id = traitnote.256 } # Notify diligent
		}
	}
}

###############################################################################
# Scholarship focus
###############################################################################

# Persisting through a bad season at the observatory: perhaps diligence,
# perhaps stress (scholarship.14, 25).
# Scope: the stargazer.
scholarship_persist_effect = {
	if = {
		limit = { NOR = { trait = diligent trait = slothful } }
		random = {
			chance = 30
			add_trait = diligent
			hidden_tooltip = { character_event = { id = traitnote.256 } } # Notify diligent
		}
	}
	if = {
		limit = { NOT = { trait = stressed } }
		random = {
			chance = 10
			add_trait = stressed
			hidden_tooltip = { character_event = { id = traitnote.282 } } # Notify stressed
		}
	}
}

# Giving up the observatory: the diligent lose their diligence, the rest
# grow slothful, and the studies end (scholarship.14, 25).
# Scope: the stargazer.
scholarship_give_up_effect = {
	if = {
		limit = { trait = diligent }
		remove_trait = diligent
	}
	else_if = {
		limit = { NOT = { trait = slothful } }
		add_trait = slothful
	}
	remove_character_modifier = studying_the_stars
}

# The end of the road of the heavens: the studies end, and the stargazer
# becomes a scholar, or keeps the Heliocentrist Scholar modifier if they
# already have a lifestyle trait (scholarship.17).
# Scope: the stargazer.
scholarship_heavens_master_effect = {
	remove_character_modifier = studying_the_stars
	if = {
		limit = { NOT = { lifestyle_traits = 1 } }
		add_trait = scholar
	}
	else = {
		add_character_modifier = { name = heliocentrist_scholar duration = -1 }
	}
}

# Time spent with a relative: a rivalry with them ends, or else each gains
# Family Love for the other until both are fond (family.22, 32, 42).
# Scope: the ruler; FROM is the relative.
family_grow_closer_to_from_effect = {
	if = {
		limit = { is_rival = FROM }
		remove_rival = FROM
	}
	else = {
		if = {
			limit = {
				FROM = { NOT = { personal_opinion = { who = ROOT value = 50 } } }
			}
			FROM = { opinion = { who = ROOT modifier = opinion_family_love } }
		}
		if = {
			limit = {
				NOT = { personal_opinion = { who = FROM value = 50 } }
			}
			opinion = { who = FROM modifier = opinion_family_love }
		}
	}
}

# Picks the white beast of the hunter's lands from the region of the

	clr_character_flag = white_stag

	clr_character_flag = white_lion

				OR = {
					region = world_norsca
					region = world_old_world_kislev

					region = world_new_world_naggaroth
					region = world_chaos_wastes
					region = world_troll_country
				}
			}
		}
		set_character_flag = white_bear
	}
	else_if = {
		limit = {

				OR = {
					region = world_greater_araby_southlands
					region = world_new_world_lustria
					region = world_new_world_arnheim
					region = world_dragon_isles
				}
			}
		}
		set_character_flag = white_tiger

	else = {
		# The Old World, Athel Loren and everywhere else
		set_character_flag = white_stag

}

# Pays for the expedition: the full cost for a rich ruler, a share of the
# yearly income for most, a token sum for the poor (business.13, .14).
# Scope: the ruler.
business_pay_expedition_effect = {
	if = {
		limit = { yearly_income = 200 }
		wealth = -200
	}
	else_if = {
		limit = { yearly_income = 25 }
		scaled_wealth = -1
	}
	else = {
		wealth = -25
	}
}

# Pays the bandits to steal the beasts or the ship: two fifths of the cost
# (business.13, .14). One in five plots fails (fewer for a cunning ruler)
# and the expedition is exposed (business.20). Otherwise the clergy ask to
# come along (business.15).
# Scope: the ruler.
business_hire_thieves_effect = {
	set_character_flag = trade_bandits
	if = {
		limit = { yearly_income = 200 }
		wealth = -80
	}
	else_if = {
		limit = { yearly_income = 25 }
		scaled_wealth = -0.4
	}
	else = {
		wealth = -10
	}
	if = {
		limit = { NOT = { trait = arbitrary } }
		add_trait = arbitrary
	}
	hidden_tooltip = {
		random_list = {
			20 = {
				modifier = { factor = 0.7 intrigue = 8 }
				modifier = { factor = 0.7 intrigue = 11 }
				modifier = { factor = 0.7 intrigue = 14 }
				set_character_flag = trade_bandits_fail
			}
			80 = { }
		}
		character_event = { id = business.15 days = 3 }
	}
}
