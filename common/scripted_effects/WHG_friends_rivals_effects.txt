# Clears the flags of a lesson asked of a friend (friend.40 - friend.49).
# Scope: the one who asked.
friend_clear_skill_effect = {
	clr_character_flag = friend_skill_diplomacy
	clr_character_flag = friend_skill_martial
	clr_character_flag = friend_skill_stewardship
	clr_character_flag = friend_skill_intrigue
	clr_character_flag = friend_skill_learning
}

# What a friend's lesson teaches, by the flag of the skill asked for. A
# steward's lesson also brings new ideas to the province; a priest's brings
# the clergy's favour.
# Scope: the one who asked.
friend_lesson_effect = {
	if = { limit = { has_character_flag = friend_skill_diplomacy } change_diplomacy = 1 }
	else_if = { limit = { has_character_flag = friend_skill_martial } change_martial = 1 }
	else_if = {
		limit = { has_character_flag = friend_skill_stewardship }
		change_stewardship = 1
		if = {
			limit = {
				is_ruler = yes
				location = { NOT = { has_province_modifier = friendly_innovations } }
			}
			location = { add_province_modifier = { name = friendly_innovations duration = 1460 } }
		}
	}
	else_if = { limit = { has_character_flag = friend_skill_intrigue } change_intrigue = 1 }
	else_if = {
		limit = { has_character_flag = friend_skill_learning }
		change_learning = 1
		if = {
			limit = { religion_group = ormazd_gods_group }
			add_character_modifier = { name = pious_imams duration = 1095 }
		}
		else = { add_character_modifier = { name = pious_priests duration = 1095 } }
	}
}

# What it costs to be turned away by a friend, by the skill asked for: a
# would-be diplomat is shamed, a would-be warrior angered, a would-be spy
# grows suspicious, a would-be scholar frets.
# Scope: the one who asked.
friend_refused_effect = {
	if = {
		limit = { has_character_flag = friend_skill_diplomacy }
		prestige = -10
		if = {
			limit = { NOT = { trait = depressed } }
			random = {
				chance = 15
				add_trait = depressed
				hidden_effect = { character_event = { id = traitnote.288 } } # Notify depressed
			}
		}
	}
	else_if = {
		limit = {
			has_character_flag = friend_skill_martial
			NOT = { trait = wroth }
			NOT = { trait = patient }
			NOT = { num_of_traits = 5 }
		}
		random = {
			chance = 15
			add_trait = wroth
			hidden_effect = { character_event = { id = traitnote.254 } } # Notify wroth
		}
	}
	else_if = {
		limit = {
			has_character_flag = friend_skill_intrigue
			NOT = { trait = paranoid }
			NOT = { trait = trusting }
			NOT = { num_of_traits = 5 }
		}
		random = {
			chance = 25
			add_trait = paranoid
			hidden_effect = { character_event = { id = traitnote.275 } } # Notify paranoid
		}
	}
	else_if = {
		limit = { has_character_flag = friend_skill_learning NOT = { trait = stressed } }
		random = {
			chance = 25
			add_trait = stressed
			hidden_effect = { character_event = { id = traitnote.282 } } # Notify stressed
		}
	}
}

# The rival's heir may swear to avenge them: a child of ten or more at home,
# one time in four. Run it before the rival dies.
# Scope: the rival. ROOT is the killer.
rival_heir_vengeance_effect = {
	random_child = {
		limit = { is_primary_heir = yes is_abroad = no age = 10 }
		random = {
			chance = 25
			opinion = { modifier = opinion_killed_father who = ROOT years = 100 }
			character_event = { id = rival.74 days = 3 random = 7 tooltip = EVTTOOLTIPrival74 }
		}
	}
}

# A stray arrow kills the rival, unless something keeps them from dying.
# Scope: the rival. ROOT loosed it.
rival_arrow_death_effect = {
	if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
	else = { rival_heir_vengeance_effect = yes death = { death_reason = death_accident_arrow } }
}

# Poison kills the rival, unless something keeps them from dying.
# Scope: the rival. ROOT poured it.
rival_poison_death_effect = {
	if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
	else = { rival_heir_vengeance_effect = yes death = { death_reason = death_murder_poison } }
}

# Hired cutthroats kill the rival, unless something keeps them from dying.
# Scope: the rival. ROOT paid them.
rival_rabble_death_effect = {
	if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
	else = { rival_heir_vengeance_effect = yes death = { death_reason = death_rabble } }
}

# A sickness from a poisoned cup: the plagues of Reaper's Due where it is
# played, a plain illness otherwise. Peoples who never sicken shrug it off.
# Scope: the poisoned.
rival_poison_sickness_effect = {
	if = {
		limit = { NOT = { is_disease_immune_race = yes } }
		if = {
			limit = { has_dlc = "Reapers" }
			set_character_flag = disease_not_serious
			add_trait = vomiting
			character_event = { id = treatment.6 days = 10 random = 30 }
		}
		else = { add_trait = ill }
	}
}

# Sends hired blades after a rival. A cunning employer, or one with a
# cunning spymaster, finds better blades. Sets revenge_result_escaped,
# revenge_result_wounded, revenge_result_maimed or revenge_result_killed on
# ROOT. With the flag revenge_try_maim on ROOT, the blades are told not to
# kill.
# Scope: the rival. ROOT sent the blades.
rival_revenge_attempt_effect = {
	random_list = {
		40 = {
			modifier = {
				factor = 0.5
				ROOT = { OR = { intrigue = 15 job_spymaster = { intrigue = 15 } } }
			}
			ROOT = { set_character_flag = revenge_result_escaped }
		}
		25 = {
			modifier = { factor = 0 trait = wounded }
			add_trait = wounded
			ROOT = { set_character_flag = revenge_result_wounded }
		}
		20 = {
			modifier = { factor = 0 can_be_maimed_trigger = no }
			add_maimed_trait_effect = yes
			ROOT = { set_character_flag = revenge_result_maimed }
		}
		15 = {
			modifier = { factor = 0 ROOT = { has_character_flag = revenge_try_maim } }
			modifier = { factor = 2 is_maimed_trigger = yes }
			ROOT = { set_character_flag = revenge_result_killed }
			rival_rabble_death_effect = yes
		}
	}
	ROOT = { clr_character_flag = revenge_try_maim }
}

# Clears the outcome flags of a revenge attempt.
# Scope: the one who sought revenge.
rival_clear_revenge_effect = {
	clr_character_flag = revenge_result_escaped
	clr_character_flag = revenge_result_wounded
	clr_character_flag = revenge_result_maimed
	clr_character_flag = revenge_result_killed
	clr_character_flag = revenge_result_slandered
	clr_character_flag = revenge_for_father
}

# Spreads tales about the target of a revenge: half the time they stick,
# more often for a cunning slanderer. A father's killer is called a
# murderer, anyone else merely loses respect. The target hears of it
# (rival.92) and the slanderer learns how it went (rival.94).
# Scope: the one who seeks revenge, with event_target:revenge_target.
rival_revenge_slander_effect = {
	random = {
		chance = 50
		modifier = { factor = 1.5 intrigue = 15 }
		set_character_flag = revenge_result_slandered
	}
	if = {
		limit = { has_character_flag = revenge_result_slandered }
		event_target:revenge_target = {
			prestige = -10
			if = {
				limit = { ROOT = { has_character_flag = revenge_for_father } }
				any_vassal = { opinion = { modifier = evil_murderer who = PREV years = 5 } }
			}
			else = {
				any_vassal = { opinion = { modifier = opinion_lost_respect who = PREV years = 5 } }
			}
			character_event = { id = rival.92 days = 7 random = 7 tooltip = EVTTOOLTIPrival92 }
		}
	}
	else = { set_character_flag = revenge_result_escaped }
	character_event = { id = rival.94 days = 14 random = 7 tooltip = EVTTOOLTIPrival94 }
}

# Sends hired blades after the target of a revenge (see
# rival_revenge_attempt_effect). A survivor hears of it (rival.93) and the
# one who paid learns how it went (rival.94).
# Scope: the one who seeks revenge, with event_target:revenge_target.
rival_revenge_blades_effect = {
	event_target:revenge_target = {
		rival_revenge_attempt_effect = yes
		if = {
			limit = { ROOT = { NOT = { has_character_flag = revenge_result_killed } } }
			character_event = { id = rival.93 days = 7 random = 7 tooltip = EVTTOOLTIPrival93 }
		}
	}
	character_event = { id = rival.94 days = 14 random = 7 tooltip = EVTTOOLTIPrival94 }
}
