# The wife just created by gamerule_wife_effect.
# Scope: the lord.
# Used by: WHG_game_rule_effects
gamerule_new_wife_effect = {
	set_character_flag = recieved_royal_marriage_aid_duty
	new_character = {
		add_spouse = ROOT
		save_event_target_as = mother_target
		character_event = { id = aiclean.31 } # Racial trait
	}
	gamerule_family_prestige_effect = yes
}

# The husband just created by gamerule_husband_effect.

# A grand wedding raises the house's name.
# Scope: the ruler.
# Used by: WHG_game_rule_effects
gamerule_family_prestige_effect = {
	trigger_switch = {
		on_trigger = tier
		EMPEROR = { prestige = 400 }
		KING = { prestige = 300 }
		DUKE = { prestige = 200 }
		COUNT = { prestige = 100 }
	}
}

# A daughter, or a son for the peoples without women.
# Scope: the parent.
# Used by: WHG_game_rule_effects

	if = {
		limit = { is_no_females_race = yes }
		DR_generate_male_child_effect = yes
	}
	else = {
		DR_generate_female_child_effect = yes
	}
}

# Exclave purge (gamerule.20): frees the parts of the realm that the chosen
# exclave_purge rule no longer counts as connected to the capital. Counties
# held by kings, then dukes, then counts go first: a vassal whose own
# capital lies with the exclave goes independent with it; otherwise a
# peasant leader rises and takes it. Loose baronies follow.
# Scope: the independent ruler (ROOT).
# Used by: WHG_maintenance_events
exclave_purge_effect = {
	# Counties are set free first, to avoid releasing vassals of vassals we first loop through provinces held by Kings, then Dukes, and finally Counts
	while = { # Eliminate County exclaves (Kings)
		count = 100
		limit = {
			capital_scope = {
				any_disconnected_province = {
					sub_realm = no
					county = {
						holder_scope = {
							same_realm = ROOT
							tier = KING
						}
					}
					NOR = {
						exclave_dejure_kept_trigger = yes
						has_province_flag = peasant_cleansed
						exclave_connected_trigger = yes
					}
				}
			}
		}
		capital_scope = {
			random_disconnected_province = {
				sub_realm = no
				limit = {
					county = {
						holder_scope = {
							same_realm = ROOT
							tier = KING
						}
					}
					NOR = {
						exclave_dejure_kept_trigger = yes
						has_province_flag = peasant_cleansed
						exclave_connected_trigger = yes
					}
				}
				county = {
					save_event_target_as = county_to_clear
				}
				holder_scope = {
					save_event_target_as = owner_independent
					liege = {
						save_event_target_as = owner_liege
					}
				}
				if = { # If there's a vassal with NO land in the capital region of the top liege, set them free
					limit = {
						holder_scope = {
							NOT = {
								character = ROOT
							}
							capital_scope = {
								is_connected_to = {
									target = event_target:county_to_clear
								}
							}
							NOT = {
								any_realm_province = {
									county = {
										holder_scope = {
											OR = {
												character = event_target:owner_independent
												any_liege = {
													character = event_target:owner_independent
												}
											}
										}
									}
									exclave_connected_trigger = yes
								}
							}
						}
					}
					holder_scope = {
						save_event_target_as = text_target
						character_event = { id = gamerule.24 } # The holder goes independent
						any_liege = {
							character_event = { id = gamerule.21 } # The liege loses a vassal
						}
						set_defacto_liege = event_target:owner_independent
					}
				}
				else_if = { # If the county belongs to a vassal that holds land in the capital region of the top liege, set them free but leave all land that IS in the capital region of the top liege
					limit = {
						holder_scope = {
							NOT = {
								character = ROOT
							}
							capital_scope = {
								is_connected_to = {
									target = event_target:county_to_clear
								}

							character_event = { id = gamerule.24 } # The holder goes independent
							any_liege = {
								character_event = { id = gamerule.21 } # The liege loses a vassal
							}
							set_defacto_liege = THIS
						}
					}
					else_if = { # If the holder is higher tier than baron, but doesn't hold ANY count titles, transfer all vassals/baronies in the capital area to their liege and set them independent
						limit = {
							holder_scope = {
								NOT = {
									character = ROOT
								}
								higher_tier_than = BARON
								NOT = {
									any_demesne_title = {
										tier = COUNT
									}
								}
								any_realm_province = {
									exclave_connected_in_realm_trigger = yes
								}
							}
						}
						holder_scope = {
							any_vassal = {
								limit = {
									any_realm_province = {
										exclave_connected_in_realm_trigger = yes
									}
								}
								set_defacto_liege = event_target:owner_liege
							}
							any_demesne_title = {
								limit = {
									tier = BARON
									exclave_connected_in_realm_trigger = yes
								}
								grant_title = ROOT
							}
							save_event_target_as = text_target
							character_event = { id = gamerule.24 } # The holder goes independent
							any_liege = {
								character_event = { id = gamerule.21 } # The liege loses a vassal
							}
							set_defacto_liege = THIS
						}
					}
					else_if = { # If the holder has land that connects to the liege's capital, set the baronies free under peasants
						limit = {
							holder_scope = {
								NOT = {
									character = ROOT
								}
								any_realm_province = {
									exclave_connected_in_realm_trigger = yes
								}
							}
						}
						create_character = {
							random_traits = yes
							dynasty = culture
							religion = PREV
							culture = PREV
							age = 25
							trait = peasant_leader
						}
						new_character = {
							save_event_target_as = new_holder
							# Racial trait
							character_event = {
			  					id = aiclean.31 }
						}
						usurp_title = { target = event_target:new_holder type = revolt }
						save_event_target_as = text_target_barony
						holder_scope = {
							save_event_target_as = text_target
							any_liege = {
								character_event = { id = gamerule.23 } # The liege loses a barony to a peasant leader
							}
							set_defacto_liege = event_target:new_holder
						}
					}
				}
			}
		}
	}
