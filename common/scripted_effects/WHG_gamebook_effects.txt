# The game-book stories (events/WHG_gamebook_events.txt)

# The grief story is over.
# Scope: the mourner.
grief_end_effect = {
	clr_character_flag = grief_knows_friend_ill
	clr_character_flag = grief_took_owl
	event_target:grief_friend = { clr_character_flag = grief_new_friend }
}

# Clears the boar hunt result flags.
# Scope: the newlywed.
courtly_clear_hunt_effect = {
	clr_character_flag = courtly_shot_hit
	clr_character_flag = courtly_shot_missed
	clr_character_flag = courtly_charge_won
	clr_character_flag = courtly_charge_hurt
	clr_character_flag = courtly_froze_maimed
	clr_character_flag = courtly_froze_death
}

# The boar kills the married one, unless something keeps them from dying.
# Scope: the married one.
courtly_boar_death_effect = {
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		death = { death_reason = death_accident }
	}
}

# The court talks of the affair for ten years.
# Scope: one of the lovers.
courtly_adulterer_effect = {
	if = {
		limit = { is_woman_trigger = yes }
		add_character_modifier = {
			name = adulteress
			years = 10
		}
	}
	else = {
		add_character_modifier = {
			name = adulterer
			years = 10
		}
	}
}
