###############################################################################
# The grand hunt (events/WHG_hunting_events.txt, Part 4, hunting.100 - .199)
#
# hunting_grand_end_effect   ROOT: the hunt is over, modifier and flags gone
# hunting_diligence_effect   the hunt teaches diligence (or prestige)
# hunting_wound_effect       ROOT is wounded, with the notice
# hunting_accident_effect    ROOT dies in a hunting accident (unless
#                            invincible)
# hunting_murdered_effect    ROOT is killed by a plotter (hunting.111)
# hunting_trophy_effect      rare trophy: 10% once a lifetime, no other trophy
#                            held; great_boar_tusks with flag hunt_took_boar,
#                            else bearskin_cloak
###############################################################################

# ROOT is the host.
hunting_grand_end_effect = {
	remove_character_modifier = holding_grand_hunt
	clr_character_flag = hunt_paranoia
	clr_character_flag = hunt_good_hounds
	clr_character_flag = hunt_killed_something
	clr_character_flag = hunt_event_happened
	clr_character_flag = hunt_begin
	clr_character_flag = hunt_scene_1
	clr_character_flag = hunt_scene_2
	clr_character_flag = hunt_scene_3
	clr_character_flag = hunt_scene_4
	clr_character_flag = hunt_took_boar
	clr_character_flag = do_not_disturb
}

# ROOT is the hunter.
hunting_diligence_effect = {
	if = {
		limit = { trait = diligent }
		prestige = 10
	}
	else_if = {
		limit = { NOT = { trait = slothful } }
		add_trait = diligent
		hidden_tooltip = { character_event = { id = 38256 } } # notify diligent
	}
	else = {
		prestige = 10
	}
}

# ROOT is the hunter.
hunting_wound_effect = {
	add_trait = wounded
	hidden_tooltip = { character_event = { id = 38280 } } # notify wounded
}

# ROOT is the hunter.
hunting_accident_effect = {
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		death = { death_reason = death_accident }
	}
}

# ROOT is the hunter.
hunting_trophy_effect = {
	if = {
		limit = { hunting_trophy_allowed_trigger = yes }
		random = {
			chance = 10
			set_character_flag = grand_hunt_trophy_given
			if = {
				limit = { has_character_flag = hunt_took_boar }
				add_artifact = great_boar_tusks
			}
			else = {
				add_artifact = bearskin_cloak
			}
		}
	}
}

# ROOT is the host, murdered at the hunt by a plotter (hunt_plotter); every
# plotter hears of it (hunting.112).
hunting_murdered_effect = {
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		any_courtier = {
			limit = { hunting_plotter_trigger = yes }
			character_event = { id = hunting.112 days = 1 }
		}
		death = {
			death_reason = death_murder_unknown_arrow
			killer = event_target:hunt_plotter
		}
	}
}
