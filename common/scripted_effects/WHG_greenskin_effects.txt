# Makes a newly created greenskin a shaman of the Waaagh! and a son of ROOT.
# Scope: the new character. ROOT is the father.
# Used by: WHG_greenskin_events
greenskin_new_shaman_effect = {
	add_trait = greenskin_shaman
	magic_power_random_effect = yes
	set_father = ROOT
	enable_magic_xp_effect = yes
}

# Grows a greenskin one size step: small to average, average to large, large
# to huge, for orcs and goblins alike.
# Scope: the greenskin.
# Used by: WHG_greenskin_events
greenskin_grow_effect = {
	if = {
		limit = { trait = orc_small }
		remove_trait = orc_small
		add_trait = orc_average
	}
	else_if = {
		limit = { trait = orc_average }
		remove_trait = orc_average
		add_trait = orc_large
	}
	else_if = {
		limit = { trait = orc_large }
		remove_trait = orc_large
		add_trait = orc_huge
	}
	else_if = {
		limit = { trait = goblin_small }
		remove_trait = goblin_small
		add_trait = goblin_average
	}
	else_if = {
		limit = { trait = goblin_average }
		remove_trait = goblin_average
		add_trait = goblin_large
	}
	else_if = {
		limit = { trait = goblin_large }
		remove_trait = goblin_large
		add_trait = goblin_huge
	}
}

# Lowers the Greenskin Power law of the primary title by one step.
# Scope: the ruler.
# Used by: WHG_greenskin_events
warboss_power_falls_effect = {
	primary_title = {
		if = {
			limit = { has_law = greenskin_power_4 }
			add_law_w_cooldown = greenskin_power_3
		}
		else_if = {
			limit = { has_law = greenskin_power_3 }
			add_law_w_cooldown = greenskin_power_2
		}
		else_if = {
			limit = { has_law = greenskin_power_2 }
			add_law_w_cooldown = greenskin_power_1
		}
		else_if = {
			limit = { has_law = greenskin_power_1 }
			add_law_w_cooldown = greenskin_power_0
		}
	}
}

# Gives a new horde leader his horde: a temporary rebel duchy named for his
# culture, the magic of the Waaagh! if he is a shaman, and one of four
# bands.
# Scope: the horde leader. ROOT is the province.
# Used by: WHG_greenskin_events
warboss_horde_army_effect = {
	if = {
		limit = { trait = lore_waaagh }
		add_character_modifier = {
			name = magic_power_2
			duration = -1
		}
	}
	if = {
		limit = { culture = goblin_main }
		create_title = { tier = DUKE landless = yes temporary = yes rebel = yes culture = goblin_main name = "GOBLIN_HORDE" holder = THIS }
	}
	else_if = {
		limit = { culture = goblin_night }
		create_title = { tier = DUKE landless = yes temporary = yes rebel = yes culture = goblin_night name = "NIGHT_GOBLIN_HORDE" holder = THIS }
	}
	else_if = {
		limit = { culture = goblin_forest }
		create_title = { tier = DUKE landless = yes temporary = yes rebel = yes culture = goblin_forest name = "FOREST_GOBLIN_HORDE" holder = THIS }
	}
	else_if = {
		limit = { culture = orc_savage }
		create_title = { tier = DUKE landless = yes temporary = yes rebel = yes culture = orc_savage name = "SAVAGE_ORC_HORDE" holder = THIS }
	}
	else = {
		create_title = { tier = DUKE landless = yes temporary = yes rebel = yes culture = orc_main name = "GREENSKIN_HORDE" holder = THIS }
	}
	random_list = {
		33 = { warboss_horde_band_large_effect = yes }
		34 = { warboss_horde_band_small_effect = yes }
		16 = { warboss_horde_band_raiders_effect = yes }
		17 = { warboss_horde_band_rabble_effect = yes }
	}
}

# Moves a county one step towards greenskin land: no settlers, then a medium
# and a large greenskin population, then the greenskin ruler's culture and
# religion with a tribal camp as the only holding. Counts towards the
# "make things green" bloodline ambition.
# Scope: the county title. ROOT is the greenskin ruler.
# Used by: WHG_greenskin_events
warboss_colonize_effect = {
	location = {
		if = {
			limit = { NOT = { culture_group = ROOT } }
			if = {
				limit = { has_province_modifier = large_greenskin_population }
				remove_province_modifier = large_greenskin_population
				culture = ROOT
				religion = ROOT
				any_province_holding = {
					limit = {
						is_capital = yes
						NOT = { holding_type = nomad }
					}
					convert_to = TRIBAL
					add_building = tb_greenskin_camp_1
					refill_holding_levy = yes
				}
				any_province_holding = {
					limit = { is_capital = no }
					destroy_settlement = THIS
				}
				ROOT = {
					change_variable = { which = bloodline_greenthumb_gs value = 1 }
					if = {
						limit = {
							check_variable = { which = bloodline_greenthumb_gs value = 14 }
							has_ambition = obj_forge_bloodline
						}
						narrative_event = { id = bloodline.258 days = 1 }
					}
				}
			}
			else_if = {
				limit = { has_province_modifier = medium_greenskin_population }
				remove_province_modifier = medium_greenskin_population
				add_province_modifier = {
					name = large_greenskin_population
					duration = -1
				}
			}
			else = {
				add_province_modifier = {
					name = medium_greenskin_population
					duration = -1
				}
			}
		}
	}
}

add_immortal_age_effect = {
	custom_tooltip = { text = immortal_age_T } #Tooltip reuse FTW
	ROOT={
		export_to_variable = {
			which = current_age_variable
			value = age
		}
		change_variable = { which = immortal_age_variable value = 10 } # increase by 1 for every instance of scope
	}
	if={
		limit={ not={has_character_flag=middle_agedmad}
		check_variable = { which = immortal_age_variable value >= 80 } }
		remove_trait = grail_knight
		subtract_variable = { which = age_change_variable which = current_age_variable value = 2 }
		add_age=age_change_variable
		add_age=80
		add_trait = grail_knight
		add_age=-80
		add_age=current_age_variable
		set_character_flag=middle_agedmad
		set_variable = {
			which = current_age_variable
			value = 0
		}
		set_variable = {
			which = age_change_variable
			value = 0
		}
	}
	else_if={
		limit={ not={has_character_flag=old_agedman}
		check_variable = { which = immortal_age_variable value >= 45 } }
		remove_trait = grail_knight
		subtract_variable = { which = age_change_variable which = current_age_variable value = 2 }
		add_age=age_change_variable
		add_age=45
		add_trait = grail_knight
		add_age=-45
		add_age=current_age_variable
		set_character_flag=old_agedman
		set_variable = {
			which = current_age_variable
			value = 0
		}
		set_variable = {
			which = age_change_variable
			value = 0
		}
	}
}
