# Once a life, a rare mount may stand out from the herd: a daemon-touched
# destrier or a steed of the Chaos Wastes for a follower of the Dark Gods,
# a horse of the Hung or Cathay breeds for anyone else. Only if no such
# mount is held already. horse.1.
# Scope: the rider.
horse_rare_mount_effect = {
	if = {
		limit = { NOT = { has_character_flag = horse_rare_mount_rolled } }
		set_character_flag = horse_rare_mount_rolled
		if = {
			limit = {
				OR = {
					religion_group = chaos_gods_group
					secret_religion_group = chaos_gods_group
				}
				NOR = {
					has_artifact = chaos_destrier
					has_artifact = chaos_wastes_war_horse
				}
			}
			random_list = {
				15 = { add_artifact = chaos_destrier }
				5 = { add_artifact = chaos_wastes_war_horse }
				80 = { }
			}
		}
		else_if = {
			limit = { NOT = { has_artifact = tong_steppe_horse } }
			random = {
				chance = 10
				add_artifact = tong_steppe_horse
			}
		}
	}
}

# Riding on through the storm brings a chill (steppe.23). Peoples immune to
# disease shrug it off.
# Scope: the heir.
steppe_storm_chill_effect = {
	if = {
		limit = { is_disease_immune_race = no }
		if = {
			limit = { has_dlc = "Reapers" }
			add_symptom_effect = yes
		}
		else = {
			add_trait = ill
			hidden_tooltip = { character_event = { id = 38290 } } # Notify ill
		}
	}
}

# Death on the road to the Great Maw (steppe.30, steppe.31).
# Scope: the heir.
steppe_maw_road_death_effect = {
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		death = { death_reason = death_accident }
	}
}

# The adventurer turns on a province of the target ruler's realm: war for a
# duchy there, the host stops looting, and the ruler is told (adventurer.12).
# Scope: a province; PREV is the target ruler, ROOT the adventurer.
adventurer_war_on_province_effect = {
	ROOT = {
		unsafe_war = {
			target = PREVPREV
			casus_belli = duchy_adventure
			thirdparty_title = PREV
			tier = DUKE
		}
		any_army = {
			set_can_toggle_looting = yes
			set_looting = no
		}
		PREVPREV = { character_event = { id = adventurer.12 } }
	}
}

# A raider who has looted enough picks a province of a small looted realm to
# carve a duchy from: a port for a sea raider, a border province for a land
# raider, any province otherwise. If war follows, the adventure goes on
# (flag adventurer_next_step); if not, the caller cleans up.
# Scope: the target ruler; ROOT the adventurer.
adventurer_declare_war_effect = {
	ROOT = { clr_character_flag = raiding_adventurer }
	if = {
		limit = {
			ROOT = { is_seafarer = yes }
			any_realm_province = { port = yes }
		}
		random_realm_province = {
			limit = { port = yes }
			adventurer_war_on_province_effect = yes
		}
	}
	else_if = {
		limit = {
			ROOT = { is_seafarer = no }
			any_realm_province = { adventurer_border_province_trigger = yes }
		}
		random_realm_province = {
			limit = { adventurer_border_province_trigger = yes }
			adventurer_war_on_province_effect = yes
		}
	}
	else = {
		random_realm_province = {
			adventurer_war_on_province_effect = yes
		}
	}
	if = {
		limit = { ROOT = { war = yes } }
		ROOT = { set_character_flag = adventurer_next_step }
	}
}
