###############################################################################
# Nomad clans and pillage (WHG_nomad_events.txt)
###############################################################################

# Loot carries off a scholar, a smith or a battle standard: 5 points of a
# random technology (gain_tech_custom_tooltip). nomad.20, nomad.24.
# Scope: the raider.
nomad_plunder_tech_effect = {
	random_list = {
		10 = { military_techpoints = 5 }
		10 = { economy_techpoints = 5 }
		10 = { culture_techpoints = 5 }
	}
}

###############################################################################
# Horses (WHG_horse_events.txt)
###############################################################################

# Once a life, a rare mount may stand out from the herd: a daemon-touched
# destrier or a steed of the Chaos Wastes for a follower of the Dark Gods,
# a horse of the Hung or Cathay breeds for anyone else. Only if no such
# mount is held already. horse.1.
# Scope: the rider.
horse_rare_mount_effect = {
	if = {
		limit = { NOT = { has_character_flag = horse_rare_mount_rolled } }
		set_character_flag = horse_rare_mount_rolled
		if = {
			limit = {
				OR = {
					religion_group = chaos_gods_group
					secret_religion_group = chaos_gods_group
				}
				NOR = {
					has_artifact = chaos_destrier
					has_artifact = chaos_wastes_war_horse
				}
			}
			random_list = {
				15 = { add_artifact = chaos_destrier }
				5 = { add_artifact = chaos_wastes_war_horse }
				80 = { }
			}
		}
		else_if = {
			limit = { NOT = { has_artifact = tong_steppe_horse } }
			random = {
				chance = 10
				add_artifact = tong_steppe_horse
			}
		}
	}
}
