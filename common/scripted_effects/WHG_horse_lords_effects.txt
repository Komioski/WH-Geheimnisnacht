# Once a life, a rare mount may stand out from the herd: a daemon-touched
# destrier or a steed of the Chaos Wastes for a follower of the Dark Gods,
# a horse of the Hung or Cathay breeds for anyone else. Only if no such
# mount is held already. horse.1.
# Scope: the rider.
horse_rare_mount_effect = {
	if = {
		limit = { NOT = { has_character_flag = horse_rare_mount_rolled } }
		set_character_flag = horse_rare_mount_rolled
		if = {
			limit = {
				OR = {
					religion_group = chaos_gods_group
					secret_religion_group = chaos_gods_group
				}
				NOR = {
					has_artifact = chaos_destrier
					has_artifact = chaos_wastes_war_horse
				}
			}
			random_list = {
				15 = { add_artifact = chaos_destrier }
				5 = { add_artifact = chaos_wastes_war_horse }
				80 = { }
			}
		}
		else_if = {
			limit = { NOT = { has_artifact = tong_steppe_horse } }
			random = {
				chance = 10
				add_artifact = tong_steppe_horse
			}
		}
	}
}

# Riding on through the storm brings a chill (steppe.23). Peoples immune to
# disease shrug it off.
# Scope: the heir.
steppe_storm_chill_effect = {
	if = {
		limit = { is_disease_immune_race = no }
		if = {
			limit = { has_dlc = "Reapers" }

		}
		else = {
			add_trait = ill
			hidden_tooltip = { character_event = { id = 38290 } } # Notify ill
		}
	}
}

# The adventurer turns on a province of the target ruler's realm: war for a
# duchy there, the host stops looting, and the ruler is told (adventurer.12).
# Scope: a province; PREV is the target ruler, ROOT the adventurer.
adventurer_war_on_province_effect = {
	ROOT = {
		unsafe_war = {
			target = PREVPREV
			casus_belli = duchy_adventure
			thirdparty_title = PREV
			tier = DUKE
		}
		any_army = {
			set_can_toggle_looting = yes
			set_looting = no
		}
		PREVPREV = { character_event = { id = adventurer.12 } }
	}
}

# A raider who has looted enough picks a province of a small looted realm to
# carve a duchy from: a port for a sea raider, a border province for a land
# raider, any province otherwise. If war follows, the adventure goes on
# (flag adventurer_next_step); if not, the caller cleans up.
# Scope: the target ruler; ROOT the adventurer.
adventurer_declare_war_effect = {
	ROOT = { clr_character_flag = raiding_adventurer }
	if = {
		limit = {
			ROOT = { is_seafarer = yes }
			any_realm_province = { port = yes }
		}
		random_realm_province = {
			limit = { port = yes }
			adventurer_war_on_province_effect = yes
		}
	}
	else_if = {
		limit = {
			ROOT = { is_seafarer = no }
			any_realm_province = { adventurer_border_province_trigger = yes }
		}
		random_realm_province = {
			limit = { adventurer_border_province_trigger = yes }
			adventurer_war_on_province_effect = yes
		}
	}
	else = {
		random_realm_province = {
			adventurer_war_on_province_effect = yes
		}
	}
	if = {
		limit = { ROOT = { war = yes } }
		ROOT = { set_character_flag = adventurer_next_step }
	}
}

# Picks how the life of the band changes the captain and marks it with a
# merc_growth_* flag, for the founder's news (merc.301) and
# merc_captain_growth_effect. merc.300.
# Scope: the captain.

	random_list = {
		10 = { # A sharper eye for battle
			modifier = { factor = 0 NOT = { has_lower_than_max_martial = yes } }
			set_character_flag = merc_growth_education

		10 = { # Fear burned out
			modifier = { factor = 0 NOT = { trait = craven } }
			set_character_flag = merc_growth_craven
		}
		10 = { # Courage
			modifier = { factor = 0 OR = { trait = craven trait = brave } }
			set_character_flag = merc_growth_brave
		}
		10 = { # Drive
			modifier = { factor = 0 OR = { trait = diligent trait = slothful } }
			set_character_flag = merc_growth_diligent
		}
		10 = { # Kindness worn away
			modifier = { factor = 0 NOT = { trait = kind } }
			set_character_flag = merc_growth_kind
		}
		10 = { # Humility worn away
			modifier = { factor = 0 NOT = { trait = humble } }
			set_character_flag = merc_growth_humble

			set_character_flag = merc_growth_cruel
		}
	}
}

# Applies the change picked by merc_captain_growth_pick_effect and clears
# the flag (merc.300 when the founder is dead, merc.301, merc.302).
# Scope: the captain.
merc_captain_growth_effect = {
	if = {
		limit = { has_character_flag = merc_growth_education }
		upgrade_education_effect = yes
	}
	else_if = {
		limit = { has_character_flag = merc_growth_craven }
		remove_trait = craven
	}
	else_if = {
		limit = { has_character_flag = merc_growth_brave }
		add_trait = brave
	}
	else_if = {
		limit = { has_character_flag = merc_growth_diligent }
		add_trait = diligent
	}
	else_if = {
		limit = { has_character_flag = merc_growth_kind }
		remove_trait = kind
	}
	else_if = {
		limit = { has_character_flag = merc_growth_humble }
		remove_trait = humble
	}
	else_if = {
		limit = { has_character_flag = merc_growth_lustful }
		add_trait = lustful
	}
	else_if = {
		limit = { has_character_flag = merc_growth_wroth }
		add_trait = wroth
	}
	else_if = {
		limit = { has_character_flag = merc_growth_cruel }
		add_trait = cruel
	}
	hidden_tooltip = {
		clr_character_flag = merc_growth_education
		clr_character_flag = merc_growth_craven
		clr_character_flag = merc_growth_brave
		clr_character_flag = merc_growth_diligent
		clr_character_flag = merc_growth_kind
		clr_character_flag = merc_growth_humble
		clr_character_flag = merc_growth_lustful
		clr_character_flag = merc_growth_wroth
		clr_character_flag = merc_growth_cruel
	}
}

# Sends the change to the founder (merc.301), or applies it at once when
# the founder is dead.

	if = {
		limit = { primary_title = { band_creator = { is_alive = yes } } }
		primary_title = {
			band_creator = { character_event = { id = merc.301 } }
		}
	}

		merc_captain_growth_effect = yes
	}
}

# A child of the band's lord crosses blades with the captain (merc.300).
# The outcome is kept in a merc_duel_* flag on the captain for the lord's
# (merc.309) and the founder's (merc.310) news; the flags are cleared at
# the next duel.
# Scope: the captain.
merc_captain_duel_effect = {
	clr_character_flag = merc_duel_draw
	clr_character_flag = merc_duel_wounded
	clr_character_flag = merc_duel_maimed
	clr_character_flag = merc_duel_killed
	clr_character_flag = merc_duel_lost
	primary_title = {
		capital_scope = {
			owner = {
				save_event_target_as = merc_duel_host
				random_courtier = {
					limit = { merc_duel_challenger_trigger = yes }
					save_event_target_as = merc_duel_challenger
				}
			}
		}
	}
	random_list = {
		40 = { # Blood drawn on neither side
			modifier = {
				factor = 2
				NOT = { combat_rating_diff = { who = event_target:merc_duel_challenger value = 0 } }
			}
			set_character_flag = merc_duel_draw
			event_target:merc_duel_challenger = { add_rival = ROOT }
		}
		75 = { # The challenger is wounded
			modifier = {
				factor = 2
				combat_rating_diff = { who = event_target:merc_duel_challenger value = 30 }
			}
			modifier = {
				factor = 0
				event_target:merc_duel_challenger = {
					OR = { trait = maimed trait = wounded trait = severely_injured }
				}
			}
			set_character_flag = merc_duel_wounded
			event_target:merc_duel_challenger = {
				add_trait = wounded
				add_rival = ROOT
				random = {
					chance = 50
					event_target:merc_duel_host = { add_rival = ROOT }
				}
			}
		}
		25 = { # The challenger is maimed
			modifier = {
				factor = 2
				combat_rating_diff = { who = event_target:merc_duel_challenger value = 30 }
			}
			modifier = {
				factor = 2
				combat_rating_diff = { who = event_target:merc_duel_challenger value = 40 }
			}
			set_character_flag = merc_duel_maimed
			event_target:merc_duel_challenger = {
				add_maimed_trait_effect = yes
				add_rival = ROOT
				random = {
					chance = 90
					event_target:merc_duel_host = { add_rival = ROOT }
				}
			}
		}
		10 = { # The challenger dies
			modifier = {
				factor = 2
				combat_rating_diff = { who = event_target:merc_duel_challenger value = 30 }
			}
			modifier = {
				factor = 2
				combat_rating_diff = { who = event_target:merc_duel_challenger value = 40 }
			}
			modifier = {
				factor = 2
				combat_rating_diff = { who = event_target:merc_duel_challenger value = 50 }
			}
			modifier = {
				factor = 0
				event_target:merc_duel_challenger = { is_invincible_trigger = yes }
			}
			set_character_flag = merc_duel_killed
			event_target:merc_duel_host = { add_rival = ROOT }
			event_target:merc_duel_challenger = {
				death = { death_reason = death_duel killer = ROOT }
			}
		}
		20 = { # The captain is bested
			modifier = {
				factor = 2
				event_target:merc_duel_challenger = {
					combat_rating_diff = { who = ROOT value = 10 }
				}
			}
			modifier = {
				factor = 2
				event_target:merc_duel_challenger = {
					combat_rating_diff = { who = ROOT value = 30 }
				}
			}
			modifier = {
				factor = 0
				OR = { trait = maimed trait = wounded trait = severely_injured }
			}
			set_character_flag = merc_duel_lost
			add_trait = wounded
			prestige = -25
			event_target:merc_duel_challenger = { prestige = 25 }
		}
	}
	event_target:merc_duel_host = { character_event = { id = merc.309 } }
	primary_title = {
		band_creator = {
			if = {
				limit = {
					is_alive = yes
					NOT = { character = event_target:merc_duel_host }
				}
				character_event = { id = merc.310 }
			}
		}
	}
}
