# The man the sister must hunt (kislev.6): from her Kislevite liege's realm
# first, else a neighbouring Kislevite realm. Prefers men she dislikes.
# Scope: ROOT is the sister. Saves event_target:male_target.
# Used by: WHG_kislev_events
kislev_find_male_mage_effect = {
	if = {
		limit = {
			top_liege = {
				kislev_sisterhood_realm_trigger = yes
				any_realm_character = { kislev_male_mage_trigger = yes }
			}
		}
		top_liege = {
			random_realm_character = {
				limit = { kislev_male_mage_trigger = yes }
				score_value = { value = 10 potential_rival_score = yes }
				score_value = { value = 10 disliked_target_selection_score = yes }
				save_event_target_as = male_target
			}
		}
	}
	else = {
		random_neighbor_independent_ruler = {
			limit = {
				kislev_sisterhood_realm_trigger = yes
				any_realm_character = {
					kislev_male_mage_trigger = yes
					ROOT = { capital_scope = { distance = { where = PREV distance < 1000 } } }
				}
			}
			random_realm_character = {
				limit = {
					kislev_male_mage_trigger = yes
					ROOT = { capital_scope = { distance = { where = PREV distance < 1000 } } }
				}
				score_value = { value = 10 potential_rival_score = yes }
				score_value = { value = 10 disliked_target_selection_score = yes }
				save_event_target_as = male_target
			}
		}
	}
}

# The girl the sister must recruit (kislev.11), in order: a player, her own
# daughter, a woman of her realm, someone of her Kislevite liege's realm, of
# a neighbouring Kislevite realm. Prefers young girls with strong gifts
# (ice_witch_recruit_score).
# Scope: ROOT is the sister. Saves event_target:recruit_to_be.
# Used by: WHG_kislev_events
kislev_find_recruit_effect = {
	if = {
		limit = { any_player = { kislev_witch_recruit_trigger = yes } }
		random_player = {
			limit = { kislev_witch_recruit_trigger = yes }
			save_event_target_as = recruit_to_be
		}
	}
	else_if = {
		limit = { any_child = { kislev_witch_recruit_trigger = yes } }
		random_child = {
			limit = { kislev_witch_recruit_trigger = yes }
			score_value = { value = 10 ice_witch_recruit_score = yes }
			save_event_target_as = recruit_to_be
		}
	}
	else_if = {
		limit = { any_realm_lord = { kislev_witch_recruit_trigger = yes } }
		random_realm_lord = {
			limit = { kislev_witch_recruit_trigger = yes }
			score_value = { value = 10 ice_witch_recruit_score = yes }
			save_event_target_as = recruit_to_be
		}
	}
	else_if = {
		limit = {
			top_liege = {
				kislev_sisterhood_realm_trigger = yes
				any_realm_character = { kislev_witch_recruit_trigger = yes }
			}
		}
		top_liege = {
			random_realm_character = {
				limit = { kislev_witch_recruit_trigger = yes }
				score_value = { value = 10 ice_witch_recruit_score = yes }
				save_event_target_as = recruit_to_be
			}
		}
	}
	else = {
		random_neighbor_independent_ruler = {
			limit = {
				kislev_sisterhood_realm_trigger = yes
				any_realm_character = { kislev_witch_recruit_trigger = yes }
			}
			random_realm_character = {
				limit = { kislev_witch_recruit_trigger = yes }
				score_value = { value = 10 ice_witch_recruit_score = yes }
				save_event_target_as = recruit_to_be
			}
		}
	}
}

# The lessons stop (kislev.16 - 19): the pupil leaves off her studies and the
# teacher's mission ends.
# Used by: WHG_kislev_events
kislev_end_lessons_effect = {
	event_target:recruit_to_be = {
		if = { limit = { is_alive = yes } remove_trait = studying_magic }
	}
	event_target:recruiter = {
		if = { limit = { is_alive = yes } clr_quest = quest_iwitch_recruit }
	}
}

# A Bear Champion's rare hunt prize: a bear that would not die, tamed. One
# time in twenty, once a lifetime, for a Champion or the First with no bear.
# Scope: the hunter.
# Used by: WHG_kislev_events
kislevsoc_war_bear_effect = {
	if = {
		limit = {
			society_member_of = circle_of_the_bear
			society_rank >= 3
			NOT = { has_character_flag = kislevsoc_war_bear_given }
			NOT = { has_artifact = war_bear_mount }
		}
		random = {
			chance = 5
			set_character_flag = kislevsoc_war_bear_given
			set_character_flag = has_mount
			add_artifact = war_bear_mount
		}
	}
}

# The Grand Ice Witch who hands out the sister's mission (kislev.1): another
# sister of the highest rank, or the Sisterhood's head.
# Scope: ROOT is the sister. Saves event_target:lodge_hero.
# Used by: WHG_bretonnia_events, WHG_kislev_events
kislev_save_matriarch_effect = {
	if = {
		limit = {
			any_society_member = {
				NOT = { character = ROOT }
				society_rank == 4
				same_society_as = ROOT
			}
		}
		random_society_member = {
			limit = { NOT = { character = ROOT } society_rank == 4 same_society_as = ROOT }
			save_event_target_as = lodge_hero
		}
	}
	else = {
		random_society_member = {
			limit = { is_society_grandmaster = yes same_society_as = ROOT }
			save_event_target_as = lodge_hero
		}
	}
}

# The boyars come to the Ice Queen's Ball and eat the crown's bread and salt
# (guest right for two years, tsar.50 - 52), then one thing happens among
# the dancers (tsar.61 - 64), or nothing does.
# Scope: ROOT is the ruler of Kislev.
# Used by: WHG_kislev_crown_events
tsar_ball_guests_effect = {
	any_vassal = {
		limit = {
			culture_group = kislevite_group
			prisoner = no
		}
		set_character_flag = salt_guest
		opinion = { who = ROOT modifier = opinion_bread_and_salt }
		opinion = { who = ROOT modifier = opinion_tsar_ball_guest }
	}
	random_list = {
		30 = {
			trigger = {
				any_vassal = {
					prisoner = no
					opinion = { who = ROOT value < 0 }
				}
			}
			character_event = { id = tsar.61 days = 1 }
		}
		25 = {
			trigger = { any_child = { tsar_ball_kin_trigger = yes } }
			character_event = { id = tsar.62 days = 1 }
		}
		20 = {
			trigger = {
				any_courtier = {
					is_adult = yes
					prisoner = no
					OR = { is_cultist_trigger = yes whunter_suspect_courtier_trigger = yes }
				}
			}
			character_event = { id = tsar.63 days = 1 }
		}
		15 = {
			trigger = { any_courtier = { kislev_witch_recruit_trigger = yes } }
			character_event = { id = tsar.64 days = 1 }
		}
		10 = { }
	}
	# An Ice Witch of the court keeps watch over the Ball (WHG_kislev_faith_events)
	if = {
		limit = {
			any_courtier = {
				society_member_of = ice_witch_sisterhood
				society_rank >= 2
				prisoner = no
			}
		}
		random_courtier = {
			limit = {
				society_member_of = ice_witch_sisterhood
				society_rank >= 2
				prisoner = no
			}
			character_event = { id = frostwitch.30 days = 1 }
		}
	}
}

# The Wounded Bear is healed: every lord of Kislev's gods gains Ursun Wakes
# for twenty-five years, and the player lords hear of it (ursun.7).
# Scope: the quest giver (ursun.6).
# Used by: WHG_kislev_faith_events
ursun_wakes_effect = {
	custom_tooltip = { text = ursun_wakes_tt }
	hidden_tooltip = {
		any_character = {
			limit = { ursun_kislev_lord_trigger = yes }
			remove_character_modifier = ursun_long_sleep
			add_character_modifier = { name = ursun_wakes years = 25 }
			if = {
				limit = {
					ai = no
					NOT = { character = ROOT }
				}
				character_event = { id = ursun.7 days = 7 }
			}
		}
	}
	piety = 300
	prestige = 200
}
