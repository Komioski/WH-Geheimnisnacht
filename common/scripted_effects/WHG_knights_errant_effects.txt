# ROOT is the host.
errant_end_visit_effect = {
	clr_character_flag = errant_visiting
	clr_character_flag = errant_recruited
	clr_character_flag = errant_seduction
	clr_character_flag = errant_scene_1
	clr_character_flag = errant_scene_2
	clr_character_flag = errant_scene_3
	clr_character_flag = errant_band_hedge
}

# The seduced woman. The knight is the new_character afterwards.
errant_seducer_effect = {
	add_trait = lustful
	create_character = {
		random_traits = yes
		dynasty = random
		fertility = 0.6
		female = no
		age = 24
		religion = ROOT
		culture = ROOT
	}
	new_character = {
		character_event = { id = aiclean.31 } # racial trait
		impregnate = PREV
		if = {
			limit = { PREV = { is_no_females_race = no } }
			add_lover = PREV
		}
		add_trait = lustful
	}
}
