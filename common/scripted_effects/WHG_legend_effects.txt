# WHG_legend_effects.txt: scripted effects
# Subject: ambitions, deeds of legend, great hunts, artifacts, books and
# debates
#
# A scripted effect is a named list of commands, used anywhere an effect can go
# as "name = yes". It runs in the scope it is called from. Names are global
# across the folder and must be unique; the load order of files does not
# matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Shared: used by three or more files (3)
#   Used by WHG_ambition_events (4)
#   Used by WHG_ambitions (7)
#   Used by WHG_armoury_events (12)
#   Used by WHG_artifact_events (13)
#   Used by WHG_gamebook_events (3)
#   Used by WHG_hunting_events (5)
#   Used by WHG_legend_events (5)
#   Other helpers used by one or two files (7)
###############################################################################

# Used by: WHG_book_events, WHG_chaos_cult_events, WHG_chaos_effects +1 more
give_pneumonic_effect = {
	if = {
		limit = { NOT = { trait = pneumonic } }

		if = {
			limit = {
				has_dlc = "Reapers"
				NOT = { has_character_flag = developing_illness }
				NOT = { has_character_flag = disease_not_serious }
			}
			tooltip = { add_trait = pneumonic hidden_effect = { if = { limit = { is_disease_immune_race = yes } remove_trait = pneumonic } } }
			set_character_flag = got_pneumonia
			set_character_flag = developing_illness
			hidden_tooltip = { character_event = { id = symptom.12 days = 3 random = 10 } }
		}
		if = {
			limit = { NOT = { has_dlc = "Reapers" } }
			add_trait = pneumonic hidden_effect = { if = { limit = { is_disease_immune_race = yes } remove_trait = pneumonic } }
		}
	}
}

# Used by: WHG_book_events, WHG_chaos_effects, WHG_chaos_cult_events +3 more
give_flu_effect = {
	if = {
		limit = {
			NOT = { trait = flu }
			has_dlc = "Reapers"
			NOT = { has_character_flag = developing_illness }
			NOT = { has_character_flag = disease_not_serious }
		}

		set_character_flag = got_flu
		set_character_flag = developing_illness
		hidden_tooltip = { character_event = { id = symptom.21 days = 3 random = 3 } }
	}
	else_if = { # without Reaper's Due: plain illness
		limit = { NOT = { has_dlc = "Reapers" } }
		give_ill_effect = yes
	}
}

# Used by: WHG_warrior_society_events, WHG_duel_events +5 more
add_legend_progress_trivial_effect = {
	if = {
		limit = {
			society = {
				NOT = { has_flag = block_society_progress }  #if legendary journey is not already on-going
				society_has_active_progress = yes #A society-member has used the Legendary Gathering
			}
		}
		custom_tooltip = { text = society_legend_increases_tt }
		hidden_effect = {
			add_society_progress = 1
		}
	}
}

# Marks the "win a war against" ambitions of the war's winner, judged by the
# loser saved as event_target:amb_war_loser (ambition.200 and 201).
# Scope: the winner.
# Used by: WHG_ambition_events
WHG_ambition_war_won_effect = {
	if = {
		limit = {
			has_ambition = obj_serve_the_great_plan
			event_target:amb_war_loser = {
				OR = {
					religion_group = chaos_gods_group
					religion_group = horned_rat_god_group
				}
			}
		}
		set_character_flag = amb_great_plan_served
	}
	else_if = {
		limit = {
			has_ambition = obj_conquer_for_myrmidia
			event_target:amb_war_loser = { culture_group = arabyan_group }
		}
		set_character_flag = amb_myrmidia_conquest
	}
	else_if = {
		limit = {
			has_ambition = obj_beat_da_umies
			event_target:amb_war_loser = { trait = creature_human }
		}
		set_character_flag = amb_umies_beaten
	}
	else_if = {
		limit = {
			has_ambition = obj_walk_the_wild_hunt
			month >= 2
			NOT = { month >= 5 }
		}
		set_character_flag = amb_wild_hunt
	}
	else_if = {
		limit = {
			has_ambition = obj_humble_a_rival_emperor
			event_target:amb_war_loser = { higher_tier_than = KING }
		}
		set_character_flag = amb_emperor_humbled
	}
	else_if = {
		limit = {
			has_ambition = obj_avenge_the_sundering
			event_target:amb_war_loser = { has_landed_title = e_naggaroth }
		}
		set_character_flag = amb_sundering_avenged
	}
	else_if = {
		limit = {
			has_ambition = obj_destroy_the_great_betrayers
			event_target:amb_war_loser = {
				higher_tier_than = DUKE
				religion = nagashi
			}
		}
		set_character_flag = amb_betrayers_destroyed
	}
}

# The ruler who framed an innocent for their theft loses their honesty, and
# may grow deceitful (skill.115, skill.122). Scope: the character.
# Used by: WHG_ambition_events
WHG_skill_deceit_effect = {
	if = {
		limit = { trait = honest }
		remove_trait = honest
	}
	else_if = {
		limit = { NOT = { trait = deceitful } }
		random = {
			chance = 60
			add_trait = deceitful
			hidden_effect = { character_event = { id = traitnote.255 } } # Notify Deceitful
		}
	}
}

# Flags each sin the character holds when Atone for my sins is taken, so the
# ambition ends when one of them is gone (WHG_ambition_sin_atoned_trigger).
# Scope: the character.
# Used by: WHG_ambitions

	if = { limit = { trait = gluttonous } set_character_flag = amb_had_gluttonous }
	if = { limit = { trait = greedy } set_character_flag = amb_had_greedy }
	if = { limit = { trait = slothful } set_character_flag = amb_had_slothful }
	if = { limit = { trait = wroth } set_character_flag = amb_had_wroth }
	if = { limit = { trait = envious } set_character_flag = amb_had_envious }
	if = { limit = { trait = proud } set_character_flag = amb_had_proud }
}

# Flags the virtues the character's god wants broken when a Shed a virtue
# ambition is taken (WHG_ambition_virtue_shed_trigger reads them). Only the
# god's own list is flagged, so losing some other virtue does not count.
# Scope: the character.
# Used by: WHG_ambitions
WHG_ambition_mark_virtues_effect = {
	if = {
		limit = { religion = slaanesh }
		if = { limit = { trait = chaste } set_character_flag = amb_had_chaste }
		if = { limit = { trait = celibate } set_character_flag = amb_had_celibate }
		if = { limit = { trait = temperate } set_character_flag = amb_had_temperate }
		if = { limit = { trait = charitable } set_character_flag = amb_had_charitable }
		if = { limit = { trait = humble } set_character_flag = amb_had_humble }
		if = { limit = { trait = kind } set_character_flag = amb_had_kind }
		if = { limit = { trait = wroth } set_character_flag = amb_had_wroth }
		if = { limit = { trait = shy } set_character_flag = amb_had_shy }
		if = {
			limit = {
				OR = {
					trait = ugly
					trait = disfigured
				}
			}
			set_character_flag = amb_had_ugly
		}
	}
	else_if = {
		limit = { religion = khorne }
		if = { limit = { trait = patient } set_character_flag = amb_had_patient }
		if = { limit = { trait = kind } set_character_flag = amb_had_kind }
		if = { limit = { trait = craven } set_character_flag = amb_had_craven }
	}
	else_if = {
		limit = { religion = nurgle }
		if = { limit = { trait = chaste } set_character_flag = amb_had_chaste }
		if = { limit = { trait = temperate } set_character_flag = amb_had_temperate }
		if = { limit = { trait = diligent } set_character_flag = amb_had_diligent }
	}
	else_if = {
		limit = { religion = tzeentch }
		if = { limit = { trait = honest } set_character_flag = amb_had_honest }
		if = { limit = { trait = trusting } set_character_flag = amb_had_trusting }
		if = { limit = { trait = content } set_character_flag = amb_had_content }
	}
}

# A small raider fleet, once in a lifetime and only one time in four.
# Scope: the ambition holder.
# Used by: WHG_ambitions
WHG_ambition_fleet_reward_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = amb_fleet_given }
			capital_scope = { port = yes }
		}
		random = {
			chance = 25
			set_character_flag = amb_fleet_given
			custom_tooltip = { text = WHG_ambition_fleet_reward_tt }
			hidden_effect = {
				spawn_fleet = {
					province = closest # closest sea zone
					owner = ROOT
					disband_on_peace = no
					troops = { galleys = { 20 20 } }
					earmark = amb_raider_fleet
				}
			}
		}
	}
}

# (artifact_hunt_pay_effect); the search begins in a month (artifact.10).
# Used by: WHG_artifact_events

	set_character_flag = relic_hunter
	save_event_target_as = relic_hunter
	character_event = { id = artifact.10 days = 30 }
}

# Scope: a ruler with a rumor. The hunt costs him a share of his taxes for as
# long as it lasts, more for a greater rumor.
# Used by: WHG_artifact_events, mnm_society_decisions
artifact_hunt_pay_effect = {
	if = {
		limit = { has_character_modifier = relic_rumor_major }
		add_character_modifier = { name = pays_for_artifact_hunt_major duration = -1 }
	}
	else_if = {
		limit = { has_character_modifier = relic_rumor_medium }
		add_character_modifier = { name = pays_for_artifact_hunt_medium duration = -1 }
	}
	else_if = {
		limit = { has_character_modifier = relic_rumor_minor }
		add_character_modifier = { name = pays_for_artifact_hunt_minor duration = -1 }
	}
}

# Scope: any. The hunt of event_target:relic_ruler ends: no more payments,
# no more help lent, and his hunter goes back to his duties. The rumor stays,
# so another councillor can take up the trail (search_for_artifact).
# Used by: WHG_artifact_events, WHG_legend_effects
artifact_hunt_clear_searcher_effect = {
	event_target:relic_ruler = {
		if = {
			limit = { has_character_modifier = pays_for_artifact_hunt_minor }
			remove_character_modifier = pays_for_artifact_hunt_minor
		}
		if = {
			limit = { has_character_modifier = pays_for_artifact_hunt_medium }
			remove_character_modifier = pays_for_artifact_hunt_medium
		}
		if = {
			limit = { has_character_modifier = pays_for_artifact_hunt_major }
			remove_character_modifier = pays_for_artifact_hunt_major
		}
		if = {
			limit = { has_character_modifier = treasure_hunting_diplomats }
			remove_character_modifier = treasure_hunting_diplomats
		}
		if = {
			limit = { has_character_modifier = treasure_hunting_soldiers }
			remove_character_modifier = treasure_hunting_soldiers
		}
		if = {
			limit = { has_character_modifier = treasure_hunting_stewards }
			remove_character_modifier = treasure_hunting_stewards
		}
		if = {
			limit = { has_character_modifier = treasure_hunting_spies }
			remove_character_modifier = treasure_hunting_spies
		}
		if = {
			limit = { has_character_modifier = treasure_hunting_scholars }
			remove_character_modifier = treasure_hunting_scholars
		}
		clr_character_flag = they_asked_for_your_help
		clr_character_flag = artifact_hunt_gold
	}
	hidden_effect = {
		event_target:relic_hunter = { clr_character_flag = relic_hunter }
	}
}

artifact_hunt_reward_money_effect = {
	if = {
		limit = { has_character_modifier = relic_rumor_major }
		transfer_scaled_wealth = { to = FROM value = 1 min = 50 }
		reverse_opinion = { who = FROM modifier = opinion_was_rewarded years = 15 }
		opinion = { who = FROM modifier = opinion_found_major_artifact years = 15 }
	}
	else_if = {
		limit = { has_character_modifier = relic_rumor_medium }
		transfer_scaled_wealth = { to = FROM value = 0.5 min = 35 }
		reverse_opinion = { who = FROM modifier = opinion_was_rewarded years = 15 }
		opinion = { who = FROM modifier = opinion_found_medium_artifact years = 15 }
	}
	else = {
		transfer_scaled_wealth = { to = FROM value = 0.25 min = 20 }
		reverse_opinion = { who = FROM modifier = opinion_was_rewarded years = 10 }
		opinion = { who = FROM modifier = opinion_found_minor_artifact years = 10 }
	}
}

# Scope: the ruler, FROM the hunter (artifact.11). The ruler owes the hunter
# a favor.
# Used by: WHG_artifact_events
artifact_hunt_reward_favor_effect = {
	reverse_add_favor = FROM
	if = {
		limit = { has_character_modifier = relic_rumor_major }
		reverse_opinion = { who = FROM modifier = opinion_was_rewarded years = 15 }
		opinion = { who = FROM modifier = opinion_found_major_artifact years = 15 }
	}
	else_if = {
		limit = { has_character_modifier = relic_rumor_medium }
		reverse_opinion = { who = FROM modifier = opinion_was_rewarded years = 15 }
		opinion = { who = FROM modifier = opinion_found_medium_artifact years = 15 }
	}
	else = {
		reverse_opinion = { who = FROM modifier = opinion_was_rewarded years = 10 }
		opinion = { who = FROM modifier = opinion_found_minor_artifact years = 10 }
	}
}

# Scope: the ruler, FROM the hunter (artifact.11). The ruler sings the
# hunter's praises before the court, and they may become friends.
# Used by: WHG_artifact_events
artifact_hunt_reward_praise_effect = {
	if = {
		limit = { has_character_modifier = relic_rumor_major }
		FROM = { prestige = 100 }
		reverse_opinion = { who = FROM modifier = opinion_was_praised years = 15 }
		opinion = { who = FROM modifier = opinion_found_major_artifact years = 15 }
	}
	else_if = {
		limit = { has_character_modifier = relic_rumor_medium }
		FROM = { prestige = 50 }
		reverse_opinion = { who = FROM modifier = opinion_was_praised years = 15 }
		opinion = { who = FROM modifier = opinion_found_medium_artifact years = 15 }
	}
	else = {
		FROM = { prestige = 25 }
		reverse_opinion = { who = FROM modifier = opinion_was_praised years = 10 }
		opinion = { who = FROM modifier = opinion_found_minor_artifact years = 10 }
	}
	random = {
		chance = 20
		add_friend = FROM
	}
}

# Scope: the victor of a sack, FROM the lord or courtier who kept a treasure
# in the fallen capital (artifact.103). Takes a treasure of the quality
# rolled in artifact.102 (variable artifact_raid_quality), one the victor
# does not already have. A great treasure the victor already has one of is
# marked for burning as a forgery instead (looted_artifact_destroy_duplicate).
# A holy relic of another faith marks the find as heathen
# (artifact_raid_heathen_relic).
# Used by: WHG_artifact_events
artifact_raid_take_effect = {
	if = {
		limit = { is_variable_equal = { which = artifact_raid_quality value = 1 } }
		FROM = {
			random_artifact = {
				limit = {
					quality == 1
					NOT = { artifact_type_owned_by = ROOT }
				}
				artifact_raid_mark_effect = yes
				transfer_artifact = { from = PREV to = ROOT }
			}
		}
	}
	else_if = {
		limit = { is_variable_equal = { which = artifact_raid_quality value = 2 } }
		FROM = {
			random_artifact = {
				limit = {
					quality == 2
					NOT = { artifact_type_owned_by = ROOT }
				}
				artifact_raid_mark_effect = yes
				transfer_artifact = { from = PREV to = ROOT }
			}
		}
	}
	else_if = {
		limit = { is_variable_equal = { which = artifact_raid_quality value = 3 } }
		FROM = {
			random_artifact = {
				limit = {
					quality == 3
					NOT = { artifact_type_owned_by = ROOT }
				}
				artifact_raid_mark_effect = yes
				transfer_artifact = { from = PREV to = ROOT }
			}
		}
		if = {
			limit = {
				NOT = { any_artifact = { has_artifact_flag = looted_artifact_tooltip_1 } }
			}
			set_character_flag = looted_artifact_destroy_duplicate
			FROM = {
				random_artifact = {
					limit = { quality == 3 }
					artifact_raid_mark_effect = yes
				}
			}
		}
	}
	else_if = {
		limit = { is_variable_equal = { which = artifact_raid_quality value = 4 } }
		FROM = {
			random_artifact = {
				limit = {
					quality == 4
					NOT = { artifact_type_owned_by = ROOT }
				}
				artifact_raid_mark_effect = yes
				transfer_artifact = { from = PREV to = ROOT }
			}
		}
		if = {
			limit = {
				NOT = { any_artifact = { has_artifact_flag = looted_artifact_tooltip_1 } }
			}
			set_character_flag = looted_artifact_destroy_duplicate
			FROM = {
				random_artifact = {
					limit = { quality == 4 }
					artifact_raid_mark_effect = yes
				}
			}
		}
	}
	else_if = {
		limit = { is_variable_equal = { which = artifact_raid_quality value = 5 } }
		FROM = {
			random_artifact = {
				limit = {
					quality == 5
					NOT = { artifact_type_owned_by = ROOT }
				}
				artifact_raid_mark_effect = yes
				transfer_artifact = { from = PREV to = ROOT }
			}
		}
		if = {
			limit = {
				NOT = { any_artifact = { has_artifact_flag = looted_artifact_tooltip_1 } }
			}
			set_character_flag = looted_artifact_destroy_duplicate
			FROM = {
				random_artifact = {
					limit = { quality == 5 }
					artifact_raid_mark_effect = yes
				}
			}
		}
	}
}

set_higher_quality_of_book_flags_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = decent_book_potential }
			has_character_flag = good_book_potential
			has_character_flag = great_book_potential
		}
		clr_character_flag = good_book_potential
	}
	if = {
		limit = {
			NOT = { has_character_flag = decent_book_potential }
			has_character_flag = good_book_potential
			NOT = { has_character_flag = great_book_potential }
		}
		set_character_flag = great_book_potential
	}
	if = {
		limit = {
			has_character_flag = decent_book_potential
			has_character_flag = good_book_potential
			has_character_flag = great_book_potential
		}
		clr_character_flag = decent_book_potential
	}
	if = {
		limit = {
			has_character_flag = decent_book_potential
			has_character_flag = good_book_potential
			NOT = { has_character_flag = great_book_potential }
		}
		set_character_flag = great_book_potential
	}
	if = {
		limit = {
			has_character_flag = decent_book_potential
			NOT = { has_character_flag = good_book_potential }
			NOT = { has_character_flag = great_book_potential }
		}
		set_character_flag = good_book_potential
	}
	#Fallback in case flags have cleared somehow...
	if = {
		limit = {
			NOR = {
				has_character_flag = decent_book_potential
        		has_character_flag = good_book_potential
        		has_character_flag = great_book_potential
			}
		}
		set_character_flag = decent_book_potential
	}
}
