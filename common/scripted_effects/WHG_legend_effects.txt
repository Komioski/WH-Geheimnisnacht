# legend_outlaw_end_effect
# Scope: the ruler. The Herrimault chain is over.
legend_outlaw_end_effect = {
	clr_character_flag = legend_outlaw_hunt
	clr_character_flag = legend_outlaw_ignored
	clr_character_flag = legend_outlaw_angered
}

# legend_outlaw_outfoxed_effect
# Scope: the ruler. The outlaw made a fool of the ruler: outfoxed for five
# years, and a new robber band takes heart in the ruler's lands.
legend_outlaw_outfoxed_effect = {
	legend_outlaw_end_effect = yes
	add_character_modifier = { name = modifier_outfoxed duration = 1825 }
	random_demesne_province = {
		limit = { NOT = { has_province_modifier = highway_robber_band } }
		province_event = { id = debt.54 days = 2 random = 10 tooltip = EVTTOOLTIPdebt54 }
	}
}

# legend_outlaw_friends_effect
# Scope: the ruler. The ruler rode with the outlaws and left them be: five
# years of their goodwill, and one robber band in the ruler's lands melts
# into the Herrimault's.
legend_outlaw_friends_effect = {
	legend_outlaw_end_effect = yes
	set_character_flag = legend_outlaw_met
	add_character_modifier = { name = met_merry_men duration = 1825 }
	random_demesne_province = {
		limit = { has_province_modifier = highway_robber_band }
		remove_province_modifier = highway_robber_band
	}
}

# legend_outlaw_hang_effect
# Scope: the ruler. The Herrimault hangs. The hard of the court approve,
# and the ruler grows wroth, or cruel if already wroth.
legend_outlaw_hang_effect = {
	legend_outlaw_end_effect = yes
	set_character_flag = legend_outlaw_dead
	prestige = 10
	any_courtier = {
		limit = { NOR = { trait = charitable trait = kind } }
		opinion = { modifier = opinion_killed_merrymen who = ROOT years = 5 }
	}
	if = { limit = { NOT = { trait = wroth } } add_trait = wroth }
	else_if = { limit = { NOT = { trait = cruel } } add_trait = cruel }
}

# legend_outlaw_pardon_effect
# Scope: the ruler. The Herrimault is pardoned and joins the court with his
# sweetheart (only among peoples with women); they wed a week or two later
# (legend.30).
legend_outlaw_pardon_effect = {
	legend_outlaw_end_effect = yes
	add_character_modifier = { name = the_merciful duration = 365 }
	create_character = {
		random_traits = no
		dynasty = random
		female = no
		age = 30
		culture = ROOT
		religion = ROOT
		trait = misguided_warrior
		trait = brave
		trait = ambitious
		trait = proud
		trait = zealous
		health = 10
		attributes = { martial = 13 intrigue = 13 }
	}
	new_character = {
		character_event = { id = aiclean.31 } # Racial trait
		save_event_target_as = legend_outlaw
	}
	if = {
		limit = { is_no_females_race = no }
		create_character = {
			random_traits = no
			dynasty = random
			female = yes
			age = 25
			culture = ROOT
			religion = ROOT
			trait = naive_appeaser
			trait = brave
			trait = ambitious
			trait = proud
			trait = zealous
			health = 10
			attributes = { martial = 10 intrigue = 12 }
		}
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			save_event_target_as = legend_sweetheart
		}
		narrative_event = { id = legend.30 days = 7 random = 14 tooltip = EVTTOOLTIPlegend30 }
	}
}
