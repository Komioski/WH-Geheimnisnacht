# lover_end_affair_effect
# Scope: the lover being left, called by the one who ends it. The lover
# hears it is over (lover.52), which ends the bond.
lover_end_affair_effect = { character_event = { id = lover.52 tooltip = EVTTOOLTIPlover52 } }

# lover_sickbed_effect
# Scope: a sick lover. A 1 in 10 chance each to throw off illness and
# pneumonia (lover.10).
lover_sickbed_effect = {
	if = { limit = { is_ill = yes } random = { chance = 10 remove_trait = ill } }
	if = { limit = { trait = pneumonic } random = { chance = 10 remove_trait = pneumonic } }
}

# lover_imprisoned_kin_anger_effect
# Scope: an imprisoned character, ROOT the jailer. Their dynasty's close
# family resent ROOT for two years (lover.79).
lover_imprisoned_kin_anger_effect = {
	any_dynasty_member = {
		limit = { is_close_relative = PREV }
		opinion = { modifier = opinion_angry who = ROOT years = 2 }
	}
}
