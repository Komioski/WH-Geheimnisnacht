###############################################################################

# Behead the character, unless fate will not allow it.
# Scope: the condemned. ROOT is the one who orders it.
marriage_behead_effect = {
	if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
	else = { death = { death_reason = death_execution_beheading killer = ROOT } }
}

# Good advice that did not stick: the lord grows content, unless he is
# already content or too ambitious for it.
# Scope: the advised lord.
marriage_advice_content_effect = {
	if = {
		limit = { NOR = { trait = content trait = ambitious } }
		add_trait = content
		hidden_tooltip = { character_event = { id = traitnote.279 } } # notify content
	}
}

# A palfrey from the stables, for a spouse who rides horses and has none.
# Scope: the spouse.
marriage_gift_palfrey_effect = {
	if = {
		limit = {
			has_mount_trigger = no
			mount_beast_rider_trigger = no
			NOR = { trait = creature_dwarf trait = creature_chaos_dwarf }
		}
		add_artifact = palfry
		set_character_flag = has_mount
	}
}

# The lady's choice of favourite is spent.
# Scope: the lady.
marriage_clear_favourite_flags_effect = {
	clr_character_flag = marriage_fav_chancellor
	clr_character_flag = marriage_fav_marshal
	clr_character_flag = marriage_fav_steward
	clr_character_flag = marriage_fav_spymaster
	clr_character_flag = marriage_fav_spiritual
}

# How the pastime went: a little better than even odds of a friend.
# Scope: the lady.
marriage_pastime_roll_effect = {
	random_list = {
		55 = {
			modifier = { factor = 1.5 trait = gregarious }
			modifier = { factor = 0.5 trait = shy }
			character_event = { id = marriage.77 }
		}
		45 = { character_event = { id = marriage.78 } }
	}
}

# The bored wife's search for a friend is over.
# Scope: the lady.
marriage_clear_pastime_flags_effect = {
	clr_character_flag = marriage_making_friends
	clr_character_flag = marriage_tried_boat
	clr_character_flag = marriage_tried_picnic
	clr_character_flag = marriage_tried_hunt
	clr_character_flag = marriage_tried_dinner
}
