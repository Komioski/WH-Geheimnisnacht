# young one of the member's faith, a sister if the flag

}

# Scope: the thief. Takes the relic from the motherhouse.
monastic_relic_take_effect = {
	set_character_flag = monastic_stole_artifact

		on_trigger = has_character_flag
		monastic_relic_tear = {
			add_artifact = shallya_tear_phial
			set_global_flag = monastic_relic_tear_taken
		}
		monastic_relic_heart = {
			add_artifact = bleeding_heart_reliquary
			set_global_flag = monastic_relic_heart_taken
		}
		monastic_relic_shroud = {
			add_artifact = pergunda_shroud
			set_global_flag = monastic_relic_shroud_taken
		}
	}
}

# Scope: the thief. Every member despises the thief, and the order casts
# them out for good.
monastic_relic_cast_out_effect = {
	hidden_effect = {
		society = {
			any_society_member = {
				limit = { NOT = { character = ROOT } }
				opinion = {
					who = ROOT
					modifier = stole_artifact
					years = 20
				}
			}
		}
	}
	if = {
		limit = { society_member_of = cult_of_shallya }
		leave_society = yes
		hidden_effect = { set_character_flag = betrayed_cult_of_shallya }
	}
}
