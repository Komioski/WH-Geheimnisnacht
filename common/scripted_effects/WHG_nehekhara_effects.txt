# A county changes hands: the curse of Nehekhara lies on the land of the
# Nehekharan desert unless a Tomb King or his kin hold it, and the sleeping
# king's malus follows a Tomb King who lies struck down.
# Scope: the province.
nehekhara_county_upkeep_effect = {
	if = {
		limit = { region = world_nehekhara }
		if = {
			limit = { owner = { trait = creature_nehekharan } }
			remove_province_modifier = nehekharan_curse
		}
		else_if = {
			limit = { NOT = { has_province_modifier = nehekharan_curse } }
			add_province_modifier = { name = nehekharan_curse duration = -1 }
		}
	}
	if = {
		limit = {
			owner = {
				trait = creature_nehekharan
				trait = incapacitated
			}
		}
		if = {
			limit = { NOT = { has_province_modifier = incapacitated_tomb_king } }
			add_province_modifier = { name = incapacitated_tomb_king duration = 9000 }
		}
	}
	else = {
		remove_province_modifier = incapacitated_tomb_king
	}
}
