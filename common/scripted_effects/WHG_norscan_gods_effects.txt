# Norscan patron gods (events/WHG_norscan_gods_events.txt)

# An AI Norscan picks a patron god, weighted by tribe, temper, the faith of
# the capital and the realm's pantheon law.
# Scope: the Norscan (no patron yet).
norscan_pick_patron_effect = {
	random_list = {
		20 = { # Kjorn
			modifier = { factor = 0 norscan_dark_patron_allowed_trigger = no }
			modifier = { factor = 0 norscan_old_gods_culture_trigger = yes }
			modifier = { factor = 0.5 trait = kind }
			modifier = { factor = 1.5 trait = cruel }
			modifier = { factor = 1.5 trait = wroth }
			modifier = { factor = 10 capital_scope = { religion = khorne } }
			add_trait = norscan_patron_kharneth
		}
		20 = { # Nierg
			modifier = { factor = 0 norscan_dark_patron_allowed_trigger = no }
			modifier = { factor = 0 norscan_old_gods_culture_trigger = yes }
			modifier = { factor = 1.5 trait = kind }
			modifier = { factor = 0.1 trait = wroth }
			modifier = { factor = 10 capital_scope = { religion = nurgle } }
			add_trait = norscan_patron_nurgleth
		}
		20 = { # Loesh
			modifier = { factor = 0 norscan_dark_patron_allowed_trigger = no }
			modifier = { factor = 1.25 norscan_old_gods_culture_trigger = yes }
			modifier = { factor = 1.5 trait = lustful }
			modifier = { factor = 0.1 NOT = { trait = lustful } }
			modifier = { factor = 10 capital_scope = { religion = slaanesh } }
			add_trait = norscan_patron_shornaal
		}
		15 = { # Tchar
			modifier = { factor = 0 norscan_dark_patron_allowed_trigger = no }
			modifier = { factor = 0.66 norscan_old_gods_culture_trigger = yes }
			modifier = { factor = 15 capital_scope = { religion = tzeentch } }
			add_trait = norscan_patron_tchar
		}
		5 = { # Olric
			modifier = { factor = 0 norscan_old_patron_allowed_trigger = no }
			modifier = { factor = 0 norscan_old_gods_culture_trigger = no NOT = { culture = varg } }
			modifier = { factor = 5 norscan_old_gods_culture_trigger = yes }
			modifier = { factor = 1.5 trait = brave }
			modifier = { factor = 1.5 trait = proud }
			modifier = { factor = 1.6 culture = skaeling }
			add_trait = norscan_patron_ulric
		}
		5 = { # Orsen
			modifier = { factor = 0 culture = varg }
			modifier = { factor = 0 norscan_old_patron_allowed_trigger = no }
			modifier = { factor = 5 norscan_old_gods_culture_trigger = yes }
			modifier = { factor = 1.6 norscan_old_gods_culture_trigger = yes NOT = { culture = skaeling } }
			modifier = { factor = 1.5 trait = slothful }
			modifier = { factor = 1.5 trait = temperate }
			modifier = { factor = 2.6 tier = king }
			add_trait = norscan_patron_ursun
		}
	}
}

# Close kin at court who follow no patron take up the same god.
# Scope: the Norscan who just chose a patron.
norscan_share_patron_effect = {
	save_event_target_as = norscan_patron_chooser
	any_close_relative = {
		limit = {
			host = { character = event_target:norscan_patron_chooser }
			has_norscan_patron_trigger = no
			OR = {
				religion = norscan_gods
				religion = norscan_gods_reformed
			}
		}
		if = {
			limit = { event_target:norscan_patron_chooser = { trait = norscan_patron_kharneth } }
			add_trait = norscan_patron_kharneth
		}
		else_if = {
			limit = { event_target:norscan_patron_chooser = { trait = norscan_patron_shornaal } }
			add_trait = norscan_patron_shornaal
		}
		else_if = {
			limit = { event_target:norscan_patron_chooser = { trait = norscan_patron_nurgleth } }
			add_trait = norscan_patron_nurgleth
		}
		else_if = {
			limit = { event_target:norscan_patron_chooser = { trait = norscan_patron_tchar } }
			add_trait = norscan_patron_tchar
		}
		else_if = {
			limit = { event_target:norscan_patron_chooser = { trait = norscan_patron_ulric } }
			add_trait = norscan_patron_ulric
		}
		else_if = {
			limit = { event_target:norscan_patron_chooser = { trait = norscan_patron_ursun } }
			add_trait = norscan_patron_ursun
		}
	}
}

# A Norscan gives the old gods up for the Dark Gods.
# Scope: the convert. Pays from the warband's favour when given the chance.
norscan_dark_pact_cost_effect = {
	if = {
		limit = {
			is_member_of_any_warband_trigger = yes
			society_currency >= 1000
		}
		change_society_currency = -1000
		piety = -250
	}
	else = {
		piety = -1000
	}
}
