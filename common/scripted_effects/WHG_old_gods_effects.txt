# How the fight ends once the winner is known: the loser flees, yields or
# dies, and the winner may carry a scar from it. Sets the outcome flags the
# result events read: holmgang_won or holmgang_lost, one of holmgang_kill,
# holmgang_yield or holmgang_flight on both, holmgang_scarred on the winner
# and holmgang_maimed on a loser who yields.
# A craven may flee (never a berserker). A berserker or a man of the Blood
# God kills more often, a kind one spares more often, and a much weaker foe
# dies more often and leaves fewer scars.
# Scope: any; needs the event targets holmgang_winner and holmgang_loser.
holmgang_outcome_effect = {
	event_target:holmgang_winner = { set_character_flag = holmgang_won }
	event_target:holmgang_loser = { set_character_flag = holmgang_lost }
	event_target:holmgang_winner = {
		random_list = {
			40 = { # The loser runs for the boats
				modifier = {
					factor = 0
					event_target:holmgang_loser = {
						OR = {
							NOT = { trait = craven }
							trait = berserker
						}
					}
				}
				set_character_flag = holmgang_flight
				event_target:holmgang_loser = { set_character_flag = holmgang_flight }
			}
			20 = { # The loser dies on the holm
				modifier = {
					factor = 0
					event_target:holmgang_loser = {
						trait = craven
						NOT = { trait = berserker }
					}
				}
				modifier = {
					factor = 2
					OR = {
						trait = berserker
						religion_khorne_worshipper_trigger = yes
					}
				}
				modifier = {
					factor = 1.5
					trait = wroth
				}
				modifier = {
					factor = 0.5
					trait = kind
				}
				modifier = {
					factor = 2
					event_target:holmgang_loser = {
						OR = {
							is_the_worst_dueler_compared_to_prev_trigger = yes
							is_terrible_dueler_compared_to_prev_trigger = yes
						}
					}
				}
				set_character_flag = holmgang_kill
				event_target:holmgang_loser = { set_character_flag = holmgang_kill }
			}
			60 = { # The loser yields
				modifier = {
					factor = 0.67
					event_target:holmgang_loser = {
						trait = craven
						NOT = { trait = berserker }
					}
				}
				set_character_flag = holmgang_yield
				event_target:holmgang_loser = { set_character_flag = holmgang_yield }
			}
		}
		if = {
			limit = { NOT = { has_character_flag = holmgang_flight } }
			random_list = {
				60 = {}
				40 = {
					modifier = {
						factor = 0.5
						event_target:holmgang_loser = {
							OR = {
								is_the_worst_dueler_compared_to_prev_trigger = yes
								is_terrible_dueler_compared_to_prev_trigger = yes
							}
						}
					}
					modifier = {
						factor = 1.5
						event_target:holmgang_loser = {
							OR = {
								is_good_dueler_compared_to_prev_trigger = yes
								is_great_dueler_compared_to_prev_trigger = yes
								is_fantastic_dueler_compared_to_prev_trigger = yes
								is_the_best_dueler_compared_to_prev_trigger = yes
							}
						}
					}
					set_character_flag = holmgang_scarred
				}
			}
		}
	}
	event_target:holmgang_loser = {
		if = {
			limit = { has_character_flag = holmgang_yield }
			random = {
				chance = 30
				set_character_flag = holmgang_maimed
			}
		}
	}
}

# Clears every holmgang flag. Run for both fighters when it is over, and by
# the safety event holmgang.90.
# Scope: a fighter.
holmgang_clear_flags_effect = {
	clr_character_flag = holmgang_challenger
	clr_character_flag = holmgang_defender
	clr_character_flag = holmgang_won
	clr_character_flag = holmgang_lost
	clr_character_flag = holmgang_drew
	clr_character_flag = holmgang_kill
	clr_character_flag = holmgang_yield
	clr_character_flag = holmgang_flight
	clr_character_flag = holmgang_scarred
	clr_character_flag = holmgang_maimed
	clr_character_flag = do_not_disturb
}

# The chances shown before the fight, from the foe's skill against PREV's
# (the same bands holmgang.11 weighs the fight on).
# Scope: the foe; PREV is the fighter who reads the tooltip.
holmgang_foe_chances_effect = {
	if = {
		limit = { is_the_worst_dueler_compared_to_prev_trigger = yes }
		custom_tooltip = { text = holmgang_chances_great }
	}
	else_if = {
		limit = { is_terrible_dueler_compared_to_prev_trigger = yes }
		custom_tooltip = { text = holmgang_chances_very_good }
	}
	else_if = {
		limit = { is_bad_dueler_compared_to_prev_trigger = yes }
		custom_tooltip = { text = holmgang_chances_good }
	}
	else_if = {
		limit = {
			OR = {
				is_ok_dueler_compared_to_prev_trigger = yes
				is_decent_dueler_compared_to_prev_trigger = yes
				is_good_dueler_compared_to_prev_trigger = yes
			}
		}
		custom_tooltip = { text = holmgang_chances_even }
	}
	else_if = {
		limit = { is_great_dueler_compared_to_prev_trigger = yes }
		custom_tooltip = { text = holmgang_chances_poor }
	}
	else_if = {
		limit = { is_fantastic_dueler_compared_to_prev_trigger = yes }
		custom_tooltip = { text = holmgang_chances_very_poor }
	}
	else = {
		custom_tooltip = { text = holmgang_chances_terrible }
	}
}

# The loser dies on the holm: the winner takes a fifth of the dead man's
# wealth by holm law, then death (the invincible cheat it).
# Scope: the loser; needs the event target holmgang_winner.
holmgang_death_effect = {
	holmgang_clear_flags_effect = yes
	transfer_scaled_wealth = {
		to = event_target:holmgang_winner
		value = 0.2
	}
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		death = {
			death_reason = death_duel
			killer = event_target:holmgang_winner
		}
	}
}
