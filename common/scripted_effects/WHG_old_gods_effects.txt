# How the fight ends once the winner is known: the loser flees, yields or
# dies, and the winner may carry a scar from it. Sets the outcome flags the
# result events read: holmgang_won or holmgang_lost, one of holmgang_kill,
# holmgang_yield or holmgang_flight on both, holmgang_scarred on the winner
# and holmgang_maimed on a loser who yields.
# A craven may flee (never a berserker). A berserker or a man of the Blood
# God kills more often, a kind one spares more often, and a much weaker foe
# dies more often and leaves fewer scars.
# Scope: any; needs the event targets holmgang_winner and holmgang_loser.
holmgang_outcome_effect = {
	event_target:holmgang_winner = { set_character_flag = holmgang_won }
	event_target:holmgang_loser = { set_character_flag = holmgang_lost }
	event_target:holmgang_winner = {
		random_list = {
			40 = { # The loser runs for the boats
				modifier = {
					factor = 0
					event_target:holmgang_loser = {
						OR = {
							NOT = { trait = craven }
							trait = berserker
						}
					}
				}
				set_character_flag = holmgang_flight
				event_target:holmgang_loser = { set_character_flag = holmgang_flight }
			}
			20 = { # The loser dies on the holm
				modifier = {
					factor = 0
					event_target:holmgang_loser = {
						trait = craven
						NOT = { trait = berserker }
					}
				}
				modifier = {
					factor = 2
					OR = {
						trait = berserker
						religion_khorne_worshipper_trigger = yes
					}
				}
				modifier = {
					factor = 1.5
					trait = wroth
				}
				modifier = {
					factor = 0.5
					trait = kind
				}
				modifier = {
					factor = 2
					event_target:holmgang_loser = {
						OR = {
							is_the_worst_dueler_compared_to_prev_trigger = yes
							is_terrible_dueler_compared_to_prev_trigger = yes
						}
					}
				}
				set_character_flag = holmgang_kill
				event_target:holmgang_loser = { set_character_flag = holmgang_kill }
			}
			60 = { # The loser yields
				modifier = {
					factor = 0.67
					event_target:holmgang_loser = {
						trait = craven
						NOT = { trait = berserker }
					}
				}
				set_character_flag = holmgang_yield
				event_target:holmgang_loser = { set_character_flag = holmgang_yield }
			}
		}
		if = {
			limit = { NOT = { has_character_flag = holmgang_flight } }
			random_list = {
				60 = {}
				40 = {
					modifier = {
						factor = 0.5
						event_target:holmgang_loser = {
							OR = {
								is_the_worst_dueler_compared_to_prev_trigger = yes
								is_terrible_dueler_compared_to_prev_trigger = yes
							}
						}
					}
					modifier = {
						factor = 1.5
						event_target:holmgang_loser = {
							OR = {
								is_good_dueler_compared_to_prev_trigger = yes
								is_great_dueler_compared_to_prev_trigger = yes
								is_fantastic_dueler_compared_to_prev_trigger = yes
								is_the_best_dueler_compared_to_prev_trigger = yes
							}
						}
					}
					set_character_flag = holmgang_scarred
				}
			}
		}
	}
	event_target:holmgang_loser = {
		if = {
			limit = { has_character_flag = holmgang_yield }
			random = {
				chance = 30
				set_character_flag = holmgang_maimed
			}
		}
	}
}

# The chances shown before the fight, from the foe's skill against PREV's
# (the same bands holmgang.11 weighs the fight on).
# Scope: the foe; PREV is the fighter who reads the tooltip.
holmgang_foe_chances_effect = {
	if = {
		limit = { is_the_worst_dueler_compared_to_prev_trigger = yes }
		custom_tooltip = { text = holmgang_chances_great }
	}
	else_if = {
		limit = { is_terrible_dueler_compared_to_prev_trigger = yes }
		custom_tooltip = { text = holmgang_chances_very_good }
	}
	else_if = {
		limit = { is_bad_dueler_compared_to_prev_trigger = yes }
		custom_tooltip = { text = holmgang_chances_good }
	}
	else_if = {
		limit = {
			OR = {
				is_ok_dueler_compared_to_prev_trigger = yes
				is_decent_dueler_compared_to_prev_trigger = yes
				is_good_dueler_compared_to_prev_trigger = yes
			}
		}
		custom_tooltip = { text = holmgang_chances_even }
	}
	else_if = {
		limit = { is_great_dueler_compared_to_prev_trigger = yes }
		custom_tooltip = { text = holmgang_chances_poor }
	}
	else_if = {
		limit = { is_fantastic_dueler_compared_to_prev_trigger = yes }
		custom_tooltip = { text = holmgang_chances_very_poor }
	}
	else = {
		custom_tooltip = { text = holmgang_chances_terrible }
	}
}

# The loser dies on the holm: the winner takes a fifth of the dead man's
# wealth by holm law, then death (the invincible cheat it).
# Scope: the loser; needs the event target holmgang_winner.
holmgang_death_effect = {
	holmgang_clear_flags_effect = yes
	transfer_scaled_wealth = {
		to = event_target:holmgang_winner
		value = 0.2
	}
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		death = {
			death_reason = death_duel
			killer = event_target:holmgang_winner
		}
	}
}

# Sorts the prisoner brought to the blood-tree into one kind, kept as a flag
# on the host for the desc and option variants of rites.12 and rites.13:
# kin first (rites_victim_kin plus the bond), then the head of a faith, then
# dwarf, elf, greenskin, lizardman, a servant of the Dark Gods (only for a
# host who does not serve them), a faithful of the Old World gods (all of
# these also rites_victim_foreign), else a common captive in one of four
# moods.
# Scope: the host; FROM is the prisoner.
rites_blot_classify_victim_effect = {
	rites_blot_clear_victim_flags_effect = yes
	if = {
		limit = {
			OR = {
				sibling = FROM
				father = { character = FROM }
				mother = { character = FROM }
				FROM = { father = { character = ROOT } }
				FROM = { mother = { character = ROOT } }
				any_spouse = { character = FROM }
				any_dynasty_member = { character = FROM }
			}
		}
		set_character_flag = rites_victim_kin
		if = {
			limit = { sibling = FROM FROM = { is_woman_trigger = no } }
			set_character_flag = rites_victim_brother
		}
		else_if = {
			limit = { sibling = FROM }
			set_character_flag = rites_victim_sister
		}
		else_if = {
			limit = { father = { character = FROM } }
			set_character_flag = rites_victim_father
		}
		else_if = {
			limit = { mother = { character = FROM } }
			set_character_flag = rites_victim_mother
		}
		else_if = {
			limit = {
				FROM = {
					OR = {
						father = { character = ROOT }
						mother = { character = ROOT }
					}
					is_woman_trigger = no
				}
			}
			set_character_flag = rites_victim_son
		}
		else_if = {
			limit = {
				FROM = {
					OR = {
						father = { character = ROOT }
						mother = { character = ROOT }
					}
				}
			}
			set_character_flag = rites_victim_daughter
		}
		else_if = {
			limit = { any_spouse = { character = FROM } }
			clr_character_flag = rites_victim_kin # A spouse is no blood kin
			set_character_flag = rites_victim_spouse
		}
		else = {
			set_character_flag = rites_victim_kinsman
		}
	}
	else_if = {
		limit = { FROM = { controls_religion = yes } }
		set_character_flag = rites_victim_foreign
		set_character_flag = rites_victim_head
	}
	else_if = {
		limit = { FROM = { culture_group = dwarf_group } }
		set_character_flag = rites_victim_foreign
		set_character_flag = rites_victim_dwarf
	}
	else_if = {
		limit = { FROM = { is_elf = yes } }
		set_character_flag = rites_victim_foreign
		set_character_flag = rites_victim_elf
	}
	else_if = {
		limit = {
			FROM = {
				OR = {
					culture_group = orc_group
					culture_group = goblin_group
				}
			}
		}
		set_character_flag = rites_victim_foreign
		set_character_flag = rites_victim_greenskin
	}
	else_if = {
		limit = { FROM = { culture_group = lizardman_group } }
		set_character_flag = rites_victim_foreign
		set_character_flag = rites_victim_lizardman
	}
	else_if = {
		limit = {
			FROM = { religion_group = chaos_gods_group }
			NOT = { religion_group = chaos_gods_group }
		}
		set_character_flag = rites_victim_foreign
		set_character_flag = rites_victim_chaos
	}
	else_if = {
		limit = { FROM = { religion_group = old_world_group } }
		set_character_flag = rites_victim_foreign
		set_character_flag = rites_victim_old_world
	}
	else = {
		set_character_flag = rites_victim_common
		random_list = {
			25 = { set_character_flag = rites_victim_mood_piss }
			25 = { set_character_flag = rites_victim_mood_run }
			25 = { set_character_flag = rites_victim_mood_calm }
			25 = { set_character_flag = rites_victim_mood_spit }
		}
	}
}

# The prisoner hangs from the blood-tree. A common captive is worth 25
# prestige and 15 piety, a foreign one 50 and 25, the head of a faith 250 and
# 100 with a blow to that faith. A man of the Blood God gains 25 piety more.
# Kin make the host a kinslayer. The guests see it (rites.13), then the next
# prisoner is fetched (rites.10).
# Scope: the host; FROM is the prisoner.
rites_blot_offer_effect = {
	if = {
		limit = { FROM = { controls_religion = yes } }
		prestige = 250
		piety = 100
		religion_authority = { modifier = sacrificed_head_good years = 20 }
		FROM = { religion_authority = { modifier = sacrificed_head_bad years = 20 } }
		if = {
			limit = {
				ai = no
				NOT = { has_character_flag = achievement_holy_smoke }
			}
			hidden_effect = { set_character_flag = achievement_holy_smoke }
		}
	}
	else_if = {
		limit = { has_character_flag = rites_victim_foreign }
		prestige = 50
		piety = 25
	}
	else = {
		prestige = 25
		piety = 15
	}
	if = {
		limit = { religion_khorne_worshipper_trigger = yes }
		piety = 25
	}
	if = {
		limit = { has_character_flag = rites_victim_kin }
		FROM = { save_event_target_as = kinslayer_target }
		add_kinslayer_trait_effect = yes
	}
	change_variable = { which = blot_offerings value = 1 }
	hidden_effect = {
		any_vassal = {
			limit = { has_character_flag = rites_blot_guest }
			character_event = { id = rites.13 }
		}
	}
	FROM = {
		clr_character_flag = rites_blot_victim
		if = {
			limit = { is_invincible_trigger = yes }
			antideath_effect = yes
		}
		else = {
			death = {
				death_reason = death_sacrificed_norse
				killer = ROOT
			}
		}
	}
	hidden_effect = { character_event = { id = rites.10 days = 2 } }
}

	}
	hidden_effect = { character_event = { id = rites.10 days = 2 } }
}

# The two guests at a feast become lovers (never in races without women).
# Scope: either lover; event targets rites_feast_culprit, rites_feast_partner.

	if = {
		limit = { character = event_target:rites_feast_culprit }

			limit = {
				is_no_females_race = no
				event_target:rites_feast_partner = { is_no_females_race = no }
			}
			add_lover = event_target:rites_feast_partner
		}
	}
	else = {
		tooltip = {
			if = {
				limit = {
					is_no_females_race = no
					event_target:rites_feast_culprit = { is_no_females_race = no }
				}
				add_lover = event_target:rites_feast_culprit
			}
		}
	}
}

# How the dead parent died, as a flag on ROOT for the inscription
# (runestone.10): by ROOT's own hand first (murder, execution, dungeon or
# sacrifice), then natural, battle, executed, murdered, accident (and murder
# by an unknown hand), dungeon, sacrificed, duel, suicide, else other.
# Scope: the commissioner; event target runestone_parent.
runestone_parent_death_effect = {
	event_target:runestone_parent = {
		if = {
			limit = {
				killer = { character = ROOT }
				OR = {
					death_murder_known_trigger = yes
					death_execution_trigger = yes
					death_reason = death_dungeon
					death_sacrificed_trigger = yes
				}
			}
			ROOT = { set_character_flag = runestone_death_by_you }
		}
		else_if = {
			limit = { death_reason = death_natural }
			ROOT = { set_character_flag = runestone_death_natural }
		}
		else_if = {
			limit = { death_reason = death_battle }
			ROOT = { set_character_flag = runestone_death_battle }
		}
		else_if = {
			limit = { death_execution_trigger = yes }
			ROOT = { set_character_flag = runestone_death_executed }
		}
		else_if = {
			limit = { death_murder_known_trigger = yes }
			ROOT = { set_character_flag = runestone_death_murdered }
		}
		else_if = {
			limit = {
				OR = {
					death_accident_trigger = yes
					death_murder_unknown_trigger = yes
				}
			}
			ROOT = { set_character_flag = runestone_death_accident }
		}
		else_if = {
			limit = { death_reason = death_dungeon }
			ROOT = { set_character_flag = runestone_death_dungeon }
		}
		else_if = {
			limit = { death_sacrificed_trigger = yes }
			ROOT = { set_character_flag = runestone_death_sacrificed }
		}
		else_if = {
			limit = { death_reason = death_duel }
			ROOT = { set_character_flag = runestone_death_duel }
		}
		else_if = {
			limit = { death_suicide_trigger = yes }
			ROOT = { set_character_flag = runestone_death_suicide }
		}
		else = {
			ROOT = { set_character_flag = runestone_death_other }
		}
	}
}

# The reward for a raised stone: 200 prestige and 100 piety for most stones;
# 300 prestige for a stone of beauty, wisdom, strength, pride, courage, raids
# or a parent who fell in battle (250 for a duel); only 100 prestige for a
# mad, possessed or witless boast, a parent's suicide or a parent ROOT killed.
# Then two years of raised_runestone and a chronicle entry.
# Scope: the commissioner.
runestone_reward_effect = {
	if = {
		limit = {
			OR = {
				has_character_flag = runestone_self_fair
				has_character_flag = runestone_self_genius
				has_character_flag = runestone_self_strong
				has_character_flag = runestone_self_proud
				has_character_flag = runestone_self_brave
				has_character_flag = runestone_self_raider
				has_character_flag = runestone_death_battle
			}
		}
		prestige = 300
	}
	else_if = {
		limit = { has_character_flag = runestone_death_duel }
		prestige = 250
	}
	else_if = {
		limit = {
			OR = {
				has_character_flag = runestone_self_lunatic
				has_character_flag = runestone_self_possessed
				has_character_flag = runestone_self_imbecile
				has_character_flag = runestone_death_suicide
				has_character_flag = runestone_death_by_you
			}
		}
		prestige = 100
	}
