###############################################################################
# Life in the dungeon (events/WHG_prison_events.txt, prison.70 - .86) and the
# battle captures (events/WHG_battle_events.txt)
#
# prison_clear_cell_effect   the prisoner leaves house arrest, the dungeon
#                            and the oubliette
# prison_sacrifice_effect    ROOT sacrifices captured_commander: the stacking
#                            sacrificed_prisoner modifier (at most three,
#                            refreshed), the captive's death (prison.82), and
#                            burned hearts for faiths of ritual sacrifice
###############################################################################

# ROOT is the prisoner.
prison_clear_cell_effect = {
	remove_character_modifier = house_arrest
	remove_character_modifier = the_dungeon
	remove_character_modifier = the_oubliette
}

# ROOT is the captor's liege; event_target:captured_commander the captive.
prison_sacrifice_effect = {
	if = {
		limit = {
			has_instances_of_character_modifier = { modifier = sacrificed_prisoner amount = 3 }
		}
		hidden_effect = { # refresh the end date of all three
			remove_character_modifier = sacrificed_prisoner
			add_character_modifier = { name = sacrificed_prisoner duration = 1000 stacking = yes }
			add_character_modifier = { name = sacrificed_prisoner duration = 1000 stacking = yes }
		}
		add_character_modifier = { name = sacrificed_prisoner duration = 1000 stacking = yes }
	}
	else = {
		add_character_modifier = { name = sacrificed_prisoner duration = 1000 stacking = yes }
	}
	hidden_effect = {
		event_target:captured_commander = { character_event = { id = prison.82 } }
		if = {
			limit = {
				OR = {
					has_religion_feature = religion_ritual_sacrifice
					religion_group = old_ones_group
				}
			}
			if = { # a foreign faith head is a fine offering
				limit = {
					event_target:captured_commander = {
						NOT = { religion = ROOT }
						controls_religion = yes
					}
				}
				change_variable = { which = burned_hearts_points value = 2 }
			}
			if = { # so is a foreign emperor
				limit = {
					event_target:captured_commander = {
						NOT = { religion = ROOT }
						higher_tier_than = KING
					}
				}
				change_variable = { which = burned_hearts_points value = 2 }
			}
			bloodthirsty_tiered_traits_effect = yes
		}
	}
}
