# Scope: the slaver.
# Used by: WHG_realm_effects, WHG_slavery_decisions, WHG_slavery_events
slave_stock_update_effect = {
	hidden_effect = {
		if = {
			limit = { check_variable = { which = slaves value > 10 } }
			set_variable = { which = slaves value = 10 }
		}
		if = {
			limit = { NOT = { check_variable = { which = slaves value = 0 } } }
			set_variable = { which = slaves value = 0 }
		}
		remove_character_modifier = slave_pens_1
		remove_character_modifier = slave_pens_2
		remove_character_modifier = slave_pens_3
		if = {
			limit = { check_variable = { which = slaves value = 6 } }
			add_character_modifier = { name = slave_pens_3 duration = -1 }
			set_character_flag = slave_master
		}
		else_if = {
			limit = { check_variable = { which = slaves value = 3 } }
			add_character_modifier = { name = slave_pens_2 duration = -1 }
			set_character_flag = slave_master
		}
		else_if = {
			limit = { check_variable = { which = slaves value = 1 } }
			add_character_modifier = { name = slave_pens_1 duration = -1 }
			set_character_flag = slave_master
		}
		else = {
			clr_character_flag = slave_master
		}
		if = { # Deed: Hashut's due, the pens full (chronicle.131)
			limit = {
				ai = no
				culture_group = dwarf_chaos_group
				check_variable = { which = slaves value = 10 }
				NOT = { has_character_flag = deed_hashut }
			}
			character_event = { id = chronicle.131 }
		}
	}
}

# One lot of slaves more. Scope: the slaver.

# The shattered_succession rule's law for every demesne title.
# Scope: a landed feudal or tribal ruler.
# Used by: WHG_maintenance_events
altstart_succession_effect = {
	if = {
		limit = { has_alternate_start_setting = { setting = shattered_succession option = all_primogeniture } }
		any_demesne_title = { succession = primogeniture }
	}
	else_if = {
		limit = { has_alternate_start_setting = { setting = shattered_succession option = all_gavelkind } }
		any_demesne_title = { succession = gavelkind }
	}
	else_if = {
		limit = { has_alternate_start_setting = { setting = shattered_succession option = all_elective_gavelkind } }
		any_demesne_title = {
			succession = {
				base_type = elective_gavelkind
				voting_rules = elective_gavelkind
			}
		}
	}
	else_if = {
		limit = { has_alternate_start_setting = { setting = shattered_succession option = all_elective } }
		any_demesne_title = {
			succession = {
				base_type = feudal_elective
				voting_rules = feudal_elective
			}
		}
	}
	else_if = {
		limit = { has_alternate_start_setting = { setting = shattered_succession option = all_seniority } }
		any_demesne_title = { succession = seniority }
	}
	else_if = {
		limit = { has_alternate_start_setting = { setting = shattered_succession option = all_ultimogeniture } }
		any_demesne_title = { succession = ultimogeniture }
	}
	else_if = {
		limit = { has_alternate_start_setting = { setting = shattered_succession option = all_tanistry } }

# One step of campaign favour: 50 prestige, counted in election_prestige so
# that republic_election_clear_effect can take it back.
# Scope: the candidate.
# Used by: WHG_republic_events
republic_election_gain_effect = {
	prestige = 50
	change_variable = { which = election_prestige value = 1 }
}

# One step of favour lost (a smear, a flop, a miser's reputation). Counted
# too, so it is given back when the election is over.
# Scope: the candidate.
# Used by: WHG_republic_events
republic_election_lose_effect = {
	prestige = -50
	change_variable = { which = election_prestige value = -1 }
}

# The bought favour fades: takes back every counted step (or gives back
# every lost one) and clears the candidate's election state.
# Scope: the candidate.
# Used by: WHG_republic_events, WHG_realm_effects
republic_election_clear_effect = {
	while = {
		limit = { check_variable = { which = election_prestige value = 1 } }
		prestige = -50
		change_variable = { which = election_prestige value = -1 }
	}
	while = {
		limit = { NOT = { check_variable = { which = election_prestige value = 0 } } }
		prestige = 50
		change_variable = { which = election_prestige value = 1 }
	}
	set_variable = { which = election_prestige value = 0 }
	set_variable = { which = election_votes value = 0 }
	clr_character_flag = election_running
	clr_character_flag = election_candidate
	clr_character_flag = election_bribed
	clr_character_flag = election_smeared
}

# Ends the election: every candidate's favour fades and each hears the
# result (election.60). If the doge stood in the election and is left with
# negative prestige, he is impeached (election.65) instead of hearing it.
# Scope: the doge holding the seal after the count, saved as
# event_target:election_new_doge. event_target:election_doge called it.
# Used by: WHG_republic_events
republic_election_end_effect = {
	if = {
		limit = { has_character_flag = election_candidate }
		set_character_flag = election_fraud_check
	}
	republic_election_clear_effect = yes
	any_vassal = {
		limit = { has_character_flag = election_candidate }
		republic_election_clear_effect = yes
		character_event = { id = election.60 }
	}
	if = {
		limit = { has_character_flag = election_fraud_check NOT = { prestige = 0 } }
		character_event = { id = election.65 }
	}
	else = { character_event = { id = election.60 } }
	clr_character_flag = election_fraud_check
}

# and their top liege are furious for three years; everyone in the top
# liege's realm who is not cruel thinks the commander dishonourable for
# ten.
# Used by: WHG_wonder_events
wonder_sack_outrage_effect = {
	FROM = {
		holder_scope = {
			show_scope_change = no
			opinion = { modifier = opinion_furious who = ROOT years = 3 }
			top_liege = {
				show_scope_change = no
				opinion = { modifier = opinion_furious who = ROOT years = 3 }
				custom_tooltip = { text = EVTTOOLTIPwonder210c }
				hidden_effect = {
					any_realm_character = {
						limit = { NOT = { trait = cruel } }
						opinion = { modifier = opinion_dishonorable who = ROOT years = 10 }
					}
				}
			}
		}
	}
}
