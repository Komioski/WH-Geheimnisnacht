###############################################################################
# Merchant republics: scripted effects (events/WHG_republic_events.txt)
#
# Republic elections (Part 5, namespace election):
# republic_election_term_effect    doge scope: starts a term, sets
#                                  election_timer by the title's term law
# republic_election_gain_effect    candidate scope: one step of campaign
#                                  favour (+50 prestige, counted)
# republic_election_lose_effect    candidate scope: loses one step
# republic_election_clear_effect   candidate scope: takes back the counted
#                                  favour and clears the election state
# republic_election_end_effect     scope: the doge holding the seal after the
#                                  count; ends the election for everyone
###############################################################################

# Starts a doge's term. The timer runs a year short of the term, since the
# yearly pulse calls the election within a year of it running out.
# Scope: the doge.
republic_election_term_effect = {
	set_character_flag = election_term_set
	remove_character_modifier = election_timer
	primary_title = {
		if = {
			limit = { has_law = law_election_cycles_0 }
			PREV = { add_character_modifier = { name = election_timer years = 3 } }
		}
		else_if = {
			limit = { has_law = law_election_cycles_1 }
			PREV = { add_character_modifier = { name = election_timer years = 7 } }
		}
		else_if = {
			limit = { has_law = law_election_cycles_2 }
			PREV = { add_character_modifier = { name = election_timer years = 15 } }
		}
		else_if = {
			limit = { has_law = law_election_cycles_3 }
			PREV = { add_character_modifier = { name = election_timer years = 9 } }
		}
		else_if = {
			limit = { has_law = law_election_cycles_4 }
			PREV = { add_character_modifier = { name = election_timer years = 14 } }
		}
		else_if = {
			limit = { has_law = law_election_cycles_5 }
			PREV = { add_character_modifier = { name = election_timer years = 19 } }
		}
	}
}

# One step of campaign favour: 50 prestige, counted in election_prestige so
# that republic_election_clear_effect can take it back.
# Scope: the candidate.
republic_election_gain_effect = {
	prestige = 50
	change_variable = { which = election_prestige value = 1 }
}

# One step of favour lost (a smear, a flop, a miser's reputation). Counted
# too, so it is given back when the election is over.
# Scope: the candidate.
republic_election_lose_effect = {
	prestige = -50
	change_variable = { which = election_prestige value = -1 }
}

# The bought favour fades: takes back every counted step (or gives back
# every lost one) and clears the candidate's election state.
# Scope: the candidate.
republic_election_clear_effect = {
	while = {
		limit = { check_variable = { which = election_prestige value = 1 } }
		prestige = -50
		change_variable = { which = election_prestige value = -1 }
	}
	while = {
		limit = { NOT = { check_variable = { which = election_prestige value = 0 } } }
		prestige = 50
		change_variable = { which = election_prestige value = 1 }
	}
	set_variable = { which = election_prestige value = 0 }
	set_variable = { which = election_votes value = 0 }
	clr_character_flag = election_running
	clr_character_flag = election_candidate
	clr_character_flag = election_bribed
	clr_character_flag = election_smeared
}

# Ends the election: every candidate's favour fades and each hears the
# result (election.60). If the doge stood in the election and is left with
# negative prestige, he is impeached (election.65) instead of hearing it.
# Scope: the doge holding the seal after the count, saved as
# event_target:election_new_doge. event_target:election_doge called it.
republic_election_end_effect = {
	if = {
		limit = { has_character_flag = election_candidate }
		set_character_flag = election_fraud_check
	}
	republic_election_clear_effect = yes
	any_vassal = {
		limit = { has_character_flag = election_candidate }
		republic_election_clear_effect = yes
		character_event = { id = election.60 }
	}
	if = {
		limit = { has_character_flag = election_fraud_check NOT = { prestige = 0 } }
		character_event = { id = election.65 }
	}
	else = { character_event = { id = election.60 } }
	clr_character_flag = election_fraud_check
}
