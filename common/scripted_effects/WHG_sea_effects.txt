###############################################################################
# The seas of the world: effects for the pirate and raider module
# (events/WHG_sea_events.txt and the other WHG_sea_* files). Triggers in
# common/scripted_triggers/WHG_sea_triggers.txt.
#
# Theatre flags on a captain:
#   sea_home_<theatre>  the sea off the captain's seat (sea_set_home_theatre_effect)
#   sea_in_<theatre>    where the voyage or encounter is now
# Theatres: claws, western, tilean, araby, southern, lustria, naggaroth,
# ulthuan, east, chaos, skaven.
###############################################################################

# Clears the theatre the captain is in.
# Scope: the captain.
sea_clear_theatre_effect = {
	clr_character_flag = sea_in_claws
	clr_character_flag = sea_in_western
	clr_character_flag = sea_in_tilean
	clr_character_flag = sea_in_araby
	clr_character_flag = sea_in_southern
	clr_character_flag = sea_in_lustria
	clr_character_flag = sea_in_naggaroth
	clr_character_flag = sea_in_ulthuan
	clr_character_flag = sea_in_east
	clr_character_flag = sea_in_chaos
	clr_character_flag = sea_in_skaven
}

# Finds the sea off the captain's seat (or the liege's, for a courtier) and
# flags it as home. A seat on two seas takes the first in the order below.
# Without a port, the home sea is the nearest theatre of the region.
# Scope: the captain.
sea_set_home_theatre_effect = {
	clr_character_flag = sea_home_claws
	clr_character_flag = sea_home_western
	clr_character_flag = sea_home_tilean
	clr_character_flag = sea_home_araby
	clr_character_flag = sea_home_southern
	clr_character_flag = sea_home_lustria
	clr_character_flag = sea_home_naggaroth
	clr_character_flag = sea_home_ulthuan
	clr_character_flag = sea_home_east
	clr_character_flag = sea_home_chaos
	clr_character_flag = sea_home_skaven
	if = {
		limit = { is_landed = yes }
		capital_scope = { save_event_target_as = sea_home_seat }
	}
	else_if = {
		limit = { liege = { is_landed = yes } }
		liege = { capital_scope = { save_event_target_as = sea_home_seat } }
	}
	else = {
		location = { save_event_target_as = sea_home_seat }
	}
	event_target:sea_home_seat = {
		if = {
			limit = { any_neighbor_province = { is_land = no sea_zone_claws_trigger = yes } }
			ROOT = { set_character_flag = sea_home_claws }
		}
		else_if = {
			limit = { any_neighbor_province = { is_land = no sea_zone_western_trigger = yes } }
			ROOT = { set_character_flag = sea_home_western }
		}
		else_if = {
			limit = { any_neighbor_province = { is_land = no sea_zone_tilean_trigger = yes } }
			ROOT = { set_character_flag = sea_home_tilean }
		}
		else_if = {
			limit = { any_neighbor_province = { is_land = no sea_zone_araby_trigger = yes } }
			ROOT = { set_character_flag = sea_home_araby }
		}
		else_if = {
			limit = { any_neighbor_province = { is_land = no sea_zone_southern_trigger = yes } }
			ROOT = { set_character_flag = sea_home_southern }
		}
		else_if = {
			limit = { any_neighbor_province = { is_land = no sea_zone_lustria_trigger = yes } }
			ROOT = { set_character_flag = sea_home_lustria }
		}
		else_if = {
			limit = { any_neighbor_province = { is_land = no sea_zone_naggaroth_trigger = yes } }
			ROOT = { set_character_flag = sea_home_naggaroth }
		}
		else_if = {
			limit = { any_neighbor_province = { is_land = no sea_zone_ulthuan_trigger = yes } }
			ROOT = { set_character_flag = sea_home_ulthuan }
		}
		else_if = {
			limit = { any_neighbor_province = { is_land = no sea_zone_east_trigger = yes } }
			ROOT = { set_character_flag = sea_home_east }
		}
		else_if = {
			limit = { any_neighbor_province = { is_land = no sea_zone_chaos_trigger = yes } }
			ROOT = { set_character_flag = sea_home_chaos }
		}
		else_if = {
			limit = { any_neighbor_province = { is_land = no sea_zone_skaven_trigger = yes } }
			ROOT = { set_character_flag = sea_home_skaven }
		}
		else_if = {
			limit = { OR = { region = world_norsca region = world_old_world_kislev region = world_old_world_empire } }
			ROOT = { set_character_flag = sea_home_claws }
		}
		else_if = {
			limit = { OR = { region = world_old_world_bretonnia region = world_albion region = world_athel_loren } }
			ROOT = { set_character_flag = sea_home_western }
		}
		else_if = {
			limit = { region = world_greater_araby_araby }
			ROOT = { set_character_flag = sea_home_araby }
		}
		else_if = {
			limit = { OR = { region = world_greater_araby_southlands region = world_eastern_southlands region = world_nehekhara } }
			ROOT = { set_character_flag = sea_home_southern }
		}
		else_if = {
			limit = { region = world_new_world_lustria }
			ROOT = { set_character_flag = sea_home_lustria }
		}
		else_if = {
			limit = { region = world_new_world }
			ROOT = { set_character_flag = sea_home_naggaroth }
		}
		else_if = {
			limit = { region = world_ulthuan }
			ROOT = { set_character_flag = sea_home_ulthuan }
		}
		else_if = {
			limit = { OR = { region = world_dragon_isles region = world_darklands } }
			ROOT = { set_character_flag = sea_home_east }
		}
		else_if = {
			limit = { region = world_chaos_wastes }
			ROOT = { set_character_flag = sea_home_chaos }
		}
		else_if = {
			limit = { OR = { region = world_skavenblight region = world_inner_burrows region = world_outer_burrows region = world_surface_burrows } }
			ROOT = { set_character_flag = sea_home_skaven }
		}
		else = {
			ROOT = { set_character_flag = sea_home_tilean }
		}
	}
}

# Flags the theatre of the sea zone the captain is in now (a real voyage).
# Off the sea (a river, a lake, the land), nothing is set.
# Scope: the captain.
sea_set_theatre_from_location_effect = {
	sea_clear_theatre_effect = yes

###############################################################################
# Manann's mood (see WHG_sea_triggers.txt). The variable stays between -5
# and 5. The steps say which way the sea leans; the tooltips name the power
# the captain's people pray to (GetSeaPower).
# Scope: the captain.
###############################################################################

manann_mood_up_effect = {
	custom_tooltip = { text = manann_mood_up_tt }
	hidden_effect = {
		change_variable = { which = manann_mood value = 1 }
		if = {
			limit = { check_variable = { which = manann_mood value = 6 } }
			set_variable = { which = manann_mood value = 5 }
		}
	}
}

manann_mood_down_effect = {
	custom_tooltip = { text = manann_mood_down_tt }
	hidden_effect = {
		change_variable = { which = manann_mood value = -1 }
		if = {
			limit = { NOT = { check_variable = { which = manann_mood value = -5 } } }
			set_variable = { which = manann_mood value = -5 }
		}
	}
}

manann_mood_curse_effect = {
	custom_tooltip = { text = manann_mood_curse_tt }
	hidden_effect = {
		change_variable = { which = manann_mood value = -2 }
		if = {
			limit = { NOT = { check_variable = { which = manann_mood value = -5 } } }
			set_variable = { which = manann_mood value = -5 }
		}
	}
}

# After a voyage the sea forgets a little: one step back towards nothing.
manann_mood_settle_effect = {
	if = {
		limit = { check_variable = { which = manann_mood value = 1 } }
		change_variable = { which = manann_mood value = -1 }
	}
	else_if = {
		limit = { NOT = { check_variable = { which = manann_mood value = 0 } } }
		change_variable = { which = manann_mood value = 1 }
	}
}

###############################################################################
# The named sea lords of 2010
# Run once at the start of a game (seas.1). Only lords our history has
# alive and in place in 2010 are named: Luthor Harkon of the Vampire Coast,
# Lokhir Fellheart (raiding in Lustria from Tohuate), Duriath Helbane of
# Khand, and the Norse jarl of Skeggi. They are found by the titles they
# hold at the start, so a different holder in a later save is never taken
# for them. Encounters check sea_named_lord_can_meet_trigger before naming
# them, so a lord who dies in play is simply never met again.
# Scope: any.
###############################################################################

sea_find_named_lords_effect = {
	k_vampire_coast = {
		holder_scope = {
			limit = { is_vampire_trigger = yes }
			set_character_flag = sea_named_harkon
			save_global_event_target_as = sea_lord_harkon
		}
	}
	c_tohuate = {
		holder_scope = {
			limit = { religion_group = cytharai_group }
			set_character_flag = sea_named_lokhir
			save_global_event_target_as = sea_lord_lokhir
		}
	}
	d_khand = {
		holder_scope = {
			limit = { religion_group = cytharai_group }
			set_character_flag = sea_named_helbane
			save_global_event_target_as = sea_lord_helbane
		}
	}
	d_skeggi = {
		holder_scope = {
			limit = { culture_group = norscan_group }
			set_character_flag = sea_named_skeggi
			save_global_event_target_as = sea_lord_skeggi
		}
	}
}
