# Finds a free island (castaway_island_trigger), flags it and notes its
# waters on ROOT. Callers check any_province first.
# Scope: ROOT, the castaway.
# Used by: WHG_castaway_events
castaway_pick_island_effect = {
	random_province = {
		limit = { castaway_island_trigger = yes }
		set_province_flag = castaway_island
		save_event_target_as = castaway_island
		if = {
			limit = { OR = { province_id = 1199 province_id = 1200 } }
			ROOT = { set_character_flag = castaway_north }
		}
		else_if = {
			limit = { OR = { province_id = 2479 province_id = 2520 } }
			ROOT = { set_character_flag = castaway_albion }
		}
		else_if = {
			limit = { OR = { province_id = 1781 province_id = 1764 province_id = 1765 } }
			ROOT = { set_character_flag = castaway_lustria }
		}
		else_if = {
			limit = { province_id = 1536 }
			ROOT = { set_character_flag = castaway_ulthuan }
		}
		else = {
			ROOT = { set_character_flag = castaway_south }
		}
	}
}

# Washed ashore, presumed drowned: every title at home goes to the heir as
# on a death (abdicate), and on the same day the castaway takes the island,
# so a player is never left unlanded. The island's count is the hall
# monster's first victim: he dies, or flees to his liege's court. The
# island stays in its liege's realm.
# Scope: ROOT, the castaway.
# Used by: WHG_castaway_events
castaway_wash_ashore_effect = {
	capital_scope = { set_province_flag = castaway_old_seat }
	any_demesne_title = { set_title_flag = castaway_old_title }
	any_spouse = { set_character_flag = castaway_spouse }
	current_heir = {
		set_character_flag = castaway_heir
		save_event_target_as = castaway_heir
	}
	random_province = {
		limit = { has_province_flag = castaway_island }
		save_event_target_as = castaway_island
		owner = {
			save_event_target_as = castaway_island_count
			if = {
				limit = { independent = no }
				liege = { save_event_target_as = castaway_island_liege }
				ROOT = { set_character_flag = castaway_had_liege }
			}
		}
	}
	abdicate = yes
	event_target:castaway_island = {
		county = { ROOT = { grant_title = PREV } }
		any_province_holding = {
			limit = { holder_scope = { character = event_target:castaway_island_count } }
			ROOT = { grant_title = PREV }
		}
	}
	if = {
		limit = {
			has_character_flag = castaway_had_liege
			event_target:castaway_island_liege = { character = ROOT }
		}
		# The island was the castaway's own vassal: it now answers to the heir.
		set_defacto_liege = event_target:castaway_heir
	}
	else_if = {
		limit = { has_character_flag = castaway_had_liege }
		set_defacto_liege = event_target:castaway_island_liege
	}
	else = {
		set_defacto_liege = THIS
	}
	event_target:castaway_island_count = {
		if = {
			limit = { is_landed = no }
			random_list = {
				50 = {
					death = { death_reason = death_accident_generic_beast }
				}
				50 = {
					trigger = {
						ROOT = { has_character_flag = castaway_had_liege }
						event_target:castaway_island_liege = { NOT = { character = ROOT } }
					}
					move_character = event_target:castaway_island_liege
				}
			}
		}
	}
	add_character_modifier = { name = castaway_marooned duration = -1 }
}

# The companion who left the footprints: a native of the island's waters.

			religion = cadai

# The sorcerer of the far shore and his daughter: an exiled magister of
# the Celestial College, or off Ulthuan an exiled mage of Saphery. Saved as

# The magister teaches the castaway: a gifted pupil who is no mage yet
# becomes one in his lore; anyone else gains learning.
# Scope: the castaway. event_target:castaway_magister is the teacher.
# Used by: WHG_castaway_events
castaway_magister_tutor_effect = {
	if = {
		limit = {
			apprentice_has_gift_trigger = yes
			is_trained_mage_trigger = no
		}
		event_target:castaway_magister = { save_event_target_as = mage_tutor }
		apprentice_tutor_lore_effect = yes
		apprentice_new_mage_effect = yes
	}
	else = {
		apprentice_learning_large_effect = yes
	}
}

# Going home: pressed claims on every title the castaway held at home, now
# in the hands of the lord of the old seat.
# Scope: ROOT, the castaway. event_target:castaway_home_lord is that lord.
# Used by: WHG_castaway_events
castaway_claim_home_effect = {
	event_target:castaway_home_lord = {
		any_demesne_title = {
			limit = { has_title_flag = castaway_old_title }
			ROOT = { add_pressed_claim = PREV }
		}
	}
}

castaway_find_home_lord_effect = {

		limit = { has_province_flag = castaway_old_seat }
		owner = { save_event_target_as = castaway_home_lord }
	}
}

# The castaway's heir hands the old titles back: every flagged title, the
# vassals under them, and the heir comes to the castaway's court if left
# landless. The castaway then answers to the heir's old liege (or none).
# Scope: the lord of the old seat. FROM: the castaway.
# Used by: WHG_castaway_events
castaway_hand_back_effect = {
	if = {
		limit = { independent = no }
		liege = { save_event_target_as = castaway_home_liege }
		FROM = { set_character_flag = castaway_home_had_liege }
	}
	any_demesne_title = {
		limit = { has_title_flag = castaway_old_title }
		FROM = { grant_title = PREV }
	}
	any_vassal = {
		limit = { NOT = { character = FROM } }
		set_defacto_liege = FROM
	}
	FROM = {
		if = {
			limit = { has_character_flag = castaway_home_had_liege }
			set_defacto_liege = event_target:castaway_home_liege
		}
		else = {
			set_defacto_liege = THIS
		}
	}
	if = {
		limit = { is_landed = no }
		move_character = FROM
	}
}

# The next ship in the line: no ship -> the small flagship -> the flagship ->
# the enormous flagship. Resets the five-year refit clock.
# Scope: the captain.
# Used by: WHG_sea_decisions, WHG_sea_events
sea_refit_flagship_effect = {
	if = {
		limit = { has_artifact = regular_flagship }
		any_artifact = {
			limit = { artifact_type = regular_flagship }
			destroy_artifact = yes
		}
		add_artifact = enormous_flagship
	}
	else_if = {
		limit = { has_artifact = small_flagship }
		any_artifact = {
			limit = { artifact_type = small_flagship }
			destroy_artifact = yes
		}
		add_artifact = regular_flagship
	}
	else = {
		add_artifact = small_flagship
	}
	hidden_effect = {
		clr_character_flag = sea_refit
		set_character_flag = sea_refit
		if = {
			limit = { has_quest = quest_sea_flagship }
			character_event = { id = seaquest.65 }
		}
	}
}
