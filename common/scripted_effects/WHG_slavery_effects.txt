###############################################################################
# Slavery
# Used by events/WHG_slavery_events.txt, decisions/WHG_slavery_decisions.txt,
# the slave decisions in decisions/wh_general_decisions.txt and dawizharr.36.
#
# The stock is the character variable slaves, in lots (a column of a few
# hundred captives), at most 10. Call slave_stock_update_effect after every
# change: it caps the stock, shows it as the pens modifier and keeps the
# flag slave_master that lets the yearly pens event (slave.10) find the
# owner cheaply.
###############################################################################

# Scope: the slaver.
slave_stock_update_effect = {
	hidden_effect = {
		if = {
			limit = { check_variable = { which = slaves value > 10 } }
			set_variable = { which = slaves value = 10 }
		}
		if = {
			limit = { NOT = { check_variable = { which = slaves value = 0 } } }
			set_variable = { which = slaves value = 0 }
		}
		remove_character_modifier = slave_pens_1
		remove_character_modifier = slave_pens_2
		remove_character_modifier = slave_pens_3
		if = {
			limit = { check_variable = { which = slaves value = 6 } }
			add_character_modifier = { name = slave_pens_3 duration = -1 }
			set_character_flag = slave_master
		}
		else_if = {
			limit = { check_variable = { which = slaves value = 3 } }
			add_character_modifier = { name = slave_pens_2 duration = -1 }
			set_character_flag = slave_master
		}
		else_if = {
			limit = { check_variable = { which = slaves value = 1 } }
			add_character_modifier = { name = slave_pens_1 duration = -1 }
			set_character_flag = slave_master
		}
		else = {
			clr_character_flag = slave_master
		}
	}
}

# One lot of slaves more. Scope: the slaver.
slave_gain_lot_effect = {
	custom_tooltip = { text = slave_gain_lot_tt }
	hidden_effect = { change_variable = { which = slaves value = 1 } }
	slave_stock_update_effect = yes
}

# Two lots more. Scope: the slaver.
slave_gain_two_lots_effect = {
	custom_tooltip = { text = slave_gain_two_lots_tt }
	hidden_effect = { change_variable = { which = slaves value = 2 } }
	slave_stock_update_effect = yes
}

# One lot fewer. Scope: the slaver.
slave_lose_lot_effect = {
	custom_tooltip = { text = slave_lose_lot_tt }
	hidden_effect = { subtract_variable = { which = slaves value = 1 } }
	slave_stock_update_effect = yes
}

# Two lots fewer. Scope: the slaver.
slave_lose_two_lots_effect = {
	custom_tooltip = { text = slave_lose_two_lots_tt }
	hidden_effect = { subtract_variable = { which = slaves value = 2 } }
	slave_stock_update_effect = yes
}

# The price of one lot at the market of slave.2, by the market's mood:
# 35 gold in a glut, 50 as usual, 75 when the buyers are hungry.
# Scope: the slaver.
slave_sell_lot_effect = {
	if = {
		limit = { has_character_flag = slave_market_glut }
		wealth = 35
	}
	else_if = {
		limit = { has_character_flag = slave_market_hungry }
		wealth = 75
	}
	else = { wealth = 50 }
	hidden_effect = { subtract_variable = { which = slaves value = 1 } }
}

# Clears the market's mood after slave.2. Scope: the slaver.
slave_market_close_effect = {
	clr_character_flag = slave_market_glut
	clr_character_flag = slave_market_hungry
	slave_stock_update_effect = yes
}
