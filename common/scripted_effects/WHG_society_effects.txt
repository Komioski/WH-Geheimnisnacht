# Scope: an AI who has just joined a society (society_ai_join_effect).
# Gives a young society some influence and a grandmaster if it has none.
society_ai_join_settle_effect = {
	random_list = { # Influence for balance: AI-filled societies would be weak otherwise
		75 = {
			modifier = {
				factor = 0
				society_influence >= 30
			}
			add_mission_succeed_influence_effect = yes
		}
		25 = { }
	}
	if = {
		limit = {
			society = {
				NOT = {
					any_society_member = { is_society_grandmaster = yes }
				}
			}
		}
		set_society_grandmaster = yes
	}
}

# Scope: a society member whose secret is out (society.65). Branded as a
# cultist or an apostate, and cast out of a faith that excommunicates.
society_exposed_effect = {
	if = {
		limit = {
			is_devil_worshiper_trigger = yes
			NOT = { has_character_modifier = known_satanist }
		}
		add_character_modifier = {
			name = known_satanist
			duration = -1
		}
	}
	else_if = {
		limit = {
			is_devil_worshiper_trigger = no
			NOT = { has_character_modifier = accused_apostate }
		}
		add_character_modifier = {
			name = accused_apostate
			duration = -1
		}
	}
	if = {
		limit = { society_faith_excommunicates_trigger = yes }
		add_trait = excommunicated
	}
}

# The Huskarl of the Norscan Freigattur (freigattur.40). A shieldmaiden
# one time in three; 20, 30 or 40 years old (30 most often). Saves the
# new warrior as event_target:freig_huskarl.
freigattur_create_huskarl_effect = {
	random_list = {
		8 = { create_character = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = yes age = 20 } }
		17 = { create_character = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = yes age = 30 } }
		8 = { create_character = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = yes age = 40 } }
		17 = { create_character = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = no age = 20 } }
		33 = { create_character = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = no age = 30 } }
		17 = { create_character = { random_traits = yes dynasty = none religion = ROOT culture = ROOT female = no age = 40 } }
	}
	new_character = {
		character_event = { id = aiclean.31 } # Racial trait
		save_event_target_as = freig_huskarl
	}
}
