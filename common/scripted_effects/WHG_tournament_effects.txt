# Effects for events/WHG_tournament_events.txt (the Bretonnian tourney)
#
# event_target:tourney_host is always the host. A tilt runs with ROOT as one
# rider and event_target:tourney_foe as the other.

###############################################################################
# Housekeeping
###############################################################################

# Scope: a guest. Leaves the tourney: clears every tourney flag and the
# opinion links to the host (event_target:tourney_host). Records of past
# placings (grand_tourny_1st and the rest) stay.
tourney_clear_guest_effect = {
	clr_character_flag = tourney_rider
	clr_character_flag = tourney_watcher
	clr_character_flag = tourney_out
	clr_character_flag = tourney_paired
	clr_character_flag = tourney_tilting
	clr_character_flag = tourney_bribed
	clr_character_flag = tourney_blessed
	clr_character_flag = tourney_first
	clr_character_flag = tourney_second
	clr_character_flag = tourney_third
	clr_character_flag = tourney_happening
	clr_character_flag = tourney_fate_unhorsed
	clr_character_flag = tourney_fate_wounded
	clr_character_flag = tourney_fate_crippled
	clr_character_flag = tourney_fate_killed
	clr_character_flag = tourney_fate_yielded
	clr_character_flag = tourney_fate_threw
	tourney_clear_lesson_effect = yes
	event_target:tourney_host = {
		remove_opinion = { modifier = opinion_attending_tournament who = PREV }
		remove_opinion = { modifier = opinion_spectating_tournament who = PREV }
		reverse_remove_opinion = { modifier = opinion_attending_tournament who = PREV }
		reverse_remove_opinion = { modifier = opinion_spectating_tournament who = PREV }
	}
}

# Scope: the host. A hedge knight rides in to fill the lists: a Knight of the
# Realm of the host's people and faith, sound of body.
tourney_hedge_knight_effect = {
	capital_scope = {
		ROOT = {
			create_character = {
				random_traits = yes
				dynasty = culture
				culture = PREV
				religion = ROOT
				female = no
				age = 24
				trait = tough_soldier
			}
			new_character = {
				character_event = { id = aiclean.31 } # Racial trait
				join_society = { society = knights_of_bretonnia rank = 2 }
				give_minor_title = title_sir_bretonnian
				remove_trait = weak
				remove_trait = imbecile
				remove_trait = clubfooted
				remove_trait = inbred
				tourney_join_lists_effect = yes
			}
		}
	}
}

# Scope: a new rider. Enters the lists of event_target:tourney_host.
tourney_join_lists_effect = {
	set_character_flag = tourney_rider
	opinion = { modifier = opinion_attending_tournament who = event_target:tourney_host years = 1 }
	reverse_opinion = { modifier = opinion_attending_tournament who = event_target:tourney_host years = 1 }
	character_event = { id = tourney.14 days = 200 } # Safety catch
}

# Scope: a rider. Leaves the lists for good (unhorsed, hurt, thrown out or
# yielded). At the semi-finals or the final, earns third or second place.
tourney_leave_lists_effect = {
	set_character_flag = tourney_out
	clr_character_flag = tourney_tilting
	if = {
		limit = { event_target:tourney_host = { has_character_flag = tourney_final_round } }
		set_character_flag = tourney_second
	}
	else_if = {
		limit = { event_target:tourney_host = { has_character_flag = tourney_semi_round } }
		set_character_flag = tourney_third
	}
}

# Scope: a rider (ROOT). event_target:tourney_foe is the other. The two ride
# at each other until one is unhorsed. Skill at arms decides most tilts; a
# rider who took a purse to lose always falls, and the Lady's blessing or a
# brave heart help. The loser may be hurt, crippled or, one time in a
# hundred, killed. Players are told the result (tourney.27, tourney.28).
tourney_tilt_effect = {
	random_list = {
		50 = {
			modifier = {
				factor = 0
				has_character_flag = tourney_bribed
			}
			modifier = {
				factor = 1000
				event_target:tourney_foe = { has_character_flag = tourney_bribed }
				NOT = { has_character_flag = tourney_bribed }
			}
			modifier = {
				factor = 1.5
				combat_rating_diff = { who = event_target:tourney_foe value = 10 }
			}
			modifier = {
				factor = 1.5
				combat_rating_diff = { who = event_target:tourney_foe value = 25 }
			}
			modifier = {
				factor = 2
				combat_rating_diff = { who = event_target:tourney_foe value = 50 }
			}
			modifier = {
				factor = 2
				combat_rating_diff = { who = event_target:tourney_foe value = 100 }
			}
			modifier = {
				factor = 1.5
				has_character_flag = tourney_blessed
			}
			modifier = {
				factor = 1.2
				trait = brave
			}
			modifier = {
				factor = 0.8
				trait = craven
			}
			modifier = {
				factor = 0.7
				OR = {
					trait = wounded
					is_ill = yes
				}
			}
			save_event_target_as = tourney_tilt_winner
			event_target:tourney_foe = { save_event_target_as = tourney_tilt_loser }
		}
		50 = {
			modifier = {
				factor = 0
				event_target:tourney_foe = { has_character_flag = tourney_bribed }
			}
			modifier = {
				factor = 1000
				has_character_flag = tourney_bribed
				event_target:tourney_foe = { NOT = { has_character_flag = tourney_bribed } }
			}
			modifier = {
				factor = 1.5
				event_target:tourney_foe = { combat_rating_diff = { who = ROOT value = 10 } }
			}
			modifier = {
				factor = 1.5
				event_target:tourney_foe = { combat_rating_diff = { who = ROOT value = 25 } }
			}
			modifier = {
				factor = 2
				event_target:tourney_foe = { combat_rating_diff = { who = ROOT value = 50 } }
			}
			modifier = {
				factor = 2
				event_target:tourney_foe = { combat_rating_diff = { who = ROOT value = 100 } }
			}
			modifier = {
				factor = 1.5
				event_target:tourney_foe = { has_character_flag = tourney_blessed }
			}
			modifier = {
				factor = 1.2
				event_target:tourney_foe = { trait = brave }
			}
			modifier = {
				factor = 0.8
				event_target:tourney_foe = { trait = craven }
			}
			modifier = {
				factor = 0.7
				event_target:tourney_foe = {
					OR = {
						trait = wounded
						is_ill = yes
					}
				}
			}
			event_target:tourney_foe = { save_event_target_as = tourney_tilt_winner }
			save_event_target_as = tourney_tilt_loser
		}
	}
	event_target:tourney_tilt_winner = {
		clr_character_flag = tourney_tilting
		clr_character_flag = tourney_blessed
		if = {
			limit = { event_target:tourney_tilt_loser = { higher_tier_than = COUNT } }
			prestige = 10
		}
		else = {
			prestige = 5
		}
		small_chance_for_one_duel_experience_modifier = yes
	}
	event_target:tourney_tilt_loser = {
		clr_character_flag = tourney_blessed
		prestige = -10
		tourney_leave_lists_effect = yes
		tourney_fall_effect = yes
	}
	event_target:tourney_tilt_winner = {
		if = {
			limit = { ai = no }
			character_event = { id = tourney.27 }
		}
	}
	event_target:tourney_tilt_loser = {
		if = {
			limit = {
				ai = no
				is_alive = yes
			}
			character_event = { id = tourney.28 }
		}
	}
}

# Scope: the loser of a tilt. How the fall goes: most are merely unhorsed,
# some wounded, a few crippled and one in a hundred killed. A rider who took
# a purse falls on purpose and comes to no harm. The host and the players at
# the tourney hear of a crippling or a death (tourney.29).
tourney_fall_effect = {
	if = {
		limit = { has_character_flag = tourney_bribed }
		set_character_flag = tourney_fate_threw
	}
	else = {
		random_list = {
			88 = {
				set_character_flag = tourney_fate_unhorsed
			}
			8 = {
				set_character_flag = tourney_fate_wounded
				if = {
					limit = { NOT = { trait = wounded } }
					add_trait = wounded
				}
			}
			3 = {
				set_character_flag = tourney_fate_crippled
				random_list = {
					25 = {
						modifier = { factor = 0 trait = one_eyed }
						add_trait = one_eyed
					}
					25 = {
						modifier = { factor = 0 trait = one_legged }
						add_trait = one_legged
					}
					25 = {
						modifier = { factor = 0 trait = disfigured }
						add_trait = disfigured
					}
					25 = {
						modifier = { factor = 0 trait = maimed }
						add_trait = maimed
					}
				}
				if = {
					limit = { NOT = { trait = wounded } }
					add_trait = wounded
				}
			}
			1 = {
				set_character_flag = tourney_fate_killed
			}
		}
	}
	if = {
		limit = {
			OR = {
				has_character_flag = tourney_fate_crippled
				has_character_flag = tourney_fate_killed
			}
		}
		event_target:tourney_host = {
			if = {
				limit = { NOT = { has_character_flag = tourney_no_msg_fallen } }
				character_event = { id = tourney.29 }
			}
			any_opinion_modifier_target = {
				limit = {
					ai = no
					NOT = { has_character_flag = tourney_no_msg_fallen }
					NOT = { character = event_target:tourney_tilt_winner }
					NOT = { character = event_target:tourney_tilt_loser }
					OR = {
						has_opinion_modifier = { who = PREV modifier = opinion_attending_tournament }
						has_opinion_modifier = { who = PREV modifier = opinion_spectating_tournament }
					}
				}
				character_event = { id = tourney.29 }
			}
		}
	}
	if = {
		limit = { has_character_flag = tourney_fate_killed }
		if = {
			limit = { is_invincible_trigger = yes }
			antideath_effect = yes
		}
		else = {
			death = {
				death_reason = death_duel
				killer = event_target:tourney_tilt_winner
			}
		}
	}
}

		}
	}
	else_if = {
		limit = {
			has_character_flag = tourney_third
			event_target:tourney_host = { has_character_flag = tourney_local }
		}
		set_character_flag = small_tourny_3rd
		if = {
			limit = { has_character_modifier = small_tourny_3rd_4 }
			remove_character_modifier = small_tourny_3rd_4

			remove_character_modifier = small_tourny_3rd_3
			add_character_modifier = { name = small_tourny_3rd_4 duration = -1 }

		else_if = {
			limit = { has_character_modifier = small_tourny_3rd_2 }
			remove_character_modifier = small_tourny_3rd_2
			add_character_modifier = { name = small_tourny_3rd_3 duration = -1 }
		}
		else_if = {
			limit = { has_character_modifier = small_tourny_3rd_1 }
			remove_character_modifier = small_tourny_3rd_1

# Scope: the champion of a grand or regional tourney. Once in a life, may
# ride home on a Bretonnian charger from the host's stables or carry off the
# Champion's Lance: one in two at a grand tourney, one in ten at a regional.
tourney_prize_effect = {
	if = {
		limit = { tourney_can_win_prize_trigger = yes }
		random = {
			chance = 10
			modifier = {
				factor = 5
				event_target:tourney_host = { has_character_flag = tourney_grand }
			}
			set_character_flag = tourney_prize_won
			if = {
				limit = {
					NOT = { any_artifact = { has_artifact_flag = mount } }
				}
				add_artifact = bretonnian_charger
			}
			else_if = {
				limit = {
					NOT = { any_artifact = { has_artifact_flag = tourney_lance } }
				}
				add_artifact = tourney_champion_lance
			}
		}
	}
}

###############################################################################

# Scope: a character. ROOT is the host. Sends a letter to ride (tourney.2) to
# a fit rider good enough for this tourney's size, or else, to a lord, a
# letter to watch (tourney.3).
tourney_send_invitation_effect = {
	if = {
		limit = {
			NOT = { character = ROOT }
			tourney_can_ride_trigger = yes
			OR = {
				AND = {
					ROOT = { has_character_flag = tourney_local }
					tourney_local_rider_trigger = yes
				}
				tourney_worthy_rider_trigger = yes
			}
		}
		letter_event = { id = tourney.2 days = 2 random = 5 }
	}
	else_if = {
		limit = { tourney_can_watch_trigger = yes }
		letter_event = { id = tourney.3 days = 7 random = 5 }
	}
}

# Scope: the host. Invites the right people for the tourney's size:
# - local: the host's own realm and court, the host's liege and the liege's
#   other vassals;
# - regional: the realm of the host's king-tier liege (a Bretonnian
#   dukedom), or the host's own realm if the host is of that rank or higher;
# - grand: the whole realm of the top liege.
tourney_invite_effect = {
	if = {
		limit = { has_character_flag = tourney_grand }
		top_liege = {
			if = {
				limit = { NOT = { character = ROOT } }
				tourney_send_invitation_effect = yes
			}
			any_realm_character = { tourney_send_invitation_effect = yes }
		}
	}
	else_if = {
		limit = {
			has_character_flag = tourney_regional
			lower_tier_than = KING
			any_liege = { tier = KING }
		}
		any_liege = {
			limit = { tier = KING }
			tourney_send_invitation_effect = yes
			any_realm_character = { tourney_send_invitation_effect = yes }
		}
	}
	else = {
		any_realm_character = { tourney_send_invitation_effect = yes }
		liege = {
			if = {
				limit = { NOT = { character = ROOT } }
				tourney_send_invitation_effect = yes
				any_vassal = {
					limit = { NOT = { character = ROOT } }
					tourney_send_invitation_effect = yes
				}
			}
		}
	}
}

# Scope: the champion. A wife or betrothed passed over for
# event_target:tourney_queen takes it as an insult.
tourney_snub_effect = {
	any_spouse = {
		limit = {
			is_woman_trigger = yes
			NOT = { character = event_target:tourney_queen }
		}
		opinion = { modifier = insulted who = ROOT multiplier = 1.5 }
	}
	betrothed = {
		if = {
			limit = {
				is_woman_trigger = yes
				NOT = { character = event_target:tourney_queen }
			}
			opinion = { modifier = insulted who = ROOT multiplier = 1.5 }
		}
	}
}
