# Put a vampire into the society of their bloodline. Vampire pirates are left
# alone. Only Lahmians and von Carsteins can hide, so a hidden vampire of any
# other line is made open first. A thrall turned into another line's vampire
# moves to that line's society at rank 1.
# Scope: the vampire.
# Used by: WHG_undead_effects, WHG_vampire_events, wh_vampire_decisions
vamp_join_bloodline_effect = {
	if = {
		limit = {
			is_vampire_trigger = yes
			vamp_pirate_trigger = no
		}
		if = {
			limit = { trait = vampire_blood_hidden }
			remove_trait = vampire_blood_hidden
			add_trait = vampire_blood_visible
		}
		if = {
			limit = { trait = vampire_necrarch_hidden }
			remove_trait = vampire_necrarch_hidden
			add_trait = vampire_necrarch_visible
		}
		if = {
			limit = { trait = vampire_strigoi_hidden }
			remove_trait = vampire_strigoi_hidden
			add_trait = vampire_strigoi_visible
		}
		if = {
			limit = { trait = vampire_jade_hidden }
			remove_trait = vampire_jade_hidden
			add_trait = vampire_jade_visible
		}
		if = {
			limit = { trait = vampire_mahtmasi_hidden }
			remove_trait = vampire_mahtmasi_hidden
			add_trait = vampire_mahtmasi_visible
		}
		if = {
			limit = {
				vamp_carstein_trigger = yes
				NOT = { society_member_of = carstein_court }
			}
			if = {
				limit = { is_in_society = yes }
				leave_society = yes
			}
			join_society = carstein_court
		}
		else_if = {
			limit = {
				vamp_blood_trigger = yes
				NOT = { society_member_of = ordo_draconis }
			}
			if = {
				limit = { is_in_society = yes }
				leave_society = yes
			}
			join_society = ordo_draconis
		}
		else_if = {
			limit = {
				vamp_lahmian_trigger = yes
				NOT = { society_member_of = lahmian_sisterhood }
			}
			if = {
				limit = { is_in_society = yes }
				leave_society = yes
			}
			join_society = lahmian_sisterhood
		}
		else_if = {
			limit = {
				vamp_necrarch_trigger = yes
				NOT = { society_member_of = necrarch_brotherhood }
			}
			if = {
				limit = { is_in_society = yes }
				leave_society = yes
			}
			join_society = necrarch_brotherhood
		}
		else_if = {
			limit = {
				vamp_strigoi_trigger = yes
				NOT = { society_member_of = strigoi_brood }
			}
			if = {
				limit = { is_in_society = yes }
				leave_society = yes
			}
			join_society = strigoi_brood
		}
		else_if = {
			limit = {
				vamp_jade_trigger = yes
				NOT = { society_member_of = jade_coil }
			}
			if = {
				limit = { is_in_society = yes }
				leave_society = yes
			}
			join_society = jade_coil
		}
		else_if = {
			limit = {
				vamp_mahtmasi_trigger = yes
				NOT = { society_member_of = carrion_children }
			}
			if = {
				limit = { is_in_society = yes }
				leave_society = yes
			}
			join_society = carrion_children
		}
		if = {
			limit = { has_character_modifier = vamp_blood_thrall }
			remove_character_modifier = vamp_blood_thrall
		}
	}
}

# A Jade vampire drains one courtier of a raided court: 3 in 4 die (the
# invincible are spared by antideath_effect). A survivor keeps the
# vamp_jade_drained flag and cannot be drained again for a year; after that
# the flag counts as stale and is replaced.
# Scope: the raided court's holder. ROOT is the Jade raider.
# Used by: WHG_undead_effects
vamp_jade_drain_courtier_effect = {
	random_courtier = {
		limit = {
			is_abroad = no
			NOT = { religion_group = necromantic_group }
			OR = {
				NOT = { has_character_flag = vamp_jade_drained }
				had_character_flag = { flag = vamp_jade_drained days = 365 } # a year has passed
			}
		}
		clr_character_flag = vamp_jade_drained
		set_character_flag = vamp_jade_drained
		random = {
			chance = 75
			if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
			else = { death = { death_reason = death_murder killer = ROOT } }
		}
	}
}

# Works out the strength of the leader's host into the variable
# blood_host_strength (0 to about 35): rank, nightmare, martial skill,
# fighting skill, body, nerve, age and wounds. Knights who join a fight add 3
# each (vamp.312).
# Scope: the leader.
# Used by: WHG_vampire_events
vamp_bquest_strength_effect = {
	set_variable = { which = blood_host_strength value = 0 }
	if = { limit = { tier = EMPEROR } set_variable = { which = blood_host_strength value = 4 } }
	else_if = { limit = { tier = KING } set_variable = { which = blood_host_strength value = 3 } }
	else_if = { limit = { tier = DUKE } set_variable = { which = blood_host_strength value = 2 } }
	else_if = { limit = { tier = COUNT } set_variable = { which = blood_host_strength value = 1 } }
	if = {
		limit = { has_artifact = nightmare_mount }
		change_variable = { which = blood_host_strength value = 2 }
	}
	if = { limit = { martial = 10 } change_variable = { which = blood_host_strength value = 1 } }
	if = { limit = { martial = 15 } change_variable = { which = blood_host_strength value = 1 } }
	if = { limit = { martial = 20 } change_variable = { which = blood_host_strength value = 1 } }
	if = { limit = { martial = 25 } change_variable = { which = blood_host_strength value = 1 } }
	if = {
		limit = { combat_rating = 2 }
		change_variable = { which = blood_host_strength value = 1 }
	}
	if = {
		limit = { combat_rating = 4 }
		change_variable = { which = blood_host_strength value = 2 }
	}
	if = {
		limit = { combat_rating = 6 }
		change_variable = { which = blood_host_strength value = 2 }
	}
	if = {
		limit = { combat_rating = 8 }
		change_variable = { which = blood_host_strength value = 2 }
	}
	if = {
		limit = { combat_rating = 10 }
		change_variable = { which = blood_host_strength value = 2 }
	}
	if = { limit = { trait = strong } change_variable = { which = blood_host_strength value = 3 } }
	else_if = {
		limit = { trait = tough }
		change_variable = { which = blood_host_strength value = 2 }
	}
	else_if = {
		limit = { trait = hardy }
		change_variable = { which = blood_host_strength value = 1 }
	}
	if = { limit = { trait = weak } change_variable = { which = blood_host_strength value = -3 } }
	else_if = {
		limit = { trait = fragile }
		change_variable = { which = blood_host_strength value = -2 }
	}
	else_if = {
		limit = { trait = soft }
		change_variable = { which = blood_host_strength value = -1 }
	}
	if = { limit = { trait = brave } change_variable = { which = blood_host_strength value = 1 } }
	if = { limit = { trait = craven } change_variable = { which = blood_host_strength value = -3 } }
	if = { # Centuries of practice
		limit = { age = 75 }
		change_variable = { which = blood_host_strength value = 1 }
	}
	if = { limit = { age = 100 } change_variable = { which = blood_host_strength value = 1 } }
	if = { limit = { age = 150 } change_variable = { which = blood_host_strength value = 2 } }
	if = { limit = { age = 200 } change_variable = { which = blood_host_strength value = 2 } }
	if = { limit = { age = 400 } change_variable = { which = blood_host_strength value = 2 } }
	if = {
		limit = { trait = wounded }
		change_variable = { which = blood_host_strength value = -2 }
	}
	if = { limit = { trait = maimed } change_variable = { which = blood_host_strength value = -5 } }
	if = {
		limit = { trait = blinded }
		change_variable = { which = blood_host_strength value = -10 }
	}
	if = { limit = { trait = infirm } change_variable = { which = blood_host_strength value = -4 } }
	if = {
		limit = { OR = { trait = incapable trait = resting } }
		change_variable = { which = blood_host_strength value = -15 }
	}
}

				factor = 1.25
				FROM = { check_variable = { which = blood_host_strength value = 20 } }
			}

				factor = 1.25

# Vlad declares the vlad_subjugation war for the kingdom he marches on: on
# its holder's top liege, or, when nobody holds it, on the top liege of the
# owner of its capital (vamp.601, vamp.606).

# event_target:vamp_vlad_goal_seat its capital province.

	if = {
		limit = { event_target:vamp_vlad_goal = { has_holder = yes } }
		event_target:vamp_vlad_goal = {
			holder_scope = { top_liege = { save_event_target_as = vamp_vlad_foe } }
		}
	}
	else = {
		event_target:vamp_vlad_goal_seat = {
			owner = { top_liege = { save_event_target_as = vamp_vlad_foe } }
		}
	}
	if = {
		limit = { event_target:vamp_vlad_foe = { NOT = { same_realm = ROOT } } }
		unsafe_war = {
			target = event_target:vamp_vlad_foe
			casus_belli = vlad_subjugation
			thirdparty_title = event_target:vamp_vlad_goal
		}
	}
}

# A fight decided by prowess: sets vampsoc_won on the character on a win.
# Against event_target:vampsoc_foe when it is set, else against an unnamed
# champion of fair skill.
# Scope: the fighter.
# Used by: WHG_vampire_events
vampsoc_fight_effect = {
	clr_character_flag = vampsoc_won
	if = {
		limit = { event_target:vampsoc_foe = { always = yes } }
		random_list = {
			50 = {
				modifier = { factor = 1.5 combat_rating_diff = { character = event_target:vampsoc_foe value >= 10 } }
				modifier = { factor = 1.5 combat_rating_diff = { character = event_target:vampsoc_foe value >= 25 } }
				modifier = { factor = 2 combat_rating_diff = { character = event_target:vampsoc_foe value >= 50 } }
				set_character_flag = vampsoc_won
			}
			50 = {
				modifier = { factor = 1.5 event_target:vampsoc_foe = { combat_rating_diff = { character = PREV value >= 10 } } }
				modifier = { factor = 1.5 event_target:vampsoc_foe = { combat_rating_diff = { character = PREV value >= 25 } } }
				modifier = { factor = 2 event_target:vampsoc_foe = { combat_rating_diff = { character = PREV value >= 50 } } }
			}
		}
	}
	else = {
		random_list = {
			50 = {
				modifier = { factor = 1.5 combat_rating >= 20 }
				modifier = { factor = 1.5 combat_rating >= 40 }
				modifier = { factor = 1.5 combat_rating >= 60 }
				modifier = { factor = 1.5 martial >= 15 }
				set_character_flag = vampsoc_won
			}
			40 = {
				modifier = { factor = 0.5 trait = vampire_blood_visible }
			}
		}
	}
}
