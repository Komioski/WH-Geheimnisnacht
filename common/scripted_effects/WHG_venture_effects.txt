# Effects of the Venturers (events/WHG_venture_events.txt,
# decisions/WHG_venture_decisions.txt). Triggers in
# common/scripted_triggers/WHG_venture_triggers.txt.
#
# An expedition keeps its state on the Venturer: the destination flag
# (venture_dest_coast/temple/southlands/naggaroth), the stake flag
# (venture_stake_small/medium/large), venture_leading or a captain saved as
# event_target:venture_captain (flag venture_captain on him), and the
# variable venture_odds that the stake, guides, charts and the choices on
# the way push up or down. venture.4 rolls the outcome from it.
###############################################################################

# Scope: the sponsor. Finds the captain of a sponsored expedition: a
# courtier who is an adventurer or a fair soldier, else a hired one of the
# sponsor's people. Saves event_target:venture_captain.
venture_pick_captain_effect = {
	if = {
		limit = {
			any_courtier = {
				is_adult = yes
				is_ruler = no
				prisoner = no
				is_inaccessible_trigger = no
				NOT = { is_close_relative = ROOT }
				OR = {
					trait = adventurer
					martial = 10
				}
			}
		}
		random_courtier = {
			limit = {
				is_adult = yes
				is_ruler = no
				prisoner = no
				is_inaccessible_trigger = no
				NOT = { is_close_relative = ROOT }
				OR = {
					trait = adventurer
					martial = 10
				}
			}
			preferred_limit = { trait = adventurer }
			preferred_limit = { martial = 14 }
			save_event_target_as = venture_captain
		}
	}
	else = {
		create_character = {
			random_traits = yes
			dynasty = random
			religion = ROOT
			culture = ROOT
			female = no
			age = 35
			attributes = { martial = 10 }
			trait = adventurer
		}
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			save_event_target_as = venture_captain
		}
	}
	event_target:venture_captain = { set_character_flag = venture_captain }
}

# Scope: the Venturer. Stakes, guides, charts and the leader set the odds
# before the ships sail.
venture_set_odds_effect = {
	set_variable = { which = venture_odds value = 0 }
	if = {
		limit = { has_character_flag = venture_stake_medium }
		change_variable = { which = venture_odds value = 2 }
	}
	else_if = {
		limit = { has_character_flag = venture_stake_large }
		change_variable = { which = venture_odds value = 4 }
	}
	if = {
		limit = { has_character_modifier = venture_guides_hired }
		change_variable = { which = venture_odds value = 2 }
		remove_character_modifier = venture_guides_hired
	}
	if = {
		limit = { has_character_modifier = venture_fresh_charts }
		change_variable = { which = venture_odds value = 2 }
	}
	if = {
		limit = { has_artifact = venture_navigators_compass }
		change_variable = { which = venture_odds value = 1 }
	}
	if = {
		limit = { has_character_modifier = venture_slann_treaty }
		change_variable = { which = venture_odds value = 1 }
	}
	if = { # A seasoned leader, in person or as captain
		limit = {
			OR = {
				AND = {
					venture_leading_trigger = yes
					OR = { trait = adventurer martial = 14 }
				}
				AND = {
					venture_leading_trigger = no
					event_target:venture_captain = { OR = { trait = adventurer martial = 14 } }
				}
			}
		}
		change_variable = { which = venture_odds value = 1 }
	}
}

# Scope: the Venturer. One step better or worse odds.
venture_odds_up_effect = {
	change_variable = { which = venture_odds value = 1 }
	custom_tooltip = { text = venture_odds_up_tt }
}
venture_odds_down_effect = {
	change_variable = { which = venture_odds value = -1 }
	custom_tooltip = { text = venture_odds_down_tt }
}

# Scope: the Venturer. A deed of exploration, counted for the Great Voyage.
venture_deed_effect = {
	hidden_effect = { change_variable = { which = venture_deeds value = 1 } }
	custom_tooltip = { text = venture_deed_tt }
}

# Scope: the Venturer. A good site for a colony is marked on the charts:
# a port province of the destination that a colonial charter can take.
venture_survey_site_effect = {
	if = {
		limit = { any_province = { venture_site_trigger = yes } }
		random_province = {
			limit = { venture_site_trigger = yes }
			preferred_limit = { owner = { is_tribal = yes } }
			preferred_limit = { owner = { lower_tier_than = DUKE } }
			set_province_flag = venture_surveyed
			save_event_target_as = venture_site
		}
		set_character_flag = venture_has_survey
		custom_tooltip = { text = venture_survey_tt }
	}
}

# Scope: the Venturer. A rare find from the jungle: once per lifetime.

		add_artifact = venture_jungle_idol
	}
}
venture_compass_effect = {

	}
}

	clr_character_flag = venture_hz_shrine
	clr_character_flag = venture_hz_friends

# Scope: the Venturer. Finds a governor for a new colony: a courtier who
# is an adventurer or a good steward, else a hired one of the Venturer's
# people. Saves event_target:venture_governor.
venture_pick_governor_effect = {
	if = {
		limit = {
			any_courtier = {
				is_adult = yes
				is_ruler = no
				prisoner = no
				is_inaccessible_trigger = no
				can_be_given_away = yes
				OR = {
					trait = adventurer
					stewardship = 10
					martial = 12
				}
			}
		}
		random_courtier = {
			limit = {
				is_adult = yes
				is_ruler = no
				prisoner = no
				is_inaccessible_trigger = no
				can_be_given_away = yes
				OR = {
					trait = adventurer
					stewardship = 10
					martial = 12
				}
			}
			preferred_limit = { has_character_flag = venture_captain }
			preferred_limit = { trait = adventurer }
			save_event_target_as = venture_governor
		}
	}
	else = {
		create_character = {
			random_traits = yes
			dynasty = random
			religion = ROOT
			culture = ROOT
			female = no
			age = 40
			attributes = { stewardship = 10 martial = 8 }
			trait = adventurer
		}
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			save_event_target_as = venture_governor
		}
	}
}

# Scope: a colony province. ROOT is the Venturer. Settlers from home turn
# the colony to the colonial culture of the Venturer's people, or to the
# Venturer's own where the lore has none.
venture_settle_colony_effect = {
	if = {
		limit = {
			ROOT = {
				culture_group = imperial_group
				capital_scope = { region = world_greater_araby }
			}
		}
		culture = sudenburger
	}
	else_if = {
		limit = { ROOT = { culture_group = imperial_group } }
		culture = neueslander
	}
	else_if = {
		limit = {
			ROOT = {
				OR = {
					culture_group = tilean_group
					culture_group = sartosan_group
				}
			}
		}
		culture = oltremari
	}
	else_if = {
		limit = { ROOT = { culture_group = estalian_group } }
		culture = comarcano
	}
	else_if = {
		limit = { ROOT = { culture_group = bretonnian_group } }
		culture = marcheur
	}
	else_if = {
		limit = { ROOT = { culture_group = arabyan_group } }
		culture = hadmauti
	}
	else = {
		culture = ROOT
	}
	add_province_modifier = { name = venture_settlers_came years = 20 }
}

# Scope: a colony province. A colony building is lost (venture.51, 55).
venture_colony_lose_fortifications_effect = {
	any_province_holding = {
		if = {
			limit = { has_building = ca_venture_fortifications }
			remove_building = ca_venture_fortifications
		}
		if = {
			limit = { has_building = ct_venture_fortifications }

		}
		if = {
			limit = { has_building = tp_venture_fortifications }
			remove_building = tp_venture_fortifications
		}
	}
}

# Scope: the holder of a colony of 2010 (venture.5). Marks each overseas
# port province of the realm as a chartered colony.
venture_mark_colonies_effect = {
	any_realm_province = {
		limit = {
			port = yes
			OR = {
				region = world_new_world
				region = world_greater_araby_southlands
				region = world_eastern_southlands
			}
		}
		set_province_flag = venture_colony
	}
}

# Scope: a colony province. The temple host razed it (venture.75): every
# colony building is gone and the colony smoulders for fifteen years.
venture_colony_raze_effect = {
	any_province_holding = {
		if = { limit = { has_building = ca_venture_capitol } remove_building = ca_venture_capitol }
		if = { limit = { has_building = ct_venture_capitol } remove_building = ct_venture_capitol }
		if = { limit = { has_building = tp_venture_capitol } remove_building = tp_venture_capitol }
		if = { limit = { has_building = ca_venture_venue } remove_building = ca_venture_venue }
		if = { limit = { has_building = ct_venture_venue } remove_building = ct_venture_venue }
		if = { limit = { has_building = tp_venture_venue } remove_building = tp_venture_venue }
		if = { limit = { has_building = ca_venture_monument } remove_building = ca_venture_monument }
		if = { limit = { has_building = ct_venture_monument } remove_building = ct_venture_monument }
		if = { limit = { has_building = tp_venture_monument } remove_building = tp_venture_monument }
		if = { limit = { has_building = ca_venture_port } remove_building = ca_venture_port }
		if = { limit = { has_building = ct_venture_port } remove_building = ct_venture_port }
		if = { limit = { has_building = tp_venture_port } remove_building = tp_venture_port }
		if = { limit = { has_building = ca_venture_fortifications } remove_building = ca_venture_fortifications }
		if = { limit = { has_building = ct_venture_fortifications } remove_building = ct_venture_fortifications }
		if = { limit = { has_building = tp_venture_fortifications } remove_building = tp_venture_fortifications }
		if = { limit = { has_building = ca_venture_farmsteads } remove_building = ca_venture_farmsteads }
		if = { limit = { has_building = ct_venture_farmsteads } remove_building = ct_venture_farmsteads }
		if = { limit = { has_building = tp_venture_farmsteads } remove_building = tp_venture_farmsteads }
	}
	add_province_modifier = { name = venture_colony_razed years = 15 }
}

# Scope: the Venturer (venture.74). The skinks strike: one in ten dies
# behind a doubled guard, four in ten without it (half again with great
# wrath). A survivor is left alone by the Old Ones from then on.
venture_skink_strike_effect = {
	random_list = {
		10 = {
			modifier = { factor = 4 has_character_flag = venture_unguarded }
			modifier = { factor = 1.5 check_variable = { which = venture_slann_wrath value = 4 } }
			custom_tooltip = { text = venture_skinks_kill_tt }
			hidden_effect = { death = { death_reason = death_murder_unknown_poison } }
		}
		25 = {
			modifier = { factor = 1.5 has_character_flag = venture_unguarded }
			custom_tooltip = { text = venture_skinks_wound_tt }
			add_trait = wounded
			prestige = 50
		}
		65 = {
			custom_tooltip = { text = venture_skinks_fail_tt }
			prestige = 100
		}
	}
	clr_character_flag = venture_unguarded
	clr_character_flag = venture_slann_answering
	clr_character_flag = venture_slann_warned
	set_variable = { which = venture_slann_wrath value = 0 }
}

# Scope: the Venturer. The Great Voyage is over.
venture_great_voyage_clear_effect = {
	clr_character_flag = venture_great_voyage_out
	clr_character_flag = venture_expedition_out
	clr_character_flag = venture_leading
	clr_character_flag = venture_turned_back

	clr_character_flag = venture_gv_north
	set_variable = { which = venture_odds value = 0 }

		limit = { has_character_modifier = venture_on_great_voyage }
		remove_character_modifier = venture_on_great_voyage
	}

		limit = { event_target:venture_captain = { always = yes } }
		event_target:venture_captain = { clr_character_flag = venture_captain }
	}
}

# Scope: the Venturer. A Great Voyage that did not come home in glory
# leaves its route open for another to sail.
venture_great_voyage_reopen_effect = {
	if = {
		limit = { has_character_flag = venture_gv_west }
		clr_global_flag = venture_voyage_west_sailed
	}
	else_if = {
		limit = { has_character_flag = venture_gv_east }
		clr_global_flag = venture_voyage_east_sailed
	}
	else_if = {
		limit = { has_character_flag = venture_gv_north }
		clr_global_flag = venture_voyage_north_sailed
	}
}
