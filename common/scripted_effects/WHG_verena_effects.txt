# The library: the size is chosen. Picks the first fitting and, three
# times in four, a mishap during the building (verena.17 - 19).
# Scope: the member.
verena_library_start_effect = {
	verena_library_next_effect = yes
	random_list = {
		25 = { }
		25 = { character_event = { id = verena.17 days = 45 } }
		25 = { character_event = { id = verena.18 days = 45 } }
		25 = { character_event = { id = verena.19 days = 45 } }
	}
}

# The library: one more fitting while lab_ambition is 1 or more, else the
# library is finished (verena.14).
# Scope: the member.
verena_library_next_effect = {
	if = {
		limit = { check_variable = { which = lab_ambition value = 1 } }
		random_list = {
			25 = {
				modifier = { factor = 0 has_character_flag = verena_library_scribes }
				set_character_flag = verena_library_next_scribes
				set_character_flag = verena_library_scribes
			}
			25 = {
				modifier = { factor = 0 has_character_flag = verena_library_lamps }
				set_character_flag = verena_library_next_lamps
				set_character_flag = verena_library_lamps
			}
			25 = {
				modifier = { factor = 0 has_character_flag = verena_library_bindings }
				set_character_flag = verena_library_next_bindings
				set_character_flag = verena_library_bindings
			}
			25 = {
				modifier = { factor = 0 has_character_flag = verena_library_books }
				set_character_flag = verena_library_next_books
				set_character_flag = verena_library_books
			}
		}
		character_event = { id = verena.13 days = 30 random = 30 }
	}
	else = {
		character_event = { id = verena.14 days = 30 }
	}
}

# The library: a fitting of verena.13 is chosen.
# Scope: the member.
verena_library_fitting_done_effect = {
	change_variable = { which = lab_ambition value = -1 }
	clr_character_flag = verena_library_next_scribes
	clr_character_flag = verena_library_next_lamps
	clr_character_flag = verena_library_next_bindings
	clr_character_flag = verena_library_next_books
	verena_library_next_effect = yes
}

# The library: the building flags are cleared once it is finished.
# Scope: the member.
verena_library_clear_effect = {
	clr_character_flag = verena_library_scribes
	clr_character_flag = verena_library_lamps
	clr_character_flag = verena_library_bindings
	clr_character_flag = verena_library_books
	clr_character_flag = verena_library_next_scribes
	clr_character_flag = verena_library_next_lamps
	clr_character_flag = verena_library_next_bindings
	clr_character_flag = verena_library_next_books
	set_variable = { which = lab_ambition value = 0 }
}

# The library: the fire of verena.17 burns someone of the household when
# the library is inside the castle.
# Scope: the member; event target sacrifice_for_science.
verena_library_fire_effect = {
	if = {
		limit = {
			has_character_flag = hidden_lab
			event_target:sacrifice_for_science = { always = yes }
		}
		event_target:sacrifice_for_science = {
			add_maimed_trait_effect = yes
			if = {
				limit = {
					is_close_relative = ROOT
					NOT = { ROOT = { trait = cruel } }
				}
				random = {
					chance = 20
					ROOT = { add_trait_depressed_effect = yes }
				}
			}
		}
	}
}

# The library: once it is finished, the mission is done (verena.22).
# Scope: the member.
verena_library_reward_effect = {
	if = {
		limit = { has_quest = quest_verena_build_library }
		hidden_effect = { character_event = { id = verena.22 } }
	}
}

# The library: a library is lost (verena.68, 69).
# Scope: the owner.
verena_library_remove_effect = {
	if = {
		limit = { has_character_modifier = cheap_laboratory }
		remove_character_modifier = cheap_laboratory
	}
	if = {
		limit = { has_character_modifier = normal_laboratory }
		remove_character_modifier = normal_laboratory
	}
	if = {
		limit = { has_character_modifier = good_laboratory }
		remove_character_modifier = good_laboratory
	}
	if = {
		limit = { has_character_modifier = great_laboratory }
		remove_character_modifier = great_laboratory
	}
	if = {
		limit = { has_character_modifier = best_laboratory }
		remove_character_modifier = best_laboratory
	}
	hidden_effect = {
		clr_character_flag = built_laboratory
		clr_character_flag = hidden_lab
		clr_character_flag = very_hidden_lab
	}
}

# The Scrollbearer's hoard: the member takes. Guards who woke up catch
# him (verena.60), else he searches the hoard (verena.66).

# The scrying fallbacks: a courtier who is neither friend, rival nor lover
# of the member, or a newcomer to court, becomes the friend or rival the
# vision spoke of (verena.119, 120).
# Scope: the member. Saves event target verena_omen_courtier.
verena_omen_find_courtier_effect = {
	if = {
		limit = {
			any_courtier = {
				is_adult = yes
				prisoner = no
				NOT = { trait = incapable }
				NOR = { is_friend = ROOT is_rival = ROOT is_lover = ROOT }
			}
		}
		random_courtier = {
			limit = {
				is_adult = yes
				prisoner = no
				NOT = { trait = incapable }
				NOR = { is_friend = ROOT is_rival = ROOT is_lover = ROOT }
			}
			save_event_target_as = verena_omen_courtier
		}
	}
	else = {
		create_character = {
			random_traits = yes
			dynasty = random
		}
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			save_event_target_as = verena_omen_courtier
		}
	}
}

# The scrying fallbacks: the omen the member carries, as flags for the
# event text once the omen is removed (verena.119, 120).
# Scope: the member.
verena_omen_flags_effect = {
	if = {
		limit = { has_character_modifier = good_battle_omen }
		set_character_flag = omen_opt_good_battle
	}
	else_if = {
		limit = { has_character_modifier = bad_battle_omen }
		set_character_flag = omen_opt_bad_battle
	}
	else_if = {
		limit = { has_character_modifier = good_relationship_omen }
		set_character_flag = omen_opt_good_relationship
	}
	else_if = {
		limit = { has_character_modifier = bad_relationship_omen }
		set_character_flag = omen_opt_bad_relationship
	}
	else_if = {
		limit = { has_character_modifier = good_power_omen }
		set_character_flag = omen_opt_good_power
	}
	else = {
		set_character_flag = omen_opt_bad_power
	}
}

# The scrying fallbacks: the flags of verena_omen_flags_effect are cleared.
# Scope: the member.
verena_omen_clear_flags_effect = {
	clr_character_flag = omen_opt_good_battle
	clr_character_flag = omen_opt_bad_battle
	clr_character_flag = omen_opt_good_relationship
	clr_character_flag = omen_opt_bad_relationship
	clr_character_flag = omen_opt_good_power
	clr_character_flag = omen_opt_bad_power
}

# The scrying: a blessing comes (verena.119). A feat of arms that the
# garrison takes pride in, a new friend, or an owl of Verena settling in
# the castle.
# Scope: the member; event target verena_omen_courtier.
verena_omen_blessing_effect = {
	if = {
		limit = {
			OR = {
				has_character_flag = omen_opt_good_battle
				has_character_flag = omen_opt_bad_battle
			}
		}
		prestige = 100
		capital_scope = {
			add_province_modifier = { name = verena_proud_garrison duration = 1825 }
		}
	}
	else_if = {
		limit = {
			OR = {
				has_character_flag = omen_opt_good_relationship
				has_character_flag = omen_opt_bad_relationship
			}
		}
		add_friend = event_target:verena_omen_courtier
	}
	else_if = {
		limit = { has_character_modifier = pet_owl }
		prestige = 100
		piety = 25
	}
	else = {
		add_pet_owl_effect = yes
	}
}

# The scrying: a misfortune comes (verena.120). The charge of a white stag,
# a slanderer at court, or a quack who sells false cures to the people.
# Scope: the member; event target verena_omen_courtier.
verena_omen_misfortune_effect = {
	if = {
		limit = {
			OR = {
				has_character_flag = omen_opt_good_battle
				has_character_flag = omen_opt_bad_battle
			}
		}
		add_maimed_trait_effect = yes
	}
	else_if = {
		limit = {
			OR = {
				has_character_flag = omen_opt_good_relationship
				has_character_flag = omen_opt_bad_relationship
			}
		}
		add_rival = event_target:verena_omen_courtier
	}
	else = {
		scaled_wealth = -0.25
		capital_scope = {
			add_province_modifier = { name = verena_quack_cures duration = 1095 }
		}
	}
}

# The horoscope: at thirteen, with an adult at court to inspire the child,
# the child weighs the destiny (verena.134). Else the chain waits a year.
# Scope: the child.
verena_horoscope_contemplate_effect = {
	if = {
		limit = {
			age = 13
			verena_horoscope_has_mentor_trigger = yes
		}
		clr_character_flag = horoscope_child_contemplates_their_destiny
		set_character_flag = horoscope_parent_is_informed
		character_event = { id = verena.134 }
	}
	else = {
		character_event = { id = verena.140 days = 365 }
	}
}
