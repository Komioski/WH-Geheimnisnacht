# The library: the size is chosen. Picks the first fitting and, three
# times in four, a mishap during the building (verena.17 - 19).
# Scope: the member.
verena_library_start_effect = {
	verena_library_next_effect = yes
	random_list = {
		25 = { }
		25 = { character_event = { id = verena.17 days = 45 } }
		25 = { character_event = { id = verena.18 days = 45 } }
		25 = { character_event = { id = verena.19 days = 45 } }
	}
}

# The library: one more fitting while lab_ambition is 1 or more, else the
# library is finished (verena.14).
# Scope: the member.
verena_library_next_effect = {
	if = {
		limit = { check_variable = { which = lab_ambition value = 1 } }
		random_list = {
			25 = {
				modifier = { factor = 0 has_character_flag = verena_library_scribes }
				set_character_flag = verena_library_next_scribes
				set_character_flag = verena_library_scribes
			}
			25 = {
				modifier = { factor = 0 has_character_flag = verena_library_lamps }
				set_character_flag = verena_library_next_lamps
				set_character_flag = verena_library_lamps
			}
			25 = {
				modifier = { factor = 0 has_character_flag = verena_library_bindings }
				set_character_flag = verena_library_next_bindings
				set_character_flag = verena_library_bindings
			}
			25 = {
				modifier = { factor = 0 has_character_flag = verena_library_books }
				set_character_flag = verena_library_next_books
				set_character_flag = verena_library_books
			}
		}
		character_event = { id = verena.13 days = 30 random = 30 }
	}
	else = {
		character_event = { id = verena.14 days = 30 }
	}
}

# The library: a fitting of verena.13 is chosen.
# Scope: the member.
verena_library_fitting_done_effect = {
	change_variable = { which = lab_ambition value = -1 }
	clr_character_flag = verena_library_next_scribes
	clr_character_flag = verena_library_next_lamps
	clr_character_flag = verena_library_next_bindings
	clr_character_flag = verena_library_next_books
	verena_library_next_effect = yes
}

# The library: the building flags are cleared once it is finished.
# Scope: the member.
verena_library_clear_effect = {
	clr_character_flag = verena_library_scribes
	clr_character_flag = verena_library_lamps
	clr_character_flag = verena_library_bindings
	clr_character_flag = verena_library_books
	clr_character_flag = verena_library_next_scribes
	clr_character_flag = verena_library_next_lamps
	clr_character_flag = verena_library_next_bindings
	clr_character_flag = verena_library_next_books
	set_variable = { which = lab_ambition value = 0 }
}

# The library: the fire of verena.17 burns someone of the household when
# the library is inside the castle.
# Scope: the member; event target sacrifice_for_science.
verena_library_fire_effect = {
	if = {
		limit = {
			has_character_flag = hidden_lab
			event_target:sacrifice_for_science = { always = yes }
		}
		event_target:sacrifice_for_science = {
			add_maimed_trait_effect = yes
			if = {
				limit = {
					is_close_relative = ROOT
					NOT = { ROOT = { trait = cruel } }
				}
				random = {
					chance = 20
					ROOT = { add_trait_depressed_effect = yes }
				}
			}
		}
	}
}

# The library: once it is finished, the mission is done (verena.22).
# Scope: the member.
verena_library_reward_effect = {
	if = {
		limit = { has_quest = quest_verena_build_library }
		hidden_effect = { character_event = { id = verena.22 } }
	}
}

# The library: a library is lost (verena.68, 69).
# Scope: the owner.
verena_library_remove_effect = {
	if = {
		limit = { has_character_modifier = cheap_laboratory }
		remove_character_modifier = cheap_laboratory
	}
	if = {
		limit = { has_character_modifier = normal_laboratory }
		remove_character_modifier = normal_laboratory
	}
	if = {
		limit = { has_character_modifier = good_laboratory }
		remove_character_modifier = good_laboratory
	}
	if = {
		limit = { has_character_modifier = great_laboratory }
		remove_character_modifier = great_laboratory
	}
	if = {
		limit = { has_character_modifier = best_laboratory }
		remove_character_modifier = best_laboratory
	}
	hidden_effect = {
		clr_character_flag = built_laboratory
		clr_character_flag = hidden_lab
		clr_character_flag = very_hidden_lab
	}
}

# The Scrollbearer's hoard: the member takes. Guards who woke up catch
# him (verena.60), else he searches the hoard (verena.66).
# Scope: the member.
verena_raid_steal_effect = {
	if = {
		limit = { verena_raid_detected_trigger = yes }
		set_character_flag = caught_stealing
		character_event = { id = verena.60 days = 25 }
	}
	else = {
		character_event = { id = verena.66 days = 25 }
	}
}

# The Scrollbearer's hoard: the member wrecks. Guards who woke up catch
# him (verena.60), else he wrecks the scriptorium (verena.68).
# Scope: the member.
verena_raid_wreck_effect = {
	if = {
		limit = { verena_raid_detected_trigger = yes }
		set_character_flag = caught_destroying
		character_event = { id = verena.60 days = 25 }
	}
	else = {
		character_event = { id = verena.68 days = 25 }
	}
}

# The Scrollbearer's hoard: the companion's bond for the mission ends.
# Scope: the member; event target infiltration_companion.
verena_raid_companion_done_effect = {
	event_target:infiltration_companion = {
		remove_opinion = { who = ROOT modifier = opinion_infiltration_companion }
	}
}
