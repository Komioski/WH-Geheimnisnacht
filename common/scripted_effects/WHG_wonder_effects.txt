# wonder_rename_effect
# Scope: a wonder. Moves the wonder one step along its seven names
# (wonder_name_1 - 7) and sets the new name. The first two steps always
# happen; later steps get rarer (90% down to 50%), and the seventh name is
# the last. Replaces wonder_name_change_effect, which used break = yes and
# so had to run last in its event.
wonder_rename_effect = {
	if = {
		limit = {
			NOR = {
				has_wonder_flag = wonder_name_1
				has_wonder_flag = wonder_name_2
				has_wonder_flag = wonder_name_3
				has_wonder_flag = wonder_name_4
				has_wonder_flag = wonder_name_5
				has_wonder_flag = wonder_name_6
				has_wonder_flag = wonder_name_7
			}
		}
		set_wonder_flag = wonder_name_1
		set_name = wonder_name_text_update
	}
	else_if = {
		limit = { has_wonder_flag = wonder_name_1 }
		clr_wonder_flag = wonder_name_1
		set_wonder_flag = wonder_name_2
		set_name = wonder_name_text_update
	}
	else_if = {
		limit = { has_wonder_flag = wonder_name_2 }
		random = {
			chance = 90
			clr_wonder_flag = wonder_name_2
			set_wonder_flag = wonder_name_3
			set_name = wonder_name_text_update
		}
	}
	else_if = {
		limit = { has_wonder_flag = wonder_name_3 }
		random = {
			chance = 80
			clr_wonder_flag = wonder_name_3
			set_wonder_flag = wonder_name_4
			set_name = wonder_name_text_update
		}
	}
	else_if = {
		limit = { has_wonder_flag = wonder_name_4 }
		random = {
			chance = 70
			clr_wonder_flag = wonder_name_4
			set_wonder_flag = wonder_name_5
			set_name = wonder_name_text_update
		}
	}
	else_if = {
		limit = { has_wonder_flag = wonder_name_5 }
		random = {
			chance = 60
			clr_wonder_flag = wonder_name_5
			set_wonder_flag = wonder_name_6
			set_name = wonder_name_text_update
		}
	}
	else_if = {
		limit = { has_wonder_flag = wonder_name_6 }
		random = {
			chance = 50
			clr_wonder_flag = wonder_name_6
			set_wonder_flag = wonder_name_7
			set_name = wonder_name_text_update
		}
	}
}

# wonder_book_forgery_effect

		set_description = wonder_book_fake_desc

		set_creation_date = 1.1.1

# wonder_book_send_emissary_effect
# Scope: the ruler, with event_target:emissary chosen. The emissary rides
# to the court of the seller, and in 50 to 150 days the book is found
# (wonder.107; likelier for a diligent or scholarly emissary), the emissary
# writes for more money (wonder.104; likelier for a deceitful one who does
# not like the ruler) or it was only a rumour (wonder.103).
wonder_book_send_emissary_effect = {
	wealth = -30
	event_target:emissary = {
		add_trait = travelling
		move_character = event_target:book_province_holder
	}
	hidden_effect = {
		random_list = {
			10 = {
				modifier = { factor = 2 event_target:emissary = { trait = diligent } }
				modifier = { factor = 2 event_target:emissary = { trait = scholar } }
				character_event = { id = wonder.107 days = 100 random = 50 }
			}
			10 = {
				modifier = {
					factor = 5
					event_target:emissary = {
						trait = deceitful
						NOT = { opinion = { who = ROOT value = 20 } }
					}
				}
				letter_event = { id = wonder.104 days = 100 random = 50 }
			}
			10 = { character_event = { id = wonder.103 days = 100 random = 50 } }
		}
	}
}

# wonder_book_emissary_home_effect
# Scope: the ruler. The emissary comes home and hands back what is left of
# the purse (their stewardship in gold, variable money_left).
wonder_book_emissary_home_effect = {
	event_target:emissary = { move_character = ROOT remove_trait = travelling }
	export_to_variable = { which = money_left value = stewardship who = event_target:emissary }
}

# wonder_burial_clear_effect
# Scope: the heir in wonder.171, FROM the dead ruler. Clears the marks
# wonder.170 left on the realm's characters (liked_dead_ruler_@ and
# hated_dead_ruler_@ the dead ruler).
wonder_burial_clear_effect = {
	any_realm_character = {
		limit = {
			OR = {
				has_character_flag = liked_dead_ruler_@FROM
				has_character_flag = hated_dead_ruler_@FROM
			}
		}
		clr_character_flag = liked_dead_ruler_@FROM
		clr_character_flag = hated_dead_ruler_@FROM
	}
}

# wonder_relic_reward_effect
# Scope: the relic (artifact), ROOT the ruler. Piety and prestige by the
# relic's quality, for laying it in the wonder (wonder.180).
wonder_relic_reward_effect = {
	if = { limit = { quality == 5 } ROOT = { piety = 500 prestige = 500 } }
	else_if = { limit = { quality == 4 } ROOT = { piety = 400 prestige = 300 } }
	else_if = { limit = { quality == 3 } ROOT = { piety = 300 prestige = 100 } }
	else_if = { limit = { quality == 2 } ROOT = { piety = 200 } }
	else = { ROOT = { piety = 100 } }
}

# wonder_sack_outrage_effect
# Scope: the commander in wonder.210, FROM the sacked holding. Its holder
# and their top liege are furious for three years; everyone in the top
# liege's realm who is not cruel thinks the commander dishonourable for
# ten.
wonder_sack_outrage_effect = {
	FROM = {
		holder_scope = {
			show_scope_change = no
			opinion = { modifier = opinion_furious who = ROOT years = 3 }
			top_liege = {
				show_scope_change = no
				opinion = { modifier = opinion_furious who = ROOT years = 3 }
				custom_tooltip = { text = EVTTOOLTIPwonder210c }
				hidden_effect = {
					any_realm_character = {
						limit = { NOT = { trait = cruel } }
						opinion = { modifier = opinion_dishonorable who = ROOT years = 10 }
					}
				}
			}
		}
	}
}
