# The mists fail: Albion is open to war and marriage (the many CBs and the
# marriage rules that read mists_of_albion), and every player hears of it
# (albion.10).
# Scope: any.
albion_mists_lift_effect = {
	if = {
		limit = { has_global_flag = mists_of_albion }
		clr_global_flag = mists_of_albion
		set_global_flag = mists_of_albion_cleared
		any_player = {
			narrative_event = { id = albion.10 }
		}
	}
}

# Counts the Ogham stones of Albion that stand dark (corrupt places of
# power) into the variable albion_dark_stones. Runs only when a stone falls
# or is raised, so the walk over the provinces is rare.
# Scope: a character.
albion_count_dark_stones_effect = {
	set_variable = { which = albion_dark_stones value = 0 }
	any_province = {
		limit = {
			has_province_modifier = corrupt_place_of_power_man
			region = world_albion
		}
		PREV = { change_variable = { which = albion_dark_stones value = 1 } }
	}
}
