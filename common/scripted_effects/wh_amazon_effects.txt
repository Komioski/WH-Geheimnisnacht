# Kills a character as a sacrifice, unless they cannot die.
# Scope: the victim.
amazon_sacrifice_effect = {
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		death = { death_reason = death_sacrificed }
	}
}

# How well a contestant does today in a mud fight, a brawl or a hunt.
# Strong, brave and skilled fighters roll well more often; the hurt, sick,
# old and craven roll badly. Sets the flag amazon_roll_great or
# amazon_roll_poor (no flag is an average day).
# Scope: a contestant.
amazon_contest_roll_effect = {
	clr_character_flag = amazon_roll_great
	clr_character_flag = amazon_roll_poor
	random_list = {
		# A great day
		25 = {
			modifier = { factor = 2 trait = strong }
			modifier = { factor = 1.5 trait = tough }
			modifier = { factor = 1.25 trait = hardy }
			modifier = { factor = 1.75 trait = brave }
			modifier = { factor = 1.85 trait = duelist }
			modifier = { factor = 1.5 trait = hunter }
			modifier = { factor = 1.25 trait = tall }
			modifier = { factor = 1.25 martial = 15 }
			modifier = { factor = 1.5 martial = 20 }
			modifier = { factor = 1.35 trait = tough_soldier }
			modifier = { factor = 1.5 trait = skilled_tactician }
			modifier = { factor = 1.75 trait = brilliant_strategist }
			modifier = { factor = 1.35 trait = barbarian_2 }
			modifier = { factor = 1.5 trait = barbarian_3 }
			modifier = { factor = 1.75 trait = barbarian_4 }
			modifier = { factor = 1.25 trait = aggressive_leader }
			modifier = { factor = 1.35 trait = trickster }
			modifier = { factor = 1.35 trait = zealous }
			modifier = { factor = 1.5 trait = ambitious }
			modifier = { factor = 1.25 trait = wroth }
			modifier = { factor = 1.25 OR = { trait = quick trait = brilliant trait = genius } }
			modifier = { factor = 1.65 age = 150 trait = immortal }
			set_character_flag = amazon_roll_great
		}
		# An ordinary day
		50 = { }
		# A bad day
		25 = {
			modifier = { factor = 1.75 trait = weak }
			modifier = { factor = 1.5 trait = fragile }
			modifier = { factor = 1.25 trait = soft }
			modifier = { factor = 2 trait = craven }
			modifier = { factor = 2 trait = wounded }
			modifier = { factor = 4 trait = maimed }
			modifier = { factor = 4 trait = blinded }
			modifier = { factor = 1.5 trait = infirm }
			modifier = { factor = 2 trait = dwarf }
			modifier = { factor = 2 trait = hunchback }
			modifier = { factor = 1.5 trait = clubfooted }
			modifier = { factor = 1.25 trait = is_fat }
			modifier = { factor = 1.25 trait = is_malnourished }
			modifier = { factor = 1.2 trait = drunkard }
			modifier = { factor = 1.2 trait = addict }
			modifier = { factor = 1.25 trait = ill }
			modifier = { factor = 2 trait = has_nurgles_rot }
			modifier = { factor = 1.3 trait = depressed }
			modifier = { factor = 1.2 trait = stressed }
			modifier = { factor = 1.2 trait = complacent }
			modifier = { factor = 1.25 trait = slothful }
			modifier = { factor = 1.15 NOT = { age = 18 } }
			modifier = { factor = 2 age = 40 NOT = { trait = immortal } }
			modifier = { factor = 3 OR = { trait = pregnant trait = hard_pregnancy trait = troubled_pregnancy } }
			set_character_flag = amazon_roll_poor
		}
	}
}

# Decides a contest between event_target:amazon_contestant_1 and
# event_target:amazon_contestant_2. Each rolls her day; the better day wins
# more often, but anyone can win. Saves event_target:amazon_contest_winner
# and event_target:amazon_contest_loser.
# Scope: any.
amazon_contest_decide_effect = {
	event_target:amazon_contestant_1 = { amazon_contest_roll_effect = yes }
	event_target:amazon_contestant_2 = { amazon_contest_roll_effect = yes }
	random_list = {
		50 = {
			modifier = {
				factor = 3
				event_target:amazon_contestant_1 = { has_character_flag = amazon_roll_great }
				event_target:amazon_contestant_2 = { NOT = { has_character_flag = amazon_roll_great } }
			}
			modifier = {
				factor = 3
				event_target:amazon_contestant_1 = { NOT = { has_character_flag = amazon_roll_poor } }
				event_target:amazon_contestant_2 = { has_character_flag = amazon_roll_poor }
			}
			event_target:amazon_contestant_1 = { save_event_target_as = amazon_contest_winner }
			event_target:amazon_contestant_2 = { save_event_target_as = amazon_contest_loser }
		}
		50 = {
			modifier = {
				factor = 3
				event_target:amazon_contestant_2 = { has_character_flag = amazon_roll_great }
				event_target:amazon_contestant_1 = { NOT = { has_character_flag = amazon_roll_great } }
			}
			modifier = {
				factor = 3
				event_target:amazon_contestant_2 = { NOT = { has_character_flag = amazon_roll_poor } }
				event_target:amazon_contestant_1 = { has_character_flag = amazon_roll_poor }
			}
			event_target:amazon_contestant_2 = { save_event_target_as = amazon_contest_winner }
			event_target:amazon_contestant_1 = { save_event_target_as = amazon_contest_loser }
		}
	}
	event_target:amazon_contestant_1 = {
		clr_character_flag = amazon_roll_great
		clr_character_flag = amazon_roll_poor
	}
	event_target:amazon_contestant_2 = {
		clr_character_flag = amazon_roll_great
		clr_character_flag = amazon_roll_poor
	}
}

# One round of the feast drinking contest: the drinker keeps her drink down,
# or collapses (flag amazon_drink_collapsed). The more rounds have been
# drunk (variable drinking_count on the host), the likelier she collapses.
# Scope: the drinker; event_target:am_drinking_host is the feast host.
amazon_drink_round_effect = {
	clr_character_flag = amazon_drink_collapsed
	random_list = {
		# Keeps it down
		88 = {
			modifier = { factor = 0.65 trait = fragile }
			modifier = { factor = 0.75 trait = weak }
			modifier = { factor = 0.85 trait = soft }
			modifier = { factor = 1.5 trait = hardy }
			modifier = { factor = 1.65 trait = tough }
			modifier = { factor = 1.85 trait = strong }
			modifier = { factor = 0.75 trait = temperate }
			modifier = { factor = 1.15 trait = diligent }
			modifier = { factor = 1.5 trait = drunkard }
			modifier = { factor = 1.25 trait = addict }
			modifier = { factor = 1.15 trait = gluttonous }
			modifier = { factor = 1.5 trait = is_fat }
			modifier = { factor = 0.75 trait = is_malnourished }
			modifier = { factor = 1.5 trait = ambitious }
			modifier = { factor = 0.85 trait = content }
			modifier = { factor = 0.85 event_target:am_drinking_host = { check_variable = { which = drinking_count value = 5 } } }
			modifier = { factor = 0.75 event_target:am_drinking_host = { check_variable = { which = drinking_count value = 7 } } }
			modifier = { factor = 0.75 event_target:am_drinking_host = { check_variable = { which = drinking_count value = 10 } } }
			modifier = { factor = 0.75 event_target:am_drinking_host = { check_variable = { which = drinking_count value = 13 } } }
			modifier = { factor = 0.75 event_target:am_drinking_host = { check_variable = { which = drinking_count value = 15 } } }
			modifier = { factor = 0.75 event_target:am_drinking_host = { check_variable = { which = drinking_count value = 17 } } }
			modifier = { factor = 0.75 event_target:am_drinking_host = { check_variable = { which = drinking_count value = 20 } } }
			modifier = { factor = 0.25 event_target:am_drinking_host = { check_variable = { which = drinking_count value = 23 } } }
			modifier = { factor = 0.15 event_target:am_drinking_host = { check_variable = { which = drinking_count value = 25 } } }
			modifier = { factor = 0 event_target:am_drinking_host = { check_variable = { which = drinking_count value = 30 } } }
		}
		# Collapses
		12 = { set_character_flag = amazon_drink_collapsed }
	}
	event_target:am_drinking_host = { change_variable = { which = drinking_count value = 1 } }
}

# Prestige for a drinking contest, by how long it lasted.
# Scope: a contestant; event_target:am_drinking_host is the feast host.
amazon_drink_prestige_effect = {
	if = {
		limit = { event_target:am_drinking_host = { check_variable = { which = drinking_count value = 20 } } }
		prestige = 150
		decadence = -1
	}
	else_if = {
		limit = { event_target:am_drinking_host = { check_variable = { which = drinking_count value = 15 } } }
		prestige = 100
	}
	else_if = {
		limit = { event_target:am_drinking_host = { check_variable = { which = drinking_count value = 10 } } }
		prestige = 50
	}
	else_if = {
		limit = { event_target:am_drinking_host = { check_variable = { which = drinking_count value = 5 } } }
		prestige = 25
	}
	else = { prestige = 20 }
}

# Strips the injuries and bad habits a girl leaves behind in the sacred pool
# at her rite of passage.
# Scope: the new Amazon.
amazon_rite_cleanse_effect = {
	remove_trait = maimed
	remove_trait = wounded
	remove_trait = blinded
	remove_trait = disfigured
	remove_trait = is_fat
	remove_trait = is_malnourished
	remove_trait = stressed
	remove_trait = depressed
}

# Clears the flags a hunt leaves on a huntress.
# Scope: a huntress.
amazon_hunt_clear_effect = {
	amazon_contest_clear_story_effect = yes
	clr_character_flag = amazon_roll_great
	clr_character_flag = amazon_roll_poor
}

# The rare chance to bring home a trophy fang from a great kill.
# Scope: a huntress.
amazon_hunt_trophy_effect = {
	if = {
		limit = {
			NOT = { has_artifact = amazon_hunt_trophy }
		}
		add_artifact = amazon_hunt_trophy
	}
	else = { prestige = 50 }
}

# The price of a strong or beautiful body from the sacred pool: the mind
# dulls by one step.
# Scope: the new Amazon.
amazon_rite_dull_mind_effect = {
	if = {
		limit = { trait = imbecile }
		change_learning = -5
	}
	else_if = {
		limit = {
			OR = {
				trait = dim
				trait = dumb
				trait = slow
				trait = idiot
			}
		}
		remove_trait = dim
		remove_trait = dumb
		remove_trait = slow
		remove_trait = idiot
		add_trait = imbecile
	}
	else_if = {
		limit = {
			OR = {
				trait = bright
				trait = smart
				trait = quick
			}
		}
		remove_trait = bright
		remove_trait = smart
		remove_trait = quick
		add_trait = dumb
	}
	else_if = {
		limit = {
			OR = {
				trait = brilliant
				trait = genius
			}
		}
		remove_trait = brilliant
		remove_trait = genius
		add_trait = dim
	}
	else = { add_trait = slow }
}

# Gives a first step of magic potential to a priestess who has none.
# Scope: the priestess.
amazon_give_magic_potential_effect = {
	if = {
		limit = {
			NOR = {
				trait = magic_potential_1
				trait = magic_potential_2
				trait = magic_potential_3
				trait = magic_potential_4
			}
		}
		add_trait = magic_potential_1
	}
}
