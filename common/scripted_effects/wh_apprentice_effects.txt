###############################################################################
# Mage apprentice effects
# Used by events/WHG_mage_apprentice_events.txt (namespace apprentice).
#
# apprentice_learning_small_effect   +1 learning, or New Insights past 15
# apprentice_learning_medium_effect  +2 learning, or Nocturnal Studies past 18
# apprentice_learning_large_effect   +3 learning, or Nocturnal Studies past 20
# apprentice_tutor_lore_effect       a new mage takes his master's lore
# apprentice_learned_lore_effect     a self-taught mage's chosen lore
# apprentice_new_mage_effect         what every new mage gets
# apprentice_join_master_college_effect  an adult apprentice joins his
#                                    master's college
###############################################################################

apprentice_learning_small_effect = {
	if = { limit = { NOT = { learning = 15 } } change_learning = 1 }
	else = { add_character_modifier = { name = new_insights duration = 1095 } }
}

apprentice_learning_medium_effect = {
	if = { limit = { NOT = { learning = 18 } } change_learning = 2 }
	else = { add_character_modifier = { name = nocturnal_studies duration = 1095 } }
}

apprentice_learning_large_effect = {
	if = { limit = { NOT = { learning = 20 } } change_learning = 3 }
	else = { add_character_modifier = { name = nocturnal_studies duration = 1095 } }
}

# THIS is the new mage, event_target:mage_tutor his master. He takes his
# master's lore. Chaos lores pass whole only to a follower of that god;
# anyone else gets the Lore of Chaos. High and Dark Magic go to elves, Light
# and Shadows to everyone else. Arabyan scholars and astromancers pass on
# their calling too.
apprentice_tutor_lore_effect = {
	event_target:mage_tutor = { save_event_target_as = lore_source }
	if = { limit = { event_target:lore_source = { trait = lore_ice } } add_trait = lore_ice }
	else_if = { limit = { event_target:lore_source = { trait = lore_hag } } add_trait = lore_hag }
	else_if = { limit = { event_target:lore_source = { trait = lore_fire } } add_trait = lore_fire }
	else_if = {
		limit = { event_target:lore_source = { trait = lore_beasts } }
		add_trait = lore_beasts
	}
	else_if = {
		limit = { event_target:lore_source = { trait = lore_metal } }
		add_trait = lore_metal
	}
	else_if = { limit = { event_target:lore_source = { trait = lore_life } } add_trait = lore_life }
	else_if = {
		limit = { event_target:lore_source = { trait = lore_heavens } }
		add_trait = lore_heavens
	}
	else_if = {
		limit = { event_target:lore_source = { trait = lore_death } }
		add_trait = lore_death
	}
	else_if = {
		limit = { event_target:lore_source = { trait = lore_tzeentch } }
		if = { limit = { religion = tzeentch } add_trait = lore_tzeentch }
		else = { add_trait = lore_chaos }
	}
	else_if = {
		limit = { event_target:lore_source = { trait = lore_nurgle } }
		if = { limit = { religion = nurgle } add_trait = lore_nurgle }
		else = { add_trait = lore_chaos }
	}
	else_if = {
		limit = { event_target:lore_source = { trait = lore_slaanesh } }
		if = { limit = { religion = slaanesh } add_trait = lore_slaanesh }
		else = { add_trait = lore_chaos }
	}
	else_if = {
		limit = { event_target:lore_source = { trait = lore_chaos } }
		add_trait = lore_chaos
	}
	else_if = {
		limit = {
			event_target:lore_source = { OR = { trait = lore_light trait = lore_high_magic } }
		}
		if = { limit = { trait = creature_elf } add_trait = lore_high_magic }
		else = { add_trait = lore_light }
	}
	else_if = {
		limit = {
			event_target:lore_source = { OR = { trait = lore_shadows trait = lore_dark_magic } }
		}
		if = { limit = { trait = creature_elf } add_trait = lore_dark_magic }
		else = { add_trait = lore_shadows }
	}
	if = {
		limit = { is_female = no event_target:lore_source = { trait = arabyan_scholar } }
		add_trait = arabyan_scholar
	}
	else_if = {
		limit = { event_target:lore_source = { trait = astromancer } }
		add_trait = astromancer
	}
}

# THIS taught himself; his learning_lore_ flag becomes the lore.
apprentice_learned_lore_effect = {
	if = { limit = { has_character_flag = learning_lore_ice } add_trait = lore_ice }
	else_if = { limit = { has_character_flag = learning_lore_fire } add_trait = lore_fire }
	else_if = { limit = { has_character_flag = learning_lore_metal } add_trait = lore_metal }
	else_if = { limit = { has_character_flag = learning_lore_beasts } add_trait = lore_beasts }
	else_if = { limit = { has_character_flag = learning_lore_light } add_trait = lore_light }
	else_if = { limit = { has_character_flag = learning_lore_shadows } add_trait = lore_shadows }
	else_if = { limit = { has_character_flag = learning_lore_death } add_trait = lore_death }
	else_if = { limit = { has_character_flag = learning_lore_heavens } add_trait = lore_heavens }
	else_if = { limit = { has_character_flag = learning_lore_life } add_trait = lore_life }
}

# THIS has just become a mage: the hidden gift shows, magic power and magic
# experience start, the study flags go, and a college member who finished
# gets his college reward (college_finished_effect).
apprentice_new_mage_effect = {
	reveal_hidden_magic_effect = yes
	magic_lvl_increase_effect = yes
	enable_magic_xp_effect = yes
	if = { limit = { is_college_of_magic_member = yes } college_finished_effect = yes }
	clr_learning_flags_effect = yes
	clr_character_flag = learning_lore_ice
	clr_character_flag = learning_magic_in_college
	clr_character_flag = college_member_learning_on_own
}

# THIS is an adult apprentice, ROOT his master. If the master belongs to a
# college and the apprentice to no society, he joins the master's college
# as a student of the master's wind, so the college finds him a new teacher
# if the master dies (college.26).
apprentice_join_master_college_effect = {
	if = {
		limit = { is_adult = yes is_in_society = no ROOT = { is_college_of_magic_member = yes } }
		if = {
			limit = { ROOT = { society_member_of = college_of_magic_imperial } }
			join_society = college_of_magic_imperial
		}
		else_if = {
			limit = { ROOT = { society_member_of = college_of_magic_dynamic_1 } }
			join_society = college_of_magic_dynamic_1
		}
		else_if = {
			limit = { ROOT = { society_member_of = college_of_magic_dynamic_2 } }
			join_society = college_of_magic_dynamic_2
		}
		else_if = {
			limit = { ROOT = { society_member_of = college_of_magic_dynamic_3 } }
			join_society = college_of_magic_dynamic_3
		}
		else_if = {
			limit = { ROOT = { society_member_of = college_of_magic_dynamic_4 } }
			join_society = college_of_magic_dynamic_4
		}
		else_if = {
			limit = { ROOT = { society_member_of = college_of_magic_dynamic_5 } }
			join_society = college_of_magic_dynamic_5
		}
		if = { limit = { ROOT = { trait = lore_metal } } set_character_flag = learning_lore_metal }
		else_if = {
			limit = { ROOT = { trait = lore_fire } }
			set_character_flag = learning_lore_fire
		}
		else_if = {
			limit = { ROOT = { trait = lore_death } }
			set_character_flag = learning_lore_death
		}
		else_if = {
			limit = { ROOT = { trait = lore_shadows } }
			set_character_flag = learning_lore_shadows
		}
		else_if = {
			limit = { ROOT = { trait = lore_beasts } }
			set_character_flag = learning_lore_beasts
		}
		else_if = {
			limit = { ROOT = { trait = lore_heavens } }
			set_character_flag = learning_lore_heavens
		}
		else_if = {
			limit = { ROOT = { trait = lore_light } }
			set_character_flag = learning_lore_light
		}
		else_if = {
			limit = { ROOT = { trait = lore_life } }
			set_character_flag = learning_lore_life
		}
		set_character_flag = learning_magic_in_college
	}
}
