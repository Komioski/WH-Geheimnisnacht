# Scope: the fasting ruler.

	clr_character_flag = holy_fast
	clr_character_flag = holy_fast_prisoners
	clr_character_flag = holy_fast_charity
	clr_character_flag = holy_fast_prayers
	clr_character_flag = holy_fast_ill
	clr_character_flag = holy_fast_djinn
	clr_character_flag = holy_fast_gluttonous
	clr_character_flag = holy_fast_lustful
	clr_character_flag = holy_fast_drunkard
	clr_character_flag = holy_fast_greedy
	clr_character_flag = holy_fast_cynical
}

# The Sultan who gave the white elephant takes offence (elephant).
# Scope: the receiving ruler.
elephant_offend_giver_effect = {
	any_playable_ruler = {
		limit = { has_character_flag = gave_white_elephant }
		add_rival = ROOT
		opinion = {
			who = ROOT
			modifier = grievously_offended
			years = 10
		}
	}
}

# The pilgrim sets out for Ind (ind.1). Schedules two road events, the
# arrival and the journey home.
# Scope: the pilgrim.
ind_set_out_effect = {
	add_trait = on_indian_pilgrimage
	set_character_flag = ind_good_pilgrimage
	random_list = {
		10 = { character_event = { id = ind.10 days = 30 random = 15 } }
		10 = { character_event = { id = ind.11 days = 30 random = 15 } }
		10 = { character_event = { id = ind.12 days = 30 random = 15 } }
		10 = { character_event = { id = ind.13 days = 30 random = 15 } }
		10 = { character_event = { id = ind.14 days = 30 random = 15 } }
	}
	random_list = {
		10 = { character_event = { id = ind.15 days = 75 random = 15 } }
		10 = { character_event = { id = ind.16 days = 75 random = 15 } }
		10 = { character_event = { id = ind.17 days = 75 random = 15 } }
		10 = { modifier = { factor = 0 is_disease_immune_race = yes } character_event = { id = ind.18 days = 75 random = 15 } }
		10 = { character_event = { id = ind.19 days = 75 random = 15 } }
	}
	narrative_event = { id = ind.30 days = 120 }
	character_event = { id = ind.50 days = 270 }
}

# The pilgrim's journey ends (ind.50).
# Scope: the pilgrim.
ind_clear_flags_effect = {
	remove_trait = on_indian_pilgrimage
	clr_character_flag = do_not_disturb
	clr_character_flag = ind_good_pilgrimage
	clr_character_flag = ind_dest_temple
	clr_character_flag = ind_dest_tiger
	clr_character_flag = ind_dest_ghats
}
