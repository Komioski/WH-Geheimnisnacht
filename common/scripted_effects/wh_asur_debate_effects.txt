# The character gives up their stance on Ulthuan's foreign policy and takes
# the other one.
# Scope: the character.
asurdebate_change_stance_effect = {
	if = {
		limit = { trait = isolationist_elf }
		remove_trait = isolationist_elf
		add_trait = interventionist_elf
	}
	else_if = {
		limit = { trait = interventionist_elf }
		remove_trait = interventionist_elf
		add_trait = isolationist_elf
	}
}

# Prestige lost to a public humiliation, by rank.
# Scope: the character.
asurdebate_tier_prestige_loss_effect = {
	if = { limit = { tier = EMPEROR } prestige = -750 }
	else_if = { limit = { tier = KING } prestige = -500 }
	else_if = { limit = { tier = DUKE } prestige = -250 }
	else_if = { limit = { tier = COUNT } prestige = -100 }
	else = { prestige = -50 }
}

# Prestige won by graciously doing a peer a service, by rank.
# Scope: the character.
asurdebate_tier_prestige_gain_effect = {
	if = { limit = { tier = EMPEROR } prestige = 750 }
	else_if = { limit = { tier = KING } prestige = 500 }
	else_if = { limit = { tier = DUKE } prestige = 250 }
	else_if = { limit = { tier = COUNT } prestige = 100 }
	else = { prestige = 50 }
}

# The two of them marry. A ruling lady keeps the children in her line.
# Scope: any. Needs the event targets asurmeet_host and asurmeet_guest.
asurmeet_wedding_effect = {
	event_target:asurmeet_host = {
		if = { limit = { is_female = yes } add_spouse_matrilineal = event_target:asurmeet_guest }
		else = { add_spouse = event_target:asurmeet_guest }
	}
}

# The guest grants what the host asked for at a private meeting: gold, a
# favour, the release of the host's debt, a divorce or a marriage.
# Scope: any. Needs the event targets asurmeet_host and asurmeet_guest.
asurmeet_grant_ask_effect = {
	event_target:asurmeet_host = {
		if = {
			limit = { has_character_flag = asurmeet_ask_gold_500 }
			wealth = 500
			event_target:asurmeet_guest = { wealth = -500 }
		}
		else_if = {
			limit = { has_character_flag = asurmeet_ask_gold_250 }
			wealth = 250
			event_target:asurmeet_guest = { wealth = -250 }
		}
		else_if = {
			limit = { has_character_flag = asurmeet_ask_gold_100 }
			wealth = 100
			event_target:asurmeet_guest = { wealth = -100 }
		}
		else_if = {
			limit = { has_character_flag = asurmeet_ask_favor }
			add_favor = event_target:asurmeet_guest
		}
		else_if = {
			limit = { has_character_flag = asurmeet_ask_clear }
			event_target:asurmeet_guest = { remove_favor = event_target:asurmeet_host }
		}
		else_if = {
			limit = { has_character_flag = asurmeet_ask_divorce }
			event_target:asurmeet_guest = {
				prestige = -100
				piety = -100
				spouse = {
					save_event_target_as = asurmeet_old_spouse
					remove_spouse = event_target:asurmeet_guest
				}
			}
			event_target:asurmeet_old_spouse = { character_event = { id = asurdebate.18 } }
		}
		else_if = {
			limit = { has_character_flag = asurmeet_ask_marriage }
			asurmeet_wedding_effect = yes
		}
	}
}

# The host pays the guest's price: 200 gold, a favour (or the guest's own
# debt to the host is forgiven), or marriage.
# Scope: any. Needs the event targets asurmeet_host and asurmeet_guest.
asurmeet_grant_counter_effect = {
	event_target:asurmeet_guest = {
		if = {
			limit = { has_character_flag = asurmeet_counter_gold }
			wealth = 200
			event_target:asurmeet_host = { wealth = -200 }
		}
		else_if = {
			limit = {
				has_character_flag = asurmeet_counter_favor
				owes_favor_to = event_target:asurmeet_host
			}
			event_target:asurmeet_host = { remove_favor = event_target:asurmeet_guest }
		}
		else_if = {
			limit = { has_character_flag = asurmeet_counter_favor }
			add_favor = event_target:asurmeet_host
		}
		else_if = {
			limit = { has_character_flag = asurmeet_counter_marriage }
			asurmeet_wedding_effect = yes
		}
	}
}

# Clears every flag of a private meeting.
# Scope: the host or the guest.
asurmeet_clear_flags_effect = {
	clr_character_flag = asurmeet_ask_gold_100
	clr_character_flag = asurmeet_ask_gold_250
	clr_character_flag = asurmeet_ask_gold_500
	clr_character_flag = asurmeet_ask_favor
	clr_character_flag = asurmeet_ask_clear
	clr_character_flag = asurmeet_ask_divorce
	clr_character_flag = asurmeet_ask_marriage
	clr_character_flag = asurmeet_counter_gold
	clr_character_flag = asurmeet_counter_favor
	clr_character_flag = asurmeet_counter_marriage
	clr_character_flag = asurmeet_deal_done
}
