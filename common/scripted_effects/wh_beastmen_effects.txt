# The warherd declares war on the province's top liege.
# Scope: the chieftain. ROOT is the province.
beastmen_warherd_war_effect = {
	ROOT = {
		owner = {
			top_liege = {
				reverse_war = {
					target = PREVPREVPREV
					casus_belli = beastmen_uprising
					thirdparty_title = ROOT # The county
				}
				reverse_opinion = { who = PREVPREVPREV modifier = opinion_prey }
			}
		}
	}
}

# Kills the scoped character, murdered by ROOT. Invincible characters survive.
# Scope: the victim.
beastmen_slay_effect = {
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		death = { death_reason = death_murder killer = ROOT }
	}
}

# Raiders butcher up to five courtiers who are not beastmen, each with a 75%
# chance. ROOT is the killer.
# Scope: the ruler whose court is raided.
beastmen_slaughter_court_effect = {
	random_courtier = {
		limit = { beastmen_raid_victim_trigger = yes }
		set_character_flag = beastmen_raid_victim
		random = { chance = 75 beastmen_slay_effect = yes }
	}
	random_courtier = {
		limit = { beastmen_raid_victim_trigger = yes }
		set_character_flag = beastmen_raid_victim
		random = { chance = 75 beastmen_slay_effect = yes }
	}
	random_courtier = {
		limit = { beastmen_raid_victim_trigger = yes }
		set_character_flag = beastmen_raid_victim
		random = { chance = 75 beastmen_slay_effect = yes }
	}
	random_courtier = {
		limit = { beastmen_raid_victim_trigger = yes }
		set_character_flag = beastmen_raid_victim
		random = { chance = 75 beastmen_slay_effect = yes }
	}
	random_courtier = {
		limit = { beastmen_raid_victim_trigger = yes }
		set_character_flag = beastmen_raid_victim
		random = { chance = 75 beastmen_slay_effect = yes }
	}
	any_courtier = {
		limit = { has_character_flag = beastmen_raid_victim }
		clr_character_flag = beastmen_raid_victim
	}
}

# A herdstone ritual or a scourging leaves its mark: the scourging level of
# the province goes up one step (1 to 3).
# Scope: the province.
beastmen_scourge_province_effect = {
	if = {
		limit = { has_province_modifier = beastmen_scourging_2 }
		remove_province_modifier = beastmen_scourging_2
		add_province_modifier = { name = beastmen_scourging_3 duration = 4000 }
	}
	else_if = {
		limit = { has_province_modifier = beastmen_scourging_1 }
		remove_province_modifier = beastmen_scourging_1
		add_province_modifier = { name = beastmen_scourging_2 duration = 3600 }
	}
	else_if = {
		limit = { NOT = { has_province_modifier = beastmen_scourging_3 } }
		add_province_modifier = { name = beastmen_scourging_1 duration = 3200 }
	}
}

# The marshal's work is done: the province becomes beastman land with a
# herdstone. The capital turns nomadic, one other holding is razed and the
# scourging ends.
# Scope: the province.
beastmen_raise_herdstone_effect = {
	capital_holding = { convert_to = NOMAD }
	random_province_holding = { limit = { is_capital = no } destroy_settlement = THIS }
	remove_province_modifier = beastmen_scourging_1
	remove_province_modifier = beastmen_scourging_2
	remove_province_modifier = beastmen_scourging_3
	add_province_modifier = { name = beastman_herdstone duration = -1 }
	add_province_modifier = { name = province_scourged duration = 4650 }
}

# Challenges the liege to a duel for the herd through the duel engine
# (HFP.10095), and tells the liege (beastmen.21). Moved here unchanged from
# decision duel_beastman_decision so beastmen.23 can use it too.
# Scope: the challenger.
beastmen_challenge_liege_effect = {
	add_character_modifier = {
		modifier = recent_duel_timer
		duration = 720
		hidden = yes
	}
	save_event_target_as = combatant_1
	set_character_flag = duel
	set_character_flag = flag_duel_no_draws
	set_character_flag = deadly_duel@event_target:combatant_2
	liege = {
		save_event_target_as = combatant_2
		set_character_flag = duel
		set_character_flag = flag_duel_no_draws
		character_event = { id = beastmen.21 }
		hidden_tooltip = { character_event = { id = HFP.10095 } }
	}
}

# Declares a beastman_migration war on the de jure liege of a civilised
# province next to the realm. Check beastmen_has_prey_trigger first.
# Scope: the beastman ruler (ROOT).
beastmen_attack_prey_effect = {
	random_realm_province = {
		limit = {
			any_neighbor_province = {
				owner = { NOT = { trait = creature_beastman } NOT = { is_liege_or_above = ROOT } }
			}
		}
		random_neighbor_province = {
			limit = {
				owner = { NOT = { trait = creature_beastman } NOT = { is_liege_or_above = ROOT } }
			}
			dejure_liege_title = {
				owner = {
					ROOT = {
						war = {
							casus_belli = beastman_migration
							target = PREV
							thirdparty_title = PREVPREV
						}
					}
				}
			}
		}
	}
}
