# A lord who pledges adds prestige and piety to the war chest by rank.
# Scope: the pledging lord.
bcrusade_pledge_pot_effect = {
	trigger_switch = {
		on_trigger = tier
		EMPEROR = { add_to_crusade_prestige_pot = 2000 add_to_crusade_piety_pot = 800 }
		KING = { add_to_crusade_prestige_pot = 1000 add_to_crusade_piety_pot = 400 }
		DUKE = { add_to_crusade_prestige_pot = 500 add_to_crusade_piety_pot = 200 }
		COUNT = { add_to_crusade_prestige_pot = 250 add_to_crusade_piety_pot = 100 }
		BARON = { add_to_crusade_prestige_pot = 125 add_to_crusade_piety_pot = 100 }
	}
}

# crusadefl.4 The merchant caravan: pick the piece of regalia the caravan
# offers, by the faith of the land and the commander's rank. Sets one
# crusadefl_ware_* flag. Scope: the commander, with event target
# current_location.
crusadefl_caravan_pick_effect = {
	if = {
		limit = {
			event_target:current_location = {
				OR = {
					religion_group = old_world_group
					religion_group = stromfels_group
					religion_group = halfling_gods_group
				}
			}
		}
		random_list = {
			10 = { set_character_flag = crusadefl_ware_crown_of_majesty }
			10 = { set_character_flag = crusadefl_ware_emerald_scepter }
			10 = { set_character_flag = crusadefl_ware_sword_of_heroes }
			10 = { set_character_flag = crusadefl_ware_crown_of_pearls }
			10 = { set_character_flag = crusadefl_ware_ruby_scepter }
			10 = { set_character_flag = crusadefl_ware_golden_sword }
			10 = { trigger = { NOT = { higher_tier_than = DUKE } } set_character_flag = crusadefl_ware_the_lily_crown }
			10 = { trigger = { NOT = { higher_tier_than = DUKE } } set_character_flag = crusadefl_ware_noble_scepter }
			10 = { trigger = { NOT = { higher_tier_than = DUKE } } set_character_flag = crusadefl_ware_engraved_sword }
		}
	}
	else_if = {
		limit = {
			event_target:current_location = {
				OR = {
					religion_group = ormazd_gods_group
					religion_group = nehekharan_gods_group
					religion_group = old_ones_group
				}
			}
		}
		random_list = {
			10 = { set_character_flag = crusadefl_ware_dagger_of_glory }
			10 = { set_character_flag = crusadefl_ware_ruby_bracelets }
			10 = { set_character_flag = crusadefl_ware_bejeweled_dagger }
			10 = { set_character_flag = crusadefl_ware_golden_bracelets }
			10 = { trigger = { NOT = { higher_tier_than = DUKE } } set_character_flag = crusadefl_ware_engraved_dagger }
			10 = { trigger = { NOT = { higher_tier_than = DUKE } } set_character_flag = crusadefl_ware_silver_bracelets }
		}
	}
	else_if = {
		limit = {
			event_target:current_location = {
				OR = {
					religion_group = cadai_group
					religion_group = cytharai_group
				}
			}
		}
		random_list = {
			10 = { set_character_flag = crusadefl_ware_circlet_of_the_heavens }
			10 = { set_character_flag = crusadefl_ware_staff_of_majesty }
			10 = { set_character_flag = crusadefl_ware_circlet_of_the_sun }
			10 = { set_character_flag = crusadefl_ware_golden_staff }
			10 = { trigger = { NOT = { higher_tier_than = DUKE } } set_character_flag = crusadefl_ware_circlet_of_twilight }
			10 = { trigger = { NOT = { higher_tier_than = DUKE } } set_character_flag = crusadefl_ware_silver_staff }
		}
	}
	else_if = {
		limit = {
			event_target:current_location = {
				religion_group = oriental_gods_group
			}
		}
		random_list = {
			10 = { set_character_flag = crusadefl_ware_celestial_necklace }
			10 = { set_character_flag = crusadefl_ware_necklace_of_radiance }
			10 = { trigger = { NOT = { higher_tier_than = DUKE } } set_character_flag = crusadefl_ware_collar_of_pearls }
		}
	}
	else = {
		random_list = {
			10 = { set_character_flag = crusadefl_ware_crown_of_crows }
			10 = { set_character_flag = crusadefl_ware_crown_of_the_sea }
			10 = { set_character_flag = crusadefl_ware_heros_crown }
			10 = { set_character_flag = crusadefl_ware_axe_of_the_ancestors }
			10 = { set_character_flag = crusadefl_ware_amber_crown }
			10 = { set_character_flag = crusadefl_ware_obsidian_axe }
			10 = { trigger = { NOT = { higher_tier_than = DUKE } } set_character_flag = crusadefl_ware_iron_crown }
			10 = { trigger = { NOT = { higher_tier_than = DUKE } } set_character_flag = crusadefl_ware_engraved_crown }
			10 = { trigger = { NOT = { higher_tier_than = DUKE } } set_character_flag = crusadefl_ware_ceremonial_axe }
		}
	}
}
