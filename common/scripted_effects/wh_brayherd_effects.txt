# Uses up one road event and rolls the next one (or the arrival) in two to
# three months.
# Scope: the pilgrim.
brayherd_return_to_road_effect = {
	hidden_effect = {
		subtract_variable = { which = brayherd_road_events value = 1 }
		character_event = { id = brayherd.3 days = 60 random = 30 }
	}
}

# Uses up one event at the Heart of the Dark and rolls the next one (or the
# rite at the Herdstone) in about six weeks.
# Scope: the pilgrim.
brayherd_return_to_heart_effect = {
	hidden_effect = {
		subtract_variable = { which = brayherd_heart_events value = 1 }
		character_event = { id = brayherd.30 days = 40 random = 10 }
	}
}

# Ends the pilgrimage: the pilgrim is home, can be disturbed again and can
# never make the journey a second time. Clears every flag and variable the
# journey used.
# Scope: the pilgrim.
brayherd_pilgrimage_end_effect = {
	remove_trait = travelling
	set_character_flag = brayherd_pilgrimage_done
	clr_character_flag = brayherd_on_pilgrimage
	clr_character_flag = went_on_kow_tow_pilgrimage
	clr_character_flag = do_not_disturb
	clr_character_flag = brayherd_had_rain
	clr_character_flag = brayherd_wrote_home
	clr_character_flag = brayherd_met_merchant
	clr_character_flag = brayherd_was_hosted
	clr_character_flag = brayherd_hunted
	clr_character_flag = brayherd_befriended_hunter
	clr_character_flag = brayherd_killed_hunter
	clr_character_flag = brayherd_saw_shrine
	clr_character_flag = brayherd_fought_ring
	clr_character_flag = brayherd_saw_market
	clr_character_flag = brayherd_rite_bad
	clr_character_flag = brayherd_rite_okay
	clr_character_flag = brayherd_rite_great
	set_variable = { which = brayherd_road_events value = 0 }
	set_variable = { which = brayherd_heart_events value = 0 }
	set_variable = { which = brayherd_herdstone_favour value = 0 }
}

# Gives a travelling trader one beastman artifact the pilgrim does not own
# yet, saved as event_target:brayherd_trader_artifact.
# Scope: the trader. ROOT is the pilgrim.
brayherd_trader_stock_effect = {
	random_list = {

			add_artifact = beastmen_artifact_beast_banner

			add_artifact = beastmen_artifact_dark_heart

			trigger = { ROOT = { NOT = { has_artifact = beastmen_artifact_shard_herd_stone } } }
			add_artifact = beastmen_artifact_shard_herd_stone
		}
		10 = {
			trigger = { ROOT = { NOT = { has_artifact = beastmen_artifact_dark_rain_chalice } } }
			add_artifact = beastmen_artifact_dark_rain_chalice

		10 = {

		}
		10 = {
			trigger = { ROOT = { NOT = { has_artifact = beastmen_artifact_fur_sharrgu } } }

# Removes a trader (and whatever they still carry) from the game.
# Scope: the trader.
brayherd_trader_leaves_effect = {
	any_artifact = { destroy_artifact = yes }
	if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
	else = { death = { death_reason = death_missing } }
}
