# Ends a recruitment: clears the flags that say which cult is courting the
# character.
# Scope: the recruit.
chaoscult_clear_recruit_flags_effect = {
	clr_character_flag = being_recruited_by_khorne
	clr_character_flag = being_recruited_by_nurgle
	clr_character_flag = being_recruited_by_tzeentch
	clr_character_flag = being_recruited_by_slaanesh
	clr_character_flag = being_recruited_by_undivided
}

# Finds the grand master of the recruiter's cult and saves them as
# event_target:grand_master (used for the letter about how a recruitment
# went, chaoscult.49 and 50).
# Scope: any. Needs event_target:chaos_cult_recruiter.
chaoscult_find_grand_master_effect = {
	event_target:chaos_cult_recruiter = {
		random_society_member = {
			limit = { is_society_grandmaster = yes }
			save_event_target_as = grand_master
		}
	}
}

# No fitting cultist could come to meet the recruit (chaoscult.37), so the
# cult sends a new one: a servant shaped by their god, who joins the cult and
# takes the recruit's own secret faith. Saves them as
# event_target:chaos_cult_inductor.
# Scope: the new character. ROOT is the recruit.
chaoscult_new_inductor_effect = {
	character_event = { id = aiclean.31 } # Set racial trait
	if = {
		limit = { ROOT = { has_character_flag = being_recruited_by_khorne } }
		add_trait = strong
		add_trait = skilled_tactician
		join_society = cult_of_khorne
		set_secret_religion = khorne
	}
	else_if = {
		limit = { ROOT = { has_character_flag = being_recruited_by_nurgle } }
		add_trait = pneumonic
		nurgle_blessing_check_effect = yes
		add_trait = fortune_builder
		join_society = cult_of_nurgle
		set_secret_religion = nurgle
	}
	else_if = {
		limit = { ROOT = { has_character_flag = being_recruited_by_tzeentch } }
		add_trait = beak
		add_trait = elusive_shadow
		join_society = cult_of_tzeentch
		set_secret_religion = tzeentch
	}
	else_if = {
		limit = { ROOT = { has_character_flag = being_recruited_by_slaanesh } }
		remove_trait = ugly
		remove_trait = hunchback
		remove_trait = dwarf
		remove_trait = chaste
		add_trait = lustful
		add_trait = fair
		add_trait = elusive_shadow
		join_society = cult_of_slaanesh
		set_secret_religion = slaanesh
	}
	else = {
		add_trait = beak
		add_trait = scholarly_theologian
		join_society = cult_of_undivided
		set_secret_religion = chaos
	}
	save_event_target_as = chaos_cult_inductor
}

# The victim on the altar (FROM) may be the one a sacrifice mission asked
# for: a virgin, a holy person, a dwarf or a ruler. If so, clears that mission,
# plays its sound and sets chaoscult_sacrifice_for_quest so the event pays the
# mission reward instead of the plain one.
# Scope: the sacrificer.
chaoscult_sacrifice_quest_effect = {
	if = {
		limit = {
			has_quest = quest_the_devil_worshipers_sacrifice_virgin
			FROM = { is_virgin_trigger = yes }
		}
		clr_quest = quest_the_devil_worshipers_sacrifice_virgin
		if = {
			limit = { FROM = { is_female = yes } }
			sound_effect = devil_worshippers_sacrifice_virgin_female
		}
		else = {
			sound_effect = devil_worshippers_sacrifice_virgin_male
		}
		set_character_flag = chaoscult_sacrifice_for_quest
	}
	else_if = {
		limit = {
			has_quest = quest_the_devil_worshipers_sacrifice_holy_person
			FROM = { OR = { is_priest = yes is_ascetic_trigger = yes } }
		}
		clr_quest = quest_the_devil_worshipers_sacrifice_holy_person
		if = {
			limit = { FROM = { is_female = yes } }
			sound_effect = devil_worshippers_sacrifice_holy_woman
		}
		else = {
			sound_effect = devil_worshippers_sacrifice_holy_man
		}
		set_character_flag = chaoscult_sacrifice_for_quest
	}
	else_if = {
		limit = {
			has_quest = quest_the_devil_worshipers_sacrifice_dwarf
			FROM = { trait = dwarf }
		}
		clr_quest = quest_the_devil_worshipers_sacrifice_dwarf
		if = {
			limit = { FROM = { is_female = yes } }
			sound_effect = devil_worshippers_sacrifice_dwarf_female
		}
		else = {
			sound_effect = devil_worshippers_sacrifice_dwarf_male
		}
		set_character_flag = chaoscult_sacrifice_for_quest
	}
	else_if = {
		limit = {
			has_quest = quest_the_devil_worshipers_sacrifice_ruler
			FROM = { is_landed = yes }
		}
		clr_quest = quest_the_devil_worshipers_sacrifice_ruler
		if = {
			limit = { FROM = { is_female = yes } }
			sound_effect = devil_worshippers_sacrifice_ruler_female
		}
		else = {
			sound_effect = devil_worshippers_sacrifice_ruler_male
		}
		set_character_flag = chaoscult_sacrifice_for_quest
	}
}
