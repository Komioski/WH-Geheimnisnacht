# Clears a settlement's progress flags and counter (end of the chain).
# Scope: the settlement province.
chaoshorde_settlement_reset_effect = {
	clr_province_flag = start_bracket
	clr_province_flag = mid_bracket
	clr_province_flag = top_bracket
	clr_province_flag = development_is_done
	clr_province_flag = absence_of_governor1
	clr_province_flag = absence_of_governor2
	set_variable = { which = chaos_settlement_event_progress value = 0 }
}

# At war and without standing orders, the ruler is asked how to treat
# captured holdings (chaoshorde.38). Scope: the ruler.
chaoshorde_ask_sack_orders_effect = {
	if = {
		limit = {
			war = yes
			NOT = { chaoshorde_sack_orders_trigger = yes }
			NOT = { has_character_flag = chaoshorde_asked_orders }
		}
		set_character_flag = chaoshorde_asked_orders
		character_event = { id = chaoshorde.38 }
	}
}

# Settled (non-nomad) vassals above baron go free when the horde shatters.
# Scope: the dead warlord.
chaoshorde_free_settled_vassals_effect = {
	any_vassal = {
		limit = {
			is_nomadic = no
			higher_tier_than = BARON
		}
		set_defacto_liege = THIS
	}
}

# Creates the warlord of a breakaway province (new_character): a man of the
# heir's people or a beastman, with a culture and god fitting the part of the
# Wastes, and a gift of his god. Scope: the breakaway province.
chaoshorde_create_breakaway_warlord_effect = {
	random_list = {
		10 = { create_random_diplomat = { random_traits = yes culture = event_target:nomadic_heir religion = event_target:nomadic_heir } }

		10 = { create_random_intriguer = { random_traits = yes culture = event_target:nomadic_heir religion = event_target:nomadic_heir } }
		20 = { create_random_steward = { random_traits = yes culture = event_target:nomadic_heir religion = event_target:nomadic_heir } }
		15 = { create_random_priest = { random_traits = yes culture = event_target:nomadic_heir religion = event_target:nomadic_heir } }
		15 = { create_random_intriguer = { random_traits = yes culture = beastman_main religion = event_target:nomadic_heir } }
		10 = { create_random_soldier = { random_traits = yes culture = beastman_main religion = event_target:nomadic_heir } }
	}
	# The people and gods of each part of the Wastes
	if = {
		limit = { new_character = { culture_group = beastman_group } }
		new_character = {
			random_list = {
				30 = { religion = chaos }
				15 = { religion = khorne }

				12 = { religion = tzeentch }
				15 = { religion = nurgle }

			}
		}
	}
	else_if = {
		limit = {
			OR = {
				region = world_chaos_wastes_eastern_steppes
				region = world_chaos_wastes_zanbaijin
			}
		}
		new_character = {
			random_list = {
				20 = { culture = khazag }
				25 = { culture = yusak }
				20 = { culture = tokmar }
				20 = { culture = kul }
				5 = { culture = dolgan }
				10 = { }
			}
			random_list = {

				20 = { religion = khorne }
				10 = { religion = slaanesh }
				12 = { religion = tzeentch }
				20 = { religion = nurgle }
				10 = { }
			}
		}
	}
	else_if = {
		limit = { region = world_chaos_wastes_broken_lands }
		new_character = {
			random_list = {
				20 = { culture = aghol }
				25 = { culture = gharhar }

				10 = { religion = nurgle }

			}
		}
	}
	else_if = {
		limit = { region = world_chaos_wastes_far_wastes }
		new_character = {
			random_list = {
				35 = { culture = mung }
				20 = { culture = aghol }
				25 = { culture = chi_an }
				5 = { culture = kvellig }
				20 = { }
			}
			random_list = {
				20 = { religion = chaos }
				25 = { religion = khorne }
				20 = { religion = slaanesh }
				25 = { religion = tzeentch }
				10 = { religion = nurgle }
				10 = { }
			}
		}
	}
	# The god's gift

		if = {
			limit = { religion = khorne }
			random_list = {
				30 = { add_trait = khorne_berserker }
				15 = { add_trait = champion_khorne }
				10 = { add_trait = chaos_knight }
				5 = { add_trait = chosen_khorne }
				40 = { }
			}
		}
		else_if = {
			limit = { religion = tzeentch }
			random_list = {
				20 = { add_trait = chaos_knight }
				10 = { add_trait = champion_tzeentch }
				10 = {

# Every dwarf in the conqueror's realm is furious at the fall of Karag Dum.
# Scope: the conqueror.
chaoshorde_karak_dum_grudge_effect = {
	hidden_effect = {
		any_realm_character = {
			limit = { religion_group = dwarf_gods_group }
			opinion = { modifier = opinion_furious who = ROOT years = 200 }
		}
	}
}

# The four counties of Karag Dum are resettled by the tribes of the four
# gods. Scope: anyone.
chaoshorde_karak_dum_resettle_effect = {
	c_karak_dum = { location = { culture = kul religion = nurgle } }
	c_rodmarin = { location = { culture = baersonling religion = khorne } }
	c_grongol_norn = { location = { culture = hastling religion = tzeentch } }
	c_grung_dwor = { location = { culture = kul religion = chaos } }
}

# The ruler of Jong On, if not a Chaos worshipper, must kneel to the
# conqueror of Karag Dum. Scope: anyone; ROOT is the conqueror.
chaoshorde_karak_dum_jong_on_effect = {
	if = {
		limit = { c_jong_on = { holder_scope = { NOT = { religion_group = chaos_gods_group } } } }
		c_jong_on = { holder_scope = { set_defacto_liege = ROOT } }
	}
}
