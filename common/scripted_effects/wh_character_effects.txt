# A drunkard gives up the drink: no longer a drunkard, and either no longer
# gluttonous or newly temperate.
# Scope: the drunkard.
charfl_quit_drinking_effect = {
	remove_trait = drunkard
	if = {
		limit = { trait = gluttonous }
		remove_trait = gluttonous
	}
	else_if = {
		limit = { NOT = { trait = temperate } }
		add_trait = temperate
	}
}

# Freckles at the start of a game (charfl.31): three in ten of a freckled
# people get them, the thin kinds most often.
# Scope: the character.
charfl_random_freckles_effect = {
	random_list = {
		2 = { add_trait = freckles_5 }
		6 = { add_trait = freckles_4 }
		7 = { add_trait = freckles }
		7 = { add_trait = freckles_2 }
		10 = { add_trait = freckles_3 }
		70 = { }
	}
}

# The electors remember a gifted general sent away: the better the general
# (martial 12 and up), the more the liege's credibility suffers.
# Scope: the dismissed commander or marshal.
realm_discharged_hero_effect = {
	if = {
		limit = { martial = 28 }
		liege = { change_variable = { which = byzantine_credibility_discharged_hero value = -10 } }
	}
	else_if = {
		limit = { martial = 24 }
		liege = { change_variable = { which = byzantine_credibility_discharged_hero value = -5 } }
	}
	else_if = {
		limit = { martial = 20 }
		liege = { change_variable = { which = byzantine_credibility_discharged_hero value = -4 } }
	}
	else_if = {
		limit = { martial = 18 }
		liege = { change_variable = { which = byzantine_credibility_discharged_hero value = -3 } }
	}
	else_if = {
		limit = { martial = 14 }
		liege = { change_variable = { which = byzantine_credibility_discharged_hero value = -2 } }
	}
	else = {
		liege = { change_variable = { which = byzantine_credibility_discharged_hero value = -1 } }
	}
}

# The secret bear chain stops (the bear left court, died or was locked up).
# Scope: the ruler.
charfl_bear_end_effect = {
	clr_character_flag = secret_bear_chain
	if = {
		limit = { event_target:bear_character = { always = yes } }
		event_target:bear_character = { clr_character_flag = secret_bear }
	}
}

# Destroys the Serpent's Eye, loose or in its crown.
# Scope: whoever holds it.
gem_lose_diamond_effect = {
	if = {
		limit = { has_artifact = cursed_diamond }
		unsafe_destroy_artifact = cursed_diamond
	}
	if = {
		limit = { has_artifact = cursed_diamond_crown }
		unsafe_destroy_artifact = cursed_diamond_crown
	}
}

# Clears the owner's Serpent's Eye flags once the diamond is gone for good.
# Scope: the owner.
gem_clear_flags_effect = {
	clr_character_flag = serpents_eye_owner
	clr_character_flag = serpents_eye_crown
	clr_character_flag = serpents_eye_treasury
	clr_character_flag = serpents_eye_lust
	clr_character_flag = serpents_eye_sold
	clr_character_flag = serpents_eye_sold_crown
	any_spouse = { clr_character_flag = serpents_eye_spouse }
}

# The Red Cord rules the roads of the province for ten years
# (murdercult, events/WHG_realm_events.txt).
# Scope: the ruler; event target murdercult_province is the province.
murdercult_rampant_effect = {
	event_target:murdercult_province = {
		add_province_modifier = {
			name = thugs_rampant
			duration = 3650
		}
	}
}

# Picks how a winner's people celebrate a triumph (triumph, events/
# WHG_realm_events.txt). No flag means the default parade.
# Scope: the winner.
triumph_set_voice_effect = {
	triumph_clear_flags_effect = yes
	if = {
		limit = { religion_group = chaos_gods_group }
		set_character_flag = triumph_voice_chaos
	}
	else_if = {
		limit = { religion_group = dwarf_gods_group }
		set_character_flag = triumph_voice_dwarf
	}
	else_if = {
		limit = { religion_group = greenskin_gods_group }
		set_character_flag = triumph_voice_greenskin
	}
	else_if = {
		limit = { religion_group = cytharai_group }
		set_character_flag = triumph_voice_druchii
	}
	else_if = {
		limit = {
			OR = {
				religion_group = necromantic_group
				religion_group = nehekharan_gods_group
				is_undead_trigger = yes
			}
		}
		set_character_flag = triumph_voice_undead
	}
	else_if = {
		limit = { religion_group = horned_rat_god_group }
		set_character_flag = triumph_voice_skaven
	}
}

# Clears the triumph voice flags once the triumph is over.
# Scope: the winner.
triumph_clear_flags_effect = {
	clr_character_flag = triumph_vs_chaos
	clr_character_flag = triumph_voice_chaos
	clr_character_flag = triumph_voice_dwarf
	clr_character_flag = triumph_voice_greenskin
	clr_character_flag = triumph_voice_druchii
	clr_character_flag = triumph_voice_undead
	clr_character_flag = triumph_voice_skaven
}

# A small chance, once per lifetime, that the triumph's veterans swear
# themselves to the ruler (triumph.6).
# Scope: the winner.
triumph_veterans_effect = {
	if = {
		limit = { NOT = { has_character_flag = triumph_guard_sworn } }
		random = {
			chance = 15
			character_event = { id = triumph.6 days = 3 }
		}
	}
}
