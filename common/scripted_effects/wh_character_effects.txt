# A drunkard gives up the drink: no longer a drunkard, and either no longer
# gluttonous or newly temperate.
# Scope: the drunkard.
charfl_quit_drinking_effect = {
	remove_trait = drunkard
	if = {
		limit = { trait = gluttonous }
		remove_trait = gluttonous
	}
	else_if = {
		limit = { NOT = { trait = temperate } }
		add_trait = temperate
	}
}

# Freckles at the start of a game (charfl.31): three in ten of a freckled
# people get them, the thin kinds most often.
# Scope: the character.
charfl_random_freckles_effect = {
	random_list = {
		2 = { add_trait = freckles_5 }
		6 = { add_trait = freckles_4 }
		7 = { add_trait = freckles }
		7 = { add_trait = freckles_2 }
		10 = { add_trait = freckles_3 }
		70 = { }
	}
}

###############################################################################

# The electors remember a gifted general sent away: the better the general
# (martial 12 and up), the more the liege's credibility suffers.
# Scope: the dismissed commander or marshal.
realm_discharged_hero_effect = {
	if = {
		limit = { martial = 28 }
		liege = { change_variable = { which = byzantine_credibility_discharged_hero value = -10 } }
	}
	else_if = {
		limit = { martial = 24 }
		liege = { change_variable = { which = byzantine_credibility_discharged_hero value = -5 } }
	}
	else_if = {
		limit = { martial = 20 }
		liege = { change_variable = { which = byzantine_credibility_discharged_hero value = -4 } }
	}
	else_if = {
		limit = { martial = 18 }
		liege = { change_variable = { which = byzantine_credibility_discharged_hero value = -3 } }
	}
	else_if = {
		limit = { martial = 14 }
		liege = { change_variable = { which = byzantine_credibility_discharged_hero value = -2 } }
	}
	else = {
		liege = { change_variable = { which = byzantine_credibility_discharged_hero value = -1 } }
	}
}

# The secret bear chain stops (the bear left court, died or was locked up).
# Scope: the ruler.
charfl_bear_end_effect = {
	clr_character_flag = secret_bear_chain
	if = {
		limit = { event_target:bear_character = { always = yes } }
		event_target:bear_character = { clr_character_flag = secret_bear }
	}
}
