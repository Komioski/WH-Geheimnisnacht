# ROOT is the ruler. Walks the counties of the de jure empire of his capital
# once and flags every college found there: college_near_any plus
# college_near_<college>. Clear them with college_clear_near_effect.
college_mark_near_effect = {
	capital_scope = {
		EMPIRE = {
			any_de_jure_vassal_title = {
				limit = { tier = COUNT location = { has_province_flag = houses_college_of_magic } }
				ROOT = { set_character_flag = college_near_any }
				location = {
					if = {
						limit = { has_province_flag = houses_college_of_magic_imperial }
						ROOT = { set_character_flag = college_near_imperial }
					}
					if = {
						limit = { has_province_flag = houses_college_of_magic_dynamic_1 }
						ROOT = { set_character_flag = college_near_dynamic_1 }
					}
					if = {
						limit = { has_province_flag = houses_college_of_magic_dynamic_2 }
						ROOT = { set_character_flag = college_near_dynamic_2 }
					}
					if = {
						limit = { has_province_flag = houses_college_of_magic_dynamic_3 }
						ROOT = { set_character_flag = college_near_dynamic_3 }
					}
					if = {
						limit = { has_province_flag = houses_college_of_magic_dynamic_4 }
						ROOT = { set_character_flag = college_near_dynamic_4 }
					}
					if = {
						limit = { has_province_flag = houses_college_of_magic_dynamic_5 }
						ROOT = { set_character_flag = college_near_dynamic_5 }
					}
				}
			}
		}
	}
}

# THIS has just joined a college. If nobody leads it, he does.
college_take_lead_effect = {
	if = {
		limit = { society = { NOT = { any_society_member = { is_society_grandmaster = yes } } } }
		set_society_grandmaster = yes
	}
}

# Closes the requisition store after a purchase or on leaving.
college_reset_store_effect = { clr_character_flag = visiting_college_requisition }

# The Power Stone of THIS mage's wind (college.71).
college_powerstone_effect = {
	if = { limit = { member_of_metal_order = yes } add_artifact = goldstone }
	else_if = { limit = { member_of_fire_order = yes } add_artifact = fire_rubies }
	else_if = { limit = { member_of_death_order = yes } add_artifact = endstones }
	else_if = { limit = { member_of_shadows_order = yes } add_artifact = crystal_mist }
	else_if = { limit = { member_of_beasts_order = yes } add_artifact = ghost_amber }
	else_if = { limit = { member_of_heavens_order = yes } add_artifact = true_sapphires }
	else_if = { limit = { member_of_light_order = yes } add_artifact = lumen_stones }
	else_if = { limit = { member_of_life_order = yes } add_artifact = vitaellum }
}

# The grimoire of THIS mage's wind (college.73). Only once per lore.
college_grimoire_effect = {
	if = { limit = { member_of_metal_order = yes } add_artifact = mages_tome_metal }
	else_if = { limit = { member_of_fire_order = yes } add_artifact = mages_tome_fire }
	else_if = { limit = { member_of_death_order = yes } add_artifact = mages_tome_death }
	else_if = { limit = { member_of_shadows_order = yes } add_artifact = mages_tome_shadows }
	else_if = { limit = { member_of_beasts_order = yes } add_artifact = mages_tome_beasts }
	else_if = { limit = { member_of_heavens_order = yes } add_artifact = mages_tome_heavens }
	else_if = { limit = { member_of_light_order = yes } add_artifact = mages_tome_light }
	else_if = { limit = { member_of_life_order = yes } add_artifact = mages_tome_life }
}

# THIS is an AI Lord Magister (rank 4). If a player leads his wind as
# Patriarch, he may challenge him for the seat: 1 in 10 per five years,
# never twice against the same Patriarch within ten years (college.55).
college_ai_challenge_effect = {
	if = {
		limit = {
			society_rank == 4
			is_trained_mage_trigger = yes
			NOT = { has_character_modifier = lost_rank_up_duel_cooldown }
		}
		save_event_target_as = college_challenger
		society = {
			random_society_member = {
				limit = {
					ai = no
					society_rank == 5
					prisoner = no
					NOT = { has_character_modifier = college_challenge_cooldown }
					event_target:college_challenger = { check_has_same_lore_as_prev = yes }
				}
				random = {
					chance = 10
					add_character_modifier = {
						name = college_challenge_cooldown
						years = 10
						hidden = yes
					}
					character_event = { id = college.55 }
				}
			}
		}
	}
}

# THIS is an AI member. Once per five years (college.10) he spends college
# currency on one item, as a player would in the stores (college.70-73):
# a Power Stone from Magister up, else the best staff his rank allows, else
# robes of his Order.
college_ai_store_effect = {
	if = {
		limit = { ai = yes society_rank = 2 is_trained_mage_trigger = yes }
		if = {
			limit = {
				society_rank = 3
				has_society_currency_major_trigger = yes
				owns_an_active_powerstone = no
			}
			college_powerstone_effect = yes
			detract_society_currency_major_effect = yes
		}
		else_if = {
			limit = {
				society_rank = 4
				has_society_currency_major_trigger = yes
				NOT = { has_artifact = mages_staff_3 }
			}
			add_artifact = mages_staff_3
			detract_society_currency_major_effect = yes
		}
		else_if = {
			limit = {
				society_rank = 3
				has_society_currency_medium_trigger = yes
				NOR = { has_artifact = mages_staff_2 has_artifact = mages_staff_3 }
			}
			add_artifact = mages_staff_2
			detract_society_currency_medium_effect = yes
		}
		else_if = {
			limit = {
				has_society_currency_minor_trigger = yes
				NOT = { any_artifact = { has_artifact_flag = staff } }
			}
			add_artifact = mages_staff_1
			detract_society_currency_minor_effect = yes
		}
		else_if = {
			limit = {
				has_society_currency_minor_trigger = yes
				NOT = { any_artifact = { has_artifact_flag = mages_robes } }
			}
			if = { limit = { society_rank = 4 } add_artifact = mages_robes_3 }
			else_if = { limit = { society_rank = 3 } add_artifact = mages_robes_2 }
			else = { add_artifact = mages_robes_1 }
			detract_society_currency_minor_effect = yes
		}
	}
}

# THIS is a castle or city holding; event_target:college_house_wizard the
# wizard whose Order gets a house there. Adds the local college building
# and flags the province with his Order (establish_local_college_presence,
# college.80).
college_local_house_effect = {
	location = { set_province_flag = houses_college_of_magic_local }
	if = { limit = { holding_type = castle } add_building = ca_college_of_magic_local }
	else_if = { limit = { holding_type = city } add_building = ct_college_of_magic_local }
	location = {
		if = {
			limit = { event_target:college_house_wizard = { member_of_metal_order = yes } }
			set_province_flag = houses_college_of_magic_local_metal_order
		}
		else_if = {
			limit = { event_target:college_house_wizard = { member_of_fire_order = yes } }
			set_province_flag = houses_college_of_magic_local_fire_order
		}

			limit = { event_target:college_house_wizard = { member_of_death_order = yes } }

			set_province_flag = houses_college_of_magic_local_shadows_order
		}
