# Coronation effects, used by events/WHG_coronation_events.txt and
# events/WHG_kislev_crown_events.txt.

# The crowned ruler receives their realm's crown if it has not yet been
# made: the Crown of Sigmar for the Emperor, the Frost Crown for the ruler of
# Kislev. Made once a game; after that it passes with the throne
# (coronation.406, coronation.414). Scope: the crowned ruler.
coronation_regalia_effect = {
	if = {
		limit = {
			has_landed_title = e_the_empire
			NOT = { has_artifact = crown_of_sigmar }
			NOT = { has_global_flag = crown_of_sigmar_made }
		}
		set_global_flag = crown_of_sigmar_made
		add_artifact = crown_of_sigmar
	}
	if = {
		limit = {
			has_landed_title = e_kislev
			NOT = { has_artifact = kislev_frost_crown }
			NOT = { has_global_flag = kislev_frost_crown_made }
		}
		set_global_flag = kislev_frost_crown_made
		add_artifact = kislev_frost_crown
	}
}

# A dead Emperor or ruler of Kislev: save who inherits their throne, and
# have the crown handed on to them a day later if it went to someone else.
# Scope: the dead ruler (coronation.406).
coronation_crown_death_effect = {
	if = {
		limit = {
			has_landed_title = e_the_empire
			has_artifact = crown_of_sigmar
			e_the_empire = { current_heir = { always = yes } }
		}
		e_the_empire = { current_heir = { save_event_target_as = crown_heir_empire } }
	}
	if = {
		limit = {
			has_landed_title = e_kislev
			has_artifact = kislev_frost_crown
			e_kislev = { current_heir = { always = yes } }
		}
		e_kislev = { current_heir = { save_event_target_as = crown_heir_kislev } }
	}
	player_heir = { character_event = { id = coronation.407 days = 1 } }
}

# The player heir passes a crown they inherited to the heir of its throne.
# Scope: the player heir; event targets crown_heir_empire, crown_heir_kislev
# (coronation_crown_death_effect).
coronation_crown_hand_on_effect = {
	if = {
		limit = {
			has_artifact = crown_of_sigmar
			event_target:crown_heir_empire = { NOT = { character = ROOT } }
		}
		any_artifact = {
			limit = { artifact_type = crown_of_sigmar }
			transfer_artifact = { from = ROOT to = event_target:crown_heir_empire }
		}
	}
	if = {
		limit = {
			has_artifact = kislev_frost_crown
			event_target:crown_heir_kislev = { NOT = { character = ROOT } }
		}
		any_artifact = {
			limit = { artifact_type = kislev_frost_crown }
			transfer_artifact = { from = ROOT to = event_target:crown_heir_kislev }
		}
	}
}

# Strips every claim on a high seat of the cults from the pretenders of a
# beaten realm. Scope: the defeated ruler.
coronation_strip_pretenders_effect = {
	if = {
		limit = { coronation_cult_claim_trigger = yes }
		remove_claim = k_cult_of_sigmar
		remove_claim = k_cult_of_ulric
		remove_claim = k_cult_of_taal_and_rhya
		remove_claim = k_cult_of_morr
	}
	any_realm_character = {
		limit = { coronation_cult_claim_trigger = yes }
		remove_claim = k_cult_of_sigmar
		remove_claim = k_cult_of_ulric
		remove_claim = k_cult_of_taal_and_rhya
		remove_claim = k_cult_of_morr
		prestige = -100
	}
}

# The target of a war demand is gone, so the head of the cult takes gold
# instead and the crowning goes ahead. Scope: the ruler; FROM is the head of
# the cult (coronation.214, coronation.215, coronation.216).
coronation_war_demand_paid_in_gold_effect = {
	set_character_flag = flag_crowned_by_pope
	transfer_scaled_wealth = { to = FROM value = 2 min = 300 max = 1200 }
	FROM = {
		save_event_target_as = coronation_priest
		show_scope_change = no
		opinion = { modifier = opinion_supposed_to_crown months = 1 who = ROOT }
		reverse_opinion = { modifier = opinion_supposed_to_crown months = 1 who = ROOT }
	}
	hidden_tooltip = { character_event = { id = coronation.300 } } # Go to the budget
}

# The holy site is back in the realm: the head of the cult will crown the
# ruler. Scope: the ruler, with event target coronation_priest
# (coronation.221).
coronation_holy_site_crowning_effect = {
	set_character_flag = flag_preparing_coronation # Keeps the decision closed
	set_character_flag = flag_crowned_by_pope
	event_target:coronation_priest = {
		show_scope_change = no
		opinion = { modifier = opinion_supposed_to_crown months = 1 who = ROOT }
		reverse_opinion = { modifier = opinion_supposed_to_crown months = 1 who = ROOT }
	}
	hidden_tooltip = { character_event = { id = coronation.300 } } # Go to the budget
}
