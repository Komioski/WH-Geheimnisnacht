# Court effects, used by events/WHG_court_events.txt.

# The liege appoints the petitioner to the seat in question and the old
# holder is sacked (court.22, 25).
# Scope: the liege; event targets court_petitioner, court_holder.
court_petition_appoint_effect = {
	if = {
		limit = { has_character_flag = court_petition_marshal }
		job_marshal = {
			opinion = { modifier = opinion_fired_from_council who = ROOT years = 5 }
			remove_title = job_marshal
		}
		event_target:court_petitioner = { give_job_title = job_marshal }
	}
	else_if = {
		limit = { has_character_flag = court_petition_spymaster }
		job_spymaster = {
			opinion = { modifier = opinion_fired_from_council who = ROOT years = 5 }
			remove_title = job_spymaster
		}
		event_target:court_petitioner = { give_job_title = job_spymaster }
	}
	else_if = {
		limit = { has_character_flag = court_petition_treasurer }
		job_treasurer = {
			opinion = { modifier = opinion_fired_from_council who = ROOT years = 5 }
			remove_title = job_treasurer
		}
		event_target:court_petitioner = { give_job_title = job_treasurer }
	}
	else_if = {
		limit = { has_character_flag = court_petition_chancellor }
		job_chancellor = {
			opinion = { modifier = opinion_fired_from_council who = ROOT years = 5 }
			remove_title = job_chancellor
		}
		event_target:court_petitioner = { give_job_title = job_chancellor }
	}
	else_if = {
		limit = { has_character_flag = court_petition_spiritual }
		job_spiritual = {
			opinion = { modifier = opinion_fired_from_council who = ROOT years = 5 }
			remove_title = job_spiritual
		}
		event_target:court_petitioner = { give_job_title = job_spiritual }
	}
	event_target:court_petitioner = {
		clr_character_flag = asked_for_job
		hidden_effect = { letter_event = { id = court.26 } }
	}
	court_petition_clear_effect = yes
}

# The liege refuses the petition (court.22, 25).
# Scope: the liege; event target court_petitioner.
court_petition_refuse_effect = {
	event_target:court_petitioner = {
		prestige = -10
		opinion = { who = ROOT modifier = opinion_refused_council }
	}
	court_petition_clear_effect = yes
}

# Clears the liege's petition flags.
# Scope: the liege.
court_petition_clear_effect = {
	clr_character_flag = court_petition_open
	clr_character_flag = court_petition_marshal
	clr_character_flag = court_petition_spymaster
	clr_character_flag = court_petition_treasurer
	clr_character_flag = court_petition_chancellor
	clr_character_flag = court_petition_spiritual
}
