# Crusade effects, used by events/WHG_crusade_events.txt and the southern
# crusade CBs (common/cb_types/ka_cb_types.txt, 01_cb_types.txt).

# Settle whom the Knights of Antoch march on after the asker pays their price:
# the ruler the asker named if alive, else the lord of the named duchy, else
# any Arabyan count in it. Scope: the asker (crusade.16 - crusade.18).
crusade_south_pick_target_effect = {
	if = {
		limit = { event_target:northern_crusade_potential_target = { is_alive = yes } }
		event_target:northern_crusade_potential_target = {
			save_event_target_as = northern_crusade_target_character
		}
	}
	else_if = {
		limit = {
			event_target:northern_crusade_duchy_title = {
				owner = { religion_group = ormazd_gods_group }
			}
		}
		event_target:northern_crusade_duchy_title = {
			owner = { save_event_target_as = northern_crusade_target_character }
		}
	}
	else_if = {
		limit = {
			event_target:northern_crusade_duchy_title = {
				any_direct_de_jure_vassal_title = { owner = { religion_group = ormazd_gods_group } }
			}
		}
		event_target:northern_crusade_duchy_title = {
			random_direct_de_jure_vassal_title = {
				limit = { owner = { religion_group = ormazd_gods_group } }
				owner = { save_event_target_as = northern_crusade_target_character }
			}
		}
	}
}

# Tell the master of the Knights of Antoch to march (crusade.19).
# Scope: any.
crusade_south_call_order_effect = {
	if = {
		limit = { is_title_active = d_knights_of_antoch }
		d_knights_of_antoch = { holder_scope = { character_event = { id = crusade.19 } } }
	}
	else_if = {
		limit = { is_title_active = k_knights_of_antoch }
		k_knights_of_antoch = { holder_scope = { character_event = { id = crusade.19 } } }
	}
}

# The Errantry War is won: the kingdom is the Lady's. Needs event_target
# northern_crusade_kingdom. Scope: the master of the order (crusade.26 and
# the southern_crusade_war CB).
crusade_south_victory_effect = {
	event_target:northern_crusade_kingdom = { clr_title_flag = northern_crusade_active_target }
	religion_authority = { modifier = northern_crusade_won }
	narrative_event = { id = crusade.25 }
	any_playable_ruler = {
		limit = { NOT = { character = ROOT } }
		narrative_event = { id = crusade.25 }
	}
	set_global_flag = northern_crusade_success
	clr_global_flag = northern_crusades_active
}

# The Errantry War is lost. Scope: the master of the order (the
# southern_crusade_war CB); FROM is the defender.
crusade_south_failure_effect = {
	narrative_event = { id = crusade.23 }
	FROM = { narrative_event = { id = crusade.23 } }
	any_playable_ruler = {
		limit = { NOR = { character = ROOT character = FROM } }
		narrative_event = { id = crusade.23 }
	}
}

# An Errantry War qualifies every knight who rides in it: each Knight Errant
# of the Knights of Bretonnia on the winning side, and in the winners'
# realms, becomes a Knight of the Realm. Scope: the attacker
# (southern_crusade_war on_success).
crusade_south_knights_errant_effect = {
	custom_tooltip = {
		text = crusade_south_knights_errant_TT
		any_attacker = {
			if = {
				limit = { crusade_south_knight_errant_trigger = yes }
				crusade_south_raise_knight_effect = yes
			}
			any_realm_character = {
				limit = { crusade_south_knight_errant_trigger = yes }
				crusade_south_raise_knight_effect = yes
			}
		}
	}
}

# Scope: a Knight Errant.
crusade_south_raise_knight_effect = {
	society_rank_up = 1
	clr_character_flag = has_sent_request_to_rank_up
	remove_trait = bretonnian_knight_errant
	add_trait = bretonnian_knight
	prestige = 50
}

# The diverted Errantry War is over, won or lost: clear its flags and let the
# knights sail for Araby again in a year or two. Scope: any.
crusade_diverted_cleanup_effect = {
	any_character = {
		limit = {
			OR = {
				has_character_flag = diverted_errantry_attacker
				has_character_flag = diverted_errantry_target
				has_character_flag = diverted_errantry_creditor
			}
		}
		clr_character_flag = diverted_errantry_attacker
		clr_character_flag = diverted_errantry_target
		clr_character_flag = diverted_errantry_creditor
	}
	c_miragliano = { clr_title_flag = diverted_errantry_port }
	c_remas = { clr_title_flag = diverted_errantry_port }
	c_luccini = { clr_title_flag = diverted_errantry_port }
	810 = { province_event = { id = crusade.13 days = 365 random = 365 } } # Couronne
}

# Save the port the diverted Errantry War will sack, its county and the
# ruler the war is declared on. Scope: Miragliano, Remas or Luccini.
crusade_diverted_save_port_effect = {
	save_event_target_as = diverted_port
	county = { save_event_target_as = diverted_port_title }
	owner = {
		if = { limit = { independent = yes } save_event_target_as = diverted_target }
		else = { top_liege = { save_event_target_as = diverted_target } }
	}
}
