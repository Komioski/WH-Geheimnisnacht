# A deed pleases one Chaos god: its Warp currency rises by 250 and each of the
# three rival gods loses 200.
# Scope: a character.
daemon_favour_khorne_effect = {
	add_offmap_currency = { offmap = khorne_warp amount = 250 }
	add_offmap_currency = { offmap = nurgle_warp amount = -200 }
	add_offmap_currency = { offmap = slaanesh_warp amount = -200 }
	add_offmap_currency = { offmap = tzeentch_warp amount = -200 }

# A prisoner has been fed to a daemon. The summoner's Old World and elven
# vassals and the victim's kin hold it against the summoner.
# Scope: the victim (ROOT), FROM the summoner.
daemon_victim_outrage_effect = {
	FROM = {
		any_vassal = {
			limit = {
				OR = {
					religion = cadai
					religion = asrai_cadai
					religion_group = old_world_group
				}
			}
			opinion = { modifier = opinion_evil_sorcerer who = FROM }
		}
	}
	any_dynasty_member = {
		limit = { NOT = { character = FROM } }
		opinion = { modifier = opinion_destroyed_family who = FROM }
	}
}

# A summoning into a prisoner failed: a Chaos cult quest on that prisoner
# fails with it.
# Scope: the summoner (ROOT), FROM the prisoner.
daemon_summon_failed_effect = {
	if = {
		limit = {
			has_quest = quest_chaos_cult_demonic_possession
			quest_target = { character = FROM }
		}
		clr_quest = { id = quest_chaos_cult_demonic_possession failure = yes }
	}
}

# A daemon now wears the prisoner's body and serves the summoner. Completes a
# Chaos cult possession quest on that prisoner.
# Scope: the summoner (ROOT), FROM the new daemon.
daemon_summon_bound_effect = {
	if = {
		limit = {
			has_quest = quest_chaos_cult_demonic_possession
			quest_target = { character = FROM }
		}
		clr_quest = quest_chaos_cult_demonic_possession
		if = {
			limit = { FROM = { is_female = yes } }
			sound_effect = devil_worshippers_demonic_possession_female
		}
		else = {
			sound_effect = devil_worshippers_demonic_possession_male
		}
		add_society_currency_medium_effect = yes
	}
	add_friend = FROM
	set_character_flag = daemon_summoner
	prestige = 500
	FROM = { prisoner = no }
}

# The daemon possessing a character grows stronger: three times in five the
# possession counter rises by 5 (small), 10 (medium) or 15 (big). Otherwise the
# host fights back and it falls by 3 to 5. Sharp minds, wizards and the learned
# resist best; dull minds and the wild-gifted worst.
# Scope: the possessed character.
daemon_possession_small_effect = {
	random_list = {
		60 = {
			modifier = { factor = 1.75 trait = imbecile }
			modifier = { factor = 1.95 trait = idiot }
			modifier = { factor = 1.75 trait = slow }
			modifier = { factor = 1.5 trait = dumb }
			modifier = { factor = 1.25 trait = dim }
			modifier = { factor = 0.85 trait = bright }
			modifier = { factor = 0.75 trait = smart }
			modifier = { factor = 0.65 trait = quick }
			modifier = { factor = 0.5 trait = brilliant }
			modifier = { factor = 0.45 trait = genius }
			modifier = { factor = 1.25 trait = magic_potential_4 }
			modifier = { factor = 1.25 has_character_flag = magic_potential_flag_4 }
			modifier = { factor = 0.95 has_character_modifier = magic_power_1 }
			modifier = { factor = 0.85 has_character_modifier = magic_power_2 }
			modifier = { factor = 0.75 has_character_modifier = magic_power_3 }
			modifier = { factor = 0.65 has_character_modifier = magic_power_4 }
			modifier = { factor = 0.95 learning = 15 }
			modifier = { factor = 0.85 learning = 20 }
			modifier = { factor = 0.75 learning = 25 }
			modifier = { factor = 0.65 learning = 30 }
			change_variable = { which = "possession_counter" value = 5 }
		}
		20 = { change_variable = { which = "possession_counter" value = -3 } }
		10 = { change_variable = { which = "possession_counter" value = -4 } }
		10 = {
			modifier = { factor = 1.5 learning = 30 }
			modifier = { factor = 1.5 trait = genius }
			modifier = { factor = 1.5 has_character_modifier = magic_power_4 }
			change_variable = { which = "possession_counter" value = -5 }
		}
	}
}

# The possession ends: the trait, the counter and the god's mark are cleared.
# Scope: the possessed character.
daemon_possession_end_effect = {
	remove_trait = possessed
	set_variable = { which = "possession_counter" value = 0 }
	clr_character_flag = possessed_by_khornate
	clr_character_flag = possessed_by_nurglite
	clr_character_flag = possessed_by_slaaneshi
	clr_character_flag = possessed_by_tzeentchian
}
