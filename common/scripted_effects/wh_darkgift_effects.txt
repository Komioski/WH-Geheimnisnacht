# A summoned daemon or champion binds itself to its new master.
# Scope: the new character. ROOT is the master.
darkgift_bond_effect = {
	add_friend = ROOT
	reverse_opinion = { who = ROOT modifier = opinion_my_chinese_pet years = 100 }
	opinion = { who = ROOT modifier = opinion_servant months = 1200 }
	set_character_flag = no_court_invites
}

# Gains or loses favour with the patron. Scope: a character.

	else_if = {
		limit = { darkgift_patron_slaanesh_trigger = yes }

# The patron strips a rank: a Chosen falls back to Champion of the same god
# (Chosen and Champion traits are opposites, so the Champion trait has to be
# given back), a Champion loses the title. Scope: a Champion or Chosen.
darkgift_cast_down_effect = {
	if = {
		limit = { trait = chosen_khorne }
		remove_trait = chosen_khorne
		add_trait = champion_khorne
	}
	else_if = {
		limit = { trait = chosen_nurgle }
		remove_trait = chosen_nurgle
		add_trait = champion_nurgle
	}
	else_if = {
		limit = { trait = chosen_slaanesh }
		remove_trait = chosen_slaanesh
		add_trait = champion_slaanesh
	}
	else_if = {
		limit = { trait = chosen_tzeentch }
		remove_trait = chosen_tzeentch
		add_trait = champion_tzeentch
	}
	else_if = {
		limit = { trait = chaos_chosen }
		remove_trait = chaos_chosen
		add_trait = champion_undivided
	}
	else = {
		remove_trait = champion_khorne
		remove_trait = champion_nurgle
		remove_trait = champion_slaanesh
		remove_trait = champion_tzeentch
		remove_trait = champion_undivided
	}
}

# The patron names a marked aspirant Champion. Beastmen become Bestigors, as
# with the champion decisions. Scope: a marked aspirant.
darkgift_name_champion_effect = {
	if = {
		limit = { darkgift_patron_khorne_trigger = yes }
		if = {
			limit = { culture_group = beastman_group }
			add_trait = bestigor_khorne
			set_graphical_culture = bestigor_khorne
		}
		else = { add_trait = champion_khorne }
	}
	else_if = {
		limit = { darkgift_patron_nurgle_trigger = yes }
		if = {
			limit = { culture_group = beastman_group }
			add_trait = bestigor_nurgle
			set_graphical_culture = bestigor_nurgle
		}
		else = { add_trait = champion_nurgle }
	}
	else_if = {
		limit = { darkgift_patron_slaanesh_trigger = yes }
		if = {
			limit = { culture_group = beastman_group }
			add_trait = bestigor_slaanesh
			set_graphical_culture = bestigor_slaanesh
		}
		else = { add_trait = champion_slaanesh }
	}
	else_if = {
		limit = { darkgift_patron_tzeentch_trigger = yes }
		if = {
			limit = { culture_group = beastman_group }
			add_trait = bestigor_tzeentch
			set_graphical_culture = bestigor_tzeentch
		}
		else = { add_trait = champion_tzeentch }
	}
	else = { add_trait = champion_undivided }
}

	else_if = { limit = { darkgift_patron_nurgle_trigger = yes } mutations_table_nurgle = yes }
	else_if = { limit = { darkgift_patron_slaanesh_trigger = yes } mutations_table_slaanesh = yes }

# One step up the physique ladder: weak, fragile, soft, (none), hardy,
# tough, strong. Scope: a character.
darkgift_physique_effect = {
	if = { limit = { trait = weak } remove_trait = weak add_trait = fragile }
	else_if = { limit = { trait = fragile } remove_trait = fragile add_trait = soft }
	else_if = { limit = { trait = soft } remove_trait = soft }
	else_if = { limit = { trait = hardy } remove_trait = hardy add_trait = tough }
	else_if = { limit = { trait = tough } remove_trait = tough add_trait = strong }
	else_if = { limit = { NOT = { trait = strong } } add_trait = hardy }
	else = { health = 1 }
}

# The patron's own boon: Khorne, Nurgle and the Four make the body sturdy,
# Slaanesh perfects its beauty, Tzeentch sharpens the mind.
# Scope: a character.
darkgift_boon_effect = {
	if = {
		limit = { darkgift_patron_slaanesh_trigger = yes }
		if = { limit = { NOT = { trait = groomed } } add_trait = groomed }
		else = { add_trait = fair }
	}
	else_if = {
		limit = { darkgift_patron_tzeentch_trigger = yes }
		if = { limit = { NOT = { trait = shrewd } } add_trait = shrewd }
		else = { change_learning = 2 }
	}
	else = {
		if = { limit = { NOT = { trait = sturdy } } add_trait = sturdy }
		else = { change_martial = 2 }
	}
}
