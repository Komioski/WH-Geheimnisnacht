###############################################################################
# Dawi scripted effects
# Used by events/WHG_dwarf_events.txt (namespace dawi) and the grudge war CB.
#
# A grudge is an opinion modifier for its kind of wrong, held by the wronged
# Dawi (and the kin who take it up) against the offender; see
# wh_dawi_triggers.txt for the kinds. Before any of these effects, save the
# offender as event_target:grudge_offender.
###############################################################################

# Enter a grudge in ROOT's clan book against event_target:grudge_offender:
# call dawi_enter_<kind>_grudge_effect for the kind of wrong.
# A clansman who wronged his own clan is first cast out of it (with his
# children) and founds a clan of his own; then the grudge is entered.
# ROOT's adult kin are told and take the grudge up (dawi.3), and the offender
# is told (dawi.7, or dawi.8 when cast out).
# Scope: ROOT, the wronged Dawi.

# The first half of every dawi_enter_<kind>_grudge_effect: cast out a
# clansman, tell the offender, flag him for the death check.
dawi_grudge_offender_told_effect = {
	if = {
		limit = { event_target:grudge_offender = { dynasty = ROOT } }
		custom_tooltip = { text = dawi_grudge_cast_out_tt }
		event_target:grudge_offender = {
			dynasty = father_bastard
			recalc_succession = yes
			any_child = { limit = { dynasty = ROOT } dynasty = PREV }
			character_event = { id = dawi.8 }
		}
	}
	else = { event_target:grudge_offender = { character_event = { id = dawi.7 } } }
	event_target:grudge_offender = { set_character_flag = dawi_named_in_book }
}

# The second half: ROOT's adult kin take the grudge up (dawi.3).
dawi_grudge_kin_told_effect = {
	hidden_effect = {
		any_dynasty_member = {
			limit = {
				is_alive = yes
				is_adult = yes
				NOT = { character = ROOT }
				NOT = { character = event_target:grudge_offender }
				trait = creature_dwarf
			}
			character_event = { id = dawi.3 }
		}
	}
}

# A war.
dawi_enter_war_grudge_effect = {
	dawi_grudge_offender_told_effect = yes
	event_target:grudge_offender = {
		reverse_opinion = { who = ROOT modifier = opinion_dwarf_grudge months = 12000 }
	}
	custom_tooltip = { text = dawi_new_grudge_tt }
	dawi_grudge_kin_told_effect = yes
}

# Remembrance, and the shame of a father's open grudges is lifted.
# Scope: ROOT, the Dawi who settled it.

	add_character_modifier = { name = dawi_grudge_avenged duration = 1825 }
	remove_character_modifier = dawi_far_from_ancestors_1
	remove_character_modifier = dawi_far_from_ancestors_2
	remove_character_modifier = dawi_far_from_ancestors_3
}

# The offender died unpunished: this Dawi's grudges against him pass to
# event_target:dawi_grudge_heir. Ordinary grudges pass as they are; a Blood
# Grudge passes as a war grudge, for the blood debt was never paid.
# Scope: the Dawi. ROOT: the dead offender.
dawi_pass_grudges_to_heir_effect = {
	if = {
		limit = { has_opinion_modifier = { who = ROOT modifier = opinion_dwarf_grudge } }
		event_target:dawi_grudge_heir = {
			reverse_opinion = { who = PREV modifier = opinion_dwarf_grudge months = 12000 }
		}
	}
	if = {
		limit = { has_opinion_modifier = { who = ROOT modifier = opinion_dawi_grudge_slight } }
		event_target:dawi_grudge_heir = {
			reverse_opinion = { who = PREV modifier = opinion_dawi_grudge_slight months = 12000 }
		}
	}
	if = {
		limit = { has_opinion_modifier = { who = ROOT modifier = opinion_dawi_grudge_theft } }
		event_target:dawi_grudge_heir = {
			reverse_opinion = { who = PREV modifier = opinion_dawi_grudge_theft months = 12000 }
		}
	}
	if = {
		limit = { has_opinion_modifier = { who = ROOT modifier = opinion_dawi_grudge_oath } }
		event_target:dawi_grudge_heir = {
			reverse_opinion = { who = PREV modifier = opinion_dawi_grudge_oath months = 12000 }
		}
	}
	if = {
		limit = { dawi_holds_blood_grudge_against_root_trigger = yes }
		event_target:dawi_grudge_heir = {
			reverse_opinion = { who = PREV modifier = opinion_dwarf_grudge months = 12000 }
		}
	}
	event_target:dawi_grudge_heir = { set_character_flag = dawi_named_in_book }
}

# ROOT swears the Slayer oath: he gives up wife, lands and wealth, shaves
# his head and goes to seek his death (dawi.40, dawi.27).
# Scope: ROOT, the new Slayer.
dawi_take_slayer_oath_effect = {
	custom_tooltip = { text = dawi_become_slayer_tt }
	add_trait = dwarf_slayer
	any_spouse = { remove_spouse = PREV }
	any_consort = { remove_consort = PREV }
	remove_character_modifier = follower_of_grungni
	remove_character_modifier = follower_of_valaya
	add_character_modifier = { name = follower_of_grimnir duration = -1 }
	set_special_character_title = SLAYER_DWARF_TITLE
	hidden_effect = { set_graphical_culture = dwarf_slayer_culture }
	if = {
		limit = { is_ruler = yes current_heir = { always = yes } }
		current_heir = {
			ROOT = { transfer_scaled_wealth = { to = PREV value = 10 } }
			character_event = { id = dawi.41 days = 1 }
		}
		abdicate = yes
	}
}

# A good piece of work may raise a mason's skill: an apprentice may become a
# mason (dawi.156), a mason a master mason (dawi.166).
# Scope: ROOT, the mason.
dawi_mason_progress_effect = {
	if = {
		limit = { has_character_flag = dawi_mason_apprentice }
		random = { chance = 25 character_event = { id = dawi.156 tooltip = dawi_mason_improve_tt } }
	}
	else_if = {
		limit = { trait = dwarf_mason }
		random = { chance = 10 character_event = { id = dawi.166 tooltip = dawi_mason_improve_tt } }
	}
}
