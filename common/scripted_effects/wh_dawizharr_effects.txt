# Adds one level of slave factory to the capital holding (up to level 4).
# Scope: a ruler.
dawizharr_factory_up_effect = {
	capital_holding = {
		if = {
			limit = { holding_type = castle }
			if = {
				limit = { NOT = { has_building = ca_chaos_dwarf_factory_1 } }
				add_building = ca_chaos_dwarf_factory_1
			}
			else_if = {
				limit = { NOT = { has_building = ca_chaos_dwarf_factory_2 } }
				add_building = ca_chaos_dwarf_factory_2

				limit = { NOT = { has_building = ca_chaos_dwarf_factory_3 } }
				add_building = ca_chaos_dwarf_factory_3

				limit = { NOT = { has_building = ca_chaos_dwarf_factory_4 } }
				add_building = ca_chaos_dwarf_factory_4

			if = {

# Drags up to five courtiers of a raided court off in chains. Each has a 1 in 4
# chance to slip away.
# Scope: the raided ruler. ROOT is the slaver.
dawizharr_take_captives_effect = {
	while = {
		count = 5
		random_courtier = {
			limit = { dawizharr_captive_trigger = yes }
			random_list = {
				75 = { imprison = ROOT move_character = ROOT }
				25 = { set_character_flag = dawizharr_hid }
			}
		}
	}
	any_courtier = {
		limit = { has_character_flag = dawizharr_hid }
		clr_character_flag = dawizharr_hid
	}
}

# Throws up to five courtiers of a raided court into Hashut's furnaces.
# Scope: the raided ruler. ROOT is the slaver.
dawizharr_burn_captives_effect = {
	while = {
		count = 5
		random_courtier = {
			limit = { dawizharr_captive_trigger = yes }
			random_list = {
				75 = {
					if = {
						limit = { is_invincible_trigger = yes }
						antideath_effect = yes
						set_character_flag = dawizharr_hid
					}
					else = { death = { death_reason = death_sacrificed killer = ROOT } }
				}
				25 = { set_character_flag = dawizharr_hid }
			}
		}
	}
	any_courtier = {
		limit = { has_character_flag = dawizharr_hid }
		clr_character_flag = dawizharr_hid
	}
}

# The furnaces have been fed: lifts Hashut's demand and restarts the four-year
# clock (dawizharr_sacrifice_due_trigger).
# Scope: a ruler.
dawizharr_sacrifice_made_effect = {
	if = {
		limit = { has_character_modifier = must_sacrifice_hashut }
		remove_character_modifier = must_sacrifice_hashut
	}
	clr_character_flag = flag_sacrifice_made_hashut
	set_character_flag = flag_sacrifice_made_hashut
}
