# Effects for the debt events (Part 11 of events/WHG_realm_events.txt).

# The vassal is asked for gold: a grudge, and not again for five years.
# Scope: the vassal; ROOT the liege.
debt_vassal_asked_effect = {
	opinion = { modifier = got_money_request who = ROOT years = 2 }
	clr_character_flag = took_extra_tax
	set_character_flag = took_extra_tax
}

# If the title for sale was lost or given away since the offer, pick another
# lesser title, preferring one the buyer has a claim to.
# Scope: the liege; FROM the buyer.
debt_pick_title_again_effect = {
	if = {
		limit = { NOT = { has_landed_title = event_target:title_for_sale } }
		random_demesne_title = {
			limit = {
				lower_tier_than = ROOT
				NAND = {
					tier = COUNT
					location = { is_capital = yes }
				}
			}
			preferred_limit = {
				has_de_jure_pretension = FROM
				can_be_given_away = yes
			}
			preferred_limit = { can_be_given_away = yes }
			save_event_target_as = title_for_sale
		}
	}
}
