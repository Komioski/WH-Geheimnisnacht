# Disease effects, used by events/WHG_disease_events.txt and the
# expel_strigany decision (decisions/wh_vampire_decisions.txt).

###############################################################################
# Blame for the epidemic (blame)
###############################################################################

# Sends the Strigany out of the realm: every caravan in a realm province is
# broken up and every unlanded Strigany in the realm is banished, except the
# ruler's own close family and spouses.
# Scope: the ruler.
expel_strigany_effect = {
	any_realm_province = {
		limit = { has_province_modifier = strigany_caravan }
		remove_province_modifier = strigany_caravan
	}
	any_realm_character = {
		limit = {
			culture_group = strigany_group
			is_ruler = no
			NOT = { is_close_relative = ROOT }
			NOT = { is_married = ROOT }
		}
		set_variable = { which = "strigany_presence_v" value = 0 }
		banish = yes
	}
}

# Rare reward for burning a true Chaos cultist in the epidemic witch hunt:
# flagellant zealots swear themselves to the ruler who lit the pyre. Once
# in a life (the caller sets blame_pyre_sworn_raised).
# Scope: the ruler.
blame_pyre_sworn_effect = {
	custom_tooltip = { text = blame_pyre_sworn_tt }
	hidden_tooltip = {
		location = {
			ROOT = {
				spawn_unit = {
					province = PREV
					owner = ROOT
					troops = { light_infantry = { 150 150 } }
					earmark = blame_pyre_sworn
					attrition = 0
					maintenance_multiplier = 0.5
				}
			}
		}
	}
}

###############################################################################
# The dancing plague (dancing)
###############################################################################

# A character who caught the dancing plague tries to resist it: half the
# time it passes, else they are dragged out to dance (dancing.6). The
# greedy for pleasure give in more often.
# Scope: the character.
dancing_resist_effect = {
	random_list = {
		50 = { # It passes
			clr_character_flag = got_dancing_plague
			set_character_flag = immune_dancing_plague
		}
		50 = { # They cannot resist
			modifier = { factor = 2 trait = gluttonous }
			modifier = { factor = 2 trait = hedonist }
			modifier = { factor = 2 follower_of_slaanesh = yes }
			modifier = { factor = 0.5 trait = temperate }
			modifier = { factor = 0.5 trait = gardener }
			set_character_flag = forced_to_dance
			character_event = { id = dancing.6 days = 10 random = 10 }
		}
	}
}

# The dance is over: the trait and modifier go and the character will not
# catch the dancing plague again.
# Scope: the character.
dancing_end_effect = {
	remove_trait = dancing_plague
	if = {
		limit = { has_character_modifier = cant_stop_dancing }
		remove_character_modifier = cant_stop_dancing
	}
	clr_character_flag = got_dancing_plague
	set_character_flag = immune_dancing_plague
}
