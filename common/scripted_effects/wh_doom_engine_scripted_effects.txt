# The doom counter is the variable global_doom_counter of province 1, where
# doom.1 (events/WHG_doom_events.txt) reads it each year. The add_ and
# reduce_ effects can be called from any scope; they switch to province 1.
# check_doom_not_over_max_effect and check_doom_not_negative_effect must
# run in province 1.
add_doom_progress_tiny_effect = {
	custom_tooltip = { text = chaos_increases_tt }
	hidden_effect = {
		1 = { # the counter lives in province 1 (doom.1)
			change_variable = {
				which = "global_doom_counter"
				value = 1
			}
			check_doom_not_over_max_effect = yes
		}
	}
}

check_doom_not_over_max_effect = {
	if = {
		limit = {
			check_variable = {
				which = "global_doom_counter"
				value > 1500
			}
		}
		set_variable = {
			which = "global_doom_counter"
			value = 1500
		}
	}
}

check_doom_not_negative_effect = {
	if = {
		limit = {
			NOT = {
				check_variable = {
					which = "global_doom_counter"
					value = 1
				}
			}
	   }
	   set_variable = { which = "global_doom_counter" value = 1 }
	}
}
add_doom_everchosen_effect = {
	if = {
		limit = {
			any_realm_province = {
				NOT = { region = world_chaos_wastes }
			}
		}
		add_doom_progress_massive_effect = yes
	}
	else = {
		add_doom_progress_major_effect = yes
	}
}
add_doom_prince_effect = {
	if = {
		limit = {
			any_realm_province = {
				NOT = { region = world_chaos_wastes }
			}
		}
		random_list = {
			25 = {
				add_doom_progress_minor_effect = yes
				modifier = {
					factor = 0.3
					is_landed = no
				}
			}
			25 = {
				add_doom_progress_tiny_effect = yes
			}
			50 = { }
		}
	}
	else = {
		random_list = {
			5 = {
				add_doom_progress_minor_effect = yes
				modifier = {
					factor = 0.3
					is_landed = no
				}
			}
			45 = {
				add_doom_progress_tiny_effect = yes
			}
			50 = { }
		}
	}
}
