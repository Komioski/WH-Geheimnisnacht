# One month of mustering for the invasion of Ulthuan: a small host
# (druchii.6), a famous commander (druchii.5), a great host (druchii.4), or
# nothing this time (druchii.1 again). Able, famous rulers draw the bigger
# forces. Each of those events calls this again until the two years run
# out.
# Scope: the invader.
druchii_invasion_muster_effect = {
	random_list = {
		30 = { character_event = { id = druchii.6 days = 40 random = 15 } }
		20 = {
			modifier = { factor = 1.25 martial = 15 }
			modifier = { factor = 1.25 martial = 20 }
			modifier = { factor = 1.25 diplomacy = 15 }
			modifier = { factor = 1.25 diplomacy = 20 }
			modifier = { factor = 1.25 prestige = 1000 }
			modifier = { factor = 0.75 NOT = { martial = 10 } }
			modifier = { factor = 0.5 NOT = { martial = 7 } }
			modifier = { factor = 0.75 NOT = { diplomacy = 10 } }
			modifier = { factor = 0.5 NOT = { diplomacy = 7 } }
			modifier = { factor = 0.5 NOT = { prestige = 100 } }
			character_event = { id = druchii.5 days = 40 random = 15 }
		}
		10 = {
			modifier = { factor = 1.25 martial = 15 }
			modifier = { factor = 1.25 martial = 20 }
			modifier = { factor = 1.25 diplomacy = 15 }
			modifier = { factor = 1.25 diplomacy = 20 }
			modifier = { factor = 1.25 prestige = 1000 }
			modifier = { factor = 0.75 NOT = { martial = 10 } }
			modifier = { factor = 0.5 NOT = { martial = 7 } }
			modifier = { factor = 0.75 NOT = { diplomacy = 10 } }
			modifier = { factor = 0.5 NOT = { diplomacy = 7 } }
			modifier = { factor = 0.5 NOT = { prestige = 100 } }
			character_event = { id = druchii.4 days = 40 random = 15 }
		}
		40 = { character_event = { id = druchii.1 days = 40 random = 15 } }
	}
}

# The invasion was never launched: the host scatters and the invader is
# weakened for ten years (and cannot prepare another until then).
# Scope: the invader.
druchii_invasion_fails_effect = {
	disband_event_forces = prepared_invasion
	clr_character_flag = preparing_druchii_invasion
	remove_character_modifier = preparing_ulthuan_invasion
	prestige = -1000
	add_character_modifier = { name = attempted_ulthuan_invasion duration = 3650 }
}

# A prisoner fed to a beast in front of the court: everyone but the brave
# and the clever is cowed.
# Scope: the jailer.
druchii_beast_feast_fear_effect = {
	any_vassal = {
		limit = { NOR = { trait = brave trait = quick trait = genius } }
		opinion = { modifier = opinion_scared_into_submission who = ROOT }
	}
	any_courtier = {
		limit = { is_ruler = no NOR = { trait = brave trait = quick trait = genius } }
		opinion = { modifier = opinion_scared_into_submission who = ROOT }
	}
}

# Training in the temple of Khaine. A Witch Elf trains for ten years; a son
# stolen on Death Night (flag khaine_foundling) for twenty, from the cradle.
# Five lessons come along the way (druchii.42) and the training ends with
# druchii.44. About one in three Witch Elves and one in seven stolen sons
# die in training instead (druchii.43): the strong, brave and clever are
# more likely to live.
# Scope: the pupil.
druchii_khaine_training_effect = {
	random_list = {
		65 = {
			modifier = { factor = 3 has_character_flag = khaine_foundling }
			modifier = { factor = 1.5 trait = zealous }
			modifier = { factor = 0.75 trait = cynical }
			modifier = { factor = 1.75 trait = brave }
			modifier = { factor = 0.5 trait = craven }
			modifier = { factor = 1.75 trait = strong }
			modifier = { factor = 1.5 trait = robust }
			modifier = { factor = 1.25 trait = tall }
			modifier = { factor = 0.75 trait = weak }
			modifier = { factor = 0.75 trait = feeble }
			modifier = { factor = 0.5 trait = fragile }
			modifier = { factor = 1.85 trait = duelist }
			modifier = { factor = 1.5 trait = hunter }
			modifier = { factor = 1.5 trait = ambitious }
			modifier = { factor = 1.15 trait = cruel }
			modifier = { factor = 1.15 trait = diligent }
			modifier = { factor = 0.85 trait = slothful }
			modifier = { factor = 0.85 trait = content }
			modifier = { factor = 0.85 trait = drunkard }
			modifier = { factor = 0.85 trait = addict }
			modifier = { factor = 1.25 martial = 15 }
			modifier = { factor = 1.25 trait = quick }
			modifier = { factor = 1.35 trait = genius }
			modifier = { factor = 0.8 trait = slow }
			modifier = { factor = 0.75 trait = imbecile }
			modifier = { factor = 0.5 trait = maimed }
			modifier = { factor = 0.25 trait = blinded }
			modifier = { factor = 0.75 trait = infirm }
			modifier = { factor = 0.5 trait = hunchback }
			modifier = { factor = 0.75 trait = clubfooted }
			modifier = { factor = 0.5 trait = dwarf }
			if = {
				limit = { has_character_flag = khaine_foundling }
				character_event = { id = druchii.42 days = 3650 random = 365 }
				character_event = { id = druchii.42 days = 4380 random = 365 }
				character_event = { id = druchii.42 days = 5110 random = 365 }
				character_event = { id = druchii.42 days = 5840 random = 365 }
				character_event = { id = druchii.42 days = 6570 random = 365 }
				character_event = { id = druchii.44 days = 7300 }
			}
			else = {
				character_event = { id = druchii.42 days = 365 random = 365 }
				character_event = { id = druchii.42 days = 1095 random = 365 }
				character_event = { id = druchii.42 days = 1825 random = 365 }
				character_event = { id = druchii.42 days = 2555 random = 365 }
				character_event = { id = druchii.42 days = 3285 random = 300 }
				character_event = { id = druchii.44 days = 3650 }
			}
		}
		35 = {
			if = {
				limit = { has_character_flag = khaine_foundling }
				character_event = { id = druchii.43 days = 2920 random = 4000 }
			}
			else = { character_event = { id = druchii.43 days = 2150 random = 1500 } }
		}
	}
}
