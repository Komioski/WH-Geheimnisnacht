# What happens to one guest at the feast: a new friend (drufeast.20), a
# duel (drufeast.22), a night with another guest's spouse (drufeast.30),
# too much wine (drufeast.33, 35, 37), bandits on the road (drufeast.38),
# or nothing. The bigger the host's court, the likelier nothing happens to
# any one guest, so a great lord's feast does not drown in events.
# Scope: the guest (drufeast.10 for courtiers, drufeast.11 for vassals).
# Needs the event target feast_host.
drufeast_intrigue_roll_effect = {
	random_list = {
		10 = {
			modifier = { factor = 0 num_of_friends = 3 }
			modifier = { factor = 1.5 trait = gregarious }
			modifier = { factor = 0.5 trait = shy }
			character_event = { id = drufeast.20 days = 3 random = 20 }
		}
		6 = {
			modifier = { factor = 0 is_female = yes }
			modifier = {
				factor = 0
				NOR = { trait = duelist trait = brave trait = wroth trait = cruel }
			}
			modifier = { factor = 0 trait = craven }
			character_event = { id = drufeast.22 days = 3 random = 20 }
		}
		6 = {
			modifier = { factor = 0 is_ruler = no }
			modifier = { factor = 0 is_married = no }
			modifier = { factor = 2 trait = lustful }
			modifier = { factor = 1.5 trait = hedonist }
			modifier = { factor = 0.2 trait = chaste }
			character_event = { id = drufeast.30 days = 3 random = 20 }
		}
		8 = {
			modifier = { factor = 0 trait = drunkard }
			character_event = { id = drufeast.33 days = 3 random = 20 }
		}
		5 = {
			modifier = { factor = 0 NOT = { trait = drunkard } }
			character_event = { id = drufeast.35 days = 3 random = 20 }
		}
		5 = {
			modifier = { factor = 0 NOT = { trait = drunkard } }
			character_event = { id = drufeast.37 days = 3 random = 20 }
		}
		5 = {
			modifier = {
				factor = 0
				event_target:feast_host = {
					location = { NOT = { has_province_modifier = highway_robber_band } }
				}
			}
			character_event = { id = drufeast.38 days = 1 random = 5 }
		}
		30 = {
			modifier = { factor = 1.5 event_target:feast_host = { num_of_vassals = 5 } }
			modifier = { factor = 1.5 event_target:feast_host = { num_of_vassals = 10 } }
			modifier = { factor = 2 event_target:feast_host = { num_of_vassals = 20 } }
			modifier = { factor = 2 event_target:feast_host = { num_of_vassals = 40 } }
		}
	}
}
