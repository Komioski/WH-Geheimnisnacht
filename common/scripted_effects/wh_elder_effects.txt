# Eldership effects, used by events/WHG_eldership_events.txt.

# The other Elders of the ruler's Eldership titles approve of the ruler.
# Scope: ROOT is the ruler; FROM is the Elder who asked.
elder_others_pleased_effect = {
	custom_tooltip = { text = tooltip_other_elders_pleased }
	hidden_effect = {
		any_demesne_title = {
			limit = { has_law = succ_eldership }
			any_elector_character = {
				limit = { NOR = { character = FROM character = ROOT } }
				opinion = { modifier = opinion_elders_pleased who = ROOT years = 10 }
			}
		}
	}
}

# Every Elder of the scoped ruler's Eldership titles gets an opinion of them.
# Used when an Elder is killed (elder_murderer) or jailed
# (disrespectful_youngster); the two have their own effects so that the
# modifier name stays fixed. Scope: the ruler.
elder_all_hate_murderer_effect = {
	any_demesne_title = {
		limit = { has_law = succ_eldership }
		any_elector_character = {
			opinion = { modifier = elder_murderer who = PREVPREV years = 20 }
		}
	}
}

			opinion = { modifier = disrespectful_youngster who = PREVPREV years = 20 }
		}
	}
}
