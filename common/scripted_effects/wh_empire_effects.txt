# The Runefangs
# Each of the twelve Runefangs belongs to the Elector of its province. Each
# effect below sends one sword to its rightful bearer, taking it from
# whoever holds it now:
#   1. the holder of the Elector's crown, if he is a worthy bearer
#      (empire_runefang_bearer_trigger);
#   2. for the lost provinces of Solland and Drakwald, the Reiksmarshal and
#      the Emperor's Champion, who carry their swords in the Emperor's name;
#   3. else the Emperor, who keeps it until the crown has a worthy holder.
# With no worthy bearer the sword stays where it is. A sword that was lost
# (the old scripts destroyed them) is forged anew for its bearer.
# Callers: empire.40-42, the reclaim_runefangs decision, the Solland and
# Drakwald restorations (ee.1, ee.3) and the two minor titles.
# Scope: any. Uses the event target runefang_bearer.
###############################################################################

# The Runefang of Averland: mothers_ruin.
empire_runefang_averland_effect = {
	if = {
		limit = { k_averland = { holder_scope = { empire_runefang_bearer_trigger = yes } } }
		k_averland = { holder_scope = { save_event_target_as = runefang_bearer } }
	}
	else_if = {
		limit = { e_the_empire = { holder_scope = { empire_runefang_bearer_trigger = yes } } }
		e_the_empire = { holder_scope = { save_event_target_as = runefang_bearer } }
	}
	if = {
		limit = { event_target:runefang_bearer = { NOT = { has_artifact = mothers_ruin } } }
		if = {
			limit = { any_character = { has_artifact = mothers_ruin } }
			random_character = {
				limit = { has_artifact = mothers_ruin }
				random_artifact = {
					limit = { artifact_type = mothers_ruin }
					transfer_artifact = { from = PREV to = event_target:runefang_bearer }
				}
			}
		}
		else = { event_target:runefang_bearer = { add_artifact = mothers_ruin } }
	}
	clear_event_target = runefang_bearer
}

# Ghal Maraz, the hammer of Sigmar, belongs to the Emperor. It goes to a
# worthy Emperor from whoever holds it, and is forged anew if it was lost.
# While the Empire is held by one unworthy of it, the Grand Theogonist
# keeps it in Sigmar's name. Before the first coronation no one has it.
# Callers: empire.2 (coronation), empire.41 (the Empire changes hands),
# reclaim_runefangs. Scope: any. Uses the event target ghal_maraz_bearer.
empire_ghal_maraz_effect = {
	if = {
		limit = { e_the_empire = { holder_scope = { empire_runefang_bearer_trigger = yes } } }
		e_the_empire = { holder_scope = { save_event_target_as = ghal_maraz_bearer } }
	}
	else_if = {
		limit = {
			e_the_empire = { has_holder = yes }
			k_cult_of_sigmar = { holder_scope = { is_alive = yes religion = sigmarite } }
		}
		k_cult_of_sigmar = { holder_scope = { save_event_target_as = ghal_maraz_bearer } }
	}
	if = {
		limit = { event_target:ghal_maraz_bearer = { NOT = { has_artifact = ghal_maraz } } }
		if = {
			limit = { any_character = { has_artifact = ghal_maraz } }
			random_character = {
				limit = { has_artifact = ghal_maraz }
				random_artifact = {
					limit = { artifact_type = ghal_maraz }
					transfer_artifact = { from = PREV to = event_target:ghal_maraz_bearer }
				}
			}
		}
		else_if = {
			limit = { event_target:ghal_maraz_bearer = { has_landed_title = e_the_empire } }
			event_target:ghal_maraz_bearer = { add_artifact = ghal_maraz }
		}
	}
	clear_event_target = ghal_maraz_bearer
}

# A ruler of the Empire crowned by the head of his cult (the coronation
# decision, HF.20366) is endorsed by the cult: a strong claim on the
# Empire, and a flag naming the cult for one faith triumphant (empire.10).
# Scope: the newly crowned ruler. event_target:coronation_priest: the
# priest who crowned him.
empire_coronation_endorsement_effect = {
	if = {
		limit = {
			has_character_flag = flag_crowned_by_pope
			culture_group = imperial_group
			higher_tier_than = DUKE
			NOT = { has_global_flag = united_empire }
			e_the_empire = { has_holder = no }
			event_target:coronation_priest = {
				controls_religion = yes
				OR = { religion = sigmarite religion = ulrican religion = taalite }
			}
		}
		if = { limit = { NOT = { has_strong_claim = e_the_empire } } add_claim = e_the_empire }
		clr_character_flag = empire_crowned_by_sigmarite
		clr_character_flag = empire_crowned_by_ulrican
		clr_character_flag = empire_crowned_by_taalite
		if = {
			limit = { event_target:coronation_priest = { religion = ulrican } }
			set_character_flag = empire_crowned_by_ulrican
		}
		else_if = {
			limit = { event_target:coronation_priest = { religion = taalite } }
			set_character_flag = empire_crowned_by_taalite
		}
		else = { set_character_flag = empire_crowned_by_sigmarite }
		custom_tooltip = { text = empire_coronation_claim_tt }
	}
}
