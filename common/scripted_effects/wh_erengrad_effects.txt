# Ends a Convoy run: clears the destination and cargo, sends the ships home
# to be laid up and calls the Convoy Master back to court.
# Scope: the ruler who sent the Convoy.
erengrad_end_run_effect = {
	clr_character_flag = erengrad_on_run
	clr_character_flag = erengrad_dest_marienburg
	clr_character_flag = erengrad_dest_salzenmund
	clr_character_flag = erengrad_dest_languille
	clr_character_flag = erengrad_dest_norsca
	clr_character_flag = erengrad_dest_sudenburg
	clr_character_flag = erengrad_dest_ulthuan
	clr_character_flag = erengrad_cargo_furs
	clr_character_flag = erengrad_cargo_timber
	clr_character_flag = erengrad_cargo_silver
	clr_character_flag = erengrad_cargo_amber
	clr_character_flag = erengrad_cargo_grain
	clr_character_flag = erengrad_cargo_spoiled
	clr_character_flag = erengrad_manann_blessed
	disband_event_forces = erengrad_convoy_run
	any_courtier = {
		limit = { has_minor_title = title_convoy_master }
		move_character = PREV
	}
}

# An Elven sea chart from a far voyage: one time in twenty, once a lifetime,
# only for a ruler with no chart.
# Scope: the ruler.
erengrad_sea_chart_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = erengrad_chart_given }
			NOT = { has_artifact = erengrad_elven_sea_chart }
		}
		random = {
			chance = 5
			modifier = { factor = 3 has_character_flag = erengrad_dest_ulthuan }
			set_character_flag = erengrad_chart_given
			add_artifact = erengrad_elven_sea_chart
		}
	}
}

# The Convoy at sea counts storm and battle damage against its cargo: a
# spoiled cargo halves the run's pay.
# Scope: the Convoy Master.
naval_spoil_convoy_cargo_effect = {
	if = {
		limit = {
			has_minor_title = title_convoy_master
			liege = { has_character_flag = erengrad_on_run }
		}
		liege = { set_character_flag = erengrad_cargo_spoiled }
	}
}
