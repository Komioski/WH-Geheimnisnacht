# A man joins the order: he goes to the master's court, gives up his wealth
# and his wife, and takes the vows. His wife hears of it (holyorder.70).
# Scope: the recruit.
holy_order_join_effect = {
	save_event_target_as = holy_order_recruit
	move_character = event_target:holy_order_master
	hidden_effect = { clear_wealth = event_target:holy_order_master }
	any_spouse = {
		character_event = { id = holyorder.70 }
		remove_spouse = event_target:holy_order_recruit
	}
	if = {
		limit = { NOT = { trait = celibate } }
		add_trait = celibate
	}
	if = {
		limit = { trait = in_hiding }
		remove_trait = in_hiding
		clr_character_flag = do_not_disturb
		add_character_modifier = {
			name = went_out_of_hiding_timer
			duration = 180
			hidden = yes
		}
		hidden_effect = { character_event = { id = CM.6400 } } # Notify plotters and family
	}
}

# The ruler takes a loan from the order: the first free loan slot.
# Scope: the ruler; event target holy_order_master.
holy_order_take_loan_effect = {
	if = {
		limit = { NOT = { has_character_modifier = borrowed_from_holy_order } }
		add_character_modifier = { name = borrowed_from_holy_order duration = -1 }
		event_target:holy_order_master = {
			primary_title = {
				PREVPREV = { save_persistent_event_target = { name = holy_order_creditor scope = PREV } }
			}
		}
	}
	else = {
		add_character_modifier = { name = borrowed_from_holy_order_b duration = -1 }
		event_target:holy_order_master = {
			primary_title = {
				PREVPREV = { save_persistent_event_target = { name = holy_order_creditor_b scope = PREV } }
			}
		}
	}
}

# The ruler's loan from the order is settled or forgiven.
# Scope: the ruler; event target holy_order_master.
holy_order_clear_debt_effect = {
	if = {
		limit = {
			has_character_modifier = borrowed_from_holy_order
			persistent_event_target:holy_order_creditor = {
				holder_scope = { character = event_target:holy_order_master }
			}
		}
		remove_character_modifier = borrowed_from_holy_order
		clear_persistent_event_target = holy_order_creditor
	}
	else_if = {
		limit = {
			has_character_modifier = borrowed_from_holy_order_b
			persistent_event_target:holy_order_creditor_b = {
				holder_scope = { character = event_target:holy_order_master }
			}
		}
		remove_character_modifier = borrowed_from_holy_order_b
		clear_persistent_event_target = holy_order_creditor_b
	}
}
