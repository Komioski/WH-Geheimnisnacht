# A troll-hide cloak from a troll that stayed dead: one time in five, once a
# lifetime, only for a hunter without one.
# Scope: the hunter.
north_troll_hide_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = north_troll_hide_given }
			NOT = { has_artifact = north_troll_hide }
		}
		random = {
			chance = 20
			set_character_flag = north_troll_hide_given
			add_artifact = north_troll_hide
		}
	}
}
