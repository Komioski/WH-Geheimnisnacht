# A pelt from the Bear Hunt by Ursun's law: one time in eight, once a
# lifetime, and only for a hunter with no pelt.
# Scope: the hunter.
kislevsoc_pelt_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = kislevsoc_pelt_given }
			NOT = { has_artifact = kislevsoc_great_bear_pelt }
		}
		random = {
			chance = 12
			set_character_flag = kislevsoc_pelt_given
			add_artifact = kislevsoc_great_bear_pelt
		}
	}
}

# A bear-claw talisman from a Circle trial: one time in four, once a
# lifetime, only for one without a talisman.
# Scope: the member.
kislevsoc_claw_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = kislevsoc_claw_given }
			NOT = { has_artifact = kislevsoc_bear_claw_talisman }
		}
		random = {
			chance = 25
			set_character_flag = kislevsoc_claw_given
			add_artifact = kislevsoc_bear_claw_talisman
		}
	}
}

# A Duma trinket from a won trial: one time in six, once a lifetime, only
# for one with neither. A Praag filigree chain or a sable kaftan.
# Scope: the member.
kislevsoc_duma_trinket_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = kislevsoc_trinket_given }
			NOT = { has_artifact = kislevsoc_praag_filigree_chain }
			NOT = { has_artifact = kislevsoc_sable_kaftan }
		}
		random = {
			chance = 16
			set_character_flag = kislevsoc_trinket_given
			random_list = {
				50 = { add_artifact = kislevsoc_praag_filigree_chain }
				50 = { add_artifact = kislevsoc_sable_kaftan }
			}
		}
	}
}

# A Bear Champion's rare hunt prize: a bear that would not die, tamed. One
# time in twenty, once a lifetime, for a Champion or the First with no bear.
# Scope: the hunter.
kislevsoc_war_bear_effect = {
	if = {
		limit = {
			society_member_of = circle_of_the_bear
			society_rank >= 3
			NOT = { has_character_flag = kislevsoc_war_bear_given }
			NOT = { has_artifact = war_bear_mount }
		}
		random = {
			chance = 5
			set_character_flag = kislevsoc_war_bear_given
			set_character_flag = has_mount
			add_artifact = war_bear_mount
		}
	}
}
