# Effects for the lifestyle events (events/WHG_lifestyle_events.txt).

# Offers one lifestyle chain that fits the character. The weights carry the
# old MTTH modifiers, inverted.
# Scope: the character (lifestyle.1).

		10 = { # Scholar or mystic
			trigger = { lifestyle_scholar_start_trigger = yes }
			modifier = { factor = 0.8 trait = content }

			modifier = { factor = 1.3 learning = 10 }
			modifier = { factor = 0.7 trait = cynical }
			modifier = { factor = 5 trait = ambitious }
			modifier = { factor = 2.5 trait = zealous }
			modifier = { factor = 1.3 trait = scholarly_theologian }
			modifier = { factor = 2 trait = mastermind_theologian }
			character_event = { id = lifestyle.10 }
		}
		10 = { # Gardener or impaler
			trigger = { lifestyle_gardener_start_trigger = yes }
			modifier = { factor = 1.25 martial = 9 }
			modifier = { factor = 1.25 martial = 10 }
			modifier = { factor = 1.25 trait = indulgent_wastrel }
			modifier = { factor = 1.25 trait = misguided_warrior }
			modifier = { factor = 1.25 trait = tough_soldier }
			character_event = { id = lifestyle.20 }
		}
		10 = { # Duelist or poet
			trigger = { lifestyle_courtship_start_trigger = yes }
			modifier = { factor = 5 trait = charismatic_negotiator }
			modifier = { factor = 2.5 trait = skilled_tactician }

		}
		7 = { # Hunter or falconer
			trigger = { lifestyle_hunter_start_trigger = yes }
			modifier = { factor = 1.3 martial = 8 }
			modifier = { factor = 2 has_nickname = nick_the_fowler }

			trigger = { lifestyle_poet_falconer_start_trigger = yes }
			modifier = { factor = 0.25 trait = slothful }

			trigger = { lifestyle_celibacy_start_trigger = yes }
			modifier = { factor = 3 trait = martial_cleric }
			modifier = { factor = 2 trait = mastermind_theologian }

# Ends a lifestyle chain so another can start later.
# Scope: the character.
lifestyle_end_effect = {
	clr_character_flag = LE_ST
	clr_character_flag = lifestyle_tried_poetry
}

# Drops the leftovers of a chain that stalled (lifestyle.1).
# Scope: the character.
lifestyle_clear_chain_effect = {
	remove_character_modifier = illumination
	remove_character_modifier = one_hedonist
	remove_character_modifier = one_celibate
	remove_character_modifier = sweet_temptation
	remove_character_modifier = stay_strong
	remove_character_modifier = ormazd_hedonist
	remove_character_modifier = ormazd_poet
	remove_character_modifier = aspiring_faqih
	lifestyle_end_courtship_effect = yes
}

# The courted lady stops being courted (lifestyle.31).
# Scope: the suitor.
lifestyle_end_courtship_effect = {
	any_courtier = {
		limit = {
			OR = {
				has_character_flag = encourage_courtship
				has_character_flag = ignore_courtship
			}
		}
		clr_character_flag = encourage_courtship
		clr_character_flag = ignore_courtship
	}
}

# The courted lady becomes the suitor's lover (lifestyle.36, lifestyle.37).
# Scope: the suitor.
lifestyle_win_lady_effect = {
	random_courtier = {
		limit = {
			OR = {
				has_character_flag = encourage_courtship
				has_character_flag = ignore_courtship
			}
		}
		add_lover = ROOT
	}
	lifestyle_end_courtship_effect = yes
}

# Giving up on pleasure and poetry brings content, or else costs diligence
# (lifestyle.71, lifestyle.72).
# Scope: the character.
lifestyle_content_effect = {
	if = {
		limit = { NOT = { trait = content } }
		add_trait = content
	}
	else_if = {
		limit = { trait = diligent }
		remove_trait = diligent
	}
}
