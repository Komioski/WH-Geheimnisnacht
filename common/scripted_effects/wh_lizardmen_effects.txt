# A relic of the Old Ones from the jungle, a ruined temple or a plaque cache.
# Gives one Lizardmen artifact the character does not have yet (the common
# ones from common/artifacts/09_artifacts_lizardmen.txt); piety if they have
# them all. Saved as event_target:lizardmen_relic for tooltips.
# Scope: a character.
lizardmen_relic_artifact_effect = {
	random_list = {
		20 = {
			trigger = { NOT = { has_artifact = obsinite_blade } }
			add_artifact = obsinite_blade
		}
		20 = {
			trigger = {
				NOT = { has_artifact = saurus_scale_shield }
				OR = {
					trait = creature_saurus
					trait = creature_skink
				}
			}
			add_artifact = saurus_scale_shield
		}
		10 = {
			trigger = {
				NOT = { has_artifact = obsinite_shield }
				trait = creature_saurus
			}
			add_artifact = obsinite_shield
		}
		15 = {
			trigger = {
				NOT = { has_artifact = obsinite_plaque_armour }
				OR = {
					trait = creature_saurus
					trait = creature_skink
				}
			}
			add_artifact = obsinite_plaque_armour
		}
		6 = {
			trigger = {
				NOT = { has_artifact = golden_plaque_armour }
				OR = {
					trait = creature_saurus
					trait = creature_skink
				}
			}
			add_artifact = golden_plaque_armour
		}
		10 = {
			trigger = {
				NOT = { has_artifact = staff_of_obsinite }
				is_trained_mage_trigger = yes
			}
			add_artifact = staff_of_obsinite
		}
		10 = {
			trigger = {
				NOT = { has_artifact = revivication_crystal }
				religion_group = old_ones_group
			}
			add_artifact = revivication_crystal
		}
		3 = {
			trigger = { NOT = { has_artifact = blade_of_realities } }
			add_artifact = blade_of_realities
		}
		# Fallback when every relic is already in the treasury
		1 = {
			modifier = {
				factor = 0
				OR = {
					NOT = { has_artifact = obsinite_blade }
					NOT = { has_artifact = blade_of_realities }
				}
			}
			piety = 100
		}
	}
	new_artifact = { save_event_target_as = lizardmen_relic }
}

# One step more jungle in this province (light > medium > dense > oppressive).
# Scope: a province.
lustria_jungle_grow_effect = {
	if = {
		limit = { has_province_modifier = lustria_dense_jungle }
		remove_province_modifier = lustria_dense_jungle
		add_province_modifier = { name = lustria_oppressive_jungle duration = -1 }
	}
	else_if = {
		limit = { has_province_modifier = lustria_medium_jungle }
		remove_province_modifier = lustria_medium_jungle
		add_province_modifier = { name = lustria_dense_jungle duration = -1 }
	}
	else_if = {
		limit = { has_province_modifier = lustria_light_jungle }
		remove_province_modifier = lustria_light_jungle
		add_province_modifier = { name = lustria_medium_jungle duration = -1 }
	}
	else_if = {
		limit = { NOT = { has_province_modifier = lustria_oppressive_jungle } }
		add_province_modifier = { name = lustria_light_jungle duration = -1 }
	}
}

# One step less jungle in this province (oppressive > dense > medium > light).
# Scope: a province.
lustria_jungle_clear_effect = {
	if = {
		limit = { has_province_modifier = lustria_oppressive_jungle }
		remove_province_modifier = lustria_oppressive_jungle
		add_province_modifier = { name = lustria_dense_jungle duration = -1 }
	}
	else_if = {
		limit = { has_province_modifier = lustria_dense_jungle }
		remove_province_modifier = lustria_dense_jungle
		add_province_modifier = { name = lustria_medium_jungle duration = -1 }
	}
	else_if = {
		limit = { has_province_modifier = lustria_medium_jungle }
		remove_province_modifier = lustria_medium_jungle
		add_province_modifier = { name = lustria_light_jungle duration = -1 }
	}
}

# A war starts: a sleeping Slann vassal may grow restless (lizardmen.86) or
# wake to lead the war (lizardmen.88). Tablet wars wake them far more often.
# Scope: a Skink ruler with lizardmen_has_sleeping_slann_trigger.
lizardmen_slann_stirs_effect = {
	random_list = {
		# Sleeps on
		95 = {
			modifier = {
				factor = 0.85
				trait = ambitious
			}
			modifier = {
				factor = 0.85
				trait = diligent
			}
			modifier = {
				factor = 1.25
				trait = slothful
			}
		}
		# Restless
		4 = {
			modifier = {
				factor = 0
				any_war = { using_cb = tablet_search }
			}
			random_vassal = {
				limit = {
					trait = creature_slann
					trait = lizardman_sleepingslann
				}
				character_event = { id = lizardmen.86 days = 100 random = 50 }
			}
		}
		# Wakes for war
		1 = {
			modifier = {
				factor = 10
				any_war = { using_cb = tablet_search }
			}
			random_vassal = {
				limit = {
					trait = creature_slann
					trait = lizardman_sleepingslann
				}
				character_event = { id = lizardmen.88 days = 7 random = 7 }
			}
		}
	}
}
