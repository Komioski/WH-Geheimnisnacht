# A relic of the Old Ones from the jungle, a ruined temple or a plaque cache.
# Gives one Lizardmen artifact the character does not have yet (the common
# ones from common/artifacts/09_artifacts_lizardmen.txt); piety if they have
# them all. Saved as event_target:lizardmen_relic for tooltips.
# Scope: a character.
lizardmen_relic_artifact_effect = {
	random_list = {
		20 = {
			trigger = { NOT = { has_artifact = obsinite_blade } }
			add_artifact = obsinite_blade
		}
		20 = {
			trigger = {
				NOT = { has_artifact = saurus_scale_shield }
				OR = {
					trait = creature_saurus
					trait = creature_skink
				}
			}
			add_artifact = saurus_scale_shield
		}
		10 = {
			trigger = {
				NOT = { has_artifact = obsinite_shield }
				trait = creature_saurus
			}
			add_artifact = obsinite_shield
		}
		15 = {
			trigger = {
				NOT = { has_artifact = obsinite_plaque_armour }
				OR = {
					trait = creature_saurus
					trait = creature_skink
				}
			}
			add_artifact = obsinite_plaque_armour
		}
		6 = {
			trigger = {
				NOT = { has_artifact = golden_plaque_armour }
				OR = {
					trait = creature_saurus
					trait = creature_skink
				}
			}
			add_artifact = golden_plaque_armour
		}
		10 = {
			trigger = {
				NOT = { has_artifact = staff_of_obsinite }
				is_trained_mage_trigger = yes
			}
			add_artifact = staff_of_obsinite
		}
		10 = {
			trigger = {
				NOT = { has_artifact = revivication_crystal }
				religion_group = old_ones_group
			}
			add_artifact = revivication_crystal
		}
		3 = {
			trigger = { NOT = { has_artifact = blade_of_realities } }
			add_artifact = blade_of_realities
		}
		# Fallback when every relic is already in the treasury
		1 = {
			modifier = {
				factor = 0
				OR = {
					NOT = { has_artifact = obsinite_blade }
					NOT = { has_artifact = blade_of_realities }
				}
			}
			piety = 100
		}
	}
	new_artifact = { save_event_target_as = lizardmen_relic }
}

# One step more jungle in this province (light > medium > dense > oppressive).
# Scope: a province.
lustria_jungle_grow_effect = {
	if = {
		limit = { has_province_modifier = lustria_dense_jungle }
		remove_province_modifier = lustria_dense_jungle
		add_province_modifier = { name = lustria_oppressive_jungle duration = -1 }
	}
	else_if = {
		limit = { has_province_modifier = lustria_medium_jungle }
		remove_province_modifier = lustria_medium_jungle
		add_province_modifier = { name = lustria_dense_jungle duration = -1 }
	}
	else_if = {
		limit = { has_province_modifier = lustria_light_jungle }
		remove_province_modifier = lustria_light_jungle
		add_province_modifier = { name = lustria_medium_jungle duration = -1 }
	}
	else_if = {
		limit = { NOT = { has_province_modifier = lustria_oppressive_jungle } }
		add_province_modifier = { name = lustria_light_jungle duration = -1 }
	}
}

	else_if = {
		limit = { has_province_modifier = lustria_medium_jungle }
		remove_province_modifier = lustria_medium_jungle
		add_province_modifier = { name = lustria_light_jungle duration = -1 }
	}
}

# A war starts: a sleeping Slann vassal may grow restless (lizardmen.86) or
# wake to lead the war (lizardmen.88). Tablet wars wake them far more often.
# Scope: a Skink ruler with lizardmen_has_sleeping_slann_trigger.
lizardmen_slann_stirs_effect = {
	random_list = {
		# Sleeps on
		95 = {
			modifier = {
				factor = 0.85
				trait = ambitious
			}
			modifier = {
				factor = 0.85
				trait = diligent
			}
			modifier = {
				factor = 1.25
				trait = slothful
			}
		}
		# Restless
		4 = {
			modifier = {
				factor = 0
				any_war = { using_cb = tablet_search }
			}
			random_vassal = {
				limit = {
					trait = creature_slann
					trait = lizardman_sleepingslann
				}
				character_event = { id = lizardmen.86 days = 100 random = 50 }
			}
		}
		# Wakes for war
		1 = {
			modifier = {
				factor = 10
				any_war = { using_cb = tablet_search }
			}
			random_vassal = {
				limit = {
					trait = creature_slann
					trait = lizardman_sleepingslann
				}
				character_event = { id = lizardmen.88 days = 7 random = 7 }
			}
		}
	}
}

# The Temple Guard cut down a looter, unless something keeps them from dying.
# Scope: the looter (lustria.106).
lustria_temple_guard_death_effect = {
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		death = { death_reason = death_battle }
	}
}

# The host of the Old Ones comes ashore (lizardmen.131). Picks a port of
# event_target:liz_crossing_enemy nearest that ruler's capital (any province
# of the realm if the realm has no port), and lands an army there sized by
# the tier of ROOT. The army disbands at peace and shares the liz_fleet
# earmark, so the tablet war's disband_event_forces clears it too.
# Scope: ROOT is the Old Ones ruler who lands.
liz_crossing_land_host_effect = {
	event_target:liz_crossing_enemy = {
		capital_scope = { save_event_target_as = liz_crossing_front }
		if = {
			limit = { any_realm_province = { port = yes } }
			random_realm_province = {
				limit = { port = yes }
				preferred_limit = { distance = { where = event_target:liz_crossing_front distance < 100 } }
				preferred_limit = { distance = { where = event_target:liz_crossing_front distance < 250 } }
				preferred_limit = { distance = { where = event_target:liz_crossing_front distance < 500 } }
				save_event_target_as = liz_crossing_landing
			}
		}
		else = {
			random_realm_province = {
				preferred_limit = { distance = { where = event_target:liz_crossing_front distance < 100 } }
				preferred_limit = { distance = { where = event_target:liz_crossing_front distance < 250 } }
				save_event_target_as = liz_crossing_landing
			}
		}
	}
	trigger_switch = {
		on_trigger = real_tier
		EMPEROR = {
			spawn_unit = {
				province = event_target:liz_crossing_landing
				home = event_target:liz_crossing_landing
				owner = ROOT
				troops = {
					heavy_infantry = { 4000 4000 }
					light_infantry = { 3000 3000 }
					archers = { 2000 2000 }
					war_elephants = { 600 600 }
				}
				attrition = 0.5
				maintenance_multiplier = 0.25
				disband_on_peace = yes
				cannot_inherit = yes
				earmark = liz_fleet
			}
		}
		KING = {
			spawn_unit = {
				province = event_target:liz_crossing_landing
				home = event_target:liz_crossing_landing
				owner = ROOT
				troops = {
					heavy_infantry = { 2500 2500 }
					light_infantry = { 2000 2000 }
					archers = { 1500 1500 }
					war_elephants = { 400 400 }
				}
				attrition = 0.5
				maintenance_multiplier = 0.25
				disband_on_peace = yes
				cannot_inherit = yes
				earmark = liz_fleet
			}
		}
		DUKE = {
			spawn_unit = {
				province = event_target:liz_crossing_landing
				home = event_target:liz_crossing_landing
				owner = ROOT
				troops = {
					heavy_infantry = { 1200 1200 }
					light_infantry = { 1000 1000 }
					archers = { 800 800 }
					war_elephants = { 150 150 }
				}
				attrition = 0.5
				maintenance_multiplier = 0.25
				disband_on_peace = yes
				cannot_inherit = yes
				earmark = liz_fleet
			}
		}
		fallback = {
			spawn_unit = {
				province = event_target:liz_crossing_landing
				home = event_target:liz_crossing_landing
				owner = ROOT
				troops = {
					heavy_infantry = { 600 600 }
					light_infantry = { 500 500 }
					archers = { 400 400 }
					war_elephants = { 50 50 }
				}
				attrition = 0.5
				maintenance_multiplier = 0.25
				disband_on_peace = yes
				cannot_inherit = yes
				earmark = liz_fleet
			}
		}
	}
	# A waking Slann adds the temple guard
	if = {
		limit = { any_vassal = { trait = creature_slann NOT = { trait = lizardman_sleepingslann } } }
		spawn_unit = {
			province = event_target:liz_crossing_landing
			home = event_target:liz_crossing_landing
			owner = ROOT
			troops = { heavy_infantry = { 1000 1000 } }
			attrition = 0.5
			maintenance_multiplier = 0.25
			disband_on_peace = yes
			cannot_inherit = yes
			earmark = liz_fleet
		}
	}
}

# A ruler who joins an Old Ones war overseas (lizardmen.83, jade.4): a
# Lustrian AI ruler sends its host across the sea (lizardmen.131); a player
# gets the old fleet to sail with.
# Scope: ROOT is the ruler who joins; event_target:liz_crossing_enemy the
# enemy war leader, saved by the caller.
liz_crossing_joiner_effect = {
	if = {
		limit = {
			ai = yes
			liz_crossing_ruler_trigger = yes
			NOT = { has_character_flag = liz_crossing_at_sea }
			event_target:liz_crossing_enemy = { liz_crossing_enemy_trigger = yes }
		}
		set_character_flag = liz_crossing_at_sea
		character_event = { id = lizardmen.131 days = 60 random = 30 }
	}
	else_if = {
		limit = { ai = no }
		spawn_fleet = {
			province = closest
			owner = ROOT
			match_character = ROOT
			match_mult = 2
			match_min = 50
			disband_on_peace = yes
			earmark = liz_fleet
		}
	}
}
