###############################################################################
# Magic commissions (events/WHG_magic_events.txt, namespace commission)
#
# The item is a flag on the buyer: commission_<item>, and commission_waiting
# while the mage works.
#
# commission_take_effect     the mage (ROOT) takes the work: the buyer
#                            (event_target:buyer) pays, the mage gets his fee
#                            and magic experience, the item is due in time
# commission_deliver_effect  the buyer gets the item; flags cleared, the
#                            mage (FROM) is free again
# commission_clear_effect    the buyer's commission flags are cleared
###############################################################################

commission_deliver_effect = {
	if = { limit = { has_character_flag = commission_sword_1 } add_artifact = sword_1 }
	else_if = { limit = { has_character_flag = commission_sword_2 } add_artifact = sword_2 }
	else_if = { limit = { has_character_flag = commission_sword_3 } add_artifact = sword_3 }
	else_if = { limit = { has_character_flag = commission_sword_4 } add_artifact = sword_4 }
	else_if = { limit = { has_character_flag = commission_sword_5 } add_artifact = sword_5 }
	else_if = { limit = { has_character_flag = commission_amulet_1 } add_artifact = amulet_of_protection_1 }
	else_if = { limit = { has_character_flag = commission_amulet_3 } add_artifact = amulet_of_protection_3 }
	else_if = { limit = { has_character_flag = commission_amulet_5 } add_artifact = amulet_of_protection_5 }
	else_if = { limit = { has_character_flag = commission_potion_healing } add_artifact = potion_of_healing }
	else_if = { limit = { has_character_flag = commission_potion_beauty } add_artifact = potion_of_beauty }
	else_if = { limit = { has_character_flag = commission_potion_youth } add_artifact = elixir_of_eternal_youth }
	else_if = { limit = { has_character_flag = commission_ring_persuasion } add_artifact = ring_of_persuasion }
	else_if = { limit = { has_character_flag = commission_ring_invisibility } add_artifact = ring_of_invisibility }
	else_if = { limit = { has_character_flag = commission_ring_speed } add_artifact = ring_of_speed }
	else_if = { limit = { has_character_flag = commission_ring_command } add_artifact = ring_of_command }
	else_if = { limit = { has_character_flag = commission_elixir_martial } add_artifact = elixir_of_the_marshall }
	else_if = { limit = { has_character_flag = commission_elixir_stewardship } add_artifact = elixir_of_the_steward }
	else_if = { limit = { has_character_flag = commission_elixir_diplomacy } add_artifact = elixir_of_the_diplomat }
	else_if = { limit = { has_character_flag = commission_elixir_intrigue } add_artifact = elixir_of_the_spy }
	FROM = { remove_trait = crafting_magic_item }
	commission_clear_effect = yes
}
