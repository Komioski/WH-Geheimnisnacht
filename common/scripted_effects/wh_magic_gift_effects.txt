###############################################################################
# Magic gift effects
# Used by events/WHG_magic_events.txt (namespace gift).
#
# gift_hunt_success_effect  the hunter (THIS) reports the quarry burned or
#                           severed: kislev.8 for an Ice Witch, bretonmagic.5
#                           for a Damsel
# gift_hunt_failed_effect   the quarry was spared or drove the hunter off:
#                           kislev.9 or bretonmagic.6
# gift_sever_effect         THIS is cut off from the Winds of Magic for good
###############################################################################

gift_hunt_success_effect = {
	if = {
		limit = { society_member_of = ice_witch_sisterhood }
		character_event = { id = kislev.8 }
	}
	else = { character_event = { id = bretonmagic.5 } }
}

gift_hunt_failed_effect = {
	if = {
		limit = { society_member_of = ice_witch_sisterhood }
		character_event = { id = kislev.9 }
	}
	else = { character_event = { id = bretonmagic.6 } }
}

# Hollowed: no gift, no power, no lore, no apprenticeship. A landed victim
# gives up his lands.
gift_sever_effect = {
	magic_power_removal_effect = yes
	remove_magic_effect = yes
	remove_trait = mage_apprentice
	clr_character_flag = magic_potential_flag_1
	clr_character_flag = magic_potential_flag_2
	clr_character_flag = magic_potential_flag_3
	clr_character_flag = magic_potential_flag_4
	set_character_flag = failed_mage
	add_trait = hollowed
	if = { limit = { is_landed = yes } abdicate = yes }
}
