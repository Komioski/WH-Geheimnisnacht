join_prev_cult_of_morr_society = {
	if = {
		limit = { PREV = { society_member_of = cult_of_morr } }
		join_society = cult_of_morr
	}
}

join_from_cult_of_morr_society = {
	FROM = {
		if = {
			limit = {
				society_member_of = cult_of_morr
			}
			ROOT = {
				join_society = cult_of_morr
			}
		}
	}
}
# The Fellowship of the Shroud (events/WHG_morr_events.txt)

# A Raven Knight wins the undead hunt (cultofmorr.155, 157, 159, 161): the
# quest ends, the order gains influence and Dust, the knight rests a while,
# and may keep a trophy of the hunt (10%, once a lifetime, none held).
# Scope: the knight.
morr_undead_hunt_won_effect = {
	clr_quest = quest_undead_hunt
	add_society_modifier = { modifier = cult_of_morr_successful_mission years = 10 }
	change_society_currency = 500
	add_character_modifier = { name = quest_cooldown_timer hidden = yes days = 1000 }
	if = {
		limit = {
			NOT = { has_character_flag = morr_hunt_trophy_taken }
			NOT = { any_artifact = { artifact_type = monster_hunt_trophy } }
		}
		random = {
			chance = 10
			set_character_flag = morr_hunt_trophy_taken
			add_artifact = monster_hunt_trophy
		}
	}
}
