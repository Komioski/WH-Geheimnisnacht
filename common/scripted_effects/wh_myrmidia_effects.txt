# Saves the king the see holder serves, or else the emperor, as new_liege.
# Scope: the holder of the see.
eternal_save_temple_liege_effect = {
	if = {
		limit = { any_liege = { tier = KING } }
		any_liege = {
			limit = { tier = KING }
			save_event_target_as = new_liege
		}
	}
	else = {
		any_liege = {
			limit = { tier = EMPEROR }
			save_event_target_as = new_liege
		}
	}
}
