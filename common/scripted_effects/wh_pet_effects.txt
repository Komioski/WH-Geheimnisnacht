# Effects for the pet events (events/WHG_pet_events.txt, namespace pet).

# A pending relationship omen from the scrying events (MNM.1213) comes true
# when the cat dies.
# Scope: the owner.
pet_cat_death_omen_effect = {
	hidden_effect = {
		if = {
			limit = {
				OR = {
					has_character_modifier = good_relationship_omen
					has_character_modifier = bad_relationship_omen
				}
			}
			set_character_flag = cat_death_omen
			character_event = { id = MNM.1213 days = 35 random = 30 } # Scrying vision comes true
		}
	}
}

# The scoped character takes ROOT's cat. It dies in time, and the old owner
# hears how it is doing (pet.13).
# Scope: the new owner. ROOT is the old owner.
pet_give_cat_effect = {
	add_character_modifier = { name = pet_cat duration = -1 }
	hidden_effect = {
		character_event = { id = pet.3 days = 2555 random = 1000 } # The cat dies
		character_event = { id = pet.13 days = 100 }
	}
}
