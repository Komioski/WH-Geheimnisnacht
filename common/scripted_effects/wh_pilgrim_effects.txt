# Saves a holy order of the pilgrim's faith as event target pilgrim_order
# (pilgrim.27 - 29).
# Scope: the pilgrim.
pilgrim_save_holy_order_effect = {
	random_independent_ruler = {
		limit = {
			holy_order = yes
			religion = ROOT
		}
		save_event_target_as = pilgrim_order
	}
}

# A token of the shrine (pilgrim.61): an ampulla for the Old World cults,
# an ember of Asuryan, an acorn of the Oak of Ages, a vial from the
# Cauldron of Blood or an ancestor stone.
# Scope: the pilgrim.
pilgrim_token_effect = {
	set_character_flag = pilgrim_got_token
	if = {
		limit = { religion_group = old_world_group }
		add_artifact = pilgrim_ampulla
	}
	else_if = {
		limit = { has_character_flag = pilgrim_to_isle_of_flame }
		add_artifact = pilgrim_asuryan_ember
	}
	else_if = {
		limit = { has_character_flag = pilgrim_to_oak_of_ages }
		add_artifact = pilgrim_oak_acorn
	}
	else_if = {
		limit = { has_character_flag = pilgrim_to_har_ganeth }
		add_artifact = pilgrim_khaine_vial
	}
	else_if = {
		limit = { religion_group = dwarf_gods_group }
		add_artifact = pilgrim_ancestor_stone
	}
}
