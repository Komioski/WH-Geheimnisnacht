# Finds the Cult of Pleasure's grand master and saves them as
# event_target:grand_master.
# Scope: any.
pleasure_find_grand_master_effect = {
	cult_of_pleasure = {
		leader = { save_event_target_as = grand_master }
	}
}

# Sets the flags GetDWReasonToJoin reads, so the recruit asks the cult for
# help with their own trouble: a sick lover, spouse or child, debts, their
# own illness, a war or a rival.
# Scope: the recruit.
pleasure_reason_to_join_effect = {
	if = {
		limit = { any_lover = { has_disease_trigger = yes } }
		set_character_flag = has_ill_lover_loc
	}
	if = {
		limit = { any_spouse = { has_disease_trigger = yes } }
		set_character_flag = has_ill_spouse_loc
	}
	if = {
		limit = { any_child = { has_disease_trigger = yes } }
		set_character_flag = has_ill_child_loc
	}
	if = { limit = { NOT = { wealth = 0 } } set_character_flag = has_money_troubles_loc }
	if = { limit = { has_disease_trigger = yes } set_character_flag = has_some_illness_loc }
	if = { limit = { war = yes } set_character_flag = has_need_of_winning_battles_loc }
	if = { limit = { any_rival = { always = yes } } set_character_flag = has_a_rival_loc }
}

# No member of the right sex is beautiful enough to meet the recruit
# (pleasure.18), so the cult sends a new seducer, who joins the cult with
# Atharti as their secret faith. Saves them as
# event_target:cult_of_pleasure_inductor.
# Scope: the new character. ROOT is the recruit.
pleasure_new_inductor_effect = {
	character_event = { id = aiclean.31 } # Set racial trait
	remove_trait = ugly
	remove_trait = hunchback
	remove_trait = dwarf
	remove_trait = chaste
	add_trait = lustful
	add_trait = fair
	join_society = cult_of_pleasure
	set_secret_religion = atharti
	save_event_target_as = cult_of_pleasure_inductor
}

# The recruit joins the cult and takes Atharti as their secret faith, then the
# recruitment ends.
# Scope: the recruit.
pleasure_join_effect = {
	if = {
		limit = { has_character_flag = being_recruited_by_the_cult_of_pleasure }
		join_society = cult_of_pleasure
		set_secret_religion = atharti
	}
	hidden_effect = { clr_character_flag = being_recruited_by_the_cult_of_pleasure }
}

# A cultist (event_target:pleasure_knife) tries to kill the grand master's
# target (event_target:pleasure_knife_target). About 40% to succeed, better
# with a cunning killer, worse against a paranoid target, never against an
# invincible one. A kill clears the mark for death; a miss may expose the
# killer to the target. The grand master gets a letter either way.
# Scope: the grand master.
pleasure_knife_strike_effect = {
	random_list = {
		40 = {
			modifier = { factor = 1.5 event_target:pleasure_knife = { intrigue >= 12 } }
			modifier = { factor = 1.5 event_target:pleasure_knife = { intrigue >= 18 } }
			modifier = { factor = 1.25 event_target:pleasure_knife = { trait = elusive_shadow } }
			modifier = { factor = 0.5 event_target:pleasure_knife_target = { trait = paranoid } }
			modifier = { factor = 0 event_target:pleasure_knife_target = { is_invincible_trigger = yes } }
			event_target:pleasure_knife_target = {
				if = {
					limit = { has_character_flag = cop_deathmark }
					clr_character_flag = cop_deathmark
					ROOT = { clr_character_flag = cop_marked_someone_for_death }
				}
				death = { death_reason = death_murder_unknown killer = event_target:pleasure_knife }
			}
			event_target:pleasure_knife = { add_society_currency_minor_effect = yes }
			letter_event = { id = pleasure.61 days = 7 }
		}
		60 = {
			random = {
				chance = 30

				event_target:pleasure_knife_target = {
					opinion = { modifier = opinion_murder_attempt who = event_target:pleasure_knife }
				}
				set_character_flag = pleasure_knife_exposed
			}
			letter_event = { id = pleasure.62 days = 7 }
		}
	}
}
