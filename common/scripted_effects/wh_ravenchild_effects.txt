# The Raven's Child, used by events/WHG_chaos_undivided_events.txt.

# Sponsors the march: zeal, piety, the child's gratitude.
# Scope: a ruler; the child is event_target:childrens_crusade_leader.
ravenchild_sponsor_effect = {
	if = { limit = { NOT = { trait = zealous } } add_trait = zealous }
	tiered_piety_reward_effect = yes
	event_target:childrens_crusade_leader = {
		show_scope_change = no
		opinion = { modifier = opinion_believed_in_me who = ROOT years = 15 }
	}
	hidden_effect = {
		event_target:childrens_crusade_leader = {
			save_persistent_event_target = { name = childrens_crusade_sponsor_per scope = ROOT }
		}
	}
}

# Sends warriors to guard the march (heavy troops later).
# Scope: a ruler.
ravenchild_send_troops_effect = {
	if = { limit = { trait = cruel } random = { chance = 25 remove_trait = cruel } }
	if = {
		limit = { NOR = { trait = kind trait = cruel } }
		random = {
			chance = 25
			add_trait = kind
			hidden_effect = { character_event = { id = 38268 } }
		}
	}
	event_target:childrens_crusade_leader = {
		show_scope_change = no
		opinion = { modifier = opinion_grateful who = ROOT years = 15 }
		hidden_effect = { change_variable = { which = children_crusade_heavy_troops value = 2 } }
	}
	tiered_piety_reward_effect = yes
	capital_scope = {
		add_province_modifier = { name = kid_crusade_pledged_troops duration = 1200 stacking = yes }
	}
}

# A kind act may soften a cruel ruler. Scope: a ruler.
ravenchild_mercy_effect = {
	if = { limit = { trait = cruel } random = { chance = 20 remove_trait = cruel } }
	if = {
		limit = { NOR = { trait = kind trait = cruel } }
		random = {
			chance = 20
			add_trait = kind
			hidden_effect = { character_event = { id = 38268 } }
		}
	}
}

# Feeds the children and sends them on. Scope: a ruler.
ravenchild_feed_effect = {
	ravenchild_mercy_effect = yes
	hidden_effect = {
		event_target:childrens_crusade_leader = {
			change_variable = { which = children_crusade_morale value = 2 }
		}
	}
	transfer_scaled_wealth = {
		to = event_target:childrens_crusade_leader
		value = 1
		min = 50
		max = 100
	}
}

# Guards herd the children on; morale drops. Scope: a ruler.
ravenchild_guards_effect = {
	hidden_effect = {
		event_target:childrens_crusade_leader = {
			change_variable = { which = children_crusade_morale value = -1 }
		}
	}
}

# The march disbands: followers vanish, the band's title is destroyed.
# Scope: the child.
ravenchild_disband_effect = {
	any_courtier = {
		limit = { is_ruler = no is_landed = no host = { character = ROOT } }
		if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
		else = { death = { death_reason = death_missing } }
	}
	primary_title = { activate_title = { title = THIS status = no } destroy_landed_title = THIS }
}

# The ruler sells the child and the young followers to the slavers.
# Scope: a ruler.
ravenchild_sell_effect = {
	hidden_effect = {
		event_target:childrens_crusade_leader = {
			any_courtier = {
				limit = { is_ruler = no is_adult = no }
				if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
				else = { death = { death_reason = death_slavery } }
			}
			primary_title = {
				activate_title = { title = THIS status = no }
				destroy_landed_title = THIS
			}
			if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
			else = { death = { death_reason = death_slavery } }
		}
	}
}

# Moves the march to its next stop and tells the players of the Dark Gods;
# at Zanbaijin the war begins (ravenchild.21).
# Scope: the child (ravenchild.9).
ravenchild_move_effect = {
	if = { # Kislev fell to Chaos, or a Black Crusade is under way: the march scatters
		limit = {
			OR = {
				k_kislev = { has_holder = yes holder_scope = { religion_group = chaos_gods_group } }
				c_kislev = { holder_scope = { religion_group = chaos_gods_group } }
				c_kislev = { holder_scope = { is_crusade_target = yes } }
				religion_head = { is_preparing_crusade = yes }
				religion_head = {
					OR = { any_war = { using_cb = crusade } any_war = { using_cb = new_crusade } }
				}
			}
		}
		any_player = {
			limit = { religion_group = chaos_gods_group NOT = { character = ROOT } }
			narrative_event = { id = ravenchild.20 days = 30 }
		}
		primary_title = {
			activate_title = { title = THIS status = no }
			destroy_landed_title = THIS
		}
		clr_character_flag = flag_childrens_crusade_leader
		add_trait = lunatic_2
		set_variable = { which = "lunatic_counter" value = 10 }
		piety = -150
		remove_trait = disinherited
		set_variable = { which = children_crusade_forces value = 0 }
		set_variable = { which = children_crusade_morale value = 0 }
		set_variable = { which = children_crusade_heavy_troops value = 0 }
		any_courtier = { remove_trait = disinherited clr_character_flag = no_court_invites }
	}
	else_if = { # First stop: the Far Wastes
		limit = { has_character_flag = flag_children_at_beginning }
		primary_title = { set_preferred_capital = c_the_ragged_maze } # The Inevitable City
		change_variable = { which = children_crusade_forces value = 1 }
		change_variable = { which = children_crusade_morale value = -1 }
		clr_character_flag = flag_children_at_beginning
		set_character_flag = flag_children_in_far_wastes
		if = {
			limit = {
				NOT = { persistent_event_target:childrens_crusade_sponsor_per = { is_alive = yes } }
			}
			change_variable = { which = children_crusade_morale value = -2 }
		}
		any_player = {
			limit = { religion_group = chaos_gods_group NOT = { character = ROOT } }
			narrative_event = { id = ravenchild.16 days = 30 }
		}
		narrative_event = { id = ravenchild.9 days = 50 random = 100 }
	}
	else_if = { # Then the Broken Lands
		limit = { has_character_flag = flag_children_in_far_wastes }
		primary_title = { set_preferred_capital = c_eiigin }
		change_variable = { which = children_crusade_morale value = -1 }
		clr_character_flag = flag_children_in_far_wastes
		set_character_flag = flag_children_in_broken_lands
		any_player = {
			limit = { religion_group = chaos_gods_group NOT = { character = ROOT } }
			narrative_event = { id = ravenchild.17 days = 30 }
		}
		if = {
			limit = {
				NOT = { persistent_event_target:childrens_crusade_sponsor_per = { is_alive = yes } }
			}
			random = {
				chance = 25
				change_variable = { which = children_crusade_forces value = -1 }
			}
		}
		narrative_event = { id = ravenchild.9 days = 50 random = 100 }
	}
	else_if = { # Then the Eastern Steppes
		limit = { has_character_flag = flag_children_in_broken_lands }
		primary_title = { set_preferred_capital = c_kharakit }
		clr_character_flag = flag_children_in_broken_lands
		set_character_flag = flag_children_in_eastern_steppes
		change_variable = { which = children_crusade_forces value = 1 }
		any_player = {
			limit = { religion_group = chaos_gods_group NOT = { character = ROOT } }
			narrative_event = { id = ravenchild.18 days = 30 }
		}
		narrative_event = { id = ravenchild.9 days = 50 random = 100 }
	}
	else_if = { # Zanbaijin at last: the war begins
		limit = { has_character_flag = flag_children_in_eastern_steppes }
		primary_title = { set_preferred_capital = c_zanbaijin } # Where the armies gather
		clr_character_flag = flag_children_in_eastern_steppes
		set_character_flag = flag_children_in_zanbaijin
		narrative_event = { id = ravenchild.21 days = 25 }
		any_player = {
			limit = { religion_group = chaos_gods_group NOT = { character = ROOT } }
			narrative_event = { id = ravenchild.19 days = 30 }
		}
	}
	else = { # Fallback: straight to Zanbaijin
		primary_title = { set_preferred_capital = c_zanbaijin }
		clr_character_flag = flag_children_in_far_wastes
		clr_character_flag = flag_children_in_broken_lands
		clr_character_flag = flag_children_in_eastern_steppes
		set_character_flag = flag_children_in_zanbaijin
		change_variable = { which = children_crusade_forces value = 1 }
		change_variable = { which = children_crusade_morale value = -1 }
		if = {
			limit = {
				NOT = { persistent_event_target:childrens_crusade_sponsor_per = { is_alive = yes } }
			}
			change_variable = { which = children_crusade_morale value = -2 }
		}
		narrative_event = { id = ravenchild.21 days = 25 }
		any_player = {
			limit = { religion_group = chaos_gods_group NOT = { character = ROOT } }
			narrative_event = { id = ravenchild.19 days = 30 }
		}
	}
}

# Spawns light troops until children_crusade_forces is spent. Scope: the child.
ravenchild_light_troops_effect = {
	change_variable = { which = children_crusade_forces value = -1 }
	location = {
		ROOT = {
			spawn_unit = {
				province = PREV
				home = PREV
				owner = ROOT
				attrition = 0.5
				reinforces = no
				is_looter = no
				can_toggle_looting = no
				cannot_inherit = yes
				merge = yes
				disband_on_peace = yes
				earmark = childrens_crusade_infantry_earmark
				troops = { light_infantry = { 3000 3000 } archers = { 100 100 } }
			}
		}
	}
	if = {
		limit = { check_variable = { which = children_crusade_forces value = 1 } }
		character_event = { id = ravenchild.22 }
	}
}
