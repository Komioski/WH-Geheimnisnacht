# Declare the reconquest war on whoever holds the occupied province nearest
# ROOT, with the CB of its land. With nothing left to take, the reconquest
# ends. Scope: the reconquistador (reconquest.49).
reconquest_declare_war_effect = {
	if = {
		limit = { any_province = { reconquest_occupied_trigger = yes } }
		random_province = {
			limit = { reconquest_occupied_trigger = yes }
			preferred_limit = { any_neighbor_province = { owner = { same_realm = ROOT } } }
			preferred_limit = {
				any_neighbor_province = {
					any_neighbor_province = { owner = { same_realm = ROOT } }
				}
			}
			owner = {
				if = { limit = { independent = yes } save_event_target_as = reconquest_enemy }
				else = { top_liege = { save_event_target_as = reconquest_enemy } }
			}
			if = {
				limit = { region = custom_astarios }
				ROOT = {
					unsafe_war = {
						target = event_target:reconquest_enemy
						casus_belli = reconquista_astarios
					}
				}
			}

				}
			}
			else_if = {

					unsafe_war = {
						target = event_target:reconquest_enemy
						casus_belli = reconquista_tigarre
					}
				}
			}
			else_if = {
				limit = { region = custom_castillona }
				ROOT = {
					unsafe_war = {

# Pay a sellsword lord by his rank: from ROOT to FROMFROM.
# Scope: the payer (reconquest.30).
reconquest_pay_sellsword_effect = {
	if = {
		limit = { FROMFROM = { real_tier = COUNT } }
		transfer_scaled_wealth = { to = FROMFROM value = 0.25 }
	}
	else_if = {
		limit = { FROMFROM = { real_tier = DUKE } }
		transfer_scaled_wealth = { to = FROMFROM value = 0.5 }
	}
	else_if = {
		limit = { FROMFROM = { real_tier = KING } }
		transfer_scaled_wealth = { to = FROMFROM value = 0.75 }
	}
	else = { transfer_scaled_wealth = { to = FROMFROM value = 1 } }
}

# Gold and the god of greed: a lord who sells his sword grows ambitious.
# Scope: the sellsword.
reconquest_sellsword_greed_effect = {
	if = { limit = { trait = content } remove_trait = content }
	else_if = { limit = { NOT = { trait = ambitious } } add_trait = ambitious }
}
