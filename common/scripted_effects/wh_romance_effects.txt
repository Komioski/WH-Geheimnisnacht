# A night together may bring a child: a 50% chance when both are fertile,
# 75% when both are very fertile. impregnate_effect decides who carries it
# and whether the child is a cuckoo in another marriage.
# Scope: ROOT, one lover. FROM: the other.
romance_conception_effect = {
	if = {
		limit = {
			is_opposite_sex = FROM
			fertility = 0.25
			FROM = { fertility = 0.25 }
		}
		random_list = {
			50 = {
				modifier = {
					factor = 3
					fertility = 0.5
					FROM = { fertility = 0.5 }
				}
				impregnate_effect = yes
			}
			50 = { }
		}
	}
}
