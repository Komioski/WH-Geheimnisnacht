# The next step of building a forge: one of the fittings not yet chosen
# (anvil, furnace, tools, rune tablets) while ambition is left, else the
# forge is finished. Scope: the runesmith.
runesmith_forge_next_step_effect = {
	if = {
		limit = { check_variable = { which = runesmith_forge_ambition value = 1 } }
		random_list = {
			25 = {
				trigger = { NOT = { has_character_flag = runesmith_forge_anvil } }
				character_event = { id = runesmith.13 days = 30 }
			}
			25 = {
				trigger = { NOT = { has_character_flag = runesmith_forge_furnace } }
				character_event = { id = runesmith.14 days = 30 }
			}
			25 = {
				trigger = { NOT = { has_character_flag = runesmith_forge_tools } }
				character_event = { id = runesmith.15 days = 30 }
			}
			25 = {
				trigger = { NOT = { has_character_flag = runesmith_forge_tablets } }
				character_event = { id = runesmith.16 days = 30 }
			}
		}
	}
	else = {
		character_event = { id = runesmith.17 days = 60 }
	}
}

# A chance of trouble while the forge is built. Scope: the runesmith.
runesmith_forge_mishap_effect = {
	random_list = {
		25 = { }
		25 = { character_event = { id = runesmith.23 days = 45 } }
		25 = { character_event = { id = runesmith.24 days = 45 } }
		25 = { character_event = { id = runesmith.25 days = 45 } }
	}
}

# Tears down a forge. Scope: the forge's owner.
runesmith_remove_forge_effect = {
	trigger_switch = {
		on_trigger = has_character_modifier
		runesmith_forge_1 = { remove_character_modifier = runesmith_forge_1 }
		runesmith_forge_2 = { remove_character_modifier = runesmith_forge_2 }
		runesmith_forge_3 = { remove_character_modifier = runesmith_forge_3 }
		runesmith_forge_4 = { remove_character_modifier = runesmith_forge_4 }
		runesmith_forge_5 = { remove_character_modifier = runesmith_forge_5 }
	}
	clr_character_flag = runesmith_built_forge
	clr_character_flag = runesmith_hidden_forge
	clr_character_flag = runesmith_deep_forge
}

# Forges a commissioned piece for event_target:runesmith_recipient.
# Reads the smith's flags: runesmith_craft_weapon, _armour or _rune, plus
# runesmith_craft_masterwork and runesmith_craft_gromril.
# Norse dwarfs favour the axe and have no use for the Rune of Swiftness.
# Scope: the runesmith.
runesmith_forge_item_effect = {
	if = {
		limit = { has_character_flag = runesmith_craft_gromril }
		if = {
			limit = { has_character_flag = runesmith_craft_weapon }
			event_target:runesmith_recipient = { add_artifact = dawi_gromril_hammer }
		}
		else = {
			event_target:runesmith_recipient = { add_artifact = dawi_artifact_gromril_armour }
		}
	}
	else_if = {
		limit = { has_character_flag = runesmith_craft_masterwork }
		if = {
			limit = { has_character_flag = runesmith_craft_weapon }
			random_list = {
				50 = {
					modifier = { factor = 0 event_target:runesmith_recipient = { culture = dwarf_norse } }
					event_target:runesmith_recipient = { add_artifact = dawi_battleaxe }
				}
				50 = {
					event_target:runesmith_recipient = { add_artifact = dawi_great_hammer }
				}
				50 = {
					modifier = { factor = 0 event_target:runesmith_recipient = { NOT = { culture = dwarf_norse } } }
					event_target:runesmith_recipient = { add_artifact = dawi_norse_battleaxe }
				}
			}
		}
		else_if = {
			limit = { has_character_flag = runesmith_craft_armour }
			event_target:runesmith_recipient = { add_artifact = dawi_steel_armour }
		}
		else = {
			random_list = {
				10 = {
					modifier = { factor = 0 event_target:runesmith_recipient = { culture = dwarf_norse } }
					event_target:runesmith_recipient = { add_artifact = master_rune_of_swiftness }
				}
				10 = { event_target:runesmith_recipient = { add_artifact = master_rune_of_stoicism } }
				10 = { event_target:runesmith_recipient = { add_artifact = master_rune_of_fear } }
				10 = { event_target:runesmith_recipient = { add_artifact = master_rune_of_courage } }
				10 = {
					modifier = { factor = 2 event_target:runesmith_recipient = { culture = dwarf_norse } }
					event_target:runesmith_recipient = { add_artifact = master_rune_of_fury }
				}
				10 = { event_target:runesmith_recipient = { add_artifact = master_rune_of_steel } }
				10 = { event_target:runesmith_recipient = { add_artifact = master_rune_of_shielding } }
			}
		}
	}
	else_if = {
		limit = { has_character_flag = runesmith_craft_weapon }
		random_list = {
			50 = {
				modifier = { factor = 0.5 event_target:runesmith_recipient = { culture = dwarf_norse } }
				event_target:runesmith_recipient = { add_artifact = dawi_hammer }
			}
			50 = {
				event_target:runesmith_recipient = { add_artifact = dawi_axe }
			}
		}
	}
	else_if = {
		limit = { has_character_flag = runesmith_craft_armour }
		event_target:runesmith_recipient = { add_artifact = dawi_light_armour }
	}
	else = {
		random_list = {
			10 = {
				modifier = { factor = 0 event_target:runesmith_recipient = { culture = dwarf_norse } }
				event_target:runesmith_recipient = { add_artifact = rune_of_swiftness }
			}
			10 = { event_target:runesmith_recipient = { add_artifact = rune_of_dismay } }
			10 = { event_target:runesmith_recipient = { add_artifact = rune_of_steel } }
			10 = { event_target:runesmith_recipient = { add_artifact = rune_of_stoicism } }
			10 = {
				modifier = { factor = 2 event_target:runesmith_recipient = { culture = dwarf_norse } }
				event_target:runesmith_recipient = { add_artifact = rune_of_fury }
			}
			10 = { event_target:runesmith_recipient = { add_artifact = rune_of_shielding } }
			10 = { event_target:runesmith_recipient = { add_artifact = rune_of_courage } }
		}
	}
}

# A Runemoot working blows up: rolls what happens to a guest.
# Scope: a guest.
runesmith_moot_blast_roll_effect = {
	random_list = {
		20 = { set_character_flag = invention_failure_death }
		20 = { set_character_flag = invention_failure_maimed }
		20 = { set_character_flag = invention_failure_wounded }
		40 = { }
	}
}

# Applies the rolled harm. Scope: a guest.
runesmith_moot_blast_effect = {
	trigger_switch = {
		on_trigger = has_character_flag
		invention_failure_death = { death = { death_reason = death_accident } }
		invention_failure_maimed = { add_maimed_trait_effect = yes }
		invention_failure_wounded = { add_trait_silently_wounded_effect = yes }
	}
}

# Clears the roll. Scope: a guest.
runesmith_moot_blast_clear_effect = {
	clr_character_flag = invention_failure_death
	clr_character_flag = invention_failure_maimed
	clr_character_flag = invention_failure_wounded
}
