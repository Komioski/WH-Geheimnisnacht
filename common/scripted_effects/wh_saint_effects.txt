# Venerated Souls (saints) effects, used by events/WHG_saint_events.txt.

# The roll at death that decides whether the dead may one day be raised to
# a Venerated Soul. If it passes, the case comes before the head of the
# cult three to thirty years later (one to three for a former player).
# Scope: the dead character (saint.1).
saint_canonization_roll_effect = {
	random = {
		chance = 1
		modifier = { factor = 0 NOT = { rightful_religious_head_scope = { always = yes } } }
		modifier = {
			factor = 0
			NOR = { has_character_flag = beatified has_religion_feature = religion_beatification }
		}
		modifier = { factor = 0.8 trait = cynical }
		modifier = {
			factor = 0.4
			OR = {
				has_landed_title = k_cult_of_sigmar
				has_landed_title = k_cult_of_ulric
				has_landed_title = k_cult_of_taal_and_rhya
				has_landed_title = k_cult_of_morr
			}
		}
		modifier = { factor = 1.5 has_any_opinion_modifier = opinion_baptism_sponsor_target }
		modifier = {
			factor = 2 # Players without a holy bloodline
			ai = no
			NOT = { any_owned_bloodline = { has_bloodline_flag = saintly_bloodline } }
		}
		modifier = {
			factor = 0.2 # The blood of a Venerated Soul already
			any_owned_bloodline = { has_bloodline_flag = saintly_bloodline }
		}
		modifier = { factor = 1.5 trait = patient }
		modifier = { factor = 1.5 trait = diligent }
		modifier = { factor = 1.5 trait = kind }
		modifier = { factor = 1.5 trait = chaste }
		modifier = { factor = 1.5 trait = charitable }
		modifier = { factor = 1.5 trait = temperate }
		modifier = { factor = 1.5 trait = humble }
		modifier = { factor = 1.2 trait = just }
		modifier = { factor = 1.2 has_crusade_trait_trigger = yes }
		modifier = {
			factor = 2 # Martyrs above all
			martyrs_death_trigger = yes
		}
		modifier = { factor = 1.5 piety = 1000 }
		modifier = { factor = 1.5 piety = 2000 }
		modifier = { factor = 1.5 piety = 3500 }
		modifier = { factor = 1.5 piety = 5000 }
		modifier = { factor = 1.5 piety = 7500 }
		modifier = { factor = 1.5 piety = 10000 }
		modifier = {
			factor = 3 # The first of a faith comes easier
			saint_first_of_faith_trigger = yes
		}
		saint_mark_faith_effect = yes
		random_realm_province = {
			if = {
				limit = { ROOT = { trait = i_used_to_be_a_player } }
				province_event = { id = saint.11 days = 365 random = 730 }
			}
			else = { province_event = { id = saint.11 days = 1095 random = 9855 } }
		}
	}
}

# Tells the dead one's realm and kin that they were named Pious (saint.3).
# Scope: the proclaimer; FROM of saint.3 is this character.
saint_news_pious_effect = {
	any_playable_ruler = {
		limit = {
			religion_group = event_target:saint_person
			same_realm = event_target:saint_top_liege
		}
		narrative_event = { id = saint.3 }
		set_character_flag = saint_news_sent
	}
	event_target:saint_person = {
		any_dynasty_member = {
			limit = {
				is_ruler = yes
				religion_group = event_target:saint_person
				NOT = { has_character_flag = saint_news_sent }
			}
			narrative_event = { id = saint.3 }
		}
	}
	saint_news_clear_effect = yes
}

# Raises the tomb: in the soul's capital, else a free neighbour of it, else
# a demesne or realm province without a tomb.
# Scope: the head of the cult (saint.12).
saint_raise_tomb_effect = {
	event_target:saint_capital = { owner = { save_event_target_as = saint_capital_owner } }
	if = {
		limit = { event_target:saint_capital = { saint_no_tomb_trigger = yes } }
		event_target:saint_capital = { saint_tomb_here_effect = yes }
	}
	else_if = {
		limit = {
			event_target:saint_capital = {
				any_neighbor_province = {
					owner = { character = event_target:saint_capital_owner }
					saint_no_tomb_trigger = yes
				}
			}
		}
		event_target:saint_capital = {
			random_neighbor_province = {
				limit = {
					owner = { character = event_target:saint_capital_owner }
					saint_no_tomb_trigger = yes
				}
				saint_tomb_here_effect = yes
			}
		}
	}
	else_if = {
		limit = { event_target:random_saint_demesne_province = { saint_no_tomb_trigger = yes } }
		event_target:random_saint_demesne_province = { saint_tomb_here_effect = yes }
	}
	else_if = {
		limit = { event_target:random_saint_realm_province = { saint_no_tomb_trigger = yes } }
		event_target:random_saint_realm_province = { saint_tomb_here_effect = yes }
	}
}

# Scope: the province of the tomb.
saint_tomb_here_effect = {
	save_persistent_event_target = { name = province_saint scope = event_target:saint_person }
	add_province_modifier = { name = tomb_of_a_saint duration = -1 }
	owner = { character_event = { id = saint.13 } }
}

# Founds the holy bloodline of a new Venerated Soul. The kind follows the
# soul's life: defenders for zealots, monastic for the celibate; the style
# follows the faith (the imperial cults, the old faith, the rest).
# Scope: the head of the cult (saint.12).
saint_found_bloodline_effect = {
	event_target:saint_person = {
		random_list = {
			20 = {
				trigger = { OR = { trait = zealous trait = holy_warrior } }
				saint_bloodline_defender_effect = yes
			}
			20 = {
				trigger = { OR = { trait = celibate trait = monk trait = nun } }
				saint_bloodline_monastic_effect = yes
			}
			5 = { saint_bloodline_cultural_effect = yes }
			5 = {
				modifier = { factor = 4 OR = { trait = zealous trait = holy_warrior } }
				saint_bloodline_inquisitor_effect = yes
			}
			5 = { saint_bloodline_theologian_effect = yes }
		}
		new_bloodline = {
			set_name = saintly_bloodline_name
			set_description = saintly_bloodline_description
		}
	}
}

# Scope: the Venerated Soul.

# Sets where the pilgrim walks (the north, the south, the heart of the Old
# World, or far lands) and sends him on his way (saint.61).
# Scope: the pilgrim (saint.60).
saint_pilgrim_setout_effect = {
	if = {
		limit = {
			event_target:pilgrimage_destination = {
				OR = {
					region = world_old_world_kislev
					region = world_norsca
					region = world_old_world_empire_north
					region = world_karaz_ankor_northern_holds
				}
			}
		}
		set_character_flag = saint_pilgrim_north
	}
	else_if = {
		limit = {
			event_target:pilgrimage_destination = {
				OR = {
					region = world_old_world_tilea
					region = world_old_world_estalia
					region = world_old_world_border_princes
					region = world_greater_araby
					region = world_badlands
					region = world_nehekhara
				}
			}
		}
		set_character_flag = saint_pilgrim_south
	}
	else_if = {
		limit = {
			event_target:pilgrimage_destination = {
				OR = {
					region = world_old_world_empire
					region = world_old_world_bretonnia
					region = world_athel_loren
					region = world_albion
				}
			}
		}
		set_character_flag = saint_pilgrim_home
	}
	else = { set_character_flag = saint_pilgrim_far }
	character_event = { id = saint.61 days = 15 }
}
