# The AI settles a mission in the background: usually a success, less often
# the more influence the society already has.
# Scope: the member.
seaquest_ai_settles_mission_effect = {
	random_list = {
		70 = {
			modifier = { factor = 0.9 society_influence >= 30 }
			modifier = { factor = 0.9 society_influence >= 40 }
			modifier = { factor = 0.8 society_influence >= 50 }
			modifier = { factor = 0.7 society_influence >= 70 }
			modifier = { factor = 0.6 society_influence >= 80 }
			modifier = { factor = 0.5 society_influence >= 90 }
			add_mission_succeed_influence_effect = yes
		}
		30 = {
			add_mission_fail_influence_effect = yes
		}
	}
}

# The Game of Reavers begins: the score is set to nought and the quest for
# the coast of the game is set on the rival (event_target:pirate_rival).
# Scope: the member (ROOT).
seaquest_start_game_of_reavers_effect = {
	set_character_flag = game_of_reavers_flag
	set_variable = { which = game_of_reavers_value value = 0 }
	event_target:pirate_rival = {
		show_scope_change = no
		if = {
			limit = { ROOT = { has_character_flag = ne_game_of_reavers_flag } }
			set_quest_target = { id = quest_game_of_reavers_pirate_ne holder = ROOT }
		}
		else_if = {
			limit = { ROOT = { has_character_flag = b_game_of_reavers_flag } }
			set_quest_target = { id = quest_game_of_reavers_pirate_b holder = ROOT }
		}
		else_if = {
			limit = { ROOT = { has_character_flag = e_game_of_reavers_flag } }
			set_quest_target = { id = quest_game_of_reavers_pirate_e holder = ROOT }
		}
		else_if = {
			limit = { ROOT = { has_character_flag = t_game_of_reavers_flag } }
			set_quest_target = { id = quest_game_of_reavers_pirate_t holder = ROOT }
		}
		else = {
			set_quest_target = { id = quest_game_of_reavers_pirate_u holder = ROOT }
		}
	}
}

# The rival takes the challenger's coast of the game.
# Scope: the rival. ROOT: the challenger.
seaquest_copy_reavers_region_effect = {
	if = {
		limit = { ROOT = { has_character_flag = ne_game_of_reavers_flag } }
		set_character_flag = ne_game_of_reavers_flag
	}
	else_if = {
		limit = { ROOT = { has_character_flag = b_game_of_reavers_flag } }
		set_character_flag = b_game_of_reavers_flag
	}
	else_if = {
		limit = { ROOT = { has_character_flag = e_game_of_reavers_flag } }
		set_character_flag = e_game_of_reavers_flag
	}
	else_if = {
		limit = { ROOT = { has_character_flag = t_game_of_reavers_flag } }
		set_character_flag = t_game_of_reavers_flag
	}
	else = {
		set_character_flag = u_game_of_reavers_flag
	}
}

# Clears the coast flags of the Game of Reavers.
# Scope: a player.
seaquest_clear_reavers_flags_effect = {
	clr_character_flag = ne_game_of_reavers_flag
	clr_character_flag = b_game_of_reavers_flag
	clr_character_flag = e_game_of_reavers_flag
	clr_character_flag = t_game_of_reavers_flag

# The Game of Reavers ends for this player.
# Scope: a player.
seaquest_end_game_of_reavers_effect = {
	clr_quest = quest_game_of_reavers_pirate_ne
	clr_quest = quest_game_of_reavers_pirate_b
	clr_quest = quest_game_of_reavers_pirate_e
	clr_quest = quest_game_of_reavers_pirate_t
	clr_quest = quest_game_of_reavers_pirate_u
	clr_character_flag = game_of_reavers_flag
	set_variable = { which = game_of_reavers_value value = 0 }
	seaquest_clear_reavers_flags_effect = yes
	clear_delayed_event = { id = seaquest.17 }
}

# The admiral's hunt is over: his squadrons disband and the admiralty is
# dissolved.
# Scope: the admiral.
seaquest_admiral_retires_effect = {
	clr_character_flag = dealwadmiral_target_flag
	disband_event_forces = yes
	if = {
		limit = { primary_title = { has_title_flag = temporary_admiralty_title } }
		primary_title = { unsafe_destroy_landed_title = THIS }
	}
}

# A prize was taken: counts towards the three prizes mission.
# Scope: the captain.
seaquest_count_prize_effect = {
	if = {
		limit = { has_quest = quest_sea_three_prizes }
		change_variable = { which = sea_mission_prizes value = 1 }
		if = {
			limit = { check_variable = { which = sea_mission_prizes value = 3 } }
			character_event = { id = seaquest.65 }
		}
	}
}
