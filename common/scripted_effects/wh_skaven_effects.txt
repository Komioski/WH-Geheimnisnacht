# skaven)

# A province cleared of Skaven is resettled (skaven.71). Tomb Kings bring
# the living of Nehekhara; the Lost of the jade vampires bring their own;
# Blood Dragons and the witch hunter peoples, who do not settle, bring the
# people of the region (or, beyond the known lands, the Strigany who serve
# the vampires and the Reiklanders who follow the hunters). Anyone else
# brings their own people and faith.
# Scope: the province. ROOT: the lord.
skaven_resettle_province_effect = {
	if = {
		limit = { ROOT = { trait = creature_nehekharan } }
		culture = nehekka
		religion = atruhayid
	}
	else_if = {
		limit = { ROOT = { culture = living_jade } }
		culture = lizardman_lost
		religion = vampiric
	}
	else_if = {
		limit = {
			ROOT = {
				OR = {
					culture = blooddragons_culture
					is_hunter_culture_trigger = yes
				}
			}
		}
		religion = ROOT
		if = {
			limit = { region = world_old_world_empire }
			culture = reiklander
		}
		else_if = {
			limit = { region = world_old_world_border_princes }
			culture = neueslander
		}
		else_if = {
			limit = { region = world_old_world_kislev }
			culture = kislevite
		}
		else_if = {
			limit = { region = world_troll_country }
			culture = ropsmann
		}
		else_if = {
			limit = { region = world_old_world_bretonnia }
			culture = nortagnese
		}
		else_if = {
			limit = { region = world_old_world_estalia }
			culture = astur
		}
		else_if = {
			limit = { region = world_old_world_tilea }
			culture = reman
		}
		else_if = {
			limit = { region = world_badlands_strygos }
			culture = strigany
		}
		else_if = {
			limit = { region = world_badlands_magnata }
			culture = myken
		}
		else_if = {
			limit = { region = world_greater_araby_araby }
			culture = ramaccan
		}
		else_if = {
			limit = { region = world_greater_araby_southlands }
			culture = wasai
		}
		else_if = {
			limit = { region = world_nehekhara }
			culture = nehekka
		}
		else_if = {
			limit = { region = world_norsca }
			culture = aesling
		}
		else_if = {
			limit = { region = world_chaos_wastes }
			culture = kul
		}
		else_if = {
			limit = { ROOT = { culture = blooddragons_culture } }
			culture = strigany
		}
		else = {
			culture = reiklander
		}
	}
	else = {
		culture = ROOT
		religion = ROOT
	}
}
