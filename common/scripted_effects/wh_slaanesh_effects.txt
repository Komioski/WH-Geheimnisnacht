# The character stops hiding: they leave the secret cult and take up the open
# Slaaneshi faith of their people (the Cult of Pleasure for Asur and Asrai,
# Atharti for Druchii, Slaanesh for everyone else). A ruler of ten years or
# more has already turned their capital.
slaanesh_open_worship_effect = {
	leave_society = yes
	if = { limit = { religion_group = cadai_group } religion = cult_of_pleasure }
	else_if = { limit = { religion_group = cytharai_group } religion = atharti }
	else = { religion = slaanesh }
	if = { limit = { is_ruler = yes ruled_years = 10 } capital_scope = { religion = ROOT } }
}

# A prisoner was polymorphed. ROOT is the victim, FROM the caster. The caster's
# Asur vassals call it barbaric and the victim's family never forgets.
slaanesh_polymorph_outrage_effect = {
	hidden_effect = {
		FROM = {
			any_vassal = {
				limit = { religion = cadai }
				opinion = { modifier = opinion_barbaric who = FROM }
			}
		}
		any_dynasty_member = {
			limit = { religion_group = cytharai_group }
			opinion = { modifier = opinion_polymorphed_family_druchii who = FROM }
		}
		any_dynasty_member = {
			limit = { NOT = { religion_group = cytharai_group } }
			opinion = { modifier = opinion_polymorphed_family who = FROM }
		}
	}
}

# The caster's court and vassals saw what happened to the prisoner. All but the
# brave and the clever are scared into line.
slaanesh_scare_court_effect = {
	any_vassal = {
		limit = { NOR = { trait = brave trait = quick trait = brilliant trait = genius } }
		opinion = { modifier = opinion_scared_into_submission who = ROOT }
	}
	any_courtier = {
		limit = { NOR = { trait = brave trait = quick trait = brilliant trait = genius } }
		opinion = { modifier = opinion_scared_into_submission who = ROOT }
	}
}
