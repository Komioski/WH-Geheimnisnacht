# THIS (a necromancer) gains one level of necromantic power, up to 5.
spell_necro_power_up_effect = {
	if = {
		limit = { has_character_modifier = necromantic_power_4 }
		remove_character_modifier = necromantic_power_4
		add_character_modifier = { name = necromantic_power_5 duration = -1 }
	}
	else_if = {
		limit = { has_character_modifier = necromantic_power_3 }
		remove_character_modifier = necromantic_power_3
		add_character_modifier = { name = necromantic_power_4 duration = -1 }
	}
	else_if = {
		limit = { has_character_modifier = necromantic_power_2 }
		remove_character_modifier = necromantic_power_2
		add_character_modifier = { name = necromantic_power_3 duration = -1 }
	}
	else_if = {
		limit = { has_character_modifier = necromantic_power_1 }
		remove_character_modifier = necromantic_power_1
		add_character_modifier = { name = necromantic_power_2 duration = -1 }
	}
	else_if = {
		limit = { NOT = { has_character_modifier = necromantic_power_5 } }
		add_character_modifier = { name = necromantic_power_1 duration = -1 }
	}
}

# THIS (a necromancer) rolls whether the dead answer in greater numbers than
# he called for: about even odds, better for vampires, Nagash's faithful,
# Tomb Kings and stronger casters, worse for hedge wizards.
spell_horde_size_roll_effect = {
	clr_character_flag = spell_horde_big
	random_list = {
		50 = { }
		50 = {
			modifier = { factor = 1.75 religion = nagashi }
			modifier = { factor = 0.85 trait = hedge_wizard }
			modifier = { factor = 1.85 trait = creature_nehekharan }
			modifier = { factor = 1.75 is_vampire_trigger = yes }
			modifier = { factor = 1.65 has_character_modifier = magic_power_2 }
			modifier = { factor = 1.75 has_character_modifier = magic_power_3 }
			modifier = { factor = 1.85 has_character_modifier = magic_power_4 }
			set_character_flag = spell_horde_big
		}
	}
}

# THIS (a mage of Life) wakes the forest at his location. Weaker casters
# wake a small or average wood, stronger ones an average or great one.
# The trees go back to sleep after 15 months or more (spell.72).

	if = {
		limit = {
			OR = { has_character_modifier = magic_power_3 has_character_modifier = magic_power_4 }

			50 = { set_character_flag = spell_wood_3 }

			50 = { set_character_flag = spell_wood_1 }
			50 = { set_character_flag = spell_wood_2 }

	if = {
		limit = { has_character_flag = spell_wood_3 }
		character_event = { id = spell.72 days = 450 random = 690 }

				spawn_unit = {

# THIS (a mage) is turned into a beast: his race is stripped and the given
# creature trait is added by the caller. A transformed ruler loses his
# claim on the throne.
spell_polymorph_effect = {
	remove_race_trait_effect = yes
	add_trait = disinherited
	opinion = { modifier = opinion_polymorphed_me who = FROM }
}

# THIS (a ruler who has just shown his magic on a prisoner) frightens his
# court: the vassals and courtiers who are not brave or clever fear him.
spell_cowed_court_effect = {
	any_vassal = {
		limit = { NOR = { trait = brave trait = quick trait = brilliant trait = genius } }
		opinion = { modifier = opinion_scared_into_submission who = ROOT }
	}
	any_courtier = {
		limit = { NOR = { trait = brave trait = quick trait = brilliant trait = genius } }
		opinion = { modifier = opinion_scared_into_submission who = ROOT }
	}
}
