# Raise a wight at ROOT's court: a dead champion of ROOT's culture and faith,
# bound to ROOT. Set the character flag undead_raise_king on ROOT first for a
# Wight King, who rides a nightmare; the flag is cleared here. The wight is saved as
# event_target:undead_new_wight.
# Scope: ROOT is the vampire or necromancer.
wight_rise_effect = {
	set_character_flag = undead_wight_master
	if = {
		limit = { has_character_flag = undead_raise_king }
		create_character = {
			age = 50
			female = no
			culture = ROOT
			religion = ROOT
			dynasty = none
			attributes = { martial = 14 diplomacy = 2 stewardship = 2 intrigue = 3 learning = 4 }
			trait = creature_wight
			trait = brilliant_strategist
			trait = brave
			trait = wroth
		}
		new_character = {
			add_character_modifier = { name = undead_wight_king duration = -1 }
			add_artifact = nightmare_mount
			set_character_flag = has_mount
		}
	}
	else = {
		create_character = {
			age = 40
			female = no
			culture = ROOT
			religion = ROOT
			dynasty = none
			attributes = { martial = 9 diplomacy = 2 stewardship = 2 intrigue = 2 learning = 2 }
			trait = creature_wight
			trait = skilled_tactician
			trait = brave
		}
	}
	new_character = {
		set_graphical_culture = wight
		random_list = {
			40 = { }
			15 = { add_trait = duelist }
			15 = { add_trait = strategist }
			10 = { add_trait = cruel }
			10 = { add_trait = stubborn }
			10 = { change_martial = 2 }
		}
		save_event_target_as = undead_new_wight
	}
	clr_character_flag = undead_raise_king
}

# A Wight King may hand his master the blade he was buried with: 5% a
# raising, once a lifetime, never while ROOT already holds one.
# Scope: ROOT is the master.
wight_tomb_blade_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = undead_got_tomb_blade }
			NOT = { has_artifact = undead_tomb_blade }
		}
		random = {
			chance = 5
			set_character_flag = undead_got_tomb_blade
			add_artifact = undead_tomb_blade
		}
	}
}

# The Dark Enchantment turns event_target:undead_old_item, a fine weapon,
# into a necromantic one of the same kind, and destroys the old one. Swords
# become a Frostblade, Accursed Blade or Tomb Blade (pirates: a Blade of the
# Damned, Ebony Blade or Dirty Serpent); lances and spears a Dreadlance or
# Accursed Lance; axes a Black Axe. A Necrarch's work leans to the finer
# blades.
# Scope: ROOT is the owner; flag undead_enchant_necrarch if a Necrarch did
# the work.
undead_enchant_weapon_effect = {
	if = {
		limit = { event_target:undead_old_item = { has_artifact_flag = axe } }
		add_artifact = undead_black_axe
	}
	else_if = {
		limit = {
			event_target:undead_old_item = {
				OR = {
					has_artifact_flag = lance
					has_artifact_flag = spear
				}
			}
		}
		random_list = {
			50 = { add_artifact = undead_dreadlance }
			50 = {
				modifier = { factor = 1.5 has_character_flag = undead_enchant_necrarch }
				add_artifact = undead_accursed_lance
			}
		}
	}
	else_if = {
		limit = { vamp_pirate_trigger = yes }
		random_list = {
			35 = { add_artifact = undead_dirty_serpent }
			35 = {
				modifier = { factor = 1.5 has_character_flag = undead_enchant_necrarch }
				add_artifact = undead_ebony_blade
			}
			30 = {
				modifier = { factor = 0 OR = { trait = kind trait = just } }
				add_artifact = undead_blade_of_the_damned
			}
		}
	}
	else = {
		random_list = {
			40 = { add_artifact = undead_frostblade }
			30 = {
				modifier = { factor = 1.5 has_character_flag = undead_enchant_necrarch }
				add_artifact = undead_accursed_blade
			}
			30 = {
				modifier = { factor = 1.5 has_character_flag = undead_enchant_necrarch }
				add_artifact = undead_tomb_blade
			}
		}
	}
	event_target:undead_old_item = { destroy_artifact = yes }
}

# The Dark Enchantment turns event_target:undead_old_item, fine armour, into
# necromantic armour and destroys the old one. Living pirates get a Pirate
# Captain's Coat; vampire pirates a Drowned Admiral's Coat, Armour of the
# Depths or Wyrmscale Armour; everyone else one of six suits, weighted by
# their learning and nature. A Necrarch's work leans to the finer suits.
# Scope: ROOT is the owner; flag undead_enchant_necrarch if a Necrarch did
# the work.
undead_enchant_armour_effect = {
	if = {
		limit = {
			vamp_pirate_trigger = yes
			is_vampire_trigger = no
		}
		add_artifact = undead_pirate_captain_coat
	}
	else_if = {
		limit = { vamp_pirate_trigger = yes }
		random_list = {
			45 = { add_artifact = undead_drowned_admiral_coat }
			35 = { add_artifact = undead_armour_of_the_depths }
			20 = {
				modifier = { factor = 2 has_character_flag = undead_enchant_necrarch }
				add_artifact = undead_wyrmscale_armour
			}
		}
	}
	else = {
		random_list = {
			20 = {
				modifier = { factor = 2 has_education_intrigue_trigger = yes }
				modifier = { factor = 1.5 has_education_learning_trigger = yes }
				add_artifact = undead_cadaverous_cuirass
			}
			20 = {
				modifier = { factor = 2 has_education_learning_trigger = yes }
				modifier = { factor = 1.5 trait = lore_necromancy }
				add_artifact = undead_bone_cuirass
			}
			20 = {
				modifier = { factor = 0 trait = vampire_lahmian_visible }
				modifier = { factor = 0 trait = vampire_lahmian_hidden }
				modifier = { factor = 0 trait = vampire_necrarch_visible }
				modifier = { factor = 1.5 trait = cruel }
				add_artifact = undead_flayed_hauberk
			}
			20 = {
				modifier = { factor = 2 has_education_martial_trigger = yes }
				modifier = { factor = 1.5 trait = vampire_blood_visible }
				add_artifact = undead_bloody_hauberk
			}
			10 = {
				modifier = { factor = 1.5 has_education_martial_trigger = yes }
				modifier = { factor = 2 has_character_flag = undead_enchant_necrarch }
				add_artifact = undead_armour_of_night
			}
			10 = {
				modifier = { factor = 1.5 has_education_intrigue_trigger = yes }
				modifier = { factor = 2 has_character_flag = undead_enchant_necrarch }
				add_artifact = undead_accursed_armour
			}
		}
	}
	event_target:undead_old_item = { destroy_artifact = yes }
}

# Clears the Dark Enchantment flags.
# Scope: ROOT is the owner.
undead_enchant_clear_effect = {
	clr_character_flag = undead_enchant_self
	clr_character_flag = undead_enchant_necromancer
	clr_character_flag = undead_enchant_necrarch
	clr_character_flag = undead_enchant_weapon
}
