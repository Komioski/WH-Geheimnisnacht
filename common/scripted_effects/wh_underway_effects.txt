# Underway scripted effects
# Used by events/WHG_underway_events.txt (namespace underway).
#
# Each Underway wonder upgrade (upgrade_underway_<hold>) opens one segment of
# the Ungdrin Ankor: a fixed list of counties that get the ungdrin_ankor
# modifier (ungdrin_ankor2 for the lowland counties of Norsca). A Train
# Station (upgrade_trains) in the same wonder lays mining carriage rails
# through those counties (ungdrin_trains).
#
# The segment effects check has_wonder_upgrade, so they work in two scopes:
# in an upgrade scope they act on that one upgrade's segment, in a wonder
# scope on every segment the wonder has.
###############################################################################

# Open one county of the Underway. Scope: the county.
underway_open_province_effect = {
	if = {
		limit = {
			NOR = { has_province_modifier = ungdrin_ankor has_province_modifier = ungdrin_ankor2 }
		}
		add_province_modifier = { name = ungdrin_ankor duration = -1 }
	}
}

# Open one lowland county of the Norscan Underway. Scope: the county.
underway_open_lowland_province_effect = {
	if = {
		limit = {
			NOR = { has_province_modifier = ungdrin_ankor has_province_modifier = ungdrin_ankor2 }
		}
		add_province_modifier = { name = ungdrin_ankor2 duration = -1 }
	}
}

# Close one county: the tunnels and the rails are lost. Scope: the county.
underway_close_province_effect = {
	remove_province_modifier = ungdrin_ankor
	remove_province_modifier = ungdrin_ankor2
	remove_province_modifier = ungdrin_trains
}

# Lay mining carriage rails in one open county. Scope: the county.
underway_carriages_on_province_effect = {
	if = {
		limit = {
			OR = { has_province_modifier = ungdrin_ankor has_province_modifier = ungdrin_ankor2 }
			NOT = { has_province_modifier = ungdrin_trains }
		}
		add_province_modifier = { name = ungdrin_trains duration = -1 }
	}
}
