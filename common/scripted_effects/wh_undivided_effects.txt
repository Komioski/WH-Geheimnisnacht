	location = {
		ROOT = {
			spawn_unit = {
				province = PREV
				troops = {
					horse_archers = { 2700 2700 }
					light_cavalry = { 2500 2500 }
					light_infantry = { 5000 5000 }
					heavy_infantry = { 5000 5000 }
					knights = { 500 500 }
					archers = { 5000 5000 }
				}
				earmark = everchosen_troops
			}
		}
	}
}

# Marks a trial champion for a duel to the death with the seeker
# (event_target:combatant_1) and has it vanish afterwards (undivided.37).
# Scope: the champion. ROOT = the seeker.
undivided_trial_champion_setup_effect = {
	set_character_flag = duel
	set_character_flag = flag_duel_no_draws
	set_character_flag = deadly_duel@event_target:combatant_1
	set_character_flag = fought_duel_against@ROOT
	character_event = { id = undivided.37 days = 21 }
}
