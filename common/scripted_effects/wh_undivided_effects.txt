# Marks a trial champion for a duel to the death with the seeker
# (event_target:combatant_1) and has it vanish afterwards (undivided.37).

	set_character_flag = deadly_duel@event_target:combatant_1

}

# Clears whatever stage the seeker was on and restarts the reminder count.
# Scope: the seeker.
undivided_quest_clear_stage_effect = {
	clr_character_flag = everchosen_stage_altar
	clr_character_flag = everchosen_stage_coast
	clr_character_flag = everchosen_stage_necropolis
	clr_character_flag = everchosen_stage_cliff
	clr_character_flag = everchosen_stage_hunt
	clr_character_flag = everchosen_stage_zanbaijin
	clr_character_flag = everchosen_busy
	remove_character_modifier = everchosen_quest_altar
	remove_character_modifier = everchosen_quest_coast
	remove_character_modifier = everchosen_quest_necropolis
	remove_character_modifier = everchosen_quest_cliff
	remove_character_modifier = everchosen_quest_hunt
	remove_character_modifier = everchosen_quest_zanbaijin
	set_variable = { which = everchosen_reminders value = 0 }
}

# Gives each artifact of the Everchosen the character does not hold yet, so
# every path (AI, default rule or the extended quest) ends with all five.
# Scope: the Everchosen.
undivided_give_missing_artifacts_effect = {
	if = {
		limit = { NOT = { has_artifact = armor_of_morkar } }
		add_artifact = armor_of_morkar
		new_artifact = { set_creation_date = 1.1.1 }
	}
	if = {
		limit = { NOT = { has_artifact = eye_of_sheerian } }
		add_artifact = eye_of_sheerian
		new_artifact = { set_creation_date = 1.1.1 }
	}
	if = {
		limit = { NOT = { has_artifact = slayer_of_kings } }
		add_artifact = slayer_of_kings
		new_artifact = { set_creation_date = 1.1.1 }
	}
	if = {
		limit = { NOT = { has_artifact = steed_of_the_apocalypse_mount } }
		add_artifact = steed_of_the_apocalypse_mount
		new_artifact = { set_creation_date = 1.1.1 }
	}
	if = {
		limit = { NOT = { has_artifact = crown_of_domination } }
		add_artifact = crown_of_domination
		new_artifact = { set_creation_date = 1.1.1 }
	}
}

# The artifacts of the Everchosen vanish with their bearer, so the next
# seeker can win them again.
# Scope: the dead Everchosen or seeker.
undivided_destroy_artifacts_effect = {
	unsafe_destroy_artifact = armor_of_morkar
	unsafe_destroy_artifact = crown_of_domination
	unsafe_destroy_artifact = eye_of_sheerian
	unsafe_destroy_artifact = slayer_of_kings
	unsafe_destroy_artifact = steed_of_the_apocalypse_mount
}

# Ends the quest: flags, stage, the artifacts won so far, and the claim on
# the Mark, so another seeker can try.
# Scope: the seeker.
undivided_quest_reset_effect = {
	undivided_quest_clear_stage_effect = yes
	clr_character_flag = potential_everchosen
	clr_character_flag = everchosen_marked
	clr_character_flag = everchosen_has_dorghar
	clr_character_flag = everchosen_has_slayer
	clr_character_flag = everchosen_on_dorghar_road
	clr_global_flag = everchosen_quest_claimed
	undivided_destroy_artifacts_effect = yes
	if = { limit = { trait = on_pilgrimage } remove_trait = on_pilgrimage }
}

# A guardian of a treasure grows harder to beat as the Winds of Chaos weaken:
# at very high winds the gods favour the seeker.
# Scope: the guardian.
undivided_winds_temper_guardian_effect = {
	add_five_duel_experience_modifier = yes
	add_five_duel_experience_modifier = yes
	if = {
		limit = { is_doom_very_high_trigger = no }
		add_five_duel_experience_modifier = yes
	}
}

# The Everchosen is crowned: the Winds of Chaos blow at their strongest.
# Sets the doom counter to its maximum and the very high flag at once
# (doom.1 would set it on its next pass anyway).
# Scope: any.
undivided_doom_to_max_effect = {
	custom_tooltip = { text = undivided_doom_to_max_tt }
	hidden_effect = {
		set_variable = { which = "global_doom_counter" value = 1500 }
		set_global_flag = doom_very_high
		clr_global_flag = doom_very_low
		clr_global_flag = doom_low
		clr_global_flag = doom_medium
		clr_global_flag = doom_high
	}
}

# The host of the necropolis (undivided.55): the shade of Morkar takes a

	create_title = {
		tier = DUKE

		ROOT = {
