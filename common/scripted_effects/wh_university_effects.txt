# Picks the university nearest the student whose city is open, and saves it
# as event_target:university. Altdorf takes anyone whose own region has no
# open university. If even Altdorf is closed, no target is saved.
# Scope: the student.
university_pick_effect = {
	if = {
		limit = {
			location = { region = world_old_world_empire_south }
			309 = { university_city_open_trigger = yes }
		}
		d_university_of_nuln = { save_event_target_as = university }
	}
	else_if = {
		limit = {
			location = { region = world_old_world_empire_east }
			355 = { university_city_open_trigger = yes }
		}
		d_university_of_streissen = { save_event_target_as = university }
	}
	else_if = {
		limit = {
			OR = {
				location = { region = world_old_world_kislev }
				location = { region = world_troll_country }
			}
			182 = { university_city_open_trigger = yes }
		}
		d_university_of_praag = { save_event_target_as = university }
	}
	else_if = {
		limit = {
			location = { region = world_old_world_tilea }
			OR = {
				600 = { university_city_open_trigger = yes }
				584 = { university_city_open_trigger = yes }
			}
		}
		random_list = {
			50 = {
				modifier = { factor = 0 NOT = { 600 = { university_city_open_trigger = yes } } }
				d_university_of_remas = { save_event_target_as = university }
			}
			50 = {
				modifier = { factor = 0 NOT = { 584 = { university_city_open_trigger = yes } } }
				d_university_of_miragliano = { save_event_target_as = university }
			}
		}
	}
	else_if = {
		limit = {
			location = { region = world_old_world_estalia }
			553 = { university_city_open_trigger = yes }
		}
		d_university_of_magritta = { save_event_target_as = university }
	}
	else_if = {
		limit = {
			location = { region = world_old_world_border_princes }
			OR = {
				600 = { university_city_open_trigger = yes }
				AND = { is_female = no 1037 = { university_city_open_trigger = yes } }
			}
		}
		random_list = {
			50 = {
				modifier = { factor = 0 NOT = { 600 = { university_city_open_trigger = yes } } }
				d_university_of_remas = { save_event_target_as = university }
			}
			50 = {
				modifier = { factor = 0 is_female = yes }
				modifier = { factor = 0 NOT = { 1037 = { university_city_open_trigger = yes } } }
				d_university_of_martek = { save_event_target_as = university }
			}
		}
	}
	else_if = {
		limit = {
			location = { region = world_greater_araby }
			is_female = no
			1037 = { university_city_open_trigger = yes }
		}
		d_university_of_martek = { save_event_target_as = university }
	}
	else_if = {
		limit = { 299 = { university_city_open_trigger = yes } }
		d_university_of_altdorf = { save_event_target_as = university }
	}
}

# The parent pays and the student goes. Graduation is in four years
# (university.10). Scope: the parent. event_target:university_student is
# the student and event_target:university the university.
university_enrol_effect = {
	prestige = 50
	scaled_wealth = -0.10
	event_target:university_student = {
		clr_character_flag = university_wants
		set_character_flag = university_student
		add_character_modifier = { name = university_studying years = 4 }
		hidden_effect = { character_event = { id = university.10 days = 1460 } }
	}
}
