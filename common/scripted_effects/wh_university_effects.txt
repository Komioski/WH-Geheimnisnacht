# Picks the university nearest the student whose city is open, and saves it
# as event_target:university. Altdorf takes anyone whose own region has no
# open university. If even Altdorf is closed, no target is saved.
# Araby's universities take no women.
# Scope: the student.
university_pick_effect = {
	if = {
		limit = {
			location = { region = world_old_world_empire_south_great_forest }
			324 = { university_city_open_trigger = yes }
		}
		d_university_of_talabheim = { save_event_target_as = university }
	}
	else_if = {
		limit = {
			location = { region = world_old_world_empire_south }
			309 = { university_city_open_trigger = yes }
		}
		d_university_of_nuln = { save_event_target_as = university }
	}
	else_if = {
		limit = {
			location = { region = world_old_world_empire_wasteland }
			222 = { university_city_open_trigger = yes }
		}
		d_university_of_marienburg = { save_event_target_as = university }
	}
	else_if = {
		limit = {
			location = { region = world_old_world_empire_north }
			239 = { university_city_open_trigger = yes }
		}
		d_university_of_salzenmund = { save_event_target_as = university }
	}
	else_if = {
		limit = {
			OR = {
				location = { region = world_old_world_empire_east }
				location = { region = world_old_world_empire_mootland }
			}
			355 = { university_city_open_trigger = yes }
		}
		d_university_of_streissen = { save_event_target_as = university }
	}
	else_if = {
		limit = {
			OR = {
				location = { region = world_old_world_kislev }
				location = { region = world_troll_country }
			}
			182 = { university_city_open_trigger = yes }
		}
		d_university_of_praag = { save_event_target_as = university }
	}
	else_if = {
		limit = {
			location = { region = world_old_world_tilea }
			OR = {
				600 = { university_city_open_trigger = yes }
				584 = { university_city_open_trigger = yes }
			}
		}
		random_list = {
			50 = {
				modifier = { factor = 0 NOT = { 600 = { university_city_open_trigger = yes } } }
				d_university_of_remas = { save_event_target_as = university }
			}
			50 = {
				modifier = { factor = 0 NOT = { 584 = { university_city_open_trigger = yes } } }
				d_university_of_miragliano = { save_event_target_as = university }
			}
		}
	}
	else_if = {
		limit = {
			location = { region = world_old_world_estalia }
			553 = { university_city_open_trigger = yes }
		}
		d_university_of_magritta = { save_event_target_as = university }
	}
	else_if = {
		limit = {
			location = { region = world_old_world_border_princes }
			OR = {
				600 = { university_city_open_trigger = yes }
				AND = { is_female = no 1037 = { university_city_open_trigger = yes } }
			}
		}
		random_list = {
			50 = {
				modifier = { factor = 0 NOT = { 600 = { university_city_open_trigger = yes } } }
				d_university_of_remas = { save_event_target_as = university }
			}
			50 = {
				modifier = { factor = 0 NOT = { 1037 = { university_city_open_trigger = yes } } }
				modifier = { factor = 0 NOT = { is_female = no } }
				d_university_of_martek = { save_event_target_as = university }
			}
		}
	}
	else_if = {
		limit = {
			location = { region = world_greater_araby_southlands }
			983 = { university_city_open_trigger = yes }
		}
		d_university_of_sudenburg = { save_event_target_as = university }
	}
	else_if = {
		limit = {
			location = { region = world_greater_araby }
			is_female = no
			OR = {
				1037 = { university_city_open_trigger = yes }
				1035 = { university_city_open_trigger = yes }
			}
		}
		random_list = {
			50 = {
				modifier = { factor = 0 NOT = { 1037 = { university_city_open_trigger = yes } } }
				d_university_of_martek = { save_event_target_as = university }
			}
			50 = {
				modifier = { factor = 0 NOT = { 1035 = { university_city_open_trigger = yes } } }
				d_university_of_kufra = { save_event_target_as = university }
			}
		}
	}
	else_if = {
		limit = { 299 = { university_city_open_trigger = yes } }
		d_university_of_altdorf = { save_event_target_as = university }
	}
}

# The parent pays and the student goes to live at the Rector's court in
# the university's city. A letter comes home halfway (university.20) and
# the student graduates four years later (university.10). If the
# university has no Rector the student studies from home. The parent's
# primary title is kept, so the news can go to his heir if he dies.
# Scope: the parent. event_target:university_student is the student and
# event_target:university the university.
university_enrol_effect = {
	prestige = 50
	scaled_wealth = -0.10
	primary_title = { save_event_target_as = university_home_title }
	event_target:university = { holder_scope = { save_event_target_as = university_rector } }
	event_target:university_student = {
		clr_character_flag = university_wants
		set_character_flag = university_student
		add_character_modifier = { name = university_studying years = 4 }
		if = {
			limit = { event_target:university_rector = { is_alive = yes } }
			custom_tooltip = { text = university_sent_away_tt }
			hidden_effect = { move_character = event_target:university_rector }
		}
		hidden_effect = {
			character_event = { id = university.20 days = 730 random = 60 }
			character_event = { id = university.10 days = 1460 }
		}
	}
}

# The student's education trait rises one step. Scope: the student.
university_improve_education_effect = {

	else_if = {
		limit = { trait = charismatic_negotiator }

	}

		limit = { trait = underhanded_rogue }

		limit = { trait = naive_appeaser }

	else_if = {
		limit = { trait = fortune_builder }

		add_trait = martial_cleric
	}
}

# The Rectors
# Each university is held by a Rector, a learned scholar of its city, and the
# students live at his court while they study. A university without a
# Rector gets one at the game start (university.30); when a Rector dies, a
# scholar of the same faith and people takes his chair (university.31).
###############################################################################

# A new Rector for one university, if it has none (Salzenmund,
# Marienburg and Sudenburg only once founded, found_university_*). The Rectors of Martek
# and Kufra are always men and sorcerers of the flame. Scope: any character (the new
# Rector is created at its court and leaves with the title).
university_rector_altdorf_effect = {
	if = {
		limit = { d_university_of_altdorf = { has_holder = no } }
		random_list = {
			80 = {
				create_character = {
					random_traits = yes
					dynasty = RANDOM
					religion = sigmarite
					culture = reiklander
					female = no
					age = 52
					attributes = { learning = 12 }
				}
			}
			20 = {
				create_character = {
					random_traits = yes
					dynasty = RANDOM
					religion = sigmarite
					culture = reiklander
					female = yes
					age = 49
					attributes = { learning = 12 }
				}
			}
		}
		new_character = {
			university_rector_traits_effect = yes
			d_university_of_altdorf = { grant_title = PREV }
		}
	}
}
