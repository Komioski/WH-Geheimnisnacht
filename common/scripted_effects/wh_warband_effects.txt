# The summoned champion joins the ruler's warband.
# Scope: the champion; ROOT is the ruler.
warband_join_ruler_society_effect = {
	if = {
		limit = { ROOT = { society_member_of = warband_of_chaos } }
		join_society = warband_of_chaos
	}
	else_if = {
		limit = { ROOT = { society_member_of = warband_of_khorne } }
		join_society = warband_of_khorne
	}
	else_if = {
		limit = { ROOT = { society_member_of = warband_of_nurgle } }
		join_society = warband_of_nurgle
	}
	else_if = {
		limit = { ROOT = { society_member_of = warband_of_tzeentch } }
		join_society = warband_of_tzeentch
	}
	else_if = {
		limit = { ROOT = { society_member_of = warband_of_slaanesh } }
		join_society = warband_of_slaanesh
	}
	else_if = {
		limit = { ROOT = { society_member_of = skaven_warlords_society } }
		join_society = skaven_warlords_society
	}
	else_if = {
		limit = { ROOT = { society_member_of = warband_gork } }
		join_society = warband_gork
	}
	else_if = {
		limit = { ROOT = { society_member_of = warband_mork } }
		join_society = warband_mork
	}
}

# The player comes off worst in the celebration contest (waraid.14): Skaven
# food is poisoned, dawi ale wins, everyone else gets hurt.
# Scope: the player.
waraid_contest_lost_effect = {
	if = {
		limit = {
			warband_voice_skaven_trigger = yes
			NOT = { trait = ill }
		}
		add_trait = ill
	}
	else_if = {
		limit = {
			warband_voice_dawi_trigger = yes
			NOT = { trait = drunkard }
		}
		add_trait = drunkard
	}
	else = {
		add_trait_wounded_effect = yes
	}
	prestige = -25
}
