# Opens a wyrdstone mine in the province's capital holding, of the kind
# that fits it (castle, city, temple, tribe or nomad).
# Scope: a province.
warpstone_open_mine_effect = {
	capital_holding = {
		add_building = ct_blackstone_mine_1
		add_building = ca_blackstone_mine_1
		add_building = tp_blackstone_mine_1
		add_building = tb_blackstone_mine_1
		add_building = no_blackstone_mine_1
	}
}

# Pays for a Mordheim expedition (see warpstone_expedition_can_pay_trigger).
# Scope: the patron.
warpstone_expedition_pay_effect = {
	if = {
		limit = { warpstone_expedition_pays_piety_trigger = yes }
		piety = -500
	}
	else_if = {
		limit = { warpstone_expedition_pays_prestige_trigger = yes }
		prestige = -750
	}
	else_if = {
		limit = { trait = creature_skaven }
		wealth = -150
	}
	else_if = {
		limit = { ai = yes }
		wealth = -200
	}
	else = {
		wealth = -300
	}
}

# The witch hunters, priests or runesmiths purge a vein or meteor: the stone
# is gone and no mine can be dug. Old World faithful also call a witch
# hunter of their faith from the realm, if one is free (whunter.2).
# Scope: the province. ROOT: the province.
warpstone_purge_effect = {
	remove_province_modifier = warpstone_vein
	remove_province_modifier = warpstone_meteor
	set_variable = { which = blackstone_mining_progress value = 0 }
	owner = {
		piety = 50
		if = {
			limit = { religion_group = old_world_group }
			save_event_target_as = whunter_client # the ruler whunter.3 goes to
			random_realm_character = {
				limit = {
					religion = PREV
					is_witch_hunter_trigger = yes
					NOT = { trait = on_witch_hunt }
				}
				character_event = { id = whunter.2 }
			}
		}
	}
}

# A wyrdstone deposit is found: marks the world as having one, so the yearly
# mine check (warpstone.25) only runs once there is anything to mine.
# Scope: the province.
warpstone_deposit_found_effect = {
	set_global_flag = warpstone_deposits
	set_variable = { which = blackstone_mining_progress value = 0 }
}

# The Mordheim agent is lost in the ruins: slain, unless they cannot die.
# The patron may fund another expedition.
# Scope: the agent. The caller clears the patron's maintaining_agents flag.
warpstone_agent_lost_effect = {
	clr_character_flag = intrigue_agent_mordheim
	clr_character_flag = mercenary_agent_mordheim
	remove_trait = travelling
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		death = { death_reason = death_murder_unknown }
	}
}

# The Mordheim agent comes home.
# Scope: the agent.
warpstone_agent_recall_effect = {
	clr_character_flag = intrigue_agent_mordheim
	clr_character_flag = mercenary_agent_mordheim
	clr_character_flag = mark_for_fail
	clr_character_flag = mark_for_fail2
	remove_trait = travelling
}

# A faithful of the Old World or of Ormazd who deals in the shadow market
# may be talked about (warpstone.33, 15 in 100, a month later).
# Scope: the dealer.
warpstone_market_whispers_effect = {
	if = {
		limit = {
			OR = {
				religion_group = old_world_group
				religion_group = ormazd_gods_group
			}
		}
		random = {
			chance = 15
			modifier = { factor = 0.5 intrigue = 16 }
			character_event = { id = warpstone.33 days = 30 }
		}
	}
}
