# Witch hunter scripted effects
# Used by events/WHG_witch_hunter_events.txt (namespace whunter) and
# events/WHG_witch_chapter_events.txt (namespace whchapter).

# Burn the scoped character at the stake, killed by ROOT. Invincible
# characters survive.
# Scope: the condemned.
whunter_burn_at_stake_effect = {
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		add_trait = staked
		death = { death_reason = death_execution_burning killer = ROOT }
	}
}

# A witch hunter comes back from a successful mission.
# Scope: the witch hunter.
whunter_mission_won_effect = {
	remove_trait = travelling
	add_mission_succeed_influence_effect = yes
	random = {
		chance = 20
		whunter_reward_artifact_effect = yes
	}
	random = {
		chance = 50
		if = {
			limit = { NOT = { learning = 20 } }
			change_learning = 1 #learning determines how good a witch hunter is
		}
	}
}

# A witch hunter comes back from a failed mission.
# Scope: the witch hunter.
whunter_mission_lost_effect = {
	remove_trait = travelling
	add_mission_fail_influence_effect = yes
	add_character_modifier = { name = quest_cooldown_timer hidden = yes days = 365 }
}

# The witch hunter is hurt on a mission.
# Scope: the witch hunter.
whunter_hurt_effect = {
	if = {
		limit = { NOT = { trait = wounded } }
		add_trait = wounded
	}
	else_if = {
		limit = { NOT = { trait = maimed } }
		add_trait = maimed
	}
}

# The liege upholds a witch hunter's accusation (burned or imprisoned). The
# hunter gains piety and may learn from the case; chapter members who were not
# on a mission also earn chapter currency.
# Scope: the witch hunter.
whunter_accusation_upheld_effect = {
	piety = 20
	if = {
		limit = {
			whunter_chapter_member_trigger = yes
			NOT = { has_character_flag = mission_hunt }
		}
		change_society_currency = 25
	}
	random = {
		chance = 25
		if = {
			limit = { NOT = { learning = 20 } }
			change_learning = 1
		}
	}
}

# The Order rewards a successful hunt with a piece of witch hunter gear the
# hunter does not have yet (common/artifacts/16_artifacts_mortal_realms.txt).
# Scope: the witch hunter.
whunter_reward_artifact_effect = {
	random_list = {
		40 = {
			trigger = { NOT = { has_artifact = twin_tailed_comet_pendant } }
			add_artifact = twin_tailed_comet_pendant
		}
		30 = {
			trigger = { NOT = { has_artifact = deus_sigmar_copy } }
			add_artifact = deus_sigmar_copy
		}
		30 = {
			trigger = { NOT = { has_artifact = witch_hunter_hat } }
			add_artifact = witch_hunter_hat
		}
	}
}

###############################################################################
# Witch hunter chapters and court titles
# Chapters: Order of Sigmar, Holy Inquisition of Myrmidia, Templars of Solkan.
# Court titles: title_court_witchhunter, title_court_inquisitor,
# title_court_templar (common/minor_titles/wh_minor_titles.txt).
###############################################################################

# Gives the scoped hunter the trait and culture of the chapter they serve:
# Inquisitors become inquisitor / inquisitor_culture, Templars of Solkan
# solkan_templar / templar_culture, Order hunters keep witch_hunter. Hunters
# of no chapter (Ulric's, Taal's) keep what they have.
# Scope: the witch hunter.
whunter_chapter_trait_effect = {
	if = {
		limit = { society_member_of = holy_inquisition_myrmidia }
		if = { limit = { trait = witch_hunter } remove_trait = witch_hunter }
		if = { limit = { trait = solkan_templar } remove_trait = solkan_templar }
		if = { limit = { NOT = { trait = inquisitor } } add_trait = inquisitor }
		if = {
			limit = { is_ruler = no NOT = { culture = inquisitor_culture } }
			culture = inquisitor_culture
		}
	}
	else_if = {
		limit = { society_member_of = templars_of_solkan }
		if = { limit = { trait = witch_hunter } remove_trait = witch_hunter }
		if = { limit = { trait = inquisitor } remove_trait = inquisitor }
		if = { limit = { NOT = { trait = solkan_templar } } add_trait = solkan_templar }
		if = {
			limit = { is_ruler = no NOT = { culture = templar_culture } }
			culture = templar_culture
		}
	}
	else_if = {
		limit = { society_member_of = sigmar_witch_hunters }
		if = { limit = { trait = inquisitor } remove_trait = inquisitor }
		if = { limit = { trait = solkan_templar } remove_trait = solkan_templar }
		if = { limit = { NOT = { trait = witch_hunter } } add_trait = witch_hunter }
	}
}

# The scoped witch hunter's liege gives them the court title that fits the
# liege's lands.
# Scope: the witch hunter (a courtier).
whunter_give_court_title_effect = {
	liege = {
		if = {
			limit = { whunter_inquisition_lands_trigger = yes }
			PREV = { give_minor_title = title_court_inquisitor }
		}
		else_if = {
			limit = { whunter_solkan_lands_trigger = yes }
			PREV = { give_minor_title = title_court_templar }
		}
		else = {
			PREV = { give_minor_title = title_court_witchhunter }
		}
	}
}

# The scoped witch hunter joins the chapter that fits them: Solkanites and
# hunters of the Border Princes the Templars, Estalians and Tileans the
# Inquisition, the rest the Order of Sigmar. Falls through to the next
# chapter that will take them.
# Scope: the witch hunter.
whunter_join_local_chapter_effect = {
	if = {
		limit = { is_in_society = no }
		if = {
			limit = {
				OR = {
					religion = solkanite
					liege = { whunter_solkan_lands_trigger = yes }
				}
				can_join_society = templars_of_solkan
			}
			join_society = templars_of_solkan
		}
		else_if = {
			limit = { can_join_society = holy_inquisition_myrmidia }
			join_society = holy_inquisition_myrmidia
		}
		else_if = {
			limit = { can_join_society = sigmar_witch_hunters }
			join_society = sigmar_witch_hunters
		}
		else_if = {
			limit = { can_join_society = templars_of_solkan }
			join_society = templars_of_solkan
		}
	}
	whunter_chapter_trait_effect = yes
}

# A new witch hunter arriving at ROOT's court takes the faith and blood of
# the chapter that sends them: Inquisitors are Myrmidians of the Inquisitor
# culture, Templars Solkanites of the Templar culture. Others keep the
# Sigmarite witch hunter culture they were made with.
# Scope: the new witch hunter; ROOT is the ruler.
whunter_local_chapter_origin_effect = {
	if = {
		limit = { ROOT = { whunter_inquisition_lands_trigger = yes } }
		culture = inquisitor_culture
		religion = myrmidian
	}
	else_if = {
		limit = { ROOT = { whunter_solkan_lands_trigger = yes } }
		culture = templar_culture
		religion = solkanite
	}
}

# A small chance at the chapter's relic: the Inquisitor's Sigil or Solkan's
# Branding Iron. Once per lifetime, never twice.
# Scope: the chapter member.
whunter_chapter_relic_effect = {
	if = {
		limit = { NOT = { has_character_flag = whunter_chapter_relic } }
		random = {
			chance = 5
			if = {
				limit = {
					society_member_of = holy_inquisition_myrmidia
					NOT = { has_artifact = inquisitors_sigil }
				}
				add_artifact = inquisitors_sigil
				set_character_flag = whunter_chapter_relic
			}
			else_if = {
				limit = {
					society_member_of = templars_of_solkan
					NOT = { has_artifact = solkan_branding_iron }
				}
				add_artifact = solkan_branding_iron
				set_character_flag = whunter_chapter_relic
			}
		}
	}
}
