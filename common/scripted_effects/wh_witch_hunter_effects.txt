# Burn the scoped character at the stake, killed by ROOT. Invincible
# characters survive.
# Scope: the condemned.
whunter_burn_at_stake_effect = {
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		add_trait = staked
		death = { death_reason = death_execution_burning killer = ROOT }
	}
}

# A witch hunter comes back from a successful mission.
# Scope: the witch hunter.
whunter_mission_won_effect = {
	remove_trait = travelling
	add_mission_succeed_influence_effect = yes
	random = {
		chance = 50
		if = {
			limit = { NOT = { learning = 20 } }
			change_learning = 1 #learning determines how good a witch hunter is
		}
	}
}

# A witch hunter comes back from a failed mission.
# Scope: the witch hunter.
whunter_mission_lost_effect = {
	remove_trait = travelling
	add_mission_fail_influence_effect = yes
	add_character_modifier = { name = quest_cooldown_timer hidden = yes days = 365 }
}

# The witch hunter is hurt on a mission.
# Scope: the witch hunter.
whunter_hurt_effect = {
	if = {
		limit = { NOT = { trait = wounded } }
		add_trait = wounded
	}
	else_if = {
		limit = { NOT = { trait = maimed } }
		add_trait = maimed
	}
}
