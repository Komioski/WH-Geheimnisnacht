# Burn the scoped character at the stake, killed by ROOT. Invincible
# characters survive.
# Scope: the condemned.
whunter_burn_at_stake_effect = {
	if = {
		limit = { is_invincible_trigger = yes }
		antideath_effect = yes
	}
	else = {
		add_trait = staked
		death = { death_reason = death_execution_burning killer = ROOT }
	}
}

# A witch hunter comes back from a successful mission.
# Scope: the witch hunter.
whunter_mission_won_effect = {
	remove_trait = travelling
	add_mission_succeed_influence_effect = yes
	random = {
		chance = 20
		whunter_reward_artifact_effect = yes
	}
	random = {
		chance = 50
		if = {
			limit = { NOT = { learning = 20 } }
			change_learning = 1 #learning determines how good a witch hunter is
		}
	}
}

# A witch hunter comes back from a failed mission.
# Scope: the witch hunter.
whunter_mission_lost_effect = {
	remove_trait = travelling
	add_mission_fail_influence_effect = yes
	add_character_modifier = { name = quest_cooldown_timer hidden = yes days = 365 }
}

# The witch hunter is hurt on a mission.
# Scope: the witch hunter.
whunter_hurt_effect = {
	if = {
		limit = { NOT = { trait = wounded } }
		add_trait = wounded
	}
	else_if = {
		limit = { NOT = { trait = maimed } }
		add_trait = maimed
	}
}

# The liege upholds a witch hunter's accusation (burned or imprisoned). The
# hunter gains piety and may learn from the case; Order members who were not
# on a mission also earn Order currency.
# Scope: the witch hunter.
whunter_accusation_upheld_effect = {
	piety = 20
	if = {
		limit = {
			society_member_of = sigmar_witch_hunters
			NOT = { has_character_flag = mission_hunt }
		}
		change_society_currency = 25
	}
	random = {
		chance = 25
		if = {
			limit = { NOT = { learning = 20 } }
			change_learning = 1
		}
	}
}

# The Order rewards a successful hunt with a piece of witch hunter gear the
# hunter does not have yet (common/artifacts/16_artifacts_mortal_realms.txt).
# Scope: the witch hunter.
whunter_reward_artifact_effect = {
	random_list = {
		40 = {
			trigger = { NOT = { has_artifact = twin_tailed_comet_pendant } }
			add_artifact = twin_tailed_comet_pendant
		}
		30 = {
			trigger = { NOT = { has_artifact = deus_sigmar_copy } }
			add_artifact = deus_sigmar_copy
		}
		30 = {
			trigger = { NOT = { has_artifact = witch_hunter_hat } }
			add_artifact = witch_hunter_hat
		}
	}
}
