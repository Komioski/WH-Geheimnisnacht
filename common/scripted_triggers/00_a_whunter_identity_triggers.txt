###############################################################################
# Witch hunter identity triggers
# Loaded first so every other scripted trigger can use them.
# The Order of Sigmar's hunters carry the witch_hunter trait and culture; the
# Holy Inquisition of Myrmidia and the Templars of Solkan have their own
# (inquisitor / inquisitor_culture, solkan_templar / templar_culture).
###############################################################################

# Any witch hunter: Order hunter, Inquisitor or Templar of Solkan.
# Use this instead of trait = witch_hunter unless only the Order is meant.
# Scope: a character.
is_witch_hunter_trigger = {
	OR = {
		trait = witch_hunter
		trait = inquisitor
		trait = solkan_templar
	}
}

# One of the witch hunter cultures.
# Scope: a character.
is_hunter_culture_trigger = {
	OR = {
		culture = witchhunter_culture
		culture = inquisitor_culture
		culture = templar_culture
	}
}
