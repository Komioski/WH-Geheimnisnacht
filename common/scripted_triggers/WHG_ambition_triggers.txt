###############################################################################
# WHG_ambition_triggers.txt - shared checks for the ambitions
#
# Used by common/objectives/WHG_ambitions.txt and events/WHG_ambition_events.txt.
# Keep the cheap ones cheap: the AI weighs every ambition's potential for every
# adult who might take it, so nothing here may loop over provinces or courts.
###############################################################################

# Can dream at all: not a prisoner, not incapable, not a spawn, a Khemri
# soldier or a daemon. Scope: the character.
WHG_ambition_able_trigger = {
	prisoner = no
	NOR = {
		trait = incapable
		trait = creature_chaosspawn
		trait = khemri_soldier
		culture_group = daemonic_group
	}
}

# A landed ruler a player could be (count or higher, or a patrician).
# Scope: the character.
WHG_ambition_ruler_trigger = {
	is_adult = yes
	is_playable = yes
	is_landed = yes
}

# Can have a child of their own body: not a people without women, not undead,
# not a daemon. Scope: the character.
WHG_ambition_can_have_children_trigger = {
	is_no_females_race = no
	NOR = {
		is_undead_trigger = yes
		is_daemon_trigger = yes
	}
}

# Has a sin the Old World faiths want atoned for. Scope: the character.
WHG_ambition_has_sin_trigger = {
	OR = {
		trait = lustful
		trait = gluttonous
		trait = greedy
		trait = slothful
		trait = wroth
		trait = envious
		trait = proud
	}
}

# A follower of the Dark Gods (any of the four or Undivided, Norscan gods
# included, Hashut not). Scope: the character.
WHG_ambition_chaos_follower_trigger = {
	religion_group = chaos_gods_group
	NOT = { religion = hashut }
}

# Holds a virtue Slaanesh wants broken. Scope: the character.
WHG_ambition_slaanesh_virtue_trigger = {
	OR = {
		trait = chaste
		trait = celibate
		trait = temperate
		trait = charitable
		trait = humble
		trait = kind
		trait = wroth
		trait = shy
		trait = ugly
		trait = disfigured
	}
}

# Holds a virtue Khorne despises. Scope: the character.
WHG_ambition_khorne_virtue_trigger = {
	OR = {
		trait = patient
		trait = kind
		trait = craven
	}
}

# Holds a virtue Nurgle despises. Scope: the character.
WHG_ambition_nurgle_virtue_trigger = {
	OR = {
		trait = chaste
		trait = temperate
		trait = diligent
	}
}

# Holds a virtue Tzeentch despises. Scope: the character.
WHG_ambition_tzeentch_virtue_trigger = {
	OR = {
		trait = honest
		trait = trusting
		trait = content
	}
}

# Has shed one of the virtues flagged when the ambition was taken (see
# WHG_ambition_mark_virtues_effect). Scope: the character.
WHG_ambition_virtue_shed_trigger = {
	OR = {
		AND = { has_character_flag = amb_had_chaste NOT = { trait = chaste } }
		AND = { has_character_flag = amb_had_celibate NOT = { trait = celibate } }
		AND = { has_character_flag = amb_had_temperate NOT = { trait = temperate } }
		AND = { has_character_flag = amb_had_charitable NOT = { trait = charitable } }
		AND = { has_character_flag = amb_had_humble NOT = { trait = humble } }
		AND = { has_character_flag = amb_had_kind NOT = { trait = kind } }
		AND = { has_character_flag = amb_had_wroth NOT = { trait = wroth } }
		AND = { has_character_flag = amb_had_shy NOT = { trait = shy } }
		AND = { has_character_flag = amb_had_ugly NOT = { trait = ugly } NOT = { trait = disfigured } }
		AND = { has_character_flag = amb_had_patient NOT = { trait = patient } }
		AND = { has_character_flag = amb_had_craven NOT = { trait = craven } }
		AND = { has_character_flag = amb_had_diligent NOT = { trait = diligent } }
		AND = { has_character_flag = amb_had_honest NOT = { trait = honest } }
		AND = { has_character_flag = amb_had_trusting NOT = { trait = trusting } }
		AND = { has_character_flag = amb_had_content NOT = { trait = content } }
	}
}

# Has lost one of the sins flagged when Atone for my sins was taken.
# Scope: the character.
WHG_ambition_sin_atoned_trigger = {
	OR = {
		AND = { has_character_flag = amb_had_lustful NOT = { trait = lustful } }
		AND = { has_character_flag = amb_had_gluttonous NOT = { trait = gluttonous } }
		AND = { has_character_flag = amb_had_greedy NOT = { trait = greedy } }
		AND = { has_character_flag = amb_had_slothful NOT = { trait = slothful } }
		AND = { has_character_flag = amb_had_wroth NOT = { trait = wroth } }
		AND = { has_character_flag = amb_had_envious NOT = { trait = envious } }
		AND = { has_character_flag = amb_had_proud NOT = { trait = proud } }
	}
}

# Capital keep at level 3 or higher, whatever the culture's keep is called.
# Scope: a holding title (the capital holding).
WHG_ambition_keep_3_trigger = {
	OR = {
		has_building = ca_keep_3
		has_building = ca_asur_keep_3
		has_building = ca_bretonnian_keep_3
		has_building = ca_druchiikeep_3
		has_building = ca_estalian_keep_3
		has_building = ca_gothic_keep_3
		has_building = ca_lizardman_keep_3
		has_building = ca_nehekharan_keep_3
		has_building = ca_oriental_keep_3
		has_building = ca_tilean_keep_3
		has_building = ca_greenskin_fort_3
		has_building = ct_chaos_dwarf_ziggurat_keep_3
		has_building = tp_chaos_dwarf_ziggurat_keep_3
	}
}

		has_building = tp_nehkwall_2
		has_building = ca_sartosanwall_2
		has_building = ct_sartosanwall_2

# A province of this demesne is guarded: a level 2 wall in its capital
# holding, or a fort holding. Scope: a province.
WHG_ambition_guarded_province_trigger = {
	OR = {
		any_province_holding = { holding_type = fort }
		capital_holding = { WHG_ambition_wall_2_trigger = yes }
	}
}
