# An Asur or Druchii lord above count who ordered armour and has none of
# his people's great armours yet: his smith offers the rare materials
# (armoury.8) instead of the common prices.
# Scope: ROOT is the employer. Reads the flags of event_target:smith.
armoury_elven_armour_trigger = {
	higher_tier_than = COUNT
	trait = creature_elf
	event_target:smith = { has_character_flag = armor_set }
	OR = {
		AND = {
			culture_group = asur_group
			NOR = {
				has_artifact = asur_artifact_dragon_armour
				has_artifact = asur_artifact_shadow_armour
				has_artifact = asur_artifact_ithilmar_armour
			}
		}
		AND = {
			culture_group = druchii_group
			NOR = {
				has_artifact = druchii_meteorite_armour
				has_artifact = druchii_sea_dragon_cloak
			}
		}
	}
}

# A runesmith who can answer a summons (armoury.40): a dwarf or Chaos dwarf
# runesmith or runelord who is free to work.
# Scope: the runesmith.
armoury_runesmith_trigger = {
	OR = {
		trait = creature_dwarf
		trait = creature_chaos_dwarf
	}
	OR = {
		trait = runelord
		trait = runesmith
	}
	is_inaccessible_or_incapable_trigger = no
	is_adult = yes
	NOT = { has_character_flag = preparing_crafting }
}

# The smith is making one of the great elven armours (armoury.8).
# Scope: the smith.
armoury_elven_armour_flag_trigger = {
	OR = {
		has_character_flag = asur_artifact_dragon_armour_flag
		has_character_flag = asur_artifact_shadow_armour_flag
		has_character_flag = asur_artifact_ithilmar_armour_flag
		has_character_flag = druchii_sea_dragon_cloak_flag
		has_character_flag = druchii_meteorite_armour_flag
	}
}

# A courtier the smith may take as a lover (armoury.70, armoury.74): an
# adult who is free, not the employer's spouse and drawn to the smith.
# Scope: the courtier. ROOT is the employer.
armoury_smith_lover_trigger = {
	is_inaccessible_or_incapable_trigger = no
	age = 16
	NOT = { character = event_target:smith }
	NOT = { any_spouse = { character = ROOT } }
	NOR = {
		trait = chaste
		trait = celibate
	}
	OR = {
		AND = {
			event_target:smith = { is_woman_trigger = yes }
			prefers_men_trigger = no
		}
		AND = {
			event_target:smith = { is_woman_trigger = no }
			prefers_men_trigger = yes
		}
	}
}
