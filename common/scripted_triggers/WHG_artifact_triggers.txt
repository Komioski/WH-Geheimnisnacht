# Triggers for events/WHG_artifact_events.txt (the relic hunt and the
# treasures taken in raids)

# Scope: a ruler. Has heard a rumor of a lost treasure and may send a
# councillor after it.
artifact_hunt_rumor_trigger = {
	OR = {
		has_character_modifier = relic_rumor_major
		has_character_modifier = relic_rumor_medium
		has_character_modifier = relic_rumor_minor
	}
}

# Scope: a ruler with a rumor. His hunter is close enough to the treasure to
# find it on the next leg of the search: 8 steps for a common rumor, 14 for a
# rare one, 20 for a great one.
artifact_hunt_done_trigger = {
	OR = {
		AND = {
			has_character_modifier = relic_rumor_minor
			check_variable = { which = relic_score value = 8 }
		}
		AND = {
			has_character_modifier = relic_rumor_medium
			check_variable = { which = relic_score value = 14 }
		}
		AND = {
			has_character_modifier = relic_rumor_major
			check_variable = { which = relic_score value = 20 }
		}
	}
}

# Scope: the hunter. Can no longer search for event_target:relic_ruler: dead,
# captive, out of the council, called off, or the rumor ran cold.
artifact_hunt_stopped_trigger = {
	OR = {
		is_alive = no
		NOT = { has_character_flag = relic_hunter }
		is_councillor = no
		trait = incapable
		prisoner = yes
		liege = { NOT = { character = event_target:relic_ruler } }
		event_target:relic_ruler = { artifact_hunt_rumor_trigger = no }
	}
}

# Scope: a councillor of ROOT. Can bring ROOT a rumor of treasure
# (artifact.1).
artifact_rumor_councillor_trigger = {
	is_councillor = yes
	ai = yes
	prisoner = no
	NOT = { trait = incapable }
	opinion = { who = ROOT value = 20 }
}

# Holds a relic that can ward off a daemon: a saint's relic, a temple relic,
# a holy treasure or a rune-worked heirloom (chaoscult.493, artifact.120).
# Scope: a character.
holds_warding_relic_trigger = {
	any_artifact = {
		OR = {
			has_artifact_flag = saint
			has_artifact_flag = saint_item
			has_artifact_flag = temple_relic
			has_artifact_flag = religious
			has_artifact_flag = rune
		}
		NOT = { has_artifact_flag = curse }
	}
}

# Holds a relic of Shallya that can bless water (artifact.121).
# Scope: a character.
holds_shallya_relic_trigger = {
	any_artifact = {
		OR = {
			has_artifact_flag = shallya
			AND = {
				ROOT = { religion = shallyan }
				OR = {
					has_artifact_flag = saint
					has_artifact_flag = saint_item
					has_artifact_flag = temple_relic
				}
			}
		}
	}
}
