# Triggers for events/WHG_assassins_events.txt (the Hashishin of Ormazd and
# Clan Eshin of the Skaven)

# Scope: a character. Holds the seat of his order as primary title: Jebel
# Ezna (d_hashishin) for the Hashishin, the clan's kingdom (k_eshin) for
# Clan Eshin.
assassins_order_holder_trigger = {
	OR = {
		AND = {
			society_member_of = the_assassins
			primary_title = { title = d_hashishin }
		}
		AND = {
			society_member_of = eshin_society
			primary_title = { title = k_eshin }
		}
	}
}

# Scope: a Grand Master. An AI lord of the order's seat, who pays loans from
# his own purse and gets the gold back (assassins.2, assassins_pay_debt_effect).
assassins_order_bank_trigger = {
	ai = yes
	assassins_order_holder_trigger = yes
}

# Scope: a debtor. Has the gold to pay back what he owes the order.
assassins_can_pay_debt_trigger = {
	OR = {
		AND = {
			is_variable_equal = { which = assassins_debt value = 50 }
			wealth = 50
		}
		AND = {
			is_variable_equal = { which = assassins_debt value = 250 }
			wealth = 250
		}
		AND = {
			is_variable_equal = { which = assassins_debt value = 300 }
			wealth = 300
		}
		AND = {
			is_variable_equal = { which = assassins_debt value = 500 }
			wealth = 500
		}
	}
}

# Scope: a character; ROOT is a member. Owes ROOT's order and refused to
# pay: the debt collectors may call (assassins.207).
assassins_debtor_of_root_trigger = {
	is_adult = yes
	has_character_flag = assassins_refused_to_pay_debt
	OR = {
		has_character_modifier = assassins_debt
		has_character_modifier = borrowed_from_assassins
	}
	OR = {
		AND = {
			has_character_flag = assassins_debt_eshin
			ROOT = { society_member_of = eshin_society }
		}
		AND = {
			NOT = { has_character_flag = assassins_debt_eshin }
			ROOT = { society_member_of = the_assassins }
		}
	}
}

# Scope: a member. Worships the god of his order: Ormazd for the Hashishin,
# the Horned Rat for Clan Eshin.
assassins_true_believer_trigger = {
	OR = {
		AND = {
			society_member_of = the_assassins
			religion = ormazd
		}
		AND = {
			society_member_of = eshin_society
			religion_group = horned_rat_god_group
		}
	}
}

# Scope: a ruler. A cruel or zealous ruler of another faith over the
# faithful of Ormazd: a target for the Hashishin.
assassins_persecutor_trigger = {
	is_member_of_any_murderous_society = no
	OR = {
		is_evil_trigger = yes
		trait = zealous
	}
	NOT = { religion = ormazd }
	any_realm_province = { religion = ormazd }
}

# Scope: a ruler. A rat warlord grown too greedy or too impious for the
# Council of Thirteen: a target for Clan Eshin.
assassins_upstart_trigger = {
	NOT = { society_member_of = eshin_society }
	religion_group = horned_rat_god_group
	OR = {
		has_unpleasant_trait_trigger = yes
		has_impious_trait_trigger = yes
	}
}

# Scope: a ruler. Serves the Dark Gods, or is suspected of it: a target for
# the Hashishin.
assassins_chaos_servant_trigger = {
	NOT = { society_member_of = the_assassins }
	OR = {
		society_member_of = cult_of_khorne
		society_member_of = cult_of_nurgle
		society_member_of = cult_of_tzeentch
		society_member_of = cult_of_slaanesh
		society_member_of = cult_of_undivided
		has_character_modifier = under_suspicion
		has_character_modifier = highly_suspect
	}
}

# Scope: a top liege; ROOT is a member. A realm of unbelievers next to ROOT
# with a temple or castle held by one: the order may seize it
# (assassins.221). A realm of a sister faith only counts when ROOT has no
# true unbelievers for neighbours.
assassins_unbeliever_realm_trigger = {
	OR = {
		NOT = { religion_group = ROOT }
		AND = {
			NOT = { religion = ROOT }
			ROOT = {
				NOT = {
					any_realm_province = {
						any_neighbor_province = {
							owner = { top_liege = { NOT = { religion_group = ROOT } } }
						}
					}
				}
			}
		}
	}
	any_realm_province = {
		any_province_holding = {
			owner = { NOT = { religion = ROOT } }
			OR = {
				holding_type = temple
				holding_type = castle
			}
		}
	}
}

# Scope: a courtier or vassal. Someone the order could ask a recruit to
# kill as proof of loyalty (assassins.36).
assassins_sacrifice_trigger = {
	is_adult = yes
	ai = yes
	is_member_of_any_murderous_society = no
}

# Scope: a protege. Does not yet follow the god of the order he trains for,
# and will be asked to convert (assassins.257 - 259).
assassins_protege_converts_trigger = {
	OR = {
		AND = {
			has_character_flag = assassins_protege_eshin
			NOT = { religion_group = horned_rat_god_group }
		}
		AND = {
			NOT = { has_character_flag = assassins_protege_eshin }
			NOT = { religion = ormazd }
		}
	}
}
