# WHG_bretonnia_triggers.txt: scripted triggers
# Subject: Bretonnia: knights, the Lady, the Grail, tourneys
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Shared: used by three or more files (4)
#   Used by WHG_bretonnia_effects (7)
#   Used by WHG_bretonnia_events (6)
#   Used by WHG_mousillon_events (4)
#   Used by WHG_tournament_events (5)
#   Other helpers used by one or two files (4)
###############################################################################

# A girl or woman with the gift who is not yet a Damsel, and has not
# refused or been sought already (bretonmagic.8).
# Scope: the candidate.
# Used by: WHG_bretonnia_effects, WHG_bretonnia_triggers +1 more
damsel_recruit_trigger = {
	is_woman_trigger = yes
	age = 6
	prisoner = no
	is_in_society = no
	NOT = { trait = BretonDamsel }
	NOT = { has_character_flag = refused_magical_recruitment }
	NOT = { has_character_flag = priority_mage_recruit }
	OR = { culture_group = bretonnian_group religion = grail }
	has_magic_power_trigger = no
	OR = {
		trait = magic_potential_4
		trait = magic_potential_3
		trait = magic_potential_2
		trait = magic_potential_1
		has_character_flag = magic_potential_flag_4
		has_character_flag = magic_potential_flag_3
		has_character_flag = magic_potential_flag_2
		has_character_flag = magic_potential_flag_1
	}
}

# A Bretonnian realm of the Old World faiths where the Damsels hunt and
# recruit.
# Scope: a ruler.
# Used by: WHG_bretonnia_effects, WHG_bretonnia_triggers
damsel_realm_trigger = {
	culture_group = bretonnian_group
	religion_group = old_world_group
}

# Scope: a character. ROOT is the host (tourney.1). Fit to ride in the lists:
# 16 to 59, free, at peace and whole, of the Lady's faith or a knight of
# Bretonnia, and not a priest, patrician or mayor. Women ride only where the
# host's faith or laws treat them as equals. Older riders need more skill.
# Used by: WHG_bretonnia_effects
tourney_can_ride_trigger = {
	age = 16
	NOT = { age = 60 }
	prisoner = no
	war = no
	is_incapable = no
	NOT = { has_job_title = job_spiritual }
	NOR = {
		is_theocracy = yes
		is_republic = yes
		trait = blinded
		trait = maimed
		trait = infirm
		trait = one_legged
		has_character_flag = tourney_rider
		has_character_flag = tourney_watcher
		has_character_flag = no_tournies_ever
	}
	OR = {
		religion = grail
		is_member_of_bretonnia_trigger = yes
	}
	OR = {
		is_woman_trigger = no
		ROOT = { gender_equality_trigger = yes }
	}
	OR = {
		NOT = { age = 40 }
		combat_rating = 35
	}
}

# Scope: a character. ROOT is the host. Good enough for a regional or grand
# tourney: has placed at a tourney before, is a knight of Bretonnia of some
# skill, or is a lord (or a lord's kin) who can hold a lance.
# Used by: WHG_bretonnia_effects
tourney_worthy_rider_trigger = {
	OR = {
		has_character_flag = grand_tourny_1st
		has_character_flag = grand_tourny_2nd
		has_character_flag = grand_tourny_3rd
		has_character_flag = normal_tourny_1st
		has_character_flag = normal_tourny_2nd
		has_character_flag = normal_tourny_3rd
		has_character_flag = small_tourny_1st
		has_character_flag = small_tourny_2nd
		AND = {
			is_member_of_bretonnia_trigger = yes
			combat_rating = 35
		}
		AND = {
			OR = {
				higher_tier_than = BARON
				liege = {
					higher_tier_than = BARON
					OR = {
						is_close_relative = PREV
						dynasty = PREV
					}
				}
			}
			combat_rating = 50
		}
	}
}

# Scope: a character. ROOT is the host. Good enough for a local tourney: a
# lord, a lord's kin or a knight, with some skill at arms.

tourney_local_rider_trigger = {

		liege = {
			OR = {
				is_close_relative = PREV
				dynasty = PREV
			}
		}
	}
}

# Scope: a lord. ROOT is the host. May be asked to watch: a count or higher
# who is free, at peace, not riding and has not asked to be left alone.
# Used by: WHG_bretonnia_effects
tourney_can_watch_trigger = {
	higher_tier_than = BARON
	NOT = { character = ROOT }
	prisoner = no
	war = no
	is_incapable = no
	NOR = {
		has_character_flag = tourney_rider
		has_character_flag = tourney_watcher
		has_character_flag = tourney_host
		has_character_flag = no_tournies_ever
	}
}

# There is a man working magic in the Damsel's Bretonnian liege's realm or
# in a neighbouring Bretonnian realm near her capital (bretonmagic.1, 3).
# kislev_male_mage_trigger says who counts.
# Scope: ROOT is the Damsel.
# Used by: WHG_bretonnia_events
damsel_male_mage_exists_trigger = {
	OR = {
		top_liege = {
			damsel_realm_trigger = yes
			any_realm_character = { kislev_male_mage_trigger = yes }
		}
		any_neighbor_independent_ruler = {
			damsel_realm_trigger = yes
			any_realm_character = {
				kislev_male_mage_trigger = yes
				ROOT = { capital_scope = { distance = { where = PREV distance < 1000 } } }
			}
		}
	}
}

# There is someone for the Damsel to recruit (bretonmagic.1, 8): a player,
# her own daughter, a woman of her realm, someone of her Bretonnian liege's
# realm or of a neighbouring Bretonnian realm.
# Scope: ROOT is the Damsel.
# Used by: WHG_bretonnia_events
damsel_recruit_exists_trigger = {
	OR = {
		any_player = { damsel_recruit_trigger = yes }
		any_child = { damsel_recruit_trigger = yes }
		any_realm_lord = { damsel_recruit_trigger = yes }
		top_liege = {
			damsel_realm_trigger = yes
			any_realm_character = { damsel_recruit_trigger = yes }
		}
		any_neighbor_independent_ruler = {
			damsel_realm_trigger = yes
			any_realm_character = { damsel_recruit_trigger = yes }
		}
	}
}

	OR = {
		has_landed_title = c_mousillon
		has_landed_title = c_dras
		has_landed_title = c_elne
		has_landed_title = c_rachard
	}
}

# Lives in Mousillon: holds a Mousillon county, or is at court there.
# Scope: a character.
# Used by: WHG_mousillon_events, WHG_mousillon_decisions
mousillon_resident_trigger = {
	OR = {
		mousillon_county_holder_trigger = yes
		location = { mousillon_province_trigger = yes }
	}
}

# Can sit through a Mousillon tale: grown, free and in their right mind.

mousillon_tale_ready_trigger = {
	is_adult = yes
	prisoner = no
	NOT = { trait = incapable }
	is_inaccessible_trigger = no
}

# Knows virtue when Landuin's ghost looks for it: three or more virtues and
# no open cruelty.
# Scope: a character.
# Used by: WHG_mousillon_events
mousillon_landuin_worthy_trigger = {
	NOR = { trait = cruel trait = impaler trait = deceitful trait = cannibal_trait }
	calc_true_if = {
		amount = 3
		trait = brave
		trait = just
		trait = honest
		trait = chaste
		trait = humble
		trait = temperate
		trait = patient
		trait = charitable
		trait = diligent
		trait = kind
		trait = grail_knight
	}
}

# Scope: a guest. ROOT is the host. Rides at ROOT's tourney.
# Used by: WHG_tournament_events, WHG_bretonnia_triggers
tourney_rider_of_root_trigger = {
	has_character_flag = tourney_rider
	has_opinion_modifier = { who = ROOT modifier = opinion_attending_tournament }
}

# Scope: a guest. ROOT is the host. Watches ROOT's tourney.
# Used by: WHG_tournament_events
tourney_watcher_of_root_trigger = {
	has_character_flag = tourney_watcher
	has_opinion_modifier = { who = ROOT modifier = opinion_spectating_tournament }
}

# Scope: a guest. ROOT is the host. Still in the lists: rides at ROOT's
# tourney, has not been unhorsed and can still sit a horse.
# Used by: WHG_tournament_events
tourney_in_lists_trigger = {
	tourney_rider_of_root_trigger = yes
	NOT = { has_character_flag = tourney_out }
	prisoner = no
	is_incapable = no
}

# Scope: a guest. event_target:tourney_host is the host. Still in the lists
# (for events whose ROOT is a rider).
# Used by: WHG_tournament_events
tourney_in_lists_of_host_trigger = {
	has_character_flag = tourney_rider
	has_opinion_modifier = { who = event_target:tourney_host modifier = opinion_attending_tournament }
	NOT = { has_character_flag = tourney_out }
	prisoner = no
	is_incapable = no
}

# Scope: a rider. Will take a purse to lose a tilt: greedy or deceitful,
# neither honest nor just, and below the rank of king.
# Used by: WHG_tournament_events
tourney_bribable_trigger = {
	lower_tier_than = KING
	NOR = {
		trait = honest
		trait = just
		trait = zealous
	}
	OR = {
		trait = greedy
		trait = deceitful
	}
}
