# A courtier the envoy would take as a eunuch for the Celestial Court: a man
# of some worth (kin, gifted or of fine body or mind), unwed, not the heir.
# Scope: a courtier; ROOT is the lord.
cathayart_eunuch_candidate_trigger = {
	NOR = {
		has_character_flag = originated_from_cathayan_court
		has_character_flag = eunuch_gift
		is_landed = yes
		is_woman_trigger = yes
		is_betrothed = yes
		is_married = yes
		trait = incapable
		trait = inbred
		trait = imbecile
	}
	OR = {
		is_close_relative = ROOT
		diplomacy = 20
		martial = 20
		stewardship = 20
		intrigue = 20
		learning = 20
		trait = strong
		trait = genius
		trait = quick
		has_fair_trait_trigger = yes
		trait = robust
		trait = shrewd
		trait = architect
		trait = strategist
		trait = scholar
		trait = theologian
	}
	OR = {
		is_adult = yes
		AND = { is_close_relative = ROOT age = 12 }
	}
	is_heir = no
	is_ill = no
	is_inaccessible_trigger = no
}

# A close kin of the lord the envoy would take as a concubine for the
# Emperor or Empress: of the other sex, unwed, under 45 and of a race with
# women.
# Scope: a courtier; ROOT is the lord.
cathayart_concubine_candidate_trigger = {
	is_close_relative = ROOT
	is_adult = yes
	NOT = { age = 45 }
	is_no_females_race = no
	OR = {
		AND = {
			offmap_cathay = { offmap_ruler = { is_woman_trigger = yes } }
			is_female = no
		}
		AND = {
			offmap_cathay = { offmap_ruler = { is_female = no } }
			is_woman_trigger = yes
		}
	}
	NOR = {
		has_character_flag = originated_from_cathayan_court
		is_landed = yes
		is_betrothed = yes
		is_married = yes
		trait = incapable
		trait = inbred
		trait = imbecile
		trait = celibate
		trait = wounded
	}
	is_heir = no
	is_ill = no
	is_inaccessible_trigger = no
}
