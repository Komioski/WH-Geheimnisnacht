# A councillor of the Jade Viceroy who may visit a foreign court: a Cathayan
# of the eastern peoples or of the faith of the Celestial Dragon, and not the
# host.
# Scope: the councillor; ROOT is the host.
celestial_councillor_trigger = {
	is_alive = yes
	prisoner = no
	OR = {
		culture_group = oriental_group
		religion = draconite
	}
	NOT = { character = ROOT }
}

# A host who may receive a councillor's parting treasure (JD.60004): once
# per life, and only with no Cathayan treasure held.
# Scope: the host.
celestial_gift_allowed_trigger = {
	NOT = { has_character_flag = celestial_gift_received }
	NOT = { any_artifact = { has_artifact_flag = cathayan } }
}

# A province with a city whose merchants have not been curbed (celestial.30,
# celestial.32).
# Scope: a province.
celestial_reprimand_province_trigger = {
	NOT = { has_province_modifier = reprimanded_merchants }
	any_province_holding = { holding_type = CITY }
}

# A vassal above baron rank with a city whose merchants can be curbed
# (celestial.32).
# Scope: the vassal.
celestial_reprimand_vassal_trigger = {
	NOT = { tier = BARON }
	any_realm_province = { celestial_reprimand_province_trigger = yes }
}

# A courtier from the Celestial Court who came to this court and has not yet
# given a verdict on the stay (celestial.40).
# Scope: the courtier; ROOT is the host.
celestial_verdict_courtier_trigger = {
	block_general_event_trigger = no
	prisoner = no
	has_character_flag = originated_from_cathayan_court
	has_character_flag = cathayan_courtier_original_court_@ROOT
	NOR = {
		has_character_flag = is_escaped_cathayan_noble
		has_character_flag = is_cathayan_refugee
		has_character_flag = had_opinions_about_host_court
		has_character_flag = ignore_refund
	}
}

# A courtier who could travel to the Jade Viceroy's court (celestial.50,
# celestial.52): an adult AI courtier of the west, unlanded, free, not a
# commander, not already picked and not from the Celestial Court.
# Scope: the courtier.
celestial_traveller_trigger = {
	ai = yes
	age = 14
	prisoner = no
	is_landed = no
	block_general_event_trigger = no
	NOT = { culture_group = oriental_group }
	NOT = { has_minor_title = title_commander }
	NOR = {
		has_character_flag = wants_to_travel_0
		has_character_flag = wants_to_travel_1
		has_character_flag = wants_to_travel_2
		has_character_flag = originated_from_cathayan_court
	}
}

# A province on one of the roads that reach Grand Cathay: the Ivory Road,
# the Old Silk Road, the Khyprian Road or the Spice Route (celestial.70 -
# celestial.73).
# Scope: a province.
celestial_eastern_road_province_trigger = {
	OR = {
		trade_route = ivory_road
		trade_route = old_silk_road
		trade_route = khyprian_road
		trade_route = spice_route
	}
}

# A playable lord who may be called to the Emperor's trade contest
# (celestial.70): at peace with Cathay, settled, not the Viceroy, not raiding
# Cathay, not already in it, with land on the roads east.
# Scope: the lord.
celestial_trade_contestant_trigger = {
	cathay_diplo_interaction_trigger = yes
	is_nomadic = no
	is_offmap_governor = no
	NOR = {
		has_character_modifier = cathay_raid_active
		has_character_modifier = cathay_raid_grace_cd
		has_character_flag = trade_competition_participant
	}
	any_demesne_province = { celestial_eastern_road_province_trigger = yes }
}

# The contestant who has won the most renown so far, or one of those tied
# for it (celestial.78, celestial.80).
# Scope: a contestant.
celestial_contest_leader_trigger = {
	has_character_flag = trade_competition_accumulate_prestige
	NOT = {
		any_playable_ruler = {
			has_character_flag = trade_competition_accumulate_prestige
			NOT = { character = PREV }
			NOT = {
				PREV = {
					check_variable = {
						which = celestial_contest_gain
						which = PREV
					}
				}
			}
		}
	}
}

# A child a travelling Cathayan scholar may teach (celestial.100): an AI
# child of six or more at the lord's court, not yet taught, able to learn.
# Scope: the child; ROOT is the lord.
celestial_teacher_pupil_trigger = {
	ai = yes
	age = 6
	is_adult = no
	prisoner = no
	host = { character = ROOT }
	NOR = {
		has_character_flag = taught_by_traveling_teacher
		trait = incapable
		trait = inbred
		is_inaccessible_trigger = yes
	}
}

# A province on the roads east with a burgher's city where a Cathayan caravan
# house could open an office (celestial.105), and none there yet.
# Scope: the province.
celestial_caravan_office_province_trigger = {
	celestial_eastern_road_province_trigger = yes
	NOR = {
		has_province_modifier = radhanite_favor
		has_province_modifier = radhanite_favor_big
	}
	any_province_holding = {
		holding_type = city
		holder_scope = { tier = BARON }
	}
}

# An AI count or duke on the roads east who may invite a lord on the way to
# kowtow to stay (kowtow.20): playable, free to host, not the traveller.
# Scope: the host; ROOT is the traveller.
kowtow_road_host_trigger = {
	ai = yes
	is_playable = yes
	OR = {
		tier = COUNT
		tier = DUKE
	}
	NOT = { character = ROOT }
	capital_scope = { celestial_eastern_road_province_trigger = yes }
	block_general_event_trigger = no
}

# Someone a traveller on the long road east may miss enough to write to
# (kowtow.40): a lover, spouse, parent, child or friend at home.
# Scope: the one missed; ROOT is the traveller.
kowtow_missed_one_trigger = {
	is_alive = yes
	OR = {
		is_lover = ROOT
		any_spouse = { character = ROOT }
		is_mother = ROOT
		is_father = ROOT
		is_child_of = ROOT
		is_friend = ROOT
	}
}

# A dead parent or sibling a traveller may remember in the rain (kowtow.10),
# who did not die in Cathay.
# Scope: the dead relative.
kowtow_mourned_trigger = {
	is_alive = no
	death_in_cathay_trigger = no
}
