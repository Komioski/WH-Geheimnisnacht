# A councillor of the Jade Viceroy who may visit a foreign court: a Cathayan
# of the eastern peoples or of the faith of the Celestial Dragon, and not the
# host.
# Scope: the councillor; ROOT is the host.
celestial_councillor_trigger = {
	is_alive = yes
	prisoner = no
	OR = {
		culture_group = oriental_group
		religion = draconite
	}
	NOT = { character = ROOT }
}

# A host who may receive a councillor's parting treasure (JD.60004): once
# per life, and only with no Cathayan treasure held.
# Scope: the host.
celestial_gift_allowed_trigger = {
	NOT = { has_character_flag = celestial_gift_received }
	NOT = { any_artifact = { has_artifact_flag = cathayan } }
}

# A province with a city whose merchants have not been curbed (celestial.30,
# celestial.32).
# Scope: a province.
celestial_reprimand_province_trigger = {
	NOT = { has_province_modifier = reprimanded_merchants }
	any_province_holding = { holding_type = CITY }
}

# A vassal above baron rank with a city whose merchants can be curbed
# (celestial.32).
# Scope: the vassal.
celestial_reprimand_vassal_trigger = {
	NOT = { tier = BARON }
	any_realm_province = { celestial_reprimand_province_trigger = yes }
}

# A courtier from the Celestial Court who came to this court and has not yet
# given a verdict on the stay (celestial.40).
# Scope: the courtier; ROOT is the host.
celestial_verdict_courtier_trigger = {
	block_general_event_trigger = no
	prisoner = no
	has_character_flag = originated_from_cathayan_court
	has_character_flag = cathayan_courtier_original_court_@ROOT
	NOR = {
		has_character_flag = is_escaped_cathayan_noble
		has_character_flag = is_cathayan_refugee
		has_character_flag = had_opinions_about_host_court
		has_character_flag = ignore_refund
	}
}

# Guest and host share a virtue or vice, looks, wit, strength, health,
# learning or a pastime (celestial.40).
