# A commander of ROOT's realm who was not trained in Cathayan ways and can
# make room for a Cathayan.
# Scope: a character; ROOT is the lord.
cathaygrace_replaceable_commander_trigger = {
	has_minor_title = title_commander
	liege = { character = ROOT }
	NOR = {
		has_character_flag = originated_from_cathayan_court
		has_character_flag = taught_cathayan_strategy
	}
}

# Holds an end of the roads east and a quarter of the wealth of one of the
# roads: enough to force Cathay's gates open.
# Scope: a character.
cathaygrace_masters_road_east_trigger = {
	any_realm_province = {
		OR = {
			province_id = 1188 # Nuur
			province_id = 2359 # Chimarak
			province_id = 2369 # Vargenspeak
			province_id = 2373 # Skrapland
		}
	}
	OR = {
		trade_route_control = {
			trade_route = ivory_road
			value >= 0.25
			type = wealth
			indirectly_owned = yes
		}
		trade_route_control = {
			trade_route = old_silk_road
			value >= 0.25
			type = wealth
			indirectly_owned = yes
		}
		trade_route_control = {
			trade_route = khyprian_road
			value >= 0.25
			type = wealth
			indirectly_owned = yes
		}
		trade_route_control = {
			trade_route = spice_route
			value >= 0.25
			type = wealth
			indirectly_owned = yes
		}
	}
}
