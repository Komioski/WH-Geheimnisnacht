# A commander of ROOT's realm who was not trained in Cathayan ways and can
# make room for a Cathayan.
# Scope: a character; ROOT is the lord.
cathaygrace_replaceable_commander_trigger = {
	has_minor_title = title_commander
	liege = { character = ROOT }
	NOR = {
		has_character_flag = originated_from_cathayan_court
		has_character_flag = taught_cathayan_strategy
	}
}

# Holds an end of the roads east and a quarter of the wealth of one of the
# roads: enough to force Cathay's gates open.
# Scope: a character.
cathaygrace_masters_road_east_trigger = {
	any_realm_province = {
		OR = {
			province_id = 1188 # Nuur
			province_id = 2359 # Chimarak
			province_id = 2369 # Vargenspeak
			province_id = 2373 # Skrapland
		}
	}
	OR = {
		trade_route_control = {
			trade_route = ivory_road
			value >= 0.25
			type = wealth
			indirectly_owned = yes
		}
		trade_route_control = {
			trade_route = old_silk_road
			value >= 0.25
			type = wealth
			indirectly_owned = yes
		}
		trade_route_control = {
			trade_route = khyprian_road
			value >= 0.25
			type = wealth
			indirectly_owned = yes
		}
		trade_route_control = {
			trade_route = spice_route
			value >= 0.25
			type = wealth
			indirectly_owned = yes
		}
	}
}

# A realm the Jade Viceroy would break at a lord's request: an independent,
# settled king or emperor of 40 counties or more, near Cathay's lands or its
# tributaries, not at peace with Cathay and not of the Dragon House.
# Scope: a character (the target); FROM is the lord who asks.
cathaygrace_invasion_target_trigger = {
	independent = yes
	higher_tier_than = DUKE
	num_of_count_titles_in_realm = 40
	is_nomadic = no
	in_revolt = no
	NOT = { character = FROM }
	NOT = { has_landed_title = e_cathay_west_governor }
	NOT = { has_character_modifier = peace_deal_with_cathay }
	NOT = { has_character_flag = defeated_cathay }
	NOT = { suzerain = { has_landed_title = e_cathay_west_governor } }
	NOT = { offmap_cathay = { offmap_ruler = { dynasty = PREVPREV } } }
	NOT = { primary_title = { has_title_flag = rebel_general_title } }
	OR = {
		valid_offmap_cathay_target_location = yes
		valid_offmap_cathay_target_location_tributary_chain = yes
	}
	custom_tooltip = {
		text = cathayan_request_invasion_rel_head_tt
		controls_religion = no
	}
}

# A lord near enough to Cathay to pledge tribute: next to the Viceroy or one
# of his tributaries, or with land in Cathay's sphere.
# Scope: a character.
cathaygrace_can_pledge_tribute_trigger = {
	OR = {
		custom_tooltip = {
			text = cathayan_grace_become_tributary_allow_governor_neighbor_tt
			any_neighbor_independent_ruler = { has_landed_title = e_cathay_west_governor }
		}
		custom_tooltip = {
			text = cathayan_grace_become_tributary_allow_tributary_neighbor_tt
			any_neighbor_independent_ruler = {
				suzerain = { has_landed_title = e_cathay_west_governor }
			}
		}
		custom_tooltip = {
			text = cathayan_grace_become_tributary_allow_cathayan_influence_tt
			any_realm_province = { region = custom_cathayan_invasion_target_region }
		}
	}
}
