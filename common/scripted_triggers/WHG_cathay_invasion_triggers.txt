###############################################################################
# Triggers for Cathay's wars in the west (events/WHG_cathay_invasion_events.txt)
###############################################################################

# The Emperor likes this ruler and does not also dislike them.
# Scope: a character.
cathayinv_liked_trigger = {
	liked_by_offmap = { type = offmap_cathay }
	NOT = { disliked_by_offmap = { type = offmap_cathay } }
}

# The Emperor dislikes this ruler and does not also like them.
# Scope: a character.
cathayinv_disliked_trigger = {
	disliked_by_offmap = { type = offmap_cathay }
	NOT = { liked_by_offmap = { type = offmap_cathay } }
}

# A raider of Cathay who does not sell the Viceroy horses: the first lord
# the Viceroy wants to bring to heel.
# Scope: a character.
cathayinv_raider_trigger = {
	has_character_modifier = cathay_raid_active
	NOT = { has_character_modifier = cathay_supply_horses }
}

# A ruler within reach of the Viceroy's armies, directly or through
# Cathay's tributaries.
# Scope: a character.
cathayinv_in_reach_trigger = {
	valid_offmap_cathay_target = yes
	OR = {
		valid_offmap_cathay_target_location = yes
		valid_offmap_cathay_target_location_tributary_chain = yes
	}
}

# A small realm, of fewer than 20 counties.
# Scope: a character.
cathayinv_small_realm_trigger = {
	num_of_count_titles_in_realm > 0
	NOT = { num_of_count_titles_in_realm = 20 }
}

# A ruler near the Viceroy's lands who might be asked for tribute.
# Scope: a character.
cathayinv_tributary_target_trigger = {
	valid_offmap_cathay_target = yes
	valid_offmap_cathay_target_location = yes
}

# A settled realm of middling size (15 to 29 counties) near the Viceroy,
# or (10 to 39) at the eastern edge of the map when he holds no land.
# Scope: a character; the Viceroy is ROOT.
cathayinv_medium_conquest_target_trigger = {
	valid_offmap_cathay_target = yes
	is_nomadic = no
	OR = {
		AND = {
			ROOT = { has_offmap_tmp_flag = WP_is_on_map }
			valid_offmap_cathay_target_location = yes
			num_of_count_titles_in_realm = 15
			NOT = { num_of_count_titles_in_realm = 30 }
		}
		AND = {
			ROOT = { has_offmap_tmp_flag = WP_is_not_on_map }
			capital_scope = { region = custom_eastern_edge_of_map }
			num_of_count_titles_in_realm = 10
			NOT = { num_of_count_titles_in_realm = 40 }
		}
	}
}

# A settled realm of 20 counties or more near the Viceroy, or at the
# eastern edge of the map when he holds no land.
# Scope: a character; the Viceroy is ROOT.
cathayinv_big_conquest_target_trigger = {
	valid_offmap_cathay_target = yes
	is_nomadic = no
	num_of_count_titles_in_realm = 20
	OR = {
		AND = {
			ROOT = { has_offmap_tmp_flag = WP_is_on_map }
			valid_offmap_cathay_target_location = yes
		}
		AND = {
			ROOT = { has_offmap_tmp_flag = WP_is_not_on_map }
			capital_scope = { region = custom_eastern_edge_of_map }
		}
	}
}

# Some lord of the Dzhagna steppe Cathay could take the frontier from.
# Scope: the Viceroy.
cathayinv_dzhagna_open_trigger = {
	NOT = { completely_controls = d_dzhagna }
	d_dzhagna = {
		any_direct_de_jure_vassal_title = {
			holder_scope = { top_liege = { valid_offmap_cathay_target = yes } }
		}
	}
}

# A ruler a rebel general of Cathay could carve an empire from.
# Scope: a character.
cathayinv_rebel_general_target_trigger = {
	valid_offmap_cathay_target_location = yes
	in_revolt = no
	is_within_diplo_range = event_target:the_protector_general
	is_landed = yes
	is_nomadic = no
	higher_tier_than = BARON
}
