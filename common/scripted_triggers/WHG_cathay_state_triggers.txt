###############################################################################
# Triggers for the state of Cathay (events/WHG_cathay_state_events.txt)
###############################################################################

# Cathay is not fighting a war at home: no invasion and no civil war, so a
# yearly status change may happen (each war has its own resolution).
# Scope: offmap_cathay.
cathaystate_at_peace_trigger = {
	NOR = {
		has_status = cathay_mongol_invasion # cathaystate.42
		has_status = cathay_jurchen_invasion # cathaystate.52
		has_status = cathay_civil_war # cathaystate.62
		has_status = cathay_chaos_invasion # cathaywar.2
		has_status = cathay_ogre_war # cathaywar.12
		has_status = cathay_skaven_war # cathaywar.22
	}
}

# A famine younger than four years or a plague younger than two: neither
# ends, nor gives way to anything else, before then.
# Scope: offmap_cathay.
cathaystate_hardship_too_young_trigger = {
	OR = {
		AND = {
			has_status = cathay_famine
			has_offmap_flag = cathay_had_famine
			NOT = { had_offmap_tmp_flag = { flag = cathay_had_famine years = 4 } }
		}
		AND = {
			has_status = cathay_plague
			has_offmap_flag = cathay_had_plague
			NOT = { had_offmap_tmp_flag = { flag = cathay_had_plague years = 2 } }
		}
	}
}

# A status that must run its minimum course before it can change: unrest
# two years, famine four, plague two, a golden age fifty.
# Scope: offmap_cathay.
cathaystate_status_too_young_trigger = {
	OR = {
		cathaystate_hardship_too_young_trigger = yes
		AND = {
			has_status = cathay_unrest
			has_offmap_flag = cathay_had_unrest
			NOT = { had_offmap_tmp_flag = { flag = cathay_had_unrest years = 2 } }
		}
		AND = {
			has_status = cathay_golden_age
			has_offmap_flag = cathay_had_golden_age
			NOT = { had_offmap_tmp_flag = { flag = cathay_had_golden_age years = 50 } }
		}
	}
}

# Cathay is weakened by famine, plague or unrest of some years' standing.
# Scope: offmap_cathay.
cathaystate_weakened_trigger = {
	OR = {
		AND = {
			has_status = cathay_unrest
			had_offmap_tmp_flag = { flag = cathay_had_unrest years = 2 }
		}
		AND = {
			has_status = cathay_famine
			had_offmap_tmp_flag = { flag = cathay_had_famine years = 6 }
		}
		AND = {
			has_status = cathay_plague
			had_offmap_tmp_flag = { flag = cathay_had_plague years = 3 }
		}
	}
}

# The previous status was one of hardship.
# Scope: offmap_cathay.
cathaystate_prev_hardship_trigger = {
	OR = { prev_status = cathay_famine prev_status = cathay_unrest prev_status = cathay_plague }
}

# The Dragon Throne is held by a foreign house: neither Cathayan nor any
# other people of the east.
# Scope: offmap_cathay.
cathaystate_foreign_emperor_trigger = {
	offmap_ruler = { NOT = { culture_group = oriental_group } }
}

# Cathay may be invaded again: it has not just invaded a player's realm.
# Scope: offmap_cathay.
cathaystate_can_be_invaded_trigger = {
	cathaystate_at_peace_trigger = yes
	OR = {
		NOT = { has_offmap_flag = cathay_invaded_player_dynasty }
		had_offmap_tmp_flag = { flag = cathay_invaded_player_dynasty years = 200 }
	}
}

# A hobgoblin warlord on the map who could lead the hordes into Cathay: an
# independent AI hobgoblin khan of at least ducal rank.
# Scope: a character.
cathaystate_hobgoblin_khan_trigger = {
	ai = yes
	culture = hobgoblin
	is_landed = yes
	higher_real_tier_than = COUNT
}

# The policy did not change with the new ruler.
# Scope: offmap_cathay.
cathaystate_same_policy_trigger = {
	OR = {
		AND = { prev_policy = cathay_open has_policy = cathay_open }
		AND = { prev_policy = cathay_isolationist has_policy = cathay_isolationist }
		AND = { prev_policy = cathay_expansionist has_policy = cathay_expansionist }
	}
}
