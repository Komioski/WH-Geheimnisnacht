# Triggers for events/WHG_chaos_gods_events.txt (the Dark Gods test their own)

# Scope: a character. Hurt badly enough to be a burden to himself: crippled,
# bedridden or broken in body.
chaosgod_crippled_trigger = {
	OR = {
		trait = maimed
		trait = incapable
		trait = one_legged
		trait = one_handed
		trait = blinded
		trait = mangled
		trait = severely_injured
	}
}

# Scope: the foe of a Khorne trial. Beaten in the duel: hurt or down, so
# the challenger can take the skull.
khorne_trial_foe_beaten_trigger = {
	OR = {
		trait = wounded
		trait = severely_injured
		trait = incapable
		is_maimed_trigger = yes
		trait = blinded
	}
}

# Scope: a spouse, lover, friend or courtier of ROOT. A daemonette of
# Slaanesh who can give counsel.
darkprince_daemonette_trigger = {
	trait = slaanesh_daemonette
	is_alive = yes
	prisoner = no
}

# Scope: a character. Dull of wit: no gift of the mind and little learning.
tzeentch_dull_trigger = {
	NOR = {
		trait = genius
		trait = brilliant
		trait = quick
		trait = smart
		trait = bright
	}
	NOT = { learning = 8 }
}

# Scope: a character. A follower of Slaanesh, openly or in secret, who may
# hunt for a daemonette (darkprince.10): has never sought one and keeps none
# close, is not a daemon himself and can pay his cult.
darkprince_daemonette_hunt_trigger = {
	OR = {
		religion = slaanesh
		secret_religion = slaanesh
	}
	is_adult = yes
	prisoner = no
	is_incapable = no
	is_demon_trigger = no
	NOT = { has_character_flag = slaanesh_daemonette_sought }
	wealth = 50
	NOR = {
		any_spouse = { trait = slaanesh_daemonette }
		any_lover = { trait = slaanesh_daemonette }
		any_courtier = { trait = slaanesh_daemonette }
	}
}

# A mortal the Blood God may tempt into the Crimson Skull (khorne.100): a
# human lord of an Old World, Araby or Kitab faith with a warlike streak, in
# no cult, not undead, not a witch hunter, not a Grail knight.
khorne_temptable_trigger = {
	trait = creature_human
	higher_tier_than = BARON
	is_adult = yes
	prisoner = no
	is_incapacitated_trigger = no
	is_undead_trigger = no
	OR = {
		religion_group = old_world_group
		religion = ormazd
		religion = kitabid
	}
	NOR = {
		religion = grail
		is_witch_hunter_trigger = yes
		trait = honest
		society_member_of = cult_of_khorne
		society_member_of = cult_of_nurgle
		society_member_of = cult_of_slaanesh
		society_member_of = cult_of_tzeentch
		society_member_of = cult_of_undivided
		society_member_of = cult_of_pleasure
		has_character_flag = khorne_tempted
		has_character_flag = tzeentch_tempted
		has_character_flag = hidden_cultist_khorne
		has_character_flag = refusedcallofchaos
		has_character_flag = ai_flag_refuse_conversion
	}
	OR = {
		is_mutated_trigger = yes
		trait = brave
		trait = strong
		trait = duelist
		trait = scarred
		trait = hunter
		trait = impaler
		trait = cruel
		trait = wroth
		martial = 14
	}
}

# A man of ROOT's court or realm fit to be challenged or hunted.
khorne_foe_trigger = {
	is_adult = yes
	prisoner = no
	is_incapacitated_trigger = no
	NOT = { character = ROOT }
	NOT = { is_close_relative = ROOT }
	NOT = { is_married = ROOT }
	is_invincible_trigger = no
}

		society_member_of = cult_of_slaanesh
		society_member_of = cult_of_tzeentch
		society_member_of = cult_of_undivided
		society_member_of = cult_of_pleasure

		has_character_flag = refusedcallofchaos
		has_character_flag = ai_flag_refuse_conversion
	}
	OR = {

		trait = schemer

		trait = ambitious
		trait = envious
