# Triggers for events/WHG_chaos_gods_events.txt (the Dark Gods test their own)

# Scope: a character. Hurt badly enough to be a burden to himself: crippled,
# bedridden or broken in body.
chaosgod_crippled_trigger = {
	OR = {
		trait = maimed
		trait = incapable
		trait = one_legged
		trait = one_handed
		trait = blinded
		trait = mangled
		trait = severely_injured
	}
}

# Scope: the foe of a Khorne trial. Beaten in the duel: hurt or down, so
# the challenger can take the skull.
khorne_trial_foe_beaten_trigger = {
	OR = {
		trait = wounded
		trait = severely_injured
		trait = incapable
		is_maimed_trigger = yes
		trait = blinded
	}
}

# Scope: a spouse, lover, friend or courtier of ROOT. A daemonette of
# Slaanesh who can give counsel.
darkprince_daemonette_trigger = {
	trait = slaanesh_daemonette
	is_alive = yes
	prisoner = no
}

# Scope: a character. Dull of wit: no gift of the mind and little learning.
tzeentch_dull_trigger = {
	NOR = {
		trait = genius
		trait = brilliant
		trait = quick
		trait = smart
		trait = bright
	}
	NOT = { learning = 8 }
}

# Scope: a character. A follower of Slaanesh, openly or in secret, who may
# hunt for a daemonette (darkprince.10): has never sought one and keeps none
# close, is not a daemon himself and can pay his cult.
darkprince_daemonette_hunt_trigger = {
	OR = {
		religion = slaanesh
		secret_religion = slaanesh
	}
	is_adult = yes
	prisoner = no
	is_incapable = no
	is_demon_trigger = no
	NOT = { has_character_flag = slaanesh_daemonette_sought }
	wealth = 50
	NOR = {
		any_spouse = { trait = slaanesh_daemonette }
		any_lover = { trait = slaanesh_daemonette }
		any_courtier = { trait = slaanesh_daemonette }
	}
}
