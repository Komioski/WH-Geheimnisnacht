# ROOT's horde sacks this great city now (greatsack.1 fires or has fired
# today). The usual sacking events skip it, so the city is not sacked twice:
# chaoshorde_sackable_holding_trigger, nomad.24, waaagh.30, beastmen.60.
# Scope: the holding taken; ROOT is the owner of the winning army.
# Used by: WHG_beastmen_events, WHG_chaos_triggers, WHG_nomad_events +1 more
greatsack_city_falls_now_trigger = {
	ROOT = { greatsack_horde_trigger = yes }
	is_capital = yes
	has_holder = yes
	greatsack_victim_trigger = yes
	location = {
		owner = {
			top_liege = {
				higher_tier_than = DUKE
				capital_scope = { province = PREVPREVPREV }
				NOT = { character = ROOT }
				ROOT = { NOT = { is_vassal_or_below = PREV } }
			}
		}
		NOT = { had_province_flag = { flag = great_city_sacked days = 1 } }
	}
}

# The patron can pay for an expedition: 500 piety for a vampire, 750
# prestige for a boss or chief, 150 gold for a Skaven (the clans have ratmen
# to spare), 300 gold for anyone else (200 for the AI).
# Scope: the patron.
# Used by: wh_blackstone_decisions
warpstone_expedition_can_pay_trigger = {
	OR = {
		AND = {
			warpstone_expedition_pays_piety_trigger = yes
			piety = 500
		}
		AND = {
			warpstone_expedition_pays_prestige_trigger = yes
			prestige = 750
		}
		AND = {
			warpstone_expedition_pays_piety_trigger = no
			warpstone_expedition_pays_prestige_trigger = no
			trait = creature_skaven
			wealth = 150
		}
		AND = {
			warpstone_expedition_pays_piety_trigger = no
			warpstone_expedition_pays_prestige_trigger = no
			NOT = { trait = creature_skaven }
			ai = yes
			wealth = 200
		}
		AND = {
			warpstone_expedition_pays_piety_trigger = no
			warpstone_expedition_pays_prestige_trigger = no
			NOT = { trait = creature_skaven }
			ai = no
			wealth = 300
		}
	}
}

# Traits that make a newborn of beastman blood a bray shaman: Tzeentch favours
# the clever, Slaanesh the beautiful, Nurgle the marked, and the undivided
# Chaos gods the strong.
# Scope: the child.
# Used by: WHG_beastmen_events
beastmen_bray_shaman_birth_trigger = {
	OR = {
		AND = {
			religion = tzeentch
			OR = { trait = genius trait = elusive_shadow trait = brilliant trait = bright trait = quick }
		}
		AND = {
			religion = slaanesh
			OR = { trait = beautiful trait = fair trait = pretty trait = genius trait = unsightly }
		}
		AND = {
			religion = nurgle
			OR = {
				trait = genius
				trait = midas_touched
				trait = brilliant
				trait = hunchback
				trait = lisp
				trait = stutter
			}
		}
		AND = {
			religion = chaos
			OR = {
