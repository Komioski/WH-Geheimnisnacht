# Province scope. Yes when the province lies in ROOT's homeland, the lands
# the colony decisions leave alone. The homelands match the decisions in
# decisions/dow_new_world.txt: Ulthuan for the Phoenix King, the Old World
# with the north, the mountains and the Badlands for an Old World capital,
# and the capital's own part of the world for the others. A capital
# anywhere else counts every province as homeland, so nothing is taken.
colony_homeland_trigger = {
	OR = {
		AND = {
			ROOT = { has_landed_title = e_ulthuan }
			region = world_ulthuan
		}
		AND = {
			ROOT = {
				NOT = { has_landed_title = e_ulthuan }
				capital_scope = { region = world_old_world }
			}
			OR = {
				region = world_old_world
				region = world_norsca
				region = world_karaz_ankor
				region = world_drak_ankor
				region = world_badlands
				region = world_chaos_wastes
			}
		}
		AND = {
			ROOT = {
				NOT = { has_landed_title = e_ulthuan }
				capital_scope = { region = world_new_world }
			}
			region = world_new_world
		}
		AND = {
			ROOT = {
				NOT = { has_landed_title = e_ulthuan }
				capital_scope = { region = world_athel_loren }
			}
			region = world_athel_loren
		}
		AND = {
			ROOT = {
				NOT = { has_landed_title = e_ulthuan }
				capital_scope = {
					OR = {
						region = world_darklands
						region = world_badlands
					}
				}
			}
			OR = {
				region = world_darklands
				region = world_badlands
			}
		}
		AND = {
			ROOT = {
				NOT = { has_landed_title = e_ulthuan }
				capital_scope = {
					OR = {
						region = world_chaos_wastes
						region = world_norsca
						region = world_drak_ankor
					}
				}
			}
			OR = {
				region = world_chaos_wastes
				region = world_norsca
				region = world_drak_ankor
			}
		}
		AND = {
			ROOT = {
				NOT = { has_landed_title = e_ulthuan }
				capital_scope = {
					OR = {
						region = world_greater_araby
						region = world_nehekhara
					}
				}
			}
			OR = {
				region = world_greater_araby
				region = world_nehekhara
			}
		}
		ROOT = {
			NOT = { has_landed_title = e_ulthuan }
			capital_scope = {
				NOR = {
					region = world_old_world
					region = world_new_world
					region = world_athel_loren
					region = world_darklands
					region = world_badlands
					region = world_chaos_wastes
					region = world_norsca
					region = world_drak_ankor
					region = world_greater_araby
					region = world_nehekhara
				}
			}
		}
	}
}

# Character scope. Yes when the character is of ROOT's race, so the colony
# goes to a lord of the founder's own people.
colony_same_race_as_root_trigger = {
	OR = {
		AND = {
			trait = creature_elf
			ROOT = { trait = creature_elf }
		}
		AND = {
			trait = creature_human
			ROOT = { trait = creature_human }
		}
		AND = {
			trait = creature_chaos_dwarf
			ROOT = { trait = creature_chaos_dwarf }
		}
		AND = {
			trait = creature_dwarf
			ROOT = { trait = creature_dwarf }
		}
	}
}
