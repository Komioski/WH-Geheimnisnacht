# Council actions (events/WHG_council_action_events.txt)
#
# council_action_fit_trigger  the councillor can work: not a beastman (they
#                             have their own job events), not in seclusion,
#                             bedridden or incapable
# marshal_at_post_trigger     the marshal works in a province of his liege
# marshal_tutor_trigger       a province lord who will teach the marshal
# marshal_meddler_trigger     a province lord who holds back his levies
# steward_at_post_trigger     the steward works in a province of his liege
# steward_saboteur_trigger    a province lord who sabotages the works
# steward_settle_trigger      the steward can settle the province's tribe
# chancellor_charm_target_trigger    a lord the chancellor may win over
# chancellor_offend_target_trigger   a lord the chancellor may offend
# chancellor_dissent_target_trigger  a foreign vassal to turn on his liege
# chancellor_claim_holder_trigger    a title holder the chancellor may forge
#                                    a claim against
# chancellor_can_forge_trigger       the chancellor has a title to forge on
# spiritual_investiture_allows_trigger  the Grail investiture laws allow the
#                                       inquisition here
# spiritual_imperial_cult_trigger       a character or province of one of the
#                                       Empire's official cults
# spiritual_convert_province_trigger    the Lord Spiritual can convert his
#                                       province
# spiritual_convert_target_trigger      a character he may convert
# spiritual_heresy_suspect_trigger      a lord he may accuse of heresy
# spiritual_fellow_priest_trigger       a priest-lord of his faith
# spiritual_realm_priest_trigger        a priest-lord of the realm to turn
#                                       against the head of the faith
# spiritual_alms_target_trigger         a lord of his faith (charity)
# spiritual_holy_war_trigger            the liege fights a holy war
# spymaster_plotter_trigger     a lord with a plot the spymaster can uncover
# spymaster_faction_target_trigger  a fellow vassal in a faction against the
#                                   liege (faction prevention)
# spymaster_recruit_target_trigger  a fellow vassal the liege may draw into
#                                   his faction (faction recruitment)
# spymaster_vice_trigger        a vice the spymaster can blackmail with
# spymaster_law_trigger         a lord who yields to the letter of the law
# spymaster_fear_trigger        a lord who yields to threats
# spymaster_rumour_target_trigger   a lord with a secret worth spreading
# spymaster_corrupt_target_trigger  a lord of the realm the liege dislikes
# spymaster_sharp_lord_trigger  a lord who can catch a foreign spymaster
###############################################################################

# ROOT is the councillor.
council_action_fit_trigger = {
	NOT = { trait = creature_beastman }
	NOT = { trait = incapable }
	NOT = { has_character_modifier = in_seclusion }
	NOT = { has_character_modifier = bedridden_illness }
}

# ROOT is the marshal.
marshal_at_post_trigger = {
	location = { owner = { job_marshal = { character = ROOT } } }
}

# A province lord; ROOT is the marshal. Warlike, well spoken and fond of him.
marshal_tutor_trigger = {
	NOT = { character = ROOT }
	NOT = { is_liege_of = ROOT }
	martial = 8
	OR = { trait = gregarious diplomacy = 7 }
	opinion = { who = ROOT value = 30 }
}

# A province lord; ROOT is the marshal. An ambitious or deceitful vassal.
marshal_meddler_trigger = {
	NOR = { character = ROOT is_liege_of = ROOT }
	independent = no
	OR = { trait = ambitious trait = deceitful }
}

# ROOT is the steward.
steward_at_post_trigger = {
	location = { owner = { job_treasurer = { character = ROOT } } }
}

# A province lord; ROOT is the steward. A vassal (not the steward or his
# liege) whose own liege is not on bad terms with the steward.
steward_saboteur_trigger = {
	independent = no
	NOR = { character = ROOT is_liege_of = ROOT }
	liege = { NOT = { reverse_opinion = { who = ROOT value = -20 } } }
}

# ROOT is the steward. The province is his liege's, of another culture, and
# the culture game rule lets it change: with restricted conversion it must
# border the liege's culture or the sea (else the settlers would attack the
# steward forever with nothing to gain).
steward_settle_trigger = {
	location = {
		owner = {
			job_treasurer = { character = ROOT }
			NOT = { culture = PREV }
		}
		OR = {
			NOT = { has_game_rule = { name = culture_conversion value = restricted } }
			any_neighbor_province = { ROOT = { liege = { culture = PREVPREV } } }
			any_neighbor_province = { port = yes }
		}
	}
}

# A province lord; ROOT is the chancellor. A count or better, or a fellow
# vassal; not a priest, the chancellor or his liege.
chancellor_charm_target_trigger = {
	OR = {
		primary_title = { higher_tier_than = BARON }
		same_liege = ROOT
	}
	is_priest = no
	NOR = { is_liege_of = ROOT character = ROOT }
}

# A province lord; ROOT is the chancellor.
chancellor_offend_target_trigger = {
	is_priest = no
	NOR = { is_liege_of = ROOT character = ROOT }
}

# A province lord; ROOT is the chancellor. A vassal of another realm.
chancellor_dissent_target_trigger = {
	independent = no
	NOR = { is_liege_of = ROOT character = ROOT same_realm = ROOT }
}

# A title holder; ROOT is the chancellor. Not the chancellor, his liege or
# his liege's religious head.
chancellor_claim_holder_trigger = {
	NOR = { is_liege_of = ROOT character = ROOT }
	NOT = { ROOT = { liege = { religion_head = { character = PREVPREVPREV } } } }
}

# ROOT is the chancellor. The county of his province, or with diplomacy 15
# the duchy, or with the liege's become-king ambition (once) the kingdom, is
# not yet claimed by the liege.
chancellor_can_forge_trigger = {
	liege = {
		ROOT = {
			location = {
				OR = {
					county = {
						NOT = { claimed_by = PREVPREVPREV }
						holder_scope = { chancellor_claim_holder_trigger = yes }
					}
					AND = {
						ROOT = { diplomacy = 15 }
						duchy = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { chancellor_claim_holder_trigger = yes }
						}
					}
					AND = {
						ROOT = {
							liege = {
								has_ambition = obj_become_king
								NOT = { has_character_flag = kingdom_claimed_with_ambition }
							}
						}
						kingdom = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { chancellor_claim_holder_trigger = yes }
						}
					}
				}
			}
		}
	}
}

# ROOT is the Lord Spiritual. Under the Grail investiture laws the
# inquisition only works against heresy and other faiths.
spiritual_investiture_allows_trigger = {
	OR = {
		top_liege = {
			NOR = {
				has_law = investiture_grail_law_0
				has_law = investiture_grail_law_1
			}
		}
		location = {
			OR = {
				is_heresy_of = ROOT
				NOT = { religion_group = ROOT }
			}
		}
	}
}

# A character or a province.
spiritual_imperial_cult_trigger = {
	OR = {
		religion = sigmarite
		religion = ulrican
		religion = taalite
		religion = morrian
		religion = halfling_gods
	}
}

# ROOT is the Lord Spiritual. Not a Slann, Tomb King or greenskin priest,
# not in Nehekhara, and Hashut's priests leave the greenskin slaves of the
# Hobgoblin satrapies alone. The province is in the realm, or the liege
# heads one of the Empire's cults and the province is an Old World faith
# in the Empire.
spiritual_convert_province_trigger = {
	spiritual_investiture_allows_trigger = yes
	NOT = { trait = creature_nehekharan }
	NOT = { trait = creature_slann }
	NOT = { religion_group = greenskin_gods_group }
	NOT = { location = { religion = ROOT } }
	NOT = { location = { culture = nehekharan } }
	NAND = {
		religion = hashut
		location = { OR = { culture_group = orc_group culture_group = goblin_group } }
	}
	location = {
		OR = {
			owner = { same_realm = ROOT }
			AND = {
				ROOT = {
					liege = {
						controls_religion = yes
						OR = {
							religion = sigmarite
							religion = ulrican
							religion = manannite
							religion = taalite
						}
					}
				}
				region = world_old_world_empire
				religion_group = old_world_group
			}
		}
		OR = {
			NOT = { religion_group = greenskin_gods_group }
			ROOT = { religion = hashut }
		}
	}
}

# A character in the province; ROOT is the Lord Spiritual. A human, halfling
# or Amazon of the realm and another faith; not a vampire, daemon or forest
# spirit, not the liege, not the head of a faith, a band or an order.
spiritual_convert_target_trigger = {
	is_adult = yes
	same_realm = ROOT
	NOR = { character = ROOT is_liege_of = ROOT religion = ROOT }
	is_vampire_trigger = no
	is_daemon_race_trigger = no
	NOT = { trait = creature_forestspirit }
	OR = { trait = creature_human trait = creature_halfling trait = creature_amazon }
	OR = {
		is_ruler = no
		primary_title = {
			controls_religion = no
			mercenary = no
			holy_order = no
		}
	}
}

# A province lord; ROOT is the Lord Spiritual. A cynical, impious grown lord
# of his realm and faith group, not already judged.
spiritual_heresy_suspect_trigger = {
	trait = cynical
	age = 20
	same_realm = ROOT
	religion_group = ROOT
	can_change_religion = yes
	NOT = { piety = 100 }
	NOR = {
		character = ROOT
		is_liege_of = ROOT
		has_character_flag = no_heretic
		has_character_flag = branded_heretic
	}
}

# A province lord; ROOT is the Lord Spiritual.
spiritual_fellow_priest_trigger = {
	is_priest = yes
	religion = ROOT
	NOR = { character = ROOT is_liege_of = ROOT }
}

# A province lord; ROOT is the Lord Spiritual. Not the head of the faith.
spiritual_realm_priest_trigger = {
	spiritual_fellow_priest_trigger = yes
	same_realm = ROOT
	religion_head = { NOT = { character = PREV } }
}

# A province lord; ROOT is the court priest.
spiritual_alms_target_trigger = {
	religion = ROOT
	NOT = { is_liege_of = ROOT }
}

# A ruler fighting a holy war.
spiritual_holy_war_trigger = {
	any_war = {
		OR = {
			using_cb = crusade
			using_cb = religious
			using_cb = religious_revolt
			using_cb = daemonic_subjugation
			using_cb = unholy_raid
		}
	}
}

# A province lord; ROOT is the spymaster. A lord with a plot.
spymaster_plotter_trigger = {
	NOR = { is_liege_of = ROOT character = ROOT }
	OR = {
		has_plot = plot_gain_title
		has_plot = plot_kill_character
		has_plot = plot_kill_spouse
		has_plot = plot_weaken_fellow_vassal
		has_plot = plot_gain_liege_title
		has_plot = plot_gain_fellow_vassal_title
	}
}

# A lord; ROOT is the spymaster. A fellow vassal in a faction against the
# liege, not yet found out. The old event did not check in_faction at all.
spymaster_faction_target_trigger = {
	same_liege = ROOT
	independent = no
	in_faction = yes
	age = 16
	prisoner = no
	primary_title = { higher_tier_than = BARON }
	NOR = {
		is_liege_of = ROOT
		character = ROOT
		trait = incapable
		has_character_flag = faction_member
		has_opinion_modifier = { who = liege modifier = opinion_coerced_into_leaving_faction }
	}
}

# A lord; ROOT is the spymaster, whose liege leads or sits in a faction. An
# AI fellow vassal of the liege, in no faction, not pressed before.
spymaster_recruit_target_trigger = {
	ai = yes
	independent = no
	in_faction = no
	age = 16
	prisoner = no
	primary_title = { higher_tier_than = BARON }
	ROOT = {
		liege = {
			same_liege = PREVPREV
			NOT = { character = PREVPREV }
			NOT = { reverse_has_opinion_modifier = { who = PREVPREV modifier = opinion_coerced_into_joining_faction } }
		}
	}
	NOR = {
		is_liege_of = ROOT
		character = ROOT
		trait = incapable
		has_character_flag = faction_recruit
		has_opinion_modifier = { who = liege modifier = opinion_coerced_into_leaving_faction }
	}
}

# A province lord; ROOT is the spymaster. A lord of a people with women and
# men, with a secret love the court would sneer at, whom the liege dislikes.
spymaster_rumour_target_trigger = {
	NOR = { is_liege_of = ROOT character = ROOT }
	trait = homosexual
	is_no_females_race = no
	num_of_vassals = 1
	ROOT = {
		liege = {
			NOT = { opinion = { who = PREVPREV value = 0 } }
		}
	}
}

# A province lord; ROOT is the spymaster. A lord of the liege's realm whom
# the liege dislikes.
spymaster_corrupt_target_trigger = {
	NOR = { is_liege_of = ROOT character = ROOT }
	ROOT = {
		liege = {
			NOT = { opinion = { who = PREVPREV value = 0 } }
			same_realm = PREVPREV
			NOT = { character = PREVPREV }
		}
	}
}

# A province lord; ROOT is the spymaster. A lord sharp enough, or with a
# sharp enough spymaster, to catch ROOT.
spymaster_sharp_lord_trigger = {
	NOR = { is_liege_of = ROOT character = ROOT }
	OR = {
		intrigue = 10
		job_spymaster = { intrigue = 10 }
	}
}
