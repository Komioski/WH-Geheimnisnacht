# ROOT is the councillor.
council_action_fit_trigger = {
	NOT = { trait = creature_beastman }
	NOT = { trait = incapable }
	NOT = { has_character_modifier = in_seclusion }
	NOT = { has_character_modifier = bedridden_illness }
}

# ROOT is the marshal.
marshal_at_post_trigger = { location = { owner = { job_marshal = { character = ROOT } } } }

# A province lord; ROOT is the marshal. Warlike, well spoken and fond of him.
marshal_tutor_trigger = {
	NOT = { character = ROOT }
	NOT = { is_liege_of = ROOT }
	martial = 8
	OR = { trait = gregarious diplomacy = 7 }
	opinion = { who = ROOT value = 30 }
}

# A province lord; ROOT is the marshal. An ambitious or deceitful vassal.
marshal_meddler_trigger = {
	NOR = { character = ROOT is_liege_of = ROOT }
	independent = no
	OR = { trait = ambitious trait = deceitful }
}

# ROOT is the steward.
steward_at_post_trigger = { location = { owner = { job_treasurer = { character = ROOT } } } }

# A province lord; ROOT is the steward. A vassal (not the steward or his
# liege) whose own liege is not on bad terms with the steward.
steward_saboteur_trigger = {
	independent = no
	NOR = { character = ROOT is_liege_of = ROOT }
	liege = { NOT = { reverse_opinion = { who = ROOT value = -20 } } }
}

# ROOT is the steward. The province is his liege's, of another culture, and
# the culture game rule lets it change: with restricted conversion it must
# border the liege's culture or the sea (else the settlers would attack the
# steward forever with nothing to gain).
steward_settle_trigger = {
	location = {
		owner = { job_treasurer = { character = ROOT } NOT = { culture = PREV } }
		OR = {
			NOT = { has_game_rule = { name = culture_conversion value = restricted } }
			any_neighbor_province = { ROOT = { liege = { culture = PREVPREV } } }
			any_neighbor_province = { port = yes }
		}
	}
}

# A province lord; ROOT is the chancellor. A count or better, or a fellow
# vassal; not a priest, the chancellor or his liege.
chancellor_charm_target_trigger = {
	OR = { primary_title = { higher_tier_than = BARON } same_liege = ROOT }
	is_priest = no
	NOR = { is_liege_of = ROOT character = ROOT }
}

# A province lord; ROOT is the chancellor.
chancellor_offend_target_trigger = { is_priest = no NOR = { is_liege_of = ROOT character = ROOT } }

# A province lord; ROOT is the chancellor. A vassal of another realm.
chancellor_dissent_target_trigger = {
	independent = no
	NOR = { is_liege_of = ROOT character = ROOT same_realm = ROOT }
}

# A title holder; ROOT is the chancellor. Not the chancellor, his liege or
# his liege's religious head.
chancellor_claim_holder_trigger = {
	NOR = { is_liege_of = ROOT character = ROOT }
	NOT = { ROOT = { liege = { religion_head = { character = PREVPREVPREV } } } }
}

# ROOT is the chancellor. The county of his province, or with diplomacy 15
# the duchy, or with the liege's become-king ambition (once) the kingdom, is
# not yet claimed by the liege.
chancellor_can_forge_trigger = {
	liege = {
		ROOT = {
			location = {
				OR = {
					county = {
						NOT = { claimed_by = PREVPREVPREV }
						holder_scope = { chancellor_claim_holder_trigger = yes }
					}
					AND = {
						ROOT = { diplomacy = 15 }
						duchy = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { chancellor_claim_holder_trigger = yes }
						}
					}
					AND = {
						ROOT = {
							liege = {
								has_ambition = obj_become_king
								NOT = { has_character_flag = kingdom_claimed_with_ambition }
							}
						}
						kingdom = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { chancellor_claim_holder_trigger = yes }
						}
					}
				}
			}
		}
	}
}

# ROOT is the Lord Spiritual. Under the Grail investiture laws the
# inquisition only works against heresy and other faiths.
spiritual_investiture_allows_trigger = {
	OR = {
		top_liege = {
			NOR = { has_law = investiture_grail_law_0 has_law = investiture_grail_law_1 }
		}
		location = { OR = { is_heresy_of = ROOT NOT = { religion_group = ROOT } } }
	}
}

# A character or a province.

	OR = {
		religion = sigmarite
		religion = ulrican
		religion = taalite
		religion = morrian
		religion = halfling_gods
	}

# ROOT is the Lord Spiritual. Not a Slann, Tomb King or greenskin priest,
# not in Nehekhara, and Hashut's priests leave the greenskin slaves of the
# Hobgoblin satrapies alone. The province is in the realm, or the liege
# heads one of the Empire's cults and the province is an Old World faith
# in the Empire.
spiritual_convert_province_trigger = {
	spiritual_investiture_allows_trigger = yes
	NOT = { trait = creature_nehekharan }
	NOT = { trait = creature_slann }
	NOT = { religion_group = greenskin_gods_group }
	NOT = { location = { religion = ROOT } }
	NOT = { location = { culture = nehekharan } }
	NAND = {
		religion = hashut
		location = { OR = { culture_group = orc_group culture_group = goblin_group } }
	}
	location = {
		OR = {
			owner = { same_realm = ROOT }
			AND = {
				ROOT = {
					liege = {
						controls_religion = yes
						OR = {
							religion = sigmarite
							religion = ulrican
							religion = manannite
							religion = taalite
						}
					}
				}
				region = world_old_world_empire
				religion_group = old_world_group
			}
		}
		OR = { NOT = { religion_group = greenskin_gods_group } ROOT = { religion = hashut } }
	}
}

# A character in the province; ROOT is the Lord Spiritual. A human, halfling

	same_realm = ROOT
	NOR = { character = ROOT is_liege_of = ROOT religion = ROOT }
	is_vampire_trigger = no
	is_daemon_race_trigger = no
	NOT = { trait = creature_forestspirit }
	OR = { trait = creature_human trait = creature_halfling trait = creature_amazon }
	OR = { is_ruler = no primary_title = { controls_religion = no mercenary = no holy_order = no } }
}

# A lord; ROOT is the spymaster. A fellow vassal in a faction against the
# liege, not yet found out. The old event did not check in_faction at all.
spymaster_faction_target_trigger = {
	same_liege = ROOT
	independent = no
	in_faction = yes
	age = 16
	prisoner = no

		is_liege_of = ROOT

		has_character_flag = faction_member
		has_opinion_modifier = { who = liege modifier = opinion_coerced_into_leaving_faction }
	}
}

# A lord; ROOT is the spymaster, whose liege leads or sits in a faction. An

# A courtier or vassal of the liege; ROOT is the marshal. A commander the
# marshal can still drill (marshal.50).
marshal_drill_target_trigger = {
	NOT = { character = ROOT }
	is_ruler = no
	has_minor_title = title_commander
	NOT = { has_character_modifier = drill_cooldown_timer }
	NAND = { martial = 20 has_any_leadership_trait_trigger = yes }
}

# A ruler. True if the ruler already has as many commanders as the realm
# allows: 2 / 4 / 6 / 8 for count / duke / king / emperor, 2 / 4 for nomad
# kings / emperors.
marshal_commander_slots_full_trigger = {
	OR = {
		AND = {
			tier = COUNT
			any_realm_character = {
				count = 2
				liege = { character = PREVPREV }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = DUKE
			any_realm_character = {
				count = 4
				liege = { character = PREVPREV }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = KING
			is_nomadic = no
			any_realm_character = {
				count = 6
				liege = { character = PREVPREV }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = EMPEROR
			is_nomadic = no
			any_realm_character = {
				count = 8
				liege = { character = PREVPREV }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = KING
			is_nomadic = yes
			any_realm_character = {
				count = 2
				liege = { character = PREVPREV }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = EMPEROR
			is_nomadic = yes
			any_realm_character = {
				count = 4
				liege = { character = PREVPREV }
				has_minor_title = title_commander
			}
		}
	}
}

# A vassal; PREV is the liege. A landed vassal who could take a command
# under Imperial elective law (which wants vassals as commanders).
marshal_vassal_commander_trigger = {
	higher_real_tier_than = BARON
	NOT = { primary_title = { temporary = yes } }
	OR = {
		AND = {
			OR = {
				male_can_hold_minor_title_trigger = yes
				female_can_hold_minor_commander_title_trigger = yes
			}
			is_adult = yes
		}
		has_character_flag = special_marshal
	}
	prisoner = no
	NOT = { trait = incapable }
	NOT = { is_inaccessible_trigger = yes }
	NOT = { has_character_flag = guru }
	NOT = { has_character_modifier = refusing_to_lead }
	NOR = {
		has_job_title = job_chancellor
		has_job_title = job_marshal
		has_job_title = job_treasurer
		has_job_title = job_spymaster
		has_job_title = job_spiritual
		has_minor_title = title_commander
	}
}

# The liege; the found commander is event_target:found_commander. True if
# the liege has a free commander seat the new soldier may take.
marshal_may_appoint_commander_trigger = {
	marshal_commander_slots_full_trigger = no
	OR = {
		NOT = { has_law = succ_byzantine_elective }
		NOT = { any_vassal = { marshal_vassal_commander_trigger = yes } }
	}
}

# A realm character; ROOT is the liege. A low-born commander with less than

marshal_weak_commander_trigger = {

	liege = { character = ROOT }
	has_minor_title = title_commander
	NOT = { martial = 10 }
}

# Hunt apostates (spiritual.100). Scope: a realm character; PREV is the
# liege, ROOT the Lord Spiritual. Never the hunter or the liege, nor anyone
# the liege has already accused.
spiritual_apostate_base_trigger = {
	prisoner = no
	NOT = { character = ROOT }
	NOT = { character = PREV }
	NOT = { reverse_has_opinion_modifier = { modifier = opinion_accused_apostate who = PREV } }
}

# A suspect (or highly suspect) member of an outlawed society, not a
# brother of the hunter or the liege, nor of their secret faith unless a
# devil worshipper.
spiritual_apostate_society_trigger = {
	spiritual_apostate_base_trigger = yes
	NOT = { same_society_as = ROOT }
	NOT = { same_society_as = PREV }
	NAND = { has_roots_secret_religion_truly_trigger = yes is_devil_worshiper_trigger = no }
	NAND = { has_prevs_secret_religion_truly_trigger = yes is_devil_worshiper_trigger = no }
	is_in_society = yes
	society_is_criminal = yes
	OR = { has_character_modifier = highly_suspect has_character_modifier = under_suspicion }
}

# A grown secret heretic of the liege's faith.
spiritual_apostate_secret_faith_trigger = {
	spiritual_apostate_base_trigger = yes
	OR = { NOT = { same_society_as = ROOT } is_in_society = no }
	OR = { NOT = { same_society_as = PREV } is_in_society = no }
	NOT = { has_roots_secret_religion_truly_trigger = yes }
	NOT = { has_prevs_secret_religion_truly_trigger = yes }
	religion = PREV
	age = 16

}

# An AI devil worshipper under no suspicion at all; only a learned hunter

spiritual_apostate_hidden_devil_trigger = {
