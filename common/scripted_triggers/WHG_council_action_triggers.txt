###############################################################################
# Council actions (events/WHG_council_action_events.txt)
#
# council_action_fit_trigger  the councillor can work: not a beastman (they
#                             have their own job events), not in seclusion,
#                             bedridden or incapable
# marshal_at_post_trigger     the marshal works in a province of his liege
# marshal_tutor_trigger       a province lord who will teach the marshal
# marshal_meddler_trigger     a province lord who holds back his levies
# steward_at_post_trigger     the steward works in a province of his liege
# steward_saboteur_trigger    a province lord who sabotages the works
# steward_settle_trigger      the steward can settle the province's tribe
# chancellor_charm_target_trigger    a lord the chancellor may win over
# chancellor_offend_target_trigger   a lord the chancellor may offend
# chancellor_dissent_target_trigger  a foreign vassal to turn on his liege
# chancellor_claim_holder_trigger    a title holder the chancellor may forge
#                                    a claim against
# chancellor_can_forge_trigger       the chancellor has a title to forge on
###############################################################################

# ROOT is the councillor.
council_action_fit_trigger = {
	NOT = { trait = creature_beastman }
	NOT = { trait = incapable }
	NOT = { has_character_modifier = in_seclusion }
	NOT = { has_character_modifier = bedridden_illness }
}

# ROOT is the marshal.
marshal_at_post_trigger = {
	location = { owner = { job_marshal = { character = ROOT } } }
}

# A province lord; ROOT is the marshal. Warlike, well spoken and fond of him.
marshal_tutor_trigger = {
	NOT = { character = ROOT }
	NOT = { is_liege_of = ROOT }
	martial = 8
	OR = { trait = gregarious diplomacy = 7 }
	opinion = { who = ROOT value = 30 }
}

# A province lord; ROOT is the marshal. An ambitious or deceitful vassal.
marshal_meddler_trigger = {
	NOR = { character = ROOT is_liege_of = ROOT }
	independent = no
	OR = { trait = ambitious trait = deceitful }
}

# ROOT is the steward.
steward_at_post_trigger = {
	location = { owner = { job_treasurer = { character = ROOT } } }
}

# A province lord; ROOT is the steward. A vassal (not the steward or his
# liege) whose own liege is not on bad terms with the steward.
steward_saboteur_trigger = {
	independent = no
	NOR = { character = ROOT is_liege_of = ROOT }
	liege = { NOT = { reverse_opinion = { who = ROOT value = -20 } } }
}

# ROOT is the steward. The province is his liege's, of another culture, and
# the culture game rule lets it change: with restricted conversion it must
# border the liege's culture or the sea (else the settlers would attack the
# steward forever with nothing to gain).
steward_settle_trigger = {
	location = {
		owner = {
			job_treasurer = { character = ROOT }
			NOT = { culture = PREV }
		}
		OR = {
			NOT = { has_game_rule = { name = culture_conversion value = restricted } }
			any_neighbor_province = { ROOT = { liege = { culture = PREVPREV } } }
			any_neighbor_province = { port = yes }
		}
	}
}

# A province lord; ROOT is the chancellor. A count or better, or a fellow
# vassal; not a priest, the chancellor or his liege.
chancellor_charm_target_trigger = {
	OR = {
		primary_title = { higher_tier_than = BARON }
		same_liege = ROOT
	}
	is_priest = no
	NOR = { is_liege_of = ROOT character = ROOT }
}

# A province lord; ROOT is the chancellor.
chancellor_offend_target_trigger = {
	is_priest = no
	NOR = { is_liege_of = ROOT character = ROOT }
}

# A province lord; ROOT is the chancellor. A vassal of another realm.
chancellor_dissent_target_trigger = {
	independent = no
	NOR = { is_liege_of = ROOT character = ROOT same_realm = ROOT }
}

# A title holder; ROOT is the chancellor. Not the chancellor, his liege or
# his liege's religious head.
chancellor_claim_holder_trigger = {
	NOR = { is_liege_of = ROOT character = ROOT }
	NOT = { ROOT = { liege = { religion_head = { character = PREVPREVPREV } } } }
}

# ROOT is the chancellor. The county of his province, or with diplomacy 15
# the duchy, or with the liege's become-king ambition (once) the kingdom, is
# not yet claimed by the liege.
chancellor_can_forge_trigger = {
	liege = {
		ROOT = {
			location = {
				OR = {
					county = {
						NOT = { claimed_by = PREVPREVPREV }
						holder_scope = { chancellor_claim_holder_trigger = yes }
					}
					AND = {
						ROOT = { diplomacy = 15 }
						duchy = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { chancellor_claim_holder_trigger = yes }
						}
					}
					AND = {
						ROOT = {
							liege = {
								has_ambition = obj_become_king
								NOT = { has_character_flag = kingdom_claimed_with_ambition }
							}
						}
						kingdom = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { chancellor_claim_holder_trigger = yes }
						}
					}
				}
			}
		}
	}
}
