# Scope: a councillor of ROOT. True if the councillor is greedy or arbitrary
# enough, and dislikes ROOT enough, to dip into the realm coffers
# (council.100).
council_coffer_thief_trigger = {
	NOR = {
		trait = incapable
		is_inaccessible_trigger = yes
		has_character_flag = caught_stealing_from_coffers
	}
	prisoner = no
	OR = {
		trait = arbitrary
		AND = {
			trait = greedy
			NOT = { trait = just }
		}
	}
	NOT = { opinion = { who = ROOT value = 35 } }
}
