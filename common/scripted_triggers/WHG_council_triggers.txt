# Scope: a councillor of ROOT. True if the councillor is greedy or arbitrary
# enough, and dislikes ROOT enough, to dip into the realm coffers
# (council.100).
council_coffer_thief_trigger = {
	NOR = {
		trait = incapable
		is_inaccessible_trigger = yes
		has_character_flag = caught_stealing_from_coffers
	}
	prisoner = no
	OR = {
		trait = arbitrary
		AND = {
			trait = greedy
			NOT = { trait = just }
		}
	}
	NOT = { opinion = { who = ROOT value = 35 } }
}

###############################################################################
# Talented women at court (council.700 - council.727)
###############################################################################

# A faith that neither bars women from the council nor already opens it.
# Scope: the ruler.
council_talent_faith_trigger = {
	NOR = {
		has_religion_feature = religion_matriarchal
		has_religion_feature = religion_patriarchal
		has_religion_feature = religion_equal
	}
}

# As above, and the realm's law does not yet open the council to women.
# Scope: the ruler.
council_talent_realm_trigger = {
	council_talent_faith_trigger = yes
	NOR = {
		primary_title = { has_law = status_of_women_3 }
		primary_title = { has_law = status_of_women_4 }
	}
}

# A woman who holds no council seat and was not refused one.
# Scope: the woman. ROOT: the ruler.
council_talent_free_woman_trigger = {
	is_woman_trigger = yes
	is_adult = yes
	prisoner = no
	NOT = { trait = content }
	NOT = { character = ROOT }
	NOR = {
		has_job_title = job_chancellor
		has_job_title = job_marshal
		has_job_title = job_treasurer
		has_job_title = job_spymaster
		has_job_title = job_spiritual
		has_opinion_modifier = { who = ROOT modifier = opinion_refused_council }
	}
}

# The law or the faith keeps a woman from the spymaster's seat: the realm
# allows women only the lesser seats, or both serve the Dark Gods, or she
# is the ruler's wife or mother.
# Scope: the woman. Her liege is the ruler.
council_talent_spymaster_barred_trigger = {
	OR = {
		liege = { primary_title = { has_law = status_of_women_2 } }
		liege = { primary_title = { has_law = status_of_women_1 } }
		AND = {
			religion_group = chaos_gods_group
			liege = { religion_group = chaos_gods_group }
		}
		liege = { any_spouse = { character = PREVPREV } }
		liege = { mother = { character = PREVPREV } }
	}
}

# A kinswoman gifted enough (15) to ask for the seat the law keeps from her.
# Scope: the kinswoman. Her liege is the ruler.
council_talent_kin_chancellor_trigger = {
	diplomacy = 15
	NOT = { liege = { primary_title = { has_law = status_of_women_2 } } }
}
council_talent_kin_treasurer_trigger = {
	stewardship = 15
	NOT = { liege = { primary_title = { has_law = status_of_women_2 } } }
}
council_talent_kin_spymaster_trigger = {
	intrigue = 15
	council_talent_spymaster_barred_trigger = no
}

# A close kinswoman at the ruler's court who could ask for a seat.
# Scope: the kinswoman. ROOT: the ruler.
council_talent_kinswoman_trigger = {
	ai = yes
	liege = { character = ROOT }
	council_talent_free_woman_trigger = yes
	OR = {
		council_talent_kin_chancellor_trigger = yes
		martial = 15
		council_talent_kin_treasurer_trigger = yes
		council_talent_kin_spymaster_trigger = yes
	}
}

# A landed vassal woman who could ask for a seat: gifted (13), or ambitious
# or proud with the education for it.
# Scope: the vassal. ROOT: the liege.
council_talent_vassal_trigger = {
	is_landed = yes
	council_talent_free_woman_trigger = yes
	NOT = { primary_title = { temporary = yes } }
	OR = {
		AND = {
			OR = {
				diplomacy = 13
				AND = {
					has_education_diplomacy_trigger = yes
					OR = { trait = ambitious trait = proud }
				}
			}
			NOT = { ROOT = { primary_title = { has_law = status_of_women_2 } } }
		}
		martial = 13
		AND = {
			has_education_martial_trigger = yes
			OR = { trait = ambitious trait = proud }
		}
		AND = {
			OR = {
				stewardship = 13
				AND = {
					has_education_stewardship_trigger = yes
					OR = { trait = ambitious trait = proud }
				}
			}
			NOT = { ROOT = { primary_title = { has_law = status_of_women_2 } } }
		}
		AND = {
			OR = {
				intrigue = 13
				AND = {
					has_education_intrigue_trigger = yes
					OR = { trait = ambitious trait = proud }
				}
			}
			council_talent_spymaster_barred_trigger = no
		}
	}
}

# An unwed man at court of no great importance, fit to marry the stranger.
# Scope: a courtier. ROOT: the ruler. FROM: the woman.
council_talent_husband_trigger = {
	NOT = { character = ROOT }
	is_adult = yes
	is_married = no
	is_female = no
	prisoner = no
	is_unimportant_trigger = yes
	can_marry = FROM
}

# Scope: a courtier of the liege. PREV is the liege, ROOT the zealous Lord
# Spiritual. True if the courtier is a fair target for the Lord Spiritual's
# zeal: a grown, free courtier who is not a ruler, a councillor, or the
# liege's own close kin or spouse (council.840, council.850).
council_zeal_target_trigger = {
	age = 16
	prisoner = no
	is_ruler = no
	NOT = { character = ROOT }
	NOT = { is_close_relative = PREV }
	NOT = { is_married = PREV }
	NOR = {
		has_job_title = job_chancellor
		has_job_title = job_marshal
		has_job_title = job_treasurer
		has_job_title = job_spymaster
		has_job_title = job_spiritual
	}
}

# Scope: the liege of the zealous Lord Spiritual. True if the liege may
# burn a man for his faith. Araby keeps its court sorcerers and holds no
# burnings (council.840, council.850).
council_may_burn_trigger = {
	NOT = { culture_group = arabyan_group }
}
