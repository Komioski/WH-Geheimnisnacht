# Triggers for events/WHG_court_office_events.txt

# Character scope. Follows the Way of the One: the Ormazdic faith or the
# Kitabid sect, whose law the Grand Judge keeps.
court_way_of_one_trigger = {
	OR = {
		religion = ormazd
		religion = kitabid
	}
}

# Character scope. Could sit as Grand Judge: an adult, free and capable
# cleric of the southern faiths whom the liege's laws allow to hold the post.
court_judge_candidate_trigger = {
	is_adult = yes
	prisoner = no
	is_incapable = no
	religion_group = ormazd_gods_group
	NOT = { has_minor_title = title_chief_qadi }
	OR = {
		trait = detached_priest
		trait = martial_cleric
		trait = scholarly_theologian
		trait = mastermind_theologian
	}
	OR = {
		male_can_hold_minor_title_trigger = yes
		female_can_hold_minor_title_trigger = yes
	}
	NOT = { is_inaccessible_trigger = yes }
}

# Character scope, ROOT is the courtier who speaks up (judge.10). A candidate
# ROOT would put forward: no rival of ROOT, and when ROOT wants the sitting
# judge replaced (flag court_judge_replacing) a learned, diligent one.
court_judge_suggestion_trigger = {
	court_judge_candidate_trigger = yes
	NOT = { any_rival = { character = ROOT } }
	OR = {
		ROOT = { NOT = { has_character_flag = court_judge_replacing } }
		AND = {
			learning = 15
			NOT = { trait = arbitrary }
			NOT = { trait = slothful }
		}
	}
}

# Character scope. A sitting Grand Judge who is a disgrace to the post:
# unlearned, and arbitrary or slothful.
court_judge_unfit_trigger = {
	has_minor_title = title_chief_qadi
	NOT = { learning = 15 }
	OR = {
		trait = arbitrary
		trait = slothful
	}
}

# Character scope. The man ROOT has heard is lying with a lover (judge.40):
# a married vassal of the southern faiths whose wife is not the lover.
court_judge_adulterer_trigger = {
	is_female = no
	has_lover = yes
	NOT = { trait = homosexual }
	religion_group = ormazd_gods_group
	OR = {
		trait = lustful
		trait = deceitful
		NOT = { trait = zealous }
	}
	any_spouse = {
		religion_group = ormazd_gods_group
		NOT = { any_lover = { character = PREVPREV } }
	}
	NOT = { character = ROOT }
	NOT = { is_inaccessible_trigger = yes }
}

# Character scope, the host. Someone at this table could slip poison into
# the cup or the dish: a liege at the poisoner's court who is not busy at war.
court_poison_host_trigger = {
	is_liege_of = ROOT
	in_command = no
	at_location = ROOT
	NOT = { reverse_opinion = { who = ROOT value = 0 } }
}

# Character scope. Sharp enough to find out who poisoned them: good intrigue,
# or a spymaster with it.
court_poison_noticed_trigger = {
	OR = {
		intrigue = 15
		job_spymaster = { intrigue = 15 }
	}
}
