# Subject: the court: council, offices, intrigue, plots, prison, friends and
# rivals
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Shared: used by three or more files (5)
#   Used by WHG_council_action_events (40)
#   Used by WHG_council_events (9)
#   Used by WHG_court_events (15)
#   Used by WHG_court_office_events (7)
#   Building blocks of other entries in this file (9)
#   Used by WHG_friends_rivals_events (31)
#   Used by WHG_intrigue_events (7)
#   Other helpers used by one or two files (13)

# ROOT is the steward. The province is his liege's, of another culture, and
# the culture game rule lets it change: with restricted conversion it must
# border the liege's culture or the sea (else the settlers would attack the
# steward forever with nothing to gain).
# Used by: WHG_council_action_events
steward_settle_trigger = {
	location = {
		owner = { job_treasurer = { character = ROOT } NOT = { culture = PREV } }
		OR = {
			NOT = { has_game_rule = { name = culture_conversion value = restricted } }
			any_neighbor_province = { ROOT = { liege = { culture = PREVPREV } } }
			any_neighbor_province = { port = yes }
		}
	}
}

# ROOT is the chancellor. The county of his province, or with diplomacy 15
# the duchy, or with the liege's become-king ambition (once) the kingdom, is
# not yet claimed by the liege.
# Used by: WHG_council_action_events
chancellor_can_forge_trigger = {
	liege = {
		ROOT = {
			location = {
				OR = {
					county = {
						NOT = { claimed_by = PREVPREVPREV }
						holder_scope = { chancellor_claim_holder_trigger = yes }
					}
					AND = {
						ROOT = { diplomacy = 15 }
						duchy = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { chancellor_claim_holder_trigger = yes }
						}
					}
					AND = {
						ROOT = {
							liege = {
								has_ambition = obj_become_king
								NOT = { has_character_flag = kingdom_claimed_with_ambition }
							}
						}
						kingdom = {
							NOT = { claimed_by = PREVPREVPREV }
							holder_scope = { chancellor_claim_holder_trigger = yes }
						}
					}
				}
			}
		}
	}
}

# ROOT is the Lord Spiritual. Not a Slann, Tomb King or greenskin priest,
# not in Nehekhara, and Hashut's priests leave the greenskin slaves of the
# Hobgoblin satrapies alone. The province is in the realm, or the liege
# heads one of the Empire's cults and the province is an Old World faith
# in the Empire.
# Used by: WHG_council_action_events
spiritual_convert_province_trigger = {
	spiritual_investiture_allows_trigger = yes
	NOT = { trait = creature_nehekharan }
	NOT = { trait = creature_slann }
	NOT = { religion_group = greenskin_gods_group }
	NOT = { location = { religion = ROOT } }
	NOT = { location = { culture = nehekharan } }
	NAND = {
		religion = hashut
		location = { OR = { culture_group = orc_group culture_group = goblin_group } }
	}
	location = {
		OR = {
			owner = { same_realm = ROOT }
			AND = {
				ROOT = {
					liege = {
						controls_religion = yes
						OR = {
							religion = sigmarite
							religion = ulrican
							religion = manannite
							religion = taalite
						}
					}
				}
				region = world_old_world_empire
				religion_group = old_world_group
			}
		}
		OR = { NOT = { religion_group = greenskin_gods_group } ROOT = { religion = hashut } }
	}
}

# A ruler. True if the ruler already has as many commanders as the realm
# allows: 2 / 4 / 6 / 8 for count / duke / king / emperor, 2 / 4 for nomad
# kings / emperors.
# Used by: WHG_council_action_events, WHG_court_triggers
marshal_commander_slots_full_trigger = {
	OR = {
		AND = {
			tier = COUNT
			any_realm_character = {
				count = 2
				liege = { character = PREVPREV }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = DUKE
			any_realm_character = {
				count = 4
				liege = { character = PREVPREV }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = KING
			is_nomadic = no
			any_realm_character = {
				count = 6
				liege = { character = PREVPREV }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = EMPEROR
			is_nomadic = no
			any_realm_character = {
				count = 8
				liege = { character = PREVPREV }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = KING
			is_nomadic = yes
			any_realm_character = {
				count = 2
				liege = { character = PREVPREV }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = EMPEROR
			is_nomadic = yes
			any_realm_character = {
				count = 4
				liege = { character = PREVPREV }
				has_minor_title = title_commander
			}
		}
	}
}
