# mourning_loving_widower_trigger
# Scope: the husband of a wife who just died (mourning.30). A ruler at home
# who loved her (opinion 50), kind or charitable, and not already given
# to grief, celibacy or kin-killing.
mourning_loving_widower_trigger = {
	is_alive = yes
	is_abroad = no
	is_ruler = yes
	opinion = { who = ROOT value = 50 }
	OR = { trait = kind trait = charitable }
	NOR = { trait = kinslayer trait = celibate trait = depressed trait = eunuch }
}

# mourning_grieving_friend_trigger
# Scope: a courtier of the murdered (mourning.35). Someone at court, free
# and well, who loved the dead (opinion 50) and is not their liege.
mourning_grieving_friend_trigger = {
	is_abroad = no
	in_command = no
	prisoner = no
	age = 14
	NOT = { trait = incapable }
	NOT = { character = ROOT }
	NOT = { is_liege_of = ROOT }
	opinion = { who = ROOT value = 50 }
}
