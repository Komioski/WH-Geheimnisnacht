# Used by: WHG_cathay_diplomacy_events, wh_orient_decisions +3 more
cathay_diplo_interaction_trigger = {
	offmap_cathay = {
		governor = {
			NOT = { war_with = PREVPREV }
		}
		NOT = { has_policy = cathay_isolationist }
	}
	conditional_tooltip = {
		trigger = { government = celestial_empire_government }
		custom_tooltip = {
			text = cathay_offmap_decisions_pretender_emperor_tt
			always = no
		}
	}
}

# What goes into the two triggers above...
# Used by: wh_orient_decisions, WHG_cathay_decisions +2 more
cathay_offmap_decisions_enabled_trigger = {

	conditional_tooltip = {
		trigger = {
			FROM = {
				trait=isolationist_elf
			}
		}
		custom_tooltip = {
			text = isolationist_elfs_dont_interact_with_cathay
			always=no
		}
	}
	conditional_tooltip = {
		trigger = {
			FROM = {
				AND={
					religion=draconite
					OR = {
						has_dynasty_flag = link_with_cathay
						top_liege = { has_dynasty_flag = link_with_cathay }
					}
				}
				capital_scope={
					OR = {
						location = { region = world_nehekhara_eastern }
						location = { region = world_eastern_southlands }
						location = { region = world_darklands_plains_of_bones }
						location = { region = world_dragon_isles }
						location = { region = world_darklands_gnoblar_country }
					}
				}
				NOT={
					has_global_flag=dragon_isles_cleansed
				}
			}
		}
		custom_tooltip = {
			text = cathay_dragon_isles_threat_ended
			always=no
		}
	}
	custom_tooltip = {
		text = HAS_LINK_WITH_CATHAY
		FROM = {
			OR={
				has_character_modifier=cathayan_link
				AND={
					religion=draconite
					OR = {
						has_dynasty_flag = link_with_cathay
						top_liege = { has_dynasty_flag = link_with_cathay }
					}
				}
			}
		}
	}
	offmap_cathay = {
		OR={
			PREV={
				OR={
					has_artifact = dark_crystal
					religion=draconite
				}
			}
			NOT = { has_policy = cathay_isolationist }
		}
		custom_tooltip = {
			text = cathay_offmap_decisions_enabled_trigger_war_tt # "Is at peace with [From.GetTitledFirstName]!"
			NOT = { governor = { war_with = FROM } }
		}
		governor = {
			show_scope_change = no
			is_within_diplo_range = FROM
			hidden_trigger  = { NOT = { character = FROM } } #Governor cannot use boon/tributes decisions
		}
	}
	custom_tooltip = {
		text = cathay_offmap_boon_decisions_enabled_trigger_raiding_tt
		FROM = {
			NOR = {
				has_character_modifier = cathay_raid_active
				has_character_modifier = cathay_raid_grace_cd
			}
		}
	}
	hidden_trigger = { has_dlc = "Jade Dragon" }
	conditional_tooltip = {
		trigger = { FROM = { primary_title = { has_title_flag = rebel_general_title } } }
		custom_tooltip = {
			text = cathay_offmap_decisions_pretender_emperor_tt
			always = no
		}
	}
	conditional_tooltip = {
		trigger = { FROM = { has_character_modifier = failed_invasion_of_cathay } }
		custom_tooltip = {
			text = cathay_offmap_decisions_failed_invasion_tt
			always = no
		}
	}
}

# For checking the Chinese Emperor's likes & dislikes
# Used by: WHG_cathay_diplomacy_events, 00_offmap_powers +2 more
is_liked_by_cathay_trigger = {
	liked_by_offmap = {
        type = offmap_cathay
    }
    NOT = {
        disliked_by_offmap = {
            type = offmap_cathay
        }
    }
    offmap_cathay = {
        show_scope_change = no
        custom_tooltip = { text = emperor_likes_THIS_tt }
    }
}

}

			culture = cathayan

				OR = {
					region = world_nehekhara_eastern
					region = world_eastern_southlands
					region = world_darklands_plains_of_bones
					region = world_dragon_isles
					region = world_darklands_gnoblar_country
				}
			}
		}
	}
}
