# WHG_elves_triggers.txt: scripted triggers
# Subject: the elves: Ulthuan (Asur), Athel Loren (Asrai), Naggaroth (Druchii)
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Shared: used by three or more files (8)
#   Used by WHG_asur_events (7)
#   Used by WHG_druchii_events (7)
#   Other helpers used by one or two files (6)
###############################################################################

# The host and the guest of a private meeting could marry each other.
# Scope: the guest. Needs the event target asurmeet_host.
# Used by: WHG_asur_events
asurmeet_can_wed_trigger = {
	is_adult = yes
	is_married = no
	is_no_females_race = no
	NOT = { cannot_marry_trigger = yes }
	NOT = { is_close_relative = event_target:asurmeet_host }
	event_target:asurmeet_host = {
		is_adult = yes
		is_married = no
		is_no_females_race = no
		NOT = { cannot_marry_trigger = yes }
	}
	OR = {
		AND = { is_woman_trigger = yes event_target:asurmeet_host = { is_woman_trigger = no } }
		AND = { is_woman_trigger = no event_target:asurmeet_host = { is_woman_trigger = yes } }
	}
}

# A Prince of Ulthuan the other Princes might raise to the Phoenix Throne
# when the heir's line has been burned by Asuryan: a grown elf lord of a
# kingdom, untainted, and not of the burned king's house. The Everqueen
# never takes the Phoenix Crown.
# Scope: a vassal of the burned king, ROOT the burned king (asur.4).
# Used by: WHG_asur_events
asur_phoenix_candidate_trigger = {
	tier = KING
	is_feudal = yes
	is_female = no
	is_adult = yes
	trait = creature_elf
	NOT = { dynasty = ROOT }
	NOT = { trait = asuryan_stain }
	NOT = { trait = everqueen }
}

# Another guest the character could befriend over a late-night talk.
# Scope: a vassal of the host. ROOT: the guest looking for a friend.
# Used by: WHG_druchii_events
drufeast_friend_candidate_trigger = {
	has_character_flag = attending_druchii_feast
	NOT = { character = ROOT }
	is_adult = yes
	NOT = { num_of_friends = 3 }
	NOT = { is_friend = ROOT }
	NOT = { is_rival = ROOT }
	opinion = { who = ROOT value = 0 }
	NOT = { any_claim = { holder_scope = { character = ROOT } } }
	ROOT = { NOT = { any_claim = { holder_scope = { character = PREVPREV } } } }
}

# Another guest the character could challenge to a duel: someone they
# dislikes or hates, or anyone at all if they are a duelist.
# Scope: a vassal of the host. ROOT: the challenger.
# Used by: WHG_druchii_events
drufeast_duel_candidate_trigger = {
	has_character_flag = attending_druchii_feast
	NOT = { character = ROOT }
	is_adult = yes
	prisoner = no
	NOT = { trait = wounded }
	NOT = { trait = maimed }
	NOT = { trait = incapable }
	NOT = { is_friend = ROOT }
	OR = {
		is_rival = ROOT
		NOT = { reverse_opinion = { who = ROOT value = 0 } }
		ROOT = { trait = duelist }
	}
}

# Another guest's spouse who might accept a night with the character.
# Scope: the spouse. ROOT: the seducer.
# Used by: WHG_druchii_events
drufeast_seduction_target_trigger = {
	prisoner = no
	is_adult = yes
	is_alive = yes
	is_pregnant = no
	is_abroad = no
	NOT = { trait = celibate }
	NOT = { trait = eunuch }
	NOT = { trait = incapable }
	opinion = { who = ROOT value = 20 }
	NOT = { is_close_relative = ROOT }
	OR = {
		AND = { is_female = yes ROOT = { is_female = no } }
		AND = { is_female = no ROOT = { is_female = yes } }
	}
}

# A lesser courtier that a schemer might poison to rise at court: unlanded,
# not on the council, not clever or strong enough to be dangerous, and no
# friend, lover or close kin of the schemer.
# Scope: a courtier of the schemer's liege, ROOT the schemer (druchii.52).
# Used by: WHG_druchii_events
druchii_herd_victim_trigger = {
	NOT = { character = ROOT }
	is_landed = no
	is_councillor = no
	prisoner = no
	age = 50
	NOR = {
		intrigue = 12
		martial = 12
		is_friend = ROOT
		is_lover = ROOT
		is_close_relative = ROOT
		is_invincible_trigger = yes
	}
}

# A courtier who could become the rival of an ambitious courtier.

druchii_rival_candidate_trigger = {

	culture_group = druchii_group
	NOT = { num_of_rivals = 3 }
	NOR = {
		is_rival = ROOT
		is_friend = ROOT
		is_lover = ROOT
	}
}

# A Sorceress free to be hired by another court.
# Scope: the Sorceress (hire_sorceress, druchii.85, 86).
# Used by: WHG_druchii_events, wh_druchii_decisions
druchii_sorceress_for_hire_trigger = {
	trait = druchii_sorceress
	is_landed = no
	is_married = no
	is_incapacitated_trigger = no
	NOT = { has_character_modifier = magic_power_4 }
	NOT = { has_character_flag = hired_sorceress }
}

# The suitor can afford a show of wealth worthy of the beloved's house.
# Scope: the suitor, FROM the beloved.
# Used by: WHG_elven_marriage_events
elfwed_can_afford_gold_trigger = {
	OR = {
		AND = { FROM = { elfwed_house_emperor_trigger = yes } wealth = 750 }
		AND = {
			FROM = { elfwed_house_emperor_trigger = no elfwed_house_king_trigger = yes }
			wealth = 450
		}
		AND = {
			FROM = { elfwed_house_royal_trigger = no elfwed_house_duke_trigger = yes }
			wealth = 300
		}
		AND = {
			FROM = {
				elfwed_house_royal_trigger = no
				elfwed_house_duke_trigger = no
				elfwed_house_count_trigger = yes
			}
			wealth = 150
		}
		AND = {
			FROM = {
				elfwed_house_royal_trigger = no
				elfwed_house_duke_trigger = no
				elfwed_house_count_trigger = no
				elfwed_house_baron_trigger = yes
			}
			wealth = 100
		}
		AND = {
			FROM = {
				elfwed_house_royal_trigger = no
				elfwed_house_duke_trigger = no
				elfwed_house_count_trigger = no
				elfwed_house_baron_trigger = no
			}
			wealth = 50
		}
	}
}
