# Verena and Morr

# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.

#   Shared: used by three or more files (42)

# Visible requirements for co-founding or joining leagues.
# Used by: WHG_realm_events, wh_empire_decisions, wh_general_decisions
THIS_can_join_PREV_league_trigger = {
	# Not checked here: rank requirements (different for initial creation vs later additions)
	# Not checked here: adjacency requirements (graph traversal for initial creation)

	# Same religion group
	custom_tooltip = {
		text = SAME_RELIGION_GROUP  # Less confusing than e.g. "Is Christian"
		religion_group = PREV
	}

	trait = creature_human
	capital_scope = { region =  world_old_world_border_princes }

	is_primary_war_defender = no

	# Ally or potential ally
	custom_tooltip = {
		text = ELECTIVE_LEAGUE_ALLY_ETC
		OR = {
			is_allied_with = PREV                # Strong ally
			has_non_aggression_pact_with = PREV  # Potential ally, even if the AI hasn't bothered
			is_close_relative = PREV             # Potential ally, even if the AI hasn't bothered
			pays_tribute_to = PREV               # Weak ally
			is_friend = PREV                     # Weak ally
		}
	}

	# We like PREV enough to voluntarily create an alliance
	opinion = { who = PREV  value = 26 }
}

# A university city is open to students: its university has been founded,
# and its lord is a man, and no servant of Chaos or of the dead. Scope: the
# city's province.
# Used by: WHG_empire_effects, WHG_empire_events, WHG_university_decisions

	# Salzenmund, Marienburg and Sudenburg have no university until their lord
	# founds one

	OR = { NOT = { province_id = 222 } has_global_flag = university_founded_marienburg }
	OR = { NOT = { province_id = 983 } has_global_flag = university_founded_sudenburg }
	owner = {
		trait = creature_human
		NOT = { religion_group = chaos_gods_group }
		NOT = { religion_group = necromantic_group }
	}
}

# The ruler has room for another court witch hunter: emperors keep up to
# five, kings three, everyone else one. Each court title's allowed_to_grant
# uses the same caps.
# Scope: a ruler.
# Used by: WHG_witch_hunter_events, WHG_witch_chapter_decisions +1 more
whunter_court_hunter_seat_empty_trigger = {
	whunter_can_keep_court_hunter_trigger = yes
	is_state_witch_hunter_trigger = no
	trigger_if = {
		limit = { tier = EMPEROR }
		has_assigned_minor_title = { title = title_court_witchhunter count < 5 }
		has_assigned_minor_title = { title = title_court_inquisitor count < 5 }
		has_assigned_minor_title = { title = title_court_templar count < 5 }
		has_assigned_minor_title = { title = title_court_chekist count < 5 }
	}
	trigger_else = {
		trigger_if = {
			limit = { tier = KING }
			has_assigned_minor_title = { title = title_court_witchhunter count < 3 }
			has_assigned_minor_title = { title = title_court_inquisitor count < 3 }
			has_assigned_minor_title = { title = title_court_templar count < 3 }
			has_assigned_minor_title = { title = title_court_chekist count < 3 }
		}
		trigger_else = {
			has_assigned_minor_title = { title = title_court_witchhunter count < 1 }
			has_assigned_minor_title = { title = title_court_inquisitor count < 1 }
			has_assigned_minor_title = { title = title_court_templar count < 1 }
			has_assigned_minor_title = { title = title_court_chekist count < 1 }
		}
	}
}

# Master scope; FROM is a ruler. True if FROM owes money to this master's
# order (targeted decisions: ROOT is the master).
# Used by: WHG_holy_order_decisions
holy_order_owed_by_from_trigger = {
	FROM = {
		OR = {
			AND = {
				has_character_modifier = borrowed_from_holy_order
				persistent_event_target:holy_order_creditor = {
					holder_scope = { character = ROOT }
				}
			}
			AND = {
				has_character_modifier = borrowed_from_holy_order_b
				persistent_event_target:holy_order_creditor_b = {
					holder_scope = { character = ROOT }
				}
			}
		}
	}
}

# Ruler scope. True if the ruler can take on another loan from a holy
# order: anyone may owe one order; the greedy, the ambitious, the hedonist
# and the decadent may owe two.
# Used by: WHG_holy_order_decisions
holy_order_can_borrow_trigger = {
	OR = {
		NOT = { has_character_modifier = borrowed_from_holy_order }
		AND = {
			NOT = { has_character_modifier = borrowed_from_holy_order_b }
			OR = {
				trait = greedy
				trait = ambitious
				trait = hedonist
				trait = decadent
			}
		}
	}
}

# Who may join a College of Magic (every college society, imperial and
# dynamic): a human with the gift or an apprentice, in the Old World, settled,
# not Bretonnian, vampire or Chaos. Players need only learning 5.
# Scope: a character.
# Used by: WHG_magic_societies
college_can_join_trigger = {
	OR = {
		ai = no
		hidden_or_open_magic_potential_trigger = yes
		trait = mage_apprentice
	}
	learning = 5
	OR = {
		is_human = yes
		has_character_flag = founder_of_college_of_magic
	}
	OR = {
		location = { region = world_old_world }
		top_liege = { location = { region = world_old_world } }
	}
	is_bretonnian = no
	is_vampire_trigger = no
	open_follower_of_any_chaos_gods = no
	NOR = {
		is_nomadic = yes
		is_tribal = yes
	}
	hidden_trigger = { mercenary = no }
	is_adult = yes
	hidden_trigger = {
		NOT = { has_character_flag = society_join_block }
		NOT = { has_character_flag = taken_necromancer_decision }
	}
	trigger_if = {
		limit = { prisoner = yes }
		prisoner = no
	}
}

# Fewer holders of this court hunter title than the ruler's tier allows
# (emperor 5, king 3, others 1). Used by the titles' allowed_to_grant too.
# Scope: a ruler. Parameter: $title$.
whunter_court_hunter_cap_trigger = {
	trigger_if = {
		limit = { tier = EMPEROR }
		has_assigned_minor_title = { title = $title$ count < 5 }
	}
	trigger_else = {
		trigger_if = {
			limit = { tier = KING }
			has_assigned_minor_title = { title = $title$ count < 3 }
		}
		trigger_else = { has_assigned_minor_title = { title = $title$ count < 1 } }
	}
}
