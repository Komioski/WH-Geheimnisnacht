# WHG_faith_triggers.txt: scripted triggers
# Subject: faith: piety, heresy, saints, holy wars, doctrines, monks,
# coronations, competitions of the faiths
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Shared: used by three or more files (30)
#   Used by WHG_coronation_events (4)
#   Used by WHG_crusade_events (6)
#   Used by WHG_faith_effects (6)
#   Building blocks of other entries in this file (13)
#   Used by WHG_festival_events (11)
#   Used by WHG_monastic_events (11)
#   Used by WHG_northern_rites_events (8)
#   Used by WHG_patron_god_events (4)
#   Used by WHG_saint_events (18)
#   Used by WHG_seeress_events (5)
#   Used by WHG_sway_events (7)
#   Other helpers used by one or two files (33)
#   Not used by any live script yet: kept for content to come (1)
###############################################################################

# Miragliano, Remas or Luccini, under an independent ruler the diverted
# Errantry War may sack. Scope: a province.
# Used by: WHG_crusade_events
crusade_diverted_port_trigger = {
	OR = {
		province_id = 584 # Miragliano
		province_id = 600 # Remas
		province_id = 614 # Luccini
	}
	owner = {
		OR = {
			AND = { independent = yes crusade_diverted_target_ok_trigger = yes }
			AND = { independent = no top_liege = { crusade_diverted_target_ok_trigger = yes } }
		}
	}
}
