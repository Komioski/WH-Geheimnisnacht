# Scope: a daughter. Strong-willed enough to defy her parent: proud, brave,
# clever or otherwise headstrong, and not meek, dull or easily led.
daughter_strong_willed_trigger = {
	is_no_females_race = no
	OR = {
		trait = possessed
		is_smart_trigger = yes
		trait = celibate
		trait = diligent
		trait = wroth
		trait = proud
		trait = brave
		trait = ambitious
		trait = homosexual
	}
	NOR = {
		is_dumb_trigger = yes
		trait = inbred
		trait = humble
		trait = slothful
		trait = craven
		trait = shy
		trait = content
		trait = trusting
		trait = blinded
	}
}

# A landed kinsman with no children to inherit, old enough to be thinking of
# the end, and holding land the way his house can share: by blood, not by
# vote.
# Scope: the landed kinsman. ROOT: the kinsman who would ask.
dynreq_childless_elder_trigger = {
	is_landed = yes
	is_ruler = yes
	is_offmap_ruler = no
	age = 35
	prisoner = no
	NOT = { num_of_children = 1 }
	NOT = { character = ROOT }
	NOT = { has_law = succ_feudal_elective }
	NOT = { is_rival = ROOT }
	OR = {
		is_feudal = yes
		is_tribal = yes
		is_nomadic = yes
	}
	NOT = { any_demesne_title = { claimed_by = ROOT } }
}

# A claim the dynasty head could press in a war for a landless kinsman: held
# by an AI ruler outside the head's realm, not a nomad, rebel, ally,
# tributary of the head or pact partner (there is no war notice for a
# player).
# Scope: the claimed title. ROOT: the dynasty head.
dynreq_war_claim_trigger = {
	has_holder = yes
	holder_scope = {
		ai = yes
		is_nomadic = no
		in_revolt = no
		NOT = { same_realm = ROOT }
		NOT = { has_non_aggression_pact_with = ROOT }
		NOT = { is_allied_with = ROOT }
		NOT = {
			is_tributary = yes
			suzerain = {
				OR = {
					character = ROOT
					any_liege = { character = ROOT }
				}
			}
		}
		top_liege = {
			is_nomadic = no
			in_revolt = no
			NOT = { has_non_aggression_pact_with = ROOT }
			NOT = { is_allied_with = ROOT }
			NOT = {
				is_tributary = yes
				suzerain = {
					OR = {
						character = ROOT
						any_liege = { character = ROOT }
					}
				}
			}
		}
	}
}

# A landless kinsman of ROOT's court or realm, fit to lead, with a claim
# ROOT could press.
# Scope: the kinsman. ROOT: the dynasty head.
dynreq_war_claimant_trigger = {
	is_offmap_ruler = no
	is_landed = no
	is_incapable = no
	is_adult = yes
	prisoner = no
	is_liege_or_above = ROOT
	OR = {
		martial = 12
		diplomacy = 12
	}
	any_claim = { dynreq_war_claim_trigger = yes }
}
