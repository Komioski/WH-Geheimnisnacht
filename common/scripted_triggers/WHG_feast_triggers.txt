# WHG_feast_triggers.txt - scripted triggers for WHG_feast_events (feast)
###############################################################################

# A lord whose feast is under way and who is free to host it.
# Scope: the host.
feast_host_trigger = {
	has_character_modifier = holding_large_feast
	has_character_flag = host_feast_started
	war = no
	prisoner = no
}

# A vassal sitting at the feast of their liege.
# Scope: the guest.
feast_guest_trigger = {
	has_character_flag = guest_feast_started
	war = no
	prisoner = no
	NOT = { trait = incapable }
	liege = { feast_host_trigger = yes }
}

# A guest who has room for a friend, and another guest at the same table who
# could be that friend.
# Scope: the first guest.
feast_befriend_pair_trigger = {
	is_adult = yes
	NOT = { num_of_friends = 3 }
	liege = {
		any_vassal = {
			feast_guest_trigger = yes
			is_adult = yes
			NOT = { num_of_friends = 3 }
			NOT = { character = PREVPREV }
			NOT = { is_friend = PREVPREV }
			NOT = { is_rival = PREVPREV }
		}
	}
}

# Two guests who both resent the host and find each other over the wine.
# Scope: the first guest.
feast_grudge_pair_trigger = {
	is_adult = yes
	NOT = { num_of_friends = 3 }
	liege = {
		NOT = { reverse_opinion = { who = PREV value = 0 } }
		any_vassal = {
			feast_guest_trigger = yes
			is_adult = yes
			NOT = { num_of_friends = 3 }
			NOT = { character = PREVPREV }
			NOT = { is_friend = PREVPREV }
			NOT = { opinion = { who = PREV value = 0 } }
		}
	}
}

# A fighting guest with a grudge against another guest: they are rivals, or
# either thinks poorly of the other.
# Scope: the challenger.
feast_duelist_trigger = {
	is_adult = yes
	martial = 9
	NOT = { trait = wounded }
	NOT = { trait = craven }
	can_be_maimed_trigger = yes
	OR = {
		trait = duelist
		trait = brave
		trait = wroth
		trait = cruel
	}
	liege = {
		any_vassal = {
			feast_duel_foe_trigger = yes
		}
	}
}

# The guest the challenger means to fight. Called inside the liege's
# any_vassal or random_vassal, so PREVPREV is the challenger.
# Scope: the foe.
feast_duel_foe_trigger = {
	feast_guest_trigger = yes
	is_adult = yes
	NOT = { character = PREVPREV }
	NOT = { trait = wounded }
	can_be_maimed_trigger = yes
	OR = {
		is_rival = PREVPREV
		NOT = { opinion = { who = PREVPREV value = -25 } }
		NOT = { reverse_opinion = { who = PREVPREV value = -25 } }
	}
}

# A hot-blooded petty lord who might cut down a servant at table.
# Scope: the guest.
feast_cruel_guest_trigger = {
	feast_guest_trigger = yes
	OR = {
		trait = wroth
		trait = cruel
		trait = impaler
	}
	OR = {
		tier = BARON
		tier = COUNT
	}
	NOT = { trait = just }
	liege = {
		location = { NOT = { has_province_modifier = peasants_upset } }
	}
}

# A married ruler at the feast who could be drawn to another guest's spouse.
# Guarded against the peoples without women.
# Scope: the guest.
feast_wandering_eye_trigger = {
	feast_guest_trigger = yes
	is_ruler = yes
	is_married = yes
	is_adult = yes
	NOT = { age = 60 }
	fertility = 0.4
	is_no_females_race = no
	NOR = {
		trait = celibate
		trait = chaste
		trait = eunuch
	}
}

# Another guest's spouse who is free and young enough to answer a flirt. The
# event checks the sexes and the spouse's opinion of the flirt itself.
# Scope: the spouse.
feast_flirt_spouse_trigger = {
	is_adult = yes
	is_alive = yes
	is_abroad = no
	prisoner = no
	is_pregnant = no
	NOT = { age = 40 }
	is_no_females_race = no
	NOR = {
		trait = celibate
		trait = chaste
		trait = eunuch
	}
}

# The host, a man of a people with women, who might court a lady guest or a
# guest's wife.
# Scope: the host.
feast_host_eye_trigger = {
	feast_host_trigger = yes
	is_female = no
	is_adult = yes
	NOT = { age = 60 }
	fertility = 0.4
	is_no_females_race = no
	NOR = {
		trait = celibate
		trait = chaste
		trait = eunuch
		trait = homosexual
	}
}

# A lady at the feast (a guest, or a guest's wife) who is fond of the host.
# Scope: the lady. ROOT is the host.
feast_host_lady_trigger = {
	is_woman_trigger = yes
	is_adult = yes
	is_alive = yes
	is_abroad = no
	prisoner = no
	is_pregnant = no
	NOT = { age = 40 }
	fertility = 0.4
	NOT = { is_close_relative = ROOT }
	opinion = { who = ROOT value = 20 }
	NOR = {
		trait = celibate
		trait = chaste
		trait = homosexual
	}
}

# An unwed young daughter of a guest. ROOT is the lovestruck host or host's
# son.
# Scope: the daughter.
feast_guest_daughter_trigger = {
	is_woman_trigger = yes
	is_adult = yes
	NOT = { age = 26 }
	is_married = no
	is_betrothed = no
	is_pregnant = no
	has_lover = no
	is_abroad = no
	prisoner = no
	fertility = 0.4
	NOT = { is_close_relative = ROOT }
	opinion = { who = ROOT value = 20 }
}

# The host or the host's young son, free to lose his heart at the feast.
# Scope: the lovestruck man.
feast_lovestruck_trigger = {
	is_female = no
	is_adult = yes
	is_no_females_race = no
	has_lover = no
	is_married = no
	is_betrothed = no
	NOR = {
		trait = shy
		trait = celibate
		trait = chaste
		trait = eunuch
		trait = homosexual
	}
	OR = {
		feast_host_trigger = yes
		AND = {
			NOT = { age = 26 }
			is_abroad = no
			prisoner = no
			father = { feast_host_trigger = yes }
		}
	}
}

# A married guest with an eye for another guest's spouse of the other sex,
# who is fond of them. Guarded against the peoples without women.
# Scope: the guest who flirts (the spouse is reached through liege, the other
# guest and spouse, so the flirt is PREVPREVPREV there).
feast_guest_eye_trigger = {
	feast_wandering_eye_trigger = yes
	liege = {
		any_vassal = {
			feast_guest_trigger = yes
			NOT = { character = PREVPREV }
			spouse = {
				feast_flirt_spouse_trigger = yes
				NOT = { character = PREVPREVPREV }
				NOT = { is_close_relative = PREVPREVPREV }
				opinion = { who = PREVPREVPREV value = 20 }
				OR = {
					AND = {
						is_woman_trigger = yes
						PREVPREVPREV = { is_female = no }
					}
					AND = {
						is_female = no
						PREVPREVPREV = { is_woman_trigger = yes }
					}
				}
			}
		}
	}
}
