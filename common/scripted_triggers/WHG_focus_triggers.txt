# Triggers for the Way of Life focus events (events/WHG_<focus>_events.txt:
# theology, scholarship, rulership, family, carousing, hunting, business,
# intrigue, war, and seduction with the lovers). One section per focus.

###############################################################################
# Theology focus
###############################################################################

# A faith with written scripture and an ordered priesthood, studied from its
# holy books. Every other faith keeps its lore in tales told by its priests,
# shamans and seers (theology.1-2, theology.50).
# Scope: a character.
theology_scripture_faith_trigger = {
	OR = {
		religion = sigmarite
		religion = ulrican
		religion = taalite
		religion = morrian
		religion = manannite
		religion = myrmidian
		religion = shallyan
		religion = grail
		religion = false_grail
		religion = kislev_gods
		religion = ancestor_gods
		religion = cadai
		religion = cytharai
		religion = atharti
		religion = hashut
		religion = ormazd
		religion = kitabid
		religion = nehekharan_gods
		religion = draconite
	}
}

# A faith whose priests teach that their gods may be studied side by side
# with those of a neighbour (theology.40). Chaos, the greenskins, the Horned
# Rat, the Great Maw and the beast cults never share their gods.
# Scope: a character.
theology_tolerant_faith_trigger = {
	OR = {
		religion_group = old_world_group
		religion_group = ormazd_gods_group
		religion_group = cadai_group
		religion_group = cytharai_group
		religion_group = dwarf_gods_group
		religion_group = halfling_gods_group
		religion_group = oriental_gods_group
	}
}

# A faith that holds lust a sin worth curbing in seclusion (theology.91).
# Scope: a character.
theology_chaste_faith_trigger = {
	NOR = {
		religion = slaanesh
		religion = chaos
		religion = atharti
	}
}

###############################################################################
# Scholarship focus
###############################################################################

# A faith that reads the will of its gods in the sky and sees no heresy in
# the observatory: Tzeentch, the Old Ones, Cathay's dragons, Nehekhara's
# star priests and the Asur loremasters of Hoeth (scholarship.4, 10, 17).
# Scope: a character.
scholarship_star_faith_trigger = {
	OR = {
		religion = tzeentch
		religion = nehekharan_gods
		religion = cadai
		religion_group = old_ones_group
		religion_group = oriental_gods_group
	}
}

# A faith with its own voice under the stars: Tzeentch, the Slann and the
# Dawi (scholarship.3, 28).
# Scope: a character.
scholarship_star_voice_trigger = {
	OR = {
		religion = tzeentch
		religion_group = old_ones_group
		religion_group = dwarf_gods_group
	}
}

# The stargazer cannot study now: jailed, incapable or no longer on the
# scholarship focus. The observatory chain waits (scholarship.4).
# Scope: the stargazer.
scholarship_stargazer_busy_trigger = {
	OR = {
		prisoner = yes
		trait = incapable
		NOT = { has_focus = focus_scholarship }
	}
}

# A zealous AI relative or friend at the stargazer's court who would beg
# them to stop (scholarship.12, 22).
# Scope: the courtier; ROOT is the stargazer.
scholarship_worried_kin_trigger = {
	ai = yes
	is_adult = yes
	trait = zealous
	at_location = ROOT
	scholarship_star_faith_trigger = no
	OR = {
		mother = { character = ROOT }
		father = { character = ROOT }
		any_child = { character = ROOT }
		is_friend = ROOT
		AND = {
			any_sibling = { character = ROOT }
			NOT = { is_rival = ROOT }
		}
	}
}

###############################################################################
# Rulership focus
###############################################################################

# A people that holds courts of law where peasants and lords plead before
# their ruler. Greenskins, the Skaven and the ogre tribes settle such things
# with teeth (rulership.30, 40, 52 - 54).
# Scope: the ruler.
rulership_court_of_law_trigger = {
	NOR = {
		religion_group = greenskin_gods_group
		religion_group = horned_rat_god_group
		religion_group = great_maw_group
	}
}

# A faith that tries women for witchcraft: the gods of the Old World (but
# not Kislev, which honours its ice witches), the Moot and Araby
# (rulership.60).
# Scope: the ruler.
rulership_witch_trial_faith_trigger = {
	OR = {
		religion_group = old_world_group
		religion_group = halfling_gods_group
		religion_group = ormazd_gods_group
	}
	NOT = { religion = kislev_gods }
}

# An AI vassal fit to plead at court: adult, free, capable, and neither a
# close relative nor a spouse of the ruler (rulership.50 - rulership.54).
# Scope: the vassal; ROOT is the ruler.
rulership_plaintiff_vassal_trigger = {
	ai = yes
	is_adult = yes
	prisoner = no
	is_incapable = no
	NOT = { is_close_relative = ROOT }
	NOT = { any_spouse = { character = ROOT } }
}

# This vassal and PREVPREV (another vassal) are unrelated and at least one
# dislikes the other (rulership.50).
# Scope: a vassal; PREVPREV is the other vassal.
rulership_feuding_with_prevprev_trigger = {
	NOT = { character = PREVPREV }
	NOT = { is_close_relative = PREVPREV }
	NOT = { any_spouse = { character = PREVPREV } }
	OR = {
		NOT = { opinion = { who = PREVPREV value = 0 } }
		NOT = { reverse_opinion = { who = PREVPREV value = 0 } }
	}
}

###############################################################################
# Family focus
###############################################################################

# The ruler is at home and well enough for family life: no regent, not
# leading an army, not ill (family.10 - family.60).
# Scope: the ruler.
family_free_for_kin_trigger = {
	has_regent = no
	in_command = no
	is_ill = no
}

# A spouse the ruler can spend time with: alive, free, capable, not
# already a lover or a rival (family.10).
# Scope: the spouse; ROOT is the ruler.
family_spouse_at_hand_trigger = {
	is_alive = yes
	prisoner = no
	NOT = { trait = incapable }
	NOT = { is_lover = ROOT }
	NOT = { is_rival = ROOT }
}

# A sibling or child to spend time with: alive, free, of sound mind, and
# either thirteen or older or a child of four or more at the ruler's court
# (family.20, family.40).
# Scope: the relative; ROOT is the ruler.
family_kin_to_visit_trigger = {
	is_alive = yes
	prisoner = no
	NOR = {
		trait = incapable
		trait = inbred
		trait = imbecile
	}
	OR = {
		age = 13
		AND = {
			age = 4
			host = { character = ROOT }
		}
	}
}

# A parent who is not yet fond of the ruler (family.30).
# Scope: the parent; ROOT is the ruler.
family_cool_parent_trigger = {
	is_alive = yes
	prisoner = no
	NOT = { trait = incapable }
	NOT = { personal_opinion = { who = ROOT value = 50 } }
}

# Of the sex the ruler's succession law favours, or the ruler's heir: the
# kin whose friendships matter for the house (family.50 - family.60). The
# ruler is event_target:target_initiator, so the trigger also works from a
# guest's scope (family.51).
# Scope: the relative.
family_favoured_kin_trigger = {
	OR = {
		event_target:target_initiator = { has_law = true_cognatic_succession }
		is_primary_heir = event_target:target_initiator
		AND = {
			is_female = no
			event_target:target_initiator = {
				OR = {
					has_law = agnatic_succession
					has_law = cognatic_succession
				}
			}
		}
		AND = {
			is_woman_trigger = yes
			event_target:target_initiator = {
				NOR = {
					has_law = agnatic_succession
					has_law = cognatic_succession
				}
			}
		}
	}
}

# An adult relative who could be invited to a family gathering (family.50,
# family.51).
# Scope: the relative; event_target:target_initiator is the ruler.
family_reconcilable_kin_trigger = {
	is_offmap_ruler = no
	is_adult = yes
	is_alive = yes
	prisoner = no
	NOR = {
		trait = incapable
		trait = inbred
		trait = imbecile
	}
	family_favoured_kin_trigger = yes
}

# A young child of the ruler at court, five or older (family.60).
# Scope: the child; ROOT and event_target:target_initiator are the ruler.
family_young_sibling_trigger = {
	is_adult = no
	is_alive = yes
	age = 5
	prisoner = no
	NOR = {
		trait = incapable
		trait = inbred
		trait = imbecile
	}
	host = { character = ROOT }
	family_favoured_kin_trigger = yes
}

# Carousing focus
###############################################################################

# The unliving: vampires, Tomb Kings and daemons neither eat, drink nor
# tire. Their fountains, feasts and hunts take their own paths (business.63,
# carousing.16, carousing.60, hunting.11).
# Scope: a character.
focus_unliving_trigger = {
	OR = {
		is_undead_trigger = yes
		is_daemon_trigger = yes
	}
}

# A Tomb King or other risen Nehekharan: undead, but no vampire.
# Scope: a character.
focus_tomb_king_trigger = {
	is_undead_trigger = yes
	is_vampire_trigger = no
}

# A daemon who is not also undead.
# Scope: a character.
focus_daemon_host_trigger = {
	is_daemon_trigger = yes
	is_undead_trigger = no
}

# A living mortal guest at ROOT's feast, fit to be prey or plaything
# (carousing.65, carousing.67, carousing.69). AI only, like the cannibal.
# Scope: a realm character; ROOT is the host.
carousing_living_prey_trigger = {
	ai = yes
	carousing_at_party_trigger = yes
	focus_unliving_trigger = no
}

# The strict faiths of Araby, whose faithful shun the wine cup unless they
# are already lost to it (carousing.1).
# Scope: a character.
carousing_temperate_faith_trigger = {
	OR = {
		religion = ormazd
		religion = kitabid
	}
}

# A guest who accepted ROOT's invitation (carousing.10 - carousing.14,
# carousing.61).
# Scope: a realm character; ROOT is the host.
carousing_guest_trigger = {
	has_opinion_modifier = { modifier = opinion_carousing who = ROOT }
}

# A guest at ROOT's feast now running: the party_guest flag first, so the
# realm-wide checks stay cheap (carousing.16 - carousing.46).
# Scope: a realm character; ROOT is the host.
carousing_at_party_trigger = {
	has_character_flag = party_guest
	has_opinion_modifier = { modifier = opinion_carousing who = ROOT }
}

# Mad, possessed or an impaler, or owning the Omnivore: someone who might
# eat a sleeping guest (carousing.16, carousing.20).
# Scope: a character.
carousing_maniac_trigger = {
	OR = {
		trait = lunatic_1
		trait = lunatic_2
		trait = lunatic_3
		trait = possessed
		trait = impaler
		any_artifact = { artifact_type = the_omnivore }
	}
}

# Lustful or hedonist (carousing.16, carousing.25).
# Scope: a character.
carousing_lustful_trigger = {
	OR = {
		trait = lustful
		trait = hedonist
	}
}

# A guest who would share ROOT's bed: both of peoples with women, and of
# matching tastes (carousing.16 - carousing.26).
# Scope: a guest; ROOT is the host.
carousing_lust_match_trigger = {
	is_no_females_race = no
	ROOT = { is_no_females_race = no }
	OR = {
		AND = {
			is_opposite_sex = ROOT
			NOT = { trait = homosexual }
			ROOT = { NOT = { trait = homosexual } }
		}
		AND = {
			same_sex = ROOT
			trait = homosexual
			ROOT = { trait = homosexual }
		}
	}
}

# Wrathful, mad or paranoid: someone who might start a brawl
# (carousing.16, carousing.35).
# Scope: a character.
carousing_brawler_trigger = {
	OR = {
		trait = wroth
		trait = lunatic_1
		trait = lunatic_2
		trait = lunatic_3
		trait = paranoid
	}
}

# Mad (carousing.16, carousing.40).
# Scope: a character.
carousing_lunatic_trigger = {
	OR = {
		trait = lunatic_1
		trait = lunatic_2
		trait = lunatic_3
	}
}

# Knows the board: a gamer or an apprentice boardgamer (carousing.51 -
# carousing.53).
# Scope: a character.
carousing_board_player_trigger = {
	OR = {
		trait = gamer
		has_character_modifier = apprentice_boardgamer
	}
}

# Hunting focus
###############################################################################

# The White Hart is Taal's own beast to the woodsmen of the Old World, and
# Kurnous' to the Asrai: the faithful should let it run (hunting.2, .12).
# Scope: a character chasing a white beast.
hunting_taal_hart_trigger = {
	has_character_flag = white_stag
	OR = {
		religion = taalite
		religion = asrai_cadai
		religion = old_faith
		religion = old_faith_reformed
	}
}

# The White Bear of the north is Ursun's own child to a Kislevite: to kill
# it is to kill a piece of the Father of Bears (hunting.2, .12).
# Scope: a character chasing a white beast.
hunting_ursun_bear_trigger = {
	has_character_flag = white_bear
	religion = kislev_gods
}

# The White Lion of Chrace: an Asur who brings one down with an axe and
# wears its pelt follows the oldest rite of the Chracian hunters (hunting.2,
# .12).
# Scope: a character chasing a white beast.
hunting_chrace_lion_trigger = {
	has_character_flag = white_lion
	religion = cadai
}

# The faith holds this beast sacred and wants it spared.
# Scope: a character chasing a white beast.
hunting_sacred_beast_trigger = {
	OR = {
		hunting_taal_hart_trigger = yes
		hunting_ursun_bear_trigger = yes
	}
}

# Faiths that shun the hunt: Shallya's faithful heal living things, they
# do not chase them for sport (hunting.1, .6, .30).
# Scope: a character.
hunting_forbidden_faith_trigger = {
	religion = shallyan
}

# Someone at court fit to send a puppy as a gift (hunting.30).
# Scope: the possible giver.
hunting_puppy_giver_trigger = {
	NOT = { trait = incapable }
	prisoner = no
	is_adult = yes
}

# The hound has one of the names given in hunting.33.
# Scope: the hound's master.
hunting_hound_named_trigger = {
	OR = {
		has_character_flag = dog_name_faithful
		has_character_flag = dog_name_hunter
		has_character_flag = dog_name_tricky
		has_character_flag = dog_name_karanak
	}
}

# Business focus
###############################################################################

# A race that keeps markets and makes bargains: not spawn, not the soldiers
# of Khemri, not beastmen, greenskins or daemons (business.1, .22).
# Scope: a character.
business_trading_race_trigger = {
	NOR = {
		trait = creature_chaosspawn
		trait = khemri_soldier
		trait = creature_beastman
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
		culture_group = daemonic_group
	}
}

# A foreign ruler the trading expedition can deal with. The Skaven trade
# only with Skaven.
# Scope: the candidate; ROOT is the ruler sending the expedition.
business_trade_partner_trigger = {
	ai = yes
	NOT = { character = ROOT }
	controls_religion = no
	is_adult = yes
	mercenary = no
	holy_order = no
	business_trading_race_trigger = yes
	OR = {
		ROOT = { trait = creature_skaven }
		NOT = { trait = creature_skaven }
	}
	primary_title = { is_landless_type_title = no }
	NOR = {
		is_rival = ROOT
		trait = incapable
		prisoner = yes
		ROOT = {
			any_liege = { character = PREVPREV }
		}
	}
}

# Goods must go overland: no harbour at the capital, or a realm of deep
# holds, deep woods or wastes (business.12, .33).
# Scope: a province.
business_land_route_trigger = {
	OR = {
		port = no
		region = world_karaz_ankor_main_holds
		region = world_karaz_ankor_northern_holds
		region = world_karaz_ankor_western_holds
		region = world_darklands_blasted_wastes
		region = world_drak_ankor
		region = world_athel_loren
		region = world_chaos_wastes
	}
}

# A burgher of the realm who could buy land, back a guild or open a
# workshop.
# Scope: a holding; ROOT is the ruler.
business_burgher_holding_trigger = {
	holding_type = city
	holder_scope = {
		NOT = { character = ROOT }
		any_liege = { character = ROOT }
	}
}

# A faith whose priests would call a tower taller than the temples an

	}
