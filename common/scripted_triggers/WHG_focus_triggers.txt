# Triggers for the Way of Life focus events (events/WHG_<focus>_events.txt:
# theology, scholarship, rulership, family and the other focus files as they are
# refactored). One section per focus.

###############################################################################
# Theology focus
###############################################################################

# A faith with written scripture and an ordered priesthood, studied from its
# holy books. Every other faith keeps its lore in tales told by its priests,
# shamans and seers (theology.1-2, theology.50).
# Scope: a character.
theology_scripture_faith_trigger = {
	OR = {
		religion = sigmarite
		religion = ulrican
		religion = taalite
		religion = morrian
		religion = manannite
		religion = myrmidian
		religion = shallyan
		religion = grail
		religion = false_grail
		religion = kislev_gods
		religion = ancestor_gods
		religion = cadai
		religion = cytharai
		religion = atharti
		religion = hashut
		religion = ormazd
		religion = kitabid
		religion = nehekharan_gods
		religion = draconite
	}
}

# A faith whose priests teach that their gods may be studied side by side
# with those of a neighbour (theology.40). Chaos, the greenskins, the Horned
# Rat, the Great Maw and the beast cults never share their gods.
# Scope: a character.
theology_tolerant_faith_trigger = {
	OR = {
		religion_group = old_world_group
		religion_group = ormazd_gods_group
		religion_group = cadai_group
		religion_group = cytharai_group
		religion_group = dwarf_gods_group
		religion_group = halfling_gods_group
		religion_group = oriental_gods_group
	}
}

# A faith that holds lust a sin worth curbing in seclusion (theology.91).
# Scope: a character.
theology_chaste_faith_trigger = {
	NOR = {
		religion = slaanesh
		religion = chaos
		religion = atharti
	}
}

###############################################################################
# Scholarship focus
###############################################################################

# A faith that reads the will of its gods in the sky and sees no heresy in
# the observatory: Tzeentch, the Old Ones, Cathay's dragons, Nehekhara's
# star priests and the Asur loremasters of Hoeth (scholarship.4, 10, 17).
# Scope: a character.
scholarship_star_faith_trigger = {
	OR = {
		religion = tzeentch
		religion = nehekharan_gods
		religion = cadai
		religion_group = old_ones_group
		religion_group = oriental_gods_group
	}
}

# A faith with its own voice under the stars: Tzeentch, the Slann and the
# Dawi (scholarship.3, 28).
# Scope: a character.
scholarship_star_voice_trigger = {
	OR = {
		religion = tzeentch
		religion_group = old_ones_group
		religion_group = dwarf_gods_group
	}
}

# The stargazer cannot study now: jailed, incapable or no longer on the
# scholarship focus. The observatory chain waits (scholarship.4).
# Scope: the stargazer.
scholarship_stargazer_busy_trigger = {
	OR = {
		prisoner = yes
		trait = incapable
		NOT = { has_focus = focus_scholarship }
	}
}

# A zealous AI relative or friend at the stargazer's court who would beg
# them to stop (scholarship.12, 22).
# Scope: the courtier; ROOT is the stargazer.
scholarship_worried_kin_trigger = {
	ai = yes
	is_adult = yes
	trait = zealous
	at_location = ROOT
	scholarship_star_faith_trigger = no
	OR = {
		mother = { character = ROOT }
		father = { character = ROOT }
		any_child = { character = ROOT }
		is_friend = ROOT
		AND = {
			any_sibling = { character = ROOT }
			NOT = { is_rival = ROOT }
		}
	}
}

###############################################################################
# Rulership focus
###############################################################################

# A people that holds courts of law where peasants and lords plead before
# their ruler. Greenskins, the Skaven and the ogre tribes settle such things
# with teeth (rulership.30, 40, 52 - 54).
# Scope: the ruler.
rulership_court_of_law_trigger = {
	NOR = {
		religion_group = greenskin_gods_group
		religion_group = horned_rat_god_group
		religion_group = great_maw_group
	}
}

# A faith that tries women for witchcraft: the gods of the Old World (but
# not Kislev, which honours its ice witches), the Moot and Araby
# (rulership.60).
# Scope: the ruler.
rulership_witch_trial_faith_trigger = {
	OR = {
		religion_group = old_world_group
		religion_group = halfling_gods_group
		religion_group = ormazd_gods_group
	}
	NOT = { religion = kislev_gods }
}

# An AI vassal fit to plead at court: adult, free, capable, and neither a
# close relative nor a spouse of the ruler (rulership.50 - rulership.54).
# Scope: the vassal; ROOT is the ruler.
rulership_plaintiff_vassal_trigger = {
	ai = yes
	is_adult = yes
	prisoner = no
	is_incapable = no
	NOT = { is_close_relative = ROOT }
	NOT = { any_spouse = { character = ROOT } }
}

# This vassal and PREVPREV (another vassal) are unrelated and at least one
# dislikes the other (rulership.50).
# Scope: a vassal; PREVPREV is the other vassal.
rulership_feuding_with_prevprev_trigger = {
	NOT = { character = PREVPREV }
	NOT = { is_close_relative = PREVPREV }
	NOT = { any_spouse = { character = PREVPREV } }
	OR = {
		NOT = { opinion = { who = PREVPREV value = 0 } }
		NOT = { reverse_opinion = { who = PREVPREV value = 0 } }
	}
}

###############################################################################
# Family focus
###############################################################################

# The ruler is at home and well enough for family life: no regent, not
# leading an army, not ill (family.10 - family.60).
# Scope: the ruler.
family_free_for_kin_trigger = {
	has_regent = no
	in_command = no
	is_ill = no
}

# A spouse the ruler can spend time with: alive, free, capable, not
# already a lover or a rival (family.10).
# Scope: the spouse; ROOT is the ruler.
family_spouse_at_hand_trigger = {
	is_alive = yes
	prisoner = no
	NOT = { trait = incapable }
	NOT = { is_lover = ROOT }
	NOT = { is_rival = ROOT }
}

# A sibling or child to spend time with: alive, free, of sound mind, and
# either thirteen or older or a child of four or more at the ruler's court
# (family.20, family.40).
# Scope: the relative; ROOT is the ruler.
family_kin_to_visit_trigger = {
	is_alive = yes
	prisoner = no
	NOR = {
		trait = incapable
		trait = inbred
		trait = imbecile
	}
	OR = {
		age = 13
		AND = {
			age = 4
			host = { character = ROOT }
		}
	}
}

# A parent who is not yet fond of the ruler (family.30).
# Scope: the parent; ROOT is the ruler.
family_cool_parent_trigger = {
	is_alive = yes
	prisoner = no
	NOT = { trait = incapable }
	NOT = { personal_opinion = { who = ROOT value = 50 } }
}

# Of the sex the ruler's succession law favours, or the ruler's heir: the
# kin whose friendships matter for the house (family.50 - family.60). The
# ruler is event_target:target_initiator, so the trigger also works from a
# guest's scope (family.51).
# Scope: the relative.
family_favoured_kin_trigger = {
	OR = {
		event_target:target_initiator = { has_law = true_cognatic_succession }
		is_primary_heir = event_target:target_initiator
		AND = {
			is_female = no
			event_target:target_initiator = {
				OR = {
					has_law = agnatic_succession
					has_law = cognatic_succession
				}
			}
		}
		AND = {
			is_woman_trigger = yes
			event_target:target_initiator = {
				NOR = {
					has_law = agnatic_succession
					has_law = cognatic_succession
				}
			}
		}
	}
}

# An adult relative who could be invited to a family gathering (family.50,
# family.51).
# Scope: the relative; event_target:target_initiator is the ruler.
family_reconcilable_kin_trigger = {
	is_offmap_ruler = no
	is_adult = yes
	is_alive = yes
	prisoner = no
	NOR = {
		trait = incapable
		trait = inbred
		trait = imbecile
	}
	family_favoured_kin_trigger = yes
}

# A young child of the ruler at court, five or older (family.60).
# Scope: the child; ROOT and event_target:target_initiator are the ruler.
family_young_sibling_trigger = {
	is_adult = no
	is_alive = yes
	age = 5
	prisoner = no
	NOR = {
		trait = incapable
		trait = inbred
		trait = imbecile
	}
	host = { character = ROOT }
	family_favoured_kin_trigger = yes
}
