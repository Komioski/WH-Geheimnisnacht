# Someone free to make a new friend or rival: not a daemon, not a hunted
# target, at hand and not in chains.
# Scope: the character.
bond_open_trigger = {
	NOT = { culture_group = daemonic_group }
	NOT = { trait = s_target_trait }
	is_inaccessible_trigger = no
	prisoner = no
}

# A child with room for one more friend and one more rival (children keep
# two of each).
# Scope: the child.
bond_child_room_trigger = {
	bond_open_trigger = yes
	NOT = { num_of_friends = 2 }
	NOT = { num_of_rivals = 2 }
}

# A grown-up with room for one more friend and one more rival (five each).
# Scope: the character.
bond_adult_room_trigger = {
	bond_open_trigger = yes
	NOT = { num_of_friends = 5 }
	NOT = { num_of_rivals = 5 }
}

# A child at ROOT's court who is not yet ROOT's friend, rival or sibling.
# Scope: the other child. ROOT is the first child.
bond_child_peer_trigger = {
	bond_child_room_trigger = yes
	NOT = { character = ROOT }
	at_location = ROOT
	NOT = { sibling = ROOT }
	NOT = { is_friend = ROOT }
	NOT = { is_rival = ROOT }
}

# A grown-up at ROOT's court who is not yet ROOT's friend, rival or sibling.
# Scope: the other character. ROOT is the first.
bond_adult_peer_trigger = {
	bond_adult_room_trigger = yes
	NOT = { character = ROOT }
	NOT = { sibling = ROOT }
	NOT = { is_friend = ROOT }
	NOT = { is_rival = ROOT }
}

# A young unwed man, kind or outgoing and fond of women, who could teach a
# shy friend how to talk to them. Only for the peoples with women.
# Scope: the character.
bond_wingman_trigger = {
	bond_adult_room_trigger = yes
	is_female = no
	is_no_females_race = no
	is_married = no
	in_command = no
	war = no
	age = 16
	NOT = { age = 25 }
	trait = lustful
	OR = {
		trait = gregarious
		trait = kind
		trait = charitable
	}
}

# A young unwed man too shy or too chaste to talk to women. Only for the
# peoples with women.
# Scope: the character.
bond_shy_suitor_trigger = {
	bond_adult_room_trigger = yes
	is_female = no
	is_no_females_race = no
	is_married = no
	in_command = no
	war = no
	age = 16
	NOT = { age = 25 }
	OR = {
		trait = chaste
		trait = shy
		trait = paranoid
	}
	NOT = { trait = lustful }
}

# A girl at court with a busy mind: ambitious, diligent or just.
# Scope: the girl.
bond_bright_girl_trigger = {
	bond_child_room_trigger = yes
	is_woman_trigger = yes
	age = 4
	NOT = { age = 14 }
	OR = {
		trait = ambitious
		trait = diligent
		trait = just
	}
}

# A girl at court with a bright girl of her own age to befriend. Called
# inside host = { any_courtier }, so PREVPREV is the first girl.
# Scope: the second girl.
bond_bright_girl_pair_trigger = {
	bond_bright_girl_trigger = yes
	NOT = { character = PREVPREV }
	at_location = PREVPREV
	NOT = { sibling = PREVPREV }
	NOT = { is_friend = PREVPREV }
	NOT = { is_rival = PREVPREV }
}
