# WHG_friends_rivals_triggers.txt - scripted triggers for

# Someone free to make a new friend or rival: not a daemon, not a hunted
# target, at hand and not in chains.
# Scope: the character.
bond_open_trigger = {
	NOT = { culture_group = daemonic_group }
	NOT = { trait = s_target_trait }
	is_inaccessible_trigger = no
	prisoner = no
}

# A child with room for one more friend and one more rival (children keep
# two of each).
# Scope: the child.
bond_child_room_trigger = {
	bond_open_trigger = yes
	NOT = { num_of_friends = 2 }
	NOT = { num_of_rivals = 2 }
}

# A grown-up with room for one more friend and one more rival (five each).
# Scope: the character.
bond_adult_room_trigger = {
	bond_open_trigger = yes
	NOT = { num_of_friends = 5 }
	NOT = { num_of_rivals = 5 }
}

# A child at ROOT's court who is not yet ROOT's friend, rival or sibling.
# Scope: the other child. ROOT is the first child.
bond_child_peer_trigger = {
	bond_child_room_trigger = yes
	NOT = { character = ROOT }
	at_location = ROOT
	NOT = { sibling = ROOT }
	NOT = { is_friend = ROOT }
	NOT = { is_rival = ROOT }
}

# A grown-up at ROOT's court who is not yet ROOT's friend, rival or sibling.
# Scope: the other character. ROOT is the first.
bond_adult_peer_trigger = {
	bond_adult_room_trigger = yes
	NOT = { character = ROOT }
	NOT = { sibling = ROOT }
	NOT = { is_friend = ROOT }
	NOT = { is_rival = ROOT }
}

# A young unwed man, kind or outgoing and fond of women, who could teach a
# shy friend how to talk to them. Only for the peoples with women.
# Scope: the character.
bond_wingman_trigger = {
	bond_adult_room_trigger = yes
	is_female = no
	is_no_females_race = no
	is_married = no
	in_command = no
	war = no
	age = 16
	NOT = { age = 25 }
	trait = lustful
	OR = { trait = gregarious trait = kind trait = charitable }
}

# A young unwed man too shy or too chaste to talk to women. Only for the
# peoples with women.
# Scope: the character.
bond_shy_suitor_trigger = {
	bond_adult_room_trigger = yes
	is_female = no
	is_no_females_race = no
	is_married = no
	in_command = no
	war = no
	age = 16
	NOT = { age = 25 }
	OR = { trait = chaste trait = shy trait = paranoid }
	NOT = { trait = lustful }
}

# A girl at court with a busy mind: ambitious, diligent or just.
# Scope: the girl.
bond_bright_girl_trigger = {
	bond_child_room_trigger = yes
	is_woman_trigger = yes
	age = 4
	NOT = { age = 14 }
	OR = { trait = ambitious trait = diligent trait = just }
}

# A girl at court with a bright girl of her own age to befriend. Called
# inside host = { any_courtier }, so PREVPREV is the first girl.
# Scope: the second girl.
bond_bright_girl_pair_trigger = {
	bond_bright_girl_trigger = yes
	NOT = { character = PREVPREV }
	at_location = PREVPREV
	NOT = { sibling = PREVPREV }
	NOT = { is_friend = PREVPREV }
	NOT = { is_rival = PREVPREV }
}

# A friend or rival free to answer: open to bonds, grown, at peace and not
# leading an army.
# Scope: the friend or rival.
bond_at_hand_trigger = { bond_open_trigger = yes is_adult = yes war = no in_command = no }

# A friend who rules and leads one side of a war that ROOT could join, and
# whose war is not against ROOT's own liege.
# Scope: the friend. ROOT is the one who could help.
friend_war_leader_trigger = {
	is_ruler = yes
	war = yes
	prisoner = no
	is_inaccessible_trigger = no
	NOT = { is_vassal_or_below = ROOT }
	OR = { is_primary_war_attacker = yes is_primary_war_defender = yes }
	NOT = {
		any_war = {
			OR = { attacker = { is_liege_or_above = ROOT } defender = { is_liege_or_above = ROOT } }
		}
	}
}

# A ruling friend at peace who could join ROOT's war or pay for it: neither
# ROOT's vassal nor ROOT's liege, and not in seclusion.
# Scope: the friend. ROOT is at war.
friend_war_ally_trigger = {
	bond_open_trigger = yes
	is_ruler = yes
	war = no
	NOT = { has_character_modifier = in_seclusion }
	NOT = { is_vassal_or_below = ROOT }
	NOT = { is_liege_or_above = ROOT }
}

# A friend of one of ROOT's friends who could become ROOT's friend too.
# Scope: the friend's friend. ROOT is the character.
friend_of_friend_trigger = {
	bond_adult_room_trigger = yes
	is_adult = yes
	war = no
	in_command = no
	NOT = { character = ROOT }
	NOT = { is_close_relative = ROOT }
	NOT = { is_friend = ROOT }
	NOT = { is_rival = ROOT }
	NOT = { is_lover = ROOT }
	NOT = { is_consort = ROOT }
}

# A friend of ROOT's faith family who follows a heresy or another cult.
# Scope: the friend. ROOT is the believer.
friend_heretic_trigger = {
	bond_at_hand_trigger = yes
	NOT = { trait = incapable }
	religion_group = ROOT
	OR = { is_heretic = yes NOT = { religion = ROOT } }
}

# A friend far better than ROOT at diplomacy, and not the chancellor (who
# has no time to spare).

friend_tutor_diplomacy_trigger = {

	NOT = { has_job_title = job_chancellor }

# A friend far better than ROOT with a blade or an army, and not the marshal.

	NOT = { trait = incapable }

	NOT = { has_job_title = job_marshal }

# A friend who rules or sits on a council and keeps far better accounts

# A learned and devout friend of ROOT's own faith.
# Scope: the friend. ROOT asks for help.
friend_tutor_learning_trigger = {
	bond_at_hand_trigger = yes
	learning = 11
	religion = ROOT
	is_heretic = no
	OR = { piety = 100 trait = zealous }
}

# Someone not already asking a friend about the friend's way of life.
# Scope: the character.
friend_way_free_trigger = {
	NOR = {
		has_character_flag = friend_way_celibate
		has_character_flag = friend_way_hedonist
		has_character_flag = friend_way_scholar
	}
}

# A friend ROOT no longer knows: grown cruel, false or wrathful where ROOT
# is not, or simply disliked, unless the two once swore to stay friends.
# Scope: the friend. ROOT is the character.
friend_estranged_trigger = {
	bond_at_hand_trigger = yes
	NOT = { has_opinion_modifier = { who = ROOT modifier = opinion_not_abandon_friendship } }
	OR = {
		AND = { trait = cruel ROOT = { NOT = { trait = cruel } } }
		AND = { trait = deceitful ROOT = { NOT = { trait = deceitful } } }
		AND = { trait = wroth ROOT = { NOT = { trait = wroth } } }
		NOT = { reverse_opinion = { who = ROOT value = 0 } }
	}
}

# A friend who has been hurt by a rival and cannot defend themselves.
# Scope: the friend. ROOT is the would-be champion.
rival_friend_victim_trigger = {
	bond_at_hand_trigger = yes
	num_of_rivals = 1
	NOT = { martial = 10 }
	OR = { trait = wounded is_maimed_trigger = yes }
	any_rival = { rival_bully_trigger = yes }
}

# A friend's rival, whole and unhurt, whom ROOT could take on.
# Scope: the friend's rival. ROOT is the champion.
rival_bully_trigger = {
	bond_at_hand_trigger = yes
	NOT = { character = ROOT }
	NOT = { is_friend = ROOT }
	NOT = { is_rival = ROOT }
	NOT = { num_of_rivals = 5 }
	NOT = { trait = wounded }
	can_be_maimed_trigger = yes
}

# Fit to fight a duel: of age, whole, healthy, at hand, and not a priest
# of Shallya, who never fight.
# Scope: the duellist.
rival_duel_fit_trigger = {
	age = 16
	NOT = { age = 60 }
	prisoner = no
	is_inaccessible_trigger = no
	NOT = { has_character_flag = do_not_disturb }
	NOR = {
		trait = wounded
		trait = incapable
		trait = infirm
		is_maimed_trigger = yes
		religion = shallyan
	}
	is_ill = no
	has_symptom_trigger = no
}

# A rival under the same liege whom ROOT could challenge, and has not
# already fought or fled from.
# Scope: the rival. ROOT is the challenger.
rival_duel_foe_trigger = {
	bond_open_trigger = yes
	rival_duel_fit_trigger = yes
	war = no
	in_command = no
	same_liege = ROOT
	NOR = {
		has_opinion_modifier = { who = ROOT modifier = opinion_fought_duel }
		ROOT = { has_opinion_modifier = { who = PREV modifier = opinion_fought_duel } }
		ROOT = { has_opinion_modifier = { who = PREV modifier = opinion_fled_duel } }
	}
}

# A rival riding in the same tourney as ROOT and still in the lists.
# Scope: the rival. ROOT is a rider.
rival_cheat_trigger = {
	bond_open_trigger = yes
	has_character_flag = tourney_rider
	NOT = { has_character_flag = tourney_out }
	at_location = ROOT
}

# A rival who follows a heresy of ROOT's faith and is not yet cast out.
# Scope: the rival. ROOT is the believer.
rival_heretic_trigger = {
	bond_at_hand_trigger = yes
	is_heresy_of = ROOT
	NOT = { trait = excommunicated }
}

# A rival under the same liege, both in the liege's good graces.
# Scope: the rival. ROOT is the slanderer.
rival_slander_trigger = {
	bond_at_hand_trigger = yes
	same_liege = ROOT
	liege = {
		NOT = { character = ROOT }
		opinion = { who = ROOT value = 10 }
		opinion = { who = PREV value = 10 }
	}
}

# A ruling rival whose love of their own sex would shame them before their
# vassals. Not among the followers of Slaanesh, who think nothing of it.
# Scope: the rival.
rival_scandal_trigger = {
	bond_at_hand_trigger = yes
	is_ruler = yes
	num_of_vassals = 1
	trait = homosexual
	NOT = { religion = slaanesh }
}

# A ruling rival so far beneath ROOT that the rivalry is beneath ROOT too.
# Scope: the rival. ROOT is the greater lord.
rival_beneath_trigger = { bond_at_hand_trigger = yes is_ruler = yes lower_tier_than = ROOT }

# A rival at the liege's grand hunt, whole and unhurt.
# Scope: the rival. ROOT is a fellow hunter.
rival_hunt_trigger = {
	bond_at_hand_trigger = yes
	same_liege = ROOT
	NOT = { trait = wounded }
	can_be_maimed_trigger = yes
	NOT = { has_character_modifier = holding_grand_hunt }
}

# A rival who is a guest at the same feast as ROOT.
# Scope: the rival. ROOT is a guest.
rival_feast_trigger = {
	bond_open_trigger = yes
	same_liege = ROOT
	has_character_flag = guest_feast_started
}

# A rival on whom ROOT has sworn revenge.
# Scope: the rival. ROOT seeks revenge.
rival_revenge_trigger = {
	bond_open_trigger = yes
	reverse_has_opinion_modifier = { who = ROOT modifier = opinion_revenge_seeker }
}

# A rival who killed ROOT's father.
# Scope: the rival. ROOT is the child.
rival_father_killer_trigger = {
	bond_open_trigger = yes
	reverse_has_opinion_modifier = { who = ROOT modifier = opinion_killed_father }
}
