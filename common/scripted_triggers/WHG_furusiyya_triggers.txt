# Triggers for events/WHG_furusiyya_events.txt

# Scope: a character of the host's realm. ROOT is the host (furusiyya.1).
# Can be invited to compete: a free, capable warrior of 16 to 49 who is no
# priest, patrician or mayor, holds land or has some skill at arms. Women
# compete only where the host's faith or laws treat them as equals.
furusiyya_can_compete_trigger = {
	age = 16
	NOT = { age = 50 }
	prisoner = no
	is_incapable = no
	NOT = { has_job_title = job_spiritual }
	NOR = {
		is_theocracy = yes
		is_republic = yes
	}
	OR = {
		is_landed = yes
		martial = 10
		combat_rating = 30
	}
	OR = {
		is_woman_trigger = no
		ROOT = { gender_equality_trigger = yes }
	}
}

# Scope: a guest. Still in the running at a Furusiyya that has begun:
# accepted the invitation, free, capable and not yet dead or maimed out.
furusiyya_guest_trigger = {
	has_character_flag = furusiyya_guest
	prisoner = no
	is_incapable = no
	event_target:furusiyya_host = { has_character_flag = furusiyya_running }
}

# Scope: a champion. May be given a horse or camel from the host's stables:
# an adult with no mount who has never won one here (once per life).
furusiyya_can_win_mount_trigger = {
	is_adult = yes
	NOT = { has_character_flag = furusiyya_prize_mount }
	NOT = { any_artifact = { has_artifact_flag = mount } }
}
