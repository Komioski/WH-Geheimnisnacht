###############################################################################
# WHG_game_rule_triggers.txt: scripted triggers
# Subject: the mod's game rules
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Used by WHG_maintenance_events (5)
#   Other helpers used by one or two files (3)
###############################################################################

###############################################################################
# Used by WHG_maintenance_events
###############################################################################

# The exclave purge game rule is on.
# Used by: WHG_maintenance_events
exclave_purge_on_trigger = {
	OR = {
		has_game_rule = { name = exclave_purge value = limited }
		has_game_rule = { name = exclave_purge value = significant }
		has_game_rule = { name = exclave_purge value = harsh }
		has_game_rule = { name = exclave_purge value = total }
		has_game_rule = { name = exclave_purge value = limited_naval }
	}
}

# The ruler gets a generated family at the start (generate_families rule).
# Scope: the ruler.
# Used by: WHG_maintenance_events
gamerule_family_rule_trigger = {
	OR = {
		AND = {
			is_long_lived_character = no
			has_game_rule = { name = generate_families value = on }
		}
		AND = {
			ai = yes
			is_long_lived_character = no
			has_game_rule = { name = generate_families value = ai_only }
		}
		has_game_rule = { name = generate_families value = elder_races_aswell }
		AND = {
			ai = yes
			has_game_rule = { name = generate_families value = elder_races_ai_aswell }
		}
	}
}

# The peoples who raise families of their own: men, the elder races, Ogres,
# the Dawi Zharr, the Druchii and the Chaos-worshipping tribes. Skaven
# warlords get litters once they are grown.
# Used by: WHG_maintenance_events
gamerule_family_people_trigger = {
	is_vampire_trigger = no
	is_witch_hunter_trigger = no
	NOR = {
		culture_group = fimir_group
		culture_group = daemonic_group
		culture_group = goblin_group
		culture_group = orc_group
		culture_group = nehekharan_group
	}
	OR = {
		AND = {
			culture_group = skaven_group
			age = 25
		}
		religion_group = old_world_group
		religion_group = halfling_gods_group
		religion_group = ormazd_gods_group
		religion_group = chaos_gods_group
		culture_group = ogre_group
		culture_group = asur_group
		culture_group = dwarf_group
		culture_group = dwarf_chaos_group
		culture_group = druchii_group
	}
}

# The ruler has no family to speak of.
# Used by: WHG_maintenance_events
gamerule_family_lonely_trigger = {
	OR = {
		NOT = { num_of_dynasty_members = 2 }
		NOT = {
			any_close_relative = {
				is_alive = yes
				NOR = {
					character = ROOT
					is_child_of = ROOT
				}
			}
		}
		AND = {
			num_of_children >= 1
			NOT = {
				any_child = {
					mother = { always = yes }
				}
			}
			NOT = {
				any_dynasty_member = {
					NOT = { ROOT = { is_father = PREV } }
				}
			}
		}
	}
}

# A duchy or higher title whose holder holds none of its de jure counties
# (gamerule.1). The cults of Sigmar and Ulric may hold their titles anyway.
# Scope: the title. ROOT is the holder.
# Used by: WHG_maintenance_events
gamerule_landless_title_trigger = {
	higher_tier_than = COUNT
	is_titular = no
	is_landless_type_title = no
	NOR = {
		title = k_cult_of_sigmar
		title = k_cult_of_ulric
	}
	NOT = {
		ROOT = {
			any_realm_title = {
				tier = COUNT
				any_dejure_liege = { title = PREVPREVPREV }
			}
		}
	}
}

# The province (or title) still counts as connected to the realm capital under
# the chosen exclave purge rule. Limited allows land gaps and any sea
# crossing, Limited Naval land gaps and short crossings, Significant any sea
# crossing but no land gap, Harsh and Total only short crossings.
# Used by: WHG_game_rule_effects, WHG_maintenance_events
exclave_connected_trigger = {
	OR = {
		AND = {
			has_game_rule = { name = exclave_purge value = limited }
			is_connected_to = { sub_realm = no target = realm_capital land_gap = yes naval_distance = yes }
		}
		AND = {
			has_game_rule = { name = exclave_purge value = limited_naval }
			is_connected_to = { sub_realm = no target = realm_capital land_gap = yes naval_distance = 1000 }
		}
		AND = {
			has_game_rule = { name = exclave_purge value = significant }
			is_connected_to = { sub_realm = no target = realm_capital naval_distance = yes }
		}
		AND = {
			OR = {
				has_game_rule = { name = exclave_purge value = harsh }
				has_game_rule = { name = exclave_purge value = total }
			}
			is_connected_to = { sub_realm = no target = realm_capital naval_distance = 1000 }
		}
	}
}

# As exclave_connected_trigger, but only ROOT's own sub-realm counts as the
# path.
# Used by: WHG_game_rule_effects
exclave_connected_in_realm_trigger = {
	OR = {
		AND = {
			has_game_rule = { name = exclave_purge value = limited }
			is_connected_to = { sub_realm = ROOT target = realm_capital land_gap = yes naval_distance = yes }
		}
		AND = {
			has_game_rule = { name = exclave_purge value = limited_naval }
			is_connected_to = { sub_realm = ROOT target = realm_capital land_gap = yes naval_distance = 1000 }
		}
		AND = {
			has_game_rule = { name = exclave_purge value = significant }
			is_connected_to = { sub_realm = ROOT target = realm_capital naval_distance = yes }
		}
		AND = {
			OR = {
				has_game_rule = { name = exclave_purge value = harsh }
				has_game_rule = { name = exclave_purge value = total }
			}
			is_connected_to = { sub_realm = ROOT target = realm_capital naval_distance = 1000 }
		}
	}
}

# The province lies in ROOT's de jure lands, which every rule but Total
# leaves alone.
# Used by: WHG_game_rule_effects, WHG_maintenance_events
exclave_dejure_kept_trigger = {
	NOT = { has_game_rule = { name = exclave_purge value = total } }
	de_jure_liege_or_above = ROOT
}
