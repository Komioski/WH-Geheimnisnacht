# The game-book stories (events/WHG_gamebook_events.txt)

# The character carries one of the great plagues that kill.
grief_plague_trigger = {
	OR = {
		trait = has_bubonic_plague
		trait = has_small_pox
		trait = has_typhus
		trait = has_creeping_buboes
		trait = has_nurgles_rot
		trait = has_tomb_rot
		trait = has_rattlelung
		trait = has_bone_auge
	}
}

# A friend or kinsman whom the plague may take in grief.1.
# Scope: the dying one.
grief_dying_one_trigger = {
	is_adult = yes
	ai = yes
	grief_plague_trigger = yes
	is_invincible_trigger = no
	is_offmap_ruler = no
}

# A kinsman who may comfort the mourner in grief.4.
# Scope: the kinsman. ROOT is the mourner.
grief_new_friend_trigger = {
	is_adult = yes
	ai = yes
	is_offmap_ruler = no
	NOT = { trait = incapable }
	prisoner = no
	NOT = { has_character_flag = grief_new_friend }
}

# Someone whom the newlywed may fall for in courtly.2.
# Scope: the married one. ROOT is the newlywed.
courtly_love_target_trigger = {
	is_married = yes
	ai = yes
	is_adult = yes
	is_ruler = no
	is_no_females_race = no
	block_general_event_trigger = no
	NOT = { is_close_relative = ROOT }
	NOT = { is_married = ROOT }
	OR = {
		AND = {
			is_woman_trigger = yes
			ROOT = { prefers_men_trigger = no }
		}
		AND = {
			is_woman_trigger = no
			ROOT = { prefers_men_trigger = yes }
		}
	}
	spouse = { ai = yes }
}

# The newlywed saved the married one from the boar in courtly.23.
# Scope: the newlywed.
courtly_saved_trigger = {
	OR = {
		has_character_flag = courtly_shot_hit
		has_character_flag = courtly_charge_won
		has_character_flag = courtly_charge_hurt
	}
}
