# Triggers for events/WHG_grand_debate_events.txt (the Grand Debate of the
# Cult of Verena)

# Scope: a society member; ROOT is an attendee looking for company. Another
# attendee who can be met at the debate.
debate_fellow_trigger = {
	has_character_flag = attending_grand_debate
	NOT = { character = ROOT }
	is_incapable = no
	prisoner = no
	in_command = no
	is_inaccessible_trigger = no
}

# Scope: an attendee. Has a debate score of at least 30 (the strong half).
debate_scored_well_trigger = {
	check_variable = { which = debate_score value = 30 }
}

# Scope: a character; ROOT is the writer. Someone close who would welcome a
# letter from the debate (debate.28).
debate_letter_target_trigger = {
	is_alive = yes
	is_incapable = no
	prisoner = no
	NOT = { is_rival = ROOT }
}
