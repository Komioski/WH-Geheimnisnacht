###############################################################################
# The grand hunt (events/WHG_hunting_events.txt, Part 4, hunting.100 - .199)
#
# hunting_grand_on_trigger        ROOT is out on a grand hunt and fit to ride
# hunting_plotter_trigger         a courtier plotting to kill ROOT
# hunting_innocent_trigger        an adult courtier who is not
# hunting_courtier_hunter_trigger a hunter of ROOT's court riding along
# hunting_park_vassal_trigger     a vassal at court who could ask for deer
# hunting_trophy_allowed_trigger  ROOT may still win a grand hunt trophy
###############################################################################

# ROOT is the host.
hunting_grand_on_trigger = {
	has_character_modifier = holding_grand_hunt
	prisoner = no
	is_incapacitated_trigger = no
	NOT = { health_traits = 1 }
}

# A courtier of the host.
hunting_plotter_trigger = {
	has_plot = plot_kill_character
	plot_target_char = { character = ROOT }
}

# A courtier of the host.
hunting_innocent_trigger = {
	is_adult = yes
	prisoner = no
	NOT = { hunting_plotter_trigger = yes }
}

# A courtier of the host.
hunting_courtier_hunter_trigger = {
	trait = hunter
	is_adult = yes
	prisoner = no
	in_command = no
	is_ruler = no
	is_incapacitated_trigger = no
	NOT = { health_traits = 1 }
}

# A vassal of the host.
hunting_park_vassal_trigger = {
	prisoner = no
	war = no
	in_command = no
	is_adult = yes
	is_incapacitated_trigger = no
	OR = {
		is_abroad = no
		at_location = ROOT
	}
	OR = {
		trait = ambitious
		NOT = { trait = slothful }
	}
}

# ROOT is the hunter.
hunting_trophy_allowed_trigger = {
	NOT = { has_character_flag = grand_hunt_trophy_given }
	NOT = { any_artifact = { has_artifact_flag = trophy } }
}
