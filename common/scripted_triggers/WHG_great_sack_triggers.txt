###############################################################################
# The sack of great cities (events/WHG_great_sack_events.txt)
###############################################################################

# A horde that sacks cities: a Chaos horde, a greenskin or beastman lord,
# or a nomad of the steppe. Not a rebel.
# Scope: a character.
greatsack_horde_trigger = {
	rebel = no
	OR = {
		government = chaos_horde_government
		is_nomadic = yes
		religion_group = greenskin_gods_group
		culture_group = orc_group
		culture_group = goblin_group
		culture_group = beastman_group
	}
}

# A great city: the chief holding of the county that is the seat of a king
# or an emperor, not sacked in the last fifty years.
# Scope: a holding (barony title).
greatsack_great_city_trigger = {
	is_capital = yes
	has_holder = yes
	location = {
		owner = {
			top_liege = {
				higher_tier_than = DUKE
				capital_scope = { province = PREVPREVPREV }
			}
		}
		NOT = { has_province_flag = great_city_sacked }
	}
}

# Which horde it is, for the texts (greatsack.1, greatsack.2). Checked in
# this order: Chaos, then greenskins, then beastmen; anyone else is a nomad.
# Scope: the horde's lord.
greatsack_chaos_trigger = {
	OR = {
		religion_group = chaos_gods_group
		government = chaos_horde_government
	}
	NOT = { culture_group = beastman_group }
}
greatsack_greenskin_trigger = {
	greatsack_chaos_trigger = no
	OR = {
		religion_group = greenskin_gods_group
		culture_group = orc_group
		culture_group = goblin_group
	}
}
greatsack_beastman_trigger = {
	culture_group = beastman_group
}
greatsack_nomad_trigger = {
	greatsack_chaos_trigger = no
	greatsack_greenskin_trigger = no
	greatsack_beastman_trigger = no
}

# The horde does not sack its own kind: Chaos spares the followers of the
# Dark Gods, the Norscans and the Dawi Zharr (as chaoshorde_sacker_trigger
# does), greenskins spare greenskins, beastmen beastmen, nomads nomads.
# Scope: the holding; ROOT is the horde's lord.
greatsack_victim_trigger = {
	holder_scope = {
		trigger_if = {
			limit = { ROOT = { greatsack_chaos_trigger = yes } }
			open_follower_of_any_chaos_gods = no
			NOR = {
				religion = norscan_gods
				religion = norscan_gods_reformed
				religion = hashut
			}
		}
		trigger_else_if = {
			limit = { ROOT = { greatsack_greenskin_trigger = yes } }
			NOT = { religion_group = greenskin_gods_group }
		}
		trigger_else_if = {
			limit = { ROOT = { greatsack_beastman_trigger = yes } }
			NOT = { culture_group = beastman_group }
		}
		trigger_else = {
			is_nomadic = no
		}
	}
}

# ROOT's horde sacks this great city now (greatsack.1 fires or has fired
# today). The usual sacking events skip it, so the city is not sacked twice:
# chaoshorde_sackable_holding_trigger, nomad.24, waaagh.30, beastmen.60.
# Scope: the holding taken; ROOT is the owner of the winning army.
greatsack_city_falls_now_trigger = {
	ROOT = { greatsack_horde_trigger = yes }
	is_capital = yes
	has_holder = yes
	greatsack_victim_trigger = yes
	location = {
		owner = {
			top_liege = {
				higher_tier_than = DUKE
				capital_scope = { province = PREVPREVPREV }
				NOT = { character = ROOT }
				ROOT = { NOT = { is_vassal_or_below = PREV } }
			}
		}
		NOT = { had_province_flag = { flag = great_city_sacked days = 1 } }
	}
}
