# The guardian and the ward (events/WHG_guardian_events.txt)

# The guardian may turn to the gods over a ward: not a cynic, and not a
# servant of the Dark Gods, who reward vice rather than punish it.
# Scope: the guardian.
guardian_prays_trigger = {
	NOT = { trait = cynical }
	NOT = { religion_group = chaos_gods_group }
}

# A child of the liege whom a vassal may offer to raise (guardian.300).
# Scope: the child.
guardian_fosterable_trigger = {
	age = 6
	NOT = { age = 16 }
	is_ruler = no
	prisoner = no
	has_guardian = no
}

# A living father who outranks the ward's guardian: a count's ward whose
# father is a duke or more, a duke's ward whose father is a king or more,
# a king's ward whose father is an emperor (guardian.500, guardian.505).
# Scope: the ward.
ward_father_outranks_guardian_trigger = {
	has_guardian = yes
	father = {
		is_alive = yes
		OR = {
			AND = {
				tier = EMPEROR
				PREV = { guardian = { tier = KING } }
			}
			AND = {
				OR = { tier = KING tier = EMPEROR }
				PREV = { guardian = { tier = DUKE } }
			}
			AND = {
				OR = { tier = DUKE tier = KING tier = EMPEROR }
				PREV = { guardian = { tier = COUNT } }
			}
		}
	}
}

# A fosterling of school age, not of the guardian's own family
# (guardian.510). Scope: the ward.
ward_fosterling_trigger = {
	family = no
	age = 6
	NOT = { age = 16 }
	prisoner = no
	NOT = { trait = incapable }
}
