# The guardian and the ward (events/WHG_guardian_events.txt)

# The guardian may turn to the gods over a ward: not a cynic, and not a
# servant of the Dark Gods, who reward vice rather than punish it.
# Scope: the guardian.
guardian_prays_trigger = {
	NOT = { trait = cynical }
	NOT = { religion_group = chaos_gods_group }
}

# A child of the liege whom a vassal may offer to raise (guardian.300).
# Scope: the child.
guardian_fosterable_trigger = {
	age = 6
	NOT = { age = 16 }
	is_ruler = no
	prisoner = no
	has_guardian = no
}
