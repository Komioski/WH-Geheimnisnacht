# Triggers for events/WHG_hashishin_events.txt

# Any scope. The Hashishin order has a Grand Master.
hashishin_active_trigger = {
	d_hashishin = { has_holder = yes }
	NOT = { has_global_flag = assassins_destroyed }
}

# Scope: a courtier. ROOT is the ruler whose court it is. A courtier the
# Hashishin could kill as a message: an AI adult at court, free, not
# leading troops, not the ruler's spouse or heir, not a pretender or a ruler
# and not one of the Hashishin.
hashishin_victim_trigger = {
	ai = yes
	is_adult = yes
	prisoner = no
	is_ruler = no
	in_command = no
	is_primary_heir = no
	is_pretender = no
	host = { character = ROOT }
	NOT = { spouse = { character = ROOT } }
	NOT = { society_member_of = the_assassins }
}
