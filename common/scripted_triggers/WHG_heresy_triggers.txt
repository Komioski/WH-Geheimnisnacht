# Triggers for events/WHG_heresy_events.txt

# Province scope. Fanatics may rise in this county: an Old World or Arabyan
# faith (not the Grail faiths, which get Shallyan uprisings instead) with a
# settled holding, while the faith's authority is below 90%.
heresy_fanatics_can_rise_trigger = {
	OR = {
		religion_group = old_world_group
		religion_group = ormazd_gods_group
	}
	NOR = {
		religion = grail
		religion = false_grail
	}
	num_of_settlements = 1
	any_province_holding = {
		NOT = { holding_type = nomad }
	}
	NOT = { religion_authority = 0.9 }
	NOT = { has_province_modifier = fanatic_uprising }
}

# Province scope. Fanatics of a heresy hold this county, its top liege keeps
# the parent faith, and that faith's head has 100 gold to act and is not
# already acting here.
heresy_head_worried_trigger = {
	OR = {
		has_province_modifier = heretic_stronghold
		has_province_modifier = fanatic_uprising
	}
	owner = {
		top_liege = {
			is_parent_religion = PREVPREV
		}
	}
	parent_religion_head = {
		wealth = 100
	}
	NOT = { has_province_modifier = theological_dialogue }
	NOT = { has_province_modifier = active_inquisition }
}

# Province scope. The fanatics still hold out when an inquest or a dialogue
# ends.
heresy_fanatics_remain_trigger = {
	OR = {
		has_province_modifier = heretic_stronghold
		has_province_modifier = fanatic_uprising
	}
}
