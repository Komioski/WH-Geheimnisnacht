###############################################################################
# Nomad clans and pillage (WHG_nomad_events.txt)
###############################################################################

# The horse peoples of the steppe: Kurgan, Hung and Ungol. A raider of these
# peoples who takes a tribe keeps their own faith and culture (nomad.35).
# Scope: a character.
nomad_steppe_people_trigger = {
	OR = {
		culture_group = kurgan_group
		culture_group = hung_group
		culture = ungol
	}
}

# A greenskin raider, for the sack picture (nomad.24).
# Scope: a character.
nomad_greenskin_raider_trigger = {
	OR = {
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
	}
}

# A county with no castle, city, temple or tribe left (nomad.25).
# Scope: a province.
nomad_county_empty_trigger = {
	NOT = {
		any_province_holding = {
			OR = {
				holding_type = city
				holding_type = temple
				holding_type = castle
				holding_type = tribal
			}
		}
	}
}

# Another emperor-tier horde of ROOT's own people (nomad.31).
# Scope: a ruler; ROOT is the new clan chief.
nomad_great_horde_of_root_trigger = {
	culture = ROOT
	is_nomadic = yes
	tier = EMPEROR
	NOT = { character = ROOT }
}

# A restless county (nomad_agitation) away from ROOT's capital, with no
# castle, city, building site or grown tribe to hold its people down
# (nomad.34).
# Scope: a province; ROOT is the settled ruler.
nomad_restless_county_trigger = {
	has_province_modifier = nomad_agitation
	NOR = {
		any_province_holding = { holding_type = castle }
		any_province_holding = { holding_type = city }
		has_settlement_construction = yes
		AND = {
			ROOT = { is_tribal = yes }
			any_province_holding = {
				holding_type = tribal
				num_of_buildings = 2
			}
		}
	}
	NOT = { ROOT = { capital_scope = { province = PREVPREV } } }
}

# Steppe wargames, sky burials and wrestling (WHG_steppe_events.txt)
###############################################################################

# A grown warrior of ROOT's realm or close kin, free to come to the
# wargames: a man, or a woman who serves as marshal (steppe.1).
# Scope: a realm character; ROOT is the host.
steppe_wargames_invitee_trigger = {
	prisoner = no
	war = no
	NOT = { trait = incapable }
	age = 16
	NOT = { has_character_flag = do_not_disturb }
	OR = {
		liege = { character = ROOT }
		is_close_relative = ROOT
	}
	OR = {
		is_female = no
		has_character_flag = special_marshal
	}
}

# A guest at the wargames (steppe.6, steppe.7).
# Scope: a realm character.
steppe_wargames_guest_trigger = {
	has_character_flag = attending_nomad_wargames
}

# Peoples who give their dead to the birds: the horse peoples of the steppe,
# the Dark Gods' faithful, the faithful of Ormazd, Kislev's tzars, and any
# sky-burial bloodline (steppe.20, sky_burial_decision).
# Scope: a character.
steppe_sky_burial_people_trigger = {
	OR = {
		nomad_steppe_people_trigger = yes
		religion_group = chaos_gods_group
		religion_group = ormazd_gods_group
		government = tzardom_government
		any_owned_bloodline = { has_bloodline_flag = bloodline_sky_burial }
	}
}

# A grown, hale AI warrior fit to wrestle the lord (steppe.40, steppe.41).
# Scope: a realm character; ROOT is the lord.
steppe_wrestler_trigger = {
	ai = yes
	is_female = no
	is_adult = yes
	prisoner = no
	martial = 10
	NOR = {
		character = ROOT
		trait = incapable
		trait = infirm
		health_traits = 1
		trait = craven
	}
}

# Raiding adventurers (WHG_adventurer_events.txt)
###############################################################################

# Peoples whose landless sons take to raiding: the horse peoples under a
# horde, Norsca and its gods, the reavers of Stromfels, the corsairs of
# Araby (in a port), and any ambitious, gifted warrior with no inheritance
# (adventurer.1).
# Scope: a landless character.
adventurer_raider_people_trigger = {
	OR = {
		AND = {
			any_liege = {
				OR = {
					government = nomadic_government
					government = chaos_horde_government
				}
			}
			OR = {
				culture_group = kurgan_group
				culture_group = hung_group
			}
		}
		culture_group = norscan_group
		religion = norscan_gods
		religion = stromfels
		AND = {
			OR = {
				religion = atruhayid_reformed
				religion = atruhayid
			}
			location = { port = yes }
		}
		AND = {
			trait = ambitious
			martial = 15
			diplomacy = 12
			health = 4
			is_heir = no
			NOR = {
				any_child = {
					OR = {
						is_heir = yes
						is_landed = yes
					}
				}
				NAND = {
					uses_decadence = yes
					num_of_dynasty_members = 3
					biggest_realm_size_relative = {
						NOT = { character = ROOT }
						realm_size = 5
					}
				}
			}
		}
	}
}

# A grandchild (or child) of a count or better: the blood to raise a host
# (adventurer.1).
# Scope: a character.
adventurer_noble_blood_trigger = {
	OR = {
		father_even_if_dead = {
			OR = {
				primary_title = { higher_tier_than = BARON }
				father_even_if_dead = { primary_title = { higher_tier_than = BARON } }
				mother_even_if_dead = { primary_title = { higher_tier_than = BARON } }
			}
		}
		mother_even_if_dead = {
			OR = {
				primary_title = { higher_tier_than = BARON }
				father_even_if_dead = { primary_title = { higher_tier_than = BARON } }
				mother_even_if_dead = { primary_title = { higher_tier_than = BARON } }
			}
		}
	}
}

# A claim on a foreign king's title or an independent ruler's title of the
# same tier: TOG.1200 handles those claimants instead (adventurer.1).
# Scope: a character.
adventurer_tog_claimant_trigger = {
	any_claim = {
		has_holder = yes
		holder_scope = {
			independent = yes
			OR = {
				is_merchant_republic = no
				NOT = { capital_holding = { de_jure_liege_or_above = PREVPREV } }
			}
		}
		OR = {
			tier = KING
			holder_scope = { tier = PREV }
		}
	}
}

# An independent ruler the adventurer has looted, small enough to bully,
# not already negotiating and with no settled adventurer yet (adventurer.10,
# .11, .19).
# Scope: an independent ruler; ROOT the adventurer.
adventurer_looted_ruler_trigger = {
	has_opinion_modifier = { who = ROOT modifier = opinion_looted }
	NOT = { realm_levy_diff = { who = ROOT value = 3000 } }
	demesne_size = 1
	NOT = { has_character_flag = negotiating_with_adventurer }
	NOT = { any_realm_province = { has_province_modifier = settled_adventurer } }
}

# A province a great lord could hand to the adventurer: not the capital, one
# holding, no nomad unrest, and a port for a sea raider when the realm has
# any (adventurer.11, .19).
# Scope: a demesne province; PREV the ruler whose demesne it is, ROOT the
# adventurer.
adventurer_spare_province_trigger = {
	is_capital = no
	num_of_settlements = 1
	NOT = { has_province_modifier = nomad_agitation }
	OR = {
		AND = {
			ROOT = { is_seafarer = yes }
			port = yes
		}
		ROOT = { is_seafarer = no }
		PREV = { NOT = { any_demesne_province = { port = yes } } }
	}
}

# A great lord with a province to spare for negotiation (adventurer.11, .19).
# Scope: an independent ruler; ROOT the adventurer.
adventurer_negotiation_partner_trigger = {
	higher_real_tier_than = DUKE
	OR = {
		any_demesne_province = { adventurer_spare_province_trigger = yes }
		any_vassal = {
			any_demesne_province = { adventurer_spare_province_trigger = yes }
		}
	}
}

# A province on the realm's border (adventurer_declare_war_effect).
# Scope: a province; PREV the target ruler.
adventurer_border_province_trigger = {
	any_neighbor_province = {
		NOT = { owner = { same_realm = PREVPREVPREV } }
	}
}

# Mercenary bands (WHG_mercenary_events.txt, merc.300-341)

# The captain of a band raised from a clan or a mercenary family
# (create_mercenary_company), not a dynastic band of the merc.1 tombola.
# Scope: a character.
merc_band_captain_trigger = {
	has_character_flag = is_mercenary_leader
	primary_title = {
		mercenary = yes
		NOT = { has_title_flag = ze_dyn_merc }
	}
}

# The life of the band still has something to teach or wear away: an
# outcome of merc_captain_growth_pick_effect is open (merc.300).
# Scope: the captain.
merc_captain_can_grow_trigger = {
	OR = {
		has_lower_than_max_martial = yes
		trait = craven
		NOR = { trait = craven trait = brave }
		NOR = { trait = diligent trait = slothful }
		trait = kind
		trait = humble
		AND = {
			NOR = { trait = lustful trait = chaste }
			is_no_females_race = no
			NOT = { is_witch_hunter_trigger = yes }
		}
		AND = {
			NOR = { trait = wroth trait = patient }
			NOT = { is_witch_hunter_trigger = yes }
		}
		NOR = { trait = cruel trait = kind }
	}
}

# A human captain who could take an interest in the faith of the lord whose
# lands the band sits in: the Old World gods, the gods of Araby or of
# Cathay. Witch hunters, vampires and the Chaos-sworn never do.
# Scope: the captain.
merc_captain_open_minded_trigger = {
	trait = creature_human
	NOT = { trait = zealous }
	NOT = { is_witch_hunter_trigger = yes }
	NOT = { trait = vampire_blood_visible }
	NOT = { religion_group = chaos_gods_group }
}

# The lord of the band's home province follows one of those faiths.
# Scope: a ruler.
merc_faith_lord_trigger = {
	OR = {
		religion_group = old_world_group
		religion_group = ormazd_gods_group
		religion_group = oriental_gods_group
	}
}

# The captain already feels for the faith group of this lord.
# Scope: a ruler; ROOT the captain.
merc_root_sympathises_trigger = {
	OR = {
		AND = {
			religion_group = old_world_group
			ROOT = { trait = sympathy_old_world_group }
		}
		AND = {
			religion_group = ormazd_gods_group
			ROOT = { trait = sympathy_ormazd_gods_group }
		}
		AND = {
			religion_group = oriental_gods_group
			ROOT = { trait = sympathy_oriental_gods_group }
		}
	}
}

# Curiosity for a foreign faith group (merc.300 -> merc.305).
# Scope: the captain.
merc_captain_faith_curious_trigger = {
	merc_captain_open_minded_trigger = yes
	primary_title = {
		capital_scope = {
			owner = {
				merc_faith_lord_trigger = yes
				NOT = { religion_group = ROOT }
				NOT = { merc_root_sympathises_trigger = yes }
			}
		}
	}
}

# Conversion to the faith of the band's lord (merc.300 -> merc.306).

merc_captain_faith_convert_trigger = {
	merc_captain_open_minded_trigger = yes
	primary_title = {
		capital_scope = {
			owner = {
				merc_faith_lord_trigger = yes
				NOT = { religion = ROOT }
			}
		}
	}
}

# A courtier of the band's lord who might run off with the captain: unwed,
# unlanded, free, a fit match for the captain, and either the lord's own
# child or a gifted courtier (merc.300 -> merc.307).
# Scope: a courtier; PREV the lord, ROOT the captain.
merc_eloper_trigger = {
	is_adult = yes
	prisoner = no
	is_landed = no
	is_married = no
	has_lover = no
	is_consort = no
	NOT = { trait = celibate }
	NOT = { trait = incapable }
	is_human = yes
	is_no_females_race = no
	NOT = { character = ROOT }
	NOT = { is_close_relative = ROOT }
	OR = {
		is_child_of = PREV
		martial = 20
		diplomacy = 20
		intrigue = 20
		stewardship = 20
		learning = 20
	}
	OR = {
		AND = {
			is_opposite_sex = ROOT
			NOT = { trait = homosexual }
			ROOT = { NOT = { trait = homosexual } }
		}
		AND = {
			NOT = { is_opposite_sex = ROOT }
			trait = homosexual
			ROOT = { trait = homosexual }
		}
	}
}

# A free spirit of a captain with a match at the lord's court (merc.300).

		}
	}
}

# A band that still camps in the ruler's capital province, led by a captain

# A trade post on one of the roads that reach Grand Cathay: the Ivory Road,
# the Old Silk Road, the Khyprian Road and the Spice Route (ivory.1).
# Scope: a province.
ivory_road_cathay_province_trigger = {
	has_trade_post = yes
	OR = {
		trade_route = ivory_road
		trade_route = old_silk_road
		trade_route = khyprian_road
		trade_route = spice_route
	}
}

# A trade post on a busy caravan road, not already plagued by raiders or
# guarded by hired swords (ivory.50).
# Scope: a province.
ivory_caravan_province_trigger = {
	has_trade_post = yes
	NOT = { has_province_modifier = caravan_raiders }
	NOT = { has_province_modifier = caravan_guards }
	OR = {
		trade_route = ivory_road
		trade_route = old_silk_road
		trade_route = khyprian_road
		trade_route = spice_route
		trade_route = arabyan_trail
		trade_route = the_silver_road
		trade_route = the_skull_road
		trade_route = the_pass_to_the_east
		trade_route = old_dwarf_road
		trade_route = old_forest_road
		trade_route = great_north_road
		trade_route = great_reik_road
		trade_route = great_kislev_road
		trade_route = altdorf_middenheim_road
	}
}

# The Cathayan envoy still serves at ROOT's court, free (ivory.3, ivory.20).
