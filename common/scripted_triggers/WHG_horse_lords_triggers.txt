###############################################################################
# Shared triggers for the Horse Lords events: nomad clans and pillage,
# horses, steppe games and burials, raiding adventurers, mercenaries, the
# Ivory Road and the wolf's blood. One section per part.
###############################################################################

###############################################################################
# Nomad clans and pillage (WHG_nomad_events.txt)
###############################################################################

# The horse peoples of the steppe: Kurgan, Hung and Ungol. A raider of these
# peoples who takes a tribe keeps their own faith and culture (nomad.35).
# Scope: a character.
nomad_steppe_people_trigger = {
	OR = {
		culture_group = kurgan_group
		culture_group = hung_group
		culture = ungol
	}
}

# A greenskin raider, for the sack picture (nomad.24).
# Scope: a character.
nomad_greenskin_raider_trigger = {
	OR = {
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
	}
}

# A county with no castle, city, temple or tribe left (nomad.25).
# Scope: a province.
nomad_county_empty_trigger = {
	NOT = {
		any_province_holding = {
			OR = {
				holding_type = city
				holding_type = temple
				holding_type = castle
				holding_type = tribal
			}
		}
	}
}

# Another emperor-tier horde of ROOT's own people (nomad.31).
# Scope: a ruler; ROOT is the new clan chief.
nomad_great_horde_of_root_trigger = {
	culture = ROOT
	is_nomadic = yes
	tier = EMPEROR
	NOT = { character = ROOT }
}

# A restless county (nomad_agitation) away from ROOT's capital, with no
# castle, city, building site or grown tribe to hold its people down
# (nomad.34).
# Scope: a province; ROOT is the settled ruler.
nomad_restless_county_trigger = {
	has_province_modifier = nomad_agitation
	NOR = {
		any_province_holding = { holding_type = castle }
		any_province_holding = { holding_type = city }
		has_settlement_construction = yes
		AND = {
			ROOT = { is_tribal = yes }
			any_province_holding = {
				holding_type = tribal
				num_of_buildings = 2
			}
		}
	}
	NOT = { ROOT = { capital_scope = { province = PREVPREV } } }
}

###############################################################################
# Horses (WHG_horse_events.txt)
###############################################################################

# A spouse who rides with the rider and could be swept away (horse.20).
# Scope: a spouse; ROOT is the rider.
horse_river_spouse_trigger = {
	ai = yes
	at_location = ROOT
	NOT = { trait = incapable }
	NOT = { has_character_flag = do_not_disturb }
}

# An adult AI at court whose horse might bite the warhorse (horse.40).
# Scope: a vassal or courtier.
horse_biter_owner_trigger = {
	ai = yes
	is_adult = yes
	prisoner = no
	NOT = { trait = incapable }
}

# A kind courtier who likes the rider well enough to share the grief
# (horse.3).
# Scope: a courtier; ROOT is the rider.
horse_kind_mourner_trigger = {
	trait = kind
	opinion = { who = ROOT value = 0 }
	NOT = { trait = incapable }
}

###############################################################################
# Steppe wargames, sky burials and wrestling (WHG_steppe_events.txt)
###############################################################################

# A grown warrior of ROOT's realm or close kin, free to come to the
# wargames: a man, or a woman who serves as marshal (steppe.1).
# Scope: a realm character; ROOT is the host.
steppe_wargames_invitee_trigger = {
	prisoner = no
	war = no
	NOT = { trait = incapable }
	age = 16
	NOT = { has_character_flag = do_not_disturb }
	OR = {
		liege = { character = ROOT }
		is_close_relative = ROOT
	}
	OR = {
		is_female = no
		has_character_flag = special_marshal
	}
}

# A guest at the wargames (steppe.6, steppe.7).
# Scope: a realm character.
steppe_wargames_guest_trigger = {
	has_character_flag = attending_nomad_wargames
}

# Peoples who give their dead to the birds: the horse peoples of the steppe,
# the Dark Gods' faithful, the faithful of Ormazd, Kislev's tzars, and any
# sky-burial bloodline (steppe.20, sky_burial_decision).
# Scope: a character.
steppe_sky_burial_people_trigger = {
	OR = {
		nomad_steppe_people_trigger = yes
		religion_group = chaos_gods_group
		religion_group = ormazd_gods_group
		government = tzardom_government
		any_owned_bloodline = { has_bloodline_flag = bloodline_sky_burial }
	}
}

# A grown, hale AI warrior fit to wrestle the lord (steppe.40, steppe.41).
# Scope: a realm character; ROOT is the lord.
steppe_wrestler_trigger = {
	ai = yes
	is_female = no
	is_adult = yes
	prisoner = no
	martial = 10
	NOR = {
		character = ROOT
		trait = incapable
		trait = infirm
		health_traits = 1
		trait = craven
	}
}
