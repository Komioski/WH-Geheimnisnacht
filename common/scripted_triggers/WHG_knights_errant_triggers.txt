#
# errant_first_scene_trigger  ROOT host: no knight recruited, no seduction yet
# errant_offended_trigger     a proud or touchy man of the court
# errant_seduced_trigger      a young unwed woman of the court
###############################################################################

# ROOT is the host.
errant_first_scene_trigger = {
	NOT = { has_character_flag = errant_recruited }
	NOT = { has_character_flag = errant_seduction }
}

# A courtier of the host.
errant_offended_trigger = {
	age = 16
	is_female = no
	prisoner = no
	OR = {
		trait = cruel
		trait = proud
		trait = wroth
		trait = zealous
		trait = envious
		prestige = 100
	}
}

# A courtier of the host.
errant_seduced_trigger = {
	age = 16
	NOT = { age = 20 }
	is_woman_trigger = yes
	is_no_females_race = no
	prisoner = no
	is_married = no
	has_lover = no
	is_pregnant = no
	fertility = 0.3
}
