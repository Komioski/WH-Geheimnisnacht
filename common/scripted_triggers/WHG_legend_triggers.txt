# WHG_legend_triggers.txt: scripted triggers
# Subject: ambitions, deeds of legend, great hunts, artifacts, books and
# debates
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Shared: used by three or more files (13)
#   Used by WHG_ambition_events (20)
#   Used by WHG_ambitions (8)
#   Used by WHG_armoury_events (14)
#   Used by WHG_artifact_events (6)
#   Used by WHG_gamebook_events (4)
#   Used by WHG_grand_debate_events (4)
#   Used by WHG_hunting_events (5)
#   Other helpers used by one or two files (7)
###############################################################################

can_be_chancellor_trigger = {
	NOT = { has_character_flag = carnival_troupe } # the Carnival of Chaos
		liege = {
			NOT = { trait = vampire_necrarch_visible }
		}
		NOT = { trait = vampire_necrarch_visible }		#Necrarchs are reclusive, they do not get any diplomatic action
		OR = {
			AND = {
				OR = {
					is_female = no
					trait = horse
					AND = {
						is_female = yes
						liege = { religion = amazon }
					}
					AND = {
						is_female = yes
						liege = { religion_group = cadai_group }
					}
					AND = {
						is_female = yes
						liege = { religion_group = cytharai_group }
					}
					AND = {
						liege = {
							primary_title = {
								OR = {
									has_law = status_of_women_2
									has_law = status_of_women_3
								}
							}
						}
						OR = {
							liege = {
								OR = {
									dynasty = ROOT
									is_close_relative = ROOT
									any_spouse = { character = ROOT }
									any_consort = { character = ROOT }
								}
							}
							is_landed = yes

						}
					}
					has_law = status_of_women_4
				}
				is_adult = yes
			}
			AND = {
				is_female = yes
					liege = {
						primary_title = {
							OR = {
								has_law = enatic_succession
								has_law = enatic_cognatic_succession
								has_law = true_cognatic_succession
							}
						}
					}
					is_adult = yes
			}
            has_character_flag = special_chancellor
        }
        # warhammer restriction
		NOT = { trait = creature_slann }
		OR = {
			is_female = yes
			NOT = {
				liege = { trait = creature_amazon }
			}
		}
        OR = {
		   AND = {
				NOT = { culture_group = beastman_group }
				NOT = {
					FROM = { culture_group = beastman_group }
				}
			}
		   AND = {
				culture_group = beastman_group
				FROM = { culture_group = beastman_group }
			}
        }
		OR = {
		   AND = {
				NOT = {
					OR = {
						culture_group = orc_group
						culture_group = goblin_group
					}
				}
				NOT = {
						FROM = {
							OR = {
							culture_group = orc_group
							culture_group = goblin_group
						}
					}
				}
			}
		   AND = {
				OR = {
					culture_group = orc_group
					culture_group = goblin_group
				}
				FROM = {
						OR = {
						culture_group = orc_group
						culture_group = goblin_group
						}
				}
			}
		}
        religion_group = FROM

		prisoner = no
		is_incapable = no
		NOT = { trait = in_hiding }
}

can_be_spiritual_trigger = {
	NOT = { has_character_flag = carnival_troupe } # the Carnival of Chaos
	liege = { religion = ROOT }
	religion_group = FROM
	liege = {
		NOT = { trait = vampire_necrarch_visible }
	}
	NOT = { trait = vampire_necrarch_visible }		#Necrarchs are reclusive, they do not get any diplomatic action
	prisoner = no
	OR = {
		is_incapable = no
		trait = creature_slann
	}
	NOT = { trait = in_hiding }
	OR = {
		AND = {
			is_adult = yes
			OR = {
				is_female = no
				trait = everqueen
				trait = everqueen_heir
				religion_allows_female_temple_holders = yes
				AND = {
					is_female = yes
					liege = { religion = amazon }
				}
			}
		}
		has_character_flag = special_spiritual
	}
        # warhammer restriction
	OR = {
		trait = khemri_liche_priest
		NOT = {
			FROM = { trait = creature_nehekharan }
		}
	}
	# Lizardmen
	OR = {
		AND = {
			NOT = {
				trait = creature_skink
			}
			FROM = {
				NOR = {
					trait = creature_slann
					trait = creature_skink
				}
			}
		}
		AND = {
			OR = {
				trait = skink_priest
				trait = creature_slann
			}
			FROM = {
				OR = {
					trait = creature_slann
					trait = creature_skink
				}
            }
		}
    }
	#Beastmen
    OR = {
		NOR = {
			culture_group = beastman_group
			FROM = { culture_group = beastman_group }
		}
		AND = {
			culture_group = beastman_group
			FROM = { culture_group = beastman_group }
		}
    }
	#Skaven
    OR = {
        NOR = {
			trait = creature_skaven
            FROM = { trait = creature_skaven }
	    }
        AND = {
			trait = grey_seer
            FROM = { trait = creature_skaven }
		}
    }
	OR = {
		is_theocracy = yes
		AND = {
			liege = { is_tribal = yes }
			is_ruler = no
		}
		liege = {
			OR = {
				religion = ormazd
				religion = kitabid
			}
		}
		AND = {
			clan = yes
			liege = {
				is_nomadic = yes
			}
		}
		AND = {
			has_autocephaly = no
			is_ruler = no
		}
		AND = {
			has_autocephaly = yes
			is_ruler = no
			NOT = {
				any_vassal = { is_theocracy = yes }
			}
		}
	}

	OR = {
		NOT = { religion_group = ormazd_gods_group }
		NOT = { liege = { religion_group = ormazd_gods_group } }
		trait = grey_seer
		trait = creature_slann
		trait = khemri_liche_priest
		trait = BretonDamsel
		trait = everqueen
		trait = fay_enchantress
		trait = detached_priest
		trait = martial_cleric
		trait = scholarly_theologian
		trait = mastermind_theologian
		trait = skink_priest
	}
	#Greenskin
	OR = {
		NOR = {
			culture_group = orc_group
			culture_group = goblin_group
			FROM = {
				OR = {
					culture_group = orc_group
					culture_group = goblin_group
				}
			}
		}
	   AND = {
			OR = {
				culture_group = orc_group
				culture_group = goblin_group
			}
			FROM = {
				OR = {
					culture_group = orc_group
					culture_group = goblin_group
				}
			}
		}
	}
}

# returns true if scoped character already likes ROOT and could be physically
# attracted to them
# Used by: WHG_ai_score_values, WHG_book_events +2 more
possibly_romantically_interested_in_ROOT_trigger = {
	NOT = { is_rival = ROOT }
	NOT = { is_lover = ROOT }
	attacted_to_root_trigger = yes
	OR = {
		opinion = { who = ROOT value = 45 }
		has_opinion_modifier = { who = ROOT modifier = opinion_budding_romance }
	}
}

# returns true if scoped character is already liked by ROOT and ROOT could be
# physically attracted to them
# Used by: WHG_ai_score_values, WHG_grand_debate_events +4 more
ROOT_possibly_romantically_interested_trigger = {
	NOT = { is_rival = ROOT }
	NOT = { is_lover = ROOT }
	root_attraction_trigger = yes
	OR = {
		reverse_opinion = { who = ROOT value = 45 }
		reverse_has_opinion_modifier = { who = ROOT modifier = opinion_budding_romance }
	}
}

# Scope: a ruler with a rumor. His hunter is close enough to the treasure to
# find it on the next leg of the search: 8 steps for a common rumor, 14 for a
# rare one, 20 for a great one.
# Used by: WHG_artifact_events
artifact_hunt_done_trigger = {
	OR = {
		AND = {
			has_character_modifier = relic_rumor_minor
			check_variable = { which = relic_score value = 8 }
		}
		AND = {
			has_character_modifier = relic_rumor_medium
			check_variable = { which = relic_score value = 14 }
		}
		AND = {
			has_character_modifier = relic_rumor_major
			check_variable = { which = relic_score value = 20 }
		}
	}
}

	}
}
