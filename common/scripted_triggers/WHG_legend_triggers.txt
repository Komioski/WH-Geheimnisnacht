# legend_complainer_trigger
# Scope: a courtier. A free adult who could have been robbed on the road
# (legend.2).
legend_complainer_trigger = { is_adult = yes prisoner = no is_incapable = no }

# legend_rift_open_trigger
# Scope: a ruler. A burning rift is open in the ruler's lands.
legend_rift_open_trigger = {
	OR = {
		has_character_modifier = flaming_gap
		has_character_modifier = wailing_gap
		has_character_modifier = gates_hell
	}
}

# legend_rift_sacrifice_trigger
# Scope: a child, ROOT the parent. A son of 10 to 19 at home and free, with
# no older brother or sister by ROOT: the heir the rift asks for
# (legend.103).
legend_rift_sacrifice_trigger = {
	is_alive = yes
	is_abroad = no
	prisoner = no
	is_woman_trigger = no
	age = 10
	NOT = { age = 20 }
	NOT = { any_sibling = { is_older_than = PREV is_child_of = ROOT } }
}
