# lover_affair_partner_trigger
# Scope: a lover, ROOT the ruler. A lover who can love, is free and capable
# and is not ROOT's own spouse: the other party of an affair.
lover_affair_partner_trigger = {
	cant_love_trigger = yes
	prisoner = no
	NOT = { trait = incapable }
	NOT = { spouse = { character = ROOT } }
}

# lover_crush_target_trigger
# Scope: a courtier, ROOT a youth of 12 to 17. An unmarried youth of the
# other sex, free, healthy and not of ROOT's family, who might be ROOT's
# first love (lover.20). Both peoples must have women.
lover_crush_target_trigger = {
	cant_love_trigger = yes
	is_no_females_race = no
	is_married = no
	has_epidemic = no
	is_ill = no
	prisoner = no
	is_abroad = no
	is_pregnant = no
	age = 12
	NOR = {
		age = 18
		trait = homosexual
		trait = celibate
		trait = incapable
		trait = eunuch
		family = ROOT
	}
	OR = {
		AND = { is_woman_trigger = yes ROOT = { is_woman_trigger = no } }
		AND = { is_woman_trigger = no ROOT = { is_woman_trigger = yes } }
	}
}

# lover_mistress_target_trigger
# Scope: a courtier, ROOT a married ruler. A free, healthy woman of 16 to 49
# with no lover, no kin of ROOT, who might become his mistress (lover.30).
lover_mistress_target_trigger = {
	is_woman_trigger = yes
	cant_love_trigger = yes
	prisoner = no
	has_epidemic = no
	is_ill = no
	has_lover = no
	age = 16
	NOT = { age = 50 }
	NOT = { is_close_relative = ROOT }
	NOR = { trait = homosexual trait = incapable trait = celibate }
}

# lover_sweetheart_grown_trigger
# Scope: a courtier, ROOT a young adult. ROOT's sweetheart from their teens:
# both still carry the infatuation, like each other well and have no
# lovers (lover.25).
lover_sweetheart_grown_trigger = {
	cant_love_trigger = yes
	has_character_flag = teenage_infatuation
	has_lover = no
	age = 18
	is_abroad = no
	same_realm = ROOT
	has_opinion_modifier = { who = ROOT modifier = teenage_infatuation }
	reverse_has_opinion_modifier = { who = ROOT modifier = teenage_infatuation }
	opinion = { who = ROOT value = 25 }
	reverse_opinion = { who = ROOT value = 25 }
}

# lover_teen_pair_trigger
# Scope: a courtier, PREV a youth. The two are each other's teenage
# infatuation (lover.28).
lover_teen_pair_trigger = {
	has_character_flag = teenage_infatuation
	is_abroad = no
	prisoner = no
	NOT = { spouse = { character = PREV } }
	has_opinion_modifier = { who = PREV modifier = teenage_infatuation }
	reverse_has_opinion_modifier = { who = PREV modifier = teenage_infatuation }
}

# lover_cooling_trigger
# Scope: a lover, ROOT the other lover. One of the two no longer likes the
# other much (lover.50, lover.51).
lover_cooling_trigger = {
	cant_love_trigger = yes
	OR = {
		NOT = { opinion = { who = ROOT value = 25 } }
		NOT = { reverse_opinion = { who = ROOT value = 25 } }
	}
}

# lover_young_mistress_trigger
# Scope: a lover, ROOT the ruler. A fertile woman under 40, free, capable
# and not pregnant, who is not ROOT's wife (lover.66, lover.72).
lover_young_mistress_trigger = {
	lover_affair_partner_trigger = yes
	is_woman_trigger = yes
	is_pregnant = no
	fertility = 0.5
	NOT = { age = 40 }
}

# lover_own_subject_trigger
# Scope: a character, ROOT the ruler. Someone ROOT may arrest: a courtier
# of ROOT's or one of ROOT's vassals (lover.79).
lover_own_subject_trigger = { OR = { host = { character = ROOT } is_vassal_or_below = ROOT } }

# lover_elope_partner_trigger
# Scope: a lover, ROOT an unmarried character. An unmarried lover, neither
# ruler nor consort, free and capable, whom ROOT may marry (lover.67).
lover_elope_partner_trigger = {
	cant_love_trigger = yes
	is_ruler = no
	is_married = no
	is_consort = no
	prisoner = no
	NOT = { trait = incapable }
	NOT = { num_of_lovers = 2 }
	can_marry = ROOT
	ROOT = { can_marry = PREV }
}
