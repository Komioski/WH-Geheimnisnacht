# A courtier of her husband whom the lady adores (or beds) and who is
# skilled enough for a council seat her husband has already filled.
# Scope: a courtier of the husband. ROOT is the lady.
marriage_favourite_candidate_trigger = {
	is_adult = yes
	is_ruler = no
	is_councillor = no
	is_female = no
	prisoner = no
	NOT = { trait = incapable }
	NOT = { character = ROOT }
	NOT = { any_spouse = { character = ROOT } }
	OR = { any_lover = { character = ROOT } reverse_opinion = { who = ROOT value = 40 } }
	OR = {
		AND = { diplomacy = 10 ROOT = { spouse = { job_chancellor = { always = yes } } } }
		AND = { martial = 10 ROOT = { spouse = { job_marshal = { always = yes } } } }
		AND = { stewardship = 10 ROOT = { spouse = { job_treasurer = { always = yes } } } }
		AND = { intrigue = 10 ROOT = { spouse = { job_spymaster = { always = yes } } } }
		AND = {
			learning = 10
			ROOT = {
				NOT = { religion_group = ormazd_gods_group }
				spouse = {
					NOT = { religion_group = ormazd_gods_group }
					job_spiritual = { always = yes }
				}
			}
		}
	}
}

# A man at court who could father the lady's child: he loves her (or is her
# lover) and is no kin of hers.
# Scope: a courtier. ROOT is the lady.
marriage_sire_candidate_trigger = {
	is_female = no
	is_no_females_race = no
	prisoner = no
	age = 20
	NOT = { character = ROOT }
	NOT = { any_spouse = { character = ROOT } }
	NOT = { is_close_relative = ROOT }
	NOT = { dynasty = ROOT }
	NOR = {
		age = 65
		trait = chaste
		trait = celibate
		trait = homosexual
		trait = eunuch
		is_ascetic_trigger = yes
	}
	OR = { any_lover = { character = ROOT } reverse_opinion = { who = ROOT value = 40 } }
}

# A young, fertile, unattached man at court whom the lady likes.
# Scope: a courtier. ROOT is the lady.
marriage_young_father_trigger = {
	is_female = no
	is_no_females_race = no
	prisoner = no
	has_lover = no
	fertility = 0.5
	age = 20
	NOT = { age = 35 }
	NOT = { any_spouse = { character = ROOT } }
	NOT = { is_close_relative = ROOT }
	NOT = { dynasty = ROOT }
	NOR = { trait = eunuch trait = chaste trait = celibate is_ascetic_trigger = yes }
	reverse_opinion = { who = ROOT value = 40 }
}

# Someone at the lord's court his bored wife could befriend.
# Scope: a courtier. ROOT is the lord.
marriage_bored_wife_friend_trigger = {
	age = 16
	ai = yes
	prisoner = no
	is_councillor = no
	NOT = { character = ROOT }
	NOT = { any_spouse = { character = ROOT } }
}

# A husband whose heart can give out.
# Scope: a character.
marriage_mortal_heart_trigger = { is_invincible_trigger = no NOT = { trait = immortal } }
