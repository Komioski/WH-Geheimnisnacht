# Triggers for events/WHG_monastic_events.txt (the monastic orders)

# Scope: a character. A member of the Order of the Bleeding Heart sitting in
# penance (monastic.22 - 47).
monastic_in_seclusion_trigger = {
	is_oldworld_monastic_member_trigger = yes
	OR = {
		has_quest = quest_monastic_order_penance
		has_quest = quest_monastic_order_meditation
	}
}

# Scope: a character. Has a sin to wash away, or lacks humility or charity:
# the first stage of the purity of soul (monastic.83).
monastic_purity_sin_trigger = {
	OR = {
		trait = cynical
		trait = gluttonous
		trait = greedy
		trait = slothful
		trait = envious
		trait = wroth
		trait = proud
		trait = deceitful
		trait = cruel
		trait = lustful
		NOT = { trait = humble }
		NOT = { trait = charitable }
	}
}

# Scope: a character. Still has some way to go on the path to purity
# (monastic.83).
monastic_purity_possible_trigger = {
	OR = {
		monastic_purity_sin_trigger = yes
		trait = ambitious
		NOT = { trait = kind }
		NOT = { trait = temperate }
		NOT = { trait = honest }
		NOT = { trait = chaste }
		NOT = { trait = patient }
		NOT = { trait = diligent }
		NOT = { trait = content }
	}
}

# Scope: a fellow member, ROOT the member. An AI of the same sex in the
# order, on good terms, not yet close, and not a schemer or an ascetic, who
# may become a friend (monastic.90).
monastic_possible_friend_trigger = {
	same_society_as = ROOT
	NOT = { character = ROOT }
	ai = yes
	opinion = { who = ROOT value = 0 }
	reverse_opinion = { who = ROOT value = 0 }
	NOR = {
		trait = amateurish_plotter
		trait = flamboyant_schemer
		trait = intricate_webweaver
		trait = elusive_shadow
		trait = chaste
		trait = celibate
		is_rival = ROOT
		is_friend = ROOT
		is_lover = ROOT
	}
	OR = {
		AND = {
			is_woman_trigger = no
			ROOT = { is_woman_trigger = no }
		}
		AND = {
			is_woman_trigger = yes
			ROOT = { is_woman_trigger = yes }
		}
	}
}

# Scope: a fellow member, ROOT the thief. The lover won in the order's
# friendship (monastic.96), who may help with the theft (monastic.101).
monastic_relic_lover_trigger = {
	ai = yes
	is_lover = ROOT
	has_character_flag = monastic_friend_lover_@ROOT
	is_alive = yes
	prisoner = no
}

# Scope: a character. A relic of Shallya is still at the motherhouse
# (monastic.100).
monastic_relic_available_trigger = {
	OR = {
		NOT = { has_global_flag = monastic_relic_tear_taken }
		NOT = { has_global_flag = monastic_relic_heart_taken }
		NOT = { has_global_flag = monastic_relic_shroud_taken }
	}
}

# Scope: the thief. Still holds the relic that was stolen (monastic.114).
monastic_relic_held_trigger = {
	OR = {
		AND = {
			has_character_flag = monastic_relic_tear
			has_artifact = shallya_tear_phial
		}
		AND = {
			has_character_flag = monastic_relic_heart
			has_artifact = bleeding_heart_reliquary
		}
		AND = {
			has_character_flag = monastic_relic_shroud
			has_artifact = pergunda_shroud
		}
	}
}

# Scope: a fellow member, ROOT the member. A poet or a learned or eloquent
# AI member who can be reached (monastic.130).
monastic_poet_trigger = {
	ai = yes
	is_incapable = no
	prisoner = no
	in_command = no
	is_inaccessible_trigger = no
	NOT = { character = ROOT }
	OR = {
		trait = poet
		learning = 15
		diplomacy = 15
	}
}

# Scope: a fellow member, ROOT the member. A gardener or a learned AI

monastic_gardener_trigger = {

	is_incapable = no
	prisoner = no
	in_command = no
	is_inaccessible_trigger = no
	NOT = { character = ROOT }
	OR = {
		trait = gardener
		learning = 15
	}
}

# Scope: a fellow member, ROOT the member. An AI member with a vice who can
# be reached (monastic.160).
monastic_sleeper_trigger = {
	ai = yes
	is_incapable = no
	prisoner = no
	in_command = no
	is_inaccessible_trigger = no
	NOT = { character = ROOT }
	has_vice_trigger = yes
}

# Scope: an artifact. A forbidden or hermetic book (monastic.170).
monastic_heretical_book_trigger = {
	has_artifact_flag = book
	OR = {
		has_artifact_flag = unholy
		has_artifact_flag = hermetics
	}
}

# Scope: the former ward. Has a sin to give up or a virtue to take
# (monastic.183).
monastic_ward_can_improve_trigger = {
	OR = {
		has_vice_trigger = yes
		trait = deceitful
		trait = cruel
		trait = cynical
		NOT = { trait = humble }
		NOT = { trait = charitable }
	}
}
