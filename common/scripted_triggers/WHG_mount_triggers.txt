###############################################################################

# The races that ride their own beasts, not horses, camels or elephants.
# Scope: a character.
mount_beast_rider_trigger = {
	OR = {
		trait = creature_amazon
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
		trait = creature_fimir
		trait = creature_skaven
		trait = creature_slann
		trait = creature_saurus
		trait = creature_skink
		trait = creature_ogre
	}
}

# A leader who fights from a fast horse.
# Scope: a character.
mount_light_horse_leader_trigger = {
	OR = { trait = light_foot_leader trait = flanker trait = flat_terrain_leader }
}

# A leader who leads the charge from a heavy horse.
# Scope: a character.
mount_heavy_horse_leader_trigger = {
	OR = {
		trait = experimenter # the direct leader
		trait = inspiring_leader
		trait = aggressive_leader
		trait = cavalry_leader
	}
}
