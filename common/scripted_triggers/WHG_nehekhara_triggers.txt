# The undead and daemonic body mends what the living cannot: a Tomb King,
# a daemon or a vampire with a wound, a mutilation, a sickness or a body out
# of shape. Nurgle's daemons keep their diseases. Missing limbs or eyes the
# character chose to keep (one_*_flag) stay.
# Scope: the character.
undead_needs_mending_trigger = {
	OR = {
		trait = creature_nehekharan
		is_daemon_race_trigger = yes
		is_vampire_trigger = yes
	}
	OR = {
		trait = incapable
		trait = is_fat
		trait = is_malnourished
		trait = infirm
		trait = wounded
		trait = maimed
		trait = blinded
		trait = disfigured
		trait = eunuch
		trait = mangled
		trait = severely_injured
		trait = scarred
		AND = {
			NOT = { has_character_flag = one_eyed_flag }
			trait = one_eyed
		}
		AND = {
			NOT = { has_character_flag = one_handed_flag }
			trait = one_handed
		}
		AND = {
			NOT = { has_character_flag = one_legged_flag }
			trait = one_legged
		}
		AND = {
			NOT = { trait = nurgle_demon }
			OR = {
				trait = abdominal_pain
				trait = cough
				trait = chest_pain
				trait = cramps
				trait = diarrhea
				trait = fatigue
				trait = fever
				trait = headache
				trait = infection
				trait = malaise
				trait = rash
				trait = vomiting
				trait = has_nurgles_rot
				trait = has_creeping_buboes
				trait = has_measles
				trait = has_typhus
				trait = has_tomb_rot
				trait = has_bone_auge
				trait = pneumonic
				trait = flu
				trait = dysentery
				trait = gout
				trait = food_poisoning
				trait = rabies
				trait = cancer
				trait = scurvy
				trait = syphilitic
				trait = lovers_pox
				trait = has_tuberculosis
			}
		}
	}
}

# A Nehekkan tribe may come to roam this province: the desert of Nehekhara
# or land of the Tomb Kings, not already Nehekkan, no tribe yet, not refused
# for good, and (for the dead) no more than two tribes in the demesne.
# Scope: the province.
nehekka_nomads_may_come_trigger = {
	OR = {
		region = world_nehekhara
		owner = { trait = creature_nehekharan }
	}
	NOT = { culture = nehekka }
	NOT = { has_province_modifier = nomad_presence }
	NOT = { has_province_flag = denied_nomads }
	owner = {
		NOR = {
			trait = creature_orc
			trait = creature_goblin
			trait = creature_beastman
			trait = creature_hobgoblin
		}
		OR = {
			NOT = { trait = creature_nehekharan }
			NOT = {
				any_demesne_province = {
					count = 2
					has_province_modifier = nomad_presence
				}
			}
		}
	}
}

# A roaming tribe (or a Nehekkan province) may send emissaries.
# Scope: the province.
nehekka_nomads_may_settle_trigger = {
	OR = {
		culture = nehekka
		has_province_modifier = nomad_presence
	}
}
