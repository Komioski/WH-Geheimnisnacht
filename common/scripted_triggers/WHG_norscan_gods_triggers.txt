# Norscan patron gods (events/WHG_norscan_gods_events.txt)

# The character already follows one of the six Norscan patron gods.
has_norscan_patron_trigger = {
	OR = {
		trait = norscan_patron_kharneth
		trait = norscan_patron_shornaal
		trait = norscan_patron_nurgleth
		trait = norscan_patron_tchar
		trait = norscan_patron_ulric
		trait = norscan_patron_ursun
	}
}

# The realm lets its people take a Dark God as patron (Kjorn, Loesh, Nierg,
# Tchar): anything but the Order Aligned Pantheon law.
norscan_dark_patron_allowed_trigger = {
	top_liege = { NOT = { has_law = sub_norse_tolerance_0 } }
}

# The realm lets its people take an old god as patron (Olric, Orsen):
# anything but the Dark Gods Aligned Pantheon law.
norscan_old_patron_allowed_trigger = {
	top_liege = { NOT = { has_law = sub_norse_tolerance_2 } }
}

# Tribes that lean to the old gods of the north: the bear-folk Baersonlings
# and Bjornlings, the Skaelings and the Sarl.
norscan_old_gods_culture_trigger = {
	OR = {
		culture = baersonling
		culture = bjornling
		culture = skaeling
		culture = sarl
	}
}
# A small Norscan chieftain whom the Dark Gods may call (norgods.20 - 24):
# a human, not sworn to Olric or Orsen, who has not refused the call before.
# The undead are not tempted.
norscan_dark_call_trigger = {
	trait = creature_human
	NOR = {
		has_character_flag = refusedcallofchaos
		trait = norscan_patron_ulric
		trait = norscan_patron_ursun
		trait = creature_nehekharan
		is_vampire_trigger = yes
	}
}

# A northern realm where only the strong may lead (norgods.30, 32): Norscan,
# Hung or Kurgan, of the Dark Gods, Olric or Orsen.
norscan_strength_realm_trigger = {
	OR = { culture_group = norscan_group culture_group = hung_group culture_group = kurgan_group }
	OR = { religion_group = chaos_gods_group religion = ursun religion = ulrican }
}
