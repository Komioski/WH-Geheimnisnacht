###############################################################################
# Holmgang (WHG_holmgang_events.txt)
###############################################################################

# The peoples who settle a grudge on the holm: Norscans of every faith and
# anyone who keeps the old Norscan gods.
# Scope: a character.
holmgang_people_trigger = {
	OR = {
		culture_group = norscan_group
		religion = norscan_gods
		religion = norscan_gods_reformed
	}
}

# Fit to row out to the holm and fight: grown, hale, free, reachable and not
# already in a holmgang. Women fight only as shieldmaidens, berserkers or
# duellists.
# Scope: a character.
holmgang_fighter_trigger = {
	is_alive = yes
	age = 16
	NOT = { age = 50 }
	prisoner = no
	is_inaccessible_trigger = no
	holmgang_people_trigger = yes
	OR = {
		is_woman_trigger = no
		trait = shieldmaiden
		trait = berserker
		trait = duelist
	}
	NOT = {
		trait = incapable
		trait = wounded
		is_maimed_trigger = yes
		is_ill = yes
		has_character_flag = holmgang_challenger
		has_character_flag = holmgang_defender
	}
}

# A lord who hates ROOT enough to call ROOT to the holm: no friend, no loyal
# servant, and no grudge already settled on the holm in the last years.
# Scope: the would-be challenger; ROOT is the man he hates.
holmgang_grudge_against_root_trigger = {
	is_ruler = yes
	NOT = { character = ROOT }
	holmgang_fighter_trigger = yes
	NOT = { opinion = { who = ROOT value = -40 } }
	NOT = { has_opinion_modifier = { who = ROOT modifier = opinion_holmgang_settled } }
	NOT = { reverse_has_opinion_modifier = { who = ROOT modifier = opinion_loyal_servant } }
}

###############################################################################
# Northern rites: the blot and the feasts after it (WHG_northern_rites_events)
###############################################################################

# The faiths that hold a blot: the old Norscan gods, the reformed faith and
# Norscans who serve the Dark Gods. Who shares it attends; who does not must
# pay to be excused (rites.2, rites.3). The hold_blot decision checks it too.
# Scope: a character.
rites_blot_faith_trigger = {
	OR = {
		religion = norscan_gods
		religion = norscan_gods_reformed
		AND = {
			culture_group = norscan_group
			religion_group = chaos_gods_group
		}
	}
}

# A vassal who can be summoned to a rite: grown, free, reachable, at peace.
# Scope: a vassal.
rites_summonable_vassal_trigger = {
	age = 16
	prisoner = no
	war = no
	is_inaccessible_trigger = no
	NOT = { trait = incapable }
}

# A prisoner ROOT may hang at the blot: grown, still in ROOT's dungeon, not
# already spared at this or an earlier blot.
# Scope: a courtier; ROOT is the host.
rites_blot_victim_trigger = {
	prisoner = yes
	host = { character = ROOT }
	age = 16
	NOT = { has_character_flag = rites_blot_victim }
	NOT = { has_opinion_modifier = { who = ROOT modifier = opinion_spared_at_blot } }
}

# A guest at a feast who can still make a scene.
# Scope: a vassal.
rites_feast_guest_trigger = {
	has_character_flag = rites_feast_guest
	is_alive = yes
	prisoner = no
}

# A guest who might find a lover at the feast: drawn to their own sex, no
# lover yet, no scene made yet. Races without women never take lovers.
# Scope: a vassal.
rites_feast_lover_trigger = {
	rites_feast_guest_trigger = yes
	NOT = { has_character_flag = rites_feast_did_scene }
	trait = homosexual
	has_lover = no
	is_no_females_race = no
}

# follower is below 15 in it.

		}
		AND = {
			NOT = { martial = 15 }
			OR = {
				has_character_modifier = follower_of_grimnir
				has_character_modifier = follower_of_tor
				has_character_modifier = follower_of_phakth
			}
		}
		AND = {
			NOT = { stewardship = 15 }
			OR = {
				has_character_modifier = follower_of_grungni
				has_character_modifier = follower_of_dazh
				has_character_modifier = follower_of_ptra

			NOT = { intrigue = 15 }
			OR = {
				has_character_modifier = follower_of_kalita

# The Great Hunt (WHG_great_hunt_events.txt)
###############################################################################

# A chieftain who must answer the horns: grown, a man, hale, free, reachable,
# at peace and above a baron.
# Scope: a vassal of the Beastlord.
herdhunt_can_hunt_trigger = {
	age = 16
	NOT = { age = 55 }
	is_woman_trigger = no
	prisoner = no
	war = no
	higher_tier_than = baron
	is_inaccessible_trigger = no
	NOT = {
		trait = incapable
		trait = wounded
		trait = infirm
		is_maimed_trigger = yes
	}
}

# A chieftain in the ring who has had no trouble yet this hunt.
# Scope: a vassal of the Beastlord.
herdhunt_free_hunter_trigger = {
	has_character_flag = herdhunt_hunter
	NOT = { has_character_flag = herdhunt_incident }
	is_alive = yes
	prisoner = no
	is_inaccessible_trigger = no
}

# A courtier the razorgor may maul: grown, free, not close kin or the
# heir, not the spouse, not leading troops.
# Scope: a courtier; ROOT is the owner.
herdhunt_maul_victim_trigger = {
	age = 16
	prisoner = no
	is_ruler = no
	is_primary_heir = no
	is_pretender = no
	in_command = no
	NOT = { is_close_relative = ROOT }
	NOT = { spouse = { character = ROOT } }
}

###############################################################################
# Berserkers (WHG_berserker_events.txt)
###############################################################################

# Can be taken by the berserker rage: Norscans, Norse dwarfs, Ulricans, the
# men of Kislev and all who keep the Norscan gods, but not the Khornate (they
# have their own berserkers), and young enough for it (as a human under 50,
# a dwarf under 200).
# Scope: a character.
berserk_can_rage_trigger = {
	is_alive = yes
	OR = {
		culture_group = norscan_group
		culture = dwarf_norse
		religion = ulrican
		religion = norscan_gods
		religion = norscan_gods_reformed
		AND = {
			religion = kislev_gods
			is_woman_trigger = no
		}
	}
	NOT = { religion = khorne }
	OR = {
		immortal = yes
		lifespan_age_50_trigger = no
	}
}

# Can be taken by the rage of Khorne: an open follower of the Blood God, no
# daemon, chaos knight or warlord. Khorne's berserkers never grow too old.
# Scope: a character.
berserk_khorne_can_rage_trigger = {
	is_alive = yes
	open_follower_of_khorne = yes
	is_demon_trigger = no
	NOR = {
		trait = chaos_knight
		trait = norscan_warlord
	}
}

# Has the rage: a berserker of the old gods or of Khorne.
# Scope: a character.
berserk_is_berserker_trigger = {
	OR = {
		trait = berserker
		trait = khorne_berserker
	}
}

# A courtier who drones on until the berserker's rage rises: grown, free, no
# kin, no heir, no berserker, and thinks little of ROOT.
# Scope: a courtier; ROOT is the berserker.
berserk_tiresome_courtier_trigger = {
	is_adult = yes
	prisoner = no
	berserk_is_berserker_trigger = no
	NOT = { trait = incapable }
	is_primary_heir = no
	is_pretender = no
	NOT = { is_close_relative = ROOT }
	NOT = { is_married = ROOT }
	NOT = { opinion = { who = ROOT value = 10 } }
}

# Seeresses and the tainted child (WHG_seeress_events.txt)
###############################################################################

# A lord whose seer is a hag of the stanitsa: the Ungol and Kislevite faiths
# among the Kislevite peoples (who may grant the post of wise woman).
# Scope: a character.
seeress_hag_faith_trigger = {
	culture_group = kislevite_group
	OR = {
		religion = northern_gods
		religion = northern_gods_reformed
		religion = kislev_gods
	}
}

# A lord whose seer is a Norscan seeress.
# Scope: a character.
seeress_seer_faith_trigger = {
	OR = {
		religion = norscan_gods
		religion = norscan_gods_reformed
	}
}

# A hag at court: the wise woman, or an Ungol priestess of the northern gods.
# Scope: a courtier.
seeress_court_hag_trigger = {
	is_alive = yes
	prisoner = no
	OR = {
		has_minor_title = title_wise_woman
		AND = {
			has_job_title = job_spiritual
			culture = ungol
			OR = {
				religion = northern_gods
				religion = northern_gods_reformed
			}
		}
	}
}

# A hag or a seeress at court.
# Scope: a courtier.
seeress_court_seer_trigger = {
	OR = {
		seeress_court_hag_trigger = yes
		AND = {
			has_character_flag = court_seeress
			is_alive = yes
			prisoner = no
		}
	}
}

# Keeps a hag or a seeress at court.
# Scope: a ruler.
seeress_seer_at_court_trigger = {
	any_courtier = { seeress_court_seer_trigger = yes }
}

# A courtier who may die near the tainted child: grown, free, not kin, not
# the heir or a spouse of the lord, not leading troops; an AI.
# Scope: a courtier of the father; ROOT is the child.
seeress_child_victim_trigger = {
	NOT = { character = ROOT }
	NOT = { is_close_relative = ROOT }
	ai = yes
	age = 13
	prisoner = no
	in_command = no
	is_primary_heir = no
	is_pretender = no
	host = { NOT = { is_married = PREV } }
}

###############################################################################
# Revolts (WHG_revolt_events.txt)
###############################################################################

# The people of this county can rise: the undead never complain, lizardmen
# are built to obey, and greenskin, beastman or jungle trouble already in the
# county comes instead of a revolt.
# Scope: a province.
revolt_county_can_rise_trigger = {
	NOT = { culture = nehekharan }
	NOT = { culture_group = lizardman_group }
	NOT = { culture_group = orc_group }
	NOT = { culture_group = goblin_group }
	NOT = { has_province_modifier = greenskin_population_bad }
	NOT = { has_province_modifier = beastmen_unrest }
	NOT = { has_province_modifier = beastman_herdstone }
	NOT = { has_province_modifier = lustria_oppressive_jungle }
}

# A county where no peasants rise but a rival may (revolt.50): the dead of
# Nehekhara, or greenskins (a greenskin county, or greenskins troubling it).
# Scope: the county (province).
revolt_rival_county_trigger = {
	OR = {
		culture = nehekharan
		culture_group = orc_group
		culture_group = goblin_group
		has_province_modifier = greenskin_population_bad
	}
}

# Rebels hold every barony of this county (the old rebel system, without
# The Old Gods: revolt.100 - revolt.120).
# Scope: the county (province).
revolt_county_rebel_held_trigger = {
	NOT = {
		any_province_lord = {
			any_demesne_title = {
				location = { province_id = PREVPREVPREV }
				tier = BARON
				NOT = { controlled_by = e_rebels }
			}
		}
	}
}
