# Triggers for events/WHG_old_world_faith_events.txt

# Character scope. A faith of the Old World with temples, cloisters and
# brotherhoods of priests: Sigmar, Ulric, Taal, Manann, Morr, Shallya,
# Myrmidia and the Lady (true and false). The Kislevite, Norscan and old
# faiths and Solkan's inquisitors keep no such houses.
temple_monastic_faith_trigger = {
	OR = {
		religion = sigmarite
		religion = ulrican
		religion = taalite
		religion = manannite
		religion = morrian
		religion = shallyan
		religion = myrmidian
		religion = grail
		religion = false_grail
	}
}

# Character scope. May come by a saint's relic: of a temple faith, holding
# none, and never had one before (once per lifetime).
temple_can_gain_relic_trigger = {
	temple_monastic_faith_trigger = yes
	NOT = { has_artifact = temple_relic }
	NOT = { has_character_flag = had_temple_relic }
}

# Character scope (a courtier), ROOT the ruler. ROOT has the sympathy trait
# for this courtier's faith group and is not of it.
zeal_sympathy_match_trigger = {
	NOT = { religion_group = ROOT }
	OR = {
		AND = { religion_group = old_world_group ROOT = { trait = sympathy_old_world_group } }
		AND = { religion_group = ormazd_gods_group ROOT = { trait = sympathy_ormazd_gods_group } }
		AND = { religion_group = cadai_group ROOT = { trait = sympathy_cadai_group } }
		AND = { religion_group = cytharai_group ROOT = { trait = sympathy_cytharai_group } }
		AND = { religion_group = dwarf_gods_group ROOT = { trait = sympathy_dwarf_gods_group } }
		AND = { religion_group = halfling_gods_group ROOT = { trait = sympathy_halfling_gods_group } }
		AND = { religion_group = oriental_gods_group ROOT = { trait = sympathy_oriental_gods_group } }
	}
}

# Character scope (a courtier), ROOT the ruler. A grown, free, capable
# courtier of a faith ROOT sympathises with, not yet a friend.
zeal_possible_friend_trigger = {
	prisoner = no
	is_adult = yes
	NOT = { trait = incapable }
	NOT = { is_friend = ROOT }
	zeal_sympathy_match_trigger = yes
}
