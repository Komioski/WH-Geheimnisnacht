###############################################################################
# Places of power: who should tend them, and which need it
# (the place of power notices, namespace pop in events/WHG_magic_events.txt)
#
# The two title decisions in decisions/wh_magic_decisions.txt work on a
# county's place of power: restore_pop (the faiths of the Old World and of
# Ulthuan's Cadai) and corrupt_pop (the Dark Gods, the necromancers and the
# Cytharai). These triggers mirror their faith gates and province checks, so
# a notice only goes out when the decision would show.
###############################################################################

# The character's faith restores places of power (restore_pop).
# Scope: a character.
pop_restorer_trigger = {
	OR = {
		true_religion_group = old_world_group
		true_religion_group = cadai_group
	}
}

# The character's faith corrupts places of power (corrupt_pop).
# Scope: a character.
pop_corrupter_trigger = {
	OR = {
		true_religion_group = cytharai_group
		true_religion_group = necromantic_group
		true_religion_group = chaos_gods_group
	}
}

# The place of power here is corrupt or dormant and not drained, so a
# restorer can turn it to the good.
# Scope: a province.
pop_needs_restoring_trigger = {
	NOT = { has_province_modifier = drained_place_of_power }
	OR = {
		has_province_modifier = corrupt_place_of_power_man
		has_province_modifier = corrupt_place_of_power_elf
		has_province_modifier = corrupt_place_of_power_wood
		has_province_modifier = inactive_place_of_power_man
		has_province_modifier = inactive_place_of_power_elf
		has_province_modifier = inactive_place_of_power_wood
	}
}

# The place of power here is pure or dormant and not drained, so a
# corrupter can turn it to the dark.
# Scope: a province.
pop_needs_corrupting_trigger = {
	NOT = { has_province_modifier = drained_place_of_power }
	OR = {
		has_province_modifier = active_place_of_power_man
		has_province_modifier = active_place_of_power_elf
		has_province_modifier = active_place_of_power_wood
		has_province_modifier = inactive_place_of_power_man
		has_province_modifier = inactive_place_of_power_elf
		has_province_modifier = inactive_place_of_power_wood
	}
}

# This place of power is one PREV would turn: a restorer at a corrupt or
# dormant one, a corrupter at a pure or dormant one.
# Scope: a province. PREV: the character.
pop_turnable_by_prev_trigger = {
	OR = {
		AND = {
			PREV = { pop_restorer_trigger = yes }
			pop_needs_restoring_trigger = yes
		}
		AND = {
			PREV = { pop_corrupter_trigger = yes }
			pop_needs_corrupting_trigger = yes
		}
	}
}
