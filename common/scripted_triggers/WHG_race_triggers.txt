###############################################################################
# WHG_race_triggers.txt: scripted triggers
# Subject: races: racial traits, which faiths and cultures fit which race,
# races without women
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Shared: used by three or more files (5)
#   Other helpers used by one or two files (8)
###############################################################################

# The faith of this scope (a character or a province) fits the race of PREV.
# Humans keep to human faiths: not the Amazon or Lizardmen gods, and the
# Nehekharan gods only for the living of Nehekhara (nehekka).
# Chaos tempts humans, elves and beastmen (any of the four gods or Chaos
# Undivided; for elves Nurgle and Khorne are kept rare by the callers, see
# elf_rare_chaos_faith_trigger); the other races keep to their own gods.
# Scope: the faith holder. PREV: the character who would convert.
# Used by: WHG_childhood_events, WHG_council_events +9 more
faith_fits_race_of_prev_trigger = {
	OR = {
		AND = {
			PREV = { culture_group = orc_group }
			religion = greenskin_gods
		}
		AND = {
			PREV = { culture_group = goblin_group }
			religion_group = greenskin_gods_group
		}
		AND = {
			PREV = { culture_group = skaven_group }
			religion = horned_rat
		}
		AND = {
			PREV = { culture_group = ogre_group }
			religion = great_maw
		}
		AND = {
			PREV = { culture_group = lizardman_group }
			religion = old_ones
		}
		AND = {
			PREV = { culture_group = beastman_group }
			chaos_god_faith_trigger = yes
		}
		AND = {
			PREV = { culture_group = forest_group }
			religion = asrai_cadai
		}
		AND = {
			PREV = { culture_group = asrai_group }
			OR = {
				religion = asrai_cadai
				chaos_god_faith_trigger = yes
			}
		}
		AND = {
			PREV = { culture_group = asur_group }
			OR = {
				religion = cadai
				religion = cult_of_pleasure
				chaos_god_faith_trigger = yes
			}
		}
		AND = {
			PREV = { culture_group = druchii_group }
			OR = {
				religion_group = cytharai_group
				chaos_god_faith_trigger = yes
			}
		}
		AND = {
			PREV = { culture_group = dwarf_group }
			religion_group = dwarf_gods_group
		}
		AND = {
			PREV = { culture_group = dwarf_chaos_group }
			religion = hashut
		}
		AND = {
			PREV = { OR = { culture = mootlander culture = gnome } }
			religion = halfling_gods
		}
		AND = {
			PREV = { OR = { culture = miyan culture_group = amazon_group } }
			OR = {
				religion = amazon
				religion = kukaili
			}
		}
		AND = {
			PREV = { culture = nehekharan }
			OR = {
				religion = nehekharan_gods
				religion = nagashi
			}
		}
		AND = {
			PREV = {
				is_human = yes
				NOT = { culture_group = amazon_group }
			}
			NOR = {
				religion = hashut
				religion = old_ones
				religion = amazon
				religion = kukaili
				AND = {
					religion_group = nehekharan_gods_group
					PREV = { NOT = { culture = nehekka } }
				}
				religion_group = halfling_gods_group
				religion_group = dwarf_gods_group
				religion_group = greenskin_gods_group
				religion_group = cadai_group
				religion_group = cytharai_group
				religion_group = great_maw_group
				religion_group = horned_rat_god_group
			}
		}
	}
}

# The culture of this scope may be taken on by PREV: both are of the same
# people, and the change keeps PREV's body and birth. Humans move between
# human cultures (not the Amazons); elves and dwarfs only within their own
# group (no Dawi becomes a Chaos Dwarf); goblin tribes swap but no
# hobgoblins; orc tribes swap but no black orcs; the great Skaven clans
# swap but the white, grey and black furs are born, not taught; Ogre tribes
# swap but Chaos Ogres do not. Beastmen, lizardmen, halflings, gnomes,
# Tomb Kings, vampires and the rest keep the culture they were born with.
# The caller must also check that PREV's faith fits the new culture.
# Scope: the culture holder. PREV: the character who would change.
# Used by: WHG_childhood_events, WHG_guardian_events, WHG_revolt_events
culture_fits_race_of_prev_trigger = {
	NOT = { culture = PREV }
	OR = {
		AND = {
			is_human = yes
			NOT = { culture_group = amazon_group }
			PREV = {
				is_human = yes
				NOT = { culture_group = amazon_group }
			}
		}
		AND = {
			OR = {
				culture_group = asur_group
				culture_group = druchii_group
				culture_group = asrai_group
				culture_group = dwarf_group
				culture_group = dwarf_chaos_group
			}
			culture_group = PREV
		}
		AND = {
			culture_group = goblin_group
			NOT = { culture = hobgoblin }
			PREV = {
				culture_group = goblin_group
				NOT = { culture = hobgoblin }
			}
		}
		AND = {
			culture_group = orc_group
			NOT = { culture = orc_black }
			PREV = {
				culture_group = orc_group
				NOT = { culture = orc_black }
			}
		}
		AND = {
			OR = {
				culture = skaven_main
				culture = eshin
				culture = moulder
				culture = pestilens
				culture = skryre
			}
			PREV = {
				OR = {
					culture = skaven_main
					culture = eshin
					culture = moulder
					culture = pestilens
					culture = skryre
				}
			}
		}
		AND = {
			OR = {
				culture = ogre_main
				culture = irongut_ogre
				culture = firegut_ogre
			}
			PREV = {
				OR = {
					culture = ogre_main
					culture = irongut_ogre
					culture = firegut_ogre
				}
			}
		}
	}
}

# Peoples who never fall ill: spirits, the dead, daemons (not Nurgle's),
# Slann and Saurus, greenskins, Fimir, Kdaai. add_symptom_effect checks it, and
# every direct add of an illness trait removes it again from them.
# Used by: WHG_disease_events, WHG_chaos_effects, WHG_east_effects +42 more
is_disease_immune_race = {
	NOR = { trait = nurgle_demon trait = nurgle_plaguebearer } # Nurgle's own carry every plague
	OR = {
		culture_group = daemonic_group
		culture_group = lizardman_group # Skinks, Saurus, Slann
		trait = creature_forestspirit
		trait = creature_treekin
		trait = creature_nehekharan
		trait = vampire_carstein_visible
		trait = vampire_blood_visible
		trait = vampire_strigoi_visible
		trait = vampire_necrarch_visible
		trait = vampire_jade_visible
		trait = vampire_mahtmasi_visible
		trait = vampire_lahmian_visible
		trait = vampire_carstein_hidden
		trait = vampire_blood_hidden
		trait = vampire_strigoi_hidden
		trait = vampire_necrarch_hidden
		trait = vampire_jade_hidden
		trait = vampire_mahtmasi_hidden
		trait = vampire_lahmian_hidden
		trait = khorne_demon
		trait = undivided_demon
		trait = slaanesh_demon
		trait = tzeentch_demon
		trait = hashut_kdaai
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
		trait = creature_snotling
		trait = creature_fimir
	}
}

# The single source for races without women as vanilla shows them. The
# engine still makes female characters for them, and each race uses them its
# own way (WHG_race_effects.txt): orc and goblin females become
# snotlings, Skaven females become breeders, lizardmen females become
# Saurus (males are Skinks), Fimir females become the court's one Maergh or
# turn male, and Tomb Kings are dead. None of them is a woman in an event's
# sense. Events never test is_female directly for a woman's role: they use
# is_woman_trigger below, and romance partners need is_no_females_race = no.
# Checks culture as well as creature traits, since snotlings and breeders
# carry no orc, goblin or Skaven trait.
# Used by: WHG_disease_events, WHG_rumour_events +83 more
is_no_females_race = {
	OR = {
		culture_group = orc_group
		culture_group = goblin_group
		culture_group = lizardman_group
		culture_group = skaven_group
		culture_group = fimir_group
		culture = nehekharan
		trait = creature_skink
		trait = creature_saurus
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
		trait = creature_fimir
		trait = creature_snotling
		trait = creature_nehekharan
		trait = creature_slann
		trait = creature_skaven
	}
}

# A woman as events mean it: female, and of a people that has women. Use it
# instead of is_female = yes in every woman's role (daughter, wife, mother,
# sister, lover, lady of the court) and in text variants; is_woman_trigger = no
# is everyone else, so male and female text variants still cover everyone.
# Used by: WHG_bloodline_events, WHG_slaanesh_events +132 more
is_woman_trigger = {
	is_female = yes
	is_no_females_race = no
}

# Holds an end of the roads east and a quarter of the wealth of one of the
# roads: enough to force Cathay's gates open.
# Scope: a character.
# Used by: WHG_cathay_decisions, WHG_cathay_grace_events
cathaygrace_masters_road_east_trigger = {
	any_realm_province = {
		OR = {
			province_id = 1188 # Nuur
			province_id = 2359 # Chimarak
			province_id = 2369 # Vargenspeak
			province_id = 2373 # Skrapland
		}
	}
	OR = {
		trade_route_control = {
			trade_route = ivory_road
			value >= 0.25
			type = wealth
			indirectly_owned = yes
		}
		trade_route_control = {
			trade_route = old_silk_road
			value >= 0.25
			type = wealth
			indirectly_owned = yes
		}
		trade_route_control = {
			trade_route = khyprian_road
			value >= 0.25
			type = wealth
			indirectly_owned = yes
		}
		trade_route_control = {
			trade_route = spice_route
			value >= 0.25
			type = wealth
			indirectly_owned = yes
		}
	}
}

# A realm the Jade Viceroy would break at a lord's request: an independent,
# settled king or emperor of 40 counties or more, near Cathay's lands or its
# tributaries, not at peace with Cathay and not of the Dragon House.
# Scope: a character (the target); FROM is the lord who asks.
# Used by: WHG_cathay_decisions
cathaygrace_invasion_target_trigger = {
	independent = yes
	higher_tier_than = DUKE
	num_of_count_titles_in_realm = 40
	is_nomadic = no
	in_revolt = no
	NOT = { character = FROM }
	NOT = { has_landed_title = e_cathay_west_governor }
	NOT = { has_character_modifier = peace_deal_with_cathay }
	NOT = { has_character_flag = defeated_cathay }
	NOT = { suzerain = { has_landed_title = e_cathay_west_governor } }
	NOT = { offmap_cathay = { offmap_ruler = { dynasty = PREVPREV } } }
	NOT = { primary_title = { has_title_flag = rebel_general_title } }
	OR = {
		valid_offmap_cathay_target_location = yes
		valid_offmap_cathay_target_location_tributary_chain = yes
	}
	custom_tooltip = {
		text = cathayan_request_invasion_rel_head_tt
		controls_religion = no
	}
}

# A lord near enough to Cathay to pledge tribute: next to the Viceroy or one
# of his tributaries, or with land in Cathay's sphere.
# Scope: a character.
# Used by: WHG_cathay_decisions
cathaygrace_can_pledge_tribute_trigger = {
	OR = {
		custom_tooltip = {
			text = cathayan_grace_become_tributary_allow_governor_neighbor_tt
			any_neighbor_independent_ruler = { has_landed_title = e_cathay_west_governor }
		}
		custom_tooltip = {
			text = cathayan_grace_become_tributary_allow_tributary_neighbor_tt
			any_neighbor_independent_ruler = {
				suzerain = { has_landed_title = e_cathay_west_governor }
			}
		}
		custom_tooltip = {
			text = cathayan_grace_become_tributary_allow_cathayan_influence_tt
			any_realm_province = { region = custom_cathayan_invasion_target_region }
		}
	}
}

# An elf (PREV) would fall to Nurgle or Khorne, which is much rarer than
# Slaanesh or Tzeentch: callers let it through about one time in ten.
# Scope: the faith holder. PREV: the elf who would convert.
# Used by: WHG_childhood_events, WHG_council_events
elf_rare_chaos_faith_trigger = {
	PREV = { is_elf = yes }
	OR = {
		religion = nurgle
		religion = khorne
	}
}

# Character scope. Yes when the character is married across a line the
# races do not cross: Men may wed Men and Amazons; Dawi may wed Dawi and
# Chaos Dwarfs; elves may wed only elves, unless the other spouse is a
# player; beastmen, halflings, ogres and skaven wed only their own kind;
# orcs, goblins, saurus, skinks and Slann do not marry at all. Used by
# aiclean.54 - 55 to undo such marriages for the AI.
# Used by: WHG_maintenance_events
racemarry_forbidden_trigger = {
	OR = {
		AND = {
			trait = creature_human
			any_spouse = {
				NOR = {
					trait = creature_human
					trait = creature_amazon
				}
			}
		}
		AND = {
			trait = creature_beastman
			any_spouse = { NOT = { trait = creature_beastman } }
		}
		AND = {
			trait = creature_dwarf
			any_spouse = {
				NOR = {
					trait = creature_dwarf
					trait = creature_chaos_dwarf
				}
			}
		}
		AND = {
			trait = creature_elf
			any_spouse = {
				NOT = { trait = creature_elf }
				ai = yes
			}
		}
		AND = {
			trait = creature_halfling
			any_spouse = { NOT = { trait = creature_halfling } }
		}
		AND = {
			trait = creature_ogre
			any_spouse = { NOT = { trait = creature_ogre } }
		}
		AND = {
			trait = creature_skaven
			any_spouse = { NOT = { trait = creature_skaven } }
		}
		trait = creature_orc
		trait = creature_goblin
		trait = creature_saurus
		trait = creature_skink
		trait = creature_slann
	}
}

# A commander of ROOT's realm who was not trained in Cathayan ways and can
# make room for a Cathayan.
# Scope: a character; ROOT is the lord.
# Used by: WHG_race_effects
cathaygrace_replaceable_commander_trigger = {
	has_minor_title = title_commander
	liege = { character = ROOT }
	NOR = {
		has_character_flag = originated_from_cathayan_court
		has_character_flag = taught_cathayan_strategy
	}
}

# The faith of this scope is one of the Chaos gods or Chaos Undivided (not
# Hashut or the Norscan gods, who share the religion group).

	OR = {
		religion = chaos
		religion = khorne
		religion = nurgle
		religion = tzeentch
		religion = slaanesh
	}
}
