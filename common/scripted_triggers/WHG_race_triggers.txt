###############################################################################
# WHG_race_triggers.txt - which faiths and cultures fit which race
#
# Used wherever a character takes on someone else's faith or culture (a
# child from the educator, a courtier from the province). Every race only
# takes faiths it can follow, so aiclean.32 (has_wrong_religion) never has to
# undo a conversion, and greenskins never leave the greenskin faiths.
###############################################################################

# The faith of this scope (a character or a province) fits the race of PREV.
# Chaos tempts humans, elves and beastmen (any of the four gods or Chaos
# Undivided; for elves Nurgle and Khorne are kept rare by the callers, see
# elf_rare_chaos_faith_trigger); the other races keep to their own gods.
# Scope: the faith holder. PREV: the character who would convert.
faith_fits_race_of_prev_trigger = {
	OR = {
		AND = {
			PREV = { culture_group = orc_group }
			religion = greenskin_gods
		}
		AND = {
			PREV = { culture_group = goblin_group }
			religion_group = greenskin_gods_group
		}
		AND = {
			PREV = { culture_group = skaven_group }
			religion = horned_rat
		}
		AND = {
			PREV = { culture_group = ogre_group }
			religion = great_maw
		}
		AND = {
			PREV = { culture_group = lizardman_group }
			religion = old_ones
		}
		AND = {
			PREV = { culture_group = beastman_group }
			chaos_god_faith_trigger = yes
		}
		AND = {
			PREV = { culture_group = forest_group }
			religion = asrai_cadai
		}
		AND = {
			PREV = { culture_group = asrai_group }
			OR = {
				religion = asrai_cadai
				chaos_god_faith_trigger = yes
			}
		}
		AND = {
			PREV = { culture_group = asur_group }
			OR = {
				religion = cadai
				religion = cult_of_pleasure
				chaos_god_faith_trigger = yes
			}
		}
		AND = {
			PREV = { culture_group = druchii_group }
			OR = {
				religion_group = cytharai_group
				chaos_god_faith_trigger = yes
			}
		}
		AND = {
			PREV = { culture_group = dwarf_group }
			religion_group = dwarf_gods_group
		}
		AND = {
			PREV = { culture_group = dwarf_chaos_group }
			religion = hashut
		}
		AND = {
			PREV = { OR = { culture = mootlander culture = gnome } }
			religion = halfling_gods
		}
		AND = {
			PREV = { OR = { culture = miyan culture = amazon_main } }
			OR = {
				religion = amazon
				religion = kukaili
			}
		}
		AND = {
			PREV = { culture = nehekharan }
			OR = {
				religion = nehekharan_gods
				religion = nagashi
			}
		}
		AND = {
			PREV = {
				is_human = yes
				NOT = { culture = amazon_main }
			}
			NOR = {
				religion = hashut
				religion = old_ones
				religion_group = halfling_gods_group
				religion_group = dwarf_gods_group
				religion_group = greenskin_gods_group
				religion_group = cadai_group
				religion_group = cytharai_group
				religion_group = great_maw_group
				religion_group = horned_rat_god_group
			}
		}
	}
}

# The culture of this scope may be taken on by PREV: both are of the same
# people, and the change keeps PREV's body and birth. Humans move between
# human cultures (not the Amazons); elves and dwarfs only within their own
# group (no Dawi becomes a Chaos Dwarf); goblin tribes swap but no
# hobgoblins; orc tribes swap but no black orcs; the great Skaven clans
# swap but the white, grey and black furs are born, not taught; Ogre tribes
# swap but Chaos Ogres do not. Beastmen, lizardmen, halflings, gnomes,
# Tomb Kings, vampires and the rest keep the culture they were born with.
# The caller must also check that PREV's faith fits the new culture.
# Scope: the culture holder. PREV: the character who would change.
culture_fits_race_of_prev_trigger = {
	NOT = { culture = PREV }
	OR = {
		AND = {
			is_human = yes
			NOT = { culture_group = amazon_group }
			PREV = {
				is_human = yes
				NOT = { culture_group = amazon_group }
			}
		}
		AND = {
			OR = {
				culture_group = asur_group
				culture_group = druchii_group
				culture_group = asrai_group
				culture_group = dwarf_group
				culture_group = dwarf_chaos_group
			}
			culture_group = PREV
		}
		AND = {
			culture_group = goblin_group
			NOT = { culture = hobgoblin }
			PREV = {
				culture_group = goblin_group
				NOT = { culture = hobgoblin }
			}
		}
		AND = {
			culture_group = orc_group
			NOT = { culture = orc_black }
			PREV = {
				culture_group = orc_group
				NOT = { culture = orc_black }
			}
		}
		AND = {
			OR = {
				culture = skaven_main
				culture = eshin
				culture = moulder
				culture = pestilens
				culture = skryre
			}
			PREV = {
				OR = {
					culture = skaven_main
					culture = eshin
					culture = moulder
					culture = pestilens
					culture = skryre
				}
			}
		}
		AND = {
			OR = {
				culture = ogre_main
				culture = irongut_ogre
				culture = firegut_ogre
			}
			PREV = {
				OR = {
					culture = ogre_main
					culture = irongut_ogre
					culture = firegut_ogre
				}
			}
		}
	}
}

# The faith of this scope is one of the Chaos gods or Chaos Undivided (not
# Hashut or the Norscan gods, who share the religion group).
chaos_god_faith_trigger = {
	OR = {
		religion = chaos
		religion = khorne
		religion = nurgle
		religion = tzeentch
		religion = slaanesh
	}
}

# An elf (PREV) would fall to Nurgle or Khorne, which is much rarer than
# Slaanesh or Tzeentch: callers let it through about one time in ten.
# Scope: the faith holder. PREV: the elf who would convert.
elf_rare_chaos_faith_trigger = {
	PREV = { is_elf = yes }
	OR = {
		religion = nurgle
		religion = khorne
	}
}
