###############################################################################
# WHG_realm_triggers.txt: scripted triggers
# Subject: the realm: holdings, provinces, laws, economy, slaves, republics,
# revolts, wonders, alternate start
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Shared: used by three or more files (10)
#   Used by WHG_culture_events (4)
#   Used by WHG_maintenance_events (4)
#   Used by WHG_realm_effects (7)
#   Used by WHG_realm_events (10)
#   Used by WHG_republic_events (6)
#   Used by WHG_wonder_events (9)
#   Other helpers used by one or two files (27)
#   Not used by any live script yet: kept for content to come (2)
###############################################################################

# ROOT, or the duchy, kingdom or empire ROOT holds over this province,
# carries ROOT's culture as its conquest culture.
# Scope: a province owned by ROOT.
# Used by: WHG_culture_events, WHG_realm_triggers
culture_conquest_culture_trigger = {
	OR = {
		ROOT = { primary_title = { conquest_culture = ROOT } }
		county = { conquest_culture = ROOT }
		duchy = { holder_scope = { character = ROOT } conquest_culture = ROOT }
		kingdom = { holder_scope = { character = ROOT } conquest_culture = ROOT }
		empire = { holder_scope = { character = ROOT } conquest_culture = ROOT }
	}
}

# A province of ROOT's realm that may take up ROOT's culture from its
# neighbours (culture.2).
# Scope: a province owned by ROOT.
# Used by: WHG_culture_events
culture_assimilates_trigger = {
	NOT = { culture_group = skaven_group }
	culture_fixed_culture_trigger = no
	NOT = { culture = ROOT }
	religion = ROOT
	culture_has_neighbour_of_root_culture_trigger = yes
}

# ROOT's capital, cut off from ROOT's culture, may draw settlers from the
# conquerors' homeland (culture.5).
# Scope: ROOT's capital province.
# Used by: WHG_culture_events
culture_settlers_trigger = {
	culture_fixed_culture_trigger = no
	NOT = { culture = ROOT }
	OR = {
		port = yes
		NOR = {
			has_game_rule = { name = culture_conversion value = restricted }
			has_game_rule = {
				name = culture_conversion
				value = faster_melting_pots_and_slower_normal
			}
		}
	}
	culture_conquest_culture_trigger = yes
	culture_has_neighbour_of_root_culture_trigger = no
}

# The "slower normal conversion" culture conversion game rule is on.
# Used by: WHG_culture_events
culture_slower_conversion_trigger = {

		has_game_rule = { name = culture_conversion value = slower_normal }
		has_game_rule = { name = culture_conversion value = faster_melting_pots_and_slower_normal }
	}
}

		primary_title = { temporary = yes }
	}

		character = FROMFROM
		has_law = succ_military_elective
	}
	OR = {
		AND = {
			OR = {
				male_can_hold_minor_title_trigger = yes
				female_can_hold_minor_commander_title_trigger = yes
			}
			is_adult = yes
		}
		has_character_flag = special_marshal
	}

	prisoner = no
	is_incapable = no
 	NOT = { trait = sick_incapable }
	NOT = { is_inaccessible_trigger = yes }
	NOT = { has_character_flag = guru }
	NOT = { has_character_modifier = refusing_to_lead }
	NOR = {
		war_with = FROMFROM
		any_liege = {
			war_with = FROMFROM
		}
	}
	OR = {
		higher_real_tier_than = BARON #Either potential commander is vassal...
		NOT = { #...Or there are no valid vassal candidates that can be commanders.
			custom_tooltip = {
				text = tooltip_byzantine_empire_can_appoint_commanders
				liege = {
					any_vassal = {
						higher_real_tier_than = BARON
						NOT = {
							primary_title = { temporary = yes }
						}
						OR = {
							AND = {
								OR = {
									male_can_hold_minor_title_trigger = yes
									female_can_hold_minor_commander_title_trigger = yes
								}
								is_adult = yes
							}
							has_character_flag = special_marshal
						}

						prisoner = no
						is_incapable = no
						NOT = { is_inaccessible_trigger = yes }
						NOT = { has_character_modifier = refusing_to_lead }
						NOR = {
							has_job_title = job_chancellor

							has_job_title = job_spiritual

	}
}
