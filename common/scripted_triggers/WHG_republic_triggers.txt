# Merchant republics (events/WHG_republic_events.txt)
#
# republic_bachelor_trigger            courtier scope, ROOT the patrician: an
#                                      unmarried AI man of the house, not
#                                      the patrician's own son (trade.399)
# republic_trade_post_razable_trigger  province scope: a patrician's trade
#                                      post in a foreign realm whose ruler
#                                      dislikes the owner and has no pact,
#                                      war or past deal with them (trade.599)
# republic_election_law_trigger        doge scope: the republic holds
#                                      elections (term laws 0 - 5)
# republic_election_candidate_trigger  vassal scope: a patrician head who can
#                                      stand against the doge
# republic_election_running_trigger    candidate scope: the election this
#                                      candidate stands in is still running
###############################################################################

republic_bachelor_trigger = {
	ai = yes
	age = 18
	is_woman_trigger = no
	is_no_females_race = no
	is_married = no
	is_betrothed = no
	prisoner = no
	NOT = { trait = incapable }
	is_primary_heir = no
	is_pretender = no
	is_theocracy = no
	is_priest = no
	NOT = { trait = chaste }
	NOT = { has_character_flag = getting_married }
	dynasty = ROOT
	NOT = { father = { character = ROOT } }
	NOT = { mother = { character = ROOT } }
}

republic_trade_post_razable_trigger = {
	has_trade_post = yes
	owner = { is_patrician = no top_liege = { is_patrician = no } }

	trade_post_owner = {
		is_patrician = yes
		PREV = {
			owner = {
				NOT = { any_realm_character = { character = PREVPREVPREV } }
				top_liege = {
					NOR = {
						is_vassal_or_below = PREVPREVPREV #this is backwards
						opinion = { who = PREVPREVPREV value = 0 }
						has_opinion_modifier = { who = PREVPREVPREV modifier = opinion_bribed_tp }
						has_opinion_modifier = {
							who = PREVPREVPREV
							modifier = opinion_thwarted_tp_destruction
						}
						has_opinion_modifier = { who = PREVPREVPREV modifier = opinion_burned_tp }
						has_opinion_modifier = {
							who = PREVPREVPREV
							modifier = opinion_burned_tp_despite_bribe
						}
						war_with = PREVPREVPREV
					}
				}

				# Only if there is no non-aggression-pact with the trade post owner
				NOT = { has_non_aggression_pact_with = PREVPREV }
			}
		}
	}
}

# The republics whose title history sets an election term law (Komi, Nov 12:
# elections belong to these, not to every merchant republic). The term laws
# in wh_republic_laws.txt are only open to them.
# Scope: a title.
republic_term_title_trigger = {
	OR = {
		title = e_communards
		title = d_pigbarter
		title = k_norvard
		title = k_verezzo
		title = k_mootland
		title = d_boegenhafen
	}
}

# The doge's republic holds elections: one of the term laws of 4 to 20
# years. Dictatorship (law_election_cycles_6) and the Drachau (7) hold none.
# Scope: the doge.
republic_election_law_trigger = {
	primary_title = {
		republic_term_title_trigger = yes
		OR = {
			has_law = law_election_cycles_0
			has_law = law_election_cycles_1
			has_law = law_election_cycles_2
			has_law = law_election_cycles_3
			has_law = law_election_cycles_4
			has_law = law_election_cycles_5
		}
	}
}

# A patrician head who can stand against the doge (election.1).
# Scope: the vassal.
republic_election_candidate_trigger = {
	is_patrician = yes
	is_adult = yes
	prisoner = no
	is_inaccessible_or_incapable_trigger = no
}

# The election this candidate stands in is still running: the doge who
# called it lives and has not lost the seal.
# Scope: the candidate. event_target:election_doge is the doge.
republic_election_running_trigger = {
	has_character_flag = election_candidate
	event_target:election_doge = {
		is_alive = yes
		has_character_flag = election_running
	}
}
