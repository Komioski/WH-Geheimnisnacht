#   sea_voice_<people>_trigger     character scope: who the captain is
#   manann_mood_*_trigger          the captain's luck with the sea
#   sea_can_meet_*_trigger         the "no meeting yourself" rule
###############################################################################

###############################################################################
# Voices: who sails. Each power and event reads the voice, so a Druchii raid,
# an Orc raid and a Vampire Coast raid feel different without five copies of
# every event. The first that fits wins, in this order: Druchii, Vampire
# Coast, Orc freebootaz, Norse, Araby, and Sartosa for every other captain.
# Text in localisation/customizable_localisation/WHG_sea_voices.txt.
# Scope: a character.
###############################################################################

sea_voice_druchii_trigger = {
	OR = {
		religion_group = cytharai_group
		culture_group = druchii_group
	}
}

sea_voice_vampire_trigger = {
	NOT = { sea_voice_druchii_trigger = yes }
	OR = {
		religion = vampiric
		is_vampire_trigger = yes
	}
}

sea_voice_orc_trigger = {
	NOT = { sea_voice_druchii_trigger = yes }
	NOT = { sea_voice_vampire_trigger = yes }
	OR = {
		government = freeboota_government
		culture = orc_freeboota
		culture = goblin_seagit
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
	}
}

sea_voice_norse_trigger = {
	NOT = { sea_voice_druchii_trigger = yes }
	NOT = { sea_voice_vampire_trigger = yes }
	NOT = { sea_voice_orc_trigger = yes }
	OR = {
		culture_group = norscan_group
		religion = norscan_gods
		religion = norscan_gods_reformed
	}
}

sea_voice_araby_trigger = {
	NOT = { sea_voice_druchii_trigger = yes }
	NOT = { sea_voice_vampire_trigger = yes }
	NOT = { sea_voice_orc_trigger = yes }
	NOT = { sea_voice_norse_trigger = yes }
	OR = {
		religion = atruhayid
		religion = atruhayid_reformed

	}
}

# Who is a sea captain

# A ruler whose seat is a port, or a courtier whose liege's seat is.
# Scope: a character.
sea_coastal_trigger = {
	OR = {
		AND = {
			is_landed = yes
			capital_scope = { port = yes }
		}
		AND = {
			is_landed = no
			liege = { capital_scope = { port = yes } }
		}
	}
}

# A Norscan of the Freigattur who lives on the coast: the raider path.
# Scope: a character.
sea_norse_raider_trigger = {
	society_member_of = norscan_society
	sea_coastal_trigger = yes
}

# Any captain of the sea societies: the Black Ark Corsairs, the Wolves of the
# Sea, and the coastal Norscans of the Freigattur.
# Scope: a character.
sea_captain_trigger = {
	OR = {
		society_member_of = adventurers_druchii_corsairs
		society_member_of = adventurers_pirates
		sea_norse_raider_trigger = yes
	}
}

# The captain may put to sea now: free, grown, capable, and not hiding.
# Scope: a character.
sea_captain_free_trigger = {
	is_adult = yes
	prisoner = no
	is_inaccessible_trigger = no
	NOT = { trait = incapable }
	NOT = { has_character_flag = sea_on_voyage }
	NOT = { has_character_flag = castaway_on_island }
}

###############################################################################
# Manann's mood
# Every captain carries a hidden variable manann_mood, from -5 (the sea wants
# them dead) to 5 (fair winds for life). Set Out, the voyage legend and the
# ship events read it. It rises with sacrifices, a priest aboard and a good
# ship's cat, and falls with a shot albatross, a renamed ship or a curse
# thrown at the waves (Sea of Claws, "Events at Sea"). Voices name their own
# power: Manann, Stromfels, Mathlann, Gork, the Drowned, the Norse gods.
# Scope: a character.
###############################################################################

manann_mood_good_trigger = {
	check_variable = { which = manann_mood value = 2 }
}

manann_mood_bad_trigger = {
	NOT = { check_variable = { which = manann_mood value = -1 } }
}

manann_mood_dire_trigger = {
	NOT = { check_variable = { which = manann_mood value = -3 } }
}

# A named sea lord from the lore (Luthor Harkon, Lokhir Fellheart, Duriath
# Helbane, the jarl of Skeggi) may be met in person: alive, not the captain,
# not in the captain's realm, not held, and not in the same society crew.
# Only lords our history has alive at the 2010 start are used by name; later
# heroes (Wulfrik, Lucciano Soprania, Olg Bloodsalt) never appear.
# Scope: the named lord. ROOT is the captain.
sea_named_lord_can_meet_trigger = {
	is_alive = yes
	NOT = { character = ROOT }
	NOT = { same_realm = ROOT }
	prisoner = no
	NOT = { same_society_as = ROOT }
}

# The named lord is friendly to the captain: an ally, a friend, or a fellow
# of the same faith with a good opinion. They share news and prizes instead
# of fighting.
# Scope: the named lord. ROOT is the captain.
sea_named_lord_friendly_trigger = {
	OR = {
		is_allied_with = ROOT
		is_friend = ROOT
		AND = {
			religion = ROOT
			opinion = { who = ROOT value = 25 }
		}
	}
}

	any_artifact = { has_artifact_flag = flagship }
}

# Five years since the last refit (or never refitted).
# Scope: a character.
sea_refit_ready_trigger = {
	OR = {
		NOT = { has_character_flag = sea_refit }
		had_character_flag = { flag = sea_refit days = 1825 }
	}
}

# Enough behind them for the next refit: three deeds at sea for the middle
# ship, six for the great one. A lord outside the sea societies, who keeps no
# tally of deeds, needs the fame instead.
# Scope: a character.
sea_refit_deeds_trigger = {
	trigger_if = {
		limit = { has_artifact = regular_flagship }
		OR = {
			check_variable = { which = sea_deeds value = 6 }
			AND = {
				sea_captain_trigger = no
				prestige = 2000
			}
		}
	}
	trigger_else_if = {
		limit = { has_artifact = small_flagship }
		OR = {
			check_variable = { which = sea_deeds value = 3 }
			AND = {
				sea_captain_trigger = no
				prestige = 750
			}
		}
	}
	trigger_else = {
		always = yes
	}
}
