# WHG_sea_triggers.txt: scripted triggers
# Subject: the sea: ships, voyages, pirates, the sea societies
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Shared: used by three or more files (15)
#   Used by WHG_sea_effects (33)
#   Used by WHG_sea_events (14)
#   Used by WHG_sea_society_events (13)
#   Other helpers used by one or two files (9)
#   Not used by any live script yet: kept for content to come (3)
###############################################################################

# A ruler whose seat is a port, or a courtier whose liege's seat is.
# Scope: a character.
# Used by: WHG_sea_triggers, WHG_sea_decisions, WHG_warrior_societies
sea_coastal_trigger = {
	OR = {
		AND = {
			is_landed = yes
			capital_scope = { port = yes }
		}
		AND = {
			is_landed = no
			liege = { capital_scope = { port = yes } }
		}
	}
}

# May be wrecked into the castaway's saga: a player, once a game, count or
# duke, seat on the coast, a mortal Man, Dawi, elf or orc who has not yet
# played Death at chess. Callers roll their chance first; castaway.1 then
# looks for a free island (castaway_island_trigger).
# Scope: a character.
# Used by: WHG_sea_events, WHG_castaway_events, WHG_naval_events
castaway_candidate_trigger = {
	ai = no
	NOT = { has_global_flag = castaway_saga_begun }
	NOT = { has_character_flag = chess_with_death }
	is_adult = yes
	prisoner = no
	is_incapable = no
	OR = {
		real_tier = COUNT
		real_tier = DUKE
	}
	capital_scope = { port = yes }
	OR = {
		trait = creature_human
		trait = creature_dwarf
		trait = creature_elf
		trait = creature_orc
	}
	is_undead_trigger = no
	is_daemon_trigger = no
	immortal = no
	current_heir = { always = yes }
}

# A named sea lord from the lore (Luthor Harkon, Lokhir Fellheart, Duriath
# Helbane, the jarl of Skeggi) may be met in person: alive, not the captain,
# not in the captain's realm, not held, and not in the same society crew.
# Only lords our history has alive at the 2010 start are used by name; later
# heroes (Wulfrik, Lucciano Soprania, Olg Bloodsalt) never appear.
# Scope: the named lord. ROOT is the captain.
# Used by: WHG_sea_effects, WHG_sea_events
sea_named_lord_can_meet_trigger = {
	is_alive = yes
	NOT = { character = ROOT }
	NOT = { same_realm = ROOT }
	prisoner = no
	NOT = { same_society_as = ROOT }
}
