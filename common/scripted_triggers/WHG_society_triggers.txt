# WHG_society_triggers.txt: scripted triggers
# Subject: societies: warrior lodges and warbands, the assassins, bloodlines of
# the orders, other orders not tied to one race
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Shared: used by three or more files (33)
#   Used by WHG_assassins_events (12)
#   Building blocks of other entries in this file (8)
#   Used by WHG_warrior_society_events (40)
#   Other helpers used by one or two files (21)
#   Not used by any live script yet: kept for content to come (1)
###############################################################################

# Scope: a top liege; ROOT is a member. A realm of unbelievers next to ROOT
# with a temple or castle held by one: the order may seize it
# (assassins.221). A realm of a sister faith only counts when ROOT has no
# true unbelievers for neighbours.
# Used by: WHG_assassins_events
assassins_unbeliever_realm_trigger = {
	OR = {
		NOT = { religion_group = ROOT }
		AND = {
			NOT = { religion = ROOT }
			ROOT = {
				NOT = {
					any_realm_province = {
						any_neighbor_province = {
							owner = { top_liege = { NOT = { religion_group = ROOT } } }
						}
					}
				}
			}
		}
	}
	any_realm_province = {
		any_province_holding = {
			owner = { NOT = { religion = ROOT } }
			OR = {
				holding_type = temple
				holding_type = castle
			}
		}
	}
}
