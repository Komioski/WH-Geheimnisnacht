# Triggers for events/WHG_society_events.txt (the running of the societies)

# Scope: a character. Shows interest in a society that sends a recruiter to
# those who are interested (society.10, society.11).
society_recruits_by_interest_trigger = {
	OR = {
		interested_in_society = the_assassins
		interested_in_society = eshin_society
		interested_in_society = cult_of_khorne
		interested_in_society = cult_of_nurgle
		interested_in_society = cult_of_tzeentch
		interested_in_society = cult_of_slaanesh
		interested_in_society = cult_of_undivided
		interested_in_society = cult_of_pleasure
	}
}

# Scope: a character. An AI who may be pushed into a society (society.30).
society_ai_can_join_trigger = {
	ai = yes
	prisoner = no
	NOT = { trait = incapable }
	is_adult = yes
	is_in_society = no
	controls_religion = no
	block_general_event_trigger = no
}

# Scope: a character. Follows a faith whose head can cast out the faithless,
# and is not that head.
society_faith_excommunicates_trigger = {
	controls_religion = no
	NOT = { trait = excommunicated }
	OR = {
		religion = sigmarite
		religion = myrmidian
		religion = morrian
		religion = grail
		religion = solkanite
	}
	religion_head = { always = yes }
}

# Scope: a society member. A society whose members are hunted by the law
# or the faith if found out, so its letters of expulsion threaten.
society_hostile_expulsion_trigger = {
	OR = {
		society_member_of = the_assassins
		society_member_of = eshin_society
		is_monastic_member_trigger = yes
		is_devil_worshiper_trigger = yes
	}
}

# Shieldmaidens (WHG_warrior_society_events, namespace shieldmaiden)
# ROOT is a shieldmaiden in battle whose liege fights in the same place
# (shieldmaiden.10). She must not already be marked by an earlier fight.
shieldmaiden_can_guard_liege_trigger = {
	trait = shieldmaiden
	prisoner = no
	NOT = { trait = incapable }
	NOR = {
		has_character_modifier = favored_shieldmaiden
		has_character_modifier = bad_shieldmaiden
		has_character_modifier = clumsy_shieldmaiden
		has_character_modifier = inattentive_soldier
	}
	liege = {
		NOT = { character = ROOT }
		at_location = ROOT
		age = 16
		prisoner = no
		NOT = { has_character_modifier = lucky_to_be_alive }
	}
}

# ROOT is a liege in battle with a shieldmaiden commander at the same
# place (shieldmaiden.20).
shieldmaiden_fights_beside_me_trigger = {
	age = 16
	NOT = { trait = incapable }
	any_courtier_or_vassal = {
		trait = shieldmaiden
		NOT = { trait = incapable }
		prisoner = no
		has_minor_title = title_commander
		at_location = ROOT
	}
}
