###############################################################################
# WHG_south_triggers.txt: scripted triggers
# Subject: the South and the West: Araby, Lustria, Albion, the Southlands,
# colonies and ventures
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Shared: used by three or more files (2)
#   Used by WHG_albion_events (6)
#   Used by WHG_araby_events (7)
#   Used by WHG_lizardmen_events (4)
#   Used by WHG_sea_society_events (4)
#   Building blocks of other entries in this file (4)
#   Used by WHG_venture_decisions (5)
#   Used by WHG_venture_events (14)
#   Other helpers used by one or two files (14)
#   Not used by any live script yet: kept for content to come (6)
###############################################################################

# Province scope. Yes when the province lies in ROOT's homeland, the lands
# the colony decisions leave alone. The homelands match the decisions in
# decisions/dow_new_world.txt: Ulthuan for the Phoenix King, the Old World
# with the north, the mountains and the Badlands for an Old World capital,
# and the capital's own part of the world for the others. A capital
# anywhere else counts every province as homeland, so nothing is taken.
# Used by: WHG_south_effects
colony_homeland_trigger = {
	OR = {
		AND = {
			ROOT = { has_landed_title = e_ulthuan }
			region = world_ulthuan
		}
		AND = {
			ROOT = {
				NOT = { has_landed_title = e_ulthuan }
				capital_scope = { region = world_old_world }
			}
			OR = {
				region = world_old_world
				region = world_norsca
				region = world_karaz_ankor
				region = world_drak_ankor
				region = world_badlands
				region = world_chaos_wastes
			}
		}
		AND = {

# A coastal province where a colony could be planted, found by a
# successful expedition (venture_survey_site_effect).
# Scope: a province. ROOT is the Venturer.
# Used by: WHG_south_effects
venture_site_trigger = {
	port = yes
	NOT = { owned_by = ROOT }
	NOT = { has_province_flag = venture_surveyed }
	OR = {
		AND = {
			ROOT = {
				OR = {
					has_character_flag = venture_dest_coast
					has_character_flag = venture_dest_temple
				}
			}
			region = world_new_world_lustria
		}
		AND = {
			ROOT = { has_character_flag = venture_dest_southlands }
			OR = {
				region = world_greater_araby_southlands
				region = world_eastern_southlands
			}
		}
		AND = {
			ROOT = { has_character_flag = venture_dest_naggaroth }
			OR = {
				region = world_new_world_naggaroth
				region = world_new_world_black_forest
			}
		}
	}
}

# May raise colony buildings (power venturer_colony_works): a Venturer of
