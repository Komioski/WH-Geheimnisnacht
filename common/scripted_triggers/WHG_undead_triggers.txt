# Subject: the undead: vampires, necromancers, Mousillon, the Tomb Kings of
# Nehekhara
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Shared: used by three or more files (14)
#   Used by WHG_undead_events (4)
#   Used by WHG_vampire_events (10)
#   Used by WHG_vampire_societies (10)
#   Other helpers used by one or two files (9)

# A weapon the Dark Enchantment can work on: a fine sword, lance, spear or
# axe that is not necromantic already. Legendary weapons are left alone.
# Scope: an artifact.
# Used by: WHG_undead_events, wh_undead_decisions
undead_enchantable_weapon_trigger = {
	has_artifact_flag = combat_weapon
	OR = {
		has_artifact_flag = sword
		has_artifact_flag = lance
		has_artifact_flag = spear
		has_artifact_flag = axe
	}
	NOT = { has_artifact_flag = necromantic_item }
	OR = {
		quality == 2
		quality == 3
	}
}

# Armour the Dark Enchantment can work on: fine armour that is not
# necromantic already. Legendary armour is left alone.
# Scope: an artifact.
# Used by: WHG_undead_events, wh_undead_decisions
undead_enchantable_armour_trigger = {
	has_artifact_flag = armor
	NOT = { has_artifact_flag = necromantic_item }
	OR = {
		quality == 2
		quality == 3
	}
}

# The deed a rank 1 member needs to rise to rank 2. Living thralls and most
# lines need a deed done in play (flag vamp_deed_done: feeding on worthy prey,
# a Mahtmasi answering a plea, a thrall's service, a Blood Dragon's duel);
# some lines accept standing instead.
# Scope: a society member.
# Used by: wh_vampire_society_decisions
vampsoc_deed_trigger = {
	OR = {
		has_character_flag = vamp_deed_done
		AND = {
			society_member_of = carstein_court
			is_vampire_trigger = yes
			is_ruler = yes
		}
		AND = {
			society_member_of = lahmian_sisterhood
			is_vampire_trigger = yes
			OR = {
				intrigue = 12
				any_lover = { always = yes }
			}
		}
		AND = {
			society_member_of = necrarch_brotherhood
			is_vampire_trigger = yes
			learning = 14
		}
		AND = {
			society_member_of = jade_coil
			is_vampire_trigger = yes
			learning = 12
		}
	}
}

				trait = chest_pain
				trait = cramps
				trait = diarrhea

# A Nehekkan tribe may come to roam this province: the desert of Nehekhara
# or land of the Tomb Kings, not already Nehekkan, no tribe yet, not refused
# for good, and (for the dead) no more than two tribes in the demesne.
# Scope: the province.
# Used by: WHG_nehekhara_events
nehekka_nomads_may_come_trigger = {
	OR = {
		region = world_nehekhara
		owner = { trait = creature_nehekharan }
	}
	NOT = { culture = nehekka }
	NOT = { has_province_modifier = nomad_presence }
	NOT = { has_province_flag = denied_nomads }
	owner = {
		NOR = {
			trait = creature_orc
			trait = creature_goblin
			trait = creature_beastman
			trait = creature_hobgoblin
		}
		OR = {
			NOT = { trait = creature_nehekharan }
			NOT = {
				any_demesne_province = {
					count = 2
					has_province_modifier = nomad_presence
				}
			}
