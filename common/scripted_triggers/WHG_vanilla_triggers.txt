# WHG_vanilla_triggers.txt: scripted triggers
# Subject: vanilla definitions that nothing in the mod uses
#
# A scripted trigger is a named list of conditions, used anywhere a condition
# can go as "name = yes" (all of them true) or "name = no" (not all true). It
# runs in the scope it is called from. Names are global across the folder and
# must be unique; the load order of files does not matter.
#
# Vanilla definitions that no script of the mod uses. The vanilla-name files of
# this folder override vanilla, so these are only loaded from here; they are
# kept to salvage. Move one to its subject file when content starts using it.
#
# Each entry lists the files that use it ("Used by"). Sections:
#   Not used by any live script: kept for salvage (93)
###############################################################################

				AND = { is_landed = yes higher_real_tier_than = BARON }
			}
		}
		AND = {
			higher_real_tier_than = COUNT
			OR = { #Either defacto direct vassal, or independent ruler who is dejure vassal.
				vassal_of = ROOT
				AND = {
			 		any_dejure_liege = { title = ROOT }
					independent = yes
				}
			}
		}
	}
	AND = {
		OR = { #Candidate should not be a landless Courtier that is about to inherit a Theocracy or Republic
			AND = {
				is_ruler = yes
				is_landed = yes
				is_republic = no
				is_merchant_republic = no
				is_theocracy = no
			}
			NOR = {
				any_heir_title = {
					is_republic = yes
				}
				any_heir_title = {
					is_merchant_republic = yes
				}
				any_heir_title = {
					is_theocracy = yes
				}
			}
		}
		OR = { #If the title controls the religion, candidates must be of the same religion.
			ROOT = {
				NOT = {
					controls_religion = yes
				}
			}
			ROOT = {
				religion = PREV
			}
		}
		NOR = {
			trait = cat
			trait = disinherited
			trait = horse
			trait = eunuch
			trait = bastard
			is_ascetic_trigger = yes
			is_theocracy = yes

	}
