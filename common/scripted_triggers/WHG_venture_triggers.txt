###############################################################################
# The Venturers: explorers, merchant adventurers and colonists
# (adventurers_the_league). Triggers for events/WHG_venture_events.txt and
# decisions/WHG_venture_decisions.txt. Effects in
# common/scripted_effects/WHG_venture_effects.txt.
###############################################################################

# A people that sends out venturers: Men of the Old World, Araby and the
# East (not the Amazons, who are Lustria's own), the Asur and the Dawi.
# Scope: a character.
venture_race_trigger = {
	OR = {
		AND = {
			trait = creature_human
			NOT = { culture_group = amazon_group }
		}
		AND = {
			trait = creature_elf
			culture_group = asur_group
		}
		trait = creature_dwarf
	}
}

###############################################################################
# Voices: who the venturer is, for texts and lore picks. Every venturer has
# exactly one (the generic voice is everyone else: Albion, Kislev, the
# Southlands, the Border Princes).
###############################################################################

# Marienburg and the Wasteland's merchant houses.
venture_voice_westerland_trigger = {
	culture = westerlander
}
# The Empire: Nordland's Second Fleet, the colonists of Neues Emskrank and
# Sudenburg.
venture_voice_imperial_trigger = {
	culture_group = imperial_group
	NOT = { culture = westerlander }
}
# Tilea's merchant princes, Colombo's heirs.
venture_voice_tilean_trigger = {
	OR = {
		culture_group = tilean_group
		culture_group = sartosan_group
	}
}
# Estalia's conquistadores of Magritta.
venture_voice_estalian_trigger = {
	culture_group = estalian_group
}
# Bretonnia: Brionne and L'Anguille.
venture_voice_bretonnian_trigger = {
	culture_group = bretonnian_group
}
# Araby's merchant venturers.
venture_voice_araby_trigger = {
	culture_group = arabyan_group
}
# The Asur sea-houses of Cothique, Lothern and Yvresse.
venture_voice_asur_trigger = {
	culture_group = asur_group
}
# The Dawi of Barak Varr, the Norse holds and the southern holds.
venture_voice_dawi_trigger = {
	culture_group = dwarf_group
}
# Cathay, Ind and Nippon.
venture_voice_east_trigger = {
	culture_group = oriental_group
}
# The Norse who sail for strange shores, Losteriksson's heirs.
venture_voice_norse_trigger = {
	culture_group = norscan_group
}
# Anyone else.
venture_voice_generic_trigger = {
	NOR = {
		venture_voice_westerland_trigger = yes
		venture_voice_imperial_trigger = yes
		venture_voice_tilean_trigger = yes
		venture_voice_estalian_trigger = yes
		venture_voice_bretonnian_trigger = yes
		venture_voice_araby_trigger = yes
		venture_voice_asur_trigger = yes
		venture_voice_dawi_trigger = yes
		venture_voice_east_trigger = yes
		venture_voice_norse_trigger = yes
	}
}

# A Venturer. Scope: a character.
venture_member_trigger = {
	society_member_of = adventurers_the_league
}
