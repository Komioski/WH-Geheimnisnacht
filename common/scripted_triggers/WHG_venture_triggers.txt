# A people that sends out venturers: Men of the Old World, Araby and the
# East (not the Amazons, who are Lustria's own), the Asur and the Dawi.
# Scope: a character.
venture_race_trigger = {
	OR = {
		AND = {
			trait = creature_human
			NOT = { culture_group = amazon_group }
		}
		AND = {
			trait = creature_elf
			culture_group = asur_group
		}
		trait = creature_dwarf
	}
}

###############################################################################
# Expeditions (venture.1 - 31)
###############################################################################

# Free to send an expedition: none out, none in the last five years, and
# not locked away. Scope: the Venturer.
venture_free_trigger = {
	NOT = { has_character_flag = venture_expedition_out }
	NOT = { has_character_modifier = venture_expedition_cd }
	is_inaccessible_trigger = no
}

# A landed Venturer with a harbour to sail from. Scope: the Venturer.
venture_coastal_trigger = {
	is_landed = yes
	any_realm_province = { port = yes }
}

# May lead in person: the AI always, a player unless the game rule says
# sponsor only. Scope: the Venturer.
venture_may_lead_trigger = {
	OR = {
		ai = yes
		NOT = { has_game_rule = { name = warhammer_venture_lead value = sponsor_only } }
	}
}

# The expedition is the Venturer's own, in the jungle with the crew.
# Scope: the Venturer.
venture_leading_trigger = {
	has_character_flag = venture_leading
}

# The destination is green and deadly (Lustria, a temple-city, the
# Southlands): fevers, shrines, natives. Scope: the Venturer.
venture_jungle_trigger = {
	OR = {
		has_character_flag = venture_dest_coast
		has_character_flag = venture_dest_temple
		has_character_flag = venture_dest_southlands
	}
}

# A coastal province where a colony could be planted, found by a
# successful expedition (venture_survey_site_effect).
# Scope: a province. ROOT is the Venturer.
venture_site_trigger = {
	port = yes
	NOT = { owned_by = ROOT }
	NOT = { has_province_flag = venture_surveyed }
	OR = {
		AND = {
			ROOT = {
				OR = {
					has_character_flag = venture_dest_coast
					has_character_flag = venture_dest_temple
				}
			}
			region = world_new_world_lustria
		}
		AND = {
			ROOT = { has_character_flag = venture_dest_southlands }
			OR = {
				region = world_greater_araby_southlands
				region = world_eastern_southlands
			}
		}
		AND = {
			ROOT = { has_character_flag = venture_dest_naggaroth }
			OR = {
				region = world_new_world_naggaroth
				region = world_new_world_black_forest
			}
		}
	}
}

###############################################################################
# The Colonial Charter war (coastal_colonization) and colonies
###############################################################################

# A land across the sea, far from the Venturer's capital: the New World
# (Lustria, Naggaroth and the lands between), the Southlands, or Albion
# once the mists are gone. Scope: a province. ROOT is the Venturer.
venture_colony_region_trigger = {
	OR = {
		AND = {
			region = world_new_world
			ROOT = { capital_scope = { NOT = { region = world_new_world } } }
		}
		AND = {
			OR = {
				region = world_greater_araby_southlands
				region = world_eastern_southlands
			}
			ROOT = {
				capital_scope = {
					NOR = {
						region = world_greater_araby_southlands
						region = world_eastern_southlands
					}
				}
			}
		}
		AND = {
			region = world_albion
			NOT = { has_global_flag = mists_of_albion }
			ROOT = { capital_scope = { NOT = { region = world_albion } } }
		}
	}
}

# The Venturer's realm already holds a harbour in the same land, so the
# colony can grow along the coast. Scope: a province. ROOT is the Venturer.
venture_realm_in_region_trigger = {
	OR = {
		AND = {
			region = world_new_world_lustria
			ROOT = { any_realm_province = { region = world_new_world_lustria port = yes } }
		}
		AND = {
			region = world_new_world_naggaroth
			ROOT = { any_realm_province = { region = world_new_world_naggaroth port = yes } }
		}
		AND = {
			region = world_new_world
			NOR = {
				region = world_new_world_lustria
				region = world_new_world_naggaroth
			}
			ROOT = {
				any_realm_province = {
					region = world_new_world
					NOR = {
						region = world_new_world_lustria
						region = world_new_world_naggaroth
					}
					port = yes
				}
			}
		}
		AND = {
			OR = {
				region = world_greater_araby_southlands
				region = world_eastern_southlands
			}
			ROOT = {
				any_realm_province = {
					OR = {
						region = world_greater_araby_southlands
						region = world_eastern_southlands
					}
					port = yes
				}
			}
		}
		AND = {
			region = world_albion
			ROOT = { any_realm_province = { region = world_albion port = yes } }
		}
	}
}

# A people with a colonial culture of its own in our lore: the Empire
# (Neueslander, or Sudenburger from Araby), Tilea and Sartosa
# (Oltremari), Estalia (Comarcano), Bretonnia (Marcheur), Araby
# (Hadmauti). Scope: the Venturer.
venture_has_colonial_culture_trigger = {
	OR = {
		culture_group = imperial_group
		culture_group = tilean_group
		culture_group = sartosan_group
		culture_group = estalian_group
		culture_group = bretonnian_group
		culture_group = arabyan_group
	}
	NOR = {
		culture = neueslander
		culture = sudenburger
		culture = oltremari
		culture = comarcano
		culture = marcheur
		culture = hadmauti
	}
}

# May raise colony buildings (power venturer_colony_works): a Venturer of
# rank 3, or a lord whose liege is one. Scope: the holder.
venture_colony_works_trigger = {
	OR = {
		AND = {
			society_member_of = adventurers_the_league
			society_rank >= 3
		}
		any_liege = {
			society_member_of = adventurers_the_league
			society_rank >= 3
		}
	}
}

# Holds a chartered colony. Scope: a character.
venture_holds_colony_trigger = {
	any_demesne_province = { has_province_flag = venture_colony }
}

###############################################################################
# Venturer missions (seaquest.2 -> venture.100 - 106)
###############################################################################

# Send a venture: a Venturer free to send one, with the gold for it.
# Scope: the member.
venture_mission_expedition_trigger = {
	venture_coastal_trigger = yes
	venture_free_trigger = yes
	wealth = 150
}

# Plant a colony: a Charter Captain with a harbour.
venture_mission_charter_trigger = {
	society_rank >= 3
	venture_coastal_trigger = yes
	war = no
}

# Raise the colony works: holds a colony short of a Port, a Monument or a
# Venue.
venture_mission_works_trigger = {
	any_demesne_province = {
		has_province_flag = venture_colony
		OR = {
			AND = {
				port = yes
				venture_colony_has_port_trigger = no
			}
			venture_colony_has_monument_trigger = no
			venture_colony_has_venue_trigger = no
		}
	}
}

# Bring them home: a landed Venturer with a harbour and some gold.
venture_mission_rescue_trigger = {
	venture_coastal_trigger = yes
	wealth = 100
}

# Holds one of the Venturer missions.
venture_has_mission_trigger = {
	OR = {
		has_quest = quest_venture_expedition
		has_quest = quest_venture_charter
		has_quest = quest_venture_works
		has_quest = quest_venture_rescue
	}
}
