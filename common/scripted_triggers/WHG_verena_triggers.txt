# The Scrollbearer's hoard: a member who may be the hoarder of verena.51,
# near the member, neither the member nor the giver, not fond of the
# member and not marked for another raid.
# Scope: the candidate; ROOT is the member, FROM the mission giver.
verena_infiltration_free_trigger = {
	is_within_diplo_range = ROOT
	NOR = {
		character = ROOT
		character = FROM
		reverse_opinion = { who = ROOT value = 20 }
		has_character_flag = infiltration_rival
		has_character_flag = infiltration_mutual_dislike
		has_character_flag = infiltration_rich
	}
}

# The Scrollbearer's hoard: a hoard worth the raid: a book of the Order,
# ingredients, gold or a grand library.
# Scope: the candidate.
verena_rich_hoard_trigger = {
	OR = {
		any_artifact = { verena_stealable_text_trigger = yes }
		any_artifact = { has_artifact_flag = ingredient }
		wealth = 200
		has_character_modifier = best_laboratory
		has_character_modifier = great_laboratory
	}
}

# The Scrollbearer's hoard: a book worth stealing: the magnum opus of a
# brother or the Tablet of Lybaras.
# Scope: an artifact.
verena_stealable_text_trigger = {
	OR = {
		AND = {
			has_artifact_flag = verena
			has_artifact_flag = book
		}
		AND = {
			has_artifact_flag = hermetics
			has_artifact_flag = text
		}
	}
}

# The Scrollbearer's hoard: the guards woke up or came back too early.
# Scope: the member.
verena_raid_detected_trigger = {
	OR = {
		has_character_flag = guards_wake_up
		has_character_flag = distraction_failed
	}
}


# The scrying: someone close to the dead who carries a relationship omen
# and has not yet been told of this death (verena.106).
# Scope: the one who may be told.
verena_death_omen_recipient_trigger = {
	is_alive = yes
	has_relationship_omen_modifier_trigger = yes
	NOT = { trait = incapable }
	NOR = {
		has_character_flag = verena_omen_death_lover
		has_character_flag = verena_omen_death_friend
		has_character_flag = verena_omen_death_rival
		has_character_flag = verena_omen_death_parent
		has_character_flag = verena_omen_death_child
	}
}

# The horoscope: a courtier who may inspire the child on the path the stars
# foretold, with the skill of that path (verena.134).
# Scope: the courtier; ROOT is the child.
verena_horoscope_mentor_trigger = {
	is_adult = yes
	prisoner = no
	NOT = { trait = incapable }
	OR = {
		AND = {
			ROOT = { has_character_flag = destined_steward }
			stewardship = 15
			has_education_stewardship_trigger = yes
		}
		AND = {
			ROOT = { has_character_flag = destined_fighter }
			martial = 15
			has_education_martial_trigger = yes
		}
		AND = {
			ROOT = { has_character_flag = destined_diplomat }
			diplomacy = 15
			has_education_diplomacy_trigger = yes
		}
		AND = {
			ROOT = { has_character_flag = destined_schemer }
			intrigue = 15
			has_education_intrigue_trigger = yes
		}
		AND = {
			ROOT = { has_character_flag = destined_scholar }
			learning = 15
			has_education_learning_trigger = yes
		}
	}
}

# The horoscope: the child's court has an adult who may inspire the child,
# the host or a courtier (verena.134).
# Scope: the child.
verena_horoscope_has_mentor_trigger = {
	host = {
		OR = {
			AND = {
				NOT = { character = ROOT }
				is_adult = yes
			}
			any_courtier = { is_adult = yes prisoner = no }
		}
	}
}

# Gathering ingredients: the character (the gatherer or his companion) is
# poor at the search the gatherer chose: stewardship for herbs, martial for
# the hunt, diplomacy at the apothecaries (verena.151).
# Scope: the gatherer or his companion; ROOT is the gatherer.
verena_gather_skill_low_trigger = {
	OR = {
		AND = {
			ROOT = { has_character_flag = verena_gather_herbs }
			NOT = { stewardship = 6 }
		}
		AND = {
			ROOT = { has_character_flag = verena_gather_beasts }
			NOT = { martial = 6 }
		}
		AND = {
			ROOT = { has_character_flag = verena_gather_market }
			NOT = { diplomacy = 6 }
		}
	}
}

# Gathering ingredients: the character is good at the search the gatherer
# chose (verena.151).
# Scope: the gatherer or his companion; ROOT is the gatherer.
verena_gather_skill_high_trigger = {
	OR = {
		AND = {
			ROOT = { has_character_flag = verena_gather_herbs }
			stewardship = 12
		}
		AND = {
			ROOT = { has_character_flag = verena_gather_beasts }
			martial = 12
		}
		AND = {
			ROOT = { has_character_flag = verena_gather_market }
			diplomacy = 12
		}
	}
}

# Gathering ingredients: the gatherer has a knack for the search he chose:
# a gardener in the woods, a hunter or falconer on the hunt, a gregarious
# man among the apothecaries (verena.151).
# Scope: the gatherer.
verena_gather_knack_trigger = {
	OR = {
		AND = {
			has_character_flag = verena_gather_herbs
			trait = gardener
		}
		AND = {
			has_character_flag = verena_gather_beasts
			OR = {
				trait = hunter
				trait = falconer
			}
		}
		AND = {
			has_character_flag = verena_gather_market
			trait = gregarious
		}
	}
}

# Watching the stars: the spouse already feels neglected by ROOT
# (verena.211).
# Scope: the spouse; ROOT is the stargazer.
verena_out_of_touch_with_root_trigger = {
	OR = {
		has_opinion_modifier = { who = ROOT modifier = opinion_out_of_touch }
		has_opinion_modifier = { who = ROOT modifier = opinion_out_of_touch_mild }
	}
}

# The theory paper: the writer is still waiting on the reviewers
# (verena.303, 308).
# Scope: the writer.
verena_paper_under_review_trigger = {
	OR = {
		has_character_flag = theory_paper_normal_effort
		has_character_flag = theory_paper_asked_help
		has_character_flag = theory_paper_stole_work
		has_character_flag = theory_paper_heavy_investment
	}
}
