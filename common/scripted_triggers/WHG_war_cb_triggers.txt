# Triggers for the wars of hostages, faith and tribute
# (events/WHG_war_cb_events.txt)
###############################################################################

# One of the captives the rescuer went to war for: unlanded kin, friend,
# spouse or consort of event_target:warcb_rescuer, held in prison or as a
# forced consort by event_target:warcb_captor.
# Scope: a character in the captor's realm.
warcb_captive_trigger = {
	is_landed = no
	OR = {
		any_spouse = { character = event_target:warcb_rescuer }
		is_friend = event_target:warcb_rescuer
		is_close_relative = event_target:warcb_rescuer
		dynasty = event_target:warcb_rescuer
		is_consort = event_target:warcb_rescuer
	}
	OR = {
		prisoner = yes
		has_opinion_modifier = { modifier = opinion_forced_consort who = event_target:warcb_captor }
	}
}

# A captor whose gods want captives' blood: Chaos (Hashut and the Norscan
# gods too), the Dark Elves, the Lizardmen and the Khainite Wood Elves.
# Scope: the captor.
warcb_sacrificing_captor_trigger = {
	OR = {
		religion_group = chaos_gods_group
		religion_group = cytharai_group
		religion = old_ones
		religion = green_khaine
	}
}

# The ruler borders a realm of another faith that holds provinces of the
# ruler's faith in a kingdom the ruler also borders.
# Scope: the ruler.
warcb_liberation_possible_trigger = {
	any_realm_province = {
		any_neighbor_province = {
			holder_scope = {
				NOT = { same_realm = ROOT }
				top_liege = { warcb_liberation_target_trigger = yes }
			}
		}
	}
}

# An independent ruler of another faith who borders ROOT and rules
# provinces of ROOT's faith in a kingdom that also borders ROOT's realm.
# Scope: the infidel ruler; ROOT is the liberator.
warcb_liberation_target_trigger = {
	NOT = { religion = ROOT }
	any_realm_province = {
		religion = ROOT
		kingdom = {
			any_de_jure_vassal_title = {
				tier = COUNT
				holder_scope = { same_realm = PREVPREVPREVPREV }
				location = {
					any_neighbor_province = { holder_scope = { top_liege = { character = ROOT } } }
				}
			}
		}
	}
}
