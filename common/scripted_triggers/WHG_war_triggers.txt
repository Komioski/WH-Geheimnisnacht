# returns true if scoped character is already disliked by ROOT (while not being
# caught up in a friendship or love, and such)
# Used by: WHG_battle_events, WHG_warrior_society_events +2 more

	NOT = { is_rival = ROOT }
	NOT = { is_lover = ROOT }
	NOT = { is_friend = ROOT }
	NOT = { reverse_opinion = { who = ROOT value = 5 } }
	OR = {
		ROOT = { is_foe = PREV } #checks if the current scope has committed a crime against ROOT (I hope)
		is_foe = ROOT  #checks if root has committed a crime against the current scope
		NOT = { reverse_opinion = { who = ROOT value = -25 } }
		root_has_terrible_opinion_modifier_of_this_trigger = yes
		#Fill up with fun opinion modifiers, whenever you think of them...! Ideas: opinion_spurned, opinion_revoked_my_title, etc
	}
}

# Used by: 00_retinue_subunits, wh_vampire_society_retinues +3 more

	OR = {
		ai = no
		is_patrician = yes
		OR={
			AND={
				independent=yes
				prestige=500
				OR={
					government=beastman_government
					government=chaos_horde_government
				}
				is_nomadic = yes
			}
			AND={
				independent=no
				prestige=250
				is_nomadic = yes
			}
			total_years_played <= 1
		}
		AND = {
			higher_real_tier_than = BARON
			OR = {
				AND = {
					OR = {
						is_feudal = yes
						is_theocracy = yes
						is_republic = yes
					}
					# > doesn't work with demesne_size
					OR = {
						demesne_size = 2
						higher_real_tier_than = COUNT
					}
				}
				is_tribal = yes
			}
		}
	}
}

# Peoples whose landless sons take to raiding: the horse peoples under a
# horde, Norsca and its gods, the reavers of Stromfels, the corsairs of
# Araby (in a port), and any ambitious, gifted warrior with no inheritance
# (adventurer.1).
# Scope: a landless character.
# Used by: WHG_adventurer_events
adventurer_raider_people_trigger = {
	OR = {
		AND = {
			any_liege = {
				OR = {
					government = nomadic_government
					government = chaos_horde_government
				}
			}
			OR = {
				culture_group = kurgan_group
				culture_group = hung_group
			}
		}
		culture_group = norscan_group
		religion = norscan_gods
		religion = stromfels
		AND = {
			OR = {
				religion = atruhayid_reformed
				religion = atruhayid
			}
			location = { port = yes }
		}
		AND = {
			trait = ambitious
			martial = 15
			diplomacy = 12
			health = 4
			is_heir = no
			NOR = {
				any_child = {
					OR = {
						is_heir = yes
						is_landed = yes
					}
				}
				NAND = {
					uses_decadence = yes
					num_of_dynasty_members = 3
					biggest_realm_size_relative = {
						NOT = { character = ROOT }
						realm_size = 5
					}
				}
			}
		}
	}
}

}

# A province a great lord could hand to the adventurer: not the capital, one
# holding, no nomad unrest, and a port for a sea raider when the realm has
# any (adventurer.11, .19).
# Scope: a demesne province; PREV the ruler whose demesne it is, ROOT the
# adventurer.
# Used by: WHG_war_triggers
adventurer_spare_province_trigger = {
	is_capital = no
	num_of_settlements = 1
	NOT = { has_province_modifier = nomad_agitation }
	OR = {
		AND = {
			ROOT = { is_seafarer = yes }
			port = yes
		}
		ROOT = { is_seafarer = no }
		PREV = { NOT = { any_demesne_province = { port = yes } } }
	}
}

	    AND = {

			NOT = { is_rival = PREV }
			NOT = { is_lover = PREV }
			prev_attraction_trigger = yes
			OR = {
				reverse_opinion = { who = PREV value = 45 }
				reverse_has_opinion_modifier = { who = PREV modifier = opinion_budding_romance }
			}
			attacted_to_prev_trigger = yes
			OR = {
				opinion = { who = ROOT value = 45 }
				has_opinion_modifier = { who = ROOT modifier = opinion_budding_romance }
			}
	    }
		# Friendship potential...
		AND = {
			NOT = { is_friend = PREV }
			opinion = { who = PREV value = 5 }
			reverse_opinion = { who = PREV value = 5 }
			this_has_terrible_opinion_modifier_of_root_trigger = no
			root_has_terrible_opinion_modifier_of_this_trigger = no
		}

		# Rival potential...
		AND = {
			NOT = { is_rival = PREV }
			NOT = { is_lover = PREV }
			NOT = { is_friend = PREV }
			NOT = { opinion = { who = PREV value = 5 } }
			OR = {
				is_foe = PREV #checks if the current scope has committed a crime against PREV
				PREV = { is_foe = PREVPREV} #checks if PREV has committed a crime against current scope
				NOT = { opinion = { who = PREV value = -25 } }
				this_has_terrible_opinion_modifier_of_root_extended_trigger = yes
			}
		}
		AND = {
			NOT = { is_rival = PREV }
			NOT = { is_lover = PREV }
			NOT = { is_friend = PREV }
			NOT = { reverse_opinion = { who = PREV value = 5 } }
			OR = {
				is_foe = PREV #checks if the current scope has committed a crime against PREV

				NOT = { reverse_opinion = { who = PREV value = -25 } }

# A horde that sacks cities: a Chaos horde, a greenskin or beastman lord,
# or a nomad of the steppe. Not a rebel.
# Scope: a character.
# Used by: WHG_chaos_triggers, WHG_war_events
greatsack_horde_trigger = {
	rebel = no
	OR = {
		government = chaos_horde_government
		is_nomadic = yes
		religion_group = greenskin_gods_group
		culture_group = orc_group
		culture_group = goblin_group
		culture_group = beastman_group
	}
}

# The horde does not sack its own kind: Chaos spares the followers of the
# Dark Gods, the Norscans and the Dawi Zharr (as chaoshorde_sacker_trigger
# does), greenskins spare greenskins, beastmen beastmen, nomads nomads.
# Scope: the holding; ROOT is the horde's lord.
# Used by: WHG_chaos_triggers, WHG_war_events
greatsack_victim_trigger = {
	holder_scope = {
		trigger_if = {
			limit = { ROOT = { greatsack_chaos_trigger = yes } }
			open_follower_of_any_chaos_gods = no
			NOR = {
				religion = norscan_gods
				religion = norscan_gods_reformed
				religion = hashut
			}
		}
		trigger_else_if = {
			limit = { ROOT = { greatsack_greenskin_trigger = yes } }
			NOT = { religion_group = greenskin_gods_group }
		}
		trigger_else_if = {
			limit = { ROOT = { greatsack_beastman_trigger = yes } }
			NOT = { culture_group = beastman_group }
		}
		trigger_else = {
			is_nomadic = no
		}
	}
}
