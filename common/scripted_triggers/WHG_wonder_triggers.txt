# wonder_trap_victim_trigger
# Scope: a courtier. Free, adult and not a Lizardman (they know the traps
# of the old temples). Used by wonder.58.
wonder_trap_victim_trigger = {
	prisoner = no
	is_adult = yes
	NOT = { culture_group = lizardman_group }
}

# wonder_book_library_trigger
# Scope: a ruler. Some active upgrade in the realm keeps books: poetry and
# drama, architecture, history, natural sciences, or mathematics and
# philosophy. Used by wonder.100.
wonder_book_library_trigger = {
	any_realm_wonder_upgrade = {
		is_active = yes
		OR = {
			has_wonder_upgrade_flag = poetry_and_drama
			has_wonder_upgrade_flag = architecture
			has_wonder_upgrade_flag = history_flag
			has_wonder_upgrade_flag = natural_sciences
			has_wonder_upgrade_flag = math_and_philosophy
		}
	}
}

# wonder_book_seller_trigger
# Scope: an independent ruler. A far-off realm where a lost book can turn
# up: landed, not ROOT, with a province more than 500 away. Used by
# wonder.100.
wonder_book_seller_trigger = {
	is_landed = yes
	NOT = { character = ROOT }
	any_realm_province = { distance = { who = ROOT value > 500 } }
}

# wonder_book_scholar_trigger
# Scope: a courtier of a nearby ruler. A scholar who can be asked to come
# to the gathering of wonder.110 without leaving a family or office behind.
wonder_book_scholar_trigger = {
	prisoner = no
	is_adult = yes
	is_ruler = no
	is_married = no
	has_job_title = no
	is_inaccessible_or_incapable_trigger = no
	NOT = { is_close_relative = PREV }
	OR = { trait = scholar learning = 12 }
}

# wonder_emissary_trigger
# Scope: a courtier. Free to ride to a far-off court for a lost book: an
# adult who is not a ruler, holds no office and is not already away.
wonder_emissary_trigger = {
	prisoner = no
	is_adult = yes
	is_ruler = no
	has_job_title = no
	NOT = { trait = travelling }
	is_inaccessible_or_incapable_trigger = no
}

# wonder_may_seduce_trigger
# Scope: the host. The guest (event_target:target_character) may take the
# host as a lover: not close kin, or kin by a faith that allows it, or
# already wed to the host.
wonder_may_seduce_trigger = {
	OR = {
		NOT = { is_close_relative = event_target:target_character }
		any_spouse = { character = event_target:target_character }
		has_religion_feature = religion_feature_zun
		has_religion_feature = religion_holy_family
		religion = slaanesh
		religion = cult_of_pleasure
		religion = atharti
		religion = anath_raema
		religion = vampiric
		religion = amazon
	}
}

# wonder_murderous_host_trigger
# Scope: the host. Ruthless or cunning enough to plan a killing at the
# meeting: intrigue 15 or a scheming, deceitful, cruel or mad nature.
wonder_murderous_host_trigger = { OR = { intrigue = 15 wonder_murderous_nature_trigger = yes } }

# wonder_murderous_guest_trigger
# Scope: the guest. As the host, but intrigue 10 is enough: the guest
# only has to strike when the host comes close.
wonder_murderous_guest_trigger = { OR = { intrigue = 10 wonder_murderous_nature_trigger = yes } }

# wonder_murderous_nature_trigger
# Scope: a character with a scheming, deceitful, cruel or mad nature.
wonder_murderous_nature_trigger = {
	OR = {
		trait = deceitful
		trait = impaler
		trait = schemer
		trait = lunatic_1
		trait = lunatic_2
		trait = lunatic_3
		trait = amateurish_plotter
		trait = flamboyant_schemer
		trait = intricate_webweaver
		trait = elusive_shadow
	}
}
