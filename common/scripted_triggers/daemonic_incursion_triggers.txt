# Triggers for daemonic and beastmen incursions (events/daemonic_incursion_events.txt)

# Araby lies furthest from the polar gates
daemonic_incursion_araby_trigger = {
	OR = {
		culture_group = arabyan_group
		capital_scope = { empire = { title = e_araby } }
	}
}

# Daemons care less for the lands of the undead and the greenskins
daemonic_incursion_unloved_prey_trigger = {
	OR = {
		is_undead_trigger = yes
		religion_group = necromantic_group
		religion_group = nehekharan_gods_group
		religion_group = greenskin_gods_group
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
		trait = creature_snotling
	}
}

# One of the Chaos realms of the warp is ascendant
warp_any_realm_ascendant_trigger = {
	OR = {
		khorne_warp = { has_status = warp_ascendant }
		nurgle_warp = { has_status = warp_ascendant }
		slaanesh_warp = { has_status = warp_ascendant }
		tzeentch_warp = { has_status = warp_ascendant }
		undivided_warp = { has_status = warp_ascendant }
	}
}

# The Ruinous Powers stand united
warp_powers_united_trigger = {
	undivided_warp = { has_status = warp_united }
}
