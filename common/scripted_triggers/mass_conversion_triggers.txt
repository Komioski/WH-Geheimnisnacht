# Mass conversion (HF baptism) scope, kept to human faiths so it can't cause
# culture/religion drift across races (decisions/HF_baptism_decisions.txt).

# Who can be converted: a human follower of the old Norscan gods
mass_conversion_human_convert_trigger = {
	trait = creature_human
	religion = norscan_gods
}

# Who can sponsor: a human of the Old World faiths (including the northern
# and old faith pagans) or of the Dark Gods (dark baptism; not Hashut)
mass_conversion_human_sponsor_trigger = {
	trait = creature_human
	OR = {
		religion_group = old_world_group
		AND = {
			religion_group = chaos_gods_group
			NOT = { religion = hashut }
		}
	}
}
