# One of the misty sea zones around Albion: the Mists of Albion, the two
# Coasts of Albion, the Sea of Whales and the Lake of Eyes.
# Scope: a province.
albion_mist_province_trigger = {
	OR = {
		province_id = 1868
		province_id = 1938
		province_id = 1912
		province_id = 1865
		province_id = 2074
	}
}

# The innermost mist, which nobody passes.
# Scope: a province.
albion_inner_mist_province_trigger = {
	province_id = 1868
}

# One who knows the way through the outer mists: the tribes of Albion,
# whether they keep the Old Faith or follow the Dark Emissaries, and the
# Fimir, who raise mists of their own.
# Scope: a character.
albion_native_trigger = {
	OR = {
		culture_group = albion_group
		trait = creature_fimir
	}
}

# The commander, or his liege, has a guide through the outer mists.
# Scope: a commander.
albion_mist_guided_trigger = {
	OR = {
		has_character_modifier = albion_mist_pilot
		liege = { has_character_modifier = albion_mist_pilot }
	}
}

# A servant of the Dark Masters, led by the Dark Emissaries: the Chaos
# faithful (the Fimir among them), the Druchii, the dead, the Skaven, the
# greenskins and beastmen, and the dabblers in dark magic. Everyone else is
# led by the Truthsayers.
# Scope: a character.
albion_dark_path_trigger = {
	OR = {
		religion_group = chaos_gods_group
		religion_group = cytharai_group
		religion_group = necromantic_group
		religion_group = horned_rat_god_group
		religion_group = greenskin_gods_group
		trait = creature_beastman
		trait = lore_dark_magic
		trait = lore_necromancy
		is_vampire_trigger = yes
	}
}

# The lord pays for a guide in prestige, not gold: greenskin bosses,
# beastmen and tribal chiefs.
# Scope: a character.
albion_guide_pays_prestige_trigger = {
	OR = {
		trait = creature_orc
		trait = creature_goblin
		trait = creature_beastman
		is_tribal = yes
	}
}

# Six of Albion's Ogham stones stand dark, and the mists fail. Read after
# albion_count_dark_stones_effect.
# Scope: a character.
albion_mists_broken_trigger = {
	check_variable = { which = albion_dark_stones value = 6 }
}
