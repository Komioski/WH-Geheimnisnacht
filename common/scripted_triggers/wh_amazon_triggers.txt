# A captive man of the court who can be picked at the spring rite.
# Scope: a courtier.
amazon_feast_captive_trigger = {
	is_adult = yes
	is_female = no
	OR = {
		has_character_flag = amazon_exception
		trait = in_stock
	}
	NOR = {
		trait = incapable
		trait = eunuch
		has_character_flag = amazon_feast_chosen
	}
}

# A girl who can go on the final trial of the Amazons.
# Scope: a character.
amazon_can_take_trial_trigger = {
	NOR = {
		trait = incapable
		prisoner = yes
		trait = disinherited_amazon
		trait = questing
	}
}

# An Amazon fit to go out on a hunt.
# Scope: a character.
amazon_can_hunt_trigger = {
	trait = creature_amazon
	age = 16
	prisoner = no
	NOR = {
		trait = complacent
		trait = questing
		trait = in_hiding
		trait = studying_magic
		trait = disinherited_amazon
		trait = pregnant
		trait = hard_pregnancy
		trait = troubled_pregnancy
		trait = infirm
		trait = incapable
		trait = maimed
		trait = craven
	}
}
