# THIS has the gift, shown (trait) or hidden (flag). A recessive mage never
# counts.
apprentice_has_gift_trigger = {
	NOT = { has_character_flag = recessive_mage }
	OR = {
		trait = magic_potential_1
		trait = magic_potential_2
		trait = magic_potential_3
		trait = magic_potential_4
		has_character_flag = magic_potential_flag_1
		has_character_flag = magic_potential_flag_2
		has_character_flag = magic_potential_flag_3
		has_character_flag = magic_potential_flag_4
	}
}

# THIS is an apprentice who passes his master's test (apprentice.15). The
# weaker the gift, the more learning or lesson experience (magicxp) he
# needs. Elves are natural channels and need less.
apprentice_master_test_trigger = {
	is_incapable = no
	apprentice_has_gift_trigger = yes
	OR = {
		AND = {
			trait = creature_elf
			OR = { learning = 10 check_variable = { which = magicxp value = 11 } }
		}
		AND = {
			OR = { trait = magic_potential_1 has_character_flag = magic_potential_flag_1 }
			OR = { learning = 16 check_variable = { which = magicxp value = 20 } }
		}
		AND = {
			OR = { trait = magic_potential_2 has_character_flag = magic_potential_flag_2 }
			OR = { learning = 14 check_variable = { which = magicxp value = 17 } }
		}
		AND = {
			OR = { trait = magic_potential_3 has_character_flag = magic_potential_flag_3 }
			OR = { learning = 12 check_variable = { which = magicxp value = 15 } }
		}
		AND = {
			OR = { trait = magic_potential_4 has_character_flag = magic_potential_flag_4 }
			OR = { learning = 10 check_variable = { which = magicxp value = 13 } }
		}
	}
}

# THIS taught himself and passes the test (apprentice.57).
apprentice_self_test_trigger = { is_incapable = no learning = 21 apprentice_has_gift_trigger = yes }

# THIS is undead or of the blood: a vampire or a Tomb King. Death magic
# comes easier to them.
apprentice_undead_trigger = { OR = { is_vampire_trigger = yes trait = khemri_tomb_king } }

# THIS taught himself necromancy and passes the test (apprentice.81).
# Without undead blood or undead insight (the Liber Mortis, a dread liege)
# it takes learning 29; with it, 21.
apprentice_necro_self_test_trigger = {
	is_incapable = no
	OR = {
		learning = 29
		AND = {
			learning = 21
			OR = { is_vampire_trigger = yes has_character_flag = undead_insight }
		}
	}
	OR = { apprentice_has_gift_trigger = yes apprentice_undead_trigger = yes }
}

# THIS is a necromancer's apprentice and passes his master's test
# (apprentice.91).
apprentice_necro_ward_test_trigger = {
	is_incapable = no
	learning = 21
	OR = { apprentice_has_gift_trigger = yes apprentice_undead_trigger = yes }
}

# THIS (a master) teaches the Chaos apprenticeship (apprentice.30) instead
# of the common one: a sorcerer of Tzeentch, Nurgle or Slaanesh.
apprentice_chaos_master_trigger = {
	OR = {
		AND = { religion = tzeentch trait = lore_tzeentch }
		AND = { religion = nurgle trait = lore_nurgle }
		AND = { religion = slaanesh trait = lore_slaanesh }
	}
}

# THIS (an apprentice) still has lessons to take: fewer than fourteen, or
# his years of training are not over.
apprentice_lessons_left_trigger = {
	NAND = {
		check_variable = { which = "mage_teaching" value = 14 }
		magic_training_years_over = yes
	}
}
