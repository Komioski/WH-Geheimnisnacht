# A faith that keeps the Fast of the Prophets.
# Scope: the character.
holyfast_faith_trigger = {
	OR = {
		religion = ormazd
		religion = kitabid
	}
}

# A prisoner the fasting ruler may free in mercy: anyone but a decadent
# kinsman of the ruler's own dynasty.
# Scope: the prisoner; ROOT is the fasting ruler.
holyfast_prisoner_to_free_trigger = {
	prisoner = yes
	OR = {
		NOT = { trait = decadent }
		NOT = { dynasty = ROOT }
	}
}

# The fasting ruler holds at least one prisoner to free.
# Scope: the fasting ruler.
holyfast_prisoners_to_free_trigger = {
	num_of_prisoners = 1
	any_courtier = { holyfast_prisoner_to_free_trigger = yes }
}
