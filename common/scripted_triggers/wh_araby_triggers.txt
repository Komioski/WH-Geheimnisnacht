# A faith that keeps the Fast of the Prophets.
# Scope: the character.
holyfast_faith_trigger = {
	OR = {
		religion = ormazd
		religion = kitabid
	}
}

# A prisoner the fasting ruler may free in mercy: anyone but a decadent
# kinsman of the ruler's own dynasty.
# Scope: the prisoner; ROOT is the fasting ruler.
holyfast_prisoner_to_free_trigger = {
	prisoner = yes
	OR = {
		NOT = { trait = decadent }
		NOT = { dynasty = ROOT }
	}
}

# The fasting ruler holds at least one prisoner to free.
# Scope: the fasting ruler.
holyfast_prisoners_to_free_trigger = {
	num_of_prisoners = 1
	any_courtier = { holyfast_prisoner_to_free_trigger = yes }
}

# A Sultan of Araby who might give a white elephant (elephant.1).
# Scope: the giver; ROOT is the receiving ruler.
elephant_giver_trigger = {
	ai = yes
	religion_group = ormazd_gods_group
	higher_tier_than = COUNT
	NOT = { character = ROOT }
	NOT = { war_with = ROOT }
}

# The ruler's capital lies in Araby or the Southlands (souk).
# Scope: the ruler.
souk_araby_capital_trigger = {
	capital_scope = {
		OR = {
			region = world_greater_araby
			region = world_eastern_southlands
		}
	}
}

# The faiths that make the pilgrimage to Ind (ind, WHG_araby_events Part 5):
# the faithful of Ormazd and the gods of Cathay. Humans only.
# Scope: a character.
ind_pilgrimage_faith_trigger = {
	trait = creature_human
	OR = {
		religion_group = ormazd_gods_group
		religion_group = oriental_gods_group
	}
}
