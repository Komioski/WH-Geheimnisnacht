# A tree kin or a forest spirit.
# Scope: a character.
asrai_forest_being_trigger = { OR = { trait = creature_treekin trait = creature_forestspirit } }

# One of Athel Loren's sacred beings: an Avatar or a forest spirit.
# Scope: a character.
asrai_sacred_being_trigger = {
	OR = { trait = avatar_isha trait = avatar_kurnous trait = creature_forestspirit }
}

# A courtier the forest's anger or a spirit's hunger could fall on: not a
# ruler, not one of the forest's own.
# Scope: a character.
asrai_mortal_courtier_trigger = {
	is_ruler = no
	NOT = { trait = avatar_isha }
	NOT = { trait = avatar_kurnous }
	NOT = { trait = creature_forestspirit }
	NOT = { trait = creature_treekin }
}
