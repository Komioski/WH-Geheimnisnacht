# Hard to shame in public: royalty, archmages and masters of the court.
# Scope: a character.
asurdebate_eminent_trigger = {
	OR = {
		trait = phoenix_king
		trait = everqueen
		trait = everqueen_heir
		trait = asur_mage
		trait = grey_eminence
		trait = charismatic_negotiator
	}
}

# Easy to shame in public: vices, deformities and Asuryan's mark.
# Scope: a character.
asurdebate_shameful_trigger = {
	OR = {
		trait = kinslayer
		religion = cult_of_pleasure
		trait = drunkard
		trait = addict
		trait = gluttonous
		trait = slothful
		trait = lustful
		trait = hedonist
		trait = is_fat
		trait = hunchback
		trait = dwarf
		trait = burnt_by_asuryan
		trait = asuryan_stain
	}
}

# Above reproach: the virtues the Asur admire, and the haughty who will not
# bend.
# Scope: a character.
asurdebate_virtuous_trigger = {
	OR = {
		trait = chaste
		trait = charitable
		trait = kind
		trait = proud
		trait = haughty
		trait = poet
		trait = theologian
		piety = 500
	}
}

# The host and the guest of a private meeting could marry each other.
# Scope: the guest. Needs the event target asurmeet_host.
asurmeet_can_wed_trigger = {
	is_married = no
	NOT = { cannot_marry_trigger = yes }
	NOT = { is_close_relative = event_target:asurmeet_host }
	event_target:asurmeet_host = { is_married = no NOT = { cannot_marry_trigger = yes } }
	OR = {
		AND = { is_female = yes event_target:asurmeet_host = { is_female = no } }
		AND = { is_female = no event_target:asurmeet_host = { is_female = yes } }
	}
}

# What the host of a private meeting asked for.
# Scope: the host.
asurmeet_asked_gold_trigger = {
	OR = {
		has_character_flag = asurmeet_ask_gold_100
		has_character_flag = asurmeet_ask_gold_250
		has_character_flag = asurmeet_ask_gold_500
	}
}
