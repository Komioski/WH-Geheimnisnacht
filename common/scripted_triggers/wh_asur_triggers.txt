# A Prince of Ulthuan the other Princes might raise to the Phoenix Throne
# when the heir's line has been burned by Asuryan: a grown elf lord of a
# kingdom, untainted, and not of the burned king's house. The Everqueen
# never takes the Phoenix Crown.
# Scope: a vassal of the burned king, ROOT the burned king (asur.4).
asur_phoenix_candidate_trigger = {
	tier = KING
	is_feudal = yes
	is_female = no
	is_adult = yes
	trait = creature_elf
	NOT = { dynasty = ROOT }
	NOT = { trait = asuryan_stain }
	NOT = { trait = everqueen }
}

# The Everqueen ruling Avelorn or the Gaen Vale in her own right.
# Scope: a character (asur.49).
asur_everqueen_ruling_trigger = {
	independent = yes
	trait = everqueen
	trait = creature_elf
	religion_group = cadai_group
	OR = { has_landed_title = k_avelorn has_landed_title = c_gaen_vale }
}
