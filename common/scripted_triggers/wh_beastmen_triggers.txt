# Lands where warherds roam and turnskins are born: the Chaos Wastes, Kislev,
# Norsca, Araby and the wild parts of Bretonnia and the Border Princes.
# Scope: a province.
beastmen_lands_trigger = {
	OR = {
		empire = { title = e_chaos_wastes }
		empire = { title = e_kislev }
		empire = { title = e_norsca }
		empire = { title = e_araby }
		kingdom = { title = k_couronne }
		kingdom = { title = k_lyonesse }
		kingdom = { title = k_languille }
		kingdom = { title = k_stygian_wood }
		kingdom = { title = k_the_watchtowers }
	}
}

# A courtier raiders can reach who is not a beastman and was not already
# picked in this raid.
# Scope: a character.
beastmen_raid_victim_trigger = {
	is_abroad = no
	NOT = { trait = creature_beastman }
	NOT = { has_character_flag = beastmen_raid_victim }
}

# Traits that make a newborn of beastman blood a bray shaman: Tzeentch favours
# the clever, Slaanesh the beautiful, Nurgle the marked, and the undivided
# Chaos gods the strong.
# Scope: the child.
beastmen_bray_shaman_birth_trigger = {
	OR = {
		AND = {
			religion = tzeentch
			OR = { trait = genius trait = elusive_shadow trait = brilliant trait = bright trait = quick }
		}
		AND = {
			religion = slaanesh
			OR = { trait = beautiful trait = fair trait = pretty trait = genius trait = unsightly }
		}
		AND = {
			religion = nurgle
			OR = {
				trait = genius
				trait = midas_touched
				trait = brilliant
				trait = hunchback
				trait = lisp
				trait = stutter
			}
		}
		AND = {
			religion = chaos
			OR = {
				trait = strong
				trait = tough
				trait = hardy
				trait = genius
				trait = brilliant
				trait = brilliant_strategist
				trait = barbarian_4
			}
		}
	}
}

# Someone who scares easily: not brave and not clever. Used when a beastman
# lord eats a prisoner in front of the court.
# Scope: a character.
beastmen_easily_scared_trigger = {
	NOR = {
		trait = brave
		trait = quick
		trait = bright
		trait = brilliant
		trait = genius
	}
}

# A gor strong and bold enough to challenge the chieftain for the herd. Same
# limits as the duel_beastman_decision challenger, plus a reason to try.
# Scope: a vassal or courtier of the chieftain.
beastmen_herd_challenger_trigger = {
	trait = creature_beastman
	is_female = no
	age = 16
	has_regent = no
	is_incapacitated_trigger = no
	NOR = {
		trait = blinded
		trait = craven
		trait = content
		has_character_flag = duel
		has_character_flag = do_not_disturb
		has_character_modifier = recent_duel_timer
	}
	OR = {
		trait = ambitious
		trait = brave
		trait = wroth
		trait = horns_big
		martial = 12
	}
}

# A ruler with a neighbouring province held by someone who is not a beastman
# and not their own liege: someone the herd can attack.
# Scope: a beastman ruler.
beastmen_has_prey_trigger = {
	any_realm_province = {
		any_neighbor_province = {
			owner = { NOT = { trait = creature_beastman } NOT = { is_liege_or_above = ROOT } }
		}
	}
}

# A ruler whose lands lie in or next to the Moot.
# Scope: a ruler.
beastmen_near_moot_trigger = {
	any_realm_province = {
		OR = {
			kingdom = { title = k_mootland }
			any_neighbor_province = { kingdom = { title = k_mootland } }
		}
	}
}

# A beastman ruler who has a bray shaman: themself, a courtier or a vassal.
# Scope: a ruler.
beastmen_has_bray_shaman_trigger = {
	OR = {
		trait = bray_shaman
		any_courtier = { trait = bray_shaman }
		any_vassal = { trait = bray_shaman }
	}
}

# Not ambushed by a halfling (beastmen.81) in the last two years.
# Scope: a beastman.
beastmen_halfling_ambush_ready_trigger = {
	OR = {
		NOT = { has_character_flag = beastmen_halfling_ambush }
		had_character_flag = { flag = beastmen_halfling_ambush days = 730 }
	}
}
