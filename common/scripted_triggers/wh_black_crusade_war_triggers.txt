# Triggers for Part 3 of events/WHG_black_crusade_events.txt (bcwar).

# This character fights in a crusade war (the vanilla crusade CB or the new
# Black Crusade CB).
# Scope: a character.
bcwar_in_crusade_war_trigger = {
	any_war = {
		OR = {
			using_cb = crusade
			using_cb = new_crusade
		}
	}
}

# A great independent lord of ROOT's faith who could march on the Black
# Crusade but has not pledged: no truce with the target's holder, not cast
# out by the faith's head, and no infidel neighbours to guard against.
# Scope: the lord (ROOT is the vassal or priest who would complain).
bcwar_lord_shirks_trigger = {
	independent = yes
	religion = ROOT
	prisoner = no
	NOT = { trait = incapable }
	higher_tier_than = COUNT
	bcwar_lord_still_shirks_trigger = yes
	NOT = {
		any_realm_province = {
			any_neighbor_province = {
				has_owner = yes
				NOT = { religion_group = chaos_gods_group }
			}
		}
	}
}

# The lord still has not pledged or joined the crusade war.
# Scope: the lord.
bcwar_lord_still_shirks_trigger = {
	NOT = { has_pledged_crusade_participation = yes }
	bcwar_in_crusade_war_trigger = no
	crusade_target = {
		holder_scope = { NOT = { reverse_has_truce = PREVPREV } }
	}
	religion_head = {
		PREV = {
			rightful_religious_head_scope = { character = PREVPREV }
			NOT = { excommunicated_for = PREV }
		}
	}
}
