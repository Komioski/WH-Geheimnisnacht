# A beastman warlord who can host the pilgrim on the road: an AI beastman
# king or emperor in the Old World, Norsca or the Troll Country who is not
# the pilgrim.
# Scope: a ruler. ROOT is the pilgrim.
brayherd_host_trigger = {
	ai = yes
	government = beastman_government
	higher_tier_than = DUKE
	NOT = { character = ROOT }
	prisoner = no
	capital_scope = {
		OR = { region = world_old_world region = world_norsca region = world_troll_country }
	}
	NOT = { block_general_event_trigger = yes }
}

# A courtier who came over from the Brayherd court (flag set in WHG_beastmen_events).
courtier_from_brayherd_trigger = {
	custom_tooltip = {
		text = courtier_from_brayherd_trigger_tt
		has_character_flag = originated_from_brayherd_court
	}
}
