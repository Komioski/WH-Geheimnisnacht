# Someone at home the pilgrim misses: a lover, spouse, parent, child or
# friend among the courtiers and vassals.
# Scope: the person. ROOT is the pilgrim.
brayherd_missed_one_trigger = {
	OR = {
		is_lover = ROOT
		any_spouse = { character = ROOT }
		is_mother = ROOT
		is_father = ROOT
		is_child_of = ROOT
		is_friend = ROOT
	}
}

# A beastman warlord who can host the pilgrim on the road: an AI beastman
# king or emperor in the Old World, Norsca or the Troll Country who is not
# the pilgrim.
# Scope: a ruler. ROOT is the pilgrim.
brayherd_host_trigger = {
	ai = yes
	government = beastman_government
	higher_tier_than = DUKE
	NOT = { character = ROOT }
	prisoner = no
	capital_scope = {
		OR = { region = world_old_world region = world_norsca region = world_troll_country }
	}
	NOT = { block_general_event_trigger = yes }
}

# A dead parent or sibling who did not die on an offmap journey, for the
# pilgrim to remember in the rain.
# Scope: the relative.
brayherd_mourned_relative_trigger = { is_alive = no death_in_cathay_trigger = no }
