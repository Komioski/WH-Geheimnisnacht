# Northern blood: a Kislevite of Kislev's gods, hardened by generations beside
# the Wastes. A small edge against sickness, madness and mutation
# (WHG_health_events, WHG_health_events lunacy.1, mutate.5 and mutate.9,
# WHG_mutation_events mutate.1).
# Scope: character.
kislev_northern_blood_trigger = {
	culture_group = kislevite_group
	religion = kislev_gods
	# A cold hearth or the Long Sleep takes it away (WHG_kislev_faith_events)
	top_liege = {
		NOR = {
			has_character_modifier = ursun_cold_hearths
			has_character_modifier = ursun_long_sleep
		}
	}
}

# A commander the Kislev winter does not bite: Kislevites, the peoples of the
# Wastes and Norsca, the dead and daemons (bulwark.1).
# Scope: character.
bulwark_winter_immune_trigger = {
	OR = {
		culture_group = kislevite_group
		culture_group = norscan_group
		culture_group = kurgan_group
		culture_group = hung_group
		is_undead_trigger = yes
		is_daemon_race_trigger = yes
	}
}

# ROOT is defending a war against a Chaos attacker.
# Scope: ROOT, a ruler.
bulwark_defending_vs_chaos_trigger = {
	any_war = {
		defender = { character = ROOT }
		attacker = { religion_group = chaos_gods_group }
	}
}

# A ruler who holds land in northern Kislev, where the Spring Driving comes
# (bulwark.10).
# Scope: character.
bulwark_northern_lord_trigger = {
	is_ruler = yes
	culture_group = kislevite_group
	NOT = { religion_group = chaos_gods_group }
	any_demesne_province = { region = world_old_world_kislev_northern }
}

# The one who calls the Old World to arms for Kislev: the ruler of the
# Tsardom, or, while the Tsardom has no holder, a Kislevite king.
# Scope: character.
bulwark_kislev_crown_trigger = {
	OR = {
		has_landed_title = e_kislev
		AND = {
			e_kislev = { has_holder = no }
			independent = yes
			culture_group = kislevite_group
			higher_tier_than = DUKE
		}
	}
}

# A great power of order the Tzar's Call reaches: the Emperor and the
# electors, the Dwarf kings, the King of Bretonnia, the Phoenix King, and the
# princes of Tilea and Estalia (bulwark.60).
# Scope: an independent ruler; ROOT the caller.
bulwark_order_power_trigger = {
	independent = yes
	higher_tier_than = COUNT
	is_adult = yes
	prisoner = no
	NOT = { character = ROOT }
	NOT = { war_with = ROOT }
	NOT = { religion_group = chaos_gods_group }
	OR = {
		AND = {
			culture_group = imperial_group
			religion_group = old_world_group
		}
		AND = {
			culture_group = dwarf_group
			religion_group = dwarf_gods_group
			higher_tier_than = DUKE
		}
		AND = {
			culture_group = bretonnian_group
			higher_tier_than = DUKE
		}
		AND = {
			culture_group = asur_group
			higher_tier_than = KING
		}
		AND = {
			OR = { culture_group = tilean_group culture_group = estalian_group }
			religion_group = old_world_group
		}
	}
}

# A great city of Kislev: Kislev, Praag or Erengrad.
# Scope: province.
bulwark_great_city_trigger = {
	OR = {
		province_id = 202
		province_id = 182
		province_id = 175
	}
}

# The Gate Holds: Kislev, Praag and Erengrad are all held by Kislev's gods
# (bulwark.85).
bulwark_gate_holds_trigger = {
	202 = { owner = { religion = kislev_gods } }
	182 = { owner = { religion = kislev_gods } }
	175 = { owner = { religion = kislev_gods } }
}
