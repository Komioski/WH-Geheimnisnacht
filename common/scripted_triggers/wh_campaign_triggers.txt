# Triggers for the campaign events (Part 2 of events/WHG_battle_events.txt).

# Another vassal of the same liege who commands at the same place, is not at
# war with ROOT and is neither a friend nor an enemy yet.
# Scope: a vassal of ROOT's liege.
campaign_comrade_trigger = {
	NOT = { character = ROOT }
	in_command = yes
	at_location = ROOT
	NOT = { war_with = ROOT }
	opinion = { who = ROOT value = -30 }
	NOT = { opinion = { who = ROOT value = 30 } }
}
