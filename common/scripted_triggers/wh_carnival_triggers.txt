# The lands the Carnival roams: the Empire, Tilea, Estalia, Bretonnia and
# Kislev.
# Scope: a province.
carnival_region_trigger = {
	OR = {
		region = world_old_world_empire
		region = world_old_world_tilea
		region = world_old_world_estalia
		region = world_old_world_bretonnia
		region = world_old_world_kislev
	}
}

# A ruler the Carnival can visit: a living human lord, duke or higher, who
# does not openly worship the Dark Gods and has not had the Carnival before.
# Scope: a character.
carnival_host_trigger = {
	is_ruler = yes
	higher_tier_than = COUNT
	is_adult = yes
	prisoner = no
	trait = creature_human
	NOT = { religion_group = chaos_gods_group }
	NOT = { has_character_flag = carnival_visited }
	capital_scope = { carnival_region_trigger = yes }
	any_realm_province = { carnival_region_trigger = yes is_capital = no }
}

# Someone who can see through the paint: a witch hunter, an Inquisitor,
# a Templar or a Chekist, a trained wizard, an Ice Witch or hag, a Grail
# Knight or a Damsel of the Lady.
# Scope: a character.
carnival_seer_trigger = {
	OR = {
		is_witch_hunter_trigger = yes
		is_trained_mage_trigger = yes
		trait = ice_witch
		trait = hag_witch
		trait = grail_knight
		society_member_of = handmaidens_of_the_lady
	}
}

# The Master still lives and walks free.
# Scope: any.
carnival_master_free_trigger = { event_target:carnival_master = { is_alive = yes prisoner = no } }

# Regional voices: where the Carnival has stopped.
# Scope: any (reads the global target carnival_town).
carnival_voice_empire_trigger = { event_target:carnival_town = { region = world_old_world_empire } }
carnival_voice_tilea_trigger = { event_target:carnival_town = { region = world_old_world_tilea } }
carnival_voice_estalia_trigger = {
	event_target:carnival_town = { region = world_old_world_estalia }
}
carnival_voice_bretonnia_trigger = {
	event_target:carnival_town = { region = world_old_world_bretonnia }
}
carnival_voice_kislev_trigger = { event_target:carnival_town = { region = world_old_world_kislev } }
