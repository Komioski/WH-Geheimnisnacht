# A courtier the cult can name as the target of the demonic possession
# mission (chaoscult.181): an adult mortal of a race a daemon can take.
# The mission tombola (chaoscult.200) only offers the mission when the
# cultist has one at court.
# Scope: the courtier.
chaoscult_possession_target_trigger = {
	ai = yes
	is_adult = yes
	NOT = { trait = avatar_kurnous }
	NOT = { trait = avatar_isha }
	NOT = { trait = everqueen }
	OR = {
		trait = creature_elf
		trait = creature_chaos_dwarf
		trait = creature_dwarf
		trait = creature_human
		trait = creature_amazon
		trait = creature_beastman
	}
	is_vampire_trigger = no
}

###############################################################################
# The hunters come (chaoscult.440 - 441)
###############################################################################

# Scope: a fellow cultist; ROOT is the hunted cultist. Can be given up to the
# hunters: free, able and not ROOT.
chaoscult_hunt_blamable_trigger = {
	NOT = { character = ROOT }
	is_alive = yes
	prisoner = no
	is_incapable = no
}

# Scope: the hunted cultist. The hunters come for the first time.
chaoscult_hunt_first_time_trigger = {
	NOR = {
		has_character_flag = cult_hunt_sister_again
		has_character_flag = cult_hunt_mob_again
		has_character_flag = cult_hunt_templars_again
		has_character_flag = cult_hunt_hunter_again
		has_character_flag = cult_hunt_knights_again
	}
}
