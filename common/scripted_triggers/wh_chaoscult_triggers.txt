# A courtier the cult can name as the target of the demonic possession
# mission (chaoscult.181): an adult mortal of a race a daemon can take.
# The mission tombola (chaoscult.200) only offers the mission when the
# cultist has one at court.
# Scope: the courtier.
chaoscult_possession_target_trigger = {
	ai = yes
	is_adult = yes
	NOT = { trait = avatar_kurnous }
	NOT = { trait = avatar_isha }
	NOT = { trait = everqueen }
	OR = {
		trait = creature_elf
		trait = creature_chaos_dwarf
		trait = creature_dwarf
		trait = creature_human
		trait = creature_amazon
		trait = creature_beastman
	}
	is_vampire_trigger = no
}

###############################################################################
# The hunters come (chaoscult.440 - 441)
###############################################################################

# Scope: a fellow cultist; ROOT is the hunted cultist. Can be given up to the
# hunters: free, able and not ROOT.
chaoscult_hunt_blamable_trigger = {
	NOT = { character = ROOT }
	is_alive = yes
	prisoner = no
	is_incapable = no
}

# Scope: the hunted cultist. The hunters come for the first time.
chaoscult_hunt_first_time_trigger = {
	NOR = {
		has_character_flag = cult_hunt_sister_again
		has_character_flag = cult_hunt_mob_again
		has_character_flag = cult_hunt_templars_again
		has_character_flag = cult_hunt_hunter_again
		has_character_flag = cult_hunt_knights_again
	}
}

###############################################################################
# Life in the cult (chaoscult.450 - 484), salvaged from the vanilla secret
# religious societies
###############################################################################

# Scope: any character. A member of one of the five Chaos cults (not the
# Dark Elf Cult of Pleasure, which has its own events).
chaoscult_member_trigger = {
	OR = {
		society_member_of = cult_of_khorne
		society_member_of = cult_of_nurgle
		society_member_of = cult_of_tzeentch
		society_member_of = cult_of_slaanesh
		society_member_of = cult_of_undivided
	}
}

# Scope: a fellow cultist; ROOT is a stressed cultist (chaoscult.450). An AI
# member in reach who likes ROOT and is liked back: a friend, or someone
# ROOT is drawn to.
chaoscult_solace_giver_trigger = {
	ai = yes
	NOT = { character = ROOT }
	block_general_event_trigger = no
	is_within_diplo_range = ROOT
	opinion = { who = ROOT value = 20 }
	reverse_opinion = { who = ROOT value = 20 }
	NOT = { is_lover = ROOT }
	OR = {
		is_friend = ROOT
		AND = {
			root_attraction_trigger = yes
			NOT = { is_close_relative = ROOT }
		}
	}
}

# Scope: a fellow cultist; ROOT is a cultist with a flaw (chaoscult.455).
# Someone able to coach ROOT.
chaoscult_flaw_helper_trigger = {
	NOT = { character = ROOT }
	block_general_event_trigger = no
	is_within_diplo_range = ROOT
}

# Scope: an educator, guardian or parent of a child (chaoscult.465). A Chaos
# cultist who can still act.
chaoscult_child_teacher_trigger = {
	is_alive = yes
	chaoscult_member_trigger = yes
	is_incapable = no
	prisoner = no
}

# Scope: a fellow cultist; ROOT is the cultist (chaoscult.475). An AI member
# below grand master, in reach, who might want to leave.
chaoscult_cold_feet_trigger = {
	ai = yes
	NOT = { character = ROOT }
	is_society_grandmaster = no
	is_within_diplo_range = ROOT
	is_incapable = no
}

# Scope: a fellow cultist; ROOT is the cultist. Outranks ROOT in the cult.
chaoscult_outranks_root_trigger = {
	OR = {
		AND = {
			society_rank >= 2
			ROOT = { society_rank == 1 }
		}
		AND = {
			society_rank >= 3
			ROOT = { society_rank == 2 }
		}
		AND = {
			society_rank >= 4
			ROOT = { society_rank == 3 }
		}
	}
}
