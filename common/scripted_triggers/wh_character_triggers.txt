###############################################################################
# Character flavour triggers
# Used by events/WHG_character_events.txt (namespace charfl) and the
# elective-commander part of events/WHG_realm_events.txt (realm.80 - 83).
###############################################################################

# A close relative fit to beg a drunkard to stop: grown, free, sound and
# sober themselves.
# Scope: the relative.
charfl_sober_relative_trigger = {
	is_adult = yes
	prisoner = no
	is_incapable = no
	NOT = { trait = drunkard }
}

# A parent through whom freckles can pass: freckled, of a freckled people,
# or the child of a freckled people.
# Scope: the parent (mother_even_if_dead or true_father_even_if_dead).
charfl_freckled_line_trigger = {
	OR = {
		has_freckles = yes
		has_freckles_culture = yes
		mother_even_if_dead = { has_freckles_culture = yes }
		true_father_even_if_dead = { has_freckles_culture = yes }
	}
}

# The falling star (charfl.50 - 51) is voiced like the sway voices
# (wh_sway_triggers.txt), with the Dawi and the Sigmarites split off the
# court voice.
# Scope: the ruler.
charfl_voice_dawi_trigger = {
	sway_voice_court_trigger = yes
	culture_group = dwarf_group
}

charfl_voice_sigmar_trigger = {
	sway_voice_court_trigger = yes
	NOT = { culture_group = dwarf_group }
	religion = sigmarite
}

# Everyone else: ogres, lizardmen, Albion and the rest of the court voice.
charfl_voice_other_trigger = {
	charfl_voice_dawi_trigger = no
	charfl_voice_sigmar_trigger = no
	sway_voice_skaven_trigger = no
	sway_voice_greenskin_trigger = no
	sway_voice_warband_trigger = no
	sway_voice_undead_trigger = no
}

###############################################################################
# Elective commanders (realm.80 - 82 in events/WHG_realm_events.txt)
###############################################################################

# Holds no other post at court, so a dismissal really sent them away.
# Scope: the dismissed commander or marshal.
realm_holds_no_court_post_trigger = {
	NOR = {
		has_job_title = job_marshal
		has_job_title = job_chancellor
		has_job_title = job_treasurer
		has_job_title = job_spymaster
		has_job_title = job_spiritual
		has_minor_title = title_commander
	}
}

# The unicorn prophecy (charfl.70 - 75) comes to horses, the Asrai and the
# faithful of the Lady, each in their own voice (first match wins).
# Scope: the seeker.
charfl_unicorn_seeker_trigger = {
	OR = {
		graphical_culture = horsegfx
		culture_group = asrai_group
		religion = grail
	}
}

charfl_unicorn_horse_trigger = {
	graphical_culture = horsegfx
}

charfl_unicorn_asrai_trigger = {
	charfl_unicorn_horse_trigger = no
	culture_group = asrai_group
}

charfl_unicorn_grail_trigger = {
	charfl_unicorn_horse_trigger = no
	charfl_unicorn_asrai_trigger = no
	religion = grail
}

# The secret bear (charfl.80 - 86).
# A young relative at court who could be a bear in disguise.
# Scope: the relative.
charfl_bear_candidate_trigger = {
	ai = yes
	is_adult = yes
	age < 40
	prisoner = no
	is_incapable = no
	host = { character = ROOT }
	NOT = { has_character_flag = secret_bear }
	NOT = { culture = bear }
}

# The bear is still at the ruler's court.
# Scope: the ruler.
charfl_bear_at_court_trigger = {
	event_target:bear_character = {
		is_alive = yes
		prisoner = no
		host = { character = ROOT }
	}
}

# Kislev: Ursun's people.
# Scope: the ruler.
charfl_voice_kislev_trigger = {
	OR = {
		culture_group = kislevite_group
		religion = kislev_gods
	}
}
