# Coronation triggers, used by events/WHG_coronation_events.txt and
# decisions/HF_coronation_decisions.txt.

# A faith whose kings are crowned by its priests: the great cults of the
# Empire. Scope: a character.
coronation_faith_trigger = {
	OR = { religion = sigmarite religion = ulrican religion = taalite religion = morrian }
}

# The head of a great cult of the Empire: the Grand Theogonist, the Ar-Ulric,
# the High Priest of Taal and Rhya or the high priest of Morr.
# Scope: a character.
coronation_cult_head_trigger = {
	OR = {
		has_landed_title = k_cult_of_sigmar
		has_landed_title = k_cult_of_ulric
		has_landed_title = k_cult_of_taal_and_rhya
		has_landed_title = k_cult_of_morr
	}
}

# Has been crowned, by anyone. Scope: a character.
coronation_crowned_trigger = {
	OR = {
		trait = crowned_by_priest
		trait = crowned_by_bishop
		trait = crowned_by_pope
		trait = crowned_by_myself
		trait = crowned_by_priest_rd
		trait = crowned_by_bishop_rd
		trait = crowned_by_pope_rd
	}
}

# A child of the ruler old enough to be curious about the crown, at court
# and free. Scope: a character.
coronation_curious_child_trigger = {
	is_alive = yes
	age = 7
	NOT = { age = 14 }
	NOT = { trait = incapable }
	prisoner = no
	NOT = { is_inaccessible_trigger = yes }
	at_location = ROOT
}

# Claims the high seat of their own cult without holding it: a pretender
# Grand Theogonist, Ar-Ulric or high priest. Scope: a character.
coronation_cult_claim_trigger = {
	OR = {
		AND = {
			religion = sigmarite
			has_claim = k_cult_of_sigmar
			NOT = { has_landed_title = k_cult_of_sigmar }
		}
		AND = {
			religion = ulrican
			has_claim = k_cult_of_ulric
			NOT = { has_landed_title = k_cult_of_ulric }
		}
		AND = {
			religion = taalite
			has_claim = k_cult_of_taal_and_rhya
			NOT = { has_landed_title = k_cult_of_taal_and_rhya }
		}
		AND = {
			religion = morrian
			has_claim = k_cult_of_morr
			NOT = { has_landed_title = k_cult_of_morr }
		}
	}
}

# A ruler of ROOT's faith who backs a pretender to the high seat: they
# follow a false head of the cult, or shelter someone who claims the seat.
# Scope: a ruler; ROOT shares the faith.
coronation_pretender_host_trigger = {
	religion = ROOT
	OR = {
		rightful_religious_head_scope = { coronation_cult_head_trigger = no }
		coronation_cult_claim_trigger = yes
		any_realm_character = { coronation_cult_claim_trigger = yes }
	}
}

# A war demand of the cult is still open, so the coronation is on hold.
# Scope: a character.
coronation_war_demand_trigger = {
	OR = {
		has_character_flag = flag_excommunicated_for_coronation
		has_character_flag = flag_antipope_for_coronation
		has_character_flag = flag_pepin_for_coronation
		has_character_flag = flag_holy_site_for_coronation
	}
}
