# Triggers for events/WHG_court_events.txt.

# A child of ROOT, 9 or older, guarded by a hero knight, not yet taught.
# Scope: the child.
heroknight_pupil_trigger = {
	age = 9
	is_adult = no
	prisoner = no
	NOT = { has_character_flag = hero_knight_taught }
	guardian = {
		has_character_flag = hero_knight
		prisoner = no
	}
}

# A court that keeps an Imperial Eunuch: a Myken king or emperor of
# Myrmidia, or a Cathayan king or emperor (eunuch.1 - 6, and
# title_imperial_eunuch in 00_minor_titles.txt).
# Scope: the ruler.
eunuch_court_trigger = {
	is_feudal = yes
	higher_tier_than = DUKE
	OR = {
		AND = {
			culture = myken
			religion = myrmidian
		}
		culture_group = oriental_group
	}
}
