# Triggers for events/WHG_court_events.txt.

# A child of ROOT, 9 or older, guarded by a hero knight, not yet taught.
# Scope: the child.
heroknight_pupil_trigger = {
	age = 9
	is_adult = no
	prisoner = no
	NOT = { has_character_flag = hero_knight_taught }
	guardian = { has_character_flag = hero_knight prisoner = no }
}

# A court that keeps an Imperial Eunuch: a Myken king or emperor of
# Myrmidia, or a Cathayan king or emperor (eunuch.1 - 6, and
# title_imperial_eunuch in 00_minor_titles.txt).
# Scope: the ruler.
eunuch_court_trigger = {
	is_feudal = yes
	higher_tier_than = DUKE
	OR = { AND = { culture = myken religion = myrmidian } culture_group = oriental_group }
}

# A courtier who may ask for a council seat (court.20): an adult of the
# court, not a ruler or councillor, who may hold a seat in this realm.
# Scope: the courtier; ROOT is the liege.
# The holder's seat can be asked for. With Conclave a landed vassal's seat
# is his by right of power and vote, so only a courtier's seat is open.
# Scope: the councillor.
court_petition_seat_open_trigger = { OR = { NOT = { has_dlc = "Zeus" } is_ruler = no } }

court_petitioner_trigger = {
	is_adult = yes
	is_ruler = no
	is_councillor = no
	prisoner = no
	is_inaccessible_or_incapable_trigger = no
	NOT = { has_character_flag = asked_for_job }
	OR = {
		court_petition_marshal_trigger = yes
		court_petition_spymaster_trigger = yes
		court_petition_treasurer_trigger = yes
		court_petition_chancellor_trigger = yes
		court_petition_spiritual_trigger = yes
	}
}

# The courtier would make a better marshal than the holder (court.20).
# Scope: the courtier; ROOT is the liege.
court_petition_marshal_trigger = {
	can_hold_title = job_marshal
	ROOT = {
		job_marshal = {
			NOT = { martial = PREVPREV }
			NOT = { has_minor_title = title_regent }
			court_petition_seat_open_trigger = yes
		}
	}
}

# A young unwed son who may get a servant girl with child (court.16).
# Scope: the son.
court_randy_son_trigger = {
	is_female = no
	is_no_females_race = no
	is_married = no
	age = 16
	NOT = { age = 21 }
	NOT = { demesne_size = 1 }
	NOR = { trait = chaste trait = celibate trait = eunuch is_ascetic_trigger = yes }
}

# Unwell enough to care less for court life (weights in court.10, 15, 19).
# Scope: the ruler.
court_unwell_trigger = {
	OR = {
		trait = stressed
		trait = depressed
		trait = lunatic_1
		trait = lunatic_2
		trait = lunatic_3
		trait = possessed
		trait = infirm
		trait = incapable
		trait = drunkard
		is_ill = yes
		is_maimed_trigger = yes
	}
}

# court_drift_mediocre_trigger
# Scope: a courtier. An unlanded free adult whose skills all sit between 4
# and 9: the dull middle of the court that court.30 may lift or sink.
court_drift_mediocre_trigger = {
	is_adult = yes
	is_landed = no
	prisoner = no
	stewardship = 4
	diplomacy = 4
	martial = 4
	intrigue = 4
	learning = 4
	NOR = { stewardship = 10 diplomacy = 10 martial = 10 intrigue = 10 learning = 10 }
}

# court_lonely_courtier_trigger
# Scope: a courtier. An unlanded free adult with no lover, friend or rival
# (court.30).
court_lonely_courtier_trigger = {
	is_adult = yes
	is_landed = no
	prisoner = no
	has_lover = no
	NOT = { any_rival = { always = yes } }
	NOT = { any_friend = { always = yes } }
}

# court_nickname_seeker_trigger
# Scope: a courtier. An unlanded free adult without a nickname (court.32).
court_nickname_seeker_trigger = { is_adult = yes is_landed = no prisoner = no has_nickname = no }

# court_exodus_candidate_trigger
# Scope: a courtier, ROOT the ruler. Someone who may leave in the court
# exodus (court.34): a free unlanded adult who matters to no one, whose
# spouses hold no land and none of whose children is the ruler's own or
# matters to the ruler.
court_exodus_candidate_trigger = {
	prisoner = no
	is_adult = yes
	is_landed = no
	important_character_to_root_trigger = no
	NOT = { any_spouse = { is_landed = yes } }
	NOT = { any_child = { OR = { is_child_of = ROOT important_character_to_root_trigger = yes } } }
}

# court_host_guest_trigger
# Scope: a courtier, ROOT the ruler. An adult who can speak to the ruler,
# even one in seclusion (court.36).
court_host_guest_trigger = {
	is_adult = yes
	OR = {
		is_inaccessible_or_incapable_trigger = no
		AND = { ROOT = { in_seclusion = yes } prisoner = no is_incapable = no }
	}
}
