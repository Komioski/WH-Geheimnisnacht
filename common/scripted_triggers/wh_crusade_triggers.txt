# Crusade triggers, used by events/WHG_crusade_events.txt, the southern
# crusade CBs and decisions/HFP_holy_order_decisions.txt.

# Holds the Knights of Antoch (the duchy order or the kingdom it becomes).
# Scope: a character.
crusade_south_order_trigger = {
	OR = { has_landed_title = d_knights_of_antoch has_landed_title = k_knights_of_antoch }
}

# The Knights of Antoch already have a kingdom to conquer.
# Scope: any.
crusade_south_target_set_trigger = {
	OR = {
		d_knights_of_antoch = { has_title_flag = northern_crusade_target }
		k_knights_of_antoch = { has_title_flag = northern_crusade_target }
	}
}

# The lands the Errantry War may be fought over: the Old World, the
# Badlands and Araby. Scope: a title or province.
crusade_south_region_trigger = {
	OR = { region = world_old_world region = world_badlands region = world_greater_araby_araby }
}

# A ruler of Araby the knights may march on. Scope: a character.
crusade_south_target_ruler_trigger = {
	independent = yes
	religion_group = ormazd_gods_group
	NOR = { holy_order = yes mercenary = yes }
	any_demesne_title = { higher_tier_than = COUNT crusade_south_region_trigger = yes }
}

# Every county of the kingdom is held by a faithful of the Old World: the
# Errantry War for it is won. Scope: a kingdom title.
crusade_south_kingdom_won_trigger = {
	has_title_flag = northern_crusade_active_target
	NOT = {
		any_direct_de_jure_vassal_title = {
			any_direct_de_jure_vassal_title = {
				owner = { NOT = { religion_group = old_world_group } }
			}
		}
	}
}

# The Knights of Antoch are at war on the Errantry War. Scope: any.
crusade_south_war_running_trigger = {
	OR = {
		d_knights_of_antoch = { holder_scope = { any_war = { using_cb = southern_crusade_war } } }
		k_knights_of_antoch = { holder_scope = { any_war = { using_cb = southern_crusade_war } } }
	}
}

# A living Knight Errant of the Knights of Bretonnia. Scope: a character.
crusade_south_knight_errant_trigger = {
	trait = bretonnian_knight_errant
	society_member_of = knights_of_bretonnia
	society_rank == 1
	is_adult = yes
	prisoner = no
}
