# Chaos cult mission triggers, used by events/WHG_chaos_cult_events.txt
# Part 2 (cultpath).

# The member has neither of the two focuses their god loves (cultpath.31).
# Scope: the member.
cultpath_lacks_god_focus_trigger = {
	OR = {
		AND = {
			society_member_of = cult_of_nurgle
			NOR = { has_focus = focus_family has_focus = focus_business }
		}
		AND = {
			society_member_of = cult_of_slaanesh
			NOR = { has_focus = focus_seduction has_focus = focus_carousing }
		}
		AND = {
			society_member_of = cult_of_tzeentch
			NOR = { has_focus = focus_scholarship has_focus = focus_intrigue }
		}
		AND = {
			society_member_of = cult_of_khorne
			NOR = { has_focus = focus_war has_focus = focus_hunting }
		}
		AND = {
			society_member_of = cult_of_undivided
			NOR = { has_focus = focus_war has_focus = focus_rulership }
		}
	}
}

# The member now has the focus their focus mission asked for (cultpath.32).
# Scope: the member.
cultpath_focus_quest_done_trigger = {
	OR = {
		AND = { has_quest = quest_chaos_cult_weak_focus_to_family has_focus = focus_family }
		AND = { has_quest = quest_chaos_cult_weak_focus_to_business has_focus = focus_business }
		AND = { has_quest = quest_chaos_cult_weak_focus_to_seduction has_focus = focus_seduction }
		AND = { has_quest = quest_chaos_cult_weak_focus_to_carousing has_focus = focus_carousing }
		AND = { has_quest = quest_chaos_cult_weak_focus_to_scholarship has_focus = focus_scholarship }
		AND = { has_quest = quest_chaos_cult_weak_focus_to_intrigue has_focus = focus_intrigue }
		AND = { has_quest = quest_chaos_cult_weak_focus_to_war has_focus = focus_war }
		AND = { has_quest = quest_chaos_cult_un_weak_focus_to_war has_focus = focus_war }
		AND = { has_quest = quest_chaos_cult_weak_focus_to_hunting has_focus = focus_hunting }
		AND = { has_quest = quest_chaos_cult_weak_focus_to_rulership has_focus = focus_rulership }
	}
}

# A cult brother fit for a first-blood duel (cultpath.41).
# Scope: the brother.
cultpath_duel_fit_trigger = {
	age = 16
	NOT = { age = 60 }
	is_maimed_trigger = no
	has_medium_disability_trigger = no
	has_severe_disability_trigger = no
	character_disease_trigger = no
	is_inaccessible_or_incapable_trigger = no
}

# A foe for the duel to the death (cultpath.43): an adult AI outside the
# cult, healthy enough to fight, not the member's spouse, within reach.
# Scope: the foe; ROOT is the member.
cultpath_deadly_duel_target_trigger = {
	ai = yes
	is_adult = yes
	NOT = { same_society_as = ROOT }
	NOT = { character = ROOT }
	NOT = { any_spouse = { character = ROOT } }
	is_inaccessible_or_incapable_trigger = no
	NOT = { has_character_modifier = bedridden_illness }
	NOT = { has_character_modifier = severe_illness }
	is_within_diplo_range = ROOT
}

# A ruler the Slaaneshi member is to seduce (cultpath.51): of the member's
# faith, within reach, and drawn to the member.
# Scope: the target; ROOT is the member.
cultpath_seduce_target_trigger = {
	ai = yes
	NOT = { character = ROOT }
	NOT = { any_spouse = { character = ROOT } }
	NOT = { is_lover = ROOT }
	religion_group = ROOT
	is_within_diplo_range = ROOT
	is_inaccessible_trigger = no
	prisoner = no
	NOT = { trait = incapable }
	higher_tier_than = BARON
	is_no_females_race = no
	OR = {
		AND = { # A man drawn to women, or a woman to men
			ROOT = { NOT = { trait = homosexual } }
			NOT = { trait = homosexual }
			OR = {
				AND = { ROOT = { is_woman_trigger = yes } is_female = no }
				AND = { ROOT = { is_female = no } is_woman_trigger = yes }
			}
		}
		AND = { # Drawn to one's own sex
			ROOT = { trait = homosexual }
			trait = homosexual
			OR = {
				AND = { ROOT = { is_woman_trigger = yes } is_woman_trigger = yes }
				AND = { ROOT = { is_female = no } is_female = no }
			}
		}
	}
}
