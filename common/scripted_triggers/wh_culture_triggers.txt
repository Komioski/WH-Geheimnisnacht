###############################################################################

###############################################################################

# Cultures that never change and never spread by assimilation: the Tomb
# Kings, Beastmen, greenskins, the hunter cultures, the Blood Dragons and the
# living jade.
# Scope: character or province.
culture_fixed_culture_trigger = {
	OR = {
		culture_group = nehekharan_group
		culture_group = beastman_group
		culture_group = orc_group
		culture_group = goblin_group
		is_hunter_culture_trigger = yes
		culture = blooddragons_culture
		culture = living_jade
	}
}

# ROOT, or the duchy, kingdom or empire ROOT holds over this province,
# carries ROOT's culture as its conquest culture.
# Scope: a province owned by ROOT.
culture_conquest_culture_trigger = {
	OR = {
		ROOT = { primary_title = { conquest_culture = ROOT } }
		county = { conquest_culture = ROOT }
		duchy = { holder_scope = { character = ROOT } conquest_culture = ROOT }
		kingdom = { holder_scope = { character = ROOT } conquest_culture = ROOT }
		empire = { holder_scope = { character = ROOT } conquest_culture = ROOT }
	}
}

# A neighbouring province of ROOT's culture whose owner is of ROOT's culture
# too and that is not depopulated: the culture spreads from there.
# Scope: a province owned by ROOT.
culture_has_neighbour_of_root_culture_trigger = {
	any_neighbor_province = {
		has_owner = yes
		culture = ROOT
		owner = { culture = ROOT }
		NOR = {
			has_province_modifier = depopulated_1
			has_province_modifier = depopulated_2
			has_province_modifier = depopulated_3
		}
	}
}

# A province of ROOT's realm that may take up ROOT's culture from its
# neighbours (culture.2).
# Scope: a province owned by ROOT.
culture_assimilates_trigger = {
	NOT = { culture_group = skaven_group }
	culture_fixed_culture_trigger = no
	NOT = { culture = ROOT }
	religion = ROOT
	culture_has_neighbour_of_root_culture_trigger = yes
}

# ROOT's capital, cut off from ROOT's culture, may draw settlers from the
# conquerors' homeland (culture.5).
# Scope: ROOT's capital province.
culture_settlers_trigger = {
	culture_fixed_culture_trigger = no
	NOT = { culture = ROOT }
	OR = {
		port = yes
		NOR = {
			has_game_rule = { name = culture_conversion value = restricted }
			has_game_rule = {
				name = culture_conversion
				value = faster_melting_pots_and_slower_normal
			}
		}
	}
	culture_conquest_culture_trigger = yes
	culture_has_neighbour_of_root_culture_trigger = no
}

# The "slower normal conversion" culture conversion game rule is on.
culture_slower_conversion_trigger = {
	OR = {
		has_game_rule = { name = culture_conversion value = slower_normal }
		has_game_rule = { name = culture_conversion value = faster_melting_pots_and_slower_normal }
	}
}
