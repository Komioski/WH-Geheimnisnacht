# A lesser daemon bound to a mortal: a Bloodletter, Plaguebearer, Daemonette,
# Flamer, Undivided daemon or K'daai. Daemon princes and the commanders of a
# summoned daemonic host are not bound familiars.
# Scope: a character.
daemon_bound_trigger = {
	OR = {
		trait = khorne_demon
		trait = undivided_demon
		trait = nurgle_demon
		trait = slaanesh_demon
		trait = tzeentch_demon
		trait = hashut_kdaai
	}
	NOR = {
		trait = prince_khorne
		trait = prince_nurgle
		trait = prince_undivided
		trait = prince_slaanesh
		trait = prince_tzeentch
		has_character_flag = host_demon_commander
	}
}

# A daemon of a god at war with ROOT's god: Khorne, Nurgle, Slaanesh,
# Tzeentch and Hashut's K'daai all hate one another.
# Scope: a character; ROOT is a daemon.
daemon_rival_of_root_trigger = {
	NOT = { character = ROOT }
	OR = {
		AND = {
			ROOT = { trait = khorne_demon }
			OR = { trait = slaanesh_demon trait = nurgle_demon trait = tzeentch_demon trait = hashut_kdaai }
		}
		AND = {
			ROOT = { trait = slaanesh_demon }
			OR = { trait = khorne_demon trait = nurgle_demon trait = tzeentch_demon trait = hashut_kdaai }
		}
		AND = {
			ROOT = { trait = nurgle_demon }
			OR = { trait = khorne_demon trait = slaanesh_demon trait = tzeentch_demon trait = hashut_kdaai }
		}
		AND = {
			ROOT = { trait = tzeentch_demon }
			OR = { trait = khorne_demon trait = slaanesh_demon trait = nurgle_demon trait = hashut_kdaai }
		}
		AND = {
			ROOT = { trait = hashut_kdaai }
			OR = { trait = khorne_demon trait = slaanesh_demon trait = nurgle_demon trait = tzeentch_demon }
		}
	}
}

# A bound daemon at court that a witch hunter can corner.
# Scope: a character.
daemon_hunt_quarry_trigger = {
	daemon_bound_trigger = yes
	prisoner = no
	NOT = { trait = incapable }
}

# A courtier a possessed ruler can lash out at: an adult mortal, not a daemon,
# a living god or someone death cannot touch.
# Scope: a character.
daemon_possessed_victim_trigger = {
	is_adult = yes
	NOR = {
		culture_group = daemonic_group
		trait = avatar_isha
		trait = avatar_kurnous
		trait = everqueen
		has_artifact = carstein_ring
		trait = creature_nehekharan
		trait = creature_forestspirit
	}
}

# The Slaaneshi daemon in this character craves men (a woman, or a man who
# loves men); otherwise it craves women. Decides which sex the realm's laws
# favour in daemon.111.
# Scope: a character.
daemon_craves_men_trigger = {
	OR = {
		AND = { is_female = yes NOT = { trait = homosexual } }
		AND = { is_female = no trait = homosexual }
	}
}
