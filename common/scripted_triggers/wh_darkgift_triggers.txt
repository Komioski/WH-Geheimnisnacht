###############################################################################
# Dark gifts and the Eye of the Gods: triggers
# Used by events/WHG_chaos_undivided_events.txt, the dark gift effects
# (common/scripted_effects/wh_darkgift_effects.txt) and the patron names in
# localisation/customizable_localisation/wh_darkgift_customizable_localisation.txt
#
# A character's patron is the god whose rank or mark they carry: Chosen first,
# then Champion, then a bare mark. Chaos Undivided ranks are chaos_chosen,
# champion_undivided and mark_chaos.
###############################################################################

# The character serves Khorne. Scope: a character.
darkgift_patron_khorne_trigger = {
	OR = { trait = chosen_khorne trait = champion_khorne trait = mark_khorne }
}

# The character serves Nurgle. Scope: a character.
darkgift_patron_nurgle_trigger = {
	OR = { trait = chosen_nurgle trait = champion_nurgle trait = mark_nurgle }
}

# The character serves Slaanesh. Scope: a character.
darkgift_patron_slaanesh_trigger = {
	OR = { trait = chosen_slaanesh trait = champion_slaanesh trait = mark_slaanesh }
}

# The character serves Tzeentch. Scope: a character.
darkgift_patron_tzeentch_trigger = {
	OR = { trait = chosen_tzeentch trait = champion_tzeentch trait = mark_tzeentch }
}

# The character serves Chaos Undivided. A god's own rank or mark wins over
# an Undivided one. Scope: a character.
darkgift_patron_undivided_trigger = {
	darkgift_patron_khorne_trigger = no
	darkgift_patron_nurgle_trigger = no
	darkgift_patron_slaanesh_trigger = no
	darkgift_patron_tzeentch_trigger = no
	OR = { trait = chaos_chosen trait = champion_undivided trait = mark_chaos }
}

# The character is a mortal servant of the Dark Gods whom the Eye of the Gods
# can judge: a marked aspirant, Champion or Chosen. Daemon Princes and the
# Everchosen have passed beyond mortal judgement. Scope: a character.
darkgift_judged_trigger = {
	OR = {
		darkgift_patron_khorne_trigger = yes
		darkgift_patron_nurgle_trigger = yes
		darkgift_patron_slaanesh_trigger = yes
		darkgift_patron_tzeentch_trigger = yes
		darkgift_patron_undivided_trigger = yes
	}
	is_daemon_prince_trigger = no
	NOT = { trait = everchosen }
}

# The character bears a mark but holds no rank yet. Scope: a character.
darkgift_aspirant_trigger = { is_chaos_champion_trigger = no is_chaos_chosen_trigger = no }

# The character lives the way their patron loves. Scope: a character.
darkgift_pleases_patron_trigger = {
	OR = {
		AND = {
			darkgift_patron_khorne_trigger = yes
			OR = { trait = wroth trait = brave trait = berserker war = yes }
		}
		AND = {
			darkgift_patron_nurgle_trigger = yes
			OR = {
				trait = content
				trait = gregarious
				trait = slothful
				trait = gluttonous
				trait = has_nurgles_rot
			}
		}
		AND = {
			darkgift_patron_slaanesh_trigger = yes
			OR = {
				trait = lustful
				trait = hedonist
				trait = proud
				trait = seducer
				trait = seductress
			}
		}
		AND = {
			darkgift_patron_tzeentch_trigger = yes
			OR = {
				trait = ambitious
				trait = deceitful
				trait = schemer
				trait = mystic
				magic_potential_trigger = yes
			}
		}
		AND = {
			darkgift_patron_undivided_trigger = yes
			OR = { trait = zealous trait = ambitious trait = proud }
		}
	}
}

# The character lives the way their patron hates. Khorne despises sorcerers
# and cowards, Nurgle the ambitious schemers of Tzeentch, Slaanesh the chaste
# and the temperate, Tzeentch the content and the stagnant. Scope: a character.
darkgift_offends_patron_trigger = {
	OR = {
		AND = {
			darkgift_patron_khorne_trigger = yes
			OR = { trait = craven trait = content trait = patient magic_potential_trigger = yes }
		}
		AND = {
			darkgift_patron_nurgle_trigger = yes
			OR = { trait = ambitious trait = paranoid trait = deceitful }
		}
		AND = {
			darkgift_patron_slaanesh_trigger = yes
			OR = { trait = chaste trait = temperate trait = humble }
		}
		AND = {
			darkgift_patron_tzeentch_trigger = yes
			OR = { trait = content trait = honest trait = slothful }
		}
		AND = { darkgift_patron_undivided_trigger = yes OR = { trait = craven trait = cynical } }
	}
}
