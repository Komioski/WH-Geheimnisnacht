# A Dawi of the ancestor gods who could sit with the elders: no Slayer (a
# Slayer has given up his standing in the clan) and no Chaos Dwarf.
dawielder_can_sit_trigger = {
	trait = creature_dwarf
	religion_group = dwarf_gods_group
	is_adult = yes
	NOR = { trait = dwarf_slayer trait = dwarf_slayer_king }
}

# Old enough for the council, or the head of a clan. The lore makes a Dawi an
# Elder at 150 (DPG p.31); in the mod's longer lifespans that is a Longbeard,
# or a Dawi of 200. A ruler speaks for his clan at any age.
dawielder_old_enough_trigger = { OR = { is_ruler = yes trait = longbeard age = 200 } }

# An AI courtier or ruler the five-year pulse may seat on the council.
dawielder_ai_recruit_trigger = {
	ai = yes
	is_in_society = no
	dawielder_can_sit_trigger = yes
	dawielder_old_enough_trigger = yes
	prisoner = no
	is_incapable = no
	mercenary = no
	holy_order = no
}

# A member's court on the bi-yearly pulse (the dawielder_council_life list).
dawielder_pulse_trigger = { society_member_of = dawi_elder_council prisoner = no is_incapable = no }

# The lord the elder's judgements serve: the elder himself if he rules, or
# else his Dawi liege. Scope: the elder.
dawielder_has_lord_trigger = {
	OR = { is_ruler = yes liege = { religion_group = dwarf_gods_group } }
}

# A Dawi of the elder's court or realm who has no nickname yet and could be
# given one (dawielder.12). Dawi may not name themselves.
# Scope: the candidate. ROOT: the elder.
dawielder_name_candidate_trigger = {
	trait = creature_dwarf
	is_adult = yes
	has_nickname = no
	prisoner = no
	NOT = { character = ROOT }
}

# ROOT has someone to name: a courtier or vassal of his own, or of his liege.
# Scope: the elder.
dawielder_has_name_candidate_trigger = {
	OR = {
		any_courtier = { dawielder_name_candidate_trigger = yes }
		any_vassal = { dawielder_name_candidate_trigger = yes }
		liege = { any_courtier = { dawielder_name_candidate_trigger = yes } }
	}
}

# A Dawi wronged by this character: PREV holds an ordinary grudge against
# him, and both are Dawi, so the elders can end it as a feud between clans
# (dawielder.16). Neither may be the elder (ROOT): no Dawi judges his own
# feud. Blood Grudges are not the council's to end.
# Scope: the offender. PREV: the grudge holder.
dawielder_feud_offender_trigger = {
	trait = creature_dwarf
	religion_group = dwarf_gods_group
	NOT = { character = ROOT }
	OR = {
		reverse_has_opinion_modifier = { who = PREV modifier = opinion_dwarf_grudge }
		reverse_has_opinion_modifier = { who = PREV modifier = opinion_dawi_grudge_slight }
		reverse_has_opinion_modifier = { who = PREV modifier = opinion_dawi_grudge_theft }
		reverse_has_opinion_modifier = { who = PREV modifier = opinion_dawi_grudge_oath }
	}
}

# A Dawi ruler, other than the elder, holding an ordinary grudge against
# another Dawi ruler. Scope: the grudge holder.
dawielder_feud_holder_trigger = {
	dawi_grudge_keeper_trigger = yes
	NOT = { character = ROOT }
	prisoner = no
	any_playable_ruler = { dawielder_feud_offender_trigger = yes }
}

# A feud between Dawi clans exists for the elders to end. Scope: the elder.
dawielder_feud_exists_trigger = { any_playable_ruler = { dawielder_feud_holder_trigger = yes } }

# A ruler shamed by an ancestor who died with grudges open (dawi.34).
dawielder_far_from_ancestors_trigger = {
	OR = {
		has_character_modifier = dawi_far_from_ancestors_1
		has_character_modifier = dawi_far_from_ancestors_2
		has_character_modifier = dawi_far_from_ancestors_3
	}
}
