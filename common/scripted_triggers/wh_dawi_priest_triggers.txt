# A Dawi of the ancestor gods who could serve in a temple: no Slayer (a
# Slayer has forsaken clan and hold) and no Chaos Dwarf.
dawipriest_can_serve_trigger = {
	trait = creature_dwarf
	religion_group = dwarf_gods_group
	is_adult = yes
	NOR = { trait = dwarf_slayer trait = dwarf_slayer_king }
}

# An AI courtier or ruler the five-year pulse may bring into the priesthood.
dawipriest_ai_recruit_trigger = {
	ai = yes
	is_in_society = no
	dawipriest_can_serve_trigger = yes
	prisoner = no
	is_incapable = no
	OR = { trait = theologian trait = zealous learning = 12 is_priest = yes }
	NOT = { trait = cynical }
	NOT = { higher_tier_than = DUKE }
}

# This priest has taken a calling (dawipriest.12).
dawipriest_has_calling_trigger = {
	OR = {
		has_character_modifier = dawipriest_hearth_calling
		has_character_modifier = dawipriest_forge_calling
		has_character_modifier = dawipriest_doom_calling
	}
}

# This character has a Dawi patron god (decision pick_main_deity).
dawipriest_has_patron_trigger = {
	OR = {
		has_character_modifier = follower_of_grungni
		has_character_modifier = follower_of_grimnir
		has_character_modifier = follower_of_valaya
		has_character_modifier = follower_of_thungni
	}
}

# A member's court on the bi-yearly pulse (the dawipriest_temple_life list).
dawipriest_pulse_trigger = {
	society_member_of = ancestor_priesthood
	prisoner = no
	is_incapable = no
}
