# A Dawi of a clan who keeps a Book of Grudges: a dwarf of the ancestor gods
# with a dynasty (lowborn Dawi have no clan book to write in).
dawi_grudge_keeper_trigger = {
	trait = creature_dwarf
	religion_group = dwarf_gods_group
	is_lowborn = no
}

# Folk the Great Book of Grudges already names as a whole race: greenskins,
# Skaven and the Chaos Dwarfs. No clan needs a grudge of its own against them.
dawi_race_grudge_trigger = {
	OR = {
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
		trait = creature_skaven
		trait = creature_chaos_dwarf
	}
}

# Grudge kinds. Each kind of wrong is its own opinion modifier, held by the
# wronged Dawi against the offender (common/opinion_modifiers/
# wh_opinion_modifiers.txt). Ordinary grudges: opinion_dwarf_grudge (a war),
# opinion_dawi_grudge_slight, _theft, _oath. Blood Grudges:
# opinion_dawi_blood_grudge_kin, _murder, _oath, _shaving.

# ROOT's clan book names this character, for any wrong.
# Scope: the possible offender. ROOT: the Dawi.
dawi_grudge_against_trigger = {
	OR = {
		reverse_has_opinion_modifier = { who = ROOT modifier = opinion_dwarf_grudge }
		reverse_has_opinion_modifier = { who = ROOT modifier = opinion_dawi_grudge_slight }
		reverse_has_opinion_modifier = { who = ROOT modifier = opinion_dawi_grudge_theft }
		reverse_has_opinion_modifier = { who = ROOT modifier = opinion_dawi_grudge_oath }
		reverse_has_opinion_modifier = { who = ROOT modifier = opinion_dawi_blood_grudge_kin }
		reverse_has_opinion_modifier = { who = ROOT modifier = opinion_dawi_blood_grudge_murder }
		reverse_has_opinion_modifier = { who = ROOT modifier = opinion_dawi_blood_grudge_oath }
		reverse_has_opinion_modifier = { who = ROOT modifier = opinion_dawi_blood_grudge_shaving }
	}
}

# ROOT holds an ordinary grudge against this character.

# A province a Slayer can clear: greenskin warrens, Beastmen raids or a
# herdstone.
# Scope: province.
dawi_monster_province_trigger = {
	OR = {
		has_province_modifier = greenskin_population_bad
		has_province_modifier = beastmen_scourging_1
		has_province_modifier = beastmen_scourging_2
		has_province_modifier = beastmen_scourging_3
		has_province_modifier = beastman_herdstone
	}
}

# A ruler a Slayer would walk into a hall to kill: greenskins, Beastmen, Skaven,
# Fimir, Ogres, known vampires and the servants of Chaos.
# Scope: character.
dawi_slayer_foe_trigger = {
	OR = {
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
		trait = creature_beastman
		trait = creature_skaven
		trait = creature_fimir
		trait = creature_ogre
		is_vampire_visible_trigger = yes
		religion_group = chaos_gods_group
	}
}

# A province with a mine (ca_mine_1 to ca_mine_6).
# Scope: province.
dawi_mine_province_trigger = {
	OR = {
		has_building = ca_mine_1
		has_building = ca_mine_2
		has_building = ca_mine_3
		has_building = ca_mine_4
		has_building = ca_mine_5
		has_building = ca_mine_6
	}
}

# A province whose bridges a mason could build again (dawi.163).
# Scope: province.
dawi_bridge_province_trigger = { NOT = { has_province_modifier = SJ_more_bridges_built } }

# A mason or master mason.
# Scope: character.
dawi_mason_trigger = { OR = { trait = dwarf_mason trait = dwarf_master_mason } }

# A Dawi of the hold's warrior paths: warrior, Quarreller or Thunderer.
# Scope: character.
dawi_warrior_path_trigger = { OR = { trait = dwarf_warrior trait = quarreller trait = thunderer } }

# A manling or elgi lord who can serve a Dawi his ale (dawi.131).
# Scope: the host. ROOT: the Dawi.
dawi_beer_host_trigger = {
	is_adult = yes
	prisoner = no
	NOT = { trait = creature_dwarf }
	NOT = { culture_group = dwarf_group }
	NOT = { has_character_modifier = SJ_dwarfs_complain_about_beer_timer }
	war = no
}

# A courtier who could join the ruler's table at the tavern (dawi.185).
# Scope: the courtier. ROOT: the ruler.
dawi_tavern_friend_trigger = {
	is_adult = yes
	prisoner = no
	NOT = { is_friend = ROOT }
	NOT = { is_rival = ROOT }
	NOT = { num_of_friends = 2 }
	at_location = ROOT
}

# A neighbouring AI lord the manling debt collectors are hunting (dawi.191).
# Scope: the lord. ROOT: the Dawi ruler.
dawi_debtor_trigger = { ai = yes wealth = 100 NOT = { character = ROOT } }

# A courtier fit to be on his feet: free, not infirm, not incapable.
# Scope: the courtier.
dawi_able_courtier_trigger = { prisoner = no NOR = { trait = infirm trait = incapable } }

# A warrior of the court the ruler can spar with (dawi.221).
# Scope: the courtier.
dawi_sparring_partner_trigger = {
	ai = yes
	dawi_warrior_path_trigger = yes
	dawi_able_courtier_trigger = yes
	NOR = { trait = maimed trait = slothful }
}

# A Dawi commander who has had no battle deed for a year (dawi.211-218).
# Scope: character.
dawi_battle_deed_ready_trigger = {
	OR = {
		NOT = { has_character_flag = dawi_battle_deed }
		had_character_flag = { flag = dawi_battle_deed days = 365 }
	}
}

# The host of the Day of Grudgement: the High King of the Karaz Ankor, or the
# lord of Karaz-a-Karak while there is no High King.
dawi_grudgement_host_trigger = {
	dawi_grudge_keeper_trigger = yes
	OR = {
		has_landed_title = e_karak_ankor
		AND = { e_karak_ankor = { has_holder = no } has_landed_title = c_everpeak }
	}
}

# This Dawi has a Reckoner at court (title_dawi_reckoner), who carries his
# grudges to the offenders and argues the settlements.
dawi_has_reckoner_trigger = { any_courtier = { has_minor_title = title_dawi_reckoner } }

# A neighbour who may wrong the hold in the new grudge events (dawi.233,
# dawi.234): not of ROOT's own clan, not already in ROOT's book, not a
# Dwarf-friend or ally, and not of a race the Great Book already names.
# Scope: the neighbour. ROOT: the Dawi.
dawi_new_grudge_neighbour_trigger = {
	is_adult = yes
	prisoner = no
	NOT = { dynasty = ROOT }
	dawi_race_grudge_trigger = no
	dawi_grudge_against_trigger = no
	NOT = { is_allied_with = ROOT }
	NOT = { reverse_has_opinion_modifier = { who = ROOT modifier = opinion_dawi_friend } }
}
