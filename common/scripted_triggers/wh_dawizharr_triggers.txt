###############################################################################
# Dawi Zharr (Chaos Dwarf) triggers
# Used by events/WHG_chaos_dwarf_events.txt and the Chaos Dwarf decisions.
#
# The slave buildings (factory and slave levy, levels 1-4) only ever sit in a
# ruler's capital holding. Their prefix follows the holding type: ca_ castle,
# ct_ city, tp_ temple, fp_ family palace.
###############################################################################

# The capital holding has a slave factory (any level).
# Scope: a ruler.
dawizharr_factory_trigger = {
	capital_holding = {
		OR = {
			has_building = ca_chaos_dwarf_factory_1
			has_building = ct_chaos_dwarf_factory_1
			has_building = tp_chaos_dwarf_factory_1
			has_building = fp_chaos_dwarf_factory_1
		}
	}
}

# The capital holding has a slave levy barracks (any level).
# Scope: a ruler.
dawizharr_levy_trigger = {
	capital_holding = {
		OR = {
			has_building = ca_chaos_dwarf_slave_levy_1
			has_building = ct_chaos_dwarf_slave_levy_1
			has_building = tp_chaos_dwarf_slave_levy_1
			has_building = fp_chaos_dwarf_slave_levy_1
		}
	}
}

# The capital factory is level 2 or higher, so it can lose a level to decay.
# Scope: a ruler.
dawizharr_factory_2_trigger = {

		OR = {

			has_building = ct_chaos_dwarf_factory_2

}

dawizharr_levy_2_trigger = {

# The holding has any slave building.
# Scope: a barony title.
dawizharr_slave_holding_trigger = {
	OR = {
		has_building = ca_chaos_dwarf_factory_1
		has_building = ct_chaos_dwarf_factory_1
		has_building = tp_chaos_dwarf_factory_1
		has_building = fp_chaos_dwarf_factory_1
		has_building = ca_chaos_dwarf_slave_levy_1
		has_building = ct_chaos_dwarf_slave_levy_1
		has_building = tp_chaos_dwarf_slave_levy_1
		has_building = fp_chaos_dwarf_slave_levy_1
	}
}

# A courtier the slavers can drag off from a raided court: not the ruler, not
# away, not already a prisoner, not a Chaos Dwarf or a daemon, and not one who
# already slipped away this raid.
# Scope: a courtier of the raided ruler.
dawizharr_captive_trigger = {
	is_abroad = no
	is_ruler = no
	prisoner = no
	NOT = { has_character_flag = dawizharr_hid }
	NOT = { trait = creature_chaos_dwarf }
	NOT = { is_daemon_race_trigger = yes }
}

# Hashut asks for a sacrifice: a landed Chaos Dwarf (not a temple ruler, whose
# priests feed the furnaces) who has not fed the furnaces for four years.
# Scope: a ruler.
dawizharr_sacrifice_due_trigger = {
	higher_tier_than = BARON
	is_theocracy = no
	trait = creature_chaos_dwarf
	religion = hashut
	NOT = { has_character_modifier = must_sacrifice_hashut }
	had_character_flag = { flag = flag_sacrifice_made_hashut days = 1460 }
}

# The arms seller is a Dawi (Dwarf) hold rather than the Dawi Zharr.
# Scope: the seller.
dawizharr_dawi_seller_trigger = { trait = creature_dwarf }
