# Triggers for the debt events (Part 11 of events/WHG_realm_events.txt).

# A vassal who likes ROOT well enough and was not asked for gold in the
# last five years.
# Scope: the vassal.
debt_vassal_can_be_asked_trigger = {
	opinion = { who = ROOT value = 0 }
	OR = {
		NOT = { has_character_flag = took_extra_tax }
		had_character_flag = { flag = took_extra_tax days = 1825 }
	}
}

# A lesser title of the liege that ROOT has a de jure claim to and that can
# be given away; never the liege's capital county.
# Scope: a demesne title of ROOT's liege.
debt_title_for_sale_trigger = {
	has_de_jure_pretension = ROOT
	can_be_given_away = yes
	lower_tier_than = PREV
	NAND = {
		tier = COUNT
		location = { is_capital = yes }
	}
}
