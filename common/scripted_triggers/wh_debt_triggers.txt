# Triggers for the debt events (Part 11 of events/WHG_realm_events.txt).

# A vassal who likes ROOT well enough and was not asked for gold in the
# last five years.
# Scope: the vassal.
debt_vassal_can_be_asked_trigger = {
	opinion = { who = ROOT value = 0 }
	OR = {
		NOT = { has_character_flag = took_extra_tax }
		had_character_flag = { flag = took_extra_tax days = 1825 }
	}
}

# A lesser title of the liege that ROOT has a de jure claim to and that can
# be given away; never the liege's capital county.
# Scope: a demesne title of ROOT's liege.
debt_title_for_sale_trigger = {
	has_de_jure_pretension = ROOT
	can_be_given_away = yes
	lower_tier_than = PREV
	NAND = {
		tier = COUNT
		location = { is_capital = yes }
	}
}

# Who lends to this ruler: the Dawi lending-clans to the Dawi, the
# merchant houses of Araby to Arabyans, and the banking houses of
# Marienburg to everyone else who keeps coin. Greenskins, Skaven,
# beastmen, Chaos and the dead have no lenders.
# Scope: a ruler.
debt_lender_dawi_trigger = {
	trait = creature_dwarf
}
debt_lender_araby_trigger = {
	NOT = { trait = creature_dwarf }
	OR = {
		culture_group = arabyan_group
		religion_group = ormazd_gods_group
	}
}
debt_has_lender_trigger = {
	OR = {
		trait = creature_human
		trait = creature_dwarf
		trait = creature_halfling
		trait = creature_elf
	}
	NOR = {
		religion_group = chaos_gods_group
		religion_group = necromantic_group
		is_vampire_trigger = yes
	}
}

# Misrule in the red (debt.50 - debt.64).

# A province that one of the four ills of misrule can still befall.
# Scope: a province.
debt_misrule_room_trigger = {
	OR = {
		NOT = { has_province_modifier = incompetent_rule }
		NOT = { has_province_modifier = thieves_guild }
		NOT = { has_province_modifier = smugglers_ring }
		NOT = { has_province_modifier = highway_robber_band }
	}
}

# A province suffering one of the four ills of misrule.
# Scope: a province.
debt_misrule_any_trigger = {
	OR = {
		has_province_modifier = incompetent_rule
		has_province_modifier = thieves_guild
		has_province_modifier = smugglers_ring
		has_province_modifier = highway_robber_band
	}
}

# A wonder with guards (a retinue or temple guards) stands in the province
# or beside it.
# Scope: a province.
debt_wonder_guards_trigger = {
	OR = {
		AND = {
			has_wonder = yes
			wonder = {
				is_active = yes
				has_wonder_upgrade_flag = defensive
				has_wonder_upgrade_flag = unit
			}
		}
		any_neighbor_province = {
			has_wonder = yes
			wonder = {
				is_active = yes
				has_wonder_upgrade_flag = defensive
				has_wonder_upgrade_flag = unit
			}
		}
	}
}
