# Disease triggers, used by events/WHG_disease_events.txt.

###############################################################################
# Blame for the epidemic (blame)
###############################################################################

# THIS rules a settled people who would look for someone to blame when an
# epidemic empties their capital. Not greenskins, Skaven, ogres, the
# Lizardmen, the dead, or followers of Nurgle, who welcome the sickness.
blame_epidemic_ruler_trigger = {
	NOR = {
		religion_group = greenskin_gods_group
		religion_group = horned_rat_god_group
		religion_group = great_maw_group
		religion_group = old_ones_group
		religion_group = necromantic_group
		religion_group = nehekharan_gods_group
		culture_group = orc_group
		culture_group = goblin_group
		culture_group = monstrous_group
		culture_group = lizardman_group
		culture_group = nehekharan_group
		is_vampire_trigger = yes
		follower_of_nurgle = yes
	}
}

# THIS has Strigany in the realm to blame: a caravan in a realm province or
# a Strigany courtier, and has not expelled them already. Vampires and
# necromancers protect their Strigany servants.
blame_strigany_present_trigger = {
	NOT = { has_character_modifier = expelled_strigany }
	NOT = { culture_group = strigany_group }
	NOT = { religion_group = necromantic_group }
	is_vampire_trigger = no
	OR = {
		any_realm_province = { has_province_modifier = strigany_caravan }
		any_courtier = { culture_group = strigany_group }
	}
}

# THIS has someone to lead a witch hunt (a court witch hunter, a witch
# hunter of a chapter at court, or a court chaplain) and enough courtiers
# to accuse.
blame_witch_hunt_possible_trigger = {
	OR = {
		any_courtier_or_vassal = { whunter_court_hunter_title_trigger = yes prisoner = no }
		any_courtier = { whunter_chapter_member_trigger = yes prisoner = no is_adult = yes }
		job_spiritual = { always = yes }
	}
	any_courtier = { count = 5 }
}

# THIS follows a temple faith and has no temple in the capital, so the
# priests can say the gods sent the sickness because the ruler is never seen
# at worship. Old World, Araby, Cathay, Dawi, halfling and elven faiths.
blame_impious_ruler_possible_trigger = {
	OR = {
		religion_group = old_world_group
		religion_group = ormazd_gods_group
		religion_group = oriental_gods_group
		religion_group = dwarf_gods_group
		religion_group = halfling_gods_group
		religion_group = cadai_group
		religion_group = cytharai_group
	}
	capital_scope = { NOT = { any_province_holding = { holding_type = temple } } }
}

# THIS (a courtier of the ruler) can be accused of witchcraft by ROOT, the
# accuser. Not the witch hunter, not already accused or in prison, and old
# enough to be believed a witch.
blame_witch_suspect_base_trigger = {
	prisoner = no
	age = 10
	NOT = { character = ROOT }
	NOT = { character = event_target:bg_witch_hunter }
	NOT = { has_character_flag = bg_accused_witch }
}

# THIS (a courtier) is someone ROOT, the ruler, would happily name as a
# witch: a rival or someone the ruler dislikes.
blame_disliked_courtier_trigger = {
	prisoner = no
	age = 10
	OR = { is_rival = ROOT NOT = { reverse_opinion = { who = ROOT value = 0 } } }
}

# THIS (the ruler) has a spouse, a child of four or older, their mother or a
# sibling at court, whom a mob could take for a witch.
blame_relative_at_court_trigger = {
	OR = {
		any_spouse = { at_location = ROOT }
		any_child = { at_location = ROOT age = 4 }
		mother = { at_location = ROOT }
		any_sibling = { at_location = ROOT }
	}
}

###############################################################################
# The Great Plague (plague)
###############################################################################

# THIS rejoices at a plague: a follower of Nurgle or a Skaven.
plague_news_dark_trigger = {
	OR = { follower_of_nurgle = yes religion_group = horned_rat_god_group }
}

# THIS (a province next to a newly infected one) is held by a player, or a
# player's vassal, who has not been warned of the plague yet.
plague_warning_owner_trigger = {
	owner = {
		OR = {
			AND = { ai = no NOT = { has_character_flag = great_plague_close } }
			any_liege = { ai = no NOT = { has_character_flag = great_plague_close } }
		}
	}
}
