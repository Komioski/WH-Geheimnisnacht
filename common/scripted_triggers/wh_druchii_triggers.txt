# A lesser courtier that a schemer might poison to rise at court: unlanded,
# not on the council, not clever or strong enough to be dangerous, and no
# friend, lover or close kin of the schemer.
# Scope: a courtier of the schemer's liege, ROOT the schemer (druchii.52).
druchii_herd_victim_trigger = {
	NOT = { character = ROOT }
	is_landed = no
	is_councillor = no
	prisoner = no
	age = 50
	NOR = {
		intrigue = 12
		martial = 12
		is_friend = ROOT
		is_lover = ROOT
		is_close_relative = ROOT
		is_invincible_trigger = yes
	}
}

# A courtier who could become the rival of an ambitious courtier.
# Scope: a courtier of the courtier's liege, ROOT the courtier (druchii.54).
druchii_rival_candidate_trigger = {
	NOT = { character = ROOT }
	is_adult = yes
	culture_group = druchii_group
	NOT = { num_of_rivals = 3 }
	NOR = {
		is_rival = ROOT
		is_friend = ROOT
		is_lover = ROOT
	}
}

# A Sorceress free to be hired by another court.
# Scope: the Sorceress (hire_sorceress, druchii.85, 86).
druchii_sorceress_for_hire_trigger = {
	trait = druchii_sorceress
	is_landed = no
	is_married = no
	is_incapacitated_trigger = no
	NOT = { has_character_modifier = magic_power_4 }
	NOT = { has_character_flag = hired_sorceress }
}
