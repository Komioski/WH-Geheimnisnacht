# Eldership triggers, used by events/WHG_eldership_events.txt.

# An elder able to meddle: not incapable, not in a cell. Scope: a character.
elder_able_trigger = { NOR = { trait = incapable prisoner = yes } }

# ROOT is a pleased or ecstatic Elder of this title. Scope: an Eldership
# title.
elder_root_pleased_trigger = {
	ROOT = {
		OR = {
			has_elector_stance = { stance = elder_pleased title = PREV }
			has_elector_stance = { stance = elder_ecstatic title = PREV }
		}
	}
}

# ROOT is a displeased Elder of this title. Scope: an Eldership title.
elder_root_displeased_trigger = {
	ROOT = { has_elector_stance = { stance = elder_displeased title = PREV } }
}

# ROOT is an ecstatic Elder of this title. Scope: an Eldership title.
elder_root_ecstatic_trigger = {
	ROOT = { has_elector_stance = { stance = elder_ecstatic title = PREV } }
}

# The ruler has a county to spare: not the capital, and free to give away.
# Scope: a ruler.
elder_county_to_spare_trigger = {
	any_demesne_title = { can_be_given_away = yes tier = COUNT location = { is_capital = no } }
}

# An unlanded adult of the ruler's house, not the heir, whom the ruler's
# gender law allows to hold land. Scope: a character; PREV is the ruler.
elder_kin_candidate_trigger = {
	is_alive = yes
	is_adult = yes
	is_ruler = no
	prisoner = no
	NOT = { is_primary_heir = PREV }
	OR = {
		AND = {
			is_female = no
			PREV = { primary_title = { NOT = { has_law = enatic_succession } } }
		}
		AND = {
			is_female = yes
			PREV = { primary_title = { NOT = { has_law = agnatic_succession } } }
		}
	}
}

# The ruler has a county to spare and a kinsman to give it to, and has not
# been pestered lately. Scope: a ruler.
elder_county_request_trigger = {
	NOT = { has_character_modifier = elder_demand_cooldown }
	elder_county_to_spare_trigger = yes
	any_dynasty_member = { elder_kin_candidate_trigger = yes }
}

# A realm character a displeased Elder may talk into a claim on the ruler's
# title. Scope: a character; PREV is the ruler, PREVPREV the title.
elder_claimant_trigger = {
	is_adult = yes
	is_incapable = no
	is_ill = no
	prisoner = no
	NOR = { has_claim = PREVPREV is_heir = PREV character = ROOT is_rival = ROOT character = PREV }
}

# A kinsman a displeased Elder would gladly see fail: a fool, a weakling, a
# villain or someone the world laughs at. Scope: a character.
elder_unfit_kin_trigger = {
	OR = {
		is_dumb_trigger = yes
		is_weak_trigger = yes
		is_evil_trigger = yes
		trait = inbred
		trait = maimed
		trait = mangled
		trait = hunchback
		trait = ugly
		trait = clubfooted
		trait = harelip
		trait = dwarf
		trait = giant
		trait = stutter
		trait = lisp
	}
}
