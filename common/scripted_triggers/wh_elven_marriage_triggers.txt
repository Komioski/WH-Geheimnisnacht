# The rank of a character's house: their own tier, or a parent's.
# Checked from the top down (emperor first), as the old code did.
# Scope: the character.
elfwed_house_emperor_trigger = {
	OR = {
		tier = EMPEROR
		father_even_if_dead = { tier = EMPEROR }
		mother_even_if_dead = { tier = EMPEROR }
	}
}
elfwed_house_king_trigger = {
	OR = { tier = KING father_even_if_dead = { tier = KING } mother_even_if_dead = { tier = KING } }
}
elfwed_house_duke_trigger = {
	OR = { tier = DUKE father_even_if_dead = { tier = DUKE } mother_even_if_dead = { tier = DUKE } }
}
elfwed_house_count_trigger = {
	OR = {
		tier = COUNT
		father_even_if_dead = { tier = COUNT }
		mother_even_if_dead = { tier = COUNT }
	}
}
elfwed_house_baron_trigger = {
	OR = {
		tier = BARON
		father_even_if_dead = { tier = BARON }
		mother_even_if_dead = { tier = BARON }
	}
}

# A house of at least royal rank.
# Scope: the character.
elfwed_house_royal_trigger = {
	OR = {
		higher_tier_than = DUKE
		father_even_if_dead = { higher_tier_than = DUKE }
		mother_even_if_dead = { higher_tier_than = DUKE }
	}
}

# The suitor can afford a show of wealth worthy of the beloved's house.
# Scope: the suitor, FROM the beloved.
elfwed_can_afford_gold_trigger = {
	OR = {
		AND = { FROM = { elfwed_house_emperor_trigger = yes } wealth = 750 }
		AND = {
			FROM = { elfwed_house_emperor_trigger = no elfwed_house_king_trigger = yes }
			wealth = 450
		}
		AND = {
			FROM = { elfwed_house_royal_trigger = no elfwed_house_duke_trigger = yes }
			wealth = 300
		}
		AND = {
			FROM = {
				elfwed_house_royal_trigger = no
				elfwed_house_duke_trigger = no
				elfwed_house_count_trigger = yes
			}
			wealth = 150
		}
		AND = {
			FROM = {
				elfwed_house_royal_trigger = no
				elfwed_house_duke_trigger = no
				elfwed_house_count_trigger = no
				elfwed_house_baron_trigger = yes
			}
			wealth = 100
		}
		AND = {
			FROM = {
				elfwed_house_royal_trigger = no
				elfwed_house_duke_trigger = no
				elfwed_house_count_trigger = no
				elfwed_house_baron_trigger = no
			}
			wealth = 50
		}
	}
}

# The two share a pastime or a craft.
# Scope: one of them, FROM the other.
elfwed_shared_interest_trigger = {
	OR = {
		AND = { trait = poet FROM = { trait = poet } }
		AND = { trait = asur_mage FROM = { trait = asur_mage } }
		AND = { trait = hunter FROM = { OR = { trait = hunter trait = falconer } } }
		AND = { trait = falconer FROM = { OR = { trait = hunter trait = falconer } } }
		AND = { trait = gardener FROM = { trait = gardener } }
		AND = { trait = scholar FROM = { trait = scholar } }
		AND = { trait = theologian FROM = { trait = theologian } }
		AND = { trait = mystic FROM = { trait = mystic } }
		AND = { trait = socializer FROM = { trait = socializer } }
		AND = { trait = hedonist FROM = { trait = hedonist } }
		AND = { trait = gamer FROM = { trait = gamer } }
		AND = { trait = duelist FROM = { trait = duelist } }
		AND = { trait = strategist FROM = { trait = strategist } }
		AND = { trait = administrator FROM = { trait = administrator } }
		AND = { trait = architect FROM = { trait = architect } }
		AND = { trait = schemer FROM = { trait = schemer } }

	}
}
