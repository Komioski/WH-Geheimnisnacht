# One of the great cults of the Empire. Scope: the character.
empire_imperial_cult_trigger = {
	OR = {
		religion = sigmarite
		religion = ulrican
		religion = taalite
		religion = morrian
		religion = manannite
	}
}

# Holds a rival Imperial crown of the claimants' wars. Scope: the character.
empire_holds_contender_crown_trigger = {
	OR = {
		has_landed_title = e_reikland
		has_landed_title = e_middenland
		has_landed_title = e_talabecland
		has_landed_title = e_westerland
	}
}

# An independent Elector of the Empire asked to accept a new Emperor
# (empire.1). Scope: the Elector. ROOT: the claimant.
empire_elector_to_ask_trigger = {
	higher_tier_than = DUKE
	culture_group = imperial_group
	religion_group = old_world_group
	NOT = { character = ROOT }
}

# The Empire's cults face a common foe: the vampires have won a great war,
# or a Black Crusade marches on the Empire. Scope: any.
empire_common_foe_trigger = {
	k_cult_of_sigmar = {
		OR = { has_title_flag = black_crusade_invasion has_title_flag = vampire_invasion }
	}
}

# A worthy bearer of a Runefang: a living man or woman of the Empire who
# keeps the old faiths and is not one of the walking dead. Scope: the
# character.
empire_runefang_bearer_trigger = {
	is_alive = yes
	culture_group = imperial_group
	religion_group = old_world_group
	is_undead_trigger = no
}

# One of the twelve Elector crowns that carry a Runefang. Scope: the title.
empire_runefang_crown_trigger = {
	tier = KING
	OR = {
		title = k_averland
		title = k_hochland
		title = k_middenland
		title = k_nordland
		title = k_ostland
		title = k_ostermark
		title = k_reikland
		title = k_stirland
		title = k_talabecland
		title = k_wissenland
		title = k_solland
		title = k_drakwald
	}
}
