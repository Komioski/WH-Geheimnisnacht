###############################################################################
# Engineers Guild triggers
# Used by events/WHG_dwarf_guild_events.txt and decisions/WHG_engineer_decisions.txt.
###############################################################################

# An AI courtier or ruler who might join the guild (engineer.5).
# Scope: the candidate.
engineer_ai_recruit_trigger = {
	ai = yes
	prisoner = no
	is_adult = yes
	is_in_society = no
	NOT = { trait = incapable }
	controls_religion = no
	block_general_event_trigger = no
	can_join_society = engineers_guild
}

# A manling court that would take an engineer (hire_imperial_engineer):
# the Empire, Tilea or Estalia. No servant of the dead.
# Scope: the ruler.
engineer_manling_court_trigger = {
	OR = {
		culture_group = imperial_group
		culture_group = tilean_group
		culture_group = estalian_group
	}
	NOT = { religion_group = necromantic_group }
}

# An engineer of any kind. Scope: the character.
engineer_is_engineer_trigger = {
	OR = { trait = engineer trait = dwarf_engineer trait = dwarf_engineer_master }
}

# The ruler is an engineer, or keeps one at court (commission_war_machine).
# Scope: the ruler.
engineer_has_court_engineer_trigger = {
	OR = {
		engineer_is_engineer_trigger = yes
		any_courtier = { engineer_is_engineer_trigger = yes prisoner = no }
	}
}

# A war machine is already being built in the ruler's lands.
# Scope: the ruler.
engineer_machine_in_work_trigger = {
	any_demesne_province = {
		OR = {
			has_province_modifier = engineer_working_artillery
			has_province_modifier = engineer_working_steam_tank
		}
	}
}

# A Steam Tank can be ordered: an Imperial court, with the Imperial school
# open. Scope: the ruler.
engineer_steam_tank_allowed_trigger = {
	culture_group = imperial_group
	has_global_flag = imperial_engineering_school
}

###############################################################################
# The three branches of the guild
# Every member belongs to one, shown by a character modifier set when he
# joins (engineer.7): the Dawi of the holds (engineer_branch_hold), the Dawi
# of the manling cities (engineer_branch_manling_dawi), the Imperial
# Engineers School of Altdorf (engineer_branch_school, with the Tileans) and
# the Imperial Gunnery School of Nuln (engineer_branch_gunnery).
###############################################################################

# The Imperial Gunnery School stands in Nuln (the wonder upgrade on Nuln's
# great works, province 309) and Nuln is in Imperial hands.
engineer_gunnery_school_open_trigger = {
	309 = {
		owner = {
			culture_group = imperial_group
			any_realm_wonder = {
				location = { province = 309 }
				has_wonder_upgrade = upgrade_imperial_gunnery_school
			}
		}
	}
}

# The Imperial Engineers School in Altdorf has been open for a year.
engineer_school_open_trigger = {
	had_global_flag = { flag = imperial_engineering_school months >= 12 }
}

# A Dawi who lives among manlings: of an Imperial Dwarf clan, or at a manling
# court, or under a manling liege. Scope: the Dawi.
engineer_dawi_among_manlings_trigger = {
	OR = {
		culture = dwarf_imperial
		AND = { is_ruler = no liege = { trait = creature_human } }
		AND = { is_ruler = yes top_liege = { trait = creature_human } }
	}
}

# The member already belongs to a branch. Scope: the member.
engineer_has_branch_trigger = {
	OR = {
		has_character_modifier = engineer_branch_hold
		has_character_modifier = engineer_branch_manling_dawi
		has_character_modifier = engineer_branch_school
		has_character_modifier = engineer_branch_gunnery
	}
}

# A member of the guild, of his branch, on the bi-yearly society pulse.
# Scope: the member.
engineer_branch_pulse_trigger = {
	society_member_of = engineers_guild
	prisoner = no
	is_incapable = no
}

# A manling courtier the Dawi of the cities could teach (engineer.63).
# Scope: the courtier. ROOT: the Dawi engineer.
engineer_manling_pupil_trigger = {
	trait = creature_human
	is_adult = yes
	prisoner = no
	learning = 8
	NOT = { engineer_is_engineer_trigger = yes }
	NOT = { character = ROOT }
	NOT = { religion_group = chaos_gods_group }
}

# The Dawi engineer has a manling pupil at his own court or his liege's.
# Scope: the Dawi engineer.
engineer_has_manling_pupil_trigger = {
	OR = {
		any_courtier = { engineer_manling_pupil_trigger = yes }
		liege = { any_courtier = { engineer_manling_pupil_trigger = yes } }
	}
}

# The ruler is a gunner of Nuln, or keeps one at court (commission_war_machine,
# the Nuln battery). Scope: the ruler.
engineer_has_gunnery_engineer_trigger = {
	OR = {
		has_character_modifier = engineer_branch_gunnery
		any_courtier = { has_character_modifier = engineer_branch_gunnery prisoner = no }
	}
}
