###############################################################################

# An AI courtier or ruler who might join the guild (engineer.5).
# Scope: the candidate.
engineer_ai_recruit_trigger = {
	ai = yes
	prisoner = no
	is_adult = yes
	is_in_society = no
	NOT = { trait = incapable }
	controls_religion = no
	block_general_event_trigger = no
	can_join_society = engineers_guild
}

# A manling court that would take an engineer (hire_imperial_engineer):
# the Empire, Tilea, Estalia, Kislev, or a Bretonnian court outside the Grail
# faiths. No servant of the dead.
# Scope: the ruler.
engineer_manling_court_trigger = {
	OR = {
		culture_group = imperial_group
		culture_group = tilean_group
		culture_group = estalian_group
		culture_group = kislevite_group
		culture_group = bretonnian_group
	}
	NOT = { religion_group = necromantic_group }
	NOR = { religion = grail religion = false_grail }
}

# An engineer of any kind. Scope: the character.
engineer_is_engineer_trigger = {
	OR = { trait = engineer trait = dwarf_engineer trait = dwarf_engineer_master }
}

# The ruler is an engineer, or keeps one at court (commission_war_machine).
# Scope: the ruler.
engineer_has_court_engineer_trigger = {
	OR = {
		engineer_is_engineer_trigger = yes
		any_courtier = { engineer_is_engineer_trigger = yes prisoner = no }
	}
}

# A war machine is already being built in the ruler's lands.
# Scope: the ruler.
engineer_machine_in_work_trigger = {
	any_demesne_province = {
		OR = {
			has_province_modifier = engineer_working_artillery
			has_province_modifier = engineer_working_steam_tank
		}
	}
}

# A Steam Tank can be ordered: an Imperial court, with the Imperial school
# open. Scope: the ruler.
engineer_steam_tank_allowed_trigger = {
	culture_group = imperial_group
	has_global_flag = imperial_engineering_school
}
