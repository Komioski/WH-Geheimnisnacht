# Holds Erengrad and the Convoy's charter: may appoint a Convoy Master and
# send the Convoy out.
# Scope: a ruler.
erengrad_convoy_holder_trigger = {
	has_landed_title = c_erengrad
	has_global_flag = erengrad_convoy_chartered
}

# The Convoy Master of this ruler's court, if any, is free to sail.
# Scope: a ruler.
erengrad_has_convoy_master_trigger = {
	any_courtier = {
		has_minor_title = title_convoy_master
		prisoner = no
		NOT = { trait = incapable }
	}
}

# The Convoy Master is docked at the port their liege sent them to. Each
# destination takes any port of its land, so the run can't be lost to a
# port that is gone.
# Scope: the Convoy Master.
erengrad_convoy_arrived_trigger = {
	has_minor_title = title_convoy_master
	liege = { has_character_flag = erengrad_on_run }
	location = { port = yes }
	OR = {
		AND = {
			liege = { has_character_flag = erengrad_dest_marienburg }
			location = {
				OR = {
					province_id = 222 # Marienburg
					region = world_old_world_empire_wasteland
				}
			}
		}
		AND = {
			liege = { has_character_flag = erengrad_dest_salzenmund }
			location = {
				OR = {
					province_id = 239 # Salzenmund
					region = world_old_world_empire_north
				}
			}
		}
		AND = {
			liege = { has_character_flag = erengrad_dest_languille }
			location = {
				OR = {
					province_id = 799 # L'Anguille
					region = world_old_world_bretonnia
				}
			}
		}
		AND = {
			liege = { has_character_flag = erengrad_dest_norsca }
			location = { region = world_norsca }
		}
		AND = {
			liege = { has_character_flag = erengrad_dest_sudenburg }
			location = {
				OR = {
					province_id = 983 # Aruk
					province_id = 1045 # Samur
					culture = sudenburger
				}
			}
		}
		AND = {
			liege = { has_character_flag = erengrad_dest_ulthuan }
			location = { region = world_ulthuan }
		}
	}
}

# The cargo suits the port: each market pays best for two of the five.
# Scope: the ruler who sent the Convoy.
erengrad_cargo_suits_port_trigger = {
	OR = {
		AND = {
			has_character_flag = erengrad_dest_marienburg
			OR = { has_character_flag = erengrad_cargo_amber has_character_flag = erengrad_cargo_furs }
		}
		AND = {
			has_character_flag = erengrad_dest_salzenmund
			OR = { has_character_flag = erengrad_cargo_timber has_character_flag = erengrad_cargo_silver }
		}
		AND = {
			has_character_flag = erengrad_dest_languille
			OR = { has_character_flag = erengrad_cargo_furs has_character_flag = erengrad_cargo_silver }
		}
		AND = {
			has_character_flag = erengrad_dest_norsca
			OR = { has_character_flag = erengrad_cargo_grain has_character_flag = erengrad_cargo_silver }
		}
		AND = {
			has_character_flag = erengrad_dest_sudenburg
			OR = { has_character_flag = erengrad_cargo_timber has_character_flag = erengrad_cargo_grain }
		}
		AND = {
			has_character_flag = erengrad_dest_ulthuan
			OR = { has_character_flag = erengrad_cargo_furs has_character_flag = erengrad_cargo_amber }
		}
	}
}

# Sails the Sea of Claws, between Erengrad, Norsca and the Empire coast: any
# zone of the Sea of Claws theatre (sea_zone_claws_trigger in
# WHG_sea_triggers.txt). The old list also took the Claws of the Far East,
# a Skaven Run and the Sea of Chill off Naggaroth, which only share a name.
# Scope: a character at sea.
naval_sea_of_claws_trigger = {
	location = { sea_zone_claws_trigger = yes }
}

# Pirates, corsairs and Norscan raiders: they attack any foreign fleet at sea
# (naval.100) and take plunder when they win (nb_scripted_effects.txt).
naval_sea_raider_trigger = {
	OR = {
		government = pirate_government
		government = freeboota_government
		government = malekith_government
		government = druchii_government
		culture_group = norscan_group
		trait = viking
		trait = pirate
		trait = ravager
		trait = seaking
		trait = sea_queen
	}
}
