# Triggers for the restored and new event chains (daemonic and beastmen incursions, the Dragon Emperor)

# Araby lies furthest from the polar gates
daemonic_incursion_araby_trigger = {
	OR = {
		culture_group = arabyan_group
		capital_scope = { empire = { title = e_araby } }
	}
}

# Daemons care less for the lands of the undead and the greenskins
daemonic_incursion_unloved_prey_trigger = {
	OR = {
		is_undead_trigger = yes
		religion_group = necromantic_group
		religion_group = nehekharan_gods_group
		religion_group = greenskin_gods_group
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
		trait = creature_snotling
	}
}

# One of the Chaos realms of the warp is ascendant
warp_any_realm_ascendant_trigger = {
	OR = {
		khorne_warp = { has_status = warp_ascendant }
		nurgle_warp = { has_status = warp_ascendant }
		slaanesh_warp = { has_status = warp_ascendant }
		tzeentch_warp = { has_status = warp_ascendant }
		undivided_warp = { has_status = warp_ascendant }
	}
}

# The Ruinous Powers stand united
warp_powers_united_trigger = {
	undivided_warp = { has_status = warp_united }
}


# The immortal Dragon Emperor (creature_celdragon) sits on the Dragon Throne.
# Cathay's vanilla systems assume emperors die and get replaced; this lets
# them skip or replace those steps while he reigns.
dragon_emperor_reigns_trigger = {
	offmap_cathay = {
		offmap_ruler = { trait = creature_celdragon }
	}
}

# Cathay is fighting a war at home (any invasion status or civil war)
cathay_beset_trigger = {
	offmap_cathay = {
		OR = {
			has_status = cathay_civil_war
			has_status = cathay_mongol_invasion
			has_status = cathay_jurchen_invasion
			has_status = cathay_chaos_invasion
			has_status = cathay_ogre_war
			has_status = cathay_skaven_war
		}
	}
}

# A Cathayan colony in the west: a Draconite realm linked to the Dragon Throne,
# or a Cathayan settlement in the lands the colonists hold (Naparaya, the
# Dragon Isles, the Plains of Bones, Gnoblar Country, the far southlands).
# Character scope.
cathay_colony_ruler_trigger = {
	is_ruler = yes
	independent = yes
	is_landed = yes
	religion = draconite
	NOT = { has_landed_title = e_cathay_west_governor }
	OR = {
		has_dynasty_flag = link_with_cathay
		has_character_modifier = cathayan_link
		AND = {
			culture = cathayan
			capital_scope = {
				OR = {
					region = world_nehekhara_eastern
					region = world_eastern_southlands
					region = world_darklands_plains_of_bones
					region = world_dragon_isles
					region = world_darklands_gnoblar_country
				}
			}
		}
	}
}

# A colony is defending a war it did not start
cathay_colony_in_danger_trigger = {
	any_independent_ruler = {
		cathay_colony_ruler_trigger = yes
		war = yes
		is_primary_war_defender = yes
	}
}

# Someone the Dragon Throne wants brought to heel: rebel generals and
# displaced princes who fled west, and those who sacked Cathay.
# Character scope.
cathay_exile_trigger = {
	is_alive = yes
	OR = {
		has_character_flag = is_rebel_general
		has_character_flag = is_displaced_prince
		has_character_flag = cathay_sworn_vengeance
	}
	NOT = { has_character_flag = cathay_exile_pardoned }
	NOT = { has_landed_title = e_cathay_west_governor }
}

cathay_exile_at_large_trigger = {
	any_character = {
		cathay_exile_trigger = yes
	}
}

# Chaos-worshipping humans and daemon-touched who could claim the Dragon
# Throne (not Hashut's Chaos Dwarfs or the Norscan gods)
cathay_chaos_claimant_faith_trigger = {
	religion_group = chaos_gods_group
	NOR = {
		religion = hashut
		religion = norscan_gods
		religion = norscan_gods_reformed
	}
}

# Can reach Cathay: holds land on the eastern edge of the map or in the
# eastern Chaos Wastes, or borders the Western Protectorate. Character scope.
cathay_invasion_staging_ground_trigger = {
	OR = {
		any_realm_province = { region = custom_eastern_edge_of_map }
		any_realm_province = { region = world_chaos_wastes_eastern_steppes }
		any_neighbor_independent_ruler = { has_landed_title = e_cathay_west_governor }
	}
}

# Not already at war or in truce with Cathay, and nobody else is invading it.
# Character scope (ROOT = the would-be invader).
cathay_invasion_governor_free_trigger = {
	offmap_cathay = {
		governor = {
			show_scope_change = no
			NOT = { war_with = ROOT }
			NOT = { has_truce = ROOT }
			custom_tooltip = {
				text = cathay_invade_dragon_throne_war_tt
				NOT = { any_war = { using_cb = invasion_of_cathay_cb } }
			}
		}
	}
}

# A dynasty member at court who could lead a Chaos claim on the Dragon Throne.
# Courtier scope, ROOT = the Chaos ruler.
cathay_chaos_claimant_candidate_trigger = {
	dynasty = ROOT
	is_adult = yes
	is_ruler = no
	is_primary_heir = no
	prisoner = no
	is_ill = no
	NOR = {
		trait = incapable
		trait = inbred
		trait = imbecile
		is_ascetic_trigger = yes
	}
}
