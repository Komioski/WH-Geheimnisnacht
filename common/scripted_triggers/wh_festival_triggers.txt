# Great Tribal Festival scripted triggers

# An old faith that can still be reformed and has not been: the Northern
# gods, the Old Faith of Albion, the Norscan gods and Atruhayid. Their
# festival brings authority toward reform instead of piety.
# Scope: a character.
festival_unreformed_faith_trigger = {
	OR = {
		religion = northern_gods
		religion = old_faith
		religion = norscan_gods
		religion = atruhayid
	}
}

# The rite at the end of the festival. Exactly one of these is true for any
# character; the offering event (festival.160, 161) picks its text by them.
# Scope: the host.

# The shamans burn the offerings to the old gods.
festival_rite_old_gods_trigger = { festival_unreformed_faith_trigger = yes }

# A blood rite for the Dark Gods.
festival_rite_chaos_trigger = {
	religion_group = chaos_gods_group
	festival_unreformed_faith_trigger = no
}

# A Waaagh! bonfire for Gork and Mork.
festival_rite_waaagh_trigger = { religion_group = greenskin_gods_group }

# The offerings are tipped into the Great Maw.
festival_rite_maw_trigger = { religion_group = great_maw_group }

# The skink priests read the stars for the Old Ones.
festival_rite_stars_trigger = { religion_group = old_ones_group }

# A temple procession: the gods of the Old World, Stromfels, the halfling
# gods, Ormazd, the Dragon faith and the Cytharai.
festival_rite_temple_trigger = {
	festival_unreformed_faith_trigger = no
	OR = {
		religion_group = old_world_group
		religion_group = stromfels_group
		religion_group = halfling_gods_group
		religion_group = ormazd_gods_group
		religion_group = oriental_gods_group
		religion_group = cytharai_group
	}
}

# Everyone else gives the offerings to the ancestors.
festival_rite_ancestor_trigger = {
	NOR = {
		festival_rite_old_gods_trigger = yes
		festival_rite_chaos_trigger = yes
		festival_rite_waaagh_trigger = yes
		festival_rite_maw_trigger = yes
		festival_rite_stars_trigger = yes
		festival_rite_temple_trigger = yes
	}
}

# May keep a tamed beast as a mount (festival.139): no mount yet, never won
# one at a festival, and of a people with a mount artifact to win
# (beastmen tame razorgors, but there is no razorgor mount).
# Scope: the beast-taming champion.
festival_can_win_mount_trigger = {
	is_adult = yes
	NOT = { has_character_flag = festival_tamed_mount }
	NOT = { any_artifact = { has_artifact_flag = mount } }
	NOT = { culture_group = beastman_group }
}
