# Has a living mortal body that can catch a disease, fester, grow infirm or
# lose its wits to age: not a daemon, a vampire or other undead, a forest
# spirit or treekin, or Chaos spawn.
# Scope: a character.
health_mortal_body_trigger = {
	is_daemon_race_trigger = no
	is_undead_trigger = no
	is_tree_trigger = no
	NOT = { trait = creature_chaosspawn }
}

# Possessed, and hears no voice yet: neither the voice from on high, nor the
# daemon's voice, nor the twin-tailed omen.
# Scope: a character.
health_no_voice_trigger = {
	NOR = {
		has_character_modifier = voice_of_tzeentch
		has_character_modifier = voice_of_greater_daemon
		has_character_modifier = holy_comet
	}
}

# Possessed and hearing the voice from on high (voice_of_tzeentch): a voice
# that sounds like the character's god and is Tzeentch.
# Scope: a character.
health_voice_from_on_high_trigger = {
	trait = possessed
	has_character_modifier = voice_of_tzeentch
}

# Possessed and has seen the twin-tailed omen (holy_comet).
# Scope: a character.
health_twin_tailed_omen_trigger = {
	trait = possessed
	has_character_modifier = holy_comet
}

# Possessed and hearing a daemon (voice_of_greater_daemon).
# Scope: a character.
health_daemon_voice_trigger = {
	trait = possessed
	has_character_modifier = voice_of_greater_daemon
}

# Follows a faith that has a heresy, so become_heretic does something: the
# Grail, Ormazd, Akghan, Cytharai and Cadai faiths.
# Scope: a character.
health_has_heresy_trigger = {
	OR = {
		religion = grail
		religion = ormazd
		religion = akghan
		religion = cytharai
		religion = cadai
	}
}

# Can start the drowned dreams (health.170-173): follows a northern faith of
# the Old World near the Sea of Claws, is not in Lustria (health.210-213
# covers colonists), has not had any dream chain before and is not off at sea
# or in hiding.
# Scope: a character.
health_drowned_dreams_trigger = {
	NOT = { location = { region = world_new_world_lustria } }
	OR = {
		religion = kislev_gods
		religion = northern_gods
		religion = northern_gods_reformed
		religion = ulrican
		religion = taalite
		religion = old_faith
		religion = old_faith_reformed
	}
	NOT = { has_character_flag = health_drowned_dreams }
	is_inaccessible_trigger = no
}

# Can start the singing sands (health.190-193): of Araby, Sudenburg, Tilea,
# Estalia or Sartosa, not in Lustria, has not had any dream chain before and
# is not off at sea or in hiding.
# Scope: a character.
health_singing_sands_trigger = {
	OR = {
		culture_group = arabyan_group
		culture_group = tilean_group
		culture_group = estalian_group
		culture_group = sartosan_group
		culture = sudenburger
	}
	NOT = { location = { region = world_new_world_lustria } }
	NOT = { has_character_flag = health_drowned_dreams }
	is_inaccessible_trigger = no
}

# Can start the dreaming temple (health.210-213): a living Old World human in
# Lustria (a colonist, adventurer or pirate), who has not had any dream chain
# before and is not off at sea or in hiding.
# Scope: a character.
health_dreaming_temple_trigger = {
	location = { region = world_new_world_lustria }
	is_living_human_trigger = yes
	OR = {
		culture_group = imperial_group
		culture_group = bretonnian_group
		culture_group = tilean_group
		culture_group = estalian_group
		culture_group = sartosan_group
		culture_group = norscan_group
		culture_group = arabyan_group
		culture_group = albion_group
		culture_group = kislevite_group
	}
	NOT = { has_character_flag = health_drowned_dreams }
	is_inaccessible_trigger = no
}
