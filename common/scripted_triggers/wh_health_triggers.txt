# Possessed, and hears no voice yet: neither the voice from on high, nor the
# daemon's voice, nor the twin-tailed omen.
# Scope: a character.
health_no_voice_trigger = {
	NOR = {
		has_character_modifier = voice_of_tzeentch
		has_character_modifier = voice_of_greater_daemon
		has_character_modifier = holy_comet
	}
}

# Possessed and hearing the voice from on high (voice_of_tzeentch): a voice
# that sounds like the character's god and is Tzeentch.
# Scope: a character.
health_voice_from_on_high_trigger = {
	trait = possessed
	has_character_modifier = voice_of_tzeentch
}

# Possessed and has seen the twin-tailed omen (holy_comet).
# Scope: a character.
health_twin_tailed_omen_trigger = {
	trait = possessed
	has_character_modifier = holy_comet
}

# Possessed and hearing a daemon (voice_of_greater_daemon).
# Scope: a character.
health_daemon_voice_trigger = {
	trait = possessed
	has_character_modifier = voice_of_greater_daemon
}

# Follows a faith that has a heresy, so become_heretic does something: the
# Grail, Ormazd, Akghan, Cytharai and Cadai faiths.
# Scope: a character.
health_has_heresy_trigger = {
	OR = {
		religion = grail
		religion = ormazd
		religion = akghan
		religion = cytharai
		religion = cadai
	}
}

# Can start the drowned dreams (health.170-173): follows a northern faith of
# the Old World near the Sea of Claws, has not had the dreams before and is
# not off at sea or in hiding.
# Scope: a character.
health_drowned_dreams_trigger = {
	OR = {
		religion = kislev_gods
		religion = northern_gods
		religion = northern_gods_reformed
		religion = ulrican
		religion = taalite
		religion = old_faith
		religion = old_faith_reformed
	}
	NOT = { has_character_flag = health_drowned_dreams }
	is_inaccessible_trigger = no
}
