###############################################################################
# Triggers for hiding and murder in hiding (events/WHG_intrigue_events.txt,
# Part 2, namespace hiding)
###############################################################################

# The one who hears the news wishes the hidden character ill: ROOT plots
# against FROM, backs a known plot against FROM, or is FROM's rival.
# Scope: ROOT, the one told. FROM: the character going into or out of hiding.
hiding_gloats_trigger = {
	OR = {
		FROM = { is_plot_target_of = ROOT }
		FROM = { any_known_plotter = { any_plot_backer = { character = ROOT } } }
		is_rival = FROM
	}
}

# A plot to kill this character is afoot and this character does not know
# who is behind it.
# Scope: the would-be victim.
hiding_real_murder_plot_trigger = {
	any_unknown_plotter = {
		num_of_plot_backers = 1
		OR = {
			has_plot = plot_kill_character
			has_plot = plot_kill_spouse
		}
	}
}

# This character may hear rumours of a plot on their life: a real plot is
# afoot, or ROOT is paranoid and sees killers everywhere.
# Scope: the would-be victim. ROOT: the ruler who hears the rumour.
hiding_rumour_victim_trigger = {
	NOT = { has_character_modifier = suspected_murder_plot }
	OR = {
		ROOT = { trait = paranoid }
		hiding_real_murder_plot_trigger = yes
	}
}

# Some plotter against this character, or one of their backers, has a loose
# tongue: no skill at intrigue.
# Scope: the would-be victim.
hiding_clumsy_plotter_trigger = {
	any_plotter = {
		OR = {
			NOT = { intrigue = 3 }
			any_plot_backer = { NOT = { intrigue = 3 } }
		}
	}
}

# Some plotter against this character, or one of their backers, is a fool
# or a madman.
# Scope: the would-be victim.
hiding_foolish_plotter_trigger = {
	any_plotter = {
		OR = {
			is_dumb_trigger = yes
			trait = lunatic_1
			trait = lunatic_2
			trait = lunatic_3
			any_plot_backer = {
				OR = {
					is_dumb_trigger = yes
					trait = lunatic_1
					trait = lunatic_2
					trait = lunatic_3
				}
			}
		}
	}
}

# A plot against this character has six or more backers: too many mouths.
# Scope: the would-be victim.
hiding_crowded_plot_trigger = {
	any_plotter = { num_of_plot_backers = 6 }
}

# A murder plotter whose target sits in hiding.
# Scope: the plotter.
hiding_murder_plotter_trigger = {
	OR = {
		has_plot = plot_kill_character
		has_plot = plot_kill_spouse
	}
	plot_power = 1.0
	plot_target_char = {
		is_inaccessible_trigger = yes
		prisoner = no
	}
	NOT = { has_character_flag = hiding_murder_in_motion }
}
