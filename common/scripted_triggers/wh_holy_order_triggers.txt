# Holy order triggers for WH: Geheimnisnacht.

# Title scope. True for the Old World islands the Knights Mariner ask for after a holy war
# (used by check_holy_order_donation_effect and HFP.44000).
wh_mariner_island_title_trigger = {
	OR = {
		title = d_sartosa
		title = d_siren_isles
		title = d_nacary_isles
		title = d_hadmaut_islands
		title = c_nonucci_island
		AND = {
			tier = COUNT
			duchy = {
				OR = {
					title = d_sartosa
					title = d_siren_isles
					title = d_nacary_isles
					title = d_hadmaut_islands
				}
			}
		}
	}
}

# Character scope. True if the character or a liege already holds one of those islands.
wh_holds_mariner_island_trigger = {
	OR = {
		any_demesne_title = {
			tier = COUNT
			wh_mariner_island_title_trigger = yes
		}
		any_realm_lord = {
			any_demesne_title = {
				tier = COUNT
				wh_mariner_island_title_trigger = yes
			}
		}
	}
}
