# Holy order triggers for WH: Geheimnisnacht.

# Title scope. True for the Old World islands the Knights Mariner ask for after a holy war
# (used by check_holy_order_donation_effect and holyorder.1).
wh_mariner_island_title_trigger = {
	OR = {
		title = d_sartosa
		title = d_siren_isles
		title = d_nacary_isles
		title = d_hadmaut_islands
		title = c_nonucci_island
		AND = {
			tier = COUNT
			duchy = {
				OR = {
					title = d_sartosa
					title = d_siren_isles
					title = d_nacary_isles
					title = d_hadmaut_islands
				}
			}
		}
	}
}

# Character scope. True if the character or a liege already holds one of those islands.
wh_holds_mariner_island_trigger = {
	OR = {
		any_demesne_title = {
			tier = COUNT
			wh_mariner_island_title_trigger = yes
		}
		any_realm_lord = {
			any_demesne_title = {
				tier = COUNT
				wh_mariner_island_title_trigger = yes
			}
		}
	}
}

# Ruler scope; event target holy_order_master. True if the ruler owes money
# to that master's order, in either loan slot.
holy_order_debtor_trigger = {
	OR = {
		AND = {
			has_character_modifier = borrowed_from_holy_order
			persistent_event_target:holy_order_creditor = {
				holder_scope = { character = event_target:holy_order_master }
			}
		}
		AND = {
			has_character_modifier = borrowed_from_holy_order_b
			persistent_event_target:holy_order_creditor_b = {
				holder_scope = { character = event_target:holy_order_master }
			}
		}
	}
}

# Master scope; FROM is a ruler. True if FROM owes money to this master's
# order (targeted decisions: ROOT is the master).
holy_order_owed_by_from_trigger = {
	FROM = {
		OR = {
			AND = {
				has_character_modifier = borrowed_from_holy_order
				persistent_event_target:holy_order_creditor = {
					holder_scope = { character = ROOT }
				}
			}
			AND = {
				has_character_modifier = borrowed_from_holy_order_b
				persistent_event_target:holy_order_creditor_b = {
					holder_scope = { character = ROOT }
				}
			}
		}
	}
}

# Ruler scope. True if the ruler can take on another loan from a holy
# order: anyone may owe one order; the greedy, the ambitious, the hedonist
# and the decadent may owe two.
holy_order_can_borrow_trigger = {
	OR = {
		NOT = { has_character_modifier = borrowed_from_holy_order }
		AND = {
			NOT = { has_character_modifier = borrowed_from_holy_order_b }
			OR = {
				trait = greedy
				trait = ambitious
				trait = hedonist
				trait = decadent
			}
		}
	}
}

# Character scope. A son or brother the order would take: an adult unlanded
# man of the faith, able to fight, who is not the ruler's heir.
# Scope: the candidate; event target holy_order_master.
holy_order_recruit_trigger = {
	is_female = no
	is_adult = yes
	is_ruler = no
	prisoner = no
	NOT = { age = 46 }
	NOT = { trait = incapable }
	NOT = { has_character_flag = refused_holy_order }
	martial = 5
	religion = event_target:holy_order_master
	is_heretic = no
	is_primary_heir = no
}
