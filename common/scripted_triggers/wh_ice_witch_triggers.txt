# A Grand Ice Witch: rank 3 or higher in the Sisterhood.
# Scope: character.
frost_grand_witch_trigger = {
	society_member_of = ice_witch_sisterhood
	society_rank >= 3
	prisoner = no
}

# A novice of the Sisterhood fit for a winter on the glacier (frostwitch.35).
# Scope: a courtier.
frost_novice_trigger = {
	society_member_of = ice_witch_sisterhood
	society_rank == 1
	prisoner = no
	is_incapable = no
}

# An Ungol hag, a wise woman outside the Sisterhood.
# Scope: character.
frost_hag_trigger = {
	OR = {
		trait = hag_witch
		trait = lore_hag
	}
	NOT = { society_member_of = ice_witch_sisterhood }
}
