# A landed lord of Kislev's gods, count or higher.
# Scope: character.
ursun_kislev_lord_trigger = {
	is_ruler = yes
	religion = kislev_gods
	higher_tier_than = BARON
}

# May receive the omens of the Wounded Bear: the High Speaker of Ursun, or
# the First of the Circle of the Bear (ursun.1).
# Scope: character.
ursun_quest_leader_trigger = {
	religion = kislev_gods
	OR = {
		has_landed_title = k_cult_of_ursun
		AND = {
			society_member_of = circle_of_the_bear
			society_rank >= 4
		}
	}
}

# A boyar who might mock the god at the Waking Day feast (ursun.20).
# Scope: a vassal or courtier.
ursun_mocker_trigger = {
	is_adult = yes
	is_female = no
	prisoner = no
	religion = kislev_gods
	NOT = { trait = zealous }
	NOT = { has_landed_title = k_cult_of_ursun }
	NOT = { society_member_of = circle_of_the_bear }
}

# An Ungol lord still on the old faith, walking the spirit-paths toward
# Ursun (ursun.60 - 62). The Ropsmann keep apart and do not walk them.
# Scope: character.
ursun_spirit_paths_trigger = {
	is_ruler = yes
	culture = ungol
	OR = {
		religion = northern_gods
		religion = northern_gods_reformed
	}
}

# A Kislevite of the court who has mutated: the priests want them gone
# (ursun.70).
# Scope: a courtier.
ursun_tainted_trigger = {
	is_adult = yes
	prisoner = no
	is_ruler = no
	culture_group = kislevite_group
	is_mutated_trigger = yes
}

# A lord who may swear Tor's Oath at the start of a war (ursun.41).
# Scope: character.
ursun_tor_oath_candidate_trigger = {
	is_ruler = yes
	is_adult = yes
	prisoner = no
	religion = kislev_gods
	higher_tier_than = BARON
	NOT = { has_character_flag = ursun_tor_sworn }
}
