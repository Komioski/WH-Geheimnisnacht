# A Kislevite lord with a coastal province of Kislev in his demesne, and no
# truce with the Norscans (kislevnorth.40).
# Scope: character.
north_norscan_target_trigger = {
	is_ruler = yes
	culture_group = kislevite_group
	higher_tier_than = BARON
	NOT = { has_character_modifier = north_norscan_truce }
	any_demesne_province = {
		region = world_old_world_kislev
		port = yes
	}
}

# A lord with land in the troll country of northern Kislev (kislevnorth.50).
# Scope: character.
north_troll_country_trigger = {
	is_ruler = yes
	higher_tier_than = BARON
	war = no
	any_demesne_province = { region = world_old_world_kislev_northern }
}
