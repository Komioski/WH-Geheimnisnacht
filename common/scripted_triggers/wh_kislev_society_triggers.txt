# Can join the Circle of the Bear: a grown man or woman of Ursun's faith with
# a warrior's schooling or a warrior's arm.
# Scope: a character.
kislevsoc_circle_candidate_trigger = {
	religion = kislev_gods
	is_adult = yes
	trait = creature_human
	is_witch_hunter_trigger = no
	NOT = { trait = ice_witch }
	OR = {
		martial = 10
		has_education_martial_trigger = yes
	}
}

# Can join the Boyar Duma: a Kislevite noble or guild master with a head for
# talk or money. Landed rulers and patricians always; courtiers need the
# skill or the focus.
# Scope: a character.
kislevsoc_duma_candidate_trigger = {
	culture_group = kislevite_group
	is_adult = yes
	trait = creature_human
	NOT = { trait = ice_witch }
	OR = {
		is_landed = yes
		is_patrician = yes
		diplomacy = 12
		stewardship = 12
		has_focus = focus_business
		has_character_flag = kislevsoc_druzhina_raised
	}
}

# A member of the Circle in the Rota of the Dawn, Dazh's temple guard.
# Scope: a character.
kislevsoc_dawn_trigger = {
	society_member_of = circle_of_the_bear
	has_character_flag = kislevsoc_dawn
}

# Stands with the Ice Court, the power behind the Ice Throne: the Tzar(ina),
# the Grand Ice Witches, the Rota of the Dawn and the druzhina raised by
# the crown. A hook for the Ice Court and Orthodoxy system (later segment).
# Scope: a character.
kislev_ice_court_trigger = {
	OR = {
		has_landed_title = e_kislev
		AND = {
			society_member_of = ice_witch_sisterhood
			society_rank >= 3
		}
		kislevsoc_dawn_trigger = yes
		has_character_flag = kislevsoc_druzhina_raised
	}
}

# Stands with the Orthodoxy of Ursun: the High Speaker and the Circle of the
# Bear outside the Rota of the Dawn.
# Scope: a character.
kislev_orthodoxy_trigger = {
	OR = {
		has_landed_title = k_cult_of_ursun
		AND = {
			society_member_of = circle_of_the_bear
			NOT = { has_character_flag = kislevsoc_dawn }
		}
	}
}

# Has earned the next rank's trial: rank 1 slays a bear by their own hand,
# rank 2 completes a Circle trial.
# Scope: a Circle member.
kislevsoc_next_rank_open_trigger = {
	OR = {
		AND = {
			society_member_of = circle_of_the_bear
			society_rank == 1
			NOT = { is_society_rank_full = { society = circle_of_the_bear rank = 2 } }
		}
		AND = {
			society_member_of = circle_of_the_bear
			society_rank == 2
			NOT = { is_society_rank_full = { society = circle_of_the_bear rank = 3 } }
		}
		AND = {
			society_member_of = boyar_duma
			society_rank == 1
			NOT = { is_society_rank_full = { society = boyar_duma rank = 2 } }
		}
		AND = {
			society_member_of = boyar_duma
			society_rank == 2
			NOT = { is_society_rank_full = { society = boyar_duma rank = 3 } }
		}
	}
}
