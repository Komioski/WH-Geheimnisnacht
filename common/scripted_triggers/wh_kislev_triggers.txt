# A man who works magic, or is gifted and learned enough to try: the quarry
# of the Ice Witches' hunt (kislev.6).
# Scope: the candidate; ROOT the hunting sister.
kislev_male_mage_trigger = {
	is_female = no
	is_adult = yes
	trait = creature_human
	NOT = { trait = hollowed }
	is_inaccessible_or_incapable_trigger = no
	NOT = { any_spouse = { character = ROOT } }
	OR = {
		has_magic_power_trigger = yes
		trait = mage_apprentice
		AND = { trait = magic_potential_4 learning = 10 }
		AND = { trait = magic_potential_3 learning = 12 }
		AND = { trait = magic_potential_2 learning = 15 }
	}
}

# A Kislevite realm the sister may hunt or recruit in.
# Scope: a ruler.
kislev_sisterhood_realm_trigger = {
	culture_group = kislevite_group
	religion_group = old_world_group
}

# There is a male mage to hunt in the sister's Kislevite liege's realm or in
# a neighbouring Kislevite realm near her capital (kislev.1, kislev.6).
# Scope: ROOT is the sister.
kislev_male_mage_exists_trigger = {
	OR = {
		top_liege = {
			kislev_sisterhood_realm_trigger = yes
			any_realm_character = { kislev_male_mage_trigger = yes }
		}
		any_neighbor_independent_ruler = {
			kislev_sisterhood_realm_trigger = yes
			any_realm_character = {
				kislev_male_mage_trigger = yes
				ROOT = { capital_scope = { distance = { where = PREV distance < 1000 } } }
			}
		}
	}
}

# A girl or woman with the gift who is not yet a sister, and has not
# refused or been sought already (kislev.11).
# Scope: the candidate.
kislev_witch_recruit_trigger = {
	is_female = yes
	age = 6
	prisoner = no
	is_in_society = no
	NOT = { trait = ice_witch }
	NOT = { has_character_flag = refused_magical_recruitment }
	NOT = { has_character_flag = priority_mage_recruit }
	OR = { culture_group = kislevite_group religion = kislev_gods }
	OR = { has_magic_power_trigger = no trait = lore_ice }
	OR = {
		trait = magic_potential_4
		trait = magic_potential_3
		trait = magic_potential_2
		trait = magic_potential_1
		has_character_flag = magic_potential_flag_4
		has_character_flag = magic_potential_flag_3
		has_character_flag = magic_potential_flag_2
		has_character_flag = magic_potential_flag_1
		has_bloodline_flag = magic_bloodline_female_kislev
	}
}

# There is someone for the sister to recruit (kislev.1, kislev.11): a
# player, her own daughter, a woman of her realm, someone of her Kislevite
# liege's realm or of a neighbouring Kislevite realm.
# Scope: ROOT is the sister.
kislev_recruit_exists_trigger = {
	OR = {
		any_player = { kislev_witch_recruit_trigger = yes }
		any_child = { kislev_witch_recruit_trigger = yes }
		any_realm_lord = { kislev_witch_recruit_trigger = yes }
		top_liege = {
			kislev_sisterhood_realm_trigger = yes
			any_realm_character = { kislev_witch_recruit_trigger = yes }
		}
		any_neighbor_independent_ruler = {
			kislev_sisterhood_realm_trigger = yes
			any_realm_character = { kislev_witch_recruit_trigger = yes }
		}
	}
}

# The newborn's parents worship the Ruinous Powers or are mutants, so the
# birth rites find taint more often (kislev.61, kislev.63).
# Scope: the newborn.
kislev_chaos_touched_parents_trigger = {
	OR = {
		father = { religion_group = chaos_gods_group }
		mother = { religion_group = chaos_gods_group }
		father = { is_mutated_trigger = yes }
		mother = { is_mutated_trigger = yes }
	}
}

# A woman of Kislev's gods who can take up Fearfrost (kislev.70).
# Scope: a kinswoman of the dead bearer.
kislev_fearfrost_heir_trigger = {
	is_alive = yes
	is_female = yes
	religion = kislev_gods
	NOT = { character = ROOT }
}

# A Gospodar province of Kislev that is not a nomad camp: the melting pot
# can start there (kislev.150).
# Scope: province.
kislev_melting_pot_province_trigger = {
	region = world_old_world_kislev
	culture = gospodar
	any_province_holding = { NOT = { holding_type = nomad } }
}

# A province of Kislev of a Kislevite-group culture other than the new
# Kislevite one (kislev.151).
# Scope: province.
kislev_melting_pot_spread_trigger = {
	region = world_old_world_kislev
	culture_group = kislevite_group
	NOT = { culture = kislevite }
	any_province_holding = { NOT = { holding_type = nomad } }
	NOT = { has_province_modifier = stanitsa_ropsmann_apart } # stanitsa.52
	NOT = { # the Union of the Old Peoples (realm.70)
		OR = {
			culture = ungol
			culture = ropsmann
		}
		owner = { kislev_old_peoples_realm_trigger = yes }
	}
}

# A Gospodar, Ungol or other Kislevite-group character who has not taken up
# the new Kislevite culture yet (kislev.152 - 154). Ropsmann who chose to stay
# apart, or live at a court with a keeper of the tongue, hold out, and so do
# the Ungols and Ropsmenn of the Union of the Old Peoples.
# Scope: character.
kislev_melting_pot_convert_trigger = {
	culture_group = kislevite_group
	NOT = { culture = kislevite }
	NOT = { stanitsa_ropsmann_holds_out_trigger = yes }
	NOT = { kislev_old_peoples_holds_out_trigger = yes }
}

# The character's top liege holds the Union of the Old Peoples (realm.70).
# Scope: character.
kislev_old_peoples_realm_trigger = { top_liege = { has_landed_title = e_old_peoples } }

# An Ungol or Ropsmann of the Union of the Old Peoples, who will not take up
# the Kislevite culture while the union stands.
# Scope: character.
kislev_old_peoples_holds_out_trigger = {
	OR = {
		culture = ungol
		culture = ropsmann
	}
	OR = {
		kislev_old_peoples_realm_trigger = yes
		host = { kislev_old_peoples_realm_trigger = yes }
	}
}

# The character's top liege rules the Tsardom of Kislev.
# Scope: character.
kislev_in_tsardom_trigger = { top_liege = { primary_title = { title = e_kislev } } }

# The "faster melting pots" culture conversion game rule is on.
kislev_faster_melting_pots_trigger = {
	OR = {
		has_game_rule = { name = culture_conversion value = faster_melting_pots }
		has_game_rule = { name = culture_conversion value = faster_melting_pots_and_slower_normal }
	}
}
