# A man who works magic, or is gifted and learned enough to try: the quarry
# of the Ice Witches' hunt (kislev.6).
# Scope: the candidate; ROOT the hunting sister.
kislev_male_mage_trigger = {
	is_female = no
	is_adult = yes
	trait = creature_human
	NOT = { trait = hollowed }
	is_inaccessible_or_incapable_trigger = no
	NOT = { any_spouse = { character = ROOT } }
	OR = {
		has_magic_power_trigger = yes
		trait = mage_apprentice
		AND = { trait = magic_potential_4 learning = 10 }
		AND = { trait = magic_potential_3 learning = 12 }
		AND = { trait = magic_potential_2 learning = 15 }
	}
}

# A Kislevite realm the sister may hunt or recruit in.
# Scope: a ruler.
kislev_sisterhood_realm_trigger = {
	culture_group = kislevite_group
	religion_group = old_world_group
}

# There is a male mage to hunt in the sister's Kislevite liege's realm or in
# a neighbouring Kislevite realm near her capital (kislev.1, kislev.6).
# Scope: ROOT is the sister.
kislev_male_mage_exists_trigger = {
	OR = {
		top_liege = {
			kislev_sisterhood_realm_trigger = yes
			any_realm_character = { kislev_male_mage_trigger = yes }
		}
		any_neighbor_independent_ruler = {
			kislev_sisterhood_realm_trigger = yes
			any_realm_character = {
				kislev_male_mage_trigger = yes
				ROOT = { capital_scope = { distance = { where = PREV distance < 1000 } } }
			}
		}
	}
}

# A girl or woman with the gift who is not yet a sister, and has not
# refused or been sought already (kislev.11).
# Scope: the candidate.
kislev_witch_recruit_trigger = {
	is_female = yes
	age = 6
	prisoner = no
	is_in_society = no
	NOT = { trait = ice_witch }
	NOT = { has_character_flag = refused_magical_recruitment }
	NOT = { has_character_flag = priority_mage_recruit }
	OR = { culture_group = kislevite_group religion = kislev_gods }
	OR = { has_magic_power_trigger = no trait = lore_ice }
	OR = {
		trait = magic_potential_4
		trait = magic_potential_3
		trait = magic_potential_2
		trait = magic_potential_1
		has_character_flag = magic_potential_flag_4
		has_character_flag = magic_potential_flag_3
		has_character_flag = magic_potential_flag_2
		has_character_flag = magic_potential_flag_1
		has_bloodline_flag = magic_bloodline_female_kislev
	}
}

# There is someone for the sister to recruit (kislev.1, kislev.11): a
# player, her own daughter, a woman of her realm, someone of her Kislevite
# liege's realm or of a neighbouring Kislevite realm.
# Scope: ROOT is the sister.
kislev_recruit_exists_trigger = {
	OR = {
		any_player = { kislev_witch_recruit_trigger = yes }
		any_child = { kislev_witch_recruit_trigger = yes }
		any_realm_lord = { kislev_witch_recruit_trigger = yes }
		top_liege = {
			kislev_sisterhood_realm_trigger = yes
			any_realm_character = { kislev_witch_recruit_trigger = yes }
		}
		any_neighbor_independent_ruler = {
			kislev_sisterhood_realm_trigger = yes
			any_realm_character = { kislev_witch_recruit_trigger = yes }
		}
	}
}
