###############################################################################
# Lifespan and hardiness triggers
# Used to scale "slowly over time" odds (lunacy, illness, infirmity) so that
# long-lived races are not ground down by rolls tuned for a human lifetime.
# Used by events/health_events.txt (6020, 6063, 6100), mutate.5 and lunacy.90.
###############################################################################

# Dwarfs and Chaos Dwarfs: grow old at about 200, four human lifetimes.
# Scope: a character.
lifespan_dwarf_trigger = {
	OR = {
		trait = creature_dwarf
		trait = creature_chaos_dwarf
	}
}

# Elves: live for many centuries.
# Scope: a character.
lifespan_elf_trigger = {
	trait = creature_elf
}

# Other long-lived characters: immortals, Grail Knights and Fimir.
# Scope: a character.
lifespan_other_long_lived_trigger = {
	lifespan_dwarf_trigger = no
	lifespan_elf_trigger = no
	OR = {
		trait = immortal
		trait = grail_knight
		trait = creature_fimir
	}
}

# Lizardmen and greenskins: cold-blooded or fungus-hardy, almost never ill or
# mad in the ordinary way.
# Scope: a character.
lifespan_hardy_race_trigger = {
	OR = {
		trait = creature_saurus
		trait = creature_skink
		trait = creature_slann
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
		trait = creature_snotling
	}
}
