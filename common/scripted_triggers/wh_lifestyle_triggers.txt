# Triggers for the lifestyle events (events/WHG_lifestyle_events.txt).
# The start triggers are checked both by the selector (lifestyle.1) and by
# each chain's first event, which friends can also fire directly.

# A living human (not undead, not a vampire), with no lifestyle trait and no
# chain in progress. Only humans take up these callings.
# Scope: a character.
lifestyle_can_start_trigger = {
	is_living_human_trigger = yes
	NOT = { lifestyle_traits = 1 }
	NOT = { has_character_flag = LE_ST }
}

# Scholar or mystic (lifestyle.10).
lifestyle_scholar_start_trigger = {
	lifestyle_can_start_trigger = yes
	NOT = { trait = blinded }
}

# Gardener or impaler (lifestyle.20).
lifestyle_gardener_start_trigger = {
	lifestyle_can_start_trigger = yes
	NOR = {
		has_character_modifier = the_impaler
		has_character_modifier = the_gardener
	}
}

# A cold lady at court who could be courted (lifestyle.30).
# Scope: a courtier. ROOT is the suitor.
lifestyle_icy_lady_trigger = {
	is_female = yes
	is_married = no
	has_lover = no
	age = 25
	NOT = { age = 35 }
	NOR = {
		trait = celibate
		trait = homosexual
		opinion = { who = ROOT value = 20 }
	}
	reverse_opinion = { who = ROOT value = 20 }
}

# Duelist or poet: an unmarried man outside Araby with a cold lady to court
# (lifestyle.30).
lifestyle_courtship_start_trigger = {
	lifestyle_can_start_trigger = yes
	is_female = no
	is_married = no
	has_lover = no
	NOT = { trait = homosexual }
	NOT = { religion_group = ormazd_gods_group }
	any_courtier = { lifestyle_icy_lady_trigger = yes }
}

# Hunter or falconer: a man outside the faith of Shallya (lifestyle.40).
lifestyle_hunter_start_trigger = {
	lifestyle_can_start_trigger = yes
	is_female = no
	NOT = { religion = shallyan }
}

# Poet or falconer: any ruler outside the faith of Shallya who is neither yet
# (lifestyle.50). As in vanilla, another lifestyle trait does not bar it.
lifestyle_poet_falconer_start_trigger = {
	is_living_human_trigger = yes
	is_playable = yes
	NOT = { has_character_flag = LE_ST }
	NOT = { religion = shallyan }
	NOR = {
		trait = falconer
		trait = poet
	}
}

# Celibate or hedonist: the Old World faiths (lifestyle.60).
lifestyle_celibacy_start_trigger = {
	lifestyle_can_start_trigger = yes
	religion_group = old_world_group
}

# Araby: hedonist or poet (lifestyle.70).
lifestyle_araby_start_trigger = {
	lifestyle_can_start_trigger = yes
	religion_group = ormazd_gods_group
}

# Araby: the Ormazdic Jurist (lifestyle.80).
lifestyle_jurist_start_trigger = {
	is_living_human_trigger = yes
	religion_group = ormazd_gods_group
	NOT = { has_character_flag = LE_ST }
	NOT = { has_character_modifier = aspiring_faqih }
	is_dumb_trigger = no
	NOR = {
		trait = faqih
		trait = scholar
		trait = hafiz
		trait = inbred
		trait = infirm
	}
	NOT = {
		has_alternate_start_parameter = {
			key = religion_names
			value = random
		}
	}
}

# Peoples who cheer an impaler instead of pelting him (lifestyle.23).
lifestyle_impaling_expected_trigger = {
	OR = {
		religion_group = chaos_gods_group
		religion_group = cytharai_group
		religion_group = greenskin_gods_group
		religion_group = necromantic_group
		religion_group = horned_rat_god_group
	}
}

### Voices: which people a lifestyle event speaks for ###

# The Empire.
lifestyle_voice_empire_trigger = { culture_group = imperial_group }

# Bretonnia.
lifestyle_voice_bretonnia_trigger = { culture_group = bretonnian_group }

# Kislev.
lifestyle_voice_kislev_trigger = { culture_group = kislevite_group }

# Tilea, Estalia and Sartosa.
lifestyle_voice_south_trigger = {
	OR = {
		culture_group = tilean_group
		culture_group = estalian_group
		culture_group = sartosan_group
	}
}

# The Northmen: Norsca, the Kurgan and the Hung.
lifestyle_voice_north_trigger = {
	OR = {
		culture_group = norscan_group
		culture_group = kurgan_group
		culture_group = hung_group
	}
}

# Araby.
lifestyle_voice_araby_trigger = { culture_group = arabyan_group }

# Cathay, Ind and the East.
lifestyle_voice_east_trigger = { culture_group = oriental_group }
