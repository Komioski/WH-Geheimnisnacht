# An independent Skink ruler with a sleeping Slann among their vassals.
# Scope: a character.
lizardmen_has_sleeping_slann_trigger = {
	independent = yes
	trait = creature_skink
	any_vassal = {
		trait = creature_slann
		trait = lizardman_sleepingslann
	}
}

# A Skink who can be the target of plot_investigate_alignment events.
# Scope: a character.
lizardmen_alignment_target_trigger = {
	age = 16
	prisoner = no
	trait = creature_skink
	NOT = { trait = incapable }
	NOT = { trait = in_hiding }
}

# Scope: a war leader of the Old Ones. With liz_crossing_enemy_trigger on
# the enemy: the ruler sits in Lustria and the enemy's capital lies outside
# the New World, so the host must cross the ocean (the AI cannot use the
# spawned fleets and tries to walk around the world instead).
liz_crossing_ruler_trigger = {
	is_landed = yes
	religion_group = old_ones_group
	capital_scope = { region = world_new_world_lustria }
}

# Scope: the enemy of a liz_crossing_ruler_trigger ruler.
liz_crossing_enemy_trigger = {
	is_landed = yes
	capital_scope = { NOT = { region = world_new_world } }
}
