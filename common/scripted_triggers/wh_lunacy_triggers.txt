# Wears an equipped artifact that carries madness with it (madness_artifact
# flag). Such a character suffers lunacy events of every stage, though less
# often than a true lunatic.
# Scope: a character.
lunacy_madness_artifact_trigger = {
	any_artifact = {
		is_artifact_equipped = yes
		has_artifact_flag = madness_artifact
	}
}

# Twitchy stage of lunacy (lunatic_1) or a madness artifact.
# Scope: a character.
lunacy_twitchy_trigger = {
	OR = {
		trait = lunatic_1
		lunacy_madness_artifact_trigger = yes
	}
}

# Maniac stage of lunacy (lunatic_2) or a madness artifact.
# Scope: a character.
lunacy_maniac_trigger = {
	OR = {
		trait = lunatic_2
		lunacy_madness_artifact_trigger = yes
	}
}

# Raving stage of lunacy (lunatic_3) or a madness artifact.
# Scope: a character.
lunacy_raving_trigger = {
	OR = {
		trait = lunatic_3
		lunacy_madness_artifact_trigger = yes
	}
}

# Has one of the nine madnesses. A character only ever has one at a time.
# Scope: a character.
lunacy_has_madness_trigger = {
	OR = {
		has_character_modifier = blind_virtue
		has_character_modifier = fake_mage
		has_character_modifier = despairing_laughter
		has_character_modifier = unfeeling
		has_character_modifier = bloodthirsty
		has_character_modifier = god_delusion
		has_character_modifier = everchild_syndrome
		has_character_modifier = exquisite_agonies
		has_character_modifier = heart_of_melancholy
	}
}

# Awake and able to act on a madness.
# Scope: a character.
lunacy_can_act_trigger = {
	NOR = {
		trait = resting
		trait = incapable
		trait = incapacitated
	}
}

# Can be struck by lunacy at all: spirits, treekin, Nehekharan dead and
# Chaos spawn have no mind left to lose, and daemons are madness given form.
# Also used by the health events for stress, depression and possession.
# Scope: a character.
lunacy_can_go_mad_trigger = {
	NOR = {
		trait = creature_forestspirit
		trait = creature_treekin
		trait = creature_nehekharan
		trait = creature_chaosspawn
	}
	is_daemon_race_trigger = no
}

# Follows a faith that frowns on madness: anything but the Chaos gods, the
# necromantic faiths and the Horned Rat. These lose piety when their ruler goes
# mad, and their vassals and courtiers think less of the ruler.
# Scope: a character.
lunacy_sane_faith_trigger = {
	NOR = {
		religion_group = chaos_gods_group
		religion_group = necromantic_group
		religion_group = horned_rat_god_group
	}
}

# A courtier a bloodthirsty lunatic can murder: an adult who is not a ruler,
# not a spirit, daemon, vampire or Nehekharan, and has no magic to defend
# themselves with.
# Scope: a character; ROOT is the murderer.
lunacy_murder_victim_trigger = {
	is_adult = yes
	is_ruler = no
	prisoner = no
	NOT = { character = ROOT }
	NOR = {
		trait = avatar_isha
		trait = avatar_kurnous
		trait = creature_treekin
		trait = creature_forestspirit
		trait = creature_nehekharan
	}
	is_daemon_race_trigger = no
	is_vampire_trigger = no
	has_magic_power_trigger = no
}

# A courtier with a weakness a maniac with despairing laughter can mock: old
# age, wounds, scars, weight, sickness, a hard pregnancy, mutations or
# lunacy.
# Scope: a character; ROOT is the maniac.
lunacy_mock_target_trigger = {
	NOT = { character = ROOT }
	prisoner = no
	NOR = {
		trait = incapable
		trait = incapacitated
	}
	is_daemon_race_trigger = no
	OR = {
		trait = infirm
		trait = wounded
		trait = maimed
		trait = disfigured
		trait = burnt_by_asuryan
		trait = is_fat
		trait = is_malnourished
		trait = leper
		trait = pneumonic
		trait = ill
		trait = syphilitic
		trait = troubled_pregnancy
		trait = hard_pregnancy
		is_mutated_trigger = yes
		trait = lunatic_3
	}
}

# Has a lover or a spouse with the everchild syndrome.
# Scope: a character.
lunacy_everchild_partner_trigger = {
	OR = {
		any_lover = { has_character_modifier = everchild_syndrome }
		any_spouse = { has_character_modifier = everchild_syndrome }
	}
}

# Looks like a child forever: a Lahmian or von Carstein everchild.
# Scope: a character.
lunacy_everchild_body_trigger = {
	OR = {
		graphical_culture = lahmianeverkidgfx
		graphical_culture = carsteineverkidgfx
	}
}
