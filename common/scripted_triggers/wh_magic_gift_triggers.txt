# THIS (a newborn) belongs to a people that can be born with the gift. Dawi,
# greenskins, Skaven, beastmen, ogres, halflings, daemons, Tomb Kings,
# lizardmen, Asrai and the immortal and Lahmian lines have their own magic,
# or none.
gift_can_be_born_trigger = {
	NOR = {
		has_character_flag = magic_potential_rolled
		hidden_or_open_magic_potential_trigger = yes
		is_trained_mage_trigger = yes
		culture_group = beastman_group
		culture_group = daemonic_group
		culture_group = dwarf_group
		culture_group = dwarf_chaos_group
		culture_group = forest_group
		culture_group = goblin_group
		culture_group = halfling_old_world
		culture_group = halfling_lustrian
		culture_group = immortal_group
		culture_group = lahmian_group
		culture_group = lizardman_group
		culture_group = nehekharan_group
		culture_group = ogre_group
		culture_group = orc_group
		culture_group = skaven_group
	}
}

# THIS (a parent) carries the gift: shown, hidden or recessive.
gift_parent_gifted_trigger = {
	OR = {
		trait = magic_potential_1
		trait = magic_potential_2
		trait = magic_potential_3
		trait = magic_potential_4
		has_character_flag = recessive_mage
		has_character_flag = magic_potential_flag_1
		has_character_flag = magic_potential_flag_2
		has_character_flag = magic_potential_flag_3
		has_character_flag = magic_potential_flag_4
	}
}

# THIS (a parent) holds an active Ice Witch bloodline and lives in Kislev or
# the Troll Country, where the Ice Witches' gift is tied to the land.
gift_parent_ice_blood_trigger = {
	any_owned_bloodline = {
		has_bloodline_flag = magic_bloodline_female_kislev
		bloodline_is_active_for = PREV
	}
	OR = {
		location = { region = world_old_world_kislev }
		location = { region = world_troll_country }
		liege = { capital_scope = { region = world_old_world_kislev } }
	}
}

# THIS (a newborn) has a sure claim to the gift: an Ice Witch bloodline in
# Kislev, a magic bloodline, or a bloodline that passes the gift to his own
# sex.
gift_sure_bloodline_trigger = {
	OR = {
		true_father_even_if_dead = { gift_parent_ice_blood_trigger = yes }
		mother_even_if_dead = { gift_parent_ice_blood_trigger = yes }
		true_father_even_if_dead = {
			any_owned_bloodline = {
				has_bloodline_flag = magic_bloodline
				bloodline_is_active_for = PREV
			}
		}
		mother_even_if_dead = {
			any_owned_bloodline = {
				has_bloodline_flag = magic_bloodline
				bloodline_is_active_for = PREV
			}
		}
		AND = {
			is_female = yes
			OR = {
				true_father_even_if_dead = {
					any_owned_bloodline = {
						has_bloodline_flag = magic_bloodline_female
						bloodline_is_active_for = PREV
					}
				}
				mother_even_if_dead = {
					any_owned_bloodline = {
						has_bloodline_flag = magic_bloodline_female
						bloodline_is_active_for = PREV
					}
				}
			}
		}
		AND = {
			is_female = no
			OR = {
				true_father_even_if_dead = {
					any_owned_bloodline = {
						has_bloodline_flag = magic_bloodline_male
						bloodline_is_active_for = PREV
					}
				}
				mother_even_if_dead = {
					any_owned_bloodline = {
						has_bloodline_flag = magic_bloodline_male
						bloodline_is_active_for = PREV
					}
				}
			}
		}
	}
}

# THIS (a newborn) has the gift in his blood: a gifted parent, a magic
# bloodline of either sex, or the Witchling Star.
gift_heritage_trigger = {
	OR = {
		trait = the_witchling_star
		true_father_even_if_dead = { gift_parent_gifted_trigger = yes }
		mother_even_if_dead = { gift_parent_gifted_trigger = yes }
		gift_sure_bloodline_trigger = yes
		true_father_even_if_dead = {
			any_owned_bloodline = {
				OR = {
					has_bloodline_flag = magic_bloodline_female
					has_bloodline_flag = magic_bloodline_male
				}
				bloodline_is_active_for = PREV
			}
		}
		mother_even_if_dead = {
			any_owned_bloodline = {
				OR = {
					has_bloodline_flag = magic_bloodline_female
					has_bloodline_flag = magic_bloodline_male
				}
				bloodline_is_active_for = PREV
			}
		}
	}
}

# THIS (a young character whose gift has just shown) would be in danger if
# he showed it: magic is banned where he lives and his sex may not wield it,
# or he is a man in Kislev or under Malekith, or a woman in Araby.
gift_reveal_dangerous_trigger = {
	OR = {
		AND = {
			NOT = { trait = creature_elf }
			liege = { has_magic_law_0_trigger = yes }
			gender_is_allowed_magic_trigger = no
			NOT = { top_liege = { has_character_modifier = expelled_k_witch_hunters_order } }
			NOT = { top_liege = { trait = peasant_leader } }
		}
		AND = {
			is_female = no
			OR = { top_liege = { trait = malekith } liege = { religion = kislev_gods } }
		}
		AND = { is_female = yes liege = { OR = { religion = ormazd religion = kitabid } } }
	}
}

# THIS (a trained mage) lives openly under a lord who would burn him and may
# be exposed (gift.10): a witch-burning faith, a man under Kislev's gods or
# a woman under Araby's. A realm ruled by a mage protects its own.
gift_exposed_mage_trigger = {
	is_trained_mage_trigger = yes
	independent = no
	NOR = {
		trait = in_hiding
		is_daemon_race_trigger = yes
		top_liege = { is_trained_mage_trigger = yes }
		top_liege = { trait = mage_apprentice }
	}
	check_if_mage_should_be_burned_trigger = yes
	liege = {
		OR = {
			gift_witch_burning_faith_trigger = yes
			AND = { religion = kislev_gods ROOT = { is_female = no } }
			AND = { OR = { religion = ormazd religion = kitabid } ROOT = { is_female = yes } }
		}
	}
}

}
