# An Elector Count's kingdom with its own grand style (aiclean.57).
# Westerland has none.
# Scope: a title.
aiclean_elector_title_trigger = {
	OR = {
		title = k_reikland
		title = k_ostland
		title = k_averland
		title = k_stirland
		title = k_wissenland
		title = k_sylvania
		title = k_hochland
		title = k_nordland
		title = k_middenland
		title = k_talabecland
		title = k_ostermark
	}
}

# A corsair lord of the Atruhayid, the idolaters of the Arabyan coast, with
# a port for a seat (aiclean.57).
# Scope: a ruler.
aiclean_corsair_style_trigger = {
	OR = {
		religion = atruhayid
		religion = atruhayid_reformed
	}
	OR = {
		government = feudal_government
		government = muslim_government
	}
	capital_scope = { port = yes }
}
