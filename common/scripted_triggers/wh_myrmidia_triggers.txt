# A Myrmidian theocracy fit to hold a high temple: free, or under a
# Myrmidian king or emperor.
# Scope: the holder of the see.
eternal_temple_holder_trigger = {
	is_theocracy = yes
	religion = myrmidian
	OR = {
		independent = yes
		top_liege = {
			higher_real_tier_than = DUKE
			religion = ROOT
		}
	}
}

# The head of the Cult of Myrmidia holds neither high temple.
# Scope: any.
eternal_cult_head_lacks_temple_trigger = {
	k_cult_of_myrmidia = {
		has_holder = yes
		holder_scope = {
			NOT = { character = ROOT }
			NOT = { has_landed_title = d_high_temple_of_magritta }
			NOT = { has_landed_title = d_high_temple_of_remas }
		}
	}
}
