# A landed Kislevite lord, count or higher.
# Scope: character.
oblast_kislev_lord_trigger = {
	is_ruler = yes
	culture_group = kislevite_group
	higher_tier_than = BARON
}

# A lord of the steppe who takes part in the Long Ride (oblast.10).
# Scope: character.
oblast_long_ride_trigger = {
	is_ruler = yes
	is_adult = yes
	OR = {
		culture = ungol
		culture = gospodar
	}
}

# A vassal or courtier who might challenge the lord to a drinking contest
# (oblast.30).
# Scope: a vassal or courtier.
oblast_drinker_trigger = {
	is_adult = yes
	prisoner = no
	is_incapable = no
	is_female = no
	culture_group = kislevite_group
	NOT = { trait = temperate }
}

# Kin who rides with the Gryphon Legion (kislev.52 - 54).
# Scope: a relative.
oblast_legionary_trigger = {
	has_character_flag = kislev_gryphon_legionary
	is_alive = yes
	prisoner = no
}
