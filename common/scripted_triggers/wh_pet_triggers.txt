# Triggers for the pet events (events/WHG_pet_events.txt, namespace pet).

# A lunatic or possessed owner, who gets the mad funeral options.
# Scope: a character.
pet_madman_trigger = {
	OR = {
		trait = lunatic_1
		trait = lunatic_2
		trait = lunatic_3
		trait = possessed
	}
}

# A spouse, lover or friend who could demand the cat goes (pet.11).
# Scope: the spouse, lover or friend.
pet_cat_hater_trigger = {
	ai = yes
	prisoner = no
	NOR = {
		has_character_modifier = pet_cat
		trait = kind
		trait = incapable
	}
}

# Someone who could take the cat (pet.11). The cat hater is excluded.
# Scope: a friend or courtier.
pet_new_cat_owner_trigger = {
	NOT = { character = event_target:cat_hater }
	NOT = { has_character_modifier = pet_cat }
	age = 12
	ai = yes
	prisoner = no
	NOT = { trait = incapable }
}

# A courtier with a young child whose rabbit the cat could kill (pet.17).
# Scope: a courtier.
pet_rabbit_parent_trigger = {
	ai = yes
	prisoner = no
	NOT = { trait = incapable }
	any_child = {
		NOT = { age = 13 }
		prisoner = no
		NOT = { trait = incapable }
	}
}

# An owner odd or soft-hearted enough to adopt every stray (pet.18).
# Scope: the owner.
pet_cat_collector_trigger = {
	OR = {
		pet_madman_trigger = yes
		trait = shy
		trait = paranoid
		trait = imbecile
		trait = inbred
		trait = celibate
		trait = kind
	}
}

# A friend or courtier who can take the cat or the hound (pet.20).
# Scope: a friend or courtier.
pet_animal_caretaker_trigger = {
	age = 8
	prisoner = no
	ai = yes
	is_inaccessible_trigger = no
	NOR = {
		trait = incapable
		trait = cruel
		has_character_modifier = pet_cat
		has_character_modifier = pet_hunting_dog
	}
}

# A cruel, wrathful or cat-mad ruler who may kill a cat hater.
# Scope: the owner.
pet_cruel_cat_lover_trigger = {
	OR = {
		trait = wroth
		trait = cruel
		trait = arbitrary
		has_character_modifier = crazy_cat_lady
		has_character_modifier = crazy_cat_lord
	}
}

# A ruler who keeps great cats: lions in Araby and the Southlands, jaguars in
# Lustria, and the Amazons' war cats anywhere.
# Scope: the owner.
pet_great_cat_keeper_trigger = {
	OR = {
		culture_group = amazon_group
		capital_scope = {
			OR = {
				region = world_greater_araby
				region = world_eastern_southlands
				region = world_new_world_lustria
			}
		}
	}
}

# A courtier who could name the cat a witch's familiar (pet.30): a witch
# hunter, or a zealot of the Old World faiths without a kind heart.
# Scope: a courtier.
pet_familiar_accuser_candidate_trigger = {
	ai = yes
	prisoner = no
	is_inaccessible_trigger = no
	religion_group = old_world_group
	NOT = { trait = incapable }
	OR = {
		is_witch_hunter_trigger = yes
		AND = {
			trait = zealous
			NOT = { trait = kind }
		}
	}
}

# An Old World cat owner with such a courtier, accused at most once (pet.30).
# Scope: the owner.
pet_familiar_accuser_trigger = {
	has_character_modifier = pet_cat
	religion_group = old_world_group
	NOT = { has_character_flag = pet_familiar_accused }
	any_courtier = { pet_familiar_accuser_candidate_trigger = yes }
}
