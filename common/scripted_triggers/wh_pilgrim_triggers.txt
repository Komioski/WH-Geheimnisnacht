# A faith with a great shrine of its own in the pilgrimage chain.
# Scope: a character.
pilgrim_has_great_shrine_trigger = {
	OR = {
		religion = sigmarite
		religion = ulrican
		religion = myrmidian
		religion = manannite
		religion = taalite
		religion = shallyan
		religion = grail
		religion = false_grail
		religion = morrian
		religion = cadai
		religion = asrai_cadai
		religion = cytharai
		religion = true_khaine
		religion = atharti
		religion = anath_raema
		religion = ancestor_gods
		religion = norse_ancestor_gods
	}
}

# May go on a pilgrimage at all: a faith with a great shrine, or an Old
# World faith that honours Venerated Souls (their tombs).
# Scope: a character.
pilgrim_faith_trigger = {
	OR = {
		pilgrim_has_great_shrine_trigger = yes
		AND = {
			religion_group = old_world_group
			saint_no_saints_faith_trigger = no
		}
	}
}

# The shrine's province is held by a lord of the pilgrim's faith group who
# is not at war with the pilgrim.
# Scope: the shrine's province; ROOT is the pilgrim.
pilgrim_shrine_open_trigger = {
	has_owner = yes
	owner = {
		religion_group = ROOT
		NOT = { war_with = ROOT }
	}
}

# A Venerated Soul's tomb the pilgrim may walk to (saint.60).
# Scope: a character (the pilgrim, as ROOT).
pilgrim_saint_tomb_open_trigger = {
	religion_group = old_world_group
	saint_no_saints_faith_trigger = no
	has_global_flag = saintly_burial_sites_exist
	any_province = { saint_pilgrim_tomb_trigger = yes }
}

# A holy order of the pilgrim's faith exists (pilgrim.27 - 29).
# Scope: the pilgrim.
pilgrim_holy_order_trigger = {
	any_independent_ruler = {
		holy_order = yes
		religion = ROOT
	}
}

# Sick enough for the healers of Shallya or Isha (pilgrim.40).
# Scope: the pilgrim.
pilgrim_needs_healing_trigger = {
	OR = {
		is_ill = yes
		trait = pneumonic
		trait = syphilitic
		trait = wounded
		has_disease_trigger = yes
		AND = {
			has_symptom_trigger = yes
			has_character_flag = developing_illness
		}
	}
}

# A dawi who carries shame to the Shrine of Grimnir (pilgrim.50).
# Scope: the pilgrim.
pilgrim_shamed_dawi_trigger = {
	religion_group = dwarf_gods_group
	OR = {
		trait = craven
		trait = depressed
		trait = wroth
	}
}

# May find a rune in a lost hold (pilgrim.25): once per lifetime, and only
# without one.
# Scope: the pilgrim.
pilgrim_can_find_rune_trigger = {
	is_adult = yes
	NOT = { has_character_flag = pilgrim_found_rune }
	NOT = { any_artifact = { has_artifact_flag = rune } }
}

# May bring home a shrine token (pilgrim.61): once per lifetime, only
# without one, and only from a shrine that gives them.
# Scope: the pilgrim.
pilgrim_can_get_token_trigger = {
	is_adult = yes
	NOT = { has_character_flag = pilgrim_got_token }
	NOT = { any_artifact = { has_artifact_flag = pilgrim_token } }
	NOR = {
		has_character_flag = pilgrim_to_gaen_vale
	}
}
