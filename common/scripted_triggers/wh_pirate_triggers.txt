# Someone worth a ransom whose kin would pay: a living man, Dawi or elf, and
# no vampire, hidden or not.
# Scope: any character.
pirate_ransom_victim_trigger = {
	OR = {
		trait = creature_human
		trait = creature_dwarf
		trait = creature_elf
	}
	NOR = {
		trait = vampire_strigoi_hidden
		trait = vampire_necrarch_hidden
		trait = vampire_strigoi_visible
		trait = vampire_necrarch_visible
	}
}

# A raider who sacks the holdings he storms (pillage.1 - pillage.2): pirate
# lords, the Druchii, the Ogres and the Norscans of the old gods. Chaos
# Norscans sack in the Chaos manner (chaoshorde.30 - 42).
# Scope: the winner of a siege.
pillage_raider_trigger = {
	OR = {
		government = pirate_government
		culture_group = druchii_group
		culture_group = ogre_group
		AND = {
			culture_group = norscan_group
			OR = {
				religion = norscan_gods
				NOT = { religion_group = chaos_gods_group }
			}
		}
	}
}

# The temple-cities of the Lizardmen whose plaques hold the Old Ones'
# knowledge (pillage.1).
# Scope: a province.
pillage_temple_city_trigger = {
	OR = {
		title = c_itza
		title = c_xlanhuapec
		title = c_tlaxtlan
		title = c_hexoatl
		title = c_tlaqua
	}
}
