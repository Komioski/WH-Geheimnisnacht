# The recruiter can still carry on: alive and still in the cult.
# Scope: the recruiter.
pleasure_active_member_trigger = {
	is_alive = yes
	society_member_of = cult_of_pleasure
}

# Someone the cult's knives can reach (pleasure.60).
# Scope: the target. ROOT is the grand master.
pleasure_knife_target_trigger = {
	is_alive = yes
	is_adult = yes
	is_offmap_ruler = no
	NOT = { character = ROOT }
	NOT = { society_member_of = cult_of_pleasure }
	is_within_diplo_range = ROOT
}

# A skilled councillor or commander worth killing to weaken a ruler
# marked for death (pleasure.60). Malekith is never sent for.
# Scope: the courtier. ROOT is the grand master.
pleasure_knife_servant_trigger = {
	pleasure_knife_target_trigger = yes
	NOT = { trait = malekith }
	OR = {
		AND = { has_job_title = job_chancellor diplomacy >= 12 }
		AND = { has_job_title = job_marshal martial >= 12 }
		AND = { has_job_title = job_treasurer stewardship >= 12 }
		AND = { has_job_title = job_spymaster intrigue >= 12 }
		AND = { has_job_title = job_spiritual learning >= 12 }
		AND = { has_minor_title = title_commander martial >= 12 }
	}
}
