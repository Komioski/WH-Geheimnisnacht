# A servant of the Ruinous Powers, for Praag's fall.
# Scope: a character.
praag_chaos_trigger = {
	religion_group = chaos_gods_group
}

# Praag lies under its curse right now.
# Scope: any.
praag_cursed_trigger = {
	182 = { has_province_modifier = praag_curse } # c_praag
}

# The ruler lives in Praag: its capital is the city.
# Scope: a ruler.
praag_lives_in_praag_trigger = {
	has_landed_title = c_praag
	capital_scope = { province_id = 182 }
}

# The crown of Kislev is weak: a regent, a ruler who cannot rule, or the End
# Times close at hand. The Z'ra heirs wait for this (praag.60).
# Scope: the ruler of Kislev.
praag_crown_weak_trigger = {
	OR = {
		has_regent = yes
		trait = incapable
		is_doom_high_trigger = yes
		is_doom_very_high_trigger = yes
	}
}

# The Praag vampire boyars may be found out: no hunt running, none in the
# last fifty years.
# Scope: a ruler.
praag_vampires_hidden_trigger = {
	NOT = { has_character_flag = praag_vamp_hunt }
	OR = {
		NOT = { has_global_flag = praag_vampires_found }
		had_global_flag = { flag = praag_vampires_found days = 18250 }
	}
}
