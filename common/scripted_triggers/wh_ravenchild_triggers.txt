# The Raven's Child, used by events/WHG_raven_child_events.txt.

# The march has a living sponsor. Scope: any.
ravenchild_has_sponsor_trigger = {
	event_target:childrens_crusade_leader = {
		persistent_event_target:childrens_crusade_sponsor_per = { is_alive = yes }
	}
}

# This character is the sponsor of the march. Scope: a character.
ravenchild_is_sponsor_trigger = {
	event_target:childrens_crusade_leader = {
		persistent_event_target:childrens_crusade_sponsor_per = { character = ROOT }
	}
}

# A young child at court who could run off with the march.
# Scope: a ruler.
ravenchild_has_spare_child_trigger = {
	any_child = {
		age = 8
		is_adult = no
		is_ruler = no
		is_primary_heir = no
		host = { character = ROOT }
	}
}

# The province lies in the land the march is crossing.
# Scope: a province; ROOT is the child (ravenchild.9).
ravenchild_march_region_trigger = {
	OR = {
		AND = {
			ROOT = { has_character_flag = flag_children_at_beginning }
			OR = {
				region = world_chaos_wastes_broken_lands
				region = world_chaos_wastes_eastern_steppes
				region = world_chaos_wastes_far_wastes
			}
		}
		AND = {
			ROOT = { has_character_flag = flag_children_in_far_wastes }
			region = world_chaos_wastes_far_wastes
		}
		AND = {
			ROOT = { has_character_flag = flag_children_in_eastern_steppes }
			region = world_chaos_wastes_eastern_steppes
		}
		AND = {
			ROOT = { has_character_flag = flag_children_in_broken_lands }
			region = world_chaos_wastes_broken_lands
		}
		AND = {
			ROOT = { has_character_flag = flag_children_in_zanbaijin }
			region = world_chaos_wastes_zanbaijin
		}
	}
}
