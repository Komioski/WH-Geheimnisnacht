# Reconquest triggers, used by events/WHG_reconquest_events.txt,
# decisions/WHG_reconquest_decisions.txt and the reconquista_* CBs.

# Estalia or Tilea. Scope: a province or title.
reconquest_land_trigger = {
	OR = { region = world_old_world_estalia region = world_old_world_tilea }
}

# One of the 13 lands a reconquest war can be fought for (each has its CB).
# Sartosa and Santa Clair are not among them. Scope: a province or title.
reconquest_target_land_trigger = {
	OR = {
		region = custom_astarios
		region = custom_cantonia
		region = custom_obregon
		region = custom_tigarre
		region = custom_castillona
		region = custom_tobaro
		region = custom_luccini
		region = custom_verezzo
		region = custom_remas
		region = custom_pavona
		region = custom_trantio
		region = custom_miragliano
		region = custom_monte_castello
	}
}

# A province of the 13 lands under a ruler outside the Old World faiths.
# Scope: a province.
reconquest_occupied_trigger = {
	reconquest_target_land_trigger = yes
	owner = {
		OR = {
			AND = { independent = yes NOT = { religion_group = old_world_group } }
			top_liege = { NOT = { religion_group = old_world_group } }
		}
	}
}

# A lord of the south: Estalian or Tilean, of an Old World faith.
# Scope: a character.
reconquest_southerner_trigger = {
	OR = { culture_group = estalian_group culture_group = tilean_group }
	religion_group = old_world_group
}

# The war is one of the reconquest wars. Scope: a war.
reconquest_cb_trigger = {
	OR = {
		using_cb = reconquista_astarios
		using_cb = reconquista_cantonia
		using_cb = reconquista_obregon
		using_cb = reconquista_tigarre
		using_cb = reconquista_castillona
		using_cb = reconquista_tobaro
		using_cb = reconquista_luccini
		using_cb = reconquista_verezzo
		using_cb = reconquista_remas
		using_cb = reconquista_pavona
		using_cb = reconquista_trantio
		using_cb = reconquista_miragliano
		using_cb = reconquista_monte_castello
	}
}

# Leads a reconquest war as attacker. Scope: a character.
reconquest_attacker_trigger = {
	any_war = { reconquest_cb_trigger = yes attacker = { character = ROOT } }
}

# Fights on the attacking side of a reconquest war. Scope: a character.
reconquest_any_attacker_trigger = {
	any_war = { reconquest_cb_trigger = yes any_attacker = { character = PREVPREV } }
}

# A lord who sells his sword in the reconquest wars: at peace, landed in
# the south, and more greedy than devout. Scope: a character.
reconquest_sellsword_trigger = {
	war = no
	independent = yes
	mercenary = no
	holy_order = no
	higher_real_tier_than = BARON
	any_realm_province = { reconquest_land_trigger = yes }
	OR = {
		ai = no
		AND = {
			NOR = {
				trait = zealous
				trait = holy_warrior
				has_crusade_trait_trigger = yes
				ai_zeal = 5
			}
			OR = { trait = envious trait = greedy trait = cynical trait = deceitful }
		}
	}
}

# Of the Reconqueror's blood: the oath costs half. Scope: a character.
reconquest_cheap_oath_trigger = { any_owned_bloodline = { has_bloodline_flag = cheap_reconquista } }
