# This character takes no part in courtly romance: the chaste in body or
# vow, the sick and the scarred, the dead of Nehekhara, and the races that
# breed without love (Skaven, greenskins, the Slann and their spawn).
# Scope: any character.
romance_barred_trigger = {
	OR = {
		trait = infirm
		trait = incapable
		trait = celibate
		trait = eunuch
		trait = in_hiding
		trait = creature_nehekharan
		religion_group = horned_rat_god_group
		religion_group = greenskin_gods_group
		religion = old_ones
	}
}

# A married or kept courtier who would stray: the lustful, the deceitful and
# the arbitrary always might; the rest only when their spouse or keeper
# does not love them.
# Scope: the would-be lover.
romance_would_stray_trigger = {
	OR = {
		is_married = no
		trait = lustful
		trait = deceitful
		trait = arbitrary
		spouse = { NOT = { reverse_opinion = { who = PREV value = 0 } } }
	}
	OR = {
		is_consort = no
		trait = lustful
		trait = deceitful
		trait = arbitrary
		consort = { NOT = { reverse_opinion = { who = PREV value = 0 } } }
	}
}

# This character is a poor catch: inbred, simple, a dwarf by birth or ugly.
# Scope: any character.
romance_poor_catch_trigger = {
	OR = {
		trait = inbred
		trait = imbecile
		trait = dwarf
		has_ugly_trait_trigger = yes
	}
}

# Love is worship here: Slaanesh, Atharti and the Cult of Pleasure.
# Scope: any character.
romance_pleasure_faith_trigger = {
	OR = {
		religion = slaanesh
		religion = atharti
		religion = cult_of_pleasure
		society_member_of = cult_of_pleasure
	}
}

# A courtier who might welcome the liege's attention: free of a lover and
# of a spouse who loves them, or careless of their vows.
# Scope: a courtier of ROOT. ROOT: the liege.
romance_liege_admirer_trigger = {
	is_pregnant = no
	prisoner = no
	has_lover = no
	is_consort = no
	romance_barred_trigger = no
	NOT = { trait = chaste }
	NOT = { trait = disfigured }
	OR = {
		is_woman_trigger = no
		NOT = { age = 40 }
	}
	NOT = { spouse = { character = ROOT } }
	NOT = { is_rival = ROOT }
	OR = {
		is_married = no
		trait = lustful
		trait = deceitful
		trait = hedonist
		spouse = {
			NOT = { is_lover = PREV }
			NOT = { reverse_opinion = { who = PREV value = 50 } }
		}
	}
	is_valid_romance_target = ROOT
}

# Another courtier of the same court who might take ROOT's advance: of the
# right sex for both, not close kin, friendly, not a former lover. A poor
# catch is only courted by another poor catch.
# Scope: a courtier of ROOT's liege. ROOT: the courtier making the advance.
romance_courtier_match_trigger = {
	NOT = { character = ROOT }
	is_ruler = no
	has_lover = no
	is_pregnant = no
	prisoner = no
	is_marriage_adult = yes
	is_ill = no
	romance_barred_trigger = no
	NOT = { trait = chaste }
	NOT = { has_character_modifier = lustful_affair_timer }
	NOT = { is_close_relative = ROOT }
	NOT = { spouse = { character = ROOT } }
	NOT = { is_former_lover = ROOT }
	OR = {
		AND = {
			NOT = { trait = homosexual }
			ROOT = { NOT = { trait = homosexual } }
			is_opposite_sex = ROOT
		}
		AND = {
			trait = homosexual
			ROOT = { trait = homosexual }
			same_sex = ROOT
		}
	}
	OR = {
		is_woman_trigger = no
		NOT = { age = 40 }
	}
	OR = {
		ROOT = { romance_poor_catch_trigger = yes }
		NOR = {
			trait = inbred
			trait = imbecile
			trait = dwarf
		}
	}
	opinion = { who = ROOT value = 0 }
	reverse_opinion = { who = ROOT value = 0 }
	romance_would_stray_trigger = yes
}
