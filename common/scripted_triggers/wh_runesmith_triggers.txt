# A dwarf lord who could commission work from ROOT (a Guild member): not in
# ROOT's society, a count with few artifacts, reachable.
# Scope: the candidate client.
runesmith_valid_client_trigger = {
	NOT = { same_society_as = ROOT }
	OR = {
		trait = creature_dwarf
		trait = creature_chaos_dwarf
	}
	num_of_artifacts <= 3
	ai = yes
	tier = COUNT
	is_inaccessible_or_incapable_trigger = no
	is_adult = yes
	is_within_diplo_range = ROOT
}

# ROOT has a client to work for: in the realm, or a neighbour's lord within
# reach of the capital. Scope: the member (ROOT).
runesmith_has_client_trigger = {
	OR = {
		any_realm_lord = { runesmith_valid_client_trigger = yes }
		any_neighbor_independent_ruler = {
			any_realm_lord = {
				runesmith_valid_client_trigger = yes
				ROOT = { capital_scope = { distance = { where = PREV distance < 1000 } } }
			}
		}
	}
}

# Lacks at least one of the four ancestor master runes. Scope: a character.
runesmith_lacks_an_ancestor_rune_trigger = {
	NAND = {
		has_artifact = master_rune_of_valaya
		has_artifact = master_rune_of_grungi
		has_artifact = master_rune_of_grimnir
		has_artifact = master_rune_of_thungi
	}
}

# A forge worth raiding. Scope: the forge's owner.
runesmith_rich_forge_trigger = {
	OR = {
		any_artifact = { has_artifact_flag = rune }
		wealth = 200
		has_character_modifier = runesmith_forge_4
		has_character_modifier = runesmith_forge_5
	}
}

# Has a rune a thief could carry off (not one of the special runes).
# Scope: a character.
runesmith_has_stealable_rune_trigger = {
	any_artifact = {
		has_artifact_flag = rune
		NOT = { has_artifact_flag = special_rune }
	}
}
