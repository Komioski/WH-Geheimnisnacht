# Venerated Souls (saints) triggers, used by events/WHG_saint_events.txt.

# A life worthy of being named Pious: devout, virtuous, untainted.
# Scope: the dead character (saint.1).
saint_worthy_trigger = {
	is_adult = yes
	death_in_cathay_trigger = no
	death_no_trace_trigger = no
	has_vice_trigger = no
	has_virtue_trigger = yes
	is_benevolent_trigger = yes
	is_pious_trigger = yes
	saint_no_saints_faith_trigger = no # Kislev and Bretonnia have no Venerated Souls
	NOR = {
		trait = excommunicated
		trait = drunkard
		trait = hedonist
		trait = cannibal_trait
		is_any_kinslayer_trigger = yes
		has_character_modifier = voice_of_greater_daemon
		has_character_modifier = known_satanist
		has_character_modifier = suspected_demon_spawn
		has_character_modifier = suspected_witch
		has_character_modifier = known_witch
		has_character_modifier = crazy_cannibal
		has_character_modifier = demon_child_non_pagan
		has_character_modifier = corrupting_magic_minor
		has_character_modifier = corrupting_magic_major
	}
}

# A landed priest of the dead one's faith who may name them Pious.
# Scope: a character in the dead one's realm (saint.1, saint.20).
saint_local_priest_trigger = {
	is_theocracy = yes
	is_landed = yes
	religion = event_target:saint_person
	NOT = { trait = bad_priest }
}

# Cults that sit in the Grand Conclave Magnus the Pious founded.
# Scope: a character.
saint_grand_conclave_trigger = {
	OR = {
		religion = sigmarite
		religion = ulrican
		religion = taalite
		religion = morrian
		religion = shallyan
		religion = manannite
		religion = myrmidian
	}
}

# Scope: a province.
saint_has_tomb_trigger = {
	OR = {
		has_province_modifier = tomb_of_a_saint
		has_province_modifier = pillaged_tomb_of_a_saint
	}
}

# Scope: a province.
saint_no_tomb_trigger = {
	NOR = {
		has_province_modifier = tomb_of_a_saint
		has_province_modifier = pillaged_tomb_of_a_saint
	}
}

# A tomb a pilgrim may visit: outside his own realm, of an Old World soul.
# Scope: a province; ROOT is the pilgrim (saint.60, SoA.5011).
saint_pilgrim_tomb_trigger = {
	saint_has_tomb_trigger = yes
	NOT = { owner = { same_realm = ROOT } }
	event_target:province_saint = { religion_group = old_world_group }
}

# Drink or gluttony, the sins of saint.20.
# Scope: a character.
saint_shares_sin_trigger = { OR = { trait = drunkard trait = hedonist } }

# Any of the vices a corrupt Grand Theogonist may share with a favourite.
# Scope: a character.
saint_has_any_vice_trigger = {
	OR = {
		trait = impaler
		trait = cannibal_trait
		trait = hedonist
		trait = drunkard
		trait = lunatic_1
		trait = lunatic_2
		trait = lunatic_3
		trait = seducer
		trait = seductress
		trait = lustful
	}
}

# The Grand Theogonist shares a vice with ROOT, the dead favourite.
# Scope: the Grand Theogonist (saint.30).
saint_gt_shares_vice_with_root_trigger = {
	OR = {
		AND = { trait = impaler ROOT = { trait = impaler } }
		AND = { trait = cannibal_trait ROOT = { trait = cannibal_trait } }
		AND = { trait = hedonist ROOT = { trait = hedonist } }
		AND = { trait = drunkard ROOT = { trait = drunkard } }
		AND = {
			OR = { trait = lunatic_1 trait = lunatic_2 trait = lunatic_3 }
			ROOT = { OR = { trait = lunatic_1 trait = lunatic_2 trait = lunatic_3 } }
		}
		AND = {
			OR = { trait = seducer trait = seductress }
			ROOT = { OR = { trait = seducer trait = seductress } }
		}
		AND = { trait = lustful ROOT = { trait = lustful } }
	}
}

# The Grand Theogonist (ROOT) still shares a vice with his dead favourite.
# Scope: the Grand Theogonist (saint.31).
saint_gt_shares_vice_with_saint_trigger = {
	event_target:saint_person = { saint_gt_shares_vice_reverse_trigger = yes }
}

# Helper for the above. Scope: the dead favourite; ROOT is the Grand
# Theogonist.
saint_gt_shares_vice_reverse_trigger = {
	OR = {
		AND = { trait = impaler ROOT = { trait = impaler } }
		AND = { trait = cannibal_trait ROOT = { trait = cannibal_trait } }
		AND = { trait = hedonist ROOT = { trait = hedonist } }
		AND = { trait = drunkard ROOT = { trait = drunkard } }
		AND = {
			OR = { trait = lunatic_1 trait = lunatic_2 trait = lunatic_3 }
			ROOT = { OR = { trait = lunatic_1 trait = lunatic_2 trait = lunatic_3 } }
		}
		AND = {
			OR = { trait = seducer trait = seductress }
			ROOT = { OR = { trait = seducer trait = seductress } }
		}
		AND = { trait = lustful ROOT = { trait = lustful } }
	}
}

# A zealous AI king or emperor who may march on a corrupt Grand Theogonist.
# Scope: a ruler; ROOT is the Grand Theogonist (saint.31).
saint_papal_attacker_trigger = {
	ai = yes
	independent = yes
	war = no
	religion = ROOT
	NOT = { trait = cynical }
	OR = { tier = KING tier = EMPEROR }
	is_within_diplo_range = ROOT
}

# The Venerated Soul had a virtue that ROOT lacks.
# Scope: the Venerated Soul; ROOT is the lord (saint.41).
saint_has_virtue_root_lacks_trigger = {
	OR = {
		AND = { trait = chaste ROOT = { NOT = { trait = chaste } } }
		AND = { trait = temperate ROOT = { NOT = { trait = temperate } } }
		AND = { trait = charitable ROOT = { NOT = { trait = charitable } } }
		AND = { trait = diligent ROOT = { NOT = { trait = diligent } } }
		AND = { trait = patient ROOT = { NOT = { trait = patient } } }
		AND = { trait = kind ROOT = { NOT = { trait = kind } } }
		AND = { trait = humble ROOT = { NOT = { trait = humble } } }
		AND = { trait = zealous ROOT = { NOT = { trait = zealous } } }
		AND = { trait = honest ROOT = { NOT = { trait = honest } } }
		AND = { trait = just ROOT = { NOT = { trait = just } } }
	}
}

# No Venerated Soul has yet been named in this faith.
# Scope: the dead character (saint.1).
saint_first_of_faith_trigger = {
	trigger_if = {
		limit = { religion = sigmarite }
		NOT = { has_global_flag = venerated_soul_sigmarite }
	}
	trigger_else_if = {
		limit = { religion = ulrican }
		NOT = { has_global_flag = venerated_soul_ulrican }
	}
	trigger_else_if = {
		limit = { religion = taalite }
		NOT = { has_global_flag = venerated_soul_taalite }
	}
	trigger_else_if = {
		limit = { religion = morrian }
		NOT = { has_global_flag = venerated_soul_morrian }
	}
	trigger_else_if = {
		limit = { religion = shallyan }
		NOT = { has_global_flag = venerated_soul_shallyan }
	}
	trigger_else_if = {
		limit = { religion = manannite }
		NOT = { has_global_flag = venerated_soul_manannite }
	}
	trigger_else_if = {
		limit = { religion = myrmidian }
		NOT = { has_global_flag = venerated_soul_myrmidian }
	}
	trigger_else_if = {
		limit = { religion = solkanite }
		NOT = { has_global_flag = venerated_soul_solkanite }
	}
	trigger_else_if = {
		limit = { OR = { religion = old_faith religion = old_faith_reformed } }
		NOT = { has_global_flag = venerated_soul_old_faith }
	}
	trigger_else = { always = no }
}

# The Imperial cults: the Grand Conclave less Myrmidia, plus Solkan.
# Their holy lines take the Imperial bloodline art. Scope: a character.
saint_imperial_cult_trigger = {
	OR = {
		religion = sigmarite
		religion = ulrican
		religion = taalite
		religion = morrian
		religion = shallyan
		religion = manannite
		religion = solkanite
	}
}

# Faiths with no Venerated Souls: Kislev and the Ungol gods, and Bretonnia's
# Lady of the Lake (the true Grail and the false one). Scope: a character.
saint_no_saints_faith_trigger = {
	OR = {
		religion = kislev_gods
		religion = northern_gods
		religion = northern_gods_reformed
		religion = grail
		religion = false_grail
	}
}

# The Old Faith of the druids. Scope: a character.
saint_old_faith_trigger = { OR = { religion = old_faith religion = old_faith_reformed } }
