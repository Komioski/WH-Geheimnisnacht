# The member may loot at all, and a free ruler within 250 of his realm, not
# his ally, liege, tributary or fellow member, holds an unoccupied province
# his ships can reach from a port with a shipyard (lodge.201).
# Scope: the member.
seaquest_pirate_can_raid_trigger = {
	is_landed = yes
	is_allowed_to_loot = yes
	OR = {
		independent = yes
		top_liege = { is_allowed_to_loot = yes }
	}
	any_independent_ruler = {
		NOT = { war_with = ROOT }
		NOR = {
			is_liege_of = ROOT
			has_non_aggression_pact_with = ROOT
			is_allied_with = ROOT
			AND = {
				is_tributary = yes
				suzerain = {
					OR = {
						character = ROOT
						any_liege = { character = ROOT }
					}
				}
			}
		}
		NOT = { distance_from_realm = { who = ROOT value = 250 } }
		any_realm_province = {
			is_occupied = no
			capital_holding = { NOT = { holding_type = nomad } }
			holder_scope = {
				NOT = { character = ROOT }
				NOT = { same_society_as = ROOT }
				top_liege = { NOT = { same_society_as = ROOT } }
			}
			ROOT = {
				any_demesne_province = {
					port = yes
					TECH_NAVAL = 1
					OR = {
						has_building = tb_shipyard_1
						has_building = ca_shipyard_1
						has_building = tp_shipyard_1
						has_building = ct_shipyard_1
					}
					can_naval_path_to = PREVPREV
				}
			}
		}
	}
}

# The same for the League, who also raid overland: a neighbouring province
# will do, or one within 250 for a seafaring people.
# Scope: the member.
seaquest_league_can_raid_trigger = {
	is_landed = yes
	is_allowed_to_loot = yes
	OR = {
		independent = yes
		top_liege = { is_allowed_to_loot = yes }
	}
	any_independent_ruler = {
		NOT = { war_with = ROOT }
		NOR = {
			is_liege_of = ROOT
			has_non_aggression_pact_with = ROOT
			is_allied_with = ROOT
			AND = {
				is_tributary = yes
				suzerain = {
					OR = {
						character = ROOT
						any_liege = { character = ROOT }
					}
				}
			}
		}
		OR = {
			AND = {
				ROOT = {
					OR = {
						religion = norscan_gods
						has_religion_feature = religion_seafaring
						is_seafarer = yes
					}
				}
				NOT = { distance_from_realm = { who = ROOT value = 250 } }
			}
			NOT = { distance_from_realm = { who = ROOT value = 50 } }
		}
		any_realm_province = {
			is_occupied = no
			capital_holding = { NOT = { holding_type = nomad } }
			holder_scope = {
				NOT = { character = ROOT }
				NOT = { same_society_as = ROOT }
				top_liege = { NOT = { same_society_as = ROOT } }
			}
			OR = {
				any_neighbor_province = { owner = { is_liege_or_above = ROOT } }
				any_neighbor_province = { owner = { character = ROOT } }
				ROOT = {
					any_demesne_province = {
						port = yes
						TECH_NAVAL = 1
						OR = {
							has_building = tb_shipyard_1
							has_building = ca_shipyard_1
							has_building = tp_shipyard_1
							has_building = ct_shipyard_1
						}
						can_naval_path_to = PREVPREV
					}
				}
			}
		}
	}
}

# A province within reach of event_target:home_capital: by sea within
# 1000, or overland within 1000.
# Scope: a province.
seaquest_in_reach_trigger = {
	OR = {
		AND = {
			is_coastal = yes
			can_naval_path_to = { target = event_target:home_capital distance < 1000 }
		}
		can_land_path_to = { target = event_target:home_capital distance < 1000 }
	}
}

# A free ruler of the member's own size or a little bigger (between half
# and 1.2 times his strength) holding a county within reach that he could
# take in a county conquest (lodge.227).
# Scope: the member. event_target:home_capital.
seaquest_conquest_target_trigger = {
	is_landed = yes
	any_independent_ruler = {
		any_realm_title = {
			tier = COUNT
			ROOT = {
				can_use_cb = {
					target = PREVPREV
					thirdparty_title = PREV
					casus_belli = pagan_county_conquest
				}
			}
			location = { seaquest_in_reach_trigger = yes }
		}
		holder_scope = {
			top_liege = {
				NOT = { has_truce = ROOT }
				NOT = { relative_power = { who = ROOT power = 1.2 } }
				relative_power = { who = ROOT power = 0.5 }
			}
		}
	}
}

# A county of the Tomb Kings within reach, whose king is no more than 1.2
# times the member's strength nor less than a fifth of it (the tomb raid,
# seaquest.46, with the exploration casus belli of cb_types_nehekharan).
# Scope: a county title. ROOT: the member. event_target:home_capital.
seaquest_tomb_title_trigger = {
	tier = COUNT
	location = {
		culture = nehekharan
		seaquest_in_reach_trigger = yes
	}
	holder_scope = {
		top_liege = {
			trait = creature_nehekharan
			NOT = { has_truce = ROOT }
			NOT = { relative_power = { who = ROOT power = 1.2 } }
			relative_power = { who = ROOT power = 0.2 }
		}
	}
}

# A free foreign ruler, not a pirate, of another faith and people, holding
# a port county in Lustria or the Southlands: a target for a colonial
# charter (seaquest.51).
# Scope: a ruler. ROOT: the member.
seaquest_charter_ruler_trigger = {
	NOR = {
		religion_group = ROOT
		culture_group = ROOT
		government = pirate_government
		government = freeboota_government
	}
	any_realm_title = {
		tier = COUNT
		location = {
			port = yes
			OR = {
				region = world_new_world_lustria
				region = world_greater_araby_southlands
			}
		}
	}
}

# The member has a child of twelve or more not yet in a society, for the
# family business (lodge.235). The AI only recruits its heir.
# Scope: the member.
seaquest_child_recruit_trigger = {
	is_playable = yes
	is_adult = yes
	any_child = {
		is_child_of = PREV
		is_in_society = no
		is_alive = yes
		age = 12
		prisoner = no
		NOT = { has_character_flag = flag_child_abandoned_warband }
		OR = {
			PREV = { ai = no }
			is_primary_heir = PREV
		}
	}
}

# The member may fight a friendly duel with an adult fellow member
# (lodge.220).
# Scope: the member.
seaquest_duel_partner_trigger = {
	age = 16
	society = {
		any_society_member = {
			ai = yes
			age = 16
			NOT = { character = ROOT }
			is_inaccessible_or_incapable_trigger = no
			NOT = { has_character_modifier = bedridden_illness }
			NOT = { has_character_modifier = severe_illness }
			NOT = { has_character_flag = call_to_legend_active }
		}
	}
}

# A player may duel to the death: a warrior of another lodge, a rival
# within 1000, or a commander of his own or a neighbour's realm
# (lodge.225).
# Scope: the member.
seaquest_deadly_duel_trigger = {
	ai = no
	age = 16
	OR = {
		any_society_member = {
			NOT = { same_society_as = ROOT }
			is_member_of_any_warrior_lodge_trigger = yes
			is_inaccessible_or_incapable_trigger = no
			NOT = { has_character_modifier = bedridden_illness }
			NOT = { has_character_modifier = severe_illness }
			NOT = { any_spouse = { character = ROOT } }
			is_adult = yes
			ai = yes
			is_within_diplo_range = ROOT
		}
		any_rival = {
			is_inaccessible_or_incapable_trigger = no
			NOT = { has_character_modifier = bedridden_illness }
			NOT = { has_character_modifier = severe_illness }
			is_adult = yes
			ai = yes
			ROOT = { capital_scope = { distance = { where = PREV distance < 1000 } } }
		}
		any_realm_character = {
			is_inaccessible_or_incapable_trigger = no
			NOT = { has_character_modifier = bedridden_illness }
			NOT = { has_character_modifier = severe_illness }
			NOT = { any_spouse = { character = ROOT } }
			ai = yes
			is_adult = yes
			has_minor_title = title_commander
		}
	}
}

# A naval patrol can reach the member: the member's capital is coastal and
# one of the patrol bases (Galleons Graveyard, the Kuldevind Islands,
# Sartosa, Mazur or Calumel) is within 1500 by sea (seaquest.21).
# Scope: the member. event_target:home_capital.
seaquest_admiral_reach_trigger = {
	is_landed = yes
	event_target:home_capital = { is_coastal = yes }
	OR = {
		2437 = { can_naval_path_to = { target = event_target:home_capital distance < 1500 } }
		1200 = { can_naval_path_to = { target = event_target:home_capital distance < 1500 } }
		626 = { can_naval_path_to = { target = event_target:home_capital distance < 1500 } }
		1072 = { can_naval_path_to = { target = event_target:home_capital distance < 1500 } }
		1574 = { can_naval_path_to = { target = event_target:home_capital distance < 1500 } }
	}
}

# The member is free for a mission.
# Scope: the member.
seaquest_free_member_trigger = {
	has_any_quest = no
	NOT = { has_character_modifier = quest_cooldown_timer }
	is_inaccessible_trigger = no
}

# Sail the far seas: a captain who can Set Out (rank 2) and is not at sea.
# Scope: the member.
seaquest_sea_voyage_trigger = {
	society_rank >= 2
	sea_coastal_trigger = yes
	NOT = { has_character_flag = sea_on_voyage }
	NOT = { has_character_modifier = sea_set_out_cd }
}

# Three prizes: a landed captain who can take prizes at sea.
# Scope: the member.
seaquest_sea_prizes_trigger = {
	is_landed = yes
	sea_coastal_trigger = yes
}

# A flagship of your own: a Wolf of the Sea of rank 3 who has none, or whose
# flagship can be refitted.
# Scope: the member.
seaquest_sea_flagship_trigger = {
	society_member_of = adventurers_pirates
	society_rank >= 3
	is_landed = yes
	NOT = { has_artifact = enormous_flagship }
	sea_refit_ready_trigger = yes
}
