# A Skaven breeder who can bear a litter now.
# Scope: the breeder.
skaven_ready_breeder_trigger = {
	is_female = yes
	is_pregnant = no
	trait = creature_skaven
	trait = skaven_breeder
	NOR = {
		trait = eunuch
		trait = transgender
		trait = infertile
		trait = incapable
	}
}

# A childless brother at the ruler's court who sires the litter in his place.
# Scope: the brother. ROOT: the ruler.
skaven_breeding_brother_trigger = {
	is_female = no
	NOT = { num_of_children = 1 }
	in_command = no
	at_location = ROOT
}
