# A Ropsmann who holds out against the Kislevite melting pot: they, or their
# lord, chose to stay apart (stanitsa.52), or a keeper of the tongue lives at
# their court (stanitsa.53).
# Scope: a character.
stanitsa_ropsmann_holds_out_trigger = {
	culture = ropsmann
	OR = {
		has_character_flag = stanitsa_ropsmann_apart
		liege = { has_character_flag = stanitsa_ropsmann_apart }
		has_character_flag = stanitsa_tongue_keeper
		host = { any_courtier = { has_character_flag = stanitsa_tongue_keeper } }
	}
}

# An Ungol or Ropsmann stanitsa that still keeps the northern gods: a target
# for a mission to bring it to Ursun, Dazh and Tor.
# Scope: a province.
stanitsa_mission_province_trigger = {
	religion = northern_gods
	OR = {
		culture = ungol
		culture = ropsmann
	}
	NOT = { has_province_modifier = stanitsa_mission_resented }
}

# A court has a keeper of the old Ropsmann tongue.
# Scope: a ruler.
stanitsa_has_keeper_trigger = {
	any_courtier = { has_character_flag = stanitsa_tongue_keeper }
}
