###############################################################################
# Sway and antagonize: the voices of the races
# Used by events/WHG_sway_events.txt, events/WHG_sway_events.txt and
# decisions/HF_sway_antagonize_decisions.txt.
#
# Every ruler speaks in one voice. The first match wins, so a vampire is
# undead, not court, and a Chaos-sworn Kurgan chieftain is warband.
#   daemon     daemons and daemon princes
#   undead     vampires, wights, Tomb Kings, necromancers' dead
#   greenskin  orcs, goblins, hobgoblins, snotlings
#   skaven     the Skaven
#   ogre       ogres
#   lizardmen  slann, saurus, skinks
#   warband    beastmen, Chaos Dwarfs, fimir, Chaos spawn and monsters, the
#              tribes of Norsca, the Kurgan and the Hung, and any open
#              worshipper of the Chaos gods
#   albion     the tribes of Albion (Truthsayers, standing stones, blood
#              feuds), unless they are Chaos-sworn or fimir
#   court      everyone else: the Empire, Bretonnia, Kislev, the south, Cathay,
#              elves, Dawi, halflings, Amazons
# Court and undead rulers keep a court, write letters and give feasts, so the
# courtly methods (letters, art, poems, visits, invitations) work between
# them. Every other pairing uses the voice methods (sway.299 - sway.303,
# antagonize.300 - antagonize.303).
###############################################################################

sway_voice_daemon_trigger = {
	OR = {
		is_daemon_trigger = yes
		culture_group = daemonic_group
	}
}

sway_voice_undead_trigger = {
	sway_voice_daemon_trigger = no
	OR = {
		is_undead_trigger = yes
		trait = creature_wight
		trait = creature_nehekharan
		culture_group = nehekharan_group
	}
}

sway_voice_greenskin_trigger = {
	sway_voice_daemon_trigger = no
	sway_voice_undead_trigger = no
	OR = {
		trait = creature_orc
		trait = creature_goblin
		trait = creature_hobgoblin
		trait = creature_snotling
		culture_group = orc_group
		culture_group = goblin_group
	}
}

sway_voice_skaven_trigger = {
	sway_voice_daemon_trigger = no
	sway_voice_undead_trigger = no
	OR = {
		trait = creature_skaven
		culture_group = skaven_group
	}
}

sway_voice_ogre_trigger = {
	sway_voice_daemon_trigger = no
	sway_voice_undead_trigger = no
	OR = {
		trait = creature_ogre
		culture_group = ogre_group
	}
}

sway_voice_lizardmen_trigger = {
	sway_voice_daemon_trigger = no
	sway_voice_undead_trigger = no
	OR = {
		trait = creature_slann
		trait = creature_saurus
		trait = creature_skink
		culture_group = lizardman_group
	}
}

sway_voice_warband_trigger = {
	sway_voice_daemon_trigger = no
	sway_voice_undead_trigger = no
	sway_voice_greenskin_trigger = no
	sway_voice_skaven_trigger = no
	sway_voice_ogre_trigger = no
	sway_voice_lizardmen_trigger = no
	OR = {
		trait = creature_beastman
		trait = creature_chaos_dwarf
		trait = creature_fimir
		trait = creature_chaosspawn
		trait = creature_monster
		culture_group = beastman_group
		culture_group = dwarf_chaos_group
		culture_group = kurgan_group
		culture_group = hung_group
		culture_group = norscan_group
		religion_group = chaos_gods_group
	}
}

sway_voice_albion_trigger = {
	sway_voice_daemon_trigger = no
	sway_voice_undead_trigger = no
	sway_voice_greenskin_trigger = no
	sway_voice_skaven_trigger = no
	sway_voice_ogre_trigger = no
	sway_voice_lizardmen_trigger = no
	sway_voice_warband_trigger = no
	culture_group = albion_group
}

sway_voice_court_trigger = {
	sway_voice_daemon_trigger = no
	sway_voice_undead_trigger = no
	sway_voice_greenskin_trigger = no
	sway_voice_skaven_trigger = no
	sway_voice_ogre_trigger = no
	sway_voice_lizardmen_trigger = no
	sway_voice_warband_trigger = no
	sway_voice_albion_trigger = no
}

# Keeps a court: letters, art, poems, visits and invitations make sense.
sway_voice_courtly_trigger = {
	OR = {
		sway_voice_court_trigger = yes
		sway_voice_undead_trigger = yes
	}
}

# Both this character and ROOT keep a court. Symmetric.
sway_courtly_pair_trigger = {
	sway_voice_courtly_trigger = yes
	ROOT = { sway_voice_courtly_trigger = yes }
}

# This character and ROOT speak in the same voice. Symmetric.
sway_same_voice_trigger = {
	OR = {
		AND = { sway_voice_daemon_trigger = yes ROOT = { sway_voice_daemon_trigger = yes } }
		AND = { sway_voice_undead_trigger = yes ROOT = { sway_voice_undead_trigger = yes } }
		AND = { sway_voice_greenskin_trigger = yes ROOT = { sway_voice_greenskin_trigger = yes } }
		AND = { sway_voice_skaven_trigger = yes ROOT = { sway_voice_skaven_trigger = yes } }
		AND = { sway_voice_ogre_trigger = yes ROOT = { sway_voice_ogre_trigger = yes } }
		AND = { sway_voice_lizardmen_trigger = yes ROOT = { sway_voice_lizardmen_trigger = yes } }
		AND = { sway_voice_warband_trigger = yes ROOT = { sway_voice_warband_trigger = yes } }
		AND = { sway_voice_albion_trigger = yes ROOT = { sway_voice_albion_trigger = yes } }
		AND = { sway_voice_court_trigger = yes ROOT = { sway_voice_court_trigger = yes } }
	}
}

# Pairs who will never be swayed into liking each other: the Dawi and the
# greenskins, the Lizardmen and the Skaven or the daemons. Antagonize stays
# open to them. Symmetric between this character and ROOT.
sway_pair_blocked_trigger = {
	OR = {
		AND = {
			OR = { trait = creature_dwarf culture_group = dwarf_group }
			ROOT = { sway_voice_greenskin_trigger = yes }
		}
		AND = {
			sway_voice_greenskin_trigger = yes
			ROOT = { OR = { trait = creature_dwarf culture_group = dwarf_group } }
		}
		AND = {
			sway_voice_lizardmen_trigger = yes
			ROOT = { OR = { sway_voice_skaven_trigger = yes sway_voice_daemon_trigger = yes } }
		}
		AND = {
			OR = { sway_voice_skaven_trigger = yes sway_voice_daemon_trigger = yes }
			ROOT = { sway_voice_lizardmen_trigger = yes }
		}
	}
}
