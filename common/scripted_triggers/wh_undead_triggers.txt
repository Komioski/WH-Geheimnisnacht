###############################################################################
# Shared undead triggers
# Used by events/WHG_undead_events.txt (namespace undead) and
# decisions/wh_undead_decisions.txt.
###############################################################################

# A master of the dead who can hold a wight's leash: a vampire or a
# necromancer.
# Scope: a character.
undead_master_trigger = {
	OR = {
		is_vampire_trigger = yes
		trait = lore_necromancy
	}
}

# A wight still held by a master of the dead: his employer, or his
# employer's liege, is a vampire or a necromancer.
# Scope: the wight.
undead_wight_held_trigger = {
	employer = {
		OR = {
			undead_master_trigger = yes
			liege = { undead_master_trigger = yes }
		}
	}
}

# A weapon the Dark Enchantment can work on: a fine sword, lance, spear or
# axe that is not necromantic already. Legendary weapons are left alone.
# Scope: an artifact.
undead_enchantable_weapon_trigger = {
	has_artifact_flag = combat_weapon
	OR = {
		has_artifact_flag = sword
		has_artifact_flag = lance
		has_artifact_flag = spear
		has_artifact_flag = axe
	}
	NOT = { has_artifact_flag = necromantic_item }
	OR = {
		quality == 2
		quality == 3
	}
}

# Armour the Dark Enchantment can work on: fine armour that is not
# necromantic already. Legendary armour is left alone.
# Scope: an artifact.
undead_enchantable_armour_trigger = {
	has_artifact_flag = armor
	NOT = { has_artifact_flag = necromantic_item }
	OR = {
		quality == 2
		quality == 3
	}
}

# A master of the dead who can work the Dark Enchantment himself: a
# necromancer of great power, or the Red Duke.
# Scope: a character.
undead_self_enchanter_trigger = {
	OR = {
		trait = special_char_red_duke
		AND = {
			trait = lore_necromancy
			OR = {
				has_character_modifier = magic_power_3
				has_character_modifier = magic_power_4
			}
		}
	}
}
