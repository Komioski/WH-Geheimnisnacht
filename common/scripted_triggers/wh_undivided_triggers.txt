# An AI nomad warlord of the Wastes strong enough to take the mantle of the
# Everchosen (undivided.1). Must already be a horde, or the everchosen
# invasion war cannot be declared.
# Scope: character.
undivided_ai_everchosen_candidate_trigger = {
	ai = yes
	religion = chaos
	is_nomadic = yes
	OR = {
		culture_group = kurgan_group
		culture_group = hung_group
	}
	location = { region = world_chaos_wastes }
	demesne_size = 20
}

# A Wastes kingdom capital the new Everchosen can invade: it has an owner and
# the Everchosen is not already its top liege (undivided.1).
# Scope: province. Needs event_target:everchosen.
undivided_invasion_target_trigger = {
	has_owner = yes
	owner = {
		top_liege = {
			NOT = { character = event_target:everchosen }
		}
	}
}

###############################################################################
# The path of the Everchosen (undivided.30-79)
###############################################################################

# The Winds of Chaos blow strongly enough for a new Everchosen to rise. The
# warhammer_everchosen_winds game rule sets the bar: very high (default) or
# high.
# Scope: any.
undivided_winds_allow_everchosen_trigger = {
	trigger_if = {
		limit = { has_game_rule = { name = warhammer_everchosen_winds value = high } }
		OR = {
			is_doom_high_trigger = yes
			is_doom_very_high_trigger = yes
		}
	}
	trigger_else = {
		is_doom_very_high_trigger = yes
	}
}

# The seeker walks the extended quest: the extended game rule and a player.
# The AI and the default rule take the pilgrimage instead.
# Scope: the seeker.
undivided_extended_quest_trigger = {
	has_game_rule = { name = warhammer_everchosen_quest value = extended }
	ai = no
}

# Someone whose blood can quiet U'zuhl, the daemon in the Slayer of Kings:
# a lord, a commander or a marshal in the seeker's service.
# Scope: character. ROOT is the seeker.
undivided_royal_blood_trigger = {
	is_adult = yes
	prisoner = no
	is_abroad = no
	NOT = { character = ROOT }
	is_invincible_trigger = no
	OR = {
		is_ruler = yes
		has_minor_title = title_commander
		has_job_title = job_marshal
	}
}

# A companion who marched with the seeker: an adult fighter of the court.
# Scope: character. ROOT is the seeker.
undivided_companion_trigger = {
	is_adult = yes
	prisoner = no
	is_abroad = no
	is_ruler = no
	NOT = { character = ROOT }
	is_invincible_trigger = no
	martial = 8
}
