# An AI nomad warlord of the Wastes strong enough to take the mantle of the
# Everchosen (undivided.1). Must already be a horde, or the everchosen
# invasion war cannot be declared.
# Scope: character.
undivided_ai_everchosen_candidate_trigger = {
	ai = yes
	religion = chaos
	is_nomadic = yes
	OR = {
		culture_group = kurgan_group
		culture_group = hung_group
	}
	location = { region = world_chaos_wastes }
	demesne_size = 20
}

# A Wastes kingdom capital the new Everchosen can invade: it has an owner and
# the Everchosen is not already its top liege (undivided.1).
# Scope: province. Needs event_target:everchosen.
undivided_invasion_target_trigger = {
	has_owner = yes
	owner = {
		top_liege = {
			NOT = { character = event_target:everchosen }
		}
	}
}

###############################################################################
# The path of the Everchosen (undivided.30-79)
###############################################################################

# The Winds of Chaos blow strongly enough for a seeker to try for the mantle:
# high or very high doom. The quest is cheaper and its guardians weaker at
# very high doom (seektobecomeeverchosen, undivided_winds_temper_guardian_effect,
# undivided_necropolis_host_effect).
# Scope: any.
undivided_quest_doom_trigger = {
	OR = {
		is_doom_high_trigger = yes
		is_doom_very_high_trigger = yes
	}
}

# The seeker walks the extended quest: the extended game rule and a player.
# The AI and the default rule take the pilgrimage instead.
# Scope: the seeker.
undivided_extended_quest_trigger = {
	has_game_rule = { name = warhammer_everchosen_quest value = extended }
	ai = no
}

# Quest locations. Each is one place to change if the map moves.
# Scope: a character (checks where they are).

# The Altar of Ultimate Darkness: Ansar Har, northern Naggaroth.
undivided_at_altar_trigger = {
	location = { province_id = 1364 }
}
# The coast where the seeker's warband seizes the ships: Tullich or Wryn on
# the Gulf of Naggarond.
undivided_at_ship_coast_trigger = {
	location = {
		OR = {
			province_id = 1344 # Tullich
			province_id = 1363 # Wryn
		}
	}
}
# The sunless necropolis of the Armour of Morkar: the isle of Ildelver in the
# Sea of Chaos.
undivided_at_necropolis_trigger = {
	location = { province_id = 1199 }
}
# The Cliff of Beasts, lair of Flamefang: the northern coast of Norsca.
undivided_at_beast_cliff_trigger = {
	location = {
		OR = {
			province_id = 93 # Ergren's Peninsula
			province_id = 94 # Drergen Mort
			province_id = 95 # Forest of Frykt
		}
	}
}
# The Chimera Plateau, where Krakanrok sleeps: the hobgoblin lands on the
# edge of the Khuuitan Shelf.
undivided_at_chimera_plateau_trigger = {
	location = {
		OR = {
			province_id = 2442 # Khutalai
			province_id = 2438 # Avagrom
			province_id = 2443 # Dalekland
		}
	}
}
# Zanbaijin, on the Plateau of K'Datha, where the Everchosen is crowned.
undivided_at_zanbaijin_trigger = {
	location = {
		OR = {
			province_id = 405  # Zanbaijin
			province_id = 395  # Kathuul
			province_id = 404  # Sarog
			province_id = 1244 # Azkaroth
		}
	}
}

# Someone who stands with the seeker on the Chimera Plateau: a courtier,
# vassal or commander of theirs who is there with the army.
# Scope: character. ROOT is the seeker.
undivided_blood_companion_trigger = {
	is_adult = yes
	prisoner = no
	NOT = { character = ROOT }
	is_invincible_trigger = no
	undivided_at_chimera_plateau_trigger = yes
}

# A companion of royal blood: landed, noble-born, or the seeker's own kin.
# One such death is enough to quench the Slayer of Kings.
# Scope: character. ROOT is the seeker.
undivided_royal_blood_trigger = {
	undivided_blood_companion_trigger = yes
	OR = {
		is_landed = yes
		is_lowborn = no
		is_close_relative = ROOT
		dynasty = ROOT
	}
}
