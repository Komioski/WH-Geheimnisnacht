# An AI nomad warlord of the Wastes strong enough to take the mantle of the
# Everchosen (undivided.1). Must already be a horde, or the everchosen
# invasion war cannot be declared.
# Scope: character.
undivided_ai_everchosen_candidate_trigger = {
	ai = yes
	religion = chaos
	is_nomadic = yes
	OR = {
		culture_group = kurgan_group
		culture_group = hung_group
	}
	location = { region = world_chaos_wastes }
	demesne_size = 20
}

# A Wastes kingdom capital the new Everchosen can invade: it has an owner and
# the Everchosen is not already its top liege (undivided.1).
# Scope: province. Needs event_target:everchosen.
undivided_invasion_target_trigger = {
	has_owner = yes
	owner = {
		top_liege = {
			NOT = { character = event_target:everchosen }
		}
	}
}
