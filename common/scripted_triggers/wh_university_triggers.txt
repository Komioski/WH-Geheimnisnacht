# A university city is open to students: its university has been founded,
# and its lord is a man, and no servant of Chaos or of the dead. Scope: the city's province.
university_city_open_trigger = {
	# Salzenmund, Marienburg and Sudenburg have no university until their lord
	# founds one
	OR = { NOT = { province_id = 239 } has_global_flag = university_founded_salzenmund }
	OR = { NOT = { province_id = 222 } has_global_flag = university_founded_marienburg }
	OR = { NOT = { province_id = 983 } has_global_flag = university_founded_sudenburg }
	owner = {
		trait = creature_human
		NOT = { religion_group = chaos_gods_group }
		NOT = { religion_group = necromantic_group }
	}
}

# A parent who could send a child to a university: a settled human ruler of
# the Old World or Araby, outside Bretonnia's Grail faiths. Scope: the parent.
university_parent_trigger = {
	trait = creature_human
	NOT = { culture_group = halfling_old_world } # The Moot sends no students
	is_alive = yes
	is_ruler = yes
	prisoner = no
	NOT = { trait = incapable }
	OR = { religion_group = old_world_group religion_group = ormazd_gods_group }
	NOR = {
		religion = grail
		religion = false_grail
		is_tribal = yes
		is_nomadic = yes
		government = chaos_horde_government
		government = chaos_host_government
		government = norscan_government
		government = norscan_government_feudal
	}
}

# A young noble who might ask to go to a university (university.1): an
# unlanded human of the Old World or Araby at a parent's court, not yet
# learned. Araby's universities take no women.
# Scope: the young noble.
university_candidate_trigger = {
	is_landed = no
	in_command = no
	trait = creature_human
	NOT = { culture_group = halfling_old_world } # The Moot sends no students
	OR = { religion_group = old_world_group religion_group = ormazd_gods_group }
	NOR = {
		religion = grail
		religion = false_grail
		trait = academic
		trait = arabyan_scholar
		has_character_flag = university_wants
		has_character_flag = university_denied
		has_character_flag = university_student
	}
	OR = {
		is_female = no
		AND = {
			NOR = { religion = ormazd religion = kitabid }
			top_liege = { NOR = { religion = ormazd religion = kitabid } }
		}
	}
	OR = { location = { region = world_old_world } location = { region = world_greater_araby } }
	NOT = { location = { region = world_old_world_bretonnia } }
	OR = {
		father = {
			university_parent_trigger = yes
			is_liege_of = ROOT
			ROOT = { host = { character = PREVPREV } }
		}
		mother = {
			university_parent_trigger = yes
			is_liege_of = ROOT
			ROOT = { host = { character = PREVPREV } }
		}
	}
}

# A student who could learn Araby's sorcery at Martek (university.18): an
# Arabyan of learning 15 with magic in the blood but no power yet, studying
# in Araby. Scope: the student.
university_araby_magic_trigger = {
	learning = 15
	OR = { religion = ormazd religion = kitabid }
	location = { OR = { religion = ormazd religion = kitabid } }
	OR = {
		trait = magic_potential_1
		trait = magic_potential_2
		trait = magic_potential_3
		trait = magic_potential_4
		has_character_flag = magic_potential_flag_1
		has_character_flag = magic_potential_flag_2
		has_character_flag = magic_potential_flag_3
		has_character_flag = magic_potential_flag_4
	}
	NOR = {
		has_character_modifier = magic_power_1
		has_character_modifier = magic_power_2
		has_character_modifier = magic_power_3
		has_character_modifier = magic_power_4
	}
}
