# A knight riding in ROOT's Blood Quest.
# Scope: a character; ROOT is the quest leader.
vamp_bquest_band_trigger = {
	trait = blood_questing
	trait = vampire_blood_visible
	has_opinion_modifier = { who = ROOT modifier = opinion_blood_questing }
}

# A knight of ROOT's band who rides into the current fight.
# Scope: a character; ROOT is the quest leader.
vamp_bquest_fighter_trigger = { vamp_bquest_band_trigger = yes has_character_flag = bquest_joined }

# A vampire who could ride on a Blood Quest: not a prisoner, sick, worn out
# or already questing.
# Scope: a character.
vamp_bquest_recruit_trigger = {
	trait = vampire_blood_visible
	prisoner = no
	NOR = { trait = incapable trait = resting trait = blood_questing }
}

# A Lahmian free to go on a mission: unlanded, not on a mission, not a
# prisoner or incapacitated.
# Scope: a character.
vamp_lahmia_free_agent_trigger = {
	trait = vampire_lahmian_visible
	is_landed = no
	prisoner = no
	is_incapacitated_trigger = no
	NOT = { trait = on_lahmian_mission }
}

# The seductress's mark still lives.
# Scope: the seductress (ROOT), with the persistent target lahmia_mark.
vamp_lahmia_mark_alive_trigger = { event_target:lahmia_mark = { is_alive = yes } }

# A young mortal woman a recruiter could bring into the Sisterhood: beautiful,
# clever and unlanded.
# Scope: a character.
vamp_lahmia_recruit_mark_trigger = {
	is_female = yes
	is_landed = no
	prisoner = no
	trait = creature_human
	is_vampire_trigger = no
	age = 16
	NOT = { age = 25 }
	intrigue = 10
	OR = { trait = fair trait = pretty trait = beautiful }
}

# A living vampire sire (not Mahtmasi) whom ROOT, their get, might come to
# hate (vamp.520).
# Scope: ROOT's father or mother.
vamp_resented_sire_trigger = {
	is_alive = yes
	OR = {
		trait = vampire_carstein_visible
		trait = vampire_blood_visible
		trait = vampire_strigoi_visible
		trait = vampire_necrarch_visible
		trait = vampire_jade_visible
	}
	NOT = { is_rival = ROOT }
	NOT = { is_lover = ROOT }
	NOT = { is_friend = ROOT }
}

# A living Mahtmasi sire whom ROOT, their get, might come to love (vamp.523).
# Scope: ROOT's father or mother.
vamp_mahtmasi_sire_trigger = {
	is_alive = yes
	trait = vampire_mahtmasi_visible
	NOT = { is_rival = ROOT }
	NOT = { is_lover = ROOT }
	NOT = { is_friend = ROOT }
}

# A character of the Old World cultures (Empire, Kislev, Estalia, Tilea,
# Bretonnia, the Dwarfs), who hear of the Vampire Wars first (vamp.601,
# vamp.609, vamp.21).
vamp_old_world_culture_trigger = {
	OR = {
		culture_group = imperial_group
		culture_group = kislevite_group
		culture_group = estalian_group
		culture_group = tilean_group
		culture_group = bretonnian_group
		culture_group = dwarf_group
	}
}

# A vampire of the realm with a claim on the Carstein Ring holder's crown
# (vamp.615).
# Scope: a realm character; event_target:vamp_vlad_crown the crown.
vamp_crown_claimant_trigger = {
	is_vampire_trigger = yes
	NOT = { character = ROOT }
	has_claim = event_target:vamp_vlad_crown
}

# A vampire pirate. Pirates stay in the pirate society, never a bloodline one.
# Scope: a character.
vamp_pirate_trigger = {
	OR = {
		has_character_flag = vampire_pirate
		trait = pirate
		government = pirate_government
		society_member_of = adventurers_pirates
	}
}

# One trigger per bloodline: an open vampire of that line, or a hidden one for
# the two lines that can hide (Lahmians and von Carsteins).
# Scope: a character.
vamp_carstein_trigger = {
	OR = {
		trait = vampire_carstein_visible
		trait = vampire_carstein_hidden
	}
}
vamp_blood_trigger = { trait = vampire_blood_visible }
vamp_lahmian_trigger = {
	OR = {
		trait = vampire_lahmian_visible
		trait = vampire_lahmian_hidden
	}
}
vamp_necrarch_trigger = { trait = vampire_necrarch_visible }
vamp_strigoi_trigger = { trait = vampire_strigoi_visible }
vamp_jade_trigger = { trait = vampire_jade_visible }
vamp_mahtmasi_trigger = { trait = vampire_mahtmasi_visible }

# A hidden vampire (only Lahmians and von Carsteins can hide).
# Scope: a character.
vamp_hidden_trigger = {
	OR = {
		trait = vampire_lahmian_hidden
		trait = vampire_carstein_hidden
	}
}

# A member of any of the seven bloodline societies.
# Scope: a character.
vamp_society_member_trigger = {
	OR = {
		society_member_of = carstein_court
		society_member_of = ordo_draconis
		society_member_of = lahmian_sisterhood
		society_member_of = necrarch_brotherhood
		society_member_of = strigoi_brood
		society_member_of = jade_coil
		society_member_of = carrion_children
	}
}

# A vampire who belongs in a bloodline society but is not in it yet.
# Scope: a character.
vamp_needs_society_trigger = {
	is_vampire_trigger = yes
	vamp_society_member_trigger = no
	vamp_pirate_trigger = no
}

# A living thrall: a human of a necromantic faith with no vampire trait.
# Scope: a character.
vamp_thrall_trigger = {
	trait = creature_human
	is_vampire_trigger = no
	OR = {
		religion = vampiric
		religion = nagashi
		religion = shadowmen
	}
	vamp_pirate_trigger = no
}

# Who may join each society as a living thrall (ranks 1-2 only).
# Scope: a character.
vamp_carstein_thrall_trigger = {
	vamp_thrall_trigger = yes
	religion = vampiric
	OR = {
		liege = { vamp_carstein_trigger = yes }
		top_liege = { vamp_carstein_trigger = yes }
		culture = sylvanian
	}
}
vamp_lahmian_thrall_trigger = {
	vamp_thrall_trigger = yes
	is_female = yes
	religion = vampiric
	OR = {
		liege = { vamp_lahmian_trigger = yes }
		any_spouse = { vamp_lahmian_trigger = yes }
		any_lover = { vamp_lahmian_trigger = yes }
	}
}
vamp_necrarch_thrall_trigger = {
	vamp_thrall_trigger = yes
	OR = {
		religion = nagashi
		trait = lore_necromancy
	}
}
vamp_strigoi_thrall_trigger = {
	vamp_thrall_trigger = yes
	culture_group = strigany_group
}
vamp_jade_thrall_trigger = {
	vamp_thrall_trigger = yes
	culture = living_jade
}
vamp_mahtmasi_thrall_trigger = {
	vamp_thrall_trigger = yes
	religion = shadowmen
	culture = bayaji
}

# The deed a rank 1 member needs to rise to rank 2. Living thralls and most
# lines need a deed done in play (flag vamp_deed_done: feeding on worthy prey,
# a Mahtmasi answering a plea, a thrall's service, a Blood Dragon's duel);
# some lines accept standing instead.
# Scope: a society member.
vampsoc_deed_trigger = {
	OR = {
		has_character_flag = vamp_deed_done
		AND = {
			society_member_of = carstein_court
			is_vampire_trigger = yes
			is_ruler = yes
		}
		AND = {
			society_member_of = lahmian_sisterhood
			is_vampire_trigger = yes
			OR = {
				intrigue = 12
				any_lover = { always = yes }
			}
		}
		AND = {
			society_member_of = necrarch_brotherhood
			is_vampire_trigger = yes
			learning = 14
		}
		AND = {
			society_member_of = jade_coil
			is_vampire_trigger = yes
			learning = 12
		}
	}
}

			society_member_of = carstein_court

			society_rank == 1

		}

			society_member_of = lahmian_sisterhood
