# Warband and warrior lodge triggers, used by events/WHG_warrior_society_events.txt.

# The societies that can sound the Call to Glory: the warbands and the
# Warriors of Ulric (Gork and Mork have no Call to Glory decision).
# Scope: a character.
warband_glory_society_trigger = {
	OR = { is_member_of_any_warband_trigger = yes society_member_of = warriors_of_ulric }
}

# The Call to Glory still holds: at war, in the society at rank 4, under
# the cap of 20 reinforcements. Scope: the caller.
warband_glory_ok_trigger = {
	has_character_modifier = call_to_glory
	war = yes
	warband_glory_society_trigger = yes
	society_rank = 4
	NOT = { check_variable = { which = "call_to_glory_variable" value = 20 } }
}

# The Call to Glory has lapsed: peace, or no longer of rank 4 in a
# warband or the Warriors of Ulric. Scope: the caller.
warband_glory_lapsed_trigger = {
	OR = {
		war = no
		warband_glory_society_trigger = no
		NOT = { society_rank = 4 }
		NOT = { has_character_modifier = call_to_glory }
	}
}

# The caller leads an army on land. Scope: the caller.
warband_glory_leading_trigger = { in_command = yes is_at_sea = no }

warband_voice_skaven_trigger = { society_member_of = skaven_warlords_society }

# Any of the Chaos warbands (not the Skaven Warlords). Scope: a character.

		society_member_of = warband_of_chaos
		society_member_of = warband_of_khorne
		society_member_of = warband_of_nurgle

	}
}

warband_voice_greenskin_trigger = {
	OR = { society_member_of = warband_gork society_member_of = warband_mork }
}

# No vassal could take the post of commander, so a courtier may.
# Scope: the ruler.
warband_commander_no_vassal_trigger = {
	NOT = {
		any_vassal = {
			higher_real_tier_than = BARON
			NOT = { primary_title = { temporary = yes } }
			OR = {
				AND = {
					OR = {
						male_can_hold_minor_title_trigger = yes
						female_can_hold_minor_commander_title_trigger = yes
					}
					is_adult = yes
				}
				has_character_flag = special_marshal
			}
			prisoner = no
			NOT = { trait = incapable }
			is_inaccessible_trigger = no
			NOT = { has_character_flag = guru }
			NOT = { has_character_modifier = refusing_to_lead }
			NOR = {
				has_job_title = job_chancellor
				has_job_title = job_marshal
				has_job_title = job_treasurer
				has_job_title = job_spymaster
				has_job_title = job_spiritual
				has_minor_title = title_commander
			}
		}
	}
}

# Every commander post of the realm is taken. Scope: the ruler.
warband_commander_slots_full_trigger = {
	OR = {
		AND = {
			tier = COUNT
			any_realm_character = {
				count = 2
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = DUKE
			any_realm_character = {
				count = 4
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = KING
			is_nomadic = no
			any_realm_character = {
				count = 6
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = EMPEROR
			is_nomadic = no
			any_realm_character = {
				count = 8
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = KING
			is_nomadic = yes
			any_realm_character = {
				count = 2
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = EMPEROR
			is_nomadic = yes
			any_realm_character = {
				count = 4
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
	}
}

# A weak commander (martial below 10) who could make way. Scope: the ruler.

# A member of a warband or warrior lodge. Scope: the character.
waraid_member_trigger = {
	OR = { is_member_of_any_warband_trigger = yes is_member_of_any_warrior_lodge_trigger = yes }
}

# A player who could be asked to join the war between ROOT (defender) and
# FROM (attacker): free, in range, no vassal or ally of either side, and
# not asked in the last five years. Scope: the player.
waraid_can_join_trigger = {
	is_ruler = yes
	is_adult = yes
	war = no
	prisoner = no
	NOT = { trait = incapable }
	NOT = { character = ROOT }
	NOT = { character = FROM }
	OR = {
		NOT = { has_character_flag = waraid_asked }
		had_character_flag = { flag = waraid_asked days = 1825 }
	}
	is_within_diplo_range = ROOT
	is_within_diplo_range = FROM
	NOR = {
		vassal_of = ROOT
		vassal_of = FROM
		is_allied_with = ROOT
		is_allied_with = FROM
		has_non_aggression_pact_with = ROOT
		has_non_aggression_pact_with = FROM
	}
	NAND = {
		ROOT = { any_liege = { character = PREVPREV } }
		FROM = { any_liege = { character = PREVPREV } }
	}
	NAND = { is_tributary = yes suzerain = { OR = { character = ROOT character = FROM } } }
}

# The celebration contest (waraid.14): what helps most in each society's
# contest, and what helps a little. Scope: the player.
waraid_contest_strong_trigger = {
	OR = {
		AND = {
			NOR = { warband_voice_skaven_trigger = yes warband_voice_ulric_trigger = yes }
			is_strong_trigger = yes
		}
		AND = { warband_voice_skaven_trigger = yes intrigue = 14 }
		AND = { warband_voice_ulric_trigger = yes trait = hunter }
	}
}

waraid_contest_skilled_trigger = {
	OR = {
		AND = { NOT = { warband_voice_skaven_trigger = yes } martial = 14 }
		AND = { warband_voice_skaven_trigger = yes trait = paranoid }
		AND = { warband_voice_dawi_trigger = yes trait = drunkard }
	}
}

# An AI character who may be recruited into a warband that is empty or
# short of members. Scope: the character.

	OR = { has_empty_warband_trigger = yes has_warband_in_need_of_members_trigger = yes }
	forbidden_from_joining_warband_trigger = no
	is_in_society = no
	has_secret_religion_trait_trigger = no
	is_adult = yes
	NOT = { block_general_event_trigger = yes }
}

# A character who could take up the banner of a fallen warband
# (event_target:warband_to_revive): of its god's faith, a greenskin for
# Gork and Mork, a Skaven for the Warlords. Scope: the character.
warband_revive_candidate_trigger = {
	ai = yes
	is_in_society = no
	age = 16
	has_secret_religion_trait_trigger = no
	forbidden_from_joining_warband_trigger = no
	NOT = { block_general_event_trigger = yes }
	OR = {
		has_religion_matching_warband_to_revive_trigger = yes
		AND = {
			religion_group = greenskin_gods_group
			event_target:warband_to_revive = {
				OR = { is_society = warband_gork is_society = warband_mork }
			}
		}
		AND = {
			culture_group = skaven_group
			event_target:warband_to_revive = { is_society = skaven_warlords_society }
		}
	}
}

	is_in_society = no

			religion_group = dwarf_gods_group
		}
	}
}
