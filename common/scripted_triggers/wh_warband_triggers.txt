# Warband and warrior lodge triggers, used by events/WHG_warband_events.txt,
# WHG_war_aid_events.txt and WHG_warrior_lodge_events.txt.

# The Call to Glory still holds: at war, in the warband at rank 4, under
# the cap of 20 reinforcements. Scope: the caller.
warband_glory_ok_trigger = {
	has_character_modifier = call_to_glory
	war = yes
	is_member_of_any_warband_trigger = yes
	society_rank = 4
	NOT = { check_variable = { which = "call_to_glory_variable" value = 20 } }
}

# The Call to Glory has lapsed: peace, or no longer of rank 4 in a
# warband. Scope: the caller.
warband_glory_lapsed_trigger = {
	OR = {
		war = no
		is_member_of_any_warband_trigger = no
		NOT = { society_rank = 4 }
		NOT = { has_character_modifier = call_to_glory }
	}
}

# The caller leads an army on land. Scope: the caller.
warband_glory_leading_trigger = { in_command = yes is_at_sea = no }

# The voice of a warband. Scope: a character.
warband_voice_chaos_trigger = { society_member_of = warband_of_chaos }
warband_voice_khorne_trigger = { society_member_of = warband_of_khorne }
warband_voice_nurgle_trigger = { society_member_of = warband_of_nurgle }
warband_voice_tzeentch_trigger = { society_member_of = warband_of_tzeentch }
warband_voice_slaanesh_trigger = { society_member_of = warband_of_slaanesh }
warband_voice_skaven_trigger = { society_member_of = skaven_warlords_society }

# Any of the Chaos warbands (not the Skaven Warlords). Scope: a character.
warband_voice_any_chaos_trigger = {
	OR = {
		society_member_of = warband_of_chaos
		society_member_of = warband_of_khorne
		society_member_of = warband_of_nurgle
		society_member_of = warband_of_tzeentch
		society_member_of = warband_of_slaanesh
	}
}

warband_voice_greenskin_trigger = {
	OR = { society_member_of = warband_gork society_member_of = warband_mork }
}

# No vassal could take the post of commander, so a courtier may.
# Scope: the ruler.
warband_commander_no_vassal_trigger = {
	NOT = {
		any_vassal = {
			higher_real_tier_than = BARON
			NOT = { primary_title = { temporary = yes } }
			OR = {
				AND = {
					OR = {
						male_can_hold_minor_title_trigger = yes
						female_can_hold_minor_commander_title_trigger = yes
					}
					is_adult = yes
				}
				has_character_flag = special_marshal
			}
			prisoner = no
			NOT = { trait = incapable }
			is_inaccessible_trigger = no
			NOT = { has_character_flag = guru }
			NOT = { has_character_modifier = refusing_to_lead }
			NOR = {
				has_job_title = job_chancellor
				has_job_title = job_marshal
				has_job_title = job_treasurer
				has_job_title = job_spymaster
				has_job_title = job_spiritual
				has_minor_title = title_commander
			}
		}
	}
}

# Every commander post of the realm is taken. Scope: the ruler.
warband_commander_slots_full_trigger = {
	OR = {
		AND = {
			tier = COUNT
			any_realm_character = {
				count = 2
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = DUKE
			any_realm_character = {
				count = 4
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = KING
			is_nomadic = no
			any_realm_character = {
				count = 6
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = EMPEROR
			is_nomadic = no
			any_realm_character = {
				count = 8
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = KING
			is_nomadic = yes
			any_realm_character = {
				count = 2
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = EMPEROR
			is_nomadic = yes
			any_realm_character = {
				count = 4
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
	}
}

# A weak commander (martial below 10) who could make way. Scope: the ruler.
warband_commander_weak_trigger = {
	any_realm_character = {
		NOT = { higher_tier_than = BARON }
		liege = { character = ROOT }
		has_minor_title = title_commander
		NOT = { martial = 10 }
	}
}

# The warrior lodges speak in their own voices too.
warband_voice_ulric_trigger = { society_member_of = warriors_of_ulric }
warband_voice_knight_trigger = { society_member_of = knights_of_old_world }
warband_voice_dawi_trigger = { society_member_of = warriors_guild_dawi }

# A member of a warband or warrior lodge. Scope: the character.
waraid_member_trigger = {
	OR = {
		is_member_of_any_warband_trigger = yes
		is_member_of_any_warrior_lodge_trigger = yes
	}
}

# A player who could be asked to join the war between ROOT (defender) and
# FROM (attacker): free, in range, no vassal or ally of either side, and
# not asked in the last five years. Scope: the player.
waraid_can_join_trigger = {
	is_ruler = yes
	is_adult = yes
	war = no
	prisoner = no
	NOT = { trait = incapable }
	NOT = { character = ROOT }
	NOT = { character = FROM }
	OR = {
		NOT = { has_character_flag = waraid_asked }
		had_character_flag = { flag = waraid_asked days = 1825 }
	}
	is_within_diplo_range = ROOT
	is_within_diplo_range = FROM
	NOR = {
		vassal_of = ROOT
		vassal_of = FROM
		is_allied_with = ROOT
		is_allied_with = FROM
		has_non_aggression_pact_with = ROOT
		has_non_aggression_pact_with = FROM
	}
	NAND = {
		ROOT = { any_liege = { character = PREVPREV } }
		FROM = { any_liege = { character = PREVPREV } }
	}
	NAND = {
		is_tributary = yes
		suzerain = {
			OR = {
				character = ROOT
				character = FROM
			}
		}
	}
}

# The celebration contest (waraid.14): what helps most in each society's
# contest, and what helps a little. Scope: the player.
waraid_contest_strong_trigger = {
	OR = {
		AND = {
			NOR = {
				warband_voice_skaven_trigger = yes
				warband_voice_ulric_trigger = yes
			}
			is_strong_trigger = yes
		}
		AND = {
			warband_voice_skaven_trigger = yes
			intrigue = 14
		}
		AND = {
			warband_voice_ulric_trigger = yes
			trait = hunter
		}
	}
}

waraid_contest_skilled_trigger = {
	OR = {
		AND = {
			NOT = { warband_voice_skaven_trigger = yes }
			martial = 14
		}
		AND = {
			warband_voice_skaven_trigger = yes
			trait = paranoid
		}
		AND = {
			warband_voice_dawi_trigger = yes
			trait = drunkard
		}
	}
}

# An AI character who may be recruited into a warband that is empty or
# short of members. Scope: the character.
warband_recruit_trigger = {
	OR = {
		has_empty_warband_trigger = yes
		has_warband_in_need_of_members_trigger = yes
	}
	forbidden_from_joining_warband_trigger = no
	is_in_society = no
	has_secret_religion_trait_trigger = no
	is_adult = yes
	NOT = { block_general_event_trigger = yes }
}

# A character who could take up the banner of a fallen warband
# (event_target:warband_to_revive): of its god's faith, a greenskin for
# Gork and Mork, a Skaven for the Warlords. Scope: the character.
warband_revive_candidate_trigger = {
	ai = yes
	is_in_society = no
	age = 16
	has_secret_religion_trait_trigger = no
	forbidden_from_joining_warband_trigger = no
	NOT = { block_general_event_trigger = yes }
	OR = {
		has_religion_matching_warband_to_revive_trigger = yes
		AND = {
			religion_group = greenskin_gods_group
			event_target:warband_to_revive = {
				OR = {
					is_society = warband_gork
					is_society = warband_mork
				}
			}
		}
		AND = {
			culture_group = skaven_group
			event_target:warband_to_revive = { is_society = skaven_warlords_society }
		}
	}
}

# An AI character free to join a warrior lodge of their faith that is empty
# or has fewer than 30 members (Kislevites: the Circle of the Bear).
# Scope: the character.
lodge_recruit_candidate_trigger = {
	ai = yes
	is_adult = yes
	is_in_society = no
	has_secret_religion_trait_trigger = no
	NOT = { block_general_event_trigger = yes }
	OR = {
		has_religion_matching_empty_warrior_lodge_trigger = yes
		has_religion_matching_warrior_lodge_in_need_of_members_trigger = yes
		AND = {
			religion = kislev_gods
			circle_of_the_bear = { num_of_society_members < 30 }
		}
	}
}

# An AI character who could refound the lodge in event target
# warrior_lodge_to_revive. Scope: the character.
lodge_revive_candidate_trigger = {
	ai = yes
	age = 16
	is_in_society = no
	has_secret_religion_trait_trigger = no
	NOT = { block_general_event_trigger = yes }
	OR = {
		AND = {
			event_target:warrior_lodge_to_revive = { is_society = warriors_of_ulric }
			OR = {
				religion = ulrican
				trait = norscan_patron_ulric
				special_join_warrior_lodge_trigger = yes
			}
		}
		AND = {
			event_target:warrior_lodge_to_revive = { is_society = knights_of_old_world }
			religion_group = old_world_group
		}
		AND = {
			event_target:warrior_lodge_to_revive = { is_society = warriors_guild_dawi }
			religion_group = dwarf_gods_group
		}
	}
}

# An artifact a beaten Hero may yield to the winner of a seat duel
# (lodge.62): an ordinary one the winner (ROOT) has none of. Unique, legendary
# (quality 4+), crown, society, holy, cursed and personal items are kept.
# Scope: artifact.
lodge_hero_spoils_artifact_trigger = {
	NOT = { quality = 4 }
	NOT = { artifact_type_owned_by = ROOT }
	NOR = {
		has_artifact_flag = unique
		has_artifact_flag = runefang
		has_artifact_flag = crown
		has_artifact_flag = crown_jewel
		has_artifact_flag = society
		has_artifact_flag = holy_order
		has_artifact_flag = relic
		has_artifact_flag = religious
		has_artifact_flag = cursed
		has_artifact_flag = cursed_item
		has_artifact_flag = no_inherit
		has_artifact_flag = ingredient
		has_artifact_flag = alive
	}
}
