# Warband triggers, used by events/WHG_warband_events.txt.

# The Call to Glory still holds: at war, in the warband at rank 4, under
# the cap of 20 reinforcements. Scope: the caller.
warband_glory_ok_trigger = {
	has_character_modifier = call_to_glory
	war = yes
	is_member_of_any_warband_trigger = yes
	society_rank = 4
	NOT = { check_variable = { which = "call_to_glory_variable" value = 20 } }
}

# The Call to Glory has lapsed: peace, or no longer of rank 4 in a
# warband. Scope: the caller.
warband_glory_lapsed_trigger = {
	OR = {
		war = no
		is_member_of_any_warband_trigger = no
		NOT = { society_rank = 4 }
		NOT = { has_character_modifier = call_to_glory }
	}
}

# The caller leads an army on land. Scope: the caller.
warband_glory_leading_trigger = { in_command = yes is_at_sea = no }

# The voice of a warband. Scope: a character.
warband_voice_chaos_trigger = { society_member_of = warband_of_chaos }
warband_voice_khorne_trigger = { society_member_of = warband_of_khorne }
warband_voice_nurgle_trigger = { society_member_of = warband_of_nurgle }
warband_voice_tzeentch_trigger = { society_member_of = warband_of_tzeentch }
warband_voice_slaanesh_trigger = { society_member_of = warband_of_slaanesh }
warband_voice_skaven_trigger = { society_member_of = skaven_warlords_society }

# Any of the Chaos warbands (not the Skaven Warlords). Scope: a character.
warband_voice_any_chaos_trigger = {
	OR = {
		society_member_of = warband_of_chaos
		society_member_of = warband_of_khorne
		society_member_of = warband_of_nurgle
		society_member_of = warband_of_tzeentch
		society_member_of = warband_of_slaanesh
	}
}

warband_voice_greenskin_trigger = {
	OR = { society_member_of = warband_gork society_member_of = warband_mork }
}

# No vassal could take the post of commander, so a courtier may.
# Scope: the ruler.
warband_commander_no_vassal_trigger = {
	NOT = {
		any_vassal = {
			higher_real_tier_than = BARON
			NOT = { primary_title = { temporary = yes } }
			OR = {
				AND = {
					OR = {
						male_can_hold_minor_title_trigger = yes
						female_can_hold_minor_commander_title_trigger = yes
					}
					is_adult = yes
				}
				has_character_flag = special_marshal
			}
			prisoner = no
			NOT = { trait = incapable }
			is_inaccessible_trigger = no
			NOT = { has_character_flag = guru }
			NOT = { has_character_modifier = refusing_to_lead }
			NOR = {
				has_job_title = job_chancellor
				has_job_title = job_marshal
				has_job_title = job_treasurer
				has_job_title = job_spymaster
				has_job_title = job_spiritual
				has_minor_title = title_commander
			}
		}
	}
}

# Every commander post of the realm is taken. Scope: the ruler.
warband_commander_slots_full_trigger = {
	OR = {
		AND = {
			tier = COUNT
			any_realm_character = {
				count = 2
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = DUKE
			any_realm_character = {
				count = 4
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = KING
			is_nomadic = no
			any_realm_character = {
				count = 6
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = EMPEROR
			is_nomadic = no
			any_realm_character = {
				count = 8
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = KING
			is_nomadic = yes
			any_realm_character = {
				count = 2
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
		AND = {
			tier = EMPEROR
			is_nomadic = yes
			any_realm_character = {
				count = 4
				liege = { character = ROOT }
				has_minor_title = title_commander
			}
		}
	}
}

# A weak commander (martial below 10) who could make way. Scope: the ruler.
warband_commander_weak_trigger = {
	any_realm_character = {
		NOT = { higher_tier_than = BARON }
		liege = { character = ROOT }
		has_minor_title = title_commander
		NOT = { martial = 10 }
	}
}
