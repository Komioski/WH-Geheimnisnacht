# Triggers for wyrdstone (events/WHG_warpstone_events.txt, namespace
# warpstone)

# A courtier fit to lead a Mordheim expedition: unlanded, unwed, at home,
# not the heir, not the patron's close kin, and not just back from a hunt.
# Scope: a courtier. ROOT: the patron.
warpstone_mordheim_agent_trigger = {
	is_landed = no
	is_married = no
	prisoner = no
	is_adult = yes
	is_incapable = no
	NOT = { trait = travelling }
	NOT = { has_character_flag = recently_returned }
	employer = { character = ROOT }
	is_heir = no
	NOT = { is_close_relative = ROOT }
}

# One of ROOT's Mordheim captains: a tributary of ROOT who led an
# expedition.
# Scope: an independent ruler. ROOT: the patron.
warpstone_mordheim_captain_trigger = {
	OR = {
		is_tributary = { type = mordheim suzerain = ROOT }
		is_tributary = { type = permanent suzerain = ROOT }
	}
	OR = {
		has_character_flag = martial_agent_mordheim
		has_character_flag = mercenary_agent_mordheim
		has_character_flag = intrigue_agent_mordheim
	}
}

# Those who go looking for wyrdstone: the Dark Gods' faithful, the
# Skaven, the Chaos Dwarfs, the greenskins and dabblers in dark magic.
# Scope: a ruler.
warpstone_seeker_trigger = {
	OR = {
		religion_group = horned_rat_god_group
		religion_group = dwarf_chaos_group
		religion_group = chaos_gods_group
		religion_group = greenskin_gods_group
		trait = lore_necromancy
		trait = lore_dark_magic
	}
}

# Those who leave it in the ground: the elves, the Lizardmen, the Dawi and
# the pious of the Old World.
# Scope: a ruler.
warpstone_shunner_trigger = {
	OR = {
		religion_group = cadai_group
		religion_group = old_ones_group
		religion_group = dwarf_gods_group
		AND = {
			religion_group = old_world_group
			NOR = {
				trait = lore_necromancy
				trait = lore_dark_magic
			}
		}
	}
}

# A province where a wyrdstone vein may be struck: the mountains and the
# deep places, with no vein yet.
# Scope: a province.
warpstone_vein_site_trigger = {
	NOR = {
		has_province_flag = warpstone_vein
		has_province_flag = warpstone_vein_dm
		has_province_modifier = warpstone_vein
	}
	OR = {
		region = world_karaz_ankor
		region = world_underempire
		region = world_drak_ankor
		region = world_norsca_central
		region = world_troll_country
		terrain = mountain
	}
}

# A province where a meteor of wyrdstone may fall: not one already struck,
# and never Lustria, Nehekhara, Araby, Ulthuan or the Under-Empire.
# Scope: a province.
warpstone_meteor_site_trigger = {
	is_land = yes
	NOT = { has_province_flag = warpstone_meteor }
	NOR = {
		region = world_new_world_lustria
		region = world_nehekhara
		region = world_greater_araby
		region = world_ulthuan
		region = world_underempire
	}
}

# A wyrdstone mine with stone left in it.
# Scope: a province.
warpstone_mine_trigger = {
	OR = {
		has_province_modifier = warpstone_vein
		has_province_modifier = warpstone_meteor
	}
	capital_holding = {
		OR = {
			has_building = ca_blackstone_mine_1
			has_building = ct_blackstone_mine_1
			has_building = tp_blackstone_mine_1
			has_building = tb_blackstone_mine_1
			has_building = no_blackstone_mine_1
			has_building = ca_blackstone_mine_2
			has_building = ct_blackstone_mine_2
			has_building = tp_blackstone_mine_2
		}
	}
}

# The mine has been improved.
# Scope: a province.
warpstone_improved_mine_trigger = {
	capital_holding = {
		OR = {
			has_building = ca_blackstone_mine_2
			has_building = ct_blackstone_mine_2
			has_building = tp_blackstone_mine_2
		}
	}
}

# The character holds wyrdstone the shadow market will buy.
# Scope: a character.
warpstone_holds_stone_trigger = {
	any_artifact = {
		OR = {
			artifact_type = warpstone_shard
			artifact_type = warpstone_dust
			artifact_type = warpstone_cluster
		}
	}
}

# The character can use wyrdstone: a trained mage or a Skaven (the same test
# as the warpstone artifacts' active trigger).
# Scope: a character.
warpstone_user_trigger = {
	OR = {
		is_trained_mage_trigger = yes
		trait = creature_skaven
	}
}

# The character's store of wyrdstone is full: five pieces or more. Further
# yields from the mines are sold to the shadow market (warpstone.27).
# Scope: a character.
warpstone_stock_full_trigger = {
	any_artifact = {
		count = 5
		OR = {
			artifact_type = warpstone_shard
			artifact_type = warpstone_dust
			artifact_type = warpstone_cluster
		}
	}
}

# Those who send warbands into Mordheim: every people of the Old World and
# beyond except the Asur and Asrai, who will not touch the stone, and the
# Lizardmen and the Tomb Kings, who are too far away and care nothing for
# it. The Dawi go as treasure hunters, the Druchii as raiders, the Possessed
# to worship at the Pit.
# Scope: the patron.
warpstone_expedition_people_trigger = {
	NOR = {
		religion_group = cadai_group
		religion_group = old_ones_group
		trait = creature_nehekharan
	}
}

# An AI patron close enough to Mordheim to care: in the Empire, Kislev or
# the Border Princes, or a Skaven of the Under-Empire. Keeps the title
# decision cheap for the AI.
# Scope: the patron.
warpstone_expedition_ai_reach_trigger = {
	OR = {
		trait = creature_skaven
		capital_scope = {
			OR = {
				region = world_old_world_empire
				region = world_old_world_kislev
				region = world_old_world_border_princes
			}
		}
	}
}

# The patron pays for the expedition in the dark gifts of undeath (piety).
# Scope: the patron.
warpstone_expedition_pays_piety_trigger = {
	is_vampire_trigger = yes
}

# The patron pays in prestige: greenskin bosses, beastmen and tribal chiefs
# lead their warbands with promises of glory, not coin.
# Scope: the patron.
warpstone_expedition_pays_prestige_trigger = {
	warpstone_expedition_pays_piety_trigger = no
	OR = {
		trait = creature_orc
		trait = creature_goblin
		trait = creature_beastman
		is_tribal = yes
	}
}

# The patron can pay for an expedition: 500 piety for a vampire, 750
# prestige for a boss or chief, 150 gold for a Skaven (the clans have ratmen
# to spare), 300 gold for anyone else (200 for the AI).
# Scope: the patron.
warpstone_expedition_can_pay_trigger = {
	OR = {
		AND = {
			warpstone_expedition_pays_piety_trigger = yes
			piety = 500
		}
		AND = {
			warpstone_expedition_pays_prestige_trigger = yes
			prestige = 750
		}
		AND = {
			warpstone_expedition_pays_piety_trigger = no
			warpstone_expedition_pays_prestige_trigger = no
			trait = creature_skaven
			wealth = 150
		}
		AND = {
			warpstone_expedition_pays_piety_trigger = no
			warpstone_expedition_pays_prestige_trigger = no
			NOT = { trait = creature_skaven }
			ai = yes
			wealth = 200
		}
		AND = {
			warpstone_expedition_pays_piety_trigger = no
			warpstone_expedition_pays_prestige_trigger = no
			NOT = { trait = creature_skaven }
			ai = no
			wealth = 300
		}
	}
}

# Skaven gnaw at wyrdstone mines that are not their own (warpstone.35),
# unless the mine paid them off in the last ten years.
# Scope: the mine's province.
warpstone_ratmen_target_trigger = {
	owner = { NOT = { trait = creature_skaven } }
	OR = {
		NOT = { has_province_flag = warpstone_ratmen_paid }
		had_province_flag = { flag = warpstone_ratmen_paid days = 3650 }
	}
}

# A patron whose people send a warband of their own to Mordheim rather than
# hiring the free companies (warpstone.4): Skaven, the dead, Kislevites,
# Dawi, the Possessed of Chaos, greenskins, beastmen and Druchii.
# Scope: the patron.
warpstone_own_warband_trigger = {
	OR = {
		trait = creature_skaven
		trait = creature_dwarf
		trait = creature_orc
		trait = creature_goblin
		trait = creature_beastman
		is_vampire_trigger = yes
		religion_group = necromantic_group
		religion_group = chaos_gods_group
		religion_group = cytharai_group
		AND = {
			culture_group = kislevite_group
			trait = creature_human
		}
	}
}

# The year's mine yield goes to the shadow market for gold (warpstone.27):
# the store is full, or an AI owner cannot use the stone.
# Scope: the mine's owner.
warpstone_sells_yield_trigger = {
	OR = {
		warpstone_stock_full_trigger = yes
		AND = {
			ai = yes
			warpstone_user_trigger = no
		}
	}
}

# An AI that has no business buying stone on the shadow market: it already
# hunts for it in Mordheim (an expedition out, a patron of a warband, or a
# county in the City of the Damned), mines its own, or its store is full.
# Players buy as they please. Scope: a character.
warpstone_ai_skips_market_trigger = {
	ai = yes
	OR = {
		has_character_flag = maintaining_agents
		has_character_flag = mordheim_patron
		any_demesne_province = { region = world_old_world_empire_mordheim }
		any_demesne_province = { warpstone_mine_trigger = yes }
		warpstone_stock_full_trigger = yes
	}
}
