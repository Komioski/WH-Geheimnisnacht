# A courtier fit to lead a Mordheim expedition: unlanded, unwed, at home,
# not the heir, not the patron's close kin, and not just back from a hunt.
# Scope: a courtier. ROOT: the patron.
warpstone_mordheim_agent_trigger = {
	is_landed = no
	is_married = no
	prisoner = no
	is_adult = yes
	is_incapable = no
	NOT = { trait = travelling }
	NOT = { has_character_flag = recently_returned }
	employer = { character = ROOT }
	is_heir = no
	NOT = { is_close_relative = ROOT }
}

# One of ROOT's Mordheim captains: a tributary of ROOT who led an
# expedition.
# Scope: an independent ruler. ROOT: the patron.
warpstone_mordheim_captain_trigger = {
	OR = {
		is_tributary = { type = mordheim suzerain = ROOT }
		is_tributary = { type = permanent suzerain = ROOT }
	}
	OR = {
		has_character_flag = martial_agent_mordheim
		has_character_flag = mercenary_agent_mordheim
		has_character_flag = intrigue_agent_mordheim
	}
}

# Those who go looking for wyrdstone: the Dark Gods' faithful, the
# Skaven, the Chaos Dwarfs, the greenskins and dabblers in dark magic.
# Scope: a ruler.
warpstone_seeker_trigger = {
	OR = {
		religion_group = horned_rat_god_group
		religion_group = dwarf_chaos_group
		religion_group = chaos_gods_group
		religion_group = greenskin_gods_group
		trait = lore_necromancy
		trait = lore_dark_magic
	}
}

# Those who leave it in the ground: the elves, the Lizardmen, the Dawi and
# the pious of the Old World.
# Scope: a ruler.
warpstone_shunner_trigger = {
	OR = {
		religion_group = cadai_group
		religion_group = old_ones_group
		religion_group = dwarf_gods_group
		AND = {
			religion_group = old_world_group
			NOR = {
				trait = lore_necromancy
				trait = lore_dark_magic
			}
		}
	}
}

# A province where a wyrdstone vein may be struck: the mountains and the
# deep places, with no vein yet.
# Scope: a province.
warpstone_vein_site_trigger = {
	NOR = {
		has_province_flag = warpstone_vein
		has_province_flag = warpstone_vein_dm
		has_province_modifier = warpstone_vein
	}
	OR = {
		region = world_karaz_ankor
		region = world_underempire
		region = world_drak_ankor
		region = world_norsca_central
		region = world_troll_country
		terrain = mountain
	}
}

# A province where a meteor of wyrdstone may fall: not one already struck,
# and never Lustria, Nehekhara, Araby, Ulthuan or the Under-Empire.
# Scope: a province.
warpstone_meteor_site_trigger = {
	is_land = yes
	NOT = { has_province_flag = warpstone_meteor }
	NOR = {
		region = world_new_world_lustria
		region = world_nehekhara
		region = world_greater_araby
		region = world_ulthuan
		region = world_underempire
	}
}

# A wyrdstone mine with stone left in it.
# Scope: a province.
warpstone_mine_trigger = {
	OR = {
		has_province_modifier = warpstone_vein
		has_province_modifier = warpstone_meteor
	}
	capital_holding = {
		OR = {
			has_building = ca_blackstone_mine_1
			has_building = ct_blackstone_mine_1
			has_building = tp_blackstone_mine_1
			has_building = tb_blackstone_mine_1
			has_building = no_blackstone_mine_1
			has_building = ca_blackstone_mine_2
			has_building = ct_blackstone_mine_2
			has_building = tp_blackstone_mine_2
		}
	}
}

# The mine has been improved.
# Scope: a province.
warpstone_improved_mine_trigger = {
	capital_holding = {
		OR = {
			has_building = ca_blackstone_mine_2
			has_building = ct_blackstone_mine_2
			has_building = tp_blackstone_mine_2
		}
	}
}

# The character holds wyrdstone the shadow market will buy.
# Scope: a character.
warpstone_holds_stone_trigger = {
	any_artifact = {
		OR = {
			artifact_type = warpstone_shard
			artifact_type = warpstone_dust
			artifact_type = warpstone_cluster
		}
	}
}
