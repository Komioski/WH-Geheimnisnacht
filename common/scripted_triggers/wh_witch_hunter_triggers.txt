# A province with a Chaos taint a witch hunter can cleanse.
# Scope: the province.
whunter_tainted_province_trigger = {
	OR = {
		has_province_modifier = nurgle_province_disease
		has_province_modifier = beastmen_unrest
		has_province_modifier = warpdust_traces
		has_province_modifier = slaanesh_hedonism
		has_province_modifier = warpstone_meteor
	}
}

# A human Sigmarite witch hunter, 16 or older, fit to inherit a chapter: not
# incapable, sick, mad, possessed, mutated or touched by magic.
# Scope: a character.
whunter_heir_candidate_trigger = {
	religion = sigmarite
	trait = witch_hunter
	trait = creature_human
	age = 16
	NOR = {
		trait = incapable
		trait = maimed
		trait = leper
		trait = ill
		trait = has_nurgles_rot
		trait = blinded
		trait = possessed
		trait = lunatic_3
		trait = mutation_minor
		trait = mutation_major
		trait = mutation_massive
		has_character_modifier = magic_power_1
		has_character_modifier = magic_power_2
		has_character_modifier = magic_power_3
		has_character_modifier = magic_power_4
		trait = magic_potential_1
		trait = magic_potential_2
		trait = magic_potential_3
		trait = magic_potential_4
	}
}

# A free adult courtier under suspicion of Chaos worship who has not been
# accused lately.
# Scope: a character.
whunter_suspect_courtier_trigger = {
	prisoner = no
	age = 16
	NOT = { has_character_modifier = wh_accused }
	OR = {
		has_character_modifier = under_suspicion
		has_character_modifier = highly_suspect
		has_character_modifier = known_satanist
	}
}

# A mage (or someone with magic potential) the local magic laws say should
# burn, who is not hiding, incapable, a Breton damsel or a daemon.
# Scope: a character.
whunter_mage_suspect_trigger = {
	NOR = {
		trait = in_hiding
		trait = incapable
		trait = BretonDamsel
		is_daemon_race_trigger = yes
	}
	check_if_mage_should_be_burned_trigger = yes
	OR = {
		has_character_modifier = magic_power_1
		has_character_modifier = magic_power_2
		has_character_modifier = magic_power_3
		has_character_modifier = magic_power_4
		magic_potential_trigger = yes
	}
}

# Anyone a witch hunter mission can go after: an adult, free, reachable
# character who is not a witch hunter, a daemon, or recently accused.
# Scope: a character.
whunter_quarry_base_trigger = {
	is_adult = yes
	prisoner = no
	is_inaccessible_trigger = no
	NOT = { has_character_modifier = accused_recently }
	NOT = { society_member_of = sigmar_witch_hunters }
	NOT = { trait = witch_hunter }
	NOT = { is_daemon_race_trigger = yes }
}

# Quarry for "Hunt the mutant": an unlanded mutant.
whunter_mutant_quarry_trigger = {
	is_ruler = no
	is_mutated_trigger = yes
	whunter_quarry_base_trigger = yes
}

# Quarry for "Root out a cult cell": a cultist who is not an independent ruler.
whunter_cultist_quarry_trigger = {
	independent = no
	is_cultist_trigger = yes
	whunter_quarry_base_trigger = yes
}

# Quarry for "The necromancer": an unlanded user of dark lore or a vampire.
whunter_necromancer_quarry_trigger = {
	is_ruler = no
	OR = {
		has_dark_lore_trigger = yes
		is_vampire_trigger = yes
	}
	whunter_quarry_base_trigger = yes
}

# A fellow member fit for a sparring bout (whunter.205).
# Scope: a member of the Order; ROOT is the member looking for a partner.
whunter_sparring_partner_trigger = {
	NOT = { character = ROOT }
	age = 16
	prisoner = no
	is_inaccessible_or_incapable_trigger = no
	NOT = { has_character_modifier = bedridden_illness }
	NOT = { has_character_modifier = severe_illness }
	NOT = { has_character_flag = call_to_legend_active }
}

# The Order has not asked this ruler for a castle site (whunter.105 or 110) in
# the last 10 years.
# Scope: a ruler.
whunter_site_request_ready_trigger = {
	OR = {
		NOT = { has_character_flag = whunter_asked_for_site }
		had_character_flag = { flag = whunter_asked_for_site days = 3650 }
	}
}

# The Order has not asked this ruler for a son (whunter.100) in the last 15
