# A free adult courtier under suspicion of Chaos worship who has not been
# accused lately.
# Scope: a character.
whunter_suspect_courtier_trigger = {
	prisoner = no
	age = 16
	NOT = { has_character_modifier = wh_accused }
	OR = {
		has_character_modifier = under_suspicion
		has_character_modifier = highly_suspect
		has_character_modifier = known_satanist
	}
}

# A mage (or someone with magic potential) the local magic laws say should
# burn, who is not hiding, incapable, a Breton damsel or a daemon.
# Scope: a character.
whunter_mage_suspect_trigger = {
	NOR = { trait = in_hiding trait = incapable trait = BretonDamsel is_daemon_race_trigger = yes }
	check_if_mage_should_be_burned_trigger = yes
	OR = {
		has_character_modifier = magic_power_1
		has_character_modifier = magic_power_2
		has_character_modifier = magic_power_3
		has_character_modifier = magic_power_4
		magic_potential_trigger = yes
	}
}

# Anyone a witch hunter mission can go after: an adult, free, reachable
# character who is not a witch hunter, a daemon, or recently accused.
# Scope: a character.
whunter_quarry_base_trigger = {
	is_adult = yes
	prisoner = no
	is_inaccessible_trigger = no
	NOT = { has_character_modifier = accused_recently }
	whunter_chapter_member_trigger = no
	NOT = { trait = witch_hunter }
	NOT = { is_daemon_race_trigger = yes }
}

# Quarry for "Hunt the mutant": an unlanded mutant.
whunter_mutant_quarry_trigger = {

	is_mutated_trigger = yes
	whunter_quarry_base_trigger = yes

###############################################################################
# Witch hunter chapters
# Three chapters hunt witches in the Old World: the Order of Sigmar
# (sigmar_witch_hunters), the Holy Inquisition of Myrmidia
# (holy_inquisition_myrmidia) and the Templars of Solkan (templars_of_solkan).
###############################################################################

# A member of any witch hunter chapter. Use this instead of checking
# sigmar_witch_hunters directly, unless the check is about the Order of
# Sigmar's own titles (k_witch_hunters_order and its seats).
# Scope: a character.
whunter_chapter_member_trigger = {
	OR = {
		society_member_of = sigmar_witch_hunters
		society_member_of = holy_inquisition_myrmidia
		society_member_of = templars_of_solkan
	}
}

# Lands of the Holy Inquisition: Estalia and Tilea, by faith, culture or seat.
# Scope: a character.
whunter_inquisition_lands_trigger = {
	OR = {
		religion = myrmidian
		culture_group = estalian_group
		culture_group = tilean_group
		capital_scope = { region = world_old_world_estalia }
		capital_scope = { region = world_old_world_tilea }
	}
}

# The chapter that answers this character's call is the Templars of Solkan:
# Solkanites, and the Border Princes (who prefer the Templars) unless they
# are Myrmidians or Estalians/Tileans (the Inquisition) or Sigmarites,
# Ulricans and Taalites (the Order of Sigmar, see whchapter.5).
# Scope: a character.
whunter_solkan_lands_trigger = {
	OR = {
		religion = solkanite
		AND = {
			capital_scope = { region = world_old_world_border_princes }
			NOR = {
				religion = myrmidian
				religion = sigmarite
				religion = ulrican
				religion = taalite
				culture_group = estalian_group
				culture_group = tilean_group
			}
		}
	}
}

# A ruler who may keep a court witch hunter: Old World faiths, but not
# Bretonnia (no order of its own) and not the Stromfels cult, which witch
# hunters burn.
# Scope: a ruler.
whunter_can_keep_court_hunter_trigger = {
	is_ruler = yes
	religion_group = old_world_group
	NOR = { religion = grail religion = false_grail trait = witch_hunter }
}

# A courtier the ruler can train as a witch hunter: an adult of sound mind
# and body with a hard streak, who is no mage, mutant or vampire.
# Scope: the courtier; FROM is the ruler.
whunter_trainable_courtier_trigger = {
	is_adult = yes
	NOT = { age = 50 }
	is_ruler = no
	prisoner = no
	NOT = { trait = witch_hunter }
	NOT = { trait = incapable }
	is_inaccessible_trigger = no
	is_vampire_trigger = no
	is_trained_mage_trigger = no
	is_mutated_trigger = no
	is_daemon_race_trigger = no
	NOT = { magic_potential_trigger = yes }
	OR = {
		trait = zealous
		trait = brave
		trait = wroth
		trait = cruel
		trait = diligent
		trait = just
		martial = 10
		intrigue = 10
	}
	religion_group = old_world_group
}

# Holds a court witch hunter title: Court Witch Hunter, Court Inquisitor or
# Court Templar.
# Scope: a character.
whunter_court_hunter_title_trigger = {
	OR = {
		has_minor_title = title_court_witchhunter
		has_minor_title = title_court_inquisitor
		has_minor_title = title_court_templar
	}
}

# The ruler has room for another court witch hunter: emperors keep up to
# five, kings three, everyone else one. Each court title's allowed_to_grant
# uses the same caps.
# Scope: a ruler.
whunter_court_hunter_seat_empty_trigger = {
	whunter_can_keep_court_hunter_trigger = yes
	is_state_witch_hunter_trigger = no
	trigger_if = {
		limit = { tier = EMPEROR }
		has_assigned_minor_title = { title = title_court_witchhunter count < 5 }
		has_assigned_minor_title = { title = title_court_inquisitor count < 5 }
		has_assigned_minor_title = { title = title_court_templar count < 5 }
	}
	trigger_else = {
		trigger_if = {
			limit = { tier = KING }
			has_assigned_minor_title = { title = title_court_witchhunter count < 3 }
			has_assigned_minor_title = { title = title_court_inquisitor count < 3 }
			has_assigned_minor_title = { title = title_court_templar count < 3 }
		}
		trigger_else = {
			has_assigned_minor_title = { title = title_court_witchhunter count < 1 }
			has_assigned_minor_title = { title = title_court_inquisitor count < 1 }
			has_assigned_minor_title = { title = title_court_templar count < 1 }
		}
	}
}

# Fewer holders of this court hunter title than the ruler's tier allows
# (emperor 5, king 3, others 1). Used by the titles' allowed_to_grant too.
# Scope: a ruler. Parameter: $title$.
whunter_court_hunter_cap_trigger = {
	trigger_if = {
		limit = { tier = EMPEROR }
		has_assigned_minor_title = { title = $title$ count < 5 }
	}
	trigger_else = {
		trigger_if = {
			limit = { tier = KING }
			has_assigned_minor_title = { title = $title$ count < 3 }
		}
		trigger_else = { has_assigned_minor_title = { title = $title$ count < 1 } }
	}
}

# A prisoner the Inquisition may put on trial: a cultist, a suspect, a
# mutant, a user of dark lore, a mage the magic laws condemn, or a heretic.
# Scope: the prisoner.
whunter_tribunal_accused_trigger = {
	custom_tooltip = {
		text = whunter_tribunal_accused_tt
		OR = {
			is_cultist_trigger = yes
			has_character_modifier = under_suspicion
			has_character_modifier = highly_suspect
			has_character_modifier = known_satanist
			is_mutated_trigger = yes
			has_dark_lore_trigger = yes
			whunter_mage_suspect_trigger = yes
			is_heretic = yes
		}
	}
	NOT = { is_daemon_race_trigger = yes }
}

# A hunter of a different chapter than ROOT, rank 2 or higher.
# Scope: the other hunter; ROOT is the hunter comparing.
whunter_rival_chapter_trigger = {
	NOT = { character = ROOT }
	society_rank >= 2
	prisoner = no
	OR = {
		AND = {
			society_member_of = sigmar_witch_hunters
			ROOT = { NOT = { society_member_of = sigmar_witch_hunters } }
		}
		AND = {
			society_member_of = holy_inquisition_myrmidia
			ROOT = { NOT = { society_member_of = holy_inquisition_myrmidia } }
		}
		AND = {
			society_member_of = templars_of_solkan
			ROOT = { NOT = { society_member_of = templars_of_solkan } }
		}
	}
}
