###############################################################################
# Colleges of Magic
#
#   college_of_magic_imperial   The Colleges of Magic of Altdorf
#   college_of_magic_dynamic_1  Five more colleges, founded in play by a ruler
#   ...                         (each switched on by its global flag
#   college_of_magic_dynamic_5  college_of_magic_dynamic_N_active)
#
# The six societies are the same apart from their names and flags. Who may
# join is in college_can_join_trigger and college_potential_trigger
# (common/scripted_triggers/wh_colleges_of_magic_scripted_triggers.txt), so
# a change there reaches every college. A change to ranks, currency or
# member score has to be made six times.
# Six ranks: apprentice to the Supreme Patriarch, with one rank-5 Patriarch
# per wind. Not indestructible, and no members at game start
# (startup_populate is always = no): colleges fill up through the
# apprenticeship events.
#
# Events: events/wh_college_of_magic_events.txt, events/S_mage_learning_events.txt.
# Decisions: decisions/wh_colleges_of_magic_decisions.txt.
# Loc: localisation/00_college_of_magic.csv.
###############################################################################

college_of_magic_imperial = {
	primary_attribute = learning
	society_ranks_gfx = college_ranks
	opinion_to_other_members = +10
	opinion_per_rank_above = +5
	sound = hermetic_society_interface
	society_influence = no
	non_interference = { always = yes }

	active = { has_global_flag = college_of_magic_imperial_active }

	can_join_society = {
		hidden_trigger = {
			NAND = {
				ai = yes
				is_society_rank_full = { society = college_of_magic_imperial rank = 1 }
			}
		}
		college_can_join_trigger = yes # wh_colleges_of_magic_scripted_triggers.txt
	}

	show_society = {
		has_global_flag = college_of_magic_imperial_active
		OR = { religion_group = old_world_group society_member_of = college_of_magic_imperial }
	}

	potential = { college_potential_trigger = yes }

	society_rank = {
		level = 1
		limit = 200
		startup_limit = 10
		modifier = { church_opinion = -10 }
		powers = {
			college_increased_xp #Passive Power
		}
	}

	society_rank = {
		level = 2
		limit = 100
		startup_limit = 6
		modifier = { learning = 1 church_opinion = -10 }
		decisions = { political_focus_college college_equipment_requisition }
	}

	society_rank = {
		level = 3
		limit = 50
		startup_limit = 4
		modifier = { learning = 2 church_opinion = -10 }
		powers = {
			#powerstones_discovery
		}
	}

	society_rank = {
		level = 4
		limit = 20
		startup_limit = 2
		modifier = { monthly_character_prestige = 1 learning = 3 martial = 1 church_opinion = -10 }
		decisions = { favouritism_colleges }
	}
	society_rank = {
		level = 5
		limit = 8 # 1 Per wind
		startup_limit = 0
		modifier = { monthly_character_prestige = 2 learning = 3 martial = 2 church_opinion = -10 }
		decisions = { establish_local_college_presence }
	}
	society_rank = {
		level = 6
		limit = 1
		startup_limit = 1
		modifier = {
			monthly_character_prestige = 5
			general_opinion = 5
			learning = 4
			martial = 2
			church_opinion = -10
		}
		decisions = { }
	}

	plots = { }

	monthly_currency_gain = {
		name = currency_name_college_of_magic

		per_attribute = { name = learning value = 0.5 }
		per_attribute = { name = martial value = 0.15 }
		per_attribute = { name = diplomacy value = 0.3 }
		has_trait = { value = 2 trait = diligent }
		has_trait = { value = 1 trait = erudite }
		has_trait = { value = 1 trait = stubborn }
		has_trait = { value = 1 trait = patient }
		has_trait = { value = 0.5 trait = magic_potential_1 }
		has_trait = { value = 0.6 trait = magic_potential_2 }
		has_trait = { value = 0.75 trait = magic_potential_3 }
		has_trait = { value = 1 trait = magic_potential_4 }
		triggered_gain = {
			desc = CURRcollege_is_smart
			trigger = { is_smart_trigger = yes }
			value = 0.25
		}
		triggered_gain = {
			desc = CURRcollege_magic_power_1
			trigger = { has_character_modifier = magic_power_1 }
			value = 0.25
		}
		triggered_gain = {
			desc = CURRcollege_magic_power_2
			trigger = { has_character_modifier = magic_power_2 }
			value = 1
		}
		triggered_gain = {
			desc = CURRcollege_magic_power_3
			trigger = { has_character_modifier = magic_power_3 }
			value = 2
		}
		triggered_gain = {
			desc = CURRcollege_magic_power_4
			trigger = { has_character_modifier = magic_power_4 }
			value = 4
		}
		triggered_gain = {
			desc = CURRcollege_learn_on_own
			trigger = { has_character_flag = college_member_learning_on_own }
			value = 1
		}
		society_rank = { rank = 6 value = 10 }
	}

	member_score_per_attribute = 3
	member_score_per_rank = 100

	member_score = {
		value = 10

		modifier = { factor = 2 trait = diligent }

		modifier = { factor = 1.5 trait = scholar }
		modifier = { factor = 1.5 trait = socializer }
		modifier = { factor = 1.5 trait = strategist }
		modifier = { factor = 1.5 trait = erudite }
		modifier = { factor = 1.5 trait = stubborn }
		modifier = { factor = 1.5 trait = patient }
		modifier = { factor = 1.5 is_smart_trigger = yes }
		modifier = { factor = 0.1 is_landed = yes }
	}

	startup_populate = { trigger = { always = no } }
}
