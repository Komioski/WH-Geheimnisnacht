###############################################################################
# WHG Secret Cults
#
# Hidden worship inside civilised realms: the four Chaos god cults and the
# Undivided cult of the Old World (Empire, Tilea, Estalia, Araby), and the
# Cult of Pleasure among the elves.
#
# How they fit together:
#   - Membership is secret. can_join and show_society gate on faith (Old World
#     gods, Ormazd, Kitabid; Cytharai/Cadai for the Cult of Pleasure), but
#     potential does not, so converting to the cult's god keeps you in.
#   - Bretonnians (Lady/false Grail) and Kislevites never see the Chaos cults;
#     Morrians and witch hunters are kept out at startup.
#   - Rulers of kingdoms and empires have no need for a cult unless they are
#     republics or cynical. Startup members stay in the Old World, Araby or
#     the Badlands.
#   - Cult events live in events/WHG_chaos_cult_events.txt; Cult of Pleasure
#     events are in events/WHG_slaanesh_events.txt (pleasure namespace).
###############################################################################

	devil_worshipper = yes
	opinion_to_other_members = 10
	opinion_per_rank_above = 5

			is_witch_hunter_trigger = yes
			trait = honest
			religion = grail
			religion = false_grail

		}
		OR = {
			diplomacy = 10
			trait = ambitious
			trait = proud
			trait = gluttonous
			trait = greedy
			trait = lustful
			trait = wroth
			trait = cynical
			trait = deceitful
			trait = cruel
			trait = brave
			trait = hunter
			trait = impaler
			trait = stressed
			trait = depressed
			trait = organizer
			is_ill = yes
			trait = poet
			trait = strong
			is_mutated_trigger = yes
		}
	}

	show_society = {
		is_adult = yes
		OR = {
			religion = ormazd
			religion = kitabid
			religion_group = old_world_group
			society_member_of = cult_of_undivided
		}
		NOR = { religion = grail religion = false_grail religion = kislev_gods }
	}

	potential = { is_ascetic_trigger = no NOT = { culture_group = daemonic_group } }

	society_rank = {
		level = 1
		limit = 100
		startup_limit = 10
		custom_tooltip = chaos_corruption_power_tt
		modifier = {
			is_visible = {
				OR = {
					society_member_of = cult_of_undivided
