###############################################################################
# WHG Guilds
#
# Craft, trade and court guilds, whatever the faction:
#   runesmiths_society    Guild of Runesmiths: Dawi rune lore, rune crafting,
#                         Anvil of Doom. Membership ends if the member forsakes
#                         the ancestor gods.
#   daemonsmiths_society  Hashut's daemonsmiths. Switched off (active has
#                         always = no): no loc yet and its five powers
#                         (daemonsmith_*) are not written.
#   engineers_guild       Dawi, Tilean and (a year after the lord of Altdorf
#                         founds the Imperial Engineers School) Empire
#                         engineers. Events: events/WHG_engineer_events.txt
#                         (engineer); decisions:
#                         decisions/WHG_engineer_decisions.txt.
#   boyar_duma            The Boyar Duma of Kislev: nobles and Erengrad guild
#                         masters, politics, favours and trade.
#                         Events: events/WHG_kislev_society_events.txt
#                         (kislevsoc); decisions:
#                         decisions/WHG_kislev_society_decisions.txt;
#                         triggers/effects: wh_kislev_society_triggers.txt,
#                         wh_kislev_society_effects.txt. Borrows
#                         interface/wh_kislev_societies.gfx art.
#   dawi_elder_council    The Council of Elders of the Dawi: kings, thanes and
#                         Longbeards, law, feuds, debts and the sagas.
#                         Events: events/WHG_dawi_elder_events.txt
#                         (dawielder); decisions:
#                         decisions/WHG_dawi_elder_decisions.txt.
#
# Every society here is indestructible with a startup_populate that checks
# is_in_society = no (the game uses it to fill starting ranks and pick heads).
###############################################################################

###############################################################################
# Guild of Runesmiths (Dawi)
###############################################################################
runesmiths_society = {
	primary_attribute = stewardship
	opinion_to_other_members = 10
	opinion_per_rank_above = 5
	sound = hermetic_society_interface
	society_ranks_gfx = secret_ranks
	society_influence = no
	non_interference = { always = yes }

	active = { has_dlc = "Mystics" }
	indestructible = yes

	can_join_society = {
		hidden_trigger = {
			NAND = { ai = yes is_society_rank_full = { society = runesmiths_society rank = 1 } }
			OR = { is_landed = yes is_patrician = yes }
		}
		religion_group = dwarf_gods_group
		OR = {
			AND = { learning = 16 stewardship = 16 }
			learning = 25
			stewardship = 25
			trait = runesmith
			trait = runelord
		}
		NOR = { is_nomadic = yes is_tribal = yes }
		hidden_trigger = { mercenary = no }
		age = 16
		hidden_trigger = { NOT = { has_character_flag = society_join_block } }
		trigger_if = { limit = { prisoner = yes } prisoner = no }
	}

	show_society = {
		OR = { religion_group = dwarf_gods_group society_member_of = runesmiths_society }
	}

	potential = { religion_group = dwarf_gods_group } # forsaking the ancestor gods ends membership

	society_rank = {
		level = 1
		limit = 20
		startup_limit = 2
		modifier = {
			fertility_penalty = -0.15
			#economy_techpoints = 0.05
		}
		decisions = {
			smith_crafting #Enables Standard Item Crafting (Missions)
		}
	}

	society_rank = {
		level = 2
		limit = 20
		startup_limit = 2
		modifier = { learning = 1 stewardship = 1 fertility_penalty = -0.15 }
		decisions = {
			runesmith_create_rune_of_power # Enables Creation of Lesser rune of Power
			runesmith_become_runesmith # Allows taking full benefit of a Runesmith
			runesmithing # Allows Partaking in Gromril Crafting
		}
	}

	society_rank = {
		level = 3
		limit = 20
		startup_limit = 8
		modifier = { learning = 1 stewardship = 2 fertility_penalty = -0.15 }
		decisions = {
			choose_hermetic_art # Enables you to pick/change your lifestyle trait
			runemagic_aspect # Allows Choosing Aspect of Runemagic. Valaya & Grungi & Grimnir (Morale, Defense, Damage)
		}
	}

	society_rank = {
		level = 4
		limit = 13
		startup_limit = 2
		modifier = { learning = 2 stewardship = 3 fertility_penalty = -0.15 }
		decisions = {
			runesmith_create_rune_of_power # Enables Creation of Greater rune of Power
			runesmith_become_runelord # Allows taking full benefit of a Runelord
			runesmith_anvil # Acquire an Anvil of A Doom
		}
	}

	plots = { }

	monthly_currency_gain = {
		name = currency_name_runesmiths

		per_attribute = { name = learning value = 0.2 }
		per_attribute = { name = stewardship value = 0.3 }
		triggered_gain = {
			desc = CURRrunesmith_lacks_skill
			trigger = { NOR = { trait = runesmith trait = runelord } }
			value = -1.5
		}
		triggered_gain = {
			desc = CURRlow_status
			trigger = { OR = { tier = KING tier = EMPEROR } }
			value = -2
		}
		has_trait = { value = 1 trait = scholar }
		has_trait = { value = 1 trait = mystic }
		has_trait = { value = 1 trait = erudite }
		has_trait = { value = 1 trait = stubborn }
		has_trait = { value = -1 trait = wroth }
		has_trait = { value = 1 trait = patient }
		has_trait = { value = 0.5 trait = genius }
		has_trait = { value = 0.25 trait = quick }
		has_trait = { value = 1 trait = shrewd }
		has_trait = { value = 1 trait = just }
		has_trait = { value = 1 trait = runesmith }
		has_trait = { value = 1 trait = runelord }
		has_trait = { value = 0.5 trait = diligent }
		has_trait = { value = -0.5 trait = envious }
		society_rank = { rank = 4 value = 5 }
	}

	member_score_per_attribute = 2
	member_score_per_rank = 50

	member_score = {
		value = 10
		modifier = { factor = 1.5 age > 200 }
		modifier = { factor = 1.5 age > 300 }
		modifier = { factor = 1.5 age > 120 }
		modifier = { factor = 2 trait = runesmith }
		modifier = { factor = 4 trait = runelord }
		modifier = { factor = 0.2 NOR = { trait = runelord trait = runesmith } }
		modifier = { factor = 0.2 OR = { tier = king tier = emperor } }
		modifier = { factor = 2 trait = diligent }
		modifier = { factor = 1.5 trait = scholar }
		modifier = { factor = 1.5 trait = mystic }
		modifier = { factor = 1.5 trait = erudite }
		modifier = { factor = 1.5 trait = stubborn }
		modifier = { factor = 1.5 trait = patient }
		modifier = { factor = 1.5 OR = { trait = genius trait = quick trait = shrewd } }
		modifier = { factor = 1.5 trait = just }
	}

	startup_populate = {
		trigger = {
			is_female = no
			controls_religion = no
			is_adult = yes
			is_in_society = no
			is_dumb_trigger = no
			trait = creature_dwarf
			religion_group = dwarf_gods_group
			OR = {
				trait = runesmith
				trait = runelord
				father_even_if_dead = { OR = { trait = runesmith trait = runelord } }
			}
		}
	}
}
