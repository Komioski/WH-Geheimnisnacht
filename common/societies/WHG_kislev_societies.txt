###############################################################################
# Kislev societies
#
#   ice_witch_sisterhood  The Ice Witch Sisterhood: Kislev's witches of the
#                         Lore of Ice, women of Kislev's gods or Kislevite
#                         blood with the gift. Frost Ward, the Great Warding,
#                         the Ice Palace duty, witch missions.
#   circle_of_the_bear    The Circle of the Bear: Ursun's warrior order, the
#                         martial home of Kislevites (not the Knights of the
#                         Old World). Rank-2 members may take a posting in the
#                         Rota of the Dawn, Dazh's temple guard.
#   boyar_duma            The Boyar Duma: nobles and Erengrad guild masters,
#                         politics, favours and trade.
#
# Kislev's witch hunters, the Chekist, are in WHG_witch_hunter_societies.txt.
# All three are indestructible and need their startup_populate: the game uses
# it to fill the starting ranks and to pick a new head.
#
# Events: events/WHG_ice_witch_events.txt (frostwitch),
# events/WHG_kislev_society_events.txt (kislevsoc).
# Decisions: decisions/WHG_ice_witch_decisions.txt,
# decisions/WHG_kislev_society_decisions.txt.
# Triggers/effects: wh_ice_witch_triggers.txt, wh_kislev_society_triggers.txt,
# wh_kislev_society_effects.txt.
# Art: the Circle and the Duma borrow interface/wh_kislev_societies.gfx art
# until their own is made.
###############################################################################

###############################################################################
# The Ice Witch Sisterhood
# Women with the gift in Kislev or under the Ice Throne. Frosthome Lodge
# (d_frosthome_lodge) raises the member score.
###############################################################################
ice_witch_sisterhood = {
	primary_attribute = diplomacy
	society_ranks_gfx = rank_icons_ice_witch_sisterhood
	opinion_to_other_members = 15
	opinion_per_rank_above = 10
	sound = hermetic_society_interface
	is_religious = yes

	active = { has_dlc = "Mystics" }

	indestructible = yes

	can_join_society = {
		hidden_trigger = { mercenary = no }
		hidden_trigger = {
			NAND = { ai = yes is_society_rank_full = { society = ice_witch_sisterhood rank = 1 } }
		}
		OR = {
			location = { region = world_old_world_kislev }
			top_liege = { has_landed_title = e_kislev }
		}
		is_female = yes
		age = 12
		trait = creature_human
		NOR = { is_witch_hunter_trigger = yes trait = hollowed trait = hag_witch }
		OR = {
			trait = magic_potential_1
			trait = magic_potential_2
			trait = magic_potential_3
			trait = magic_potential_4
			trait = ice_witch
			trait = lore_ice
		}
		OR = { religion = kislev_gods culture_group = kislevite_group }
	}

	show_society = {
		OR = { religion = kislev_gods culture_group = kislevite_group }
		OR = {
			location = { region = world_old_world_kislev }
			top_liege = { has_landed_title = e_kislev }
		}
		OR = {
			trait = magic_potential_1
			trait = magic_potential_2
			trait = magic_potential_3
			trait = magic_potential_4
			trait = ice_witch
			trait = lore_ice
		}
		learning = 8
		trait = creature_human
	}

	potential = {
		age = 8
		# Keep this loose: anything added here can break apprenticeships. Add an
		# event if members should be able to be thrown out.
	}

	society_rank = {
		level = 1
		limit = 20
		startup_limit = 10
		modifier = { fertility = -0.05 }
		decisions = { }
	}

	society_rank = {
		level = 2
		limit = 15
		startup_limit = 5
		modifier = { diplomacy = 1 fertility = -0.10 combat_rating = 10 }
		decisions = { }
	}

	society_rank = {
		level = 3
		limit = 10
		startup_limit = 3
		modifier = { diplomacy = 2 fertility = -0.15 combat_rating = 20 }
		decisions = { }
	}

	society_rank = {
		level = 4
		limit = 1
		startup_limit = 1
		modifier = { diplomacy = 3 fertility = -0.20 combat_rating = 30 }
		decisions = { }
	}

	plots = { }

	monthly_currency_gain = {
		name = currency_name_ice_witch_sisterhood

		per_attribute = { name = diplomacy value = 0.3 }
		per_attribute = { name = learning value = 0.2 }
		has_trait = { value = 2 trait = diligent }
		has_trait = { value = 1 trait = scholar }
		has_trait = { value = 1 trait = mystic }
		has_trait = { value = 1 trait = erudite }
		has_trait = { value = 1 trait = stubborn }
		has_trait = { value = 1 trait = patient }
		has_trait = { value = 1 trait = genius }
		has_trait = { value = 1 trait = quick }
		has_trait = { value = 1 trait = shrewd }
		has_trait = { value = 1 trait = just }
		has_trait = { value = 1 trait = gregarious }
		has_trait = { value = 2 trait = inspiring_leader }
		has_trait = { value = 1 trait = lore_ice }
		has_trait = { value = 2 trait = ice_witch }
		has_trait = { value = 5 trait = studying_magic }
		society_rank = { rank = 4 value = 2 }
	}

	member_score_per_attribute = 3
	member_score_per_rank = 100

	member_score = {
		value = 10

		modifier = { factor = 1.3 culture = gospodar }
		modifier = { factor = 2 trait = diligent }
		modifier = { factor = 1.5 trait = scholar }
		modifier = { factor = 1.5 trait = mystic }
		modifier = { factor = 1.5 trait = erudite }
		modifier = { factor = 1.5 trait = stubborn }
		modifier = { factor = 1.5 trait = patient }
		modifier = { factor = 1.5 OR = { trait = genius trait = quick trait = shrewd } }
		modifier = { factor = 1.5 trait = just }
		modifier = { factor = 1.5 trait = gregarious }
		modifier = { factor = 2 trait = inspiring_leader }
		modifier = { factor = 2 trait = lore_ice }
		modifier = { factor = 3 trait = ice_witch }
		modifier = { factor = 1.25 diplomacy = 10 }
		modifier = { factor = 1.5 diplomacy = 15 }
		modifier = { factor = 2 diplomacy = 20 }
		modifier = { factor = 2 has_landed_title = d_frosthome_lodge }
	}

	startup_populate = {
		trigger = {
			controls_religion = no
			location = { region = world_old_world_kislev }
			is_female = yes
			age = 12
			trait = creature_human
			is_in_society = no
			is_dumb_trigger = no
			NOR = { is_witch_hunter_trigger = yes trait = hollowed }
			OR = {
				trait = magic_potential_1
				trait = magic_potential_2
				trait = magic_potential_3
				trait = magic_potential_4
				trait = ice_witch
				trait = lore_ice
			}
			NOT = { trait = hag_witch }
			OR = { religion = kislev_gods culture_group = kislevite_group }
		}
	}
}

###############################################################################
# The Circle of the Bear
###############################################################################
circle_of_the_bear = {
	primary_attribute = martial
	opinion_to_other_members = 10
	opinion_per_rank_above = 10
	society_ranks_gfx = warriors_ranks
	sound = warrior_lodge_ambient_02
	society_rank_up_decision = kislevsoc_circle_rank_up
	active = { has_dlc = "Mystics" }
	indestructible = yes

	can_join_society = {
		hidden_trigger = { NOT = { has_character_flag = society_join_block } }
		hidden_trigger = { mercenary = no }
		hidden_trigger = {
			NAND = { ai = yes is_society_rank_full = { society = circle_of_the_bear rank = 1 } }
		}
		kislevsoc_circle_candidate_trigger = yes
		trigger_if = { limit = { prisoner = yes } prisoner = no }
	}

	show_society = { OR = { society_member_of = circle_of_the_bear religion = kislev_gods } }

	potential = { religion = kislev_gods }

	society_rank = {
		level = 1
		limit = 100
		startup_limit = 15
		modifier = { combat_rating = 5 }
		decisions = { kislevsoc_bear_hunt kislevsoc_waking_feast }
	}

	society_rank = {
		level = 2
		limit = 30
		startup_limit = 6
		modifier = { martial = 1 combat_rating = 10 }
		decisions = {
			kislevsoc_circle_trial
			kislevsoc_call_rota
			kislevsoc_rota_of_dawn
			kislevsoc_dawn_guard
		}
	}

	# Bear Champions charge first: the Bear's Charge.
	society_rank = {
		level = 3
		limit = 8
		startup_limit = 3
		modifier = { martial = 2 combat_rating = 20 land_morale = 0.05 }
		decisions = { }
	}

	society_rank = {
		level = 4
		limit = 1
		startup_limit = 1
		modifier = {
			martial = 3
			combat_rating = 30
			land_morale = 0.10
			monthly_character_piety = 0.2
		}
		decisions = { kislevsoc_muster_circle }
		obedience_tooltip = obedience_tooltip
		obedient = { ROOT = { always = yes } }
	}

	plots = { }

	monthly_currency_gain = {
		name = currency_name_circle_of_the_bear
		per_attribute = { name = martial value = 0.5 }
		has_trait = { trait = brave value = 1 }
		has_trait = { trait = wroth value = 1 }
		has_trait = { trait = zealous value = 1 }
		has_trait = { trait = strong value = 1 }
		has_trait = { trait = robust value = 1 }
		has_trait = { trait = hunter value = 1 }
		has_trait = { trait = duelist value = 1 }
		has_trait = { trait = skilled_tactician value = 1 }
		has_trait = { trait = brilliant_strategist value = 1 }
		society_rank = { rank = 4 value = 5 }
	}

	# Who fills the starting ranks and who can be picked as the next head.
	# Without it the engine takes characters from anywhere in the world.
	startup_populate = {
		trigger = {
			kislevsoc_circle_candidate_trigger = yes
			is_in_society = no
			controls_religion = no
			prisoner = no
			NOT = { has_landed_title = e_kislev }
		}
	}

	member_score_per_attribute = 3
	member_score_per_rank = 100

	member_score = {
		value = 10
		modifier = { factor = 2 martial = 15 }
		modifier = { factor = 2 martial = 20 }
		modifier = { factor = 1.5 combat_rating = 40 }
		modifier = { factor = 1.5 trait = brave }
		modifier = { factor = 1.5 trait = hunter }
		modifier = { factor = 2 has_landed_title = k_cult_of_ursun }
		modifier = { factor = 0.5 has_character_flag = kislevsoc_dawn }
	}
}
