###############################################################################
# Witch hunter chapters
#
# The four witch hunter societies. Every chapter takes witch hunters
# (is_witch_hunter_trigger: the traits witch_hunter, inquisitor,
# solkan_templar and chekist) and their apprentices, and shares the hunts,
# court titles and relics of the witch hunter code.
#
#   sigmar_witch_hunters       Order of Sigmar: the Empire and the Old World
#   holy_inquisition_myrmidia  Holy Inquisition of Myrmidia: Estalia and Tilea
#   templars_of_solkan         Templars of Solkan: Solkanites and the merciless
#   kislev_chekist             The Chekist: Kislevites of Kislev's gods
#
# Who goes where (common/scripted_triggers/wh_witch_hunter_triggers.txt):
# whunter_inquisition_lands_trigger, whunter_solkan_lands_trigger and
# whunter_chekist_lands_trigger. The Order takes every other Old World hunter.
# Every chapter is indestructible and needs its startup_populate: the game
# uses it to fill the starting ranks and to pick a new head.
#
# Events: events/WHG_witch_hunter_events.txt, events/WHG_witch_hunter_events.txt,
# events/WHG_kislev_crown_events.txt.
# Decisions: decisions/wh_witch_hunter_decisions.txt,
# decisions/WHG_witch_chapter_decisions.txt.
# Loc: localisation/00_witch_hunter.csv, 00_WHG_witch_hunter_events.csv,
# 00_WHG_witch_hunter_events.csv, 00_WHG_kislev_crown_events.csv (the Chekist).
###############################################################################

###############################################################################
# Order of Sigmar
# The Templars of Sigmar, seated at the Holy Order (k_witch_hunters_order)
# with twelve chapter seats (d_witch_hunter_seat_1-12) and the Samur chapter.
# The seats are for Order hunters only (trait witch_hunter).
###############################################################################
sigmar_witch_hunters = {
	primary_attribute = intrigue
	is_religious = yes
	opinion_to_other_members = 5
	opinion_per_rank_above = 5
	opinion_to_pretenders = -40
	sound = monastic_orders_interface
	society_ranks_gfx = rank_icons_witchhunter

	non_interference = { always = yes }

	active = { has_dlc = "Mystics" }

	indestructible = yes

	can_join_society = {
		hidden_trigger = {
			NAND = { ai = yes is_society_rank_full = { society = sigmar_witch_hunters rank = 1 } }
		}
		is_adult = yes
		OR = { is_witch_hunter_trigger = yes liege = { is_witch_hunter_trigger = yes } }
		NOT = { is_trained_mage_trigger = yes }
		is_vampire_trigger = no
		hidden_trigger = { NOT = { has_character_flag = society_join_block } }
		trigger_if = { limit = { prisoner = yes } prisoner = no }
	}

	show_society = {
		OR = {
			religion = sigmarite
			is_witch_hunter_trigger = yes
			is_state_witch_hunter_trigger = yes
		}
	}

	potential = {
		religion_group = old_world_group
		NOT = { religion = solkanite } # Templars of Solkan
		trigger_if = { # Estalia and Tilea answer to the Holy Inquisition
			limit = { whunter_inquisition_lands_trigger = yes }
			religion = sigmarite
		}
		NOT = { whunter_chekist_lands_trigger = yes } # Kislev has the Chekist
		OR = {
			society_rank == 1
			is_witch_hunter_trigger = yes
			liege = { is_witch_hunter_trigger = yes }
		}
	}

	society_rank = { # Apprentices, taken on by landed witch hunters (decision)
		level = 1
		limit = 20
		startup_limit = 10
		modifier = { monthly_character_piety = 0.25 health = -1 combat_rating = 5 }
		decisions = { }
	}

	society_rank = { # Witch Hunter Templars
		level = 2
		limit = 60
		startup_limit = 5
		modifier = {
			monthly_character_piety = 0.5
			intrigue = 1
			martial = 1
			health = -1
			combat_rating = 10
		}
		decisions = { request_whunter_weapon }
	}

	society_rank = { # Witch Hunter Captains
		level = 3
		limit = 12
		startup_limit = 12
		modifier = {
			monthly_character_piety = 0.75
			martial = 1
			intrigue = 1
			diplomacy = 1
			health = -1
			combat_rating = 15
		}
		decisions = { scour_barony_power inquisitorial_enlisting_courtier }
	}

	society_rank = { # Witch Hunter Generals: the Grandmaster and four generals
		level = 4
		limit = 5
		startup_limit = 1
		modifier = {
			monthly_character_piety = 1
			martial = 2
			diplomacy = 1
			intrigue = 1
			health = -1
			combat_rating = 25
		}
		decisions = { request_recruit request_a_barony_inquisition }
	}

	plots = { }

	monthly_currency_gain = {
		name = "currency_name_sigmar_witch_hunters"

		per_attribute = { name = martial value = 0.5 }
		has_trait = { value = 3 trait = strong }
		has_trait = { trait = holy_warrior value = 2.5 }
		has_trait = { trait = patient value = 3 }
		has_trait = {
			trait = cruel # they get the job done
			value = 3
		}
		has_trait = { trait = diligent value = 1 }
		has_trait = { trait = brave value = 1 }
		has_trait = { trait = hunter value = 1 }
		has_trait = { trait = impaler value = 1 }
		has_trait = { trait = duelist value = 1 }
		has_trait = { trait = zealous value = 1 }
		has_trait = { trait = cynical value = -5 }
		triggered_gain = {
			desc = "grandmaster_of_order_of_sigmar"
			trigger = { primary_title = { title = k_witch_hunters_order } }
			value = 10
		}
		society_rank = { rank = 4 value = 10 }
	}

	member_score_per_attribute = 1
	member_score_per_rank = 100

	member_score = {
		value = 10
		additive_modifier = {
			localisation_key = grandmaster_of_order_of_sigmar
			primary_title = { title = k_witch_hunters_order }
			value = 10
		}
		additive_modifier = { localisation_key = landed_chapter_master tier = DUKE value = 5 }
		additive_modifier = { localisation_key = landed_chapter_master tier = KING value = 10 }
		modifier = { primary_title = { title = k_witch_hunters_order } factor = 10 }
		modifier = { primary_title = { title = d_samur_chapter } factor = 2 }
		modifier = { # the twelve chapter seats
			primary_title = {
				OR = {
					title = d_witch_hunter_seat_1
					title = d_witch_hunter_seat_2
					title = d_witch_hunter_seat_3
					title = d_witch_hunter_seat_4
					title = d_witch_hunter_seat_5
					title = d_witch_hunter_seat_6
					title = d_witch_hunter_seat_7
					title = d_witch_hunter_seat_8
					title = d_witch_hunter_seat_9
					title = d_witch_hunter_seat_10
					title = d_witch_hunter_seat_11
					title = d_witch_hunter_seat_12
				}
			}
			factor = 4
		}
		modifier = { NOT = { religion = sigmarite } factor = 0.8 }
		modifier = { trait = strong factor = 3 }
		modifier = { trait = holy_warrior factor = 2 }
		modifier = { trait = zealous factor = 1.5 }
		modifier = { trait = brave factor = 1.3 }
		modifier = { trait = patient factor = 1.3 }
		modifier = { trait = diligent factor = 1.1 }
		modifier = { trait = hunter factor = 1.1 }
		modifier = { trait = duelist factor = 1.1 }
		# Hypocritical witch hunters: cruelty and scheming get the job done
		modifier = { trait = cruel factor = 1.5 }
		modifier = { trait = wroth factor = 1.2 }
		modifier = { trait = impaler factor = 1.5 }
		modifier = { trait = schemer factor = 1.5 }
		modifier = { learning = 10 factor = 1.2 }
		modifier = { learning = 20 factor = 1.2 }
		modifier = { intrigue = 20 factor = 1.2 }
		modifier = { intrigue = 10 factor = 1.2 }
		modifier = { martial = 10 factor = 1.2 }
		modifier = { martial = 20 factor = 1.2 }
		modifier = { diplomacy = 10 factor = 1.2 is_landed = yes }
		modifier = { diplomacy = 20 factor = 1.2 is_landed = yes }
	}

	startup_populate = {
		trigger = {
			OR = {
				is_witch_hunter_trigger = yes
				AND = { liege = { is_witch_hunter_trigger = yes } is_hunter_culture_trigger = yes }
			}
			OR = { is_landed = yes liege = { independent = yes } }
			controls_religion = no
			religion = sigmarite
			is_adult = yes
			is_in_society = no
			is_vampire_trigger = no
			NOT = { is_trained_mage_trigger = yes }
		}
	}
}
