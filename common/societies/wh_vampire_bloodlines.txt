###############################################################################
# Vampire bloodline societies
#
# One society per bloodline. Every open vampire is locked into the society of
# their bloodline; hidden Lahmians and von Carsteins are secret members. Living
# thralls of a necromantic faith may join some societies at ranks 1-2.
# Vampire pirates stay in the pirate society.
#
#   carstein_court        von Carstein     Court of Sylvania
#   ordo_draconis         Blood Dragons    Ordo Draconis (open knightly order)
#   lahmian_sisterhood    Lahmians         Lahmian Sisterhood
#   necrarch_brotherhood  Necrarchs        Necrarch Brotherhood
#   strigoi_brood         Strigoi          Heirs of Strygos
#   jade_coil             Jade             The Jade Coil
#   carrion_children      Mahtmasi         The Carrion Children
#
# Joining: vamp_join_bloodline_effect (wh_vamp_effects.txt), run at startup
# (vampsoc.1), from the vampire-making effects and a five-year sweep
# (vampsoc.2). Members can't be expelled; anyone who leaves is put back by the
# sweep. Rank-ups use vampsoc_rank_up (wh_vampire_society_decisions.txt).
# The engine picks a new head (rank 4) from member_score when the head dies.
###############################################################################

	is_secret = yes

			vamp_thrall_trigger = yes

	}

		level = 1
		limit = 200
		startup_limit = 0

			vampsoc_feed
