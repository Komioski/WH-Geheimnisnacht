# Elves triggered modifiers (WHG_elves_triggered_modifiers.txt)
#
# Covers Ulthuan (the Phoenix Throne and the Everqueen) and Athel Loren (the
# avatars of Isha and Kurnous, the restless forest).
#
# Triggered modifiers are applied by the game itself to every character for
# whom both blocks are true, and removed when they stop being true; there is
# nothing to add or remove in script. The game checks them for every living
# character again and again, so they cost performance: `potential` is checked
# first and should rule out almost everyone with cheap checks (a title, a
# trait, a flag); `trigger` holds the costlier rest and is only checked when
# `potential` is true, so it need not repeat it. Keys: icon (a frame of
# gfx/interface/modifier_icons.dds, list in
# common/event_modifiers/WHG_engine_event_modifiers.txt), then any character
# modifiers. The file index is in 00_triggered_modifiers.txt.

# The Phoenix Queen - a woman on the Phoenix Throne without the Everqueen's
# crown of Avelorn (the Phoenix Throne is a man's seat)
phoenix_queen_modifier = {
	potential = {
		has_landed_title = e_ulthuan
		is_female = yes
		trait = creature_elf
	}
	trigger = {
		NOT = { has_landed_title = k_avelorn }
	}
	icon = 25
	vassal_opinion = -20
}

# Not Crowned - a Phoenix King who has not passed through the
# flames of Asuryan (the decision is never offered to a Phoenix Queen)
flames_uncrossed = {
	potential = {
		has_landed_title = e_ulthuan
		is_female = no
		trait = creature_elf
	}
	trigger = {
		age = 14
		NOT = { trait = phoenix_king }
		NOT = { trait = burnt_by_asuryan }
	}
	icon = 25
	vassal_opinion = -25
}

# Unclear Stance - an Asur lord of Ulthuan who has not chosen between
# isolation and intervention
unclear_stand = {
	potential = {
		is_ruler = yes
		culture_group = asur_group
		trait = creature_elf
		age = 16
		prisoner = no
		NOT = { has_character_flag = elf_politics_flag }
		NOT = { trait = isolationist_elf }
		NOT = { trait = interventionist_elf }
	}
	trigger = {
		capital_scope = { empire = { title = e_ulthuan } }
	}
	icon = 25
	vassal_opinion = -15
}

# Mortal - an elf on the throne of Athel Loren who is the avatar of
# neither Isha nor Kurnous
not_avatar = {
	potential = {
		has_landed_title = e_athel_loren
		trait = creature_elf
	}
	trigger = {
		NOT = { trait = avatar_kurnous }
		NOT = { trait = avatar_isha }
	}
	icon = 25
	vassal_opinion = -50
}

# Missing Avatar - Athel Loren has Isha's avatar but no living Kurnous
without_kurnous = {
	potential = {
		has_landed_title = e_athel_loren
		trait = creature_elf
	}
	trigger = {
		NOT = { trait = avatar_kurnous }
		NOT = { avatar_kurnous = { is_alive = yes } }
		avatar_isha = { is_alive = yes }
	}
	icon = 25
	vassal_opinion = -50
}

# Restless Spirits - a ruler seated in Athel Loren who does not keep faith
# with the forest (not of the Asrai faith, or with negative piety): the
# forest turns on their armies (-50% morale; Komi)
restless_spirits = {
	potential = {
		is_ruler = yes
		capital_scope = { empire = { title = e_athel_loren } }
	}
	trigger = {
		OR = {
			NOT = { piety = 0 }
			NOT = { religion = asrai_cadai }
		}
	}
	land_morale = -0.5
	icon = 26
}
