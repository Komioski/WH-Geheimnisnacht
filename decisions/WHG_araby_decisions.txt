# Araby decisions: the Fast of the Prophets, the vanilla Ramadan
# (observe_ramadan in vanilla minor_decisions) moved to the faiths of
# Ormazd and the Kitabid.
# Events: events/WHG_araby_events.txt (holyfast.1 - 40).
# The Night of a Thousand Lamps, the vanilla Diwali feast, moved to the
# fire-temples of Ormazd (lamps.1 - 60).

decisions = {
	# Keep the Fast of the Prophets for a month.
	observe_holy_fast = {
		only_playable = yes
		ai_check_interval = 60

		potential = {
			is_playable = yes
			is_adult = yes
			prisoner = no
			holyfast_faith_trigger = yes
			NOT = { has_character_flag = holy_fast }
			NOT = { has_character_modifier = observed_holy_fast }
			NOT = { has_character_flag = preparing_hajj }
			NOT = { trait = on_hajj }
			NOT = { trait = incapable }
			NOT = {
				has_alternate_start_parameter = { key = religion_names value = random }
			}
		}

		allow = {
			war = no
			prisoner = no
			is_inaccessible_trigger = no
			wealth = 100
			custom_tooltip = {
				text = IS_HEALTHY
				NOR = {
					is_ill = yes
					has_symptom_trigger = yes
				}
			}
			custom_tooltip = {
				text = is_not_busy_trigger_tooltip
				hidden_tooltip = { NOT = { has_character_flag = do_not_disturb } }
			}
		}

		effect = {
			set_character_flag = holy_fast
			character_event = { id = holyfast.1 tooltip = EVTTOOLTIPholyfast1 }
			decadence = -3
			piety = 15
			wealth = -100
			add_character_modifier = {
				name = observed_holy_fast
				duration = 1900
			}
			custom_tooltip = {
				text = capital_prospers_custom_tooltip
				hidden_tooltip = {
					capital_scope = {
						change_variable = { which = prosperity_value value = 8 }
					}
				}
			}
		}

		revoke_allowed = {
			always = no
		}

		ai_will_do = {
			factor = 0.2
			modifier = { factor = 5 trait = zealous }
			modifier = { factor = 2 decadence = 30 }
			modifier = { factor = 0 trait = cynical }
			modifier = { factor = 0 is_weak_trigger = yes }
			modifier = { factor = 0 age = 60 }
		}
	}

	# Light a thousand lamps for the Holy Flame and feast the vassals.
	hold_thousand_lamps_festival = {
		only_rulers = yes
		is_high_prio = yes
		ai_check_interval = 12

		potential = {
			is_ruler = yes
			prisoner = no
			is_incapable = no
			holyfast_faith_trigger = yes
			num_of_vassals = 1
			NOT = { has_character_modifier = holding_thousand_lamps }
			NOT = { has_character_modifier = large_feast_timer }
			NOT = { has_character_modifier = fabulous_feast }
			NOT = { has_character_flag = holy_fast }
			NOT = { liege = { has_character_modifier = holding_thousand_lamps } }
			NOT = { has_character_flag = do_not_disturb }
		}

		allow = {
			war = no
			prisoner = no
			is_inaccessible_trigger = no
			custom_tooltip = {
				text = UNOCCUPIED_DEMESNE_TITLE
				hidden_tooltip = {
					any_demesne_title = {
						NOT = { higher_tier_than = COUNT }
						is_occupied = no
					}
				}
			}
			scaled_wealth = 0.3
			month = 8
			NOT = { month = 11 }
			prestige = 100
		}

		effect = {
			scaled_wealth = -0.3
			add_character_modifier = {
				name = holding_thousand_lamps
				duration = 365
			}
			set_character_flag = recent_minor_decision
			set_character_flag = do_not_disturb
			hidden_tooltip = { character_event = { id = lamps.1 } }
			custom_tooltip = {
				text = capital_prospers_custom_tooltip
				hidden_tooltip = {
					capital_scope = {
						change_variable = { which = prosperity_value value = 8 }
					}
				}
			}
		}

		revoke_allowed = {
			always = no
		}

		ai_will_do = {
			factor = 0.1
			modifier = { factor = 0 NOT = { scaled_wealth = 2 } }
			modifier = { factor = 3 trait = gregarious }
			modifier = { factor = 2 trait = hedonist }
			modifier = { factor = 0.5 trait = shy }
		}
	}

	# The pilgrimage to Ind (events/WHG_araby_events.txt, Part 5, ind).
	# Off the map: a year away down the Spice Route. Once in twenty years.
	go_on_ind_pilgrimage = {
		ai_check_interval = 60

		potential = {
			is_adult = yes
			prisoner = no
			is_landed = yes
			ind_pilgrimage_faith_trigger = yes
			NOT = { trait = on_indian_pilgrimage }
			NOT = { trait = on_hajj }
			NOT = { trait = on_pilgrimage }
			NOT = { trait = incapable }
		}

		allow = {
			war = no
			prisoner = no
			is_inaccessible_trigger = no
			scaled_wealth = 0.3
			NOT = { had_character_flag = { flag = ind_pilgrimage_done days = 7300 } }
			NOT = { trait = pregnant }
			custom_tooltip = {
				text = IS_HEALTHY
				NOR = {
					is_ill = yes
					has_symptom_trigger = yes
				}
			}
			custom_tooltip = {
				text = is_not_busy_trigger_tooltip
				hidden_tooltip = { NOT = { has_character_flag = do_not_disturb } }
			}
		}

		effect = {
			scaled_wealth = -0.3
			clr_character_flag = ind_pilgrimage_done
			set_character_flag = ind_pilgrimage_done
			set_character_flag = do_not_disturb
			character_event = { id = ind.1 tooltip = EVTTOOLTIPind1 }
		}

		revoke_allowed = {
			always = no
		}

		ai_will_do = {
			factor = 0.05
			modifier = { factor = 3 trait = zealous }
			modifier = { factor = 2 trait = scholar }
			modifier = { factor = 0.2 trait = cynical }
			modifier = { factor = 0 NOT = { scaled_wealth = 1 } }
			modifier = { factor = 0 independent = no NOT = { liege = { opinion = { who = ROOT value = 0 } } } }
		}
	}
}
