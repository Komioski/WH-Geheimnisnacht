###############################################################################
# Kislev, the bulwark against Chaos: decisions
# Events: events/WHG_bulwark_events.txt (namespace bulwark).
#
#   bulwark_call_winter           An Ice Witch calls an early winter on the
#                                 invaders (bulwark.2)
#   bulwark_tzars_call            The Tzar(ina) calls the powers of order to
#                                 arms against Chaos (bulwark.60)
#   bulwark_brothers_in_the_north The Tzar(ina) offers the Emperor the old
#                                 treaty (bulwark.65)
#   bulwark_emperors_call         The Emperor calls on Kislev against Chaos
#                                 (bulwark.67)
###############################################################################

decisions = {
	# Call the Winter: an Ice Witch of rank 3 or more calls an early winter
	# on the enemies in her lord's Kislevite lands. Once in five years.
	bulwark_call_winter = {
		ai_check_interval = 12

		potential = {
			society_member_of = ice_witch_sisterhood
			society_rank >= 3
		}

		allow = {
			prisoner = no
			custom_tooltip = {
				text = bulwark_cost_300_tt
				society_currency >= 300
			}
			custom_tooltip = {
				text = bulwark_call_winter_cd_tt
				NOT = { has_character_modifier = bulwark_call_winter_cd }
			}
			top_liege = {
				war = yes
				any_realm_province = {
					region = world_old_world_kislev
					OR = { has_siege = yes is_occupied = yes }
				}
			}
		}

		effect = {
			change_society_currency = -300
			add_character_modifier = { name = bulwark_call_winter_cd years = 5 hidden = yes }
			character_event = { id = bulwark.2 }
		}

		ai_will_do = { factor = 1 }
	}

	# The Tzar's Call: envoys ride to the Emperor and the electors, the Dwarf
	# kings, the King of Bretonnia, the Phoenix King and the princes of Tilea
	# and Estalia. Once in ten years, while defending against Chaos.
	bulwark_tzars_call = {
		ai_check_interval = 6

		potential = {
			bulwark_kislev_crown_trigger = yes
		}

		allow = {
			prisoner = no
			NOT = { has_character_modifier = bulwark_tzars_call_cd }
			bulwark_defending_vs_chaos_trigger = yes
		}

		effect = {
			add_character_modifier = { name = bulwark_tzars_call_cd years = 10 }
			any_independent_ruler = {
				limit = { bulwark_order_power_trigger = yes }
				character_event = { id = bulwark.60 days = 10 random = 20 }
			}
		}

		ai_will_do = { factor = 1 }
	}

	# Brothers in the North: once there is an Emperor, the Tzar(ina) can offer
	# the old treaty, an alliance of the two crowns. It can be offered again
	# after either crown passes on.
	bulwark_brothers_in_the_north = {
		ai_check_interval = 24

		potential = {
			has_landed_title = e_kislev
			e_the_empire = { has_holder = yes }
		}

		allow = {
			prisoner = no
			NOT = { has_character_modifier = bulwark_brothers_cd }
			e_the_empire = {
				holder_scope = {
					NOT = { war_with = ROOT }
					NOT = { is_allied_with = ROOT }
					NOT = { religion_group = chaos_gods_group }
				}
			}
		}

		effect = {
			add_character_modifier = { name = bulwark_brothers_cd years = 10 hidden = yes }
			e_the_empire = {
				holder_scope = { character_event = { id = bulwark.65 } }
			}
		}

		ai_will_do = { factor = 1 }
	}

	# The Emperor's call: when the Empire faces Chaos, the Emperor calls on
	# the Tzar(ina). Once in ten years.
	bulwark_emperors_call = {
		ai_check_interval = 6

		potential = {
			has_landed_title = e_the_empire
			e_kislev = { has_holder = yes }
		}

		allow = {
			prisoner = no
			NOT = { has_character_modifier = bulwark_tzars_call_cd }
			bulwark_defending_vs_chaos_trigger = yes
			e_kislev = {
				holder_scope = { NOT = { war_with = ROOT } }
			}
		}

		effect = {
			add_character_modifier = { name = bulwark_tzars_call_cd years = 10 }
			e_kislev = {
				holder_scope = { character_event = { id = bulwark.67 days = 10 } }
			}
		}

		ai_will_do = { factor = 1 }
	}
}
