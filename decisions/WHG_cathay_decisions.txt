# Cathay decisions: borrowing from the caravan houses of the Ivory Road (the
# vanilla Jade Dragon Radhanite loan, lost when the mod replaced
# minor_decisions.txt, moved to Cathay's merchants).
# Events: events/WHG_cathay_diplomacy_events.txt (celestial.105 - 106).

decisions = {
	# Borrow 200 gold from the Cathayan caravan houses. The debt stays,
	# inherited, until repaid with interest, and the factors send reminders
	# (celestial.106).
	borrow_from_caravan_houses = {
		only_playable = yes
		ai_check_interval = 60

		potential = {
			has_dlc = "Jade Dragon"
			is_playable = yes
			is_nomadic = no
			NOT = { has_character_modifier = borrowed_from_radhanites }
			any_realm_province = {
				celestial_eastern_road_province_trigger = yes
			}
		}

		allow = {
			prisoner = no
			is_inaccessible_trigger = no
		}

		effect = {
			add_character_modifier = {
				name = borrowed_from_radhanites
				duration = -1
				inherit = yes
			}
			wealth = 200
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				wealth = 0
				war = no
			}
			modifier = {
				factor = 0
				wealth = 200
				war = yes
			}
			modifier = {
				factor = 0.01 # Slow it down
			}
		}
	}

	# Repay the caravan houses 250 gold and clear the debt.
	repay_caravan_houses = {
		only_playable = yes
		ai_check_interval = 60

		potential = {
			is_playable = yes
			has_character_modifier = borrowed_from_radhanites
		}

		allow = {
			wealth = 250
		}

		effect = {
			remove_character_modifier = borrowed_from_radhanites
			wealth = -250
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				NOT = { wealth = 350 }
			}
			modifier = {
				factor = 0.1 # Slow it down
			}
		}
	}
}
