# Cathay decisions: borrowing from the caravan houses of the Ivory Road (the
# vanilla Jade Dragon Radhanite loan, lost when the mod replaced
# minor_decisions.txt, moved to Cathay's merchants) and raiding Cathay (the
# vanilla raid decisions, lost the same way; under the hostile actions
# button of Cathay's window), and the Grace of Cathay boons, gifts and
# hostile acts (the vanilla grace decisions, lost the same way).
# Events: events/WHG_cathay_diplomacy_events.txt (celestial.105 - 106),
# events/WHG_cathay_raid_events.txt (cathayraid),
# events/WHG_cathay_grace_events.txt (cathaygrace).

decisions = {
	# Borrow 200 gold from the Cathayan caravan houses. The debt stays,
	# inherited, until repaid with interest, and the factors send reminders
	# (celestial.106).
	borrow_from_caravan_houses = {
		only_playable = yes
		ai_check_interval = 60

		potential = {
			has_dlc = "Jade Dragon"
			is_playable = yes
			is_nomadic = no
			NOT = { has_character_modifier = borrowed_from_radhanites }
			any_realm_province = { celestial_eastern_road_province_trigger = yes }
		}

		allow = { prisoner = no is_inaccessible_trigger = no }

		effect = {
			add_character_modifier = { name = borrowed_from_radhanites duration = -1 inherit = yes }
			wealth = 200
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 wealth = 0 war = no }
			modifier = { factor = 0 wealth = 200 war = yes }
			modifier = {
				factor = 0.01 # Slow it down
			}
		}
	}

	# Repay the caravan houses 250 gold and clear the debt.
	repay_caravan_houses = {
		only_playable = yes
		ai_check_interval = 60

		potential = { is_playable = yes has_character_modifier = borrowed_from_radhanites }

		allow = { wealth = 250 }

		effect = { remove_character_modifier = borrowed_from_radhanites wealth = -250 }

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 NOT = { wealth = 350 } }
			modifier = {
				factor = 0.1 # Slow it down
			}
		}
	}
}

offmap_decisions = {
	# Send raiders over Cathay's western border. Costs grace every month and
	# part of the levies; loot comes home every year (cathayraid.1). Only
	# for independent lords who may loot and sit near the roads east.
	cathay_raid = {
		only_playable = yes
		button_name = offmap_hostility
		ai_check_interval = 60

		potential = {
			is_offmap_tag = offmap_cathay
			FROM = {
				is_offmap_governor = no
				NOT = { religion = draconite }
				NOT = { has_character_modifier = cathay_raid_active }
				cathayraid_within_reach_trigger = yes
			}
		}

		allow = {
			show_only_failed_conditions = yes
			cathay_offmap_hostile_decisions_enabled_trigger = yes
			FROM = {
				show_scope_change = no
				independent = yes
				is_landed = yes
				is_allowed_to_loot = yes
				custom_tooltip = {
					text = cathay_offmap_boon_decisions_enabled_trigger_raiding_tt
					NOT = { has_character_modifier = cathay_raid_grace_cd }
				}
			}
		}

		effect = { FROM = { cathayraid_start_effect = yes } }

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 FROM = { war = yes } }
			modifier = {
				factor = 3
				FROM = {
					OR = {
						is_nomadic = yes
						culture_group = kurgan_group
						culture = hobgoblin
						culture_group = ogre_group
						religion_group = chaos_gods_group
					}
				}
			}
			modifier = {
				factor = 2
				offmap_cathay = {
					OR = {
						has_status = cathay_famine
						has_status = cathay_plague
						has_status = cathay_civil_war
						has_status = cathay_unrest
					}
				}
			}
			modifier = { factor = 0.2 offmap_cathay = { has_status = cathay_golden_age } }
			modifier = {
				factor = 0.1 # Slow it down
			}
		}
	}

	# Call the raiders home. No new raid for five years.
	cathay_stop_raid = {
		only_playable = yes
		button_name = offmap_hostility
		ai_check_interval = 60

		potential = {
			is_offmap_tag = offmap_cathay
			FROM = { has_character_modifier = cathay_raid_active }
		}

		allow = { always = yes }

		effect = { FROM = { cathayraid_end_effect = yes } }

		ai_will_do = {
			factor = 0
			modifier = {
				factor = 1
				FROM = {
					OR = {
						is_allowed_to_loot = no
						independent = no
						offmap_cathay = { has_status = cathay_golden_age }
					}
				}
			}
			modifier = {
				factor = 0.1 # Slow it down
			}
		}
	}
}
