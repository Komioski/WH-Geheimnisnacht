# There and Back Again: a halfling player who carries a ring all the way to

decisions = {
	cast_it_into_the_fire = {
		is_high_prio = yes
		ai = no

		potential = {
			ai = no
			NOT = { has_global_flag = deed_ring_bearer }
			OR = {
				culture = mootlander
				culture = miyan
			}
			any_artifact = { has_artifact_flag = ring }
			OR = {
				has_landed_title = c_mount_doom
				location = { county = { title = c_mount_doom } }
			}
		}

		allow = {
			prisoner = no
		}

		effect = {
			character_event = { id = chronicle.141 }
		}

		ai_will_do = {
			factor = 0
		}
	}
}
