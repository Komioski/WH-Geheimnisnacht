###############################################################################
# Priesthood of the Ancestors decisions
# The rank powers of ancestor_priesthood (listed under each rank in
# common/societies/WHG_religious_societies.txt). Events:
# events/WHG_dawi_priest_events.txt.
#
# Rank 1  dawipriest_tell_the_sagas      tell the tales of the ancestors (.10)
# Rank 2  dawipriest_take_a_calling      Hearth, Forge or Doom Priest (.12)
#         dawipriest_holy_day            keep the holy day of the calling (.13)
# Rank 3  dawipriest_counsel_the_king    speak at the king's council (.15)
#         dawipriest_bless_the_throng    bless the throng for war (.16)
#         dawipriest_tend_the_wounded    heal the wounded or sick (.17)
# Rank 4  dawipriest_master_rune         inscribe a god's master rune (.20)
#         dawipriest_consecrate_temple   consecrate a temple at the capital (.22)
###############################################################################

society_decisions = {
	dawipriest_tell_the_sagas = {
		ai_check_interval = 24

		potential = { society_member_of = ancestor_priesthood }

		allow = {
			prisoner = no
			is_inaccessible_trigger = no
			society_currency >= 50
			custom_tooltip = {
				text = dawipriest_cooldown_tt
				hidden_trigger = { NOT = { has_character_modifier = dawipriest_sagas_cooldown } }
			}
		}

		effect = {
			change_society_currency = -50
			add_character_modifier = { name = dawipriest_sagas_cooldown years = 3 hidden = yes }
			character_event = { id = dawipriest.10 }
		}

		ai_will_do = { factor = 0.5 }
	}

	dawipriest_take_a_calling = {
		ai_check_interval = 12

		potential = {
			society_member_of = ancestor_priesthood
			society_rank >= 2
			dawipriest_has_calling_trigger = no
		}

		allow = { prisoner = no }

		effect = { character_event = { id = dawipriest.12 } }

		ai_will_do = { factor = 1 }
	}

	dawipriest_holy_day = {
		ai_check_interval = 24

		potential = { society_member_of = ancestor_priesthood society_rank >= 2 }

		allow = {
			prisoner = no
			is_inaccessible_trigger = no
			war = no
			society_currency >= 150
			custom_tooltip = {
				text = dawipriest_cooldown_tt
				hidden_trigger = { NOT = { has_character_modifier = dawipriest_holy_day_cooldown } }
			}
		}

		effect = {
			change_society_currency = -150
			add_character_modifier = { name = dawipriest_holy_day_cooldown years = 5 hidden = yes }
			character_event = { id = dawipriest.13 }
		}

		ai_will_do = { factor = 0.5 }
	}

	dawipriest_counsel_the_king = {
		ai_check_interval = 24

		potential = {
			society_member_of = ancestor_priesthood
			society_rank >= 3
			OR = { is_ruler = yes liege = { is_ruler = yes } }
		}

		allow = {
			prisoner = no
			is_inaccessible_trigger = no
			society_currency >= 250
			custom_tooltip = {
				text = dawipriest_cooldown_tt
				hidden_trigger = { NOT = { has_character_modifier = dawipriest_council_cooldown } }
			}
		}

		effect = {
			change_society_currency = -250
			add_character_modifier = { name = dawipriest_council_cooldown years = 5 hidden = yes }
			character_event = { id = dawipriest.15 }
		}

		ai_will_do = { factor = 0.5 }
	}

	dawipriest_bless_the_throng = {
		ai_check_interval = 12

		potential = { society_member_of = ancestor_priesthood society_rank >= 3 is_ruler = yes }

		allow = {
			war = yes
			prisoner = no
			society_currency >= 300
			custom_tooltip = {
				text = dawipriest_cooldown_tt
				hidden_trigger = { NOT = { has_character_modifier = dawipriest_throng_cooldown } }
			}
		}

		effect = {
			change_society_currency = -300
			add_character_modifier = { name = dawipriest_throng_cooldown years = 5 hidden = yes }
			character_event = { id = dawipriest.16 }
		}

		ai_will_do = { factor = 1 }
	}

	dawipriest_tend_the_wounded = {
		ai_check_interval = 12

		potential = { society_member_of = ancestor_priesthood society_rank >= 3 }

		allow = {
			prisoner = no
			is_inaccessible_trigger = no
			society_currency >= 200
			custom_tooltip = {
				text = dawipriest_cooldown_tt
				hidden_trigger = { NOT = { has_character_modifier = dawipriest_healing_cooldown } }
			}
			custom_tooltip = {
				text = dawipriest_someone_hurt_tt
				OR = {
					trait = wounded
					trait = ill
					any_courtier = { OR = { trait = wounded trait = ill } }
				}
			}
		}

		effect = {
			change_society_currency = -200
			add_character_modifier = { name = dawipriest_healing_cooldown years = 2 hidden = yes }
			character_event = { id = dawipriest.17 }
		}

		ai_will_do = { factor = 1 }
	}

	dawipriest_master_rune = {
		ai_check_interval = 60

		potential = {
			society_member_of = ancestor_priesthood
			society_rank == 4
			NOT = { has_character_flag = dawipriest_rune_inscribed }
		}

		allow = {
			prisoner = no
			is_inaccessible_trigger = no
			dawipriest_has_calling_trigger = yes
			society_currency >= 1000
			NOT = { has_character_modifier = dawipriest_inscribing }
		}

		effect = {
			change_society_currency = -1000
			set_character_flag = dawipriest_rune_inscribed
			add_character_modifier = { name = dawipriest_inscribing years = 3 }
			custom_tooltip = { text = dawipriest_master_rune_tt }
			hidden_effect = { character_event = { id = dawipriest.20 days = 1095 } }
		}

		ai_will_do = { factor = 1 }
	}

	dawipriest_consecrate_temple = {
		ai_check_interval = 60

		potential = { society_member_of = ancestor_priesthood society_rank == 4 is_landed = yes }

		allow = {
			prisoner = no
			society_currency >= 750
			wealth = 100
			capital_scope = { NOT = { has_province_modifier = dawipriest_temple } }
		}

		effect = {
			change_society_currency = -750
			wealth = -100
			character_event = { id = dawipriest.22 }
		}

		ai_will_do = { factor = 1 }
	}
}
