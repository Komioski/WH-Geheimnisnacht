###############################################################################
# Engineers Guild decisions
# The guild's rank powers (listed under each rank in
# common/societies/WHG_guild_societies.txt), hiring an engineer, and founding
# the Imperial Engineers School. Events: events/WHG_engineer_events.txt.
#
# Rank 1  engineer_experiment             test a new device (engineer.10)
# Rank 2  engineer_training               become an Engineer / Dwarf Engineer
#         economic_engineering            cheaper, quicker building at the capital
#         military_engineering            a regiment of Thunderers for a war
#         engineer_hold_petition          Dawi of the holds: a design before the
#                                         Elders of Zhufbar (engineer.60)
#         engineer_dawi_teach_manlings    Dawi of the manling cities: teach a
#                                         manling the craft (engineer.63)
#         engineer_gun_christening        Gunnery School: christen a gun of Nuln
#                                         (engineer.65)
# Rank 3  deploy_war_machines             a battery of cannon for a war
#         dwarf_master_engineer_training  Dwarf Engineer to Master Engineer
#         engineer_school_notebooks       Engineers School: Leonardo's notebooks
#                                         (engineer.70)
# Rank 4  engineer_masterwork             a three-year masterwork (engineer.18)
#
# Not society powers:
#   hire_engineer                   a Dawi engineer of Zhufbar (engineer.30)
#   hire_imperial_engineer          a manling engineer of the Imperial school,
#                                   the Gunnery School or Tilea (engineer.32)
#   commission_war_machine          a court engineer builds a cannon battery
#                                   or a Steam Tank (engineer.33 - .35)
#   found_imperial_engineer_school  the lord of Altdorf opens the school
#                                   (engineer.40)
###############################################################################

society_decisions = {
	engineer_experiment = {
		ai_check_interval = 24

		potential = { society_member_of = engineers_guild }

		allow = {
			prisoner = no
			is_inaccessible_trigger = no
			NOT = { trait = incapable }
			NOT = { has_character_modifier = busy_engineer }
			custom_tooltip = {
				text = engineer_experiment_cooldown_tt
				hidden_trigger = { NOT = { has_character_modifier = engineer_experiment_cooldown } }
			}
		}

		effect = {
			custom_tooltip = { text = engineer_experiment_tt }
			add_character_modifier = { name = engineer_experiment_cooldown years = 2 hidden = yes }
			character_event = { id = engineer.10 }
		}

		ai_will_do = {
			factor = 0.1
			modifier = { factor = 0 has_injury_trigger = yes }
			modifier = { factor = 2 trait = ambitious }
		}
	}

	engineer_training = {
		ai_check_interval = 24

		potential = {
			society_member_of = engineers_guild
			society_rank >= 2
			NOR = { trait = engineer trait = dwarf_engineer trait = dwarf_engineer_master }
		}

		allow = { society_currency >= 250 }

		effect = {
			change_society_currency = -250
			if = { limit = { trait = creature_dwarf } add_trait = dwarf_engineer }
			else = { add_trait = engineer }
			character_event = { id = engineer.19 }
		}

		ai_will_do = { factor = 1 }
	}

	economic_engineering = {
		ai_check_interval = 36

		potential = { society_member_of = engineers_guild society_rank >= 2 is_landed = yes }

		allow = {
			society_currency >= 200
			capital_scope = { NOT = { has_province_modifier = engineering_improvements } }
		}

		effect = {
			change_society_currency = -200
			capital_scope = {
				add_province_modifier = { name = engineering_improvements years = 5 }
			}
		}

		ai_will_do = { factor = 0.5 modifier = { factor = 0 NOT = { wealth = 100 } } }
	}

	military_engineering = {
		ai_check_interval = 12

		potential = { society_member_of = engineers_guild society_rank >= 2 is_landed = yes }

		allow = {
			war = yes
			society_currency >= 300
			custom_tooltip = {
				text = engineer_war_cooldown_tt
				hidden_trigger = { NOT = { has_character_modifier = engineer_war_cooldown } }
			}
		}

		effect = {
			change_society_currency = -300
			add_character_modifier = { name = engineer_war_cooldown years = 5 hidden = yes }
			custom_tooltip = { text = engineer_thunderers_tt }
			hidden_effect = {
				capital_scope = {
					ROOT = {
						spawn_unit = {
							owner = ROOT
							home = PREV
							province = PREV
							disband_on_peace = yes
							maintenance_multiplier = 0.5
							attrition = 1.0
							troops = { gunpowder = { 300 300 } }
						}
					}
				}
			}
		}

		ai_will_do = { factor = 1 }
	}

	deploy_war_machines = {
		ai_check_interval = 12

		potential = { society_member_of = engineers_guild society_rank >= 3 is_landed = yes }

		allow = {
			war = yes
			society_currency >= 500
			custom_tooltip = {
				text = engineer_war_cooldown_tt
				hidden_trigger = { NOT = { has_character_modifier = engineer_war_cooldown } }
			}
		}

		effect = {
			change_society_currency = -500
			add_character_modifier = { name = engineer_war_cooldown years = 5 hidden = yes }
			custom_tooltip = { text = engineer_war_machines_tt }
			hidden_effect = {
				capital_scope = {
					ROOT = {
						spawn_unit = {
							owner = ROOT
							home = PREV
							province = PREV
							disband_on_peace = yes
							maintenance_multiplier = 0.5
							attrition = 1.0
							troops = { artillery = { 30 30 } gunpowder = { 200 200 } }
						}
					}
				}
			}
		}

		ai_will_do = { factor = 1 }
	}

	dwarf_master_engineer_training = {
		ai_check_interval = 24

		potential = { society_member_of = engineers_guild society_rank >= 3 trait = dwarf_engineer }

		allow = { learning = 18 society_currency >= 500 }

		effect = {
			change_society_currency = -500
			remove_trait = dwarf_engineer
			add_trait = dwarf_engineer_master
			character_event = { id = engineer.19 }
		}

		ai_will_do = { factor = 1 }
	}

	engineer_masterwork = {
		ai_check_interval = 36

		potential = { society_member_of = engineers_guild society_rank == 4 }

		allow = {
			prisoner = no
			is_inaccessible_trigger = no
			NOT = { has_character_modifier = busy_engineer }
			society_currency >= 800
		}

		effect = {
			change_society_currency = -800
			add_character_modifier = { name = busy_engineer years = 3 }
			custom_tooltip = { text = engineer_masterwork_tt }
			hidden_effect = { character_event = { id = engineer.18 days = 1095 } }
		}

		ai_will_do = { factor = 0.5 }
	}

	# The Dawi of the holds: put a new design before the Elders of Zhufbar.
	engineer_hold_petition = {
		ai_check_interval = 36

		potential = {
			society_member_of = engineers_guild
			society_rank >= 2
			has_character_modifier = engineer_branch_hold
		}

		allow = {
			prisoner = no
			is_inaccessible_trigger = no
			society_currency >= 100
			custom_tooltip = {
				text = engineer_petition_cooldown_tt
				hidden_trigger = { NOT = { has_character_modifier = engineer_petition_cooldown } }
			}
		}

		effect = {
			change_society_currency = -100
			add_character_modifier = { name = engineer_petition_cooldown years = 5 hidden = yes }
			custom_tooltip = { text = engineer_hold_petition_tt }
			hidden_effect = { character_event = { id = engineer.60 days = 30 } }
		}

		ai_will_do = {
			factor = 0.3
			modifier = { factor = 2 trait = ambitious }
			modifier = { factor = 0.5 trait = content }
		}
	}

	# The Dawi of the manling cities: teach a manling of the court the craft,
	# but never the runes. The guild takes its tithe.
	engineer_dawi_teach_manlings = {
		ai_check_interval = 36

		potential = {
			society_member_of = engineers_guild
			society_rank >= 2
			has_character_modifier = engineer_branch_manling_dawi
		}

		allow = {
			prisoner = no
			society_currency >= 100
			wealth = 25
			custom_tooltip = {
				text = engineer_no_pupil_tt
				engineer_has_manling_pupil_trigger = yes
			}
			custom_tooltip = {
				text = engineer_teach_cooldown_tt
				hidden_trigger = { NOT = { has_character_modifier = engineer_teach_cooldown } }
			}
		}

		effect = {
			change_society_currency = -100
			wealth = -25
			add_character_modifier = { name = engineer_teach_cooldown years = 5 hidden = yes }
			custom_tooltip = { text = engineer_dawi_teach_manlings_tt }
			hidden_effect = { character_event = { id = engineer.63 } }
		}

		ai_will_do = {
			factor = 0.3
			modifier = { factor = 0 NOT = { wealth = 75 } }
			modifier = { factor = 0.5 trait = greedy }
		}
	}

	# The Gunnery School of Nuln: christen a newly cast gun and read the oil.
	engineer_gun_christening = {
		ai_check_interval = 36

		potential = {
			society_member_of = engineers_guild
			society_rank >= 2
			has_character_modifier = engineer_branch_gunnery
			is_landed = yes
		}

		allow = {
			prisoner = no
			society_currency >= 150
			custom_tooltip = {
				text = engineer_christening_cooldown_tt
				hidden_trigger = {
					NOT = { has_character_modifier = engineer_christening_cooldown }
				}
			}
		}

		effect = {
			change_society_currency = -150
			add_character_modifier = { name = engineer_christening_cooldown years = 5 hidden = yes }
			custom_tooltip = { text = engineer_gun_christening_tt }
			hidden_effect = { character_event = { id = engineer.65 } }
		}

		ai_will_do = { factor = 0.5 modifier = { factor = 3 war = yes } }
	}

	# The Imperial Engineers School: two years over Leonardo of Miragliano's
	# notebooks, written backwards and half in cipher.
	engineer_school_notebooks = {
		ai_check_interval = 36

		potential = {
			society_member_of = engineers_guild
			society_rank >= 3
			has_character_modifier = engineer_branch_school
		}

		allow = {
			prisoner = no
			is_inaccessible_trigger = no
			society_currency >= 300
			NOT = { has_character_modifier = busy_engineer }
		}

		effect = {
			change_society_currency = -300
			add_character_modifier = { name = busy_engineer years = 2 }
			add_character_modifier = { name = engineer_reading_notebooks years = 2 }
			custom_tooltip = { text = engineer_school_notebooks_tt }
			hidden_effect = { character_event = { id = engineer.70 days = 730 } }
		}

		ai_will_do = { factor = 0.3 }
	}
}

decisions = {
	# The guild sends a Dawi engineer of Zhufbar to the court.
	hire_engineer = {
		only_playable = yes
		ai_check_interval = 60

		potential = {
			higher_tier_than = BARON
			OR = {
				culture_group = dwarf_group
				culture_group = tilean_group
				AND = {
					culture_group = imperial_group
					has_global_flag = imperial_engineering_school
				}
			}
		}

		allow = {
			wealth = 200
			custom_tooltip = {
				text = engineer_hire_cooldown_tt
				hidden_trigger = { NOT = { has_character_modifier = hired_engineer } }
			}
		}

		effect = {
			wealth = -200
			add_character_modifier = { name = hired_engineer years = 10 }
			custom_tooltip = { text = engineer_hire_tt }
			hidden_effect = { character_event = { id = engineer.30 } }
		}

		ai_will_do = { factor = 0.2 modifier = { factor = 0 NOT = { wealth = 400 } } }
	}

	# Founded once, by the lord of Altdorf, who paid for the school in the
	# lore: opens the guild to Imperial humans.
	found_imperial_engineer_school = {
		only_playable = yes
		ai_check_interval = 60

		potential = {
			culture_group = imperial_group
			NOT = { has_global_flag = imperial_engineering_school }
			OR = {
				has_landed_title = c_altdorf
				has_landed_title = d_altdorf
				has_landed_title = k_reikland
			}
		}

		allow = { wealth = 1000 }

		effect = {
			wealth = -1000
			custom_tooltip = { text = found_imperial_engineer_school_tt }
			character_event = { id = engineer.40 }
		}

		ai_will_do = { factor = 0.5 modifier = { factor = 0 NOT = { wealth = 1200 } } }
	}

	# A manling engineer for the court: from the Imperial Engineers School
	# once it is open, from the Gunnery School of Nuln, or from Tilea, home of
	# the great manling engineers.
	hire_imperial_engineer = {
		only_playable = yes
		ai_check_interval = 60

		potential = {
			higher_tier_than = BARON
			trait = creature_human
			engineer_manling_court_trigger = yes
			OR = {
				has_global_flag = imperial_engineering_school
				engineer_gunnery_school_open_trigger = yes
				culture_group = tilean_group
			}
		}

		allow = {
			wealth = 150
			custom_tooltip = {
				text = engineer_hire_cooldown_tt
				hidden_trigger = { NOT = { has_character_modifier = hired_engineer } }
			}
		}

		effect = {
			wealth = -150
			add_character_modifier = { name = hired_engineer years = 10 }
			custom_tooltip = { text = engineer_hire_imperial_tt }
			hidden_effect = { character_event = { id = engineer.32 } }
		}

		ai_will_do = {
			factor = 0.2
			modifier = { factor = 0 NOT = { wealth = 300 } }
			modifier = { factor = 2 war = yes }
		}
	}

	# Set the court's engineer to building war machines at the capital. Any
	# ruler with an engineer at court may order a battery of great cannon; a
	# Steam Tank needs an Imperial court and the Imperial school.
	commission_war_machine = {
		ai_check_interval = 24

		potential = { is_landed = yes engineer_has_court_engineer_trigger = yes }

		allow = {
			prisoner = no
			wealth = 100
			custom_tooltip = {
				text = engineer_machine_busy_tt
				engineer_machine_in_work_trigger = no
			}
		}

		effect = { character_event = { id = engineer.33 } }

		ai_will_do = {
			factor = 0.5
			modifier = { factor = 0 NOT = { wealth = 250 } }
			modifier = { factor = 3 war = yes }
		}
	}
}
