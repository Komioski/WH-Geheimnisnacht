###############################################################################
# Kislev: the Ice Witches and the hags, decisions
# Events: events/WHG_ice_witch_events.txt (namespace frostwitch).
# The Sisterhood's other powers: kislevsoc_frost_ward (rank 2) in
# WHG_kislev_society_decisions.txt and bulwark_call_winter (rank 3) in
# WHG_bulwark_decisions.txt.
#
#   frost_great_warding    Rank 4: the Great Warding at the Lynsk, once per
#                          witch (frostwitch.1)
#   frost_hunt_sorcerer    Rank 2: hunt a Chaos sorcerer (frostwitch.10)
#   frost_freeze_river     Rank 2: freeze a river for the lord's armies
#                          (frostwitch.20)
#   frost_seal_breach      Rank 3: seal a warp-breach in a war against
#                          Chaos (frostwitch.25)
#   frost_train_novice     Rank 3: take a novice onto the Frosthome glacier
#                          (frostwitch.35)
#   frost_hag_mother       An Ungol or Ropsmann lord or hag seeks the
#                          Hag-Mother, once a lifetime (frostwitch.60)
###############################################################################

decisions = {
	# The Great Warding: a Grand Ice Witch pours her life into the ice of the
	# Lynsk. Once per witch.
	frost_great_warding = {
		ai_check_interval = 24

		potential = {
			society_member_of = ice_witch_sisterhood
			society_rank >= 4
			NOT = { has_character_flag = frost_warding_done }
		}

		allow = {
			prisoner = no
			is_incapable = no
			custom_tooltip = {
				text = frost_cost_400_tt
				society_currency >= 400
			}
			top_liege = {
				any_realm_province = { region = world_old_world_kislev }
			}
		}

		effect = {
			change_society_currency = -400
			set_character_flag = frost_warding_done
			character_event = { id = frostwitch.1 }
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.2 is_doom_very_low_trigger = yes }
			modifier = { factor = 2 is_doom_high_trigger = yes }
			modifier = { factor = 3 is_doom_very_high_trigger = yes }
		}
	}

	# Hunt a Chaos sorcerer in the oblast. Once in five years.
	frost_hunt_sorcerer = {
		ai_check_interval = 12

		potential = {
			society_member_of = ice_witch_sisterhood
			society_rank >= 2
		}

		allow = {
			prisoner = no
			is_incapable = no
			custom_tooltip = {
				text = frost_cost_100_tt
				society_currency >= 100
			}
			custom_tooltip = {
				text = frost_hunt_cd_tt
				NOT = { has_character_modifier = frost_hunt_cd }
			}
		}

		effect = {
			change_society_currency = -100
			add_character_modifier = { name = frost_hunt_cd years = 5 hidden = yes }
			character_event = { id = frostwitch.10 days = 7 }
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.3 trait = craven }
		}
	}

	# Freeze a river so the lord's armies can cross. Once in three years, in
	# a war.
	frost_freeze_river = {
		ai_check_interval = 6

		potential = {
			society_member_of = ice_witch_sisterhood
			society_rank >= 2
		}

		allow = {
			prisoner = no
			custom_tooltip = {
				text = frost_cost_150_tt
				society_currency >= 150
			}
			custom_tooltip = {
				text = frost_river_cd_tt
				NOT = { has_character_modifier = frost_river_cd }
			}
			top_liege = {
				war = yes
				NOT = { has_character_modifier = frost_river_frozen }
			}
		}

		effect = {
			change_society_currency = -150
			add_character_modifier = { name = frost_river_cd years = 3 hidden = yes }
			top_liege = { character_event = { id = frostwitch.20 } }
		}

		ai_will_do = { factor = 1 }
	}

	# Seal a warp-breach while Chaos is in the land. Once in five years.
	frost_seal_breach = {
		ai_check_interval = 6

		potential = {
			society_member_of = ice_witch_sisterhood
			society_rank >= 3
		}

		allow = {
			prisoner = no
			is_incapable = no
			custom_tooltip = {
				text = frost_cost_150_tt
				society_currency >= 150
			}
			custom_tooltip = {
				text = frost_breach_cd_tt
				NOT = { has_character_modifier = frost_breach_cd }
			}
			top_liege = { bulwark_defending_vs_chaos_trigger = yes }
		}

		effect = {
			change_society_currency = -150
			add_character_modifier = { name = frost_breach_cd years = 5 hidden = yes }
			character_event = { id = frostwitch.25 days = 7 }
		}

		ai_will_do = { factor = 1 }
	}

	# Take a novice of the court onto the Frosthome glacier for a winter.
	# Once in five years.
	frost_train_novice = {
		ai_check_interval = 12

		potential = {
			society_member_of = ice_witch_sisterhood
			society_rank >= 3
		}

		allow = {
			prisoner = no
			is_incapable = no
			custom_tooltip = {
				text = frost_cost_100_tt
				society_currency >= 100
			}
			custom_tooltip = {
				text = frost_novice_cd_tt
				NOT = { has_character_modifier = frost_novice_cd }
			}
			custom_tooltip = {
				text = frost_novice_at_court_tt
				host = {
					any_courtier = { frost_novice_trigger = yes }
				}
			}
		}

		effect = {
			change_society_currency = -100
			add_character_modifier = { name = frost_novice_cd years = 5 hidden = yes }
			host = {
				random_courtier = {
					limit = { frost_novice_trigger = yes }
					save_event_target_as = frost_novice
				}
			}
			character_event = { id = frostwitch.35 months = 6 }
		}

		ai_will_do = { factor = 1 }
	}

	# Seek the Hag-Mother's walking hut in the deep woods. Once a lifetime.
	frost_hag_mother = {
		ai_check_interval = 60

		potential = {
			OR = {
				culture = ungol
				culture = ropsmann
			}
			OR = {
				is_ruler = yes
				frost_hag_trigger = yes
			}
			NOT = { has_character_flag = frost_sought_hag_mother }
		}

		allow = {
			is_adult = yes
			prisoner = no
			is_incapable = no
			war = no
		}

		effect = {
			set_character_flag = frost_sought_hag_mother
			character_event = { id = frostwitch.60 days = 60 }
		}

		ai_will_do = {
			factor = 0.2
			modifier = { factor = 3 frost_hag_trigger = yes }
			modifier = { factor = 0.3 trait = craven }
		}
	}
}
