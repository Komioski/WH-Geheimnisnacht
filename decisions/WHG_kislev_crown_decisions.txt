###############################################################################
# Kislev crown and court decisions
# Events: events/WHG_kislev_crown_events.txt (namespace tsar),
# events/WHG_chekist_events.txt (namespace chekist).
#
#   tsar_call_great_pulk     The Tzar(ina) calls the Great Pulk against Chaos
#                            (tsar.41 to every Kislevite vassal)
#   tsar_raise_ice_palace    An Ice Witch on the throne raises the Ice Palace
#                            again (tsar.15)
#   chekist_arrest           The Court Chekist arrests a cultist, a mutant, a
#                            suspect or a dissenting vassal (chekist.5)
###############################################################################

decisions = {
	# Call the Great Pulk: when a Chaos host invades the Tsardom, the crown
	# calls every boyar to send his rota. The pulk disbands at the peace.
	tsar_call_great_pulk = {
		ai_check_interval = 6

		potential = {
			has_landed_title = e_kislev
		}

		allow = {
			NOT = { has_character_modifier = tsar_pulk_cd }
			prisoner = no
			any_war = {
				defender = { character = ROOT }
				attacker = { religion_group = chaos_gods_group }
			}
		}

		effect = {
			add_character_modifier = { name = tsar_pulk_cd years = 5 }
			capital_scope = {
				ROOT = {
					spawn_unit = {
						province = PREV
						owner = ROOT
						troops = {
							knights = { 300 300 }
							light_cavalry = { 400 400 }
							archers = { 300 300 }
						}
						earmark = tsar_great_pulk
						disband_on_peace = yes
						maintenance_multiplier = 0.5
					}
				}
			}
			any_vassal = {
				limit = {
					higher_tier_than = BARON
					culture_group = kislevite_group
					prisoner = no
					is_adult = yes
				}
				character_event = { id = tsar.41 days = 3 random = 10 }
			}
		}

		ai_will_do = { factor = 1 }
	}

	# Raise the Ice Palace again: an Ice Witch who sits the throne can call
	# the palace back out of the frost. Piety, and great prestige.
	tsar_raise_ice_palace = {
		ai_check_interval = 24

		potential = {
			has_landed_title = e_kislev
			trait = ice_witch
			has_global_flag = ice_palace_melted
		}

		allow = {
			piety = 500
			prisoner = no
			war = no
			society_member_of = ice_witch_sisterhood
			society_rank >= 3
		}

		effect = {
			piety = -500
			character_event = { id = tsar.15 }
		}

		ai_will_do = { factor = 1 }
	}
}

targeted_decisions = {
	# Order the Court Chekist to arrest someone of the court or a vassal: a
	# cultist, a mutant, someone under suspicion, or a vassal who speaks
	# against the crown. Once in three years.
	# ROOT is the one arrested, FROM the ruler.
	chekist_arrest = {
		filter = realm
		ai_target_filter = realm
		ai_check_interval = 12

		from_potential = {
			is_ruler = yes
			any_courtier = { has_minor_title = title_court_chekist }
		}

		potential = {
			NOT = { character = FROM }
			is_adult = yes
			prisoner = no
			NOT = { has_minor_title = title_court_chekist }
			OR = {
				host = { character = FROM }
				liege = { character = FROM }
			}
		}

		allow = {
			FROM = { NOT = { has_character_modifier = chekist_arrest_cd } }
			OR = {
				is_cultist_trigger = yes
				is_mutated_trigger = yes
				whunter_suspect_courtier_trigger = yes
				AND = {
					liege = { character = FROM }
					opinion = { who = FROM value < -25 }
				}
			}
		}

		effect = {
			save_event_target_as = chekist_quarry
			FROM = { add_character_modifier = { name = chekist_arrest_cd years = 3 hidden = yes } }
			if = {
				limit = {
					NOR = {
						is_cultist_trigger = yes
						is_mutated_trigger = yes
						whunter_suspect_courtier_trigger = yes
					}
				}
				FROM = { set_character_flag = chekist_took_dissenter }
			}
			imprison = FROM
			FROM = { character_event = { id = chekist.5 } }
		}

		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				NOR = {
					is_cultist_trigger = yes
					is_mutated_trigger = yes
				}
			}
		}
	}
}
