decisions = {
	# Relight the hearth: send to the temple in Kislev for Dazh's flame. The
	# same as the pilgrimage option of ursun.31, for a lord who waited.
	ursun_relight_hearth = {
		ai_check_interval = 12

		potential = {
			has_character_modifier = ursun_cold_hearths
		}

		allow = {
			prisoner = no
			wealth = 75
			NOT = { has_character_flag = ursun_flame_sent }
		}

		effect = {
			wealth = -75
			set_character_flag = ursun_flame_sent
			character_event = { id = ursun.32 days = 120 }
		}

		ai_will_do = { factor = 1 }
	}

	# Miska's Barrow: once a lifetime, a Gospodar lord rides out onto the
	# steppe to the barrow of the Khan-Queen.
	ursun_miska_pilgrimage = {
		ai_check_interval = 60

		potential = {
			is_ruler = yes
			culture = gospodar
			NOT = { has_character_flag = ursun_miska_visited }
		}

		allow = {
			is_adult = yes
			prisoner = no
			war = no
			wealth = 50
		}

		effect = {
			wealth = -50
			set_character_flag = ursun_miska_visited
			character_event = { id = ursun.50 days = 60 }
		}

		ai_will_do = {
			factor = 0.2
			modifier = { factor = 3 trait = zealous }
		}
	}
}
