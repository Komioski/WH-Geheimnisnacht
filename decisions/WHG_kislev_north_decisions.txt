###############################################################################
# Kislev: war and the north, decisions
# Events: events/WHG_bulwark_events.txt (namespace kislevnorth).
#
#   north_raise_urskin_bear  A Tzar raises a bear of Urskin's line, once per
#                            reign (kislevnorth.1)
#   north_hire_out_kossars   Hire a Kossar company out to a southern prince
#                            for ten years (kislevnorth.30)
#   north_recall_kossars     Call the Kossars home early when Chaos strikes
###############################################################################

decisions = {
	# Urskin's Line: the Tzar's bear-master raises a great bear of the line of
	# Urskin. One per Tzar; the bear dies with its rider or goes back to the
	# wilds (kislevnorth.2).
	north_raise_urskin_bear = {
		ai_check_interval = 24

		potential = {
			has_landed_title = e_kislev
			is_female = no
			NOT = { has_character_flag = north_urskin_raised }
		}

		allow = {
			is_adult = yes
			prisoner = no
			wealth = 300
		}

		effect = {
			wealth = -300
			set_character_flag = north_urskin_raised
			character_event = { id = kislevnorth.1 days = 30 }
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 NOT = { wealth = 500 } }
		}
	}

	# Kossar contracts: hire a company of Kossars out to a southern prince
	# for ten years. Gold now; fewer levies at home until they return.
	north_hire_out_kossars = {
		ai_check_interval = 24

		potential = {
			is_ruler = yes
			culture_group = kislevite_group
			higher_tier_than = BARON
			NOT = { has_character_flag = north_kossars_out }
		}

		allow = {
			prisoner = no
			war = no
			custom_tooltip = {
				text = north_kossars_cd_tt
				NOT = { has_character_modifier = north_kossars_cd }
			}
		}

		effect = {
			wealth = 200
			set_character_flag = north_kossars_out
			add_character_modifier = { name = north_kossars_abroad years = 10 }
			add_character_modifier = { name = north_kossars_cd years = 15 hidden = yes }
			character_event = { id = kislevnorth.30 years = 10 }
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 is_doom_high_trigger = yes }
			modifier = { factor = 0 is_doom_very_high_trigger = yes }
			modifier = { factor = 0 bulwark_northern_lord_trigger = yes }
		}
	}

	# Call the Kossars home early: when Chaos strikes at home, buy out the
	# contract and bring them back.
	north_recall_kossars = {
		ai_check_interval = 3

		potential = {
			has_character_flag = north_kossars_out
		}

		allow = {
			prisoner = no
			wealth = 150
			bulwark_defending_vs_chaos_trigger = yes
		}

		effect = {
			wealth = -150
			clr_character_flag = north_kossars_out
			remove_character_modifier = north_kossars_abroad
			custom_tooltip = { text = north_kossars_recalled_tt }
		}

		ai_will_do = { factor = 1 }
	}
}
