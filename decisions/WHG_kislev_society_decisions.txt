	# Rank up. Rank 2 needs a bear slain by your own hand (kislevsoc.11);
	# rank 3 a Circle trial (kislevsoc.20). The First is chosen by the Circle.
	kislevsoc_circle_rank_up = {
		hide_in_decisions_list = yes # Shown in the society view
		ai_check_interval = 12

		potential = {
			society_member_of = circle_of_the_bear
			society_rank < 3
			days_in_society > 180
		}

		allow = {
			trigger_if = {
				limit = { society_rank == 1 }
				custom_tooltip = {
					text = kislevsoc_cost_500_tt
					society_currency >= 500
				}
				custom_tooltip = {
					text = kislevsoc_bear_slain_tt
					has_character_flag = kislevsoc_bear_slain
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = kislevsoc_cost_1000_tt
					society_currency >= 1000
				}
				custom_tooltip = {
					text = kislevsoc_circle_trial_tt
					has_character_flag = kislevsoc_circle_trial_done
				}
			}
			custom_tooltip = {
				text = kislevsoc_rank_open_tt
				kislevsoc_next_rank_open_trigger = yes
			}
			prisoner = no
		}

		effect = {
			if = {
				limit = { society_rank == 1 }
				change_society_currency = -500
			}
			else = {
				change_society_currency = -1000
				clr_character_flag = kislevsoc_circle_trial_done
			}
			society_rank_up = 1
		}

		ai_will_do = { factor = 1 }
	}

targetted_decisions = {
	# Raise a Kislevite courtier to the druzhina: the crown's new nobility.
	# Costs gold, wins the courtier's love, and the old boyars sneer.
	kislevsoc_grant_druzhina = {
		ai_check_interval = 60
		filter = court
		ai_target_filter = court

		from_potential = {
			society_member_of = boyar_duma
			society_rank >= 2
			is_ruler = yes
		}

		potential = {
			host = { character = FROM }
			NOT = { character = FROM }
			culture_group = kislevite_group
			is_adult = yes
			is_ruler = no
			is_in_society = no
			prisoner = no
			trait = creature_human
		}

		allow = {
			FROM = {
				wealth = 100
				custom_tooltip = {
					text = kislevsoc_druzhina_cd_tt
					NOT = { has_character_modifier = kislevsoc_druzhina_cd }
				}
			}
		}

		effect = {
			FROM = {
				wealth = -100
				add_character_modifier = { name = kislevsoc_druzhina_cd years = 5 hidden = yes }
				random_vassal = {
					limit = {
						culture_group = kislevite_group
						society_member_of = boyar_duma
						society_rank >= 2
					}
					opinion = { who = FROM modifier = opinion_kislevsoc_upstart_raised }
				}
			}
			set_character_flag = kislevsoc_druzhina_raised
			join_society = boyar_duma
			prestige = 100
			opinion = { who = FROM modifier = opinion_kislevsoc_raised_me }
		}

		ai_will_do = {
			factor = 0.2
			modifier = { factor = 2 FROM = { wealth = 300 } }
			modifier = { factor = 2 diplomacy = 12 }
			modifier = { factor = 2 stewardship = 12 }
		}
	}
