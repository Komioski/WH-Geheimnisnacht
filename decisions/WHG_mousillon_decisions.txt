decisions = {
	# Ride to Landuin's tomb in the ruined Chapel of the Grail. The worthy
	# may be blessed, the unworthy shamed; a master of the dead who dares
	# it meets the ghost's sword. Once in twenty years.
	mousillon_landuin_pilgrimage = {
		only_playable = yes
		ai_check_interval = 60

		potential = {
			OR = {
				religion = grail
				religion = false_grail
				culture_group = bretonnian_group
				AND = {
					undead_master_trigger = yes
					mousillon_resident_trigger = yes
				}
			}
		}

		allow = {
			is_adult = yes
			prisoner = no
			war = no
			wealth = 50
			is_inaccessible_trigger = no
			NOT = { has_character_modifier = mousillon_pilgrimage }
			NOT = { has_character_modifier = mousillon_pilgrimage_cd }
		}

		effect = {
			wealth = -50
			add_character_modifier = { name = mousillon_pilgrimage days = 120 }
			add_character_modifier = { name = mousillon_pilgrimage_cd years = 20 hidden = yes }
			character_event = { id = mousillon.20 days = 60 random = 30 }
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 0.1
			modifier = { factor = 3 trait = zealous }
			modifier = { factor = 2 trait = grail_knight }
			modifier = { factor = 0 undead_master_trigger = yes }
			modifier = { factor = 0 NOT = { wealth = 150 } }
		}
	}
}
