# Old World faith decisions: the Feast of the Blazing Sun, the vanilla
# Zunist Festival of the Sun (hold_zun_feast in vanilla minor_decisions)
# moved to Myrmidia.
# Also the declaration of repentance (issue_declaration_of_repentance) and
# the purchase of an indulgence (buy_indulgence_for_sins, faith.250), both
# restored from vanilla realm_decisions.txt.
# Events: events/WHG_old_world_faith_events.txt (sunfest.1 - 41, penance.3).

decisions = {
	# Hold the Feast of the Blazing Sun in high summer.
	hold_blazing_sun_festival = {
		only_rulers = yes
		is_high_prio = yes
		ai_check_interval = 12

		potential = {
			is_ruler = yes
			prisoner = no
			religion = myrmidian
			is_incapable = no
			num_of_vassals = 1
			NOT = { has_character_modifier = holding_blazing_sun_festival }
			NOT = { has_character_modifier = large_feast_timer }
			NOT = { has_character_modifier = fabulous_feast }
			NOT = { liege = { has_character_modifier = holding_blazing_sun_festival } }
			NOT = { has_character_flag = do_not_disturb }
		}

		allow = {
			war = no
			prisoner = no
			is_inaccessible_trigger = no
			custom_tooltip = {
				text = UNOCCUPIED_DEMESNE_TITLE
				hidden_tooltip = {
					any_demesne_title = { NOT = { higher_tier_than = COUNT } is_occupied = no }
				}
			}
			scaled_wealth = 0.2
			month = 5
			NOT = { month = 8 }
			prestige = 100
		}

		effect = {
			scaled_wealth = -0.2
			add_character_modifier = { name = holding_blazing_sun_festival duration = 365 }
			set_character_flag = recent_minor_decision
			set_character_flag = do_not_disturb
			hidden_tooltip = { character_event = { id = sunfest.1 } }
			custom_tooltip = {
				text = capital_prospers_custom_tooltip
				hidden_tooltip = {
					capital_scope = { change_variable = { which = prosperity_value value = 8 } }
				}
			}
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 0.1
			modifier = { factor = 0 NOT = { scaled_wealth = 2 } }
			modifier = { factor = 3 trait = zealous }
			modifier = { factor = 2 trait = gregarious }
			modifier = { factor = 0.5 trait = shy }
		}
	}

	# Restored from vanilla realm_decisions.txt (it was never copied into the
	# mod, so penance.3 - 23 could never fire). An excommunicated lord asks the
	# head of his faith to name the price of absolution (penance.3).
	issue_declaration_of_repentance = {
		only_playable = yes
		ai_check_interval = 80
		is_high_prio = yes

		potential = {
			age = 16
			is_playable = yes
			is_landed = yes
			higher_tier_than = BARON # Barons need someone else to ask for them
			rightful_religious_head_scope = { ROOT = { excommunicated_for = PREV } }
		}

		allow = {
			is_heretic = no
			custom_tooltip = {
				text = tooltip_issued_decla_repentance_recently
				hidden_tooltip = {
					NOT = { has_character_modifier = asked_repentance_from_sins_cooldown }
				}
			}
		}

		effect = {
			hidden_tooltip = {
				add_character_modifier = {
					name = asked_repentance_from_sins_cooldown
					months = 48
					hidden = yes
				}
			}
			rightful_religious_head_scope = {
				character_event = { id = penance.3 days = 5 random = 12 tooltip = "repentance_ask" }
			}
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.25 NOT = { scaled_wealth = 2.0 } }
			modifier = { factor = 0 trait = stubborn }
			modifier = { factor = 0 trait = cynical }
			modifier = { factor = 2 trait = zealous }
		}
	}

	# A pious ruler of the Old World pays the head of the faith to forgive
	# their sins (vanilla, cut off when the mod dropped the Catholic Church).
	buy_indulgence_for_sins = {
		only_playable = yes
		ai_check_interval = 6

		potential = {
			religion_group = old_world_group
			age = 16
			demesne_size = 1
			NOT = { has_character_flag = ask_for_indulgence }
			NOT = { trait = excommunicated }
			rightful_religious_head_scope = { always = yes }
		}

		allow = {
			prisoner = no
			NOT = { trait = incapable }
			is_inaccessible_trigger = no
			scaled_wealth = 1.0
			wealth = 25
			is_heretic = no
		}

		effect = {
			set_character_flag = ask_for_indulgence
			rightful_religious_head_scope = {
				character_event = { id = faith.250 days = 5 random = 12 tooltip = "indulgence_ask" }
			}
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 OR = { NOT = { scaled_wealth = 10.0 } piety = 0 } }
		}
	}
}
