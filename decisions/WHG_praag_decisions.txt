# Praag decisions
# Events: events/WHG_praag_events.txt (namespace praag).
#
#   praag_ease_curse   The Duma rebuilds cursed Praag: the curse lifts for 25
#                      years (praag.4 brings it back)

decisions = {
	# Rebuild cursed Praag: new stone over the screaming walls. The holder,
	# or a senior Duma boyar of Kislev (who gains Duma standing for it).
	praag_ease_curse = {
		ai_check_interval = 24

		potential = {
			praag_cursed_trigger = yes
			OR = {
				has_landed_title = c_praag
				AND = {
					society_member_of = boyar_duma
					society_rank >= 2
					c_praag = {
						holder_scope = {
							OR = {
								has_landed_title = e_kislev
								top_liege = { has_landed_title = e_kislev }
							}
						}
					}
				}
			}
		}

		allow = {
			wealth = 300
			prisoner = no
			war = no
		}

		effect = {
			wealth = -300
			prestige = 100
			if = {
				limit = { society_member_of = boyar_duma }
				change_society_currency = 300
			}
			182 = {
				remove_province_modifier = praag_curse
				add_province_modifier = { name = praag_curse_eased years = 25 }
			}
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 NOT = { wealth = 500 } }
		}
	}
}
