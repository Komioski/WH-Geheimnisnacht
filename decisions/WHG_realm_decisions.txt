decisions = {

	realm_refound_eagle = {
		only_independent = yes
		is_high_prio = yes

			is_landed = yes

				trigger = { ai = no }
				prestige = 500
			}
			wealth = 150
		}

			wealth = -150

			any_demesne_title = {
				limit = { tier = DUKE }
