# The decisions below restore the rest of vanilla HFP_realm_decisions on
# the mod's own titles (events realm.30-60).
decisions = {
	# The Eastern Marches: the Elector of Ostermark proclaims the East March.
	realm_east_march = {
		is_high_prio = yes
		only_playable = yes
		ai_check_interval = 60

		potential = {
			has_landed_title = k_ostermark
			culture_group = imperial_group
			religion_group = old_world_group
			NOT = { has_global_flag = east_march_proclaimed }
		}

		allow = {
			is_adult = yes
			war = no
			completely_controls = d_bechafen
			realm_size = 15
			conditional_tooltip = {
				trigger = { ai = no }
				prestige = 1000
			}
			wealth = 300
			trigger_if = {
				limit = { independent = no }
				custom_tooltip = {
					text = realm_emperor_approves_tt
					top_liege = { opinion = { who = ROOT value = 25 } }
				}
			}
		}

		effect = {
			wealth = -300
			set_global_flag = east_march_proclaimed
			k_ostermark = { set_title_flag = east_march }
			add_character_modifier = { modifier = warden_of_the_east_march duration = -1 }
			hidden_tooltip = { narrative_event = { id = realm.30 } }
		}

		ai_will_do = { factor = 1 }
	}

	# The Archduchy of Austria: a favoured duke of Ostland is raised to
	# Grand Prince by the Emperor.
	realm_grand_prince_ostland = {
		is_high_prio = yes
		only_playable = yes
		ai_check_interval = 60

		potential = {
			tier = DUKE
			has_landed_title = d_wolfenburg
			independent = no
			top_liege = { has_landed_title = e_the_empire }
			culture_group = imperial_group
			k_ostland = { has_holder = no }
			NOT = { has_global_flag = grand_prince_ostland_raised }
		}

		allow = {
			is_adult = yes
			war = no
			completely_controls = d_wolfenburg
			completely_controls = d_ferlangen
			OR = {
				prestige = 1500
				AND = {
					prestige = 500
					custom_tooltip = {
						text = realm_same_dynasty_as_emperor_tt
						top_liege = { dynasty = ROOT }
					}
				}
			}
			wealth = 500
			custom_tooltip = {
				text = realm_emperor_favours_tt
				top_liege = { opinion = { who = ROOT value = 50 } }
			}
		}

		effect = {
			wealth = -500
			set_global_flag = grand_prince_ostland_raised
			primary_title = { save_event_target_as = current_primary_title }
			top_liege = { save_event_target_as = the_emperor }
			k_ostland = {
				show_scope_change = no
				grant_title = ROOT
				copy_title_laws = event_target:current_primary_title
			}
			set_defacto_liege = event_target:the_emperor
			hidden_tooltip = { narrative_event = { id = realm.35 } }
		}

		ai_will_do = { factor = 1 }
	}

	# The papal guards: the Knights of the Griffon are raised again.
	realm_refound_griffons = {
		is_high_prio = yes
		only_playable = yes
		ai_check_interval = 60

		potential = {
			religion = sigmarite
			OR = {
				has_landed_title = k_cult_of_sigmar
				has_landed_title = e_the_empire
			}
			has_global_flag = knights_griffon_founded
			NOT = { is_title_active = d_knights_griffon }
		}

		allow = {
			is_adult = yes
			any_demesne_province = { always = yes }
			piety = 300
			wealth = 200
		}

		effect = {
			piety = -300
			wealth = -200
			hidden_tooltip = { narrative_event = { id = realm.60 } }
			custom_tooltip = { text = realm_refound_griffons_tt }
		}

		ai_will_do = { factor = 1 }
	}
}
