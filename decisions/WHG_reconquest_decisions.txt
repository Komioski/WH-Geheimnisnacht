# The Reconquest of the South: decisions. Restored from vanilla
# HF_realm_decisions.txt (the mod's copy had dropped them, so the reconquest
# could never start) and adapted to Estalia and Tilea.
# Events: events/WHG_reconquest_events.txt (reconquest).
# The vanilla beneficiary decisions are left out: vanilla disabled them.

decisions = {
	# Swear to take back the south: a year of preparation, then war.
	prepare_reconquista = {
		is_high_prio = yes
		only_playable = yes
		ai_check_interval = 36

		potential = {
			has_dlc = "Holy Fury"
			is_landed = yes
			independent = yes
			NOT = { has_game_rule = { name = reconquista value = off } }
			reconquest_southerner_trigger = yes
			any_demesne_title = { higher_tier_than = COUNT reconquest_land_trigger = yes }
			any_neighbor_independent_ruler = {
				NOT = { religion_group = old_world_group }
				any_realm_province = { reconquest_land_trigger = yes }
			}
			NOR = { has_character_flag = ex_reconquistador has_global_flag = reconquista_finished }
		}

		allow = {
			is_adult = yes
			war = no
			prisoner = no
			conditional_tooltip = {
				trigger = { ai = no tier = DUKE reconquest_cheap_oath_trigger = no }
				prestige = 300
				piety = 150
				wealth = 100
			}
			conditional_tooltip = {
				trigger = { ai = no tier = KING reconquest_cheap_oath_trigger = no }
				prestige = 500
				piety = 250
				wealth = 200
			}
			conditional_tooltip = {
				trigger = { ai = no tier = EMPEROR reconquest_cheap_oath_trigger = no }
				prestige = 1000
				piety = 500
				wealth = 300
			}
			conditional_tooltip = {
				trigger = { ai = no tier = DUKE reconquest_cheap_oath_trigger = yes }
				prestige = 150
				piety = 100
				wealth = 75
			}
			conditional_tooltip = {
				trigger = { ai = no tier = KING reconquest_cheap_oath_trigger = yes }
				prestige = 250
				piety = 200
				wealth = 150
			}
			conditional_tooltip = {
				trigger = { ai = no tier = EMPEROR reconquest_cheap_oath_trigger = yes }
				prestige = 500
				piety = 300
				wealth = 200
			}
			conditional_tooltip = { trigger = { ai = yes } prestige = 300 piety = 100 wealth = 100 }
			custom_tooltip = {
				text = active_reconquista_condition_TT
				NOT = { has_global_flag = active_reconquista }
			}
			conditional_tooltip = {
				trigger = { ai = yes }
				NOT = { has_global_flag = reconquista_ai_cooldown }
			}
		}

		effect = {
			if = {
				limit = { ai = yes }
				character_event = { id = reconquest.80 }
				prestige = -300
				piety = -100
				wealth = -100
			}
			else_if = {
				limit = { reconquest_cheap_oath_trigger = yes }
				if = { limit = { tier = DUKE } prestige = -150 piety = -100 wealth = -75 }
				else_if = { limit = { tier = KING } prestige = -250 piety = -200 wealth = -150 }
				else = { prestige = -500 piety = -300 wealth = -200 }
			}
			else_if = { limit = { tier = DUKE } prestige = -300 piety = -150 wealth = -100 }
			else_if = { limit = { tier = KING } prestige = -500 piety = -250 wealth = -200 }
			else = { prestige = -1000 piety = -500 wealth = -300 }
			custom_tooltip = {
				text = start_preparing_reconquista_TT
				set_character_flag = ex_reconquistador
				set_character_flag = currently_reconquering_iberia
				set_global_flag = active_reconquista
				add_character_modifier = {
					name = reconquista_preparation_duration
					hidden = yes
					years = 1
				}
				add_character_modifier = {
					name = reconquista_preparation_left
					hidden = yes
					years = 2
				}
				character_event = { id = reconquest.8 days = 730 }
				character_event = { id = reconquest.40 days = 10 random = 10 }
				narrative_event = { id = reconquest.50 }
				any_playable_ruler = {
					limit = {
						NOT = { character = ROOT }
						is_landed = yes
						any_demesne_title = {
							higher_tier_than = BARON
							reconquest_land_trigger = yes
						}
					}
					narrative_event = { id = reconquest.50 }
				}
			}
			spawn_unit = {
				province = capital_scope
				home = capital_scope
				owner = ROOT
				troops = {
					light_infantry = { 600 600 }
					heavy_infantry = { 500 500 }
					archers = { 400 400 }
				}
				earmark = reconquista_event_troops
				attrition = 1
				maintenance_multiplier = 0
				reinforces = no
			}
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 2 trait = zealous }
			modifier = { factor = 0.5 trait = cynical }
		}
	}
}
