###############################################################################
# Decisions of the sea societies: the Black Ark Corsairs, the Wolves of the
# Sea and the coastal Norscans of the Freigattur. Each is a society power,
# shown in the society screen (is_in_society). Events in events/WHG_sea_*.
# Triggers and effects: WHG_sea_triggers.txt, WHG_sea_effects.txt.
#
#   sea_set_out          Set Out / Go a-Viking (rank 2): a voyage of three
#                        to six months on a chosen sea (seas.10)
###############################################################################

targeted_decisions = {
	# Set Out (rank 2): pick a heading, meet what the sea sends, come home
	# with gold, Infamy and stories. A five-year rest between voyages; the AI
	# sails only in peace.
	sea_set_out = {
		is_in_society = yes

		filter = self
		ai_target_filter = self
		ai_check_interval = 24

		potential = {
			character = FROM
			has_dlc = "Holy Fury"
			is_playable = yes
			sea_captain_trigger = yes
			society_rank >= 2
			NOT = { has_character_flag = sea_on_voyage }
		}

		allow = {
			sea_captain_free_trigger = yes
			sea_coastal_trigger = yes
			custom_tooltip = {
				text = sea_set_out_cd_tt
				NOT = { has_character_modifier = sea_set_out_cd }
			}
			wealth = 25
			has_society_currency_minor_trigger = yes
		}

		effect = {
			wealth = -25
			detract_society_currency_minor_effect = yes
			custom_tooltip = { text = sea_set_out_tt }
			hidden_effect = { character_event = { id = seas.10 } }
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 war = yes }
			modifier = { factor = 0 NOT = { wealth = 75 } }
			modifier = { factor = 0 is_ill = yes }
			modifier = { factor = 1.5 trait = ambitious }
			modifier = { factor = 1.5 trait = greedy }
			modifier = { factor = 1.5 trait = brave }
			modifier = { factor = 0.5 trait = content }
			modifier = { factor = 0.3 trait = craven }
			modifier = { factor = 0.5 age = 60 }
		}
	}
}

# Refit the Flagship (rank 3 of the Wolves of the Sea): the next ship in the
# line without waiting for the shipwrights' offer (seas.41). A first hull
# costs 100 gold, the middle ship 200 gold and three deeds at sea, the great
# ship 400 gold and six deeds. Five years between refits.
targeted_decisions = {
	sea_refit_flagship = {
		is_in_society = yes

		filter = self
		ai_target_filter = self
		ai_check_interval = 36

		potential = {
			character = FROM
			has_dlc = "Holy Fury"
			is_playable = yes
			society_member_of = adventurers_pirates
			society_rank >= 3
			NOT = { has_artifact = enormous_flagship }
		}

		allow = {
			is_landed = yes
			capital_scope = { port = yes }
			is_inaccessible_trigger = no
			custom_tooltip = {
				text = sea_refit_ready_tt
				sea_refit_ready_trigger = yes
			}
			trigger_if = {
				limit = { has_artifact = regular_flagship }
				wealth = 400
				has_society_currency_major_trigger = yes
				custom_tooltip = {
					text = sea_refit_deeds_6_tt
					check_variable = { which = sea_deeds value = 6 }
				}
			}
			trigger_else_if = {
				limit = { has_artifact = small_flagship }
				wealth = 200
				has_society_currency_medium_trigger = yes
				custom_tooltip = {
					text = sea_refit_deeds_3_tt
					check_variable = { which = sea_deeds value = 3 }
				}
			}
			trigger_else = {
				wealth = 100
				has_society_currency_minor_trigger = yes
			}
		}

		effect = {
			if = {
				limit = { has_artifact = regular_flagship }
				detract_society_currency_major_effect = yes
			}
			else_if = {
				limit = { has_artifact = small_flagship }
				detract_society_currency_medium_effect = yes
			}
			else = {
				detract_society_currency_minor_effect = yes
			}
			sea_pay_refit_effect = yes
			sea_refit_flagship_effect = yes
			hidden_effect = { character_event = { id = seas.42 } }
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 war = yes }
			modifier = { factor = 0 NOT = { wealth = 300 } }
			modifier = { factor = 0 has_artifact = regular_flagship NOT = { wealth = 700 } }
			modifier = { factor = 2 trait = proud }
			modifier = { factor = 0.3 trait = greedy }
		}
	}

	# Raise a Dragon-ship (rank 3 of the Freigattur raiders): the Norse
	# flagship line, from a longship to a great dragon-ship. Same costs and
	# deeds as Refit the Flagship.
	sea_raise_dragonship = {
		is_in_society = yes

		filter = self
		ai_target_filter = self
		ai_check_interval = 36

		potential = {
			character = FROM
			has_dlc = "Holy Fury"
			is_playable = yes
			sea_norse_raider_trigger = yes
			society_rank >= 3
			NOT = { has_artifact = enormous_flagship }
		}

		allow = {
			is_landed = yes
			capital_scope = { port = yes }
			is_inaccessible_trigger = no
			custom_tooltip = {
				text = sea_refit_ready_tt
				sea_refit_ready_trigger = yes
			}
			trigger_if = {
				limit = { has_artifact = regular_flagship }
				wealth = 400
				has_society_currency_major_trigger = yes
				custom_tooltip = {
					text = sea_refit_deeds_6_tt
					check_variable = { which = sea_deeds value = 6 }
				}
			}
			trigger_else_if = {
				limit = { has_artifact = small_flagship }
				wealth = 200
				has_society_currency_medium_trigger = yes
				custom_tooltip = {
					text = sea_refit_deeds_3_tt
					check_variable = { which = sea_deeds value = 3 }
				}
			}
			trigger_else = {
				wealth = 100
				has_society_currency_minor_trigger = yes
			}
		}

		effect = {
			if = {
				limit = { has_artifact = regular_flagship }
				detract_society_currency_major_effect = yes
			}
			else_if = {
				limit = { has_artifact = small_flagship }
				detract_society_currency_medium_effect = yes
			}
			else = {
				detract_society_currency_minor_effect = yes
			}
			sea_pay_refit_effect = yes
			sea_refit_flagship_effect = yes
			hidden_effect = { character_event = { id = seas.42 } }
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 war = yes }
			modifier = { factor = 0 NOT = { wealth = 300 } }
			modifier = { factor = 0 has_artifact = regular_flagship NOT = { wealth = 700 } }
			modifier = { factor = 2 trait = proud }
			modifier = { factor = 0.3 trait = greedy }
		}
	}

	# Recruit Raiders (rank 3 of all three sea societies): a standing band of
	# the captain's own people at the capital, for prestige and Infamy. Half
	# upkeep, kept in peace, gone when the captain dies. One band at a time,
	# five years between musters.
	sea_recruit_raiders = {
		is_in_society = yes

		filter = self
		ai_target_filter = self
		ai_check_interval = 36

		potential = {
			character = FROM
			has_dlc = "Holy Fury"
			is_playable = yes
			sea_captain_trigger = yes
			society_rank >= 3
		}

		allow = {
			is_landed = yes
			is_inaccessible_trigger = no
			prisoner = no
			custom_tooltip = {
				text = sea_raiders_one_tt
				NOT = { has_earmarked_regiments = sea_raiders }
			}
			custom_tooltip = {
				text = sea_raiders_cd_tt
				NOT = { has_character_modifier = sea_raiders_cd }
			}
			prestige = 200
			has_society_currency_medium_trigger = yes
		}

		effect = {
			prestige = -200
			detract_society_currency_medium_effect = yes
			add_character_modifier = { name = sea_raiders_cd years = 5 hidden = yes }
			custom_tooltip = { text = sea_recruit_raiders_tt }
			hidden_effect = { character_event = { id = seas.43 } }
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = { factor = 3 war = yes }
			modifier = { factor = 0.2 war = no }
			modifier = { factor = 0 NOT = { prestige = 500 } }
			modifier = { factor = 0 NOT = { wealth = 50 } }
		}
	}
}
