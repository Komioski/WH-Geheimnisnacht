###############################################################################
# Slavery (events/WHG_slavery_events.txt)
# What a slaver can do with the slaves in the pens beyond the market and the
# fields (openslaverymenu and deploy_slaves_province in
# wh_general_decisions.txt): give them to the gods, drive them into battle,
# or feed them to the forges of Hashut; and how lords of faiths that keep no
# slaves set them free.
###############################################################################

decisions = {
	# One lot of slaves on the altar: 40 piety.
	offer_slaves_to_gods = {
		only_playable = yes
		ai_check_interval = 24

		potential = {
			has_character_flag = slave_master
			slave_sacrifice_faith_trigger = yes
		}

		allow = {
			check_variable = { which = slaves value = 1 }
		}

		effect = {
			slave_lose_lot_effect = yes
			piety = 40
		}

		ai_will_do = {
			factor = 0.1
			modifier = { factor = 3 trait = zealous }
			modifier = { factor = 2 trait = cruel }
			modifier = { factor = 3 has_character_modifier = slave_pens_3 }
		}
	}

	# Two lots of slaves driven ahead of the army: 600 light infantry at the
	# capital who go home (or die) at the peace. Once a year.
	raise_slave_levy = {
		only_playable = yes
		ai_check_interval = 6

		potential = {
			has_character_flag = slave_master
			slave_keeper_trigger = yes
		}

		allow = {
			war = yes
			check_variable = { which = slaves value = 2 }
			NOT = { has_character_modifier = slave_levy_raised }
		}

		effect = {
			slave_lose_two_lots_effect = yes
			add_character_modifier = { name = slave_levy_raised years = 1 }
			custom_tooltip = { text = raise_slave_levy_tt }
			hidden_effect = {
				capital_scope = {
					spawn_unit = {
						province = THIS
						home = THIS
						owner = ROOT
						troops = {
							light_infantry = { 600 600 }
						}
						attrition = 1.0
						disband_on_peace = yes
					}
				}
			}
		}

		ai_will_do = {
			factor = 1
		}
	}

	# Three lots of slaves fed to the Chaos Dwarfs' forges: a level of slave
	# factory at the capital (dawizharr_factory_up_effect).
	feed_slaves_to_forges = {
		only_playable = yes
		ai_check_interval = 24

		potential = {
			has_character_flag = slave_master
			culture_group = dwarf_chaos_group
			religion = hashut
		}

		allow = {
			check_variable = { which = slaves value = 3 }
		}

		effect = {
			slave_lose_two_lots_effect = yes
			slave_lose_lot_effect = yes
			custom_tooltip = { text = feed_slaves_to_forges_tt }
			hidden_effect = { dawizharr_factory_up_effect = yes }
		}

		ai_will_do = {
			factor = 1
		}
	}

	# Free every slave in the pens: piety for those whose faith keeps no
	# slaves, a loss of face for a slaver.
	free_slave_stock = {
		only_playable = yes
		ai_check_interval = 24

		potential = {
			has_character_flag = slave_master
			OR = {
				slaver_faith_trigger = no
				trait = kind
			}
		}

		allow = {
			check_variable = { which = slaves value = 1 }
		}

		effect = {
			if = {
				limit = { slaver_faith_trigger = no }
				piety = 50
			}
			else = { prestige = -50 }
			custom_tooltip = { text = free_slave_stock_tt }
			hidden_effect = {
				set_variable = { which = slaves value = 0 }
				slave_stock_update_effect = yes
			}
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.05 slaver_faith_trigger = yes }
		}
	}
}

targetted_decisions = {
	# A lord whose faith keeps no slaves frees the slaves worked in one of
	# their provinces (slave.19).
	free_worked_slaves = {
		ai_check_interval = 24
		only_playable = yes
		filter = owned
		ai_target_filter = owned
		is_high_prio = yes

		from_potential = {
			is_adult = yes
			slaver_faith_trigger = no
		}

		potential = {
			owner = { character = FROM }
			location = { has_province_modifier = deployed_slaves }
		}

		allow = {
			always = yes
		}

		effect = {
			FROM = { character_event = { id = slave.19 } }
		}

		revoke_allowed = {
			always = no
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.3 FROM = { trait = greedy } }
		}
	}
}
