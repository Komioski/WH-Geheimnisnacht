###############################################################################
# Stanitsa decisions: the Ungols and the Ropsmann
# Events: events/WHG_stanitsa_events.txt (namespace stanitsa).
#
#   stanitsa_send_mission   Send a mission to a stanitsa of the northern gods
#                           (stanitsa.25)
#   stanitsa_call_rotas     Call the Ungol horse-archer rotas to a war
###############################################################################

decisions = {
	# Send a mission to an Ungol or Ropsmann stanitsa of the northern gods
	# in your lands. Once in five years.
	stanitsa_send_mission = {
		ai_check_interval = 24

		potential = {
			religion = kislev_gods
			any_demesne_province = { stanitsa_mission_province_trigger = yes }
		}

		allow = {
			NOT = { has_character_modifier = stanitsa_mission_cd }
			OR = {
				piety = 50
				wealth = 100
			}
			prisoner = no
		}

		effect = {
			add_character_modifier = { name = stanitsa_mission_cd years = 5 hidden = yes }
			random_demesne_province = {
				limit = { stanitsa_mission_province_trigger = yes }
				preferred_limit = { culture = ungol }
				save_event_target_as = stanitsa_mission
			}
			character_event = { id = stanitsa.25 }
		}

		ai_will_do = { factor = 1 }
	}

	# Call the Ungol rotas: horse-archers and light horse from the steppe
	# stanitsas until the peace, once in ten years.
	stanitsa_call_rotas = {
		ai_check_interval = 12

		potential = {
			culture_group = kislevite_group
			OR = {
				culture = ungol
				any_demesne_province = { culture = ungol }
			}
		}

		allow = {
			war = yes
			NOT = { has_character_modifier = stanitsa_rota_cd }
			wealth = 50
		}

		effect = {
			wealth = -50
			add_character_modifier = { name = stanitsa_rota_cd years = 10 }
			capital_scope = {
				ROOT = {
					spawn_unit = {
						province = PREV
						owner = ROOT
						troops = {
							horse_archers = { 400 400 }
							light_cavalry = { 300 300 }
						}
						earmark = stanitsa_ungol_rota
						disband_on_peace = yes
						maintenance_multiplier = 0.5
					}
				}
			}
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 NOT = { wealth = 150 } }
		}
	}
}
