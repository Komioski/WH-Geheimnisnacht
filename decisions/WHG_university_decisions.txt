decisions = {

		is_high_prio = yes
		only_playable = yes
		ai_check_interval = 24

		potential = {
			trait = creature_human
			any_demesne_province = {
				university_city_trigger = yes
				NOT = { has_province_modifier = university_endowed }
			}
		}

		allow = {
			wealth = 200
			any_demesne_province = {
				university_city_trigger = yes
				university_city_open_trigger = yes
				NOT = { has_province_modifier = university_endowed }
			}
		}

		effect = {
			wealth = -200
			prestige = 100
			random_demesne_province = {
				limit = {
					university_city_trigger = yes
					university_city_open_trigger = yes
					NOT = { has_province_modifier = university_endowed }
				}
				add_province_modifier = { name = university_endowed years = 20 }
			}
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 NOT = { wealth = 500 } }
			modifier = { factor = 2 trait = scholar }
			modifier = { factor = 0.5 trait = greedy }
		}
	}
