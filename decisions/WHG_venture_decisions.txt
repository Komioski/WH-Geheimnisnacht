###############################################################################
# Decisions of the Venturers (adventurers_the_league): explorers, merchant
# adventurers and colonists. Each is a society power, shown in the society
# screen (is_in_society). Events in events/WHG_venture_events.txt.
# Triggers and effects: WHG_south_triggers.txt, WHG_south_effects.txt.
#
#   venture_sponsor_expedition  Sponsor an Expedition (rank 1): put up gold
#                               and a captain for a venture (venture.1)
#   venture_lead_expedition     Lead an Expedition (rank 2): sail with it
#                               yourself; players only under the game rule
#                               warhammer_venture_lead (on by default)
#   venture_native_guides       Native Guides (rank 2): better odds for the
#                               next expedition
#   venture_chart_coast         Chart a Coast (rank 2): a charting voyage
#                               (venture.30)
###############################################################################

targeted_decisions = {
	# Sponsor an Expedition (rank 1): gold, a ship and a captain from your
	# court. Five years between expeditions; most never come back.
	venture_sponsor_expedition = {
		is_in_society = yes

		filter = self
		ai_target_filter = self
		ai_check_interval = 36

		potential = {
			character = FROM
			has_dlc = "Holy Fury"
			is_playable = yes
			venture_member_trigger = yes
			society_rank >= 1
		}

		allow = {
			venture_coastal_trigger = yes
			custom_tooltip = {
				text = venture_free_tt
				venture_free_trigger = yes
			}
			wealth = 100
		}

		effect = {
			custom_tooltip = { text = venture_sponsor_tt }
			hidden_effect = { character_event = { id = venture.1 } }
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 NOT = { wealth = 300 } }
			modifier = { factor = 0.5 war = yes }
			modifier = { factor = 1.5 trait = ambitious }
			modifier = { factor = 1.5 trait = greedy }
			modifier = { factor = 1.5 trait = adventurer }
			modifier = { factor = 0.3 trait = content }
			modifier = { factor = 0.5 trait = paranoid }
			modifier = { factor = 2 wealth = 1000 }
		}
	}

	# Lead an Expedition (rank 2): the same, but you go yourself. The odds
	# and the rewards rise, and so does the chance of never coming home.
	venture_lead_expedition = {
		is_in_society = yes

		filter = self
		ai_target_filter = self
		ai_check_interval = 60

		potential = {
			character = FROM
			has_dlc = "Holy Fury"
			is_playable = yes
			venture_member_trigger = yes
			society_rank >= 2
			venture_may_lead_trigger = yes
		}

		allow = {
			venture_coastal_trigger = yes
			custom_tooltip = {
				text = venture_free_tt
				venture_free_trigger = yes
			}
			is_adult = yes
			war = no
			is_ill = no
			prisoner = no
			wealth = 100
		}

		effect = {
			custom_tooltip = { text = venture_lead_tt }
			hidden_effect = {
				set_character_flag = venture_leading
				character_event = { id = venture.1 }
			}
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 NOT = { wealth = 300 } }
			modifier = { factor = 0 age = 55 }
			modifier = { factor = 0 trait = craven }
			modifier = { factor = 0.2 NOT = { trait = adventurer } }
			modifier = { factor = 0.5 NOT = { trait = brave } }
			modifier = { factor = 0.5 primary_title = { higher_tier_than = DUKE } }
			modifier = { factor = 0.3 is_ruler = yes NOT = { current_heir = { is_adult = yes } } }
		}
	}

	# Native Guides (rank 2): Skink trackers, castaways, the half-mad
	# survivors of older ventures. The next expedition's odds rise.
	venture_native_guides = {
		is_in_society = yes

		filter = self
		ai_target_filter = self
		ai_check_interval = 36

		potential = {
			character = FROM
			has_dlc = "Holy Fury"
			is_playable = yes
			venture_member_trigger = yes
			society_rank >= 2
			NOT = { has_character_modifier = venture_guides_hired }
		}

		allow = {
			venture_coastal_trigger = yes
			NOT = { has_character_flag = venture_expedition_out }
			wealth = 50
			has_society_currency_minor_trigger = yes
		}

		effect = {
			wealth = -50
			detract_society_currency_minor_effect = yes
			add_character_modifier = { name = venture_guides_hired years = 10 }
			custom_tooltip = { text = venture_guides_tt }
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 NOT = { wealth = 300 } }
			modifier = { factor = 0 has_character_modifier = venture_expedition_cd }
		}
	}

	# Chart a Coast (rank 2): one ship to map an unknown shore. Charts make
	# every later expedition safer, or sell for gold (venture.30 - 31).
	venture_chart_coast = {
		is_in_society = yes

		filter = self
		ai_target_filter = self
		ai_check_interval = 60

		potential = {
			character = FROM
			has_dlc = "Holy Fury"
			is_playable = yes
			venture_member_trigger = yes
			society_rank >= 2
			NOT = { has_character_flag = venture_charting }
		}

		allow = {
			venture_coastal_trigger = yes
			custom_tooltip = {
				text = venture_chart_cd_tt
				NOT = { has_character_modifier = venture_chart_coast_cd }
			}
			wealth = 75
			has_society_currency_minor_trigger = yes
		}

		effect = {
			wealth = -75
			detract_society_currency_minor_effect = yes
			set_character_flag = venture_charting
			add_character_modifier = { name = venture_chart_coast_cd years = 5 }
			custom_tooltip = { text = venture_chart_tt }
			hidden_effect = { character_event = { id = venture.30 days = 150 random = 120 } }
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 NOT = { wealth = 200 } }
			modifier = { factor = 0.5 has_character_modifier = venture_fresh_charts }
		}
	}
}

# Reputation (rank 4): what the ports of the world say of the Venturer
# (venture.90): Conqueror, Merchant Prince, Scholar or Saviour. Ten years
# between changes.
targeted_decisions = {
	venture_choose_reputation = {
		is_in_society = yes

		filter = self
		ai_target_filter = self
		ai_check_interval = 60

		potential = {
			character = FROM
			has_dlc = "Holy Fury"
			is_playable = yes
			venture_member_trigger = yes
			society_rank >= 4
		}

		allow = {
			custom_tooltip = {
				text = venture_reputation_cd_tt
				NOT = { has_character_modifier = venture_reputation_cd }
			}
			prestige = 250
		}

		effect = {
			prestige = -250
			add_character_modifier = { name = venture_reputation_cd years = 10 }
			custom_tooltip = { text = venture_choose_reputation_tt }
			hidden_effect = { character_event = { id = venture.90 } }
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = { factor = 0 has_character_flag = venture_reputation_chosen }
		}
	}
}

# The Great Voyage (rank 4): one last venture past the edge of every
# chart, west, east or north (venture.80). Needs six deeds of exploration,
# a fortune and the society's favour. Once per lifetime, and each route
# once per game. The voyage that comes home founds a bloodline.
targeted_decisions = {
	venture_great_voyage = {
		is_in_society = yes

		filter = self
		ai_target_filter = self
		ai_check_interval = 60

		potential = {
			character = FROM
			has_dlc = "Holy Fury"
			is_playable = yes
			venture_member_trigger = yes
			society_rank >= 4
			NOT = { has_character_flag = venture_great_voyage_done }
		}

		allow = {
			venture_coastal_trigger = yes
			custom_tooltip = {
				text = venture_free_tt
				venture_free_trigger = yes
			}
			custom_tooltip = {
				text = venture_deeds_6_tt
				check_variable = { which = venture_deeds value = 6 }
			}
			custom_tooltip = {
				text = venture_route_open_tt
				venture_great_voyage_route_open_trigger = yes
			}
			war = no
			wealth = 750
			has_society_currency_major_trigger = yes
		}

		effect = {
			wealth = -750
			detract_society_currency_major_effect = yes
			set_character_flag = venture_great_voyage_done
			custom_tooltip = { text = venture_great_voyage_tt }
			hidden_effect = { character_event = { id = venture.80 } }
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 0.2
			modifier = { factor = 0 NOT = { wealth = 1500 } }
			modifier = { factor = 2 trait = ambitious }
			modifier = { factor = 2 trait = adventurer }
		}
	}
}
