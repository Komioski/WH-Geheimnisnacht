###############################################################################
# Albion decisions
# Events: events/WHG_albion_events.txt (namespace albion).
# Triggers/effects: wh_albion_triggers.txt, wh_albion_effects.txt.
#
#   albion_seek_mist_guide   Seek a guide through the Mists of Albion
#                            (albion.20, or albion.21 if the guide fails)
#
# The Ogham stones are toppled and raised with corrupt_pop and restore_pop
# (wh_magic_decisions.txt), which fire albion.30 and albion.31 in Albion.
###############################################################################

decisions = {
	# Seek a guide through the outer mists: a Truthsayer, or a Dark Emissary
	# for the servants of the Dark Masters. 100 gold, or 300 prestige for a
	# boss or chief. One in seven guides is a fraud, one in twenty for a
	# lord with a sharp eye. The AI only looks for one when it holds land in
	# Albion.
	albion_seek_mist_guide = {
		ai_check_interval = 120
		only_playable = yes

		potential = {
			has_global_flag = mists_of_albion
			is_adult = yes
			albion_native_trigger = no
			NOT = { has_character_flag = albion_seeking_guide }
			NOT = { has_character_modifier = albion_mist_pilot }
			NOT = { has_character_modifier = albion_truthsayer_curse }
		}

		allow = {
			prisoner = no
			trigger_if = {
				limit = { albion_guide_pays_prestige_trigger = yes }
				prestige = 300
			}
			trigger_else = {
				wealth = 100
			}
		}

		effect = {
			if = {
				limit = { albion_guide_pays_prestige_trigger = yes }
				prestige = -300
			}
			else = {
				wealth = -100
			}
			set_character_flag = albion_seeking_guide
			random_list = {
				85 = {
					modifier = { factor = 1.15 intrigue = 12 }
					character_event = { id = albion.20 days = 30 random = 30 }
				}
				15 = {
					modifier = { factor = 0.33 intrigue = 12 }
					character_event = { id = albion.21 days = 30 random = 30 }
				}
			}
		}

		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				NOT = { any_realm_province = { region = world_albion } }
			}
			modifier = {
				factor = 0
				NOT = { wealth = 200 }
				albion_guide_pays_prestige_trigger = no
			}
		}
	}
}
