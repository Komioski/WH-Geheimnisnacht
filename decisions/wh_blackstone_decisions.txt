# The race for Mordheim (events/WHG_warpstone_events.txt, warpstone.1 -
# warpstone.6). A lord funds an expedition against a county of d_mordheim:
# a commander with an army, an agent in the ruins or a hired captain. Close
# to the city only (750), unless a Skaven, whose tunnels run under it. Not
# against the lord's own liege or vassals, and not against the same faith
# unless the lord is cynical, arbitrary, ambitious or greedy.
title_decisions = {
	fund_mordheim_expedition = {
		ai_target_filter = all # from_potential keeps the AI to rulers near Mordheim
		ai_check_interval = 60
		is_high_prio = yes

		from_potential = {
			is_playable = yes
			is_adult = yes
			NOT = { has_character_flag = maintaining_agents }
			NOT = { has_global_flag = clean_mordheim }
			warpstone_expedition_people_trigger = yes
			OR = {
				ai = no
				warpstone_expedition_ai_reach_trigger = yes
			}
		}

		potential = {
			tier = COUNT
			de_jure_liege = d_mordheim
			has_holder = yes
			holder_scope = {
				NOT = { character = FROM }
				NOT = { is_liege_or_above = FROM }
				NOT = { is_vassal_or_below = FROM }
			}
		}

		allow = {
			custom_tooltip = {
				text = warpstone_expedition_near_tt
				OR = {
					FROM = { trait = creature_skaven }
					location = { NOT = { distance = { who = FROM value = 750 } } }
				}
			}
			holder_scope = {
				OR = {
					NOT = { religion = FROM }
					FROM = {
						OR = {
							trait = cynical
							trait = arbitrary
							trait = ambitious
							trait = greedy
						}
					}
				}
			}
			FROM = {
				war = no
				warpstone_expedition_can_pay_trigger = yes
			}
		}

		effect = {
			FROM = {
				warpstone_expedition_pay_effect = yes
				set_character_flag = maintaining_agents
			}
			holder_scope = { character_event = { id = warpstone.1 } }
		}

		ai_will_do = {
			factor = 0.1
			modifier = { factor = 2 FROM = { trait = greedy } }
			modifier = { factor = 2 FROM = { trait = ambitious } }
			modifier = { factor = 2 FROM = { warpstone_seeker_trigger = yes } }
			modifier = { factor = 0.5 FROM = { trait = content } }
			modifier = { factor = 0.5 FROM = { warpstone_shunner_trigger = yes } }
		}
	}
}
