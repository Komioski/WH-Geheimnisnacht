			cathay_invasion_declare_effect = yes
			hidden_effect = {
				any_player = {
					limit = {
						NOT = { character = ROOT }
						has_offmap_news_enabled = offmap_cathay
						is_within_diplo_range = ROOT

					narrative_event = { id = cathay_expedition.4 }
				}
			}
		}

		ai_will_do = { # Only when the End Times are here
			factor = 0
			modifier = {
				factor = 0.1
				is_doom_very_high_trigger = yes
