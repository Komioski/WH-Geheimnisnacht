###############################################################################
# Shared undead decisions
#
# Undead content shared by every master of the dead, vampire or necromancer.
# Events are in events/WHG_undead_events.txt (namespace undead), loc in
# localisation/00_WHG_undead_events.csv, effects in common/scripted_effects/
# WHG_undead_effects.txt.
#
#   undead_raise_wight        Raise a wight from an old barrow (undead.10)
#   undead_dark_enchantment   Turn a fine weapon or armour necromantic
#                             (undead.50 - undead.52)
###############################################################################

decisions = {
	# Raise a wight from an old barrow (undead.10). Vampires pay more piety
	# than necromancers, who know the rites. Not the Jade, whose magic is the
	# Dark Crystal, not the barrow. A Master Necrarch raises a Wight King
	# instead (vampsoc_raise_wight).
	undead_raise_wight = {
		only_playable = yes
		ai_check_interval = 24

		potential = {
			undead_master_trigger = yes
			NOT = { trait = vampire_jade_visible }
			is_witch_hunter_trigger = no
		}

		allow = {
			is_adult = yes
			prisoner = no
			trigger_if = {
				limit = { trait = lore_necromancy }
				piety = 50
			}
			trigger_else = {
				piety = 100
			}
			custom_tooltip = {
				text = undead_wight_cooldown_tt
				NOT = { has_character_modifier = undead_wight_cooldown }
			}
			custom_tooltip = {
				text = undead_wight_count_tt
				NOT = {
					any_courtier = {
						trait = creature_wight
						count = 3
					}
				}
			}
		}

		effect = {
			if = {
				limit = { trait = lore_necromancy }
				piety = -50
			}
			else = {
				piety = -100
			}
			add_character_modifier = { name = undead_wight_cooldown years = 10 hidden = yes }
			character_event = { id = undead.10 }
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				any_courtier = {
					trait = creature_wight
					count = 2
				}
			}
			modifier = { factor = 2 war = yes }
			modifier = { factor = 0.5 trait = greedy }
		}
	}

	# The Dark Enchantment (undead.50 - undead.52): a lord of a necromantic
	# faith has a fine weapon or suit of armour bound with death magic. It
	# shares its cooldown with hiring a smith (employ_a_smith).
	undead_dark_enchantment = {
		only_playable = yes
		ai_check_interval = 36

		potential = {
			religion_group = necromantic_group
			is_landed = yes
			higher_tier_than = BARON
			age = 16
			is_incapable = no
			OR = {
				mercenary = no
				ai = no
			}
		}

		allow = {
			is_inaccessible_trigger = no
			piety = 100
			wealth = 100
			custom_tooltip = {
				text = employ_a_smith_cooldown_tt
				hidden_trigger = {
					NOT = { has_character_modifier = employ_a_smith_cd }
				}
			}
			custom_tooltip = {
				text = undead_enchantable_item_tt
				any_artifact = {
					OR = {
						undead_enchantable_weapon_trigger = yes
						undead_enchantable_armour_trigger = yes
					}
				}
			}
		}

		effect = {
			add_character_modifier = { name = employ_a_smith_cd years = 2 hidden = yes }
			character_event = { id = undead.50 }
		}

		revoke_allowed = { always = no }

		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				any_artifact = {
					has_artifact_flag = combat_weapon
					has_artifact_flag = necromantic_item
				}
				any_artifact = {
					has_artifact_flag = armor
					has_artifact_flag = necromantic_item
				}
			}
			modifier = { factor = 0 NOT = { piety = 300 } }
			modifier = { factor = 0 NOT = { wealth = 300 } }
			modifier = { factor = 0.5 NOR = { tier = KING tier = EMPEROR } }
			modifier = { factor = 0.5 trait = greedy }
			modifier = { factor = 2 trait = ambitious }
		}
	}
}
