###############################################################################
# Vampire bloodline society decisions
#
# Powers of the seven bloodline societies (common/societies/
# wh_vampire_bloodlines.txt). Events are in events/WHG_vampire_society_events.txt
# (namespace vampsoc), loc in localisation/00_WHG_vampire_society_events.csv.
#
# Shared
#   vampsoc_rank_up            Rank up (the society's rank-up button)
#   vampsoc_begin_trial        Start the rank 2 to 3 trial (von Carstein,
#                              Necrarchs, Strigoi, Mahtmasi; the Blood Quest,
#                              the seduction missions and the Dark Crystal are
#                              the trials of the other three)
#   vampsoc_feed               Feed (every vampire member)
#   vampsoc_ask_head_aid       Ask the head of the bloodline for aid
#   vampsoc_offer_blood        Thralls: feed your vampire liege
#   vampsoc_daylight_errand    Thralls: run an errand where vampires can't go
#   vampsoc_blood_oath         Courtiers of a vampire: swear the Blood Oath
###############################################################################

society_decisions = {

	# Rank up. Rank 1 to 2 needs Blood and a deed fitting the bloodline; rank 2
	# to 3 needs more Blood and the bloodline's trial. Rank 4 is the head,
	# chosen by the society when the old head falls. Living thralls stop at 2,
	# hidden von Carsteins at 2 and hidden Lahmians at 3.
	vampsoc_rank_up = {
		only_playable = yes
		hide_in_decisions_list = yes # Shown in the society view
		ai_check_interval = 12

		potential = {
			vamp_society_member_trigger = yes
			society_rank < 3
			days_in_society > 180
			OR = {
				ai = no
				society_can_rank_up = yes
			}
		}

		allow = {
			trigger_if = {
				limit = { society_rank == 1 }
				custom_tooltip = {
					text = vampsoc_rank_up_cost_1_tt
					society_currency >= 500
				}
				custom_tooltip = {
					text = vampsoc_rank_up_deed_tt
					vampsoc_deed_trigger = yes
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = vampsoc_rank_up_cost_2_tt
					society_currency >= 1000
				}
				custom_tooltip = {
					text = vampsoc_rank_up_vampire_tt
					is_vampire_trigger = yes
				}
				custom_tooltip = {
					text = vampsoc_rank_up_hidden_tt
					NOT = { trait = vampire_carstein_hidden }
				}
				custom_tooltip = {
					text = vampsoc_rank_up_trial_tt
					has_character_flag = vamp_trial_done
				}
			}
			custom_tooltip = {
				text = vampsoc_rank_up_open_tt
				vampsoc_next_rank_open_trigger = yes
			}
			prisoner = no
		}

		effect = {
			if = {
				limit = { society_rank == 1 }
				change_society_currency = -500
			}
			else = {
				change_society_currency = -1000
				clr_character_flag = vamp_trial_done
			}
			society_rank_up = 1
			hidden_tooltip = { character_event = { id = vampsoc.5 } }
		}

		ai_will_do = { factor = 1 }
	}

	# Start the trial for rank 3. Four bloodlines run a trial of their own here;
	# the other three use their existing chains.
	vampsoc_begin_trial = {
		only_playable = yes
		ai_check_interval = 24

		potential = {
			society_rank == 2
			is_vampire_trigger = yes
			OR = {
				society_member_of = carstein_court
				society_member_of = necrarch_brotherhood
				society_member_of = strigoi_brood
				society_member_of = carrion_children
			}
			NOT = { has_character_flag = vamp_trial_done }
		}

		allow = {
			custom_tooltip = {
				text = vampsoc_on_trial_tt
				NOT = { has_character_flag = vamp_on_trial }
			}
			custom_tooltip = {
				text = vampsoc_trial_cooldown_tt
				NOT = { has_character_modifier = vampsoc_trial_cooldown }
			}
			prisoner = no
			is_incapacitated_trigger = no
		}

		effect = {
			set_character_flag = vamp_on_trial
			if = {
				limit = { society_member_of = carstein_court }
				custom_tooltip = { text = vampsoc_trial_carstein_tt }
				hidden_tooltip = { character_event = { id = vampsoc.100 } }
			}
			else_if = {
				limit = { society_member_of = necrarch_brotherhood }
				custom_tooltip = { text = vampsoc_trial_necrarch_tt }
				hidden_tooltip = { character_event = { id = vampsoc.250 } }
			}
			else_if = {
				limit = { society_member_of = strigoi_brood }
				custom_tooltip = { text = vampsoc_trial_strigoi_tt }
				hidden_tooltip = { character_event = { id = vampsoc.300 } }
			}
			else = {
				custom_tooltip = { text = vampsoc_trial_mahtmasi_tt }
				hidden_tooltip = { character_event = { id = vampsoc.400 } }
			}
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.2 } # The AI takes its time
		}
	}

	# Feed. Every two years at most. The options in vampsoc.10 depend on the
	# bloodline: the Blood Dragons lose Honour for feeding on the helpless.
	vampsoc_feed = {
		only_playable = yes
		ai_check_interval = 24

		potential = {
			vamp_society_member_trigger = yes
			is_vampire_trigger = yes
		}

		allow = {
			custom_tooltip = {
				text = vampsoc_fed_recently_tt
				NOT = { has_character_modifier = vampsoc_fed_recently }
			}
			prisoner = no
			is_incapacitated_trigger = no
		}

		effect = {
			add_character_modifier = { name = vampsoc_fed_recently years = 2 hidden = yes }
			character_event = { id = vampsoc.10 }
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 3 NOT = { health = 4 } }
		}
	}

	# Ask the head of the bloodline for gold or soldiers. The head may refuse.
	vampsoc_ask_head_aid = {
		only_playable = yes
		ai_check_interval = 60

		potential = {
			vamp_society_member_trigger = yes
			is_vampire_trigger = yes
			society_rank < 4
			is_ruler = yes
		}

		allow = {
			custom_tooltip = {
				text = vampsoc_asked_head_tt
				NOT = { has_character_modifier = vampsoc_asked_head }
			}
			custom_tooltip = {
				text = vampsoc_head_reachable_tt
				society = {
					leader = {
						NOT = { character = ROOT }
						prisoner = no
						is_incapacitated_trigger = no
					}
				}
			}
			society_currency >= 150
			war = yes
		}

		effect = {
			change_society_currency = -150
			add_character_modifier = { name = vampsoc_asked_head years = 10 hidden = yes }
			society = { leader = { letter_event = { id = vampsoc.11 } } }
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.5 }
		}
	}

	# Thralls: feed your vampire liege, who heals; you gain Blood and lose
	# some health.
	vampsoc_offer_blood = {
		only_playable = yes
		ai_check_interval = 36

		potential = {
			vamp_society_member_trigger = yes
			vamp_thrall_trigger = yes
		}

			}
			liege = {
				is_vampire_trigger = yes
				NOT = { character = ROOT }
			}
			health = 4
		}

		effect = {
			add_character_modifier = { name = vampsoc_offered_blood years = 3 hidden = yes }
			liege = { character_event = { id = vampsoc.13 } }
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.5 }
		}
	}

		}

		allow = {

			}

				is_vampire_trigger = yes
				NOT = { character = ROOT }
			}

		}

		effect = {
			add_character_modifier = { name = vampsoc_errand_cooldown years = 5 hidden = yes }
			character_event = { id = vampsoc.14 }
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.5 }
		}
	}

	###########################################################################
	# Ordo Draconis
	###########################################################################

	# Ride out to challenge a famed mortal champion (vampsoc.150). A squire's
	# victory is the deed for rank 2.
	vampsoc_challenge_champion = {
		only_playable = yes
		ai_check_interval = 24

		potential = {
			society_member_of = ordo_draconis
		}

		allow = {
			custom_tooltip = {
				text = vampsoc_challenge_cooldown_tt
				NOT = { has_character_modifier = vampsoc_challenge_cooldown }
			}
			prisoner = no
			is_incapacitated_trigger = no
			NOT = { trait = wounded }
		}

		effect = {
			add_character_modifier = { name = vampsoc_challenge_cooldown years = 3 hidden = yes }
			character_event = { id = vampsoc.150 }
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 3 society_rank == 1 }
			modifier = { factor = 0.3 trait = craven }
		}
	}

	# Challenge the Grand Master for the leadership of the Order (vampsoc.155).
	vampsoc_challenge_grand_master = {
		only_playable = yes
		ai_check_interval = 60

		potential = {
			society_member_of = ordo_draconis
			society_rank == 3
		}

		allow = {
			society_currency >= 300
			custom_tooltip = {
				text = vampsoc_challenge_gm_cooldown_tt
				NOT = { has_character_modifier = vampsoc_challenge_gm_cooldown }
			}
			custom_tooltip = {
				text = vampsoc_head_reachable_tt
				society = {
					leader = {
						NOT = { character = ROOT }
						prisoner = no
						is_incapacitated_trigger = no
					}
				}
			}
			prisoner = no
			is_incapacitated_trigger = no
			NOT = { trait = wounded }
		}

		effect = {
			change_society_currency = -300
			add_character_modifier = { name = vampsoc_challenge_gm_cooldown years = 20 hidden = yes }
			society = { leader = { save_event_target_as = vampsoc_foe } }
			character_event = { id = vampsoc.155 }
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.2 }
			modifier = { factor = 0 NOT = { combat_rating >= 30 } }
			modifier = { factor = 3 trait = ambitious }
		}
	}

	}

decisions = {

	# A courtier or vassal of a vampire swears the Blood Oath: they take the
	# vampiric faith and become a thrall. This is how a vampire realm grows a
	# loyal mortal court.
	vampsoc_blood_oath = {
		only_playable = yes
		ai_check_interval = 60

		potential = {
			trait = creature_human
			is_vampire_trigger = no
			NOT = { religion = vampiric }
			NOT = { religion_group = necromantic_group }
			liege = {
				is_vampire_trigger = yes
				religion = vampiric
				NOT = { character = ROOT }
			}
		}

		allow = {
			NOT = { trait = zealous }
			is_inaccessible_trigger = no
		}

		effect = {
			religion = vampiric
			add_character_modifier = { name = vampsoc_blood_oath duration = -1 }
			liege = { opinion = { who = ROOT modifier = opinion_vampsoc_blood_oath years = 10 } }
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.05 }
			modifier = { factor = 3 opinion = { who = liege value = 50 } }
			modifier = { factor = 0 trait = zealous }
			modifier = { factor = 0.2 trait = honest }
		}
	}
}

targetted_decisions = {

	# A Lahmian Queen helps a mortal ruler at war and catches them in the web
	# (vampsoc.200).
	vampsoc_assist_mortals = {
		ai_check_interval = 36
		filter = all
		ai_target_filter = realm
		only_playable = yes

		from_potential = {
			society_member_of = lahmian_sisterhood
			society_rank >= 3
		}

		potential = {
			is_ruler = yes
			is_vampire_trigger = no
			war = yes
			NOT = { character = FROM }
			is_within_diplo_range = FROM
		}

		allow = {
			FROM = {
				custom_tooltip = {
					text = vampsoc_assist_cooldown_tt
					NOT = { has_character_modifier = vampsoc_assist_cooldown }
				}
				wealth = 100
			}
		}

		effect = {
			save_event_target_as = vampsoc_mortal
			FROM = {
				add_character_modifier = { name = vampsoc_assist_cooldown years = 3 hidden = yes }
				character_event = { id = vampsoc.200 }
			}
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.1 }
			modifier = { factor = 3 FROM = { any_lover = { character = ROOT } } }
		}
	}
