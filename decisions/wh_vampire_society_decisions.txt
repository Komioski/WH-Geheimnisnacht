###############################################################################
# Vampire bloodline society decisions
#
# Powers of the seven bloodline societies (common/societies/
# WHG_vampire_societies.txt). Events are in events/WHG_vampire_events.txt
# (namespace vampsoc), loc in localisation/00_WHG_vampire_events.csv.
#
# Shared
#   vampsoc_rank_up            Rank up (the society's rank-up button)
#   vampsoc_begin_trial        Start the rank 2 to 3 trial (von Carstein,
#                              Necrarchs, Strigoi, Mahtmasi; the Blood Quest,
#                              the seduction missions and the Dark Crystal are
#                              the trials of the other three)
#   vampsoc_feed               Feed (every vampire member)
#   vampsoc_ask_head_aid       Ask the head of the bloodline for aid
#   vampsoc_offer_blood        Thralls: feed your vampire liege
#   vampsoc_daylight_errand    Thralls: run an errand where vampires can't go
#   vampsoc_blood_oath         Courtiers of a vampire: swear the Blood Oath
# von Carstein: throw_off_mask, raise_the_dead, grave_guard, night_court,
#   claim_sylvania, sylvania_reborn
# Ordo Draconis: challenge_champion, challenge_grand_master,
#   reclaim_blood_keep, trail_of_abhorash
# Lahmians: tithe_of_houses, veil_of_lahmia, claim_silver_pinnacle,
#   great_web, assist_mortals (targeted)
# Necrarchs: seek_nagashi_relic, join_dark_tower, raise_wight,
#   recipe_for_nagash, take_apprentice (targeted)
# Strigoi: ghoul_horde, call_strigoi_home, strygos_reborn
# Jade: scribe_examination, shed_skin, crystal_court, celestial_gaze,
#   serpent_coils (targeted)
# Mahtmasi: gwingui_vigil, open_gate, claim_sudenburg,
#   cast_out_westerlanders, devour_kin and send_dream (targeted)
# The older bloodline decisions in wh_vampire_decisions.txt check the
# society rank they belong to.
###############################################################################

society_decisions = {

	# Rank up. Rank 1 to 2 needs Blood and a deed fitting the bloodline; rank 2
	# to 3 needs more Blood and the bloodline's trial. Rank 4 is the head,
	# chosen by the society when the old head falls. Living thralls stop at 2,
	# hidden von Carsteins at 2 and hidden Lahmians at 3.
	vampsoc_rank_up = {
		only_playable = yes
		hide_in_decisions_list = yes # Shown in the society view
		ai_check_interval = 12

		potential = {
			vamp_society_member_trigger = yes
			society_rank < 3
			days_in_society > 180
			OR = {
				ai = no
				society_can_rank_up = yes
			}
		}

		allow = {
			trigger_if = {
				limit = { society_rank == 1 }
				custom_tooltip = {
					text = vampsoc_rank_up_cost_1_tt
					society_currency >= 500
				}
				custom_tooltip = {
					text = vampsoc_rank_up_deed_tt
					vampsoc_deed_trigger = yes
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = vampsoc_rank_up_cost_2_tt
					society_currency >= 1000
				}
				custom_tooltip = {
					text = vampsoc_rank_up_vampire_tt
					is_vampire_trigger = yes
				}
				custom_tooltip = {
					text = vampsoc_rank_up_hidden_tt
					NOT = { trait = vampire_carstein_hidden }
				}
				custom_tooltip = {
					text = vampsoc_rank_up_trial_tt
					has_character_flag = vamp_trial_done
				}
			}
			custom_tooltip = {
				text = vampsoc_rank_up_open_tt
				vampsoc_next_rank_open_trigger = yes
			}
			prisoner = no
		}

		effect = {
			if = {
				limit = { society_rank == 1 }
				change_society_currency = -500
			}
			else = {
				change_society_currency = -1000
				clr_character_flag = vamp_trial_done
			}
			society_rank_up = 1
			hidden_tooltip = { character_event = { id = vampsoc.5 } }
		}

		ai_will_do = { factor = 1 }
	}

	# Start the trial for rank 3. Four bloodlines run a trial of their own here;
	# the other three use their existing chains.
	vampsoc_begin_trial = {
		only_playable = yes
		ai_check_interval = 24

		potential = {
			society_rank == 2
			is_vampire_trigger = yes
			OR = {
				society_member_of = carstein_court
				society_member_of = necrarch_brotherhood
				society_member_of = strigoi_brood
				society_member_of = carrion_children
			}
			NOT = { has_character_flag = vamp_trial_done }
		}

		allow = {
			custom_tooltip = {
				text = vampsoc_on_trial_tt
				NOT = { has_character_flag = vamp_on_trial }
			}
			custom_tooltip = {
				text = vampsoc_trial_cooldown_tt
				NOT = { has_character_modifier = vampsoc_trial_cooldown }
			}
			prisoner = no
			is_incapacitated_trigger = no
		}

		effect = {
			set_character_flag = vamp_on_trial
			if = {
				limit = { society_member_of = carstein_court }
				custom_tooltip = { text = vampsoc_trial_carstein_tt }
				hidden_tooltip = { character_event = { id = vampsoc.100 } }
			}
			else_if = {
				limit = { society_member_of = necrarch_brotherhood }
				custom_tooltip = { text = vampsoc_trial_necrarch_tt }
				hidden_tooltip = { character_event = { id = vampsoc.250 } }
			}
			else_if = {
				limit = { society_member_of = strigoi_brood }
				custom_tooltip = { text = vampsoc_trial_strigoi_tt }
				hidden_tooltip = { character_event = { id = vampsoc.300 } }
			}
			else = {
				custom_tooltip = { text = vampsoc_trial_mahtmasi_tt }
				hidden_tooltip = { character_event = { id = vampsoc.400 } }
			}
		}

		ai_will_do = {
			factor = 1
			modifier = { factor = 0.2 } # The AI takes its time
		}
	}
