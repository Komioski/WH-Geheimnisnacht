# chaoscult.1 A cultist casts off the mask
# Sent by the rising cult leader (chaoscult.2) to every cultist and secret
# Chaos worshipper in the defending realm. They convert to their secret faith
# and leave the cult; landed ones join the revolt against their independent
# liege (or liege's liege) with an independence war.
# Scope: ROOT is the cultist, FROM the cult leader.
character_event = {
	id = chaoscult.1
	desc = EVTDESCchaoscult1
	picture = GFX_evt_society_joining
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	immediate = {
		set_character_flag = currently_part_of_secret_religion_uncovering
	}

	option = {
		name = EVTOPTAchaoscult1
		convert_to_secret_religion = yes
		leave_society = yes
		if = { limit = { ai = yes } set_character_flag = ai_flag_refuse_conversion }
		if = {
			limit = {
				higher_tier_than = BARON
				AND = {
					liege = { NOT = { secret_religion_group = chaos_gods_group } independent = yes }
				}
			}
			liege = {
				ROOT = { set_defacto_liege = PREV }
				reverse_war = { target = ROOT casus_belli = cultist_bid_for_independence }
			}
		}
		else_if = {
			limit = {
				higher_tier_than = BARON
				AND = {
					liege = {
						liege = {
							NOT = { secret_religion_group = chaos_gods_group }
							independent = yes
						}
					}
				}
			}
			liege = {
				liege = {
					ROOT = { set_defacto_liege = PREV }
					reverse_war = { target = ROOT casus_belli = cultist_bid_for_independence }
				}
			}
		}
	}
}

# chaoscult.2 The cult rises
# The cult leader proclaims their faith. They become a devout leader, break
# away from a top liege outside the cult with an independence war, and call on
# every cultist in that realm to rise too (chaoscult.1).
# Scope: ROOT is the cult leader.
character_event = {
	id = chaoscult.2
	desc = EVTDESCchaoscult2
	picture = GFX_evt_dark_prayer
	border = GFX_event_normal_frame_religion
	is_triggered_only = yes

	option = {
		name = EVTOPTAchaoscult2
		custom_tooltip = { text = EVTTOOLTIPchaoscult2 }
		add_character_modifier = { modifier = devout_leader duration = -1 }
		save_event_target_as = cultist_leader_revolting
		if = {
			limit = {
				top_liege = {
					NOR = {
						same_society_as = event_target:cultist_leader_revolting
						secret_religion_group = chaos_gods_group
					}
				}
			}
			top_liege = {
				ROOT = { set_defacto_liege = PREV }
				save_event_target_as = top_liege_defending
				reverse_war = { target = ROOT casus_belli = cultist_bid_for_independence }
			}
			event_target:top_liege_defending = {
				any_realm_character = {
					if = {
						limit = {
							OR = {
								same_society_as = event_target:cultist_leader_revolting
								secret_religion_group = chaos_gods_group
							}
						}
						character_event = { id = chaoscult.1 } # Convert cultists
					}
				}
			}
		}

		if = { limit = { ai = yes } set_character_flag = ai_flag_refuse_conversion }
	}
}

# chaoscult.11 Prepare the grounds: the mission
# The cult asks for a secret cell in a chosen province before it can rise.
# Scope: ROOT is the cultist.

	id = chaoscult.11
	border = GFX_event_letter_frame_religion

	desc = EVTDESCchaoscult11

	is_triggered_only = yes

	immediate = {
		random_demesne_province = {
			limit = {
				OR = {
					AND = {
						ROOT = { society_member_of = cult_of_khorne }
						NOR = { religion = khorne has_province_modifier = secret_khorne_community }
					}
					AND = {
						ROOT = { society_member_of = cult_of_nurgle }
						NOR = { religion = khorne has_province_modifier = secret_nurgle_community }
					}
					AND = {
						ROOT = { society_member_of = cult_of_tzeentch }
						NOR = {
							religion = khorne
							has_province_modifier = secret_tzeentch_community
						}
					}
					AND = {
						ROOT = { society_member_of = cult_of_slaanesh }
						NOR = {
							religion = khorne
							has_province_modifier = secret_slaanesh_community
						}
					}
					AND = {
						ROOT = { society_member_of = cult_of_undivided }
						NOR = {
							religion = khorne
							has_province_modifier = secret_undivided_community
						}
					}
				}
			}
			save_event_target_as = target_province
		}
	}

	option = {
		name = ACCEPT
		event_target:target_province = {
			set_quest_target = { id = quest_chaos_cult_prepare_grounds holder = ROOT }
		}
	}

	option = {
		name = DECLINE
		custom_tooltip = { text = decline_quest_tooltip }

		add_character_modifier = { name = quest_cooldown_timer hidden = yes days = 1000 }

		ai_chance = { factor = 0 }
	}
}

# chaoscult.14 Watch the province

# Scope: ROOT is the cultist.

	id = chaoscult.14
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				event_target:target_province = {
					county = { holder_scope = { NOT = { same_realm = ROOT } } }
				}
			}
			clear_delayed_event = { id = chaoscult.16 }
			clear_delayed_event = { id = chaoscult.17 }
			clear_delayed_event = { id = chaoscult.18 }
			clear_delayed_event = { id = chaoscult.19 }
			character_event = { id = chaoscult.15 }
		}
		else = { repeat_event = { id = chaoscult.14 days = 30 } }
	}
}

# chaoscult.20 Dinner with an open mind
# First step of recruiting someone into a Chaos cult (decisions in
# wh_cult_societies_decisions.txt and the recruitment events in
# mnm_misc_events.txt). The recruiter sounds out the target's faith. Agreeing
# moves on (chaoscult.21); refusing tells the recruiter (chaoscult.22).
# Each god's recruits favour different traits.
# Scope: ROOT is the recruit, FROMFROM the recruiter.
character_event = {
	id = chaoscult.20
	desc = EVTDESCchaoscult20
	picture = GFX_evt_shady_meeting
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes

	trigger = { }

	immediate = {
		if = {
			limit = { NOT = { event_target:chaos_cult_recruiter = { always = yes } } }
			FROMFROM = { save_event_target_as = chaos_cult_recruiter }
		}
		save_event_target_as = chaos_cult_recruit
	}

	#Approving of heretics
	option = {
		name = { text = EVTOPTAchaoscult20 }

		trigger = { OR = { ai = no NOT = { controls_religion = yes } } }

		custom_tooltip = { text = EVTTOOLTIPchaoscult20 }

		reverse_opinion = {
			who = event_target:chaos_cult_recruiter
			modifier = opinion_agreed_with_opinion
			years = 5
		}

		set_character_flag = approves_of_satanism

		hidden_effect = { character_event = { id = chaoscult.21 days = 3 } }

		ai_chance = {
			factor = 20
			modifier = { factor = 0 trait = zealous }
			modifier = { factor = 1.5 trait = trusting }
			modifier = { factor = 1.5 is_dumb_trigger = yes }
			modifier = {
				factor = 1.5
				has_character_flag = being_recruited_by_khorne
				is_strong_trigger = yes
			}

			modifier = {
				factor = 1.5
				trait = ambitious
				has_character_flag = being_recruited_by_tzeentch
			}
			modifier = {
				factor = 1.5
				trait = deceitful
				has_character_flag = being_recruited_by_tzeentch
			}
			modifier = {
				factor = 1.5
				is_smart_trigger = yes
				has_character_flag = being_recruited_by_tzeentch
			}

			modifier = { factor = 1.5 is_ill = yes has_character_flag = being_recruited_by_nurgle }

			modifier = {
				factor = 1.5
				trait = content
				has_character_flag = being_recruited_by_nurgle
			}

			modifier = {
				factor = 1.5
				has_character_flag = being_recruited_by_slaanesh
				trait = hedonist
			}

			modifier = {
				factor = 1.5
				has_character_flag = being_recruited_by_slaanesh
				trait = lustful
			}

			modifier = { factor = 1.5 trait = cynical }
		}
	}

	#Disapproving of heretics
	option = {
		name = { text = EVTOPTBchaoscult20 }

		custom_tooltip = { text = EVTTOOLTIPchaoscult20b }

		reverse_opinion = {
			who = event_target:chaos_cult_recruiter
			modifier = opinion_disagreed_with_opinion
			years = 5
		}

		set_character_flag = disapproves_of_satanism

		chaoscult_clear_recruit_flags_effect = yes

		hidden_effect = {
			event_target:chaos_cult_recruiter = {
				character_event = { id = chaoscult.22 days = 15 }
			}
		}

		ai_chance = {
			factor = 80
			modifier = { factor = 1.5 has_pious_trait_trigger = yes }
			modifier = { factor = 1.5 has_crusade_trait_trigger = yes }
		}
	}
}

# chaoscult.36 Inductor dispatch

				FROM = { character_event = { id = chaoscult.38 days = 14 } }

			chaoscult_send_flight = { FROM = { character_event = { id = chaoscult.47 days = 1 } } }
		}
		clr_character_flag = chaoscult_send_letter
		clr_character_flag = chaoscult_send_meeting
		clr_character_flag = chaoscult_send_mentor
		clr_character_flag = chaoscult_send_test
