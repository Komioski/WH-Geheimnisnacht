# Faction: Chaos
#
# When doom runs very high, a champion of all four Dark Gods rises in the
# Chaos Wastes and rides south at the head of a vast horde. When the
# Everchosen falls, the horde breaks apart and doom resets.
#
# Sections:
#   1-6    the Everchosen rises (narrative event, courtiers, host, population)
#   10-11  the Everchosen falls
#   20-22  surrender offer to Chaos and Norscan rulers
#   25     the horde grows after its first conquest
#
# Callers:
#   undivided.1   on_decade_pulse (wh_on_actions), for the Sigmarite head
#   undivided.6   on_new_holder, on_new_holder_inheritance and
#                 on_new_holder_usurpation (00_on_actions)
#   undivided.10  on_death (wh_on_actions)
#   undivided.11  on_five_year_pulse (wh_on_actions)
#   undivided.20  everchosen_invasion on_add (cb_types_chaos)
#   undivided.25  everchosen_invasion on_success (cb_types_chaos)
#
# The player path to the mantle (seektobecomeeverchosen decision and the
# claim_the_mantle_of_the_everchosen war) lives elsewhere; undivided.10 cleans
# up after any Everchosen, player or AI.

# 4 The horde's people
# The tribes follow their champion: 120,000 people and a 30-year hidden
# modifier that marks the horde as invading. Starts the yearly growth (5).
# Scope: ROOT = the Everchosen.
character_event = {
	id = undivided.4
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		population = 120000
		add_character_modifier = {
			modifier = invading_population_halt
			duration = 10950
			hidden = yes
		}
		character_event = { id = undivided.5 days = 365 }
	}
}

# 5 The horde swells
# Each year, while the horde is still a nomad horde on the march, more tribes
# join it: 80,000 people a year while it is below its population cap and
# under 350,000. The loop ends when the invading modifier runs out or the
# Everchosen settles.
# Scope: ROOT = the Everchosen.
character_event = {
	id = undivided.5
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				OR = {
					NOR = { government = nomadic_government government = chaos_horde_government }
					NOT = { has_character_modifier = invading_population_halt }
				}
			}
			remove_character_modifier = invading_population_halt
		}
		else = {
			if = {
				limit = { NOT = { population_factor = 1 } NOT = { population = 350000 } }
				population = 80000
			}
			repeat_event = { id = undivided.5 days = 365 }
		}
	}
}

# 6 Landless no longer

	hide_window = yes
	is_triggered_only = yes

	religion_group = chaos_gods_group

		trait = everchosen
		FROM = { tier = COUNT }
		primary_title = { has_title_flag = temporary_landless_title is_landless_type_title = yes }
	}

	immediate = {
		any_demesne_title = {
			limit = { has_title_flag = temporary_landless_title }
			set_title_landless = { title = THIS status = no }
		}
	}
}
