# pleasure.1 Wine and doubt
# A cultist dines with a curious elf and steers the talk towards the gods.
# Agreeing tells the recruiter (pleasure.2); refusing ends it (pleasure.3).
# Scope: ROOT is the recruit, FROM the recruiter.
character_event = {
	id = pleasure.1
	desc = EVTDESCpleasure1
	picture = GFX_evt_shady_meeting
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { event_target:cult_of_pleasure_recruiter = { always = yes } } }
			FROM = { save_event_target_as = cult_of_pleasure_recruiter }
		}
		save_event_target_as = cult_of_pleasure_recruit
	}

	option = { # Why do the gods ignore us?
		name = EVTOPTApleasure1
		trigger = { OR = { ai = no controls_religion = no } }
		custom_tooltip = { text = EVTTOOLTIPpleasure1 }
		reverse_opinion = {
			who = event_target:cult_of_pleasure_recruiter
			modifier = opinion_agreed_with_opinion
			years = 5
		}
		set_character_flag = approves_of_satanism
		hidden_effect = { character_event = { id = pleasure.2 days = 3 } }

		ai_chance = {
			factor = 20
			modifier = { factor = 0 trait = zealous }
			modifier = { factor = 0 trait = everqueen }
			modifier = { factor = 0 trait = everqueen_heir }
			modifier = { factor = 0 culture_group = asrai_group }
			modifier = { factor = 0 trait = malekith }
			modifier = { factor = 0 trait = druchii_witch_elf }
			modifier = { factor = 1.5 trait = trusting }
			modifier = { factor = 1.5 is_dumb_trigger = yes }
			modifier = { factor = 1.5 trait = hedonist }
			modifier = { factor = 1.5 trait = lustful }
			modifier = { factor = 1.5 trait = cynical }
		}
	}
	option = { # We should not speak of this
		name = EVTOPTBpleasure1
		custom_tooltip = { text = EVTTOOLTIPpleasure1b }
		reverse_opinion = {
			who = event_target:cult_of_pleasure_recruiter
			modifier = opinion_disagreed_with_opinion
			years = 5
		}
		set_character_flag = disapproves_of_satanism
		clr_character_flag = being_recruited_by_the_cult_of_pleasure
		hidden_effect = {
			event_target:cult_of_pleasure_recruiter = {
				character_event = { id = pleasure.3 days = 15 }
			}
		}

		ai_chance = {
			factor = 80
			modifier = { factor = 1.5 has_pious_trait_trigger = yes }

# pleasure.60 Knives in the dark
# Hidden. Every two years the grand master may send a cultist to kill: first
# someone they marked for death (the mark for death decision), else a skilled
# councillor or commander of a marked ruler, else a heretic hunter at their
# liege's court. Half the time nothing happens. The killing succeeds on about
# 40%, more with a cunning killer, less against a paranoid target. Letters
# follow (pleasure.61 and 62).
# Fired from on_society_bi_yearly_pulse.
# Scope: ROOT is the grand master.
character_event = {
	id = pleasure.60
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		society_member_of = cult_of_pleasure
		is_society_grandmaster = yes
		any_society_member = { NOT = { character = ROOT } prisoner = no is_adult = yes }
	}

	immediate = {
		random = {
			chance = 50

			random_list = {
				200 = { # Marked for death
					modifier = {
						factor = 0
						NOT = {
							any_character = {
								has_character_flag = cop_deathmark
								pleasure_knife_target_trigger = yes
							}
						}
					}
					random_character = {
						limit = {
							has_character_flag = cop_deathmark
							pleasure_knife_target_trigger = yes
						}
						save_event_target_as = pleasure_knife_target
					}
				}
				100 = { # A skilled servant of a marked ruler
					modifier = {
						factor = 0
						NOT = {
							any_character = {
								has_character_flag = cop_deathmark
								is_ruler = yes
								any_courtier = { pleasure_knife_servant_trigger = yes }
							}
						}
					}
					random_character = {
						limit = {
							has_character_flag = cop_deathmark
							is_ruler = yes
							any_courtier = { pleasure_knife_servant_trigger = yes }
						}
						random_courtier = {
							limit = { pleasure_knife_servant_trigger = yes }
							save_event_target_as = pleasure_knife_target
						}
					}
				}
				50 = { # A heretic hunter close to home
					modifier = {
						factor = 0
						NOT = {
							liege = {
								job_spiritual = {
									has_job_action = action_hunt_heretics
									learning = 8
									pleasure_knife_target_trigger = yes
								}
							}
						}
					}
					liege = { job_spiritual = { save_event_target_as = pleasure_knife_target } }
				}
			}

			if = {
				limit = { event_target:pleasure_knife_target = { always = yes } }
				random_society_member = {
					limit = { NOT = { character = ROOT } prisoner = no is_adult = yes }
					preferred_limit = { intrigue >= 15 }
					preferred_limit = { intrigue >= 10 }
					save_event_target_as = pleasure_knife
				}
				pleasure_knife_strike_effect = yes
			}
		}
	}
}
