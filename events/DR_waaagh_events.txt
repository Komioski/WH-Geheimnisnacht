# waaagh.1 Fightier
# One stack of fightiness for a battle or war won (AI rulers get two
# longer stacks and more prestige).
# Scope: ROOT is the greenskin.
character_event = {
	id = waaagh.1
	hide_window = yes
	religion_group = greenskin_gods_group
	only_playable = yes
	is_triggered_only = yes # battles and wars won (on_actions), cb_types_greenskin.txt
	trigger = {
		OR = {
			NOT = { has_character_modifier = greenskin_fightiness }
			has_instances_of_character_modifier = { name = greenskin_fightiness amount < 25 }
		}
	}
	immediate = {
		# Start the march events if this warlord leads a WAAAGH! war and the
		# loop is not running (for example after the old warlord died)
		if = {
			limit = {
				any_war = { using_cb = waaagh_war attacker = { character = ROOT } }
				OR = {
					NOT = { has_character_flag = waaagh_marching }
					had_character_flag = { flag = waaagh_marching days = 240 }
				}
			}
			clr_character_flag = waaagh_marching
			set_character_flag = waaagh_marching
			character_event = { id = waaagh.20 days = 30 }
		}
		if = {
			limit = { ai = no }
			prestige = 5
			piety = 10
			add_character_modifier = { modifier = greenskin_fightiness years = 2 stacking = yes }
		}
		else = {
			prestige = 25
			piety = 10
			add_character_modifier = { modifier = greenskin_fightiness years = 3 stacking = yes }
			add_character_modifier = { modifier = greenskin_fightiness years = 3 stacking = yes }
		}
	}
}

# waaagh.10 Da boyz want a WAAAGH!
# An independent greenskin ruler with fightiness 15 or more is asked to
# call a WAAAGH!. He can refuse three times (each refusal costs prestige);
# the fourth time he must call it or become a squatter king. Calling it
# turns fightiness into WAAAGH! empowerment and starts waaagh.11.
# Scope: ROOT is the greenskin ruler.
character_event = {
	id = waaagh.10
	title = EVTNAMEwaaagh10
	desc = EVTDESCwaaagh10
	picture = GFX_evt_orcs2
	religion_group = greenskin_gods_group
	only_independent = yes
	is_triggered_only = yes # on_yearly_pulse
	immediate = {
		capital_scope = { county = { save_event_target_as = waaagh_state_capital } }
	}
	trigger = {
		NOT = { has_character_modifier = denied_waaagh }
		has_instances_of_character_modifier = { name = greenskin_fightiness amount >= 15 }
		OR = { ai = no prestige = 1000 }
		check_variable = { which = denied_waaagh value < 4 }
		government = greenskin_government
	}
	option = {
		trigger = { check_variable = { which = denied_waaagh value < 3 } }
		name = EVTOPTAwaaagh10
		add_character_modifier = { modifier = denied_waaagh years = 3 }
		prestige = -100
		change_variable = { which = denied_waaagh value = 1 }
		ai_chance = {
			factor = 25
			modifier = { factor = 2 trait = slothful }
			modifier = { factor = 2 NOT = { prestige = 2000 } }
			modifier = { factor = 4 trait = content }
		}
	}
	option = {
		trigger = { check_variable = { which = denied_waaagh value >= 3 } }
		if = {
			limit = {
				NOT = { any_owned_bloodline = { has_bloodline_flag = more_raiders_bloodline } }
			}
			create_bloodline = { type = squatter_king }
		}
		ai_chance = {
			factor = 100
			modifier = { factor = 2 trait = slothful }
			modifier = { factor = 4 trait = content }
			modifier = { factor = 1.2 NOT = { prestige = 2000 } }
		}
		prestige = -100
		name = EVTOPTBwaaagh10
		add_character_modifier = { modifier = denied_waaagh years = 3 }
	}
	option = {
		trigger = {
			OR = {
				check_variable = { which = denied_waaagh value < 3 }
				AND = { check_variable = { which = denied_waaagh value < 4 } prestige = 2500 }
			}
		}
		if = {
			limit = { check_variable = { which = denied_waaagh value >= 3 } }
			prestige = -5000
			custom_tooltip = { text = hypocritical_warlord }
		}
		name = EVTOPTCwaaagh10
		hidden_effect = {
			if = {
				limit = { ai = no }
				reduce_fightiness = yes
				reduce_fightiness = yes
				reduce_fightiness = yes
				reduce_fightiness = yes
				reduce_fightiness = yes
				add_character_modifier = {
					modifier = empowered_by_waaagh
					duration = -1
					stacking = yes
				}
				add_character_modifier = {
					modifier = empowered_by_waaagh
					duration = -1
					stacking = yes
				}
			}
			else = {
				random_list = {
					50 = { reduce_fightiness = yes reduce_fightiness = yes reduce_fightiness = yes }
					25 = { reduce_fightiness = yes reduce_fightiness = yes }
					25 = { reduce_fightiness = yes }
				}
				add_character_modifier = {
					modifier = empowered_by_waaagh
					duration = -1
					stacking = yes
				}
				add_character_modifier = {
					modifier = empowered_by_waaagh
					duration = -1
					stacking = yes
				}
			}
		}
		custom_tooltip = { text = EVTOPTCwaaagh10_tt2 }
		add_character_modifier = { modifier = waagh_fervor years = 3 }
		set_character_flag = converted_to_chaos_horde
		character_event = { id = waaagh.11 }
		ai_chance = {
			factor = 5
			modifier = { factor = 2 trait = wroth }
			modifier = { factor = 2 trait = cruel }
			modifier = { factor = 2 trait = greedy }
			modifier = { factor = 10 trait = ambitious }
			modifier = { factor = 2 prestige = 2000 }
		}
	}
}

# waaagh.11 WAAAGH!
# The ruler takes the WAAAGH! government, gets a WAAAGH! empire title,
# boyz, manpower and population for his holdings, and two years to start
# the WAAAGH! war. The AI starts it within a year (waaagh.13).
# Scope: ROOT is the warlord.
character_event = {
	id = waaagh.11
	title = EVTNAMEwaaagh10
	desc = EVTDESCwaaagh11
	picture = GFX_evt_waaagh
	is_triggered_only = yes # waaagh.10

	trigger = { }

	immediate = {
		set_variable = { which = popToAddCount value = 2000 }
		set_variable = { which = moneyToAddCount value = 0 }
		any_demesne_province = {
			any_province_holding = {
				if = {
					limit = { holding_type = tribal }
					ROOT = { change_variable = { which = moneyToAddCount value = 10 } }
				}
				else = { ROOT = { change_variable = { which = moneyToAddCount value = 30 } } }
				ROOT = { change_variable = { which = popToAddCount value = 2000 } }
			}
		}
	}

	#WAAAAAAAAAAAAGH!
	option = {
		name = EVTOPTAwaaagh11
		add_character_modifier = { modifier = preparing_waaagh_invasion duration = 720 } #gives timer
		set_character_flag = preparing_waaagh_invasion_flag #Prevents AI from starting the war immediately
		if = {
			limit = { ai = yes }
			character_event = { id = waaagh.13 days = 320 } #Let ai go to waaaagh #1 year
		}
		add_character_modifier = { name = preparing_waaagh hidden = yes months = 48 }
		set_government_type = greenskin_waagh_government
		create_title = {
			tier = EMPEROR
			short_name = yes
			name = "WAAAGH_TITLE"
			custom_created = yes
			culture = ROOT
			holder = ROOT
			base_title = event_target:waaagh_state_capital
		}
		if = {
			limit = { trait = creature_orc }
			create_retinue = orc_boyz
			create_retinue = orc_arrer_boyz
		}
		if = {
			limit = { trait = creature_goblin }
			create_retinue = goblin_boyz
			create_retinue = goblin_arrer_boyz
		}
		custom_tooltip = {
			text = convert_to_chaos_nomad
			any_realm_province = {
				any_province_holding = { convert_greenskin_buildings_to_nomad_equiv = yes }
			}
			any_realm_province = {
				limit = {
					OR = {
						any_province_holding = { is_holy_site = chaos }
						any_province_holding = { is_holy_site = greenskin_gods }
					}
				}
				population = 1000
			}
		}
		population = popToAddCount
		manpower = moneyToAddCount
		manpower = moneyToAddCount
		manpower = moneyToAddCount
		hidden_tooltip = {
			capital_holding = {
				add_building = no_greenskin_camp_1
				add_building = no_orc_warcamp_1
				add_building = no_goblin_warcamp_1
				add_building = no_greenskin_palisades_0
			}
			set_variable = { which = popToAddCount value = 0 }
			set_variable = { which = moneyToAddCount value = 0 }
		}
		clr_character_flag = converted_to_chaos_horde
	}
}

# waaagh.20 The march (every six months)
# Runs for the leader of every WAAAGH! war about every six months, from the
# start of the war until it ends. Wears the warlord out over time, sends a
# small defender to its fate (waaagh.35), and rolls a march event for every
# greenskin commander in the defender's lands.
# Scope: ROOT is the warlord.
character_event = {
	id = waaagh.20
	hide_window = yes
	is_triggered_only = yes # waaagh_war on_add (cb_types_greenskin.txt), waaagh.1, and itself every six months

	religion_group = greenskin_gods_group

	trigger = {
		any_war = { using_cb = waaagh_war attacker = { character = ROOT } }
	}

	immediate = {
		# Keep marching: the next roll in about six months. The flag tells
		# waaagh.1 that the loop is running.
		clr_character_flag = waaagh_marching
		set_character_flag = waaagh_marching
		character_event = { id = waaagh.20 days = 150 random = 60 }

		any_war = {
			limit = { using_cb = waaagh_war attacker = { character = ROOT } }
			defender = {
				primary_title = { save_event_target_as = waagh_title }
				save_event_target_as = waagh_defender_target
			}
			attacker = { save_event_target_as = waagh_warboss_target }

			# The WAAAGH! wears out: weaker after a while, then exhausted
			ROOT = {
				if = {
					limit = {
						NOR = {
							has_character_modifier = waaagh_fervor_exhaustion
							has_character_modifier = waaagh_fervor
							has_character_modifier = waaagh_fervor_weaker
							has_character_flag = waaagh_fervor_weaker
						}
					}
					add_character_modifier = { name = waaagh_fervor_weaker days = 1277 }
					set_character_flag = waaagh_fervor_weaker
				}
				else_if = {
					limit = {
						has_character_flag = waaagh_fervor_weaker
						OR = {
							has_character_modifier = waaagh_fervor_exhaustion
							has_character_modifier = waaagh_fervor
							has_character_modifier = waaagh_fervor_weaker
						}
					}
					add_character_modifier = { name = waaagh_fervor_exhaustion duration = -1 }
					clr_character_flag = waaagh_fervor_weaker
				}
			}

			# A small target must decide its fate (at most once a year)
			if = {
				limit = {
