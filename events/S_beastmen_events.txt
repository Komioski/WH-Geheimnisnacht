# beastmen.1 A warherd stirs
# A ruler in the wild lands (Chaos Wastes, Kislev, Norsca, Araby, the wild
# parts of Bretonnia and the Border Princes) may see a warherd rise in their
# capital (beastmen.2). Rare outside the Chaos Wastes, never on coasts.
# Scope: ROOT is the ruler. Fired by on_five_year_pulse (random list).
character_event = {
	id = beastmen.1
	hide_window = yes
	only_rulers = yes
	is_triggered_only = yes
	is_patrician = no

	trigger = {
		NOT = { has_game_rule = { name = provincial_revolts value = none } }
		higher_tier_than = BARON
		capital_scope = { beastmen_lands_trigger = yes }
	}
	weight_multiplier = {
		factor = 1
		modifier = { factor = 0.05 lower_tier_than = DUKE independent = yes }
		modifier = { factor = 0.25 trait = creature_beastman }
		modifier = { factor = 2 has_global_flag = morrslieb_full }
		modifier = { factor = 0 trait = creature_beastman war = yes }
		modifier = {
			factor = 0.15
			NOT = { capital_scope = { empire = { title = e_chaos_wastes } } }
		}
		modifier = { factor = 0.25 capital_scope = { empire = { title = e_norsca } } }
		modifier = { factor = 0.75 capital_scope = { empire = { title = e_kislev } } }
		modifier = {
			factor = 0 #No Beastmen appearing on islands
			capital_scope = { any_neighbor_province = { NOT = { is_land = yes } } }
		}
	}

	immediate = {
		capital_scope = {
			set_province_flag = beastman_horde_inv
			province_event = { id = beastmen.2 }
		}
	}
}

# beastmen.2 A warherd rises
# A warherd rises in a province with beastmen unrest, a herdstone, or the flag
# from beastmen.1: a chieftain and two sub-chieftains spawn and declare war on
# the top liege, unless a beastmen uprising for this county is already on.
# Scope: ROOT is the province. Fired by on_rebel_revolt, beastmen.1 and 35, and
# the daemonic incursion events.
province_event = {
	id = beastmen.2
	desc = EVTDESCbeastmen2
	picture = GFX_evt_beastmen
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	trigger = {
		OR = {
			has_province_modifier = beastmen_unrest
			has_province_modifier = beastman_herdstone
			has_province_flag = beastman_horde_inv
		}
		owner = { in_revolt = no top_liege = { in_revolt = no } }
		NOT = { # Not if there is already an ongoing beastmen uprising for this province
			owner = {
				top_liege = {
					war = yes

					any_war = {
						defender = { character = PREV }
						using_cb = beastmen_uprising
						war_title = ROOT # The county
					}
				}
			}
		}
	}

	immediate = {
		add_province_modifier = { name = recent_county_uprising duration = 1825 }
		clr_province_flag = beastman_horde_inv
		beastmen_warherd_chieftain_effect = yes

		new_character = {
			culture = beastman_main
			if = {
				limit = { trait = lore_tzeentch }
				add_trait = bray_shaman
				magic_power_random_effect = yes
				enable_magic_xp_effect = yes
			}
			create_title = {
				tier = KING
				landless = yes
				temporary = yes
				rebel = yes
				culture = beastman_main
				name = "WAR_HERD"
				holder = THIS
				nomad = yes
			}
			set_government_type = beastman_government

			beastmen_warherd_army_effect = yes

			beastmen_warherd_war_effect = yes
		}

		owner = {
			any_liege = { # Inform the lieges
				character_event = { id = beastmen.3 }
			}
		}
	}

	option = {
		name = EVTOPTAbeastmen2
	}
}

# beastmen.7 Morghur corrupts a province
# Morghur (or his vassal) takes a capital: the land is scourged, his army
# grows (beastmen.10), buildings are razed and up to five courtiers die.
# Scope: ROOT is the winner of the siege, FROM the province. Fired by
# on_siege_over_winner.
character_event = {
	id = beastmen.7
	desc = EVTDESCbeastmen7
	picture = GFX_evt_beastmen

	is_triggered_only = yes

	trigger = {
		OR = { trait = morghur top_liege = { trait = morghur } }
		FROM = { is_capital = yes }
		NOT = { location = { has_province_modifier = province_scourged } }
	}

	option = {
		name = EVTOPTAbeastmen7
		location = { add_province_modifier = { name = province_scourged duration = 3600 } }
		morghur = { character_event = { id = beastmen.10 } }

		if = {
			limit = { FROM = { owner = { top_liege = { NOT = { character = PREV } } } } }
			FROM = { owner = { top_liege = { character_event = { id = beastmen.9 } } } }
		}
		FROM = { owner = { character_event = { id = beastmen.8 } } }
		FROM = { set_province_flag = flag_province_scourged }
		if = {
			limit = {
				FROM = {
					OR = {
						any_neighbor_province = { OR = { is_nomadic = yes is_tribal = yes } }
						has_province_flag = flag_province_scourged
						NOT = { num_of_settlements = 2 }
					}
				}
			}
			random = {
				chance = 1
				FROM = {
					convert_to = TRIBAL
					random_province_holding = {
						limit = { is_capital = no }
						destroy_settlement = THIS
					}
				}
			}
		}

		prestige = 50
		piety = 50
		hidden_tooltip = {
			FROM = {
				destroy_random_building = yes
				owner = { beastmen_slaughter_court_effect = yes }
			}
		}
		ai_chance = { factor = 1 }
	}
}

# beastmen.12 The herd migrates
# A new beastman chieftain is made for the lands left behind. A migrating king
# becomes independent and his old liege is saved for the heir.

# (beastmen and greenskin).

	id = beastmen.12
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		save_event_target_as = beastmen_migrator
		create_character = {
			random_traits = yes
			dynasty = RANDOM
			religion = ROOT
			culture = ROOT
			trait = creature_beastman
		}
		new_character = { character_event = { id = beastmen.13 } }
		if = {
			limit = { tier = KING }
			set_character_flag = beastmen_former_liege
			liege = { save_event_target_as = target_new_liege }
			ROOT = { set_defacto_liege = THIS prestige = 300 piety = 100 }
		}
	}
}

# otherwise a courtier. Less likely when the heir has big horns or a strong arm.

		prisoner = no

					}
				}
				else = {
					random_courtier = {
						limit = { beastmen_herd_challenger_trigger = yes }
						preferred_limit = { trait = horns_big }
						character_event = { id = beastmen.23 }

	desc = EVTDESCasrai36

	is_triggered_only = yes

		higher_tier_than = DUKE

			NOR = {
				has_province_modifier = province_slave_raid
				has_province_modifier = province_scourged
				has_province_modifier = province_slave_raid_chaos_dwarf
			}
		}
		OR = {
			AND = {
				FROM = { any_de_jure_liege_title = { has_title_flag = under_despoilment } }
				any_war = { attacker = { character = ROOT } using_cb = beastmen_despoilment }
			}
			any_war = { attacker = { character = ROOT } using_cb = despoiler_invasion }
		}
		FROM = {
			is_tribal = no
			is_capital = yes
			owner = { NOT = { religion_group = chaos_gods_group } } #only against civilized places
		}
	}

	option = {
		name = EVTOPTAasrai36
		if = {
			limit = { FROM = { owner = { top_liege = { NOT = { character = PREV } } } } }
			FROM = { owner = { top_liege = { character_event = { id = beastmen.59 } } } }
		}
		FROM = { owner = { character_event = { id = beastmen.59 } } }
		FROM = { set_province_flag = flag_province_scourged }
		if = {
			limit = {
				FROM = {
					OR = {
						any_neighbor_province = { is_tribal = yes }
						has_province_flag = flag_province_scourged

			}

				limit = { num_of_buildings >= 1 }

			factor = 25

			}
			modifier = { factor = 2 avatar_kurnous = { trait = lunatic_3 } }
		}
	}
	option = {

		if = {
			limit = { FROM = { owner = { top_liege = { NOT = { character = PREV } } } } }

# beastmen.60 The herd sacks a holding
# A beastman who wins a siege of a holding with four or more buildings can
# sack it for gold and learning (one building burnt, the holder and his top
# liege furious) or spare it for prestige. The province is then safe from
# another sack for ten years. Moved from the greenskin file (was WAAGH.9).
# Scope: ROOT is the siege winner, FROM the holding.
character_event = {
	id = beastmen.60
	desc = EVTDESCbeastmen60
	picture = GFX_evt_beastmen_charge
	border = GFX_event_normal_frame_war
	portrait = event_target:top_liege_target

	is_triggered_only = yes # on_siege_over_winner, random list raiding_events

	trigger = {
		culture_group = beastman_group
		FROM = {
			OR = {
				holding_type = city
				holding_type = temple
				holding_type = castle
				holding_type = tribal
			}
			num_of_buildings = 4
			location = { NOT = { has_province_modifier = recently_burnt_the_land_not_owner } }
		}
	}

	immediate = {
		FROM = {
			holder_scope = {
				save_event_target_as = holder_target
				top_liege = { save_event_target_as = top_liege_target }
			}
		}
		FROM = {
			location = {
				add_province_modifier = { name = recently_burnt_the_land_not_owner years = 10 }
			}
		}
	}

	option = { # Sack it
		name = EVTOPTAbeastmen60

		show_portrait = event_target:holder_target
		show_portrait = event_target:top_liege_target

		ai_chance = { factor = 95 }

		if = {
			limit = { NOT = { FROM = { holding_type = tribal } } }
			wealth = 10
			custom_tooltip = {
				text = gain_tech_custom_tooltip
				hidden_tooltip = {
					random_list = {
						10 = { military_techpoints = 5 }
						10 = { economy_techpoints = 5 }
						10 = { culture_techpoints = 5 }
					}
				}
			}
		}
		if = { limit = { FROM = { holding_type = tribal } } wealth = 5 population = 75 }

		FROM = {
			destroy_random_building = THIS
			holder_scope = {
				show_scope_change = no
				opinion = { modifier = opinion_furious who = ROOT years = 3 }
				top_liege = {
					show_scope_change = no
					opinion = { modifier = opinion_furious who = ROOT years = 3 }
				}
			}
		}
	}

	option = { # Leave it
		name = EVTOPTBbeastmen60
		ai_chance = { factor = 5 }
		prestige = 50
	}
}
