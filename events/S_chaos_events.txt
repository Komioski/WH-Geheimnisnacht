# daemon.1 Becoming a Chaos horde
# A ruler takes the Chaos horde decision: their realm becomes a nomadic horde.
# Holdings outside Zanbaijin and the holy sites are razed, and their worth
# comes back as population and wealth.
# Scope: ROOT is the ruler.
character_event = {
	id = daemon.1
	title = EVTTITLEdaemon1
	desc = EVTDESCdaemon1
	picture = GFX_evt_chaos_hordes
	is_triggered_only = yes

	immediate = {
		set_variable = { which = popToAddCount value = 1000 }
		set_variable = { which = moneyToAddCount value = 0 }
		any_realm_province = {
			any_province_holding = {
				if = {
					limit = { holding_type = tribal }
					ROOT = { change_variable = { which = moneyToAddCount value = 10 } }
				}
				else = { ROOT = { change_variable = { which = moneyToAddCount value = 30 } } }
				ROOT = { change_variable = { which = popToAddCount value = 1000 } }
			}
		}
	}
	#CALCULATE POPULATION AND ADD BUILDINGS
	option = {
		name = EVTOPTAdaemon1
		set_government_type = chaos_horde_government
		custom_tooltip = {
			text = convert_to_chaos_nomad
			any_realm_province = {
				any_province_holding = { convert_holdings_buildings_to_nomad_equiv = yes }
			}
			any_realm_province = {
				limit = {
					NOR = {
						county = {
							#The main city of the wastes
							title = c_zanbaijin
						}
						OR = {
							any_province_holding = { is_holy_site = chaos }
							any_province_holding = { is_holy_site = tzeentch }
							any_province_holding = { is_holy_site = nurgle }
							any_province_holding = { is_holy_site = slaanesh }
							any_province_holding = { is_holy_site = khorne }
							any_province_holding = { is_holy_site = norscan_gods }
						}
					}
				}
				population = 1000
				any_province_holding = {
					if = { limit = { NOT = { holding_type = nomad } } destroy_settlement = THIS }
				}
			}
		}
		population = popToAddCount
		wealth = moneyToAddCount
		hidden_tooltip = {
			capital_holding = {
				add_building = no_cattle_pens_1
				add_building = no_ox_wagon_1
				add_building = no_cattle_pens_2
				add_building = no_ox_wagon_2
			}
			set_variable = { which = popToAddCount value = 0 }
			set_variable = { which = moneyToAddCount value = 0 }
		}
		clr_character_flag = converted_to_chaos_horde
	}
	#NEVERMIND
	option = {
		name = EVTOPTBdaemon1
		trigger = { ai = no }
		clr_character_flag = converted_to_chaos_horde
	}
}

# daemon.6 A warp rift opens
# When doom is high, a rift opens at a Chaos ruler's capital in the Wastes and
# its neighbours. Their god, or all four for Chaos Undivided, drink from it.
# The rift strikes (daemon.8) until it closes (daemon.7).
# Scope: ROOT is the ruler.
character_event = {
	id = daemon.6
	desc = EVTDESCdaemon6
	picture = GFX_evt_chaos_hordes
	is_triggered_only = yes # on_decade_pulse, random list chaos

	only_rulers = yes
	religion_group = chaos_gods_group

	trigger = {
		is_doom_high_trigger = yes
		capital_scope = {
			region = world_chaos_wastes
			NOR = {
				has_province_modifier = warphole_epicentre
				any_neighbor_province = {
					has_province_modifier = warphole_epicentre
					any_neighbor_province = { has_province_modifier = warphole_epicentre }
				}
			}
		}
	}

	immediate = {
		capital_scope = { save_event_target_as = daemon_wastes_province }
		if = {
			limit = { religion = tzeentch }
			add_offmap_currency = { offmap = tzeentch_warp amount = 500 }
		}
		else_if = {
			limit = { religion = slaanesh }
			add_offmap_currency = { offmap = slaanesh_warp amount = 500 }
		}
		else_if = {
			limit = { religion = khorne }
			add_offmap_currency = { offmap = khorne_warp amount = 500 }
		}
		else_if = {
			limit = { religion = nurgle }
			add_offmap_currency = { offmap = nurgle_warp amount = 500 }
		}
		else = {
			# Chaos Undivided: every god drinks from the rift
			add_offmap_currency = { offmap = tzeentch_warp amount = 150 }
			add_offmap_currency = { offmap = slaanesh_warp amount = 150 }
			add_offmap_currency = { offmap = khorne_warp amount = 150 }
			add_offmap_currency = { offmap = nurgle_warp amount = 150 }
		}
	}

	option = {
		name = EVTOPTAdaemon6
		hidden_tooltip = {
			capital_scope = {
				add_province_modifier = { name = warphole_epicentre duration = -1 }
				province_event = { id = daemon.8 days = 25 random = 40 } # the eye strikes
				province_event = { id = daemon.7 days = 250 random = 900 } # the rift closes
				any_neighbor_province = {
					add_province_modifier = { name = warphole_epicentre duration = -1 }
					province_event = { id = daemon.8 days = 25 random = 40 }
					province_event = { id = daemon.7 days = 250 random = 900 }
				}
			}
		}
	}
}

# daemon.20 Summoning a daemon into a prisoner
# A cultist or sorcerer binds a prisoner in a circle and calls a daemon of one
# god into them. Cult or lore decides which gods answer. The summoner's power
# and the prisoner's own magic decide whether the daemon is bound, escapes, or
# the prisoner breaks free or bursts.
# Scope: ROOT is the summoner, FROM the prisoner.
character_event = {
	id = daemon.20
	desc = EVTDESCdaemon20
	picture = GFX_evt_demon_summoning
	only_rulers = yes

	is_triggered_only = yes

	option = {
		name = EVTOPTAdaemon20
		trigger = {
			OR = {
				society_member_of = cult_of_tzeentch
				society_member_of = cult_of_undivided
				trait = lore_tzeentch
				trait = lore_shadows
				trait = lore_dark_magic
			}
		}
		random_list = {
			#success, summoned tzeentch daemon
			60 = {
				modifier = { factor = 0.75 trait = hedge_wizard }
				modifier = { factor = 1.25 trait = champion_tzeentch }
				modifier = { factor = 1.5 trait = chosen_tzeentch }
				modifier = { factor = 1.75 trait = prince_tzeentch }
				modifier = { factor = 1.5 trait = lore_dark_magic }
				modifier = { factor = 1.5 has_character_modifier = magic_power_3 }

# daemon.256 Becoming a beastman warherd
# A ruler takes the warherd decision: their realm becomes a beastman warherd
# and settled holdings outside Zanbaijin and the holy sites are razed.
# Scope: ROOT is the ruler.
character_event = {
	id = daemon.256
	title = EVTTITLEdaemon256
	desc = EVTDESCdaemon256
	picture = GFX_evt_beastmen_mist

	is_triggered_only = yes

	immediate = {
		set_variable = { which = popToAddCount value = 1000 }
		set_variable = { which = moneyToAddCount value = 0 }
		any_realm_province = {
			any_province_holding = {
				if = {
					limit = { holding_type = tribal }
					ROOT = { change_variable = { which = moneyToAddCount value = 10 } }
				}
				else = { ROOT = { change_variable = { which = moneyToAddCount value = 30 } } }
				ROOT = { change_variable = { which = popToAddCount value = 1000 } }
			}
		}
	}
	#Convert Holdings and Government
	option = {
		name = EVTOPTAdaemon256
		set_government_type = beastman_government
		custom_tooltip = {
			text = convert_to_warherd
			any_realm_province = {
				limit = {
					NOR = {
						county = {
							#The main city of the wastes
							title = c_zanbaijin
						}
						OR = {
							any_province_holding = { is_holy_site = chaos }
							any_province_holding = { is_holy_site = tzeentch }
							any_province_holding = { is_holy_site = nurgle }
							any_province_holding = { is_holy_site = slaanesh }
							any_province_holding = { is_holy_site = khorne }
							any_province_holding = { is_holy_site = norscan_gods }
						}
					}
				}
				population = 1000
				any_province_holding = {
					if = { limit = { NOT = { holding_type = nomad } } destroy_settlement = THIS }
				}
			}
		}
		population = popToAddCount
		wealth = moneyToAddCount
		hidden_tooltip = {
			capital_holding = { add_building = no_beastman_herdstone }
			set_variable = { which = popToAddCount value = 0 }
			set_variable = { which = moneyToAddCount value = 0 }
		}
		clr_character_flag = converted_to_warherd
	}
	#Nevermind
	option = {
		name = EVTOPTBdaemon256
		trigger = { ai = no }
		clr_character_flag = converted_to_warherd
	}
}
