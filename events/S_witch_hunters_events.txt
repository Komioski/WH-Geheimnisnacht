###############################################################################
# Witch Hunters: the Order of Sigmar's witch hunters and the hunts they run
#
# Namespace: whunter (merged 2026-10 from witchhunter and swhunters)
# Faction: Empire
#
# How hunts work: Learning makes a good witch hunter (it rises after each
# mission); Intrigue makes a good hidden cultist. Exposure compares the
# cultist's Intrigue with the hunter's Learning. Paranoid hunters take even a
# good explanation as proof of guilt.
#
# Contents
#   1-6      Calling in a witch hunter
#   10-20    Witch hunt at court (17-19 are shared verdicts)
#   30-31    Cleansing tainted provinces
#   35-46    Vampires and Chaos worshippers in the realm
#   50-58    Chapter succession
#   60-72    Discipline in the Order
#   80-89    Rulers and the Order: loans, donations, expulsion
#   100-116  Demands from an AI Grand Master
#   120-147  Requests to the Order (player decisions)
#   150-152  Hunting mages
#   160-168  Recruits
#   170-183  Missions: inquisitorial investigation
#   185-187  Purge reports (letters during a province purge)
#   190-201  Missions: hunts (mutant, cult cell, necromancer)
#   205-211  Missions: Order duties (sparring, sermons)
#
# Fired from: common/on_actions/00_on_actions_witch_hunters.txt and
# wh_on_actions.txt (yearly, two-, five- and ten-year pulses, on_death,
# society pulse, join requests),
# decisions/wh_witch_hunter_decisions.txt, and other event files (mutations,
# magic potential, Chaos, Blackstone, random flavour, MNM).
###############################################################################

namespace = whunter

###############################################################################
# Calling in a witch hunter
# A ruler with a Chaos-tainted province sends a court witch hunter on a
# three-year hunt at the court that asked for help.
###############################################################################

# whunter.1 A province bears the taint of Chaos
# A ruler with a Chaos-tainted demesne province (Nurgle disease, beastmen,
# warpdust, warpstone, Slaanesh hedonism) decides whether to send their court
# witch hunter after it. Cultist rulers may refuse.
# Scope: ROOT is the ruler. Fired by on_yearly_pulse.
character_event = {
	id = whunter.1
	title = WITCH_HUNTERS # Witch Hunters
	desc = EVTDESCwhunter1 #"One of your demesne provinces bears the taint of chaos. Your courtiers and barons want you to call in a professional to track down the root of this evil, and cleanse it from your lands."
	picture = "GFX_evt_witch_hunters"

	only_rulers = yes
	capable_only = yes
	religion_group = old_world_group

	is_triggered_only = yes

	trigger = {
		prisoner = no
		higher_tier_than = BARON
		OR = {
			NOT = { trait = witch_hunter }
			any_courtier = { has_minor_title = title_knight_captain_heir }
		}
		any_demesne_province = {
			whunter_tainted_province_trigger = yes
			top_liege = { NOT = { has_character_modifier = expelled_k_witch_hunters_order } }
			NOT = { has_province_modifier = witch_hunter_purge_in_progress }
		}
	}

	option = {
		name = EVTOPTAwhunter1 #"We must send for a witch hunter immediately."
		trigger = {
			NOT = {
				any_courtier = {
					OR = {
						has_minor_title = title_court_witchhunter
						has_minor_title = title_knight_captain_heir
					}
					NOT = { trait = on_witch_hunt }
				}
			}
		}
		custom_tooltip = { text = assign_witch_hunter }
		piety = -200
		ai_chance = { factor = 20 }
	}
	option = {
		name = EVTOPTBwhunter1 #"We must send for a witch hunter immediately."
		trigger = {
			any_courtier = {
				OR = {
					has_minor_title = title_court_witchhunter
					has_minor_title = title_knight_captain_heir
				}
				NOT = { trait = on_witch_hunt }
			}
		}
		random_demesne_province = {
			limit = {
				whunter_tainted_province_trigger = yes
				NOT = { has_province_modifier = witch_hunter_purge_in_progress }
			}
			add_province_modifier = { name = witch_hunter_purge_in_progress duration = 1100 }
		}
		save_event_target_as = whunter_client
		random_courtier = {
			limit = {
				OR = {
					has_minor_title = title_court_witchhunter
					has_minor_title = title_knight_captain_heir
				}
				NOT = { trait = on_witch_hunt }
			}
			character_event = { id = whunter.2 }
		}
		ai_chance = { factor = 40 }
	}

	option = {
		name = EVTOPTCwhunter1 #humanflavor.3.b # No, Witch hunters cause too much trouble.
		trigger = {
			OR = {
				ai = no
				society_member_of = cult_of_slaanesh
				society_member_of = cult_of_tzeentch
				society_member_of = cult_of_khorne
				society_member_of = cult_of_nurgle
			}
		}
		increase_visibility_minor_effect = yes
		ai_chance = { factor = 20 }
	}
}

# whunter.2 Evil stirs
# The witch hunter receives the call for help and sets out. Province events
# that fire this save the ruler as event_target:whunter_client.
# Scope: ROOT is the witch hunter, FROM the ruler who asked.
character_event = {
	id = whunter.2
	title = EVIL_STIRS # Evil Stirs
	desc = EVTDESCwhunter2 #"A province lord has sent for you, asking your aid in uncovering and rooting out the taint of chaos in one of his provinces."
	picture = "GFX_evt_witch_hunters"

	is_triggered_only = yes

	option = {
		name = EVTOPTAwhunter2 #"I leave immediately."
		save_event_target_as = witch_hunter
		liege = { save_event_target_as = original_liege }
		# Callers that are province events save the ruler as whunter_client.
		if = {
			limit = { event_target:whunter_client = { always = yes } }
			event_target:whunter_client = { character_event = { id = whunter.3 } }
		}
		else = { FROM = { character_event = { id = whunter.3 } } }
	}
}

# whunter.4 Time to get to work
# The witch hunter starts a three-year hunt at the ruler's court and moves
# there unless they already hold a court witch hunter title.
# Scope: ROOT is the witch hunter, FROM the ruler.
character_event = {
	id = whunter.4
	title = ON_A_DARK_AND_STORMY_NIGHT # On a dark and stormy night...
	desc = EVTDESCwhunter4 #"You arrive at the court. Now, it's time to get to work..."
	picture = "GFX_evt_beheading"

	is_triggered_only = yes

	option = {
		name = EVTOPTAwhunter4 #"Indeed!"
		add_trait = on_witch_hunt
		set_character_flag = whunter_on_hunt # cheap pre-trigger for the hunt events
		set_character_flag = noob_witch_hunter
		if = {
			limit = {
				NOR = {
					has_minor_title = title_court_witchhunter
					has_minor_title = title_knight_captain_heir
				}
			}
			move_character = FROM
		}
		character_event = { id = whunter.5 days = 1060 } #witch hunts last for 3 years, then the witch hunter leaves again#
	}
}

# whunter.5 The investigation is finished
# After three years the hunt ends. The witch hunter gains skill, tells the
# liege, and goes home.
# Scope: ROOT is the witch hunter.
character_event = {
	id = whunter.5
	title = EVTNAMEWITCHHUNTER.5 # Your investigation is finished
	desc = EVTDESCwhunter5 #"You've done what you can here. It's time to move on again and root out new evils."
	picture = "GFX_evt_witch_hunters"

	is_triggered_only = yes

	option = {
		name = EVTOPTAwhunter5 #"Indeed, time to return to the temple."
		remove_trait = on_witch_hunt
		clr_character_flag = whunter_on_hunt
		change_learning = 1 #learning determines how good a witch hunter is#
		random_list = {
			50 = { if = { limit = { NOT = { martial = 10 } } change_martial = 1 } }
			50 = { if = { limit = { NOT = { intrigue = 10 } } change_intrigue = 1 } }
		}
		liege = { character_event = { id = whunter.6 } }
		if = {
			limit = { NOT = { has_minor_title = title_court_witchhunter } }
			move_character = event_target:original_liege
		}
	}
}

# whunter.10 A suspicious courtier
# A witch hunter on a hunt picks a suspicious-looking courtier of their liege
# (mutated, ugly, lustful, mystic...) and confronts them.
# Scope: ROOT is the witch hunter. Fired by on_yearly_pulse (whunter_events
# list).
character_event = {
	id = whunter.10
	title = WITCH_HUNT # Witch Hunt
	desc = EVTDESCwhunter10 #"No one is immune to the taint of chaos, and while it is better to burn an entire village than risk contamination by one individual, you must be more careful here in the court. But there is one courtier who you suspect to be a cultist..."
	picture = "GFX_evt_witch_hunters"

	has_character_flag = whunter_on_hunt
	religion_group = old_world_group

	is_triggered_only = yes

	trigger = {
		prisoner = no
		is_adult = yes
		trait = on_witch_hunt
		NOT = { trait = incapable }
		liege = { religion_group = old_world_group }
		top_liege = { NOT = { has_character_modifier = expelled_k_witch_hunters_order } }
	}

	weight_multiplier = {
		factor = 2
		modifier = { factor = 0.5 intrigue = 12 }
		modifier = { factor = 0.5 martial = 12 }
	}

	option = {
		name = EVTOPTAwhunter10 #"I will confront this individual."
		save_event_target_as = witch_hunter
		liege = {
			random_courtier = {
				limit = {
					NOT = { character = ROOT }
					prisoner = no
					is_adult = yes
					NOR = {
						has_character_modifier = wh_accused
						trait = BretonDamsel
						is_daemon_race_trigger = yes
						check_if_mage_should_be_burned_trigger = no
					}
					OR = { lower_tier_than = DUKE ai = no }
					OR = {
						trait = lustful
						trait = hedonist
						trait = mystic
						trait = paranoid
						trait = hunchback
						trait = ugly
						trait = dwarf
						trait = imbecile
						religion_group = chaos_gods_group
						is_mutated_trigger = yes
					}
				}
				character_event = { id = whunter.12 }
			}
		}
	}
}

# Scope: ROOT is the ruler (Sons of Abraham). Fired by on_decade_pulse
# (random: whunter_order_holding, about 1 in 50 years). Shares a 25-year
# cooldown with whunter.110.
character_event = {
	id = whunter.105
	is_triggered_only = yes

	hide_window = yes

	only_playable = yes
	religion_group = old_world_group

	trigger = {
		whunter_site_request_ready_trigger = yes
		OR = {
			religion = sigmarite
			religion = myrmidian
			religion = ulrican
			religion = taalite
			religion = shallyan
			religion = manannite
		}
		is_heretic = no

		has_character_modifier = borrowed_from_k_witch_hunters_order
		NOT = { has_character_flag = sent_son_to_witch_hunters }

		any_demesne_province = { has_empty_holding = yes }

		k_witch_hunters_order = { holder_scope = { ai = yes wealth = 300 } }
		has_dlc = "Sons of Abraham"

		NOT = { has_character_modifier = expelled_k_witch_hunters_order }
		NOT = { any_liege = { has_character_modifier = expelled_k_witch_hunters_order } }
	}

	weight_multiplier = {
		factor = 1
	}

	immediate = {
		clr_character_flag = whunter_asked_for_site
		set_character_flag = whunter_asked_for_site
		random_demesne_province = {
			limit = { has_empty_holding = yes }
			province_event = { id = whunter.106 }
		}
	}

	option = {
		name = OK
	}
}

# whunter.110 The Order asks for a castle site
# An AI Grand Master with a small realm asks a ruler (debt or not) for an
# empty holding in the Old World.
# Scope: ROOT is the ruler (Sons of Abraham). Fired by on_decade_pulse
# (random: whunter_order_site, about 1 in 100 years). Shares a 25-year
# cooldown with whunter.105.
character_event = {
	id = whunter.110
	is_triggered_only = yes

	hide_window = yes

	only_playable = yes
	religion_group = old_world_group

	trigger = {
		whunter_site_request_ready_trigger = yes
		OR = {
			religion = sigmarite
			religion = myrmidian
			religion = ulrican
			religion = taalite
			religion = shallyan
			religion = manannite
		}
		is_heretic = no

		k_witch_hunters_order = {
			holder_scope = {
				ai = yes
				wealth = 300
				NOT = { realm_size = 10 }
				NOT = { num_of_count_titles = 2 }
			}
		}

		any_demesne_province = { has_empty_holding = yes region = world_old_world }
		has_dlc = "Sons of Abraham"

		NOT = { has_character_modifier = expelled_k_witch_hunters_order }
		NOT = { any_liege = { has_character_modifier = expelled_k_witch_hunters_order } }
	}

	weight_multiplier = {
		factor = 1
		modifier = {
			factor = 2
			k_witch_hunters_order = { holder_scope = { NOT = { demesne_size = 1 } } }
		}
		modifier = {
			factor = 2
			k_witch_hunters_order = { holder_scope = { NOT = { demesne_size = 2 } } }
		}
		modifier = { factor = 0.5 k_witch_hunters_order = { holder_scope = { realm_size = 5 } } }
	}

	immediate = {
		clr_character_flag = whunter_asked_for_site
		set_character_flag = whunter_asked_for_site
		random_demesne_province = {
			limit = { has_empty_holding = yes region = world_old_world }
			province_event = { id = whunter.111 }
		}
	}
}

			factor = 75
