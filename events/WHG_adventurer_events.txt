###############################################################################
# Raiding adventurers
# Namespace: adventurer (was HL.100 - HL.200 in
#   hl_raiding_adventurers_events.txt, TOG.1200 - TOG.1215 in
#   adventures_the_old_gods.txt, and JD.1000 - JD.1005 in
#   jd_muslim_adventurers.txt)
# Faction: Generic
# Serves: Norsca, the Kurgan and Hung of the hordes, the reavers of Stromfels
#   and the corsairs of Araby
#
# A landless son of a noble house with nothing to inherit gathers a host and
# goes raiding abroad. Every year he weighs his loot: rich enough, he seeks
# land from a realm he has plundered, by bargain with a great lord (the
# negotiation, Part 2 of this file) or by war on a small one, and
# a lord who refused him a title may go home at war. A horde clan driven from
# its pastures takes to raiding in the same way. Defeat, capture or a raid
# called off ends it.
#
# Claimants and landless lords: a landless claimant with nothing to inherit
# gathers exiles and mercenaries for two years and presses a claim by war; a
# landless child of a raiding people's noble house does the same to carve a
# duchy out of a foreign realm. The target is warned, and the court they
# leave hears of it.
#
# The sons of Araby: once in 50 years an ambitious landless son of an
# Arabyan house dreams of finishing Sultan Jaffar's war and sets out to take
# a kingdom in Estalia or Tilea. The target is warned and may raise a coast
# watch; two years later the adventurer lands with a host.
#
# The raider sets out                  adventurer.1 - adventurer.2
# Seeking land                         adventurer.10 - adventurer.19
# Buying land                          adventurer.20 - adventurer.21
# The landless horde                   adventurer.30 - adventurer.31
# The end of the raid                  adventurer.40 - adventurer.49
# Claimants and landless lords         adventurer.50 - adventurer.59
# The sons of Araby                    adventurer.60 - adventurer.62
#
# Fired from: the raiding_adventurers list on on_five_year_pulse in
#   wh_on_actions.txt (adventurer.1), the buy-land decision in
#   HL_title_decisions.txt (adventurer.20), 00_on_actions.txt (on_new_holder*
#   adventurer.40, on_battle_lost_owner adventurer.41,
#   on_become_imprisoned_any_reason adventurer.42, on_ai_end_raid
#   adventurer.43, on_unlanded adventurer.30) and the
#   Cathay invasion in jd_chinese_invasion_events.txt (adventurer.12, .13,
#   .49). The claimant_adventures list on on_five_year_pulse in
#   wh_on_actions.txt (adventurer.50, .53); the Nehekharan claimants
#   (WHG_nehekhara_events.txt) and the Asur explorers (WHG_asur_events.txt)
#   fire adventurer.51, .52, .54, .55 and .59. on_five_year_pulse in
#   00_on_actions.txt (adventurer.60). Shared triggers and effects
#   are in WHG_horse_lords_triggers.txt and WHG_horse_lords_effects.txt.
#
# Also in this file (each part keeps its own header and namespace):
#   Part 2: Bargaining with a raider (namespace advneg, was ADV in
#   adventurers_negotiation.txt)
###############################################################################

namespace = adventurer

# adventurer.1 A host of one's own
# Hidden. A landless son of a noble house of a raiding people, with nothing
# to inherit and no claim for adventurer.50 to pursue, raises a host (3 to 12
# bands, horse or foot, and a fleet for a seafarer) and sets out. The liege
# is told (adventurer.2); the raider takes stock every three years
# (adventurer.10).
# Scope: the would-be raider.
character_event = {
	id = adventurer.1
	hide_window = yes
	is_triggered_only = yes # on_five_year_pulse (raiding_adventurers)

	ai = yes
	min_age = 16
	max_age = 50
	only_men = yes
	capable_only = yes
	prisoner = no

	trigger = {
		has_dlc = "Horse Lords"
		NOT = { has_game_rule = { name = adventurers value = none } }
		is_ruler = no
		NOR = {
			is_witch_hunter_trigger = yes
			is_hunter_culture_trigger = yes
			trait = content
			trait = craven
			trait = decadent
		}
		is_allowed_to_loot = yes
		adventurer_raider_people_trigger = yes
		NOT = { any_heir_title = { always = yes } }
		adventurer_tog_claimant_trigger = no
		adventurer_noble_blood_trigger = yes
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 2 trait = ambitious }
		modifier = { factor = 5 has_character_flag = flag_denied_title }
		modifier = { factor = 5 has_character_flag = demon_child_non_pagan }
		modifier = { factor = 1.33 diplomacy = 12 }
		modifier = { factor = 1.33 martial = 12 }
		modifier = { factor = 1.33 combat_rating = 30 }
		modifier = { factor = 1.33 trait = brave }
		modifier = { factor = 1.18 trait = proud }
		modifier = { factor = 0.33 trait = slothful }
		modifier = { factor = 0.33 trait = humble }
		modifier = { factor = 0.33 trait = kind }
		modifier = { factor = 0.33 any_liege = { ai = no } }
		modifier = { factor = 0.04 has_game_rule = { name = adventurers value = rare } }
	}

	immediate = {
		end_inaccessibility_effect = yes
		add_trait = adventurer
		liege = {
			primary_title = { holder_scope = { save_event_target_as = previous_liege } }
			character_event = { id = adventurer.2 days = 1 }
			capital_scope = {
				save_event_target_as = target_province
				ROOT = {
					set_defacto_liege = ROOT
					create_title = {
						tier = DUKE
						landless = yes
						adventurer = yes
						culture = ROOT
						name = "CLAIMANT_ADVENTURE"
						holder = ROOT
						base_title = THIS
					}
					set_character_flag = raiding_adventurer1
					set_character_flag = raiding_adventurer
					if = {
						limit = {
							event_target:previous_liege = {
								OR = {
									government = nomadic_government
									government = chaos_horde_government
								}
							}
						}
						adventurer_horse_host_effect = yes
					}
					else = { adventurer_raid_host_effect = yes }
					create_random_soldier = {
						random_traits = yes
						dynasty = none
						religion = ROOT
						culture = ROOT
						female = no
						age = 30
					}
					new_character = {
						character_event = { id = aiclean.31 } # Racial trait
					}
					create_random_soldier = {
						random_traits = yes
						dynasty = none
						religion = ROOT
						culture = ROOT
						female = no
						age = 25
					}
					new_character = {
						character_event = { id = aiclean.31 } # Racial trait
					}
					if = {
						limit = { is_seafarer = yes }
						spawn_fleet = {
							province = closest
							owner = ROOT
							disband_on_peace = yes
							troops = { galleys = { 110 110 } }
						}
					}
				}
			}
		}
		character_event = { id = adventurer.10 days = 1095 }
	}
}

# adventurer.10 Taking stock
# Hidden. Every year (first after three) the raider weighs the loot. Rich
# enough (or five years out), with a looted realm in reach, he seeks land
# (adventurer.11), goes home at war if he was once refused a title
# (adventurer.13), or keeps raiding. After thirty years, one last try
# (adventurer.19).
# Scope: the raider.
character_event = {
	id = adventurer.10
	hide_window = yes
	is_triggered_only = yes # adventurer.1, .30, .44, itself

	has_character_flag = raiding_adventurer

	trigger = {
		primary_title = { temporary = yes }
	}

	immediate = {
		if = {
			limit = {
				OR = { wealth = 300 had_character_flag = { flag = raiding_adventurer days = 1825 } }
				any_independent_ruler = { adventurer_looted_ruler_trigger = yes }
			}
			random_list = {
				50 = { # Seek land
					modifier = { factor = 2 trait = cruel }
					modifier = { factor = 2 trait = ambitious }
					character_event = { id = adventurer.11 }
				}
				20 = { # Home, at war
					modifier = { factor = 0 NOT = { has_character_flag = flag_denied_title } }
					modifier = { factor = 2 trait = ambitious }
					modifier = { factor = 3 trait = ambitious trait = arbitrary }
					modifier = { factor = 2 trait = envious }
					character_event = { id = adventurer.13 }
				}
				50 = { # Raid on
					modifier = {
						factor = 0.5
						had_character_flag = { flag = raiding_adventurer days = 2555 }
					}
					modifier = {
						factor = 0.5
						had_character_flag = { flag = raiding_adventurer days = 3285 }
					}
					modifier = {
						factor = 0.5
						had_character_flag = { flag = raiding_adventurer days = 4015 }
					}
					modifier = {
						factor = 0
						had_character_flag = { flag = raiding_adventurer days = 10950 }
					}
					repeat_event = { id = adventurer.10 days = 365 }
				}
			}
		}
		else_if = {
			limit = { had_character_flag = { flag = raiding_adventurer days = 10950 } }
			character_event = { id = adventurer.19 }
		}
		else = { repeat_event = { id = adventurer.10 days = 365 } }
	}
}

# adventurer.11 Land, by bargain or by war
# Hidden. Picks a looted realm: a great lord with land to spare is offered
# a bargain (advneg.1 in Part 2, with a check in 65 days whether the raid goes on,
# adventurer.44); a small one is attacked for a duchy. If neither works the
# adventure ends (adventurer.49).
# Scope: the raider.
character_event = {
	id = adventurer.11
	hide_window = yes
	is_triggered_only = yes # adventurer.10

	immediate = {
		random_independent_ruler = {
			limit = { adventurer_looted_ruler_trigger = yes }
			if = {
				limit = { adventurer_negotiation_partner_trigger = yes }
				set_character_flag = negotiating_with_adventurer
				character_event = { id = advneg.1 }
				ROOT = {
					set_character_flag = adventurer_next_step
					character_event = { id = adventurer.44 days = 65 }
				}
			}
			else_if = {
				limit = { NOT = { higher_real_tier_than = DUKE } }
				adventurer_declare_war_effect = yes
			}
		}
		if = {
			limit = { NOT = { has_character_flag = adventurer_next_step } }
			clr_character_flag = raiding_adventurer
			character_event = { id = adventurer.49 }
		}
		clr_character_flag = adventurer_next_step
	}
}

	id = adventurer.21
	desc = EVTDESCadventurer21

	is_triggered_only = yes # adventurer.20

	option = { # Sell
		name = EVTOPTAadventurer21
		event_target:target_title = { grant_title = FROM }
		hidden_tooltip = {
			FROM = {
				wealth = -150
				if = {
					limit = { ROOT = { primary_title = { higher_tier_than = COUNT } } }

	}

# adventurer.30 Driven from the pastures
# Hidden. A horde clan that loses its last land takes to raiding: its
# overlord and its vassals part ways with it, the clan becomes a landless
# host, and its manpower rides out as looting bands (adventurer.31).
# Scope: the displaced clan chief.
character_event = {
	id = adventurer.30
	hide_window = yes
	is_triggered_only = yes # on_unlanded

	trigger = {
		has_game_started = yes
		is_alive = yes
		is_dying = no
		is_nomadic = yes
		prisoner = no
		NOR = { is_witch_hunter_trigger = yes is_hunter_culture_trigger = yes trait = incapable }
	}

	immediate = {
		clan_title = {
			capital_scope = {
				save_event_target_as = last_province
				holder_scope = {
					top_liege = {
						reverse_opinion = {
							modifier = opinion_driven_from_my_lands
							who = ROOT
							months = 1200
						}
						reverse_opinion = {
							modifier = forbidden_raid_target
							who = ROOT
							months = 1200
						}
					}
				}
			}
		}
		liege = { ROOT = { any_vassal = { set_defacto_liege = PREVPREV } } }
		if = {
			limit = { independent = yes }
			if = {
				limit = { NOT = { any_vassal = { clan = yes } } }
				any_vassal = { set_defacto_liege = THIS }
			}
			random_vassal = { limit = { clan = yes } primary_title = { gain_title = PREVPREV } }
		}
		set_defacto_liege = THIS
		clan_title = {
			set_title_adventurer = { title = THIS status = yes }
			set_title_landless = { title = THIS status = yes }
			set_defacto_liege = THIS
		}
		set_character_flag = raiding_adventurer_nomad
		set_character_flag = raiding_adventurer
		any_army = { set_can_toggle_looting = yes set_looting = yes set_can_toggle_looting = no }
		if = { limit = { NOT = { unused_manpower = 250 } } manpower = 250 }
		character_event = { id = adventurer.31 }
		character_event = { id = adventurer.10 days = 1095 }
	}
}

# Hidden. Every 250 unused manpower becomes a band of 250 horse raiders at

# Scope: the clan chief.

	id = adventurer.31
	hide_window = yes
	is_triggered_only = yes # adventurer.30, itself

	trigger = {
		unused_manpower = 250
	}

	immediate = {
		spawn_unit = {
			owner = ROOT
			province = event_target:last_province
			home = event_target:last_province

			attrition = 1.0
			is_looter = yes
			can_toggle_looting = no
			reinforces = yes
			merge = yes

	}

# adventurer.49 Clean-up
# Hidden. The host disbands and its titles go; a free raider goes back to
# his father, or his father's lord.
# Scope: the raider.
character_event = {
	id = adventurer.49
	hide_window = yes
	is_triggered_only = yes # adventurer.11, .19, .41, .42, .43, Cathay invasion

	immediate = {
		clr_character_flag = raiding_adventurer
		disband_event_forces = yes
		clan_title = { activate_title = { title = THIS status = no } destroy_landed_title = THIS }
		primary_title = {
			if = {
				limit = { OR = { AND = { tier = DUKE adventurer = yes } tier = EMPEROR } }
				activate_title = { title = THIS status = no }
				destroy_landed_title = THIS
			}
		}
		if = {
			limit = { prisoner = no father = { is_alive = yes } }
			father = {
				if = { limit = { is_ruler = yes } ROOT = { set_defacto_liege = PREV } }
				else = { liege = { ROOT = { set_defacto_liege = PREV } } }
			}
		}
	}
}

# adventurer.50 A claim to press
# Hidden. A landless claimant with nothing to inherit, bold enough and of good
# health, plans to press a claim on a kingdom (or the claimed title's own
# tier) held by an independent ruler, unless the ruler is the claimant's own
# liege and the claimant is not ambitious. The holder is warned
# (adventurer.51); the claimant strikes in two years (adventurer.52), from
# exile if the target is the claimant's own lord.
# Scope: the claimant. From the claimant_adventures list on
#   on_five_year_pulse in wh_on_actions.txt (was an MTTH of 50 years on every
#   landless adult), and the Nehekharan claimants in WHG_nehekhara_events.txt
#   (which fire adventurer.51, .52 and .59 directly).
character_event = {
	id = adventurer.50

	hide_window = yes
	is_triggered_only = yes

	min_age = 16
	max_age = 50
	capable_only = yes
	prisoner = no

	trigger = {
		has_dlc = "The Old Gods"

		NOT = { has_game_rule = { name = adventurers value = none } }

		OR = {
			is_female = no
			liege = { has_law = status_of_women_4 }
			liege = { has_law = true_cognatic_succession }
		}

		is_ruler = no

		NOT = { any_spouse = { is_ruler = yes } }

		any_claim = {
			has_holder = yes
			holder_scope = {
				independent = yes # Only do this vs independent rulers
				NAND = {
					OR = { government = chaos_horde_government government = nomadic_government }
					OR = { clan_title = { title = PREVPREV } primary_title = { title = PREVPREV } }
				}
				OR = {
					# Don't target my liege unless I am ambitious
					NOT = { is_vassal_or_below = ROOT }
					ROOT = { trait = ambitious }
				}
				OR = {
					is_merchant_republic = no
					NOT = { capital_holding = { de_jure_liege_or_above = PREVPREV } }
				}
				ROOT = { NOR = { obedient = PREV non_interference = PREV } }
			}
			OR = { tier = KING holder_scope = { tier = PREV } }
		}

		NOT = { # Somewhat clunky way of checking that the character does not stand to inherit any titles
			any_heir_title = { always = yes }
		}

		diplomacy = 7

		health = 4
		NOT = { trait = content }
		NOT = { trait = craven }
		NOT = { is_witch_hunter_trigger = yes }

		NOT = { has_character_modifier = planning_claimant_adventure }

# advneg.10 The raider weighs the price
# Hidden. A raider who can pay may accept (advneg.4); a greedy, ambitious or
# clever one haggles, offering what they can (advneg.5).
# Scope: ROOT is the raider.
character_event = {
	id = advneg.10
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		#If they can afford the price, see if they haggle or not
		export_to_variable = { which = adventurer_treasury value = treasury }
		subtract_variable = { which = adventurer_treasury which = price which = event_target:selling_province }
		if = {
			limit = {
				NOT = { check_variable = { which = adventurer_treasury value = 0 } }
			}
			log = "Adventurer [This.GetBestName]([This.GetID]) is forced to haggle since [This.GetSheHe] can't afford the province."
		}
		random_list = {
			50 = {
				#Deal
				trigger = { check_variable = { which = adventurer_treasury value = 0 } }
				modifier = { NOT = { stewardship = 8 } factor = 2 }
				modifier = { age = 50 factor = 2 }
				modifier = { trait = craven factor = 2 }
				modifier = { trait = content factor = 2 }
				modifier = { trait = charitable factor = 2 }
				modifier = { trait = just factor = 2 }
				event_target:adventurer_target = { letter_event = { id = advneg.4 } }
				ROOT = { set_character_flag = nobreak_advneg_10 }
			}
			50 = { #Haggle
				modifier = { stewardship = 14 factor = 2 }
				modifier = { trait = greedy factor = 2 }
				modifier = { trait = ambitious factor = 2 }
				modifier = { trait = arbitrary factor = 2 }

				export_to_variable = { which = adventurer_treasury value = treasury }
				subtract_variable = { which = adventurer_treasury which = haggle_price which = event_target:selling_province }

				if = {
					limit = { check_variable = { which = adventurer_treasury value = 0 } }
					set_variable = { which = province_offer value = 0 }
					change_variable = { which = province_offer which = haggle_price which = event_target:selling_province }
					set_character_flag = haggle_price
				}

				if = {
					limit = {
						NOT = { check_variable = { which = adventurer_treasury value = 0 } }
						NOT = {
							wealth = 100
						}
					}
					set_character_flag = no_haggling
					log = "[This.GetBestName]([This.GetID]) can't even afford the reduced haggle price."
				}

				if = {
					limit = {
						NOT = { check_variable = { which = adventurer_treasury value = 0 } }
						wealth = 100
					}
					set_variable = { which = province_offer value = 0 }
					export_to_variable = { which = province_offer value = treasury }
					set_character_flag = haggle_price
					log = "[This.GetBestName]([This.GetID]) can't even afford the reduced haggle price - they are offering what they have instead."
				}

				event_target:adventurer_target = { letter_event = { id = advneg.5 } }
			}
		}
		ROOT = { clr_character_flag = nobreak_advneg_10 }
	}
}

# advneg.4 The raider accepts
# The price is paid, the raider takes the county as the lord's vassal, may
# take the lord's faith (advneg.20), and the lord's lands are safe from
# raiders for ten years.
# Scope: ROOT is the lord, FROM the raider.

	id = advneg.4
	desc = EVTDESCadvneg4
	border = GFX_event_letter_frame_diplomacy

	is_triggered_only = yes

			set_variable = { which = transfer_money value = 0 }

			event_target:raiding_adventurer = {
				set_variable = { which = transfer_money value = 0 }
				change_variable = { which = transfer_money which = price which = event_target:selling_province }
				multiply_variable = { which = transfer_money value = -1 }
			}
		}
	}

	option = {
		name = EVTOPTAadvneg4

		#Transfers the price of the selling province
		wealth = transfer_money

		hidden_effect = {
			event_target:raiding_adventurer = {
				wealth = transfer_money
				clr_character_flag = raiding_adventurer # flag set in WHG_adventurer_events.txt
			}

			#Non-Aggression Pact
			opinion = {
				modifier = in_non_aggression_pact
				years = 3
				who = FROM
				origin_description = EVTTITLEadvneg4
			}
			FROM = {
				opinion = {
					modifier = in_non_aggression_pact
					years = 3
					who = ROOT
					origin_description = EVTTITLEadvneg4
				}
			}
		}

		#Changes opinions/prestige if not your province
		if = {
			limit = { NOT = { event_target:selling_province = { owner = { character = ROOT } } } }

			prestige = -50

			event_target:selling_province = {
				owner = {
					opinion = {
						who = ROOT
						modifier = sold_my_province
						years = 100
					}
				}
			}
			if = {
				limit = { any_vassal = { NOT = { any_demesne_province = { province = event_target:selling_province } } } }
				custom_tooltip = { text = EVTOPTAadvneg5_tooltip }
			}
			hidden_effect = {
				any_vassal = {
					limit = { NOT = { any_demesne_province = { province = event_target:selling_province } } }
					opinion = {
						who = ROOT
						modifier = sold_vassal_province
						years = 5
					}
				}
			}
		}

		#Government switch
		if = {
			limit = {
				event_target:raiding_adventurer = { is_nomadic = yes }
				is_nomadic = no
			}
			event_target:raiding_adventurer = {
				if = {
					limit = {
						religion_group = ormazd_gods_group
					}
					set_government_type = muslim_government
				}
				else = {
					set_government_type = feudal_government
				}
			}
		}

		#Grants province and sets liege
		event_target:selling_province = {
			county = { grant_title = event_target:raiding_adventurer }
		}
		event_target:raiding_adventurer = { set_defacto_liege = ROOT }

		#Logging for displaced prince...
		if = {
			limit = { event_target:raiding_adventurer = { has_character_flag = is_displaced_prince } }
			log = "Displaced royal: [raiding_adventurer.GetBestName] settled after buying the province [selling_province.GetName] to the price of [transfer_money.GetValue] ducats, from [Root.GetBestName]."
		}

			log = "Adventurer [raiding_adventurer.GetBestName] settled after buying the province [selling_province.GetName] to the price of [transfer_money.GetValue] ducats."
		}

			if = {
				limit = {
					event_target:raiding_adventurer = {
						NOR = {
							has_character_flag = is_displaced_prince
							religion = ROOT
						}
					}
				}
				event_target:raiding_adventurer = { character_event = { id = advneg.20 } }
			}
		}

			any_realm_province = {

					name = settled_adventurer
					years = 10
				}
			}
			event_target:raiding_adventurer = {
				any_unit = {
					set_can_toggle_looting = yes
					set_looting = no
				}
				set_looter_hostility_days = {
					who = ROOT
					days = 0
				}
			}
			set_looter_hostility_days = {
				who = event_target:raiding_adventurer
				days = 0
			}
		}
	}

	after = {
		clr_character_flag = adventurer_unsafe_duchy_war
		clr_character_flag = adventurer_safe_duchy_war
		clr_character_flag = negotiating_with_adventurer
	}

					set_government_type = feudal_government
				}
			}
		}

		#Gives province and sets liege
		event_target:selling_province = {
			county = { grant_title = event_target:raiding_adventurer }
		}
		event_target:raiding_adventurer = { set_defacto_liege = ROOT }

		#Logging for displaced prince...
		if = {
			limit = { event_target:raiding_adventurer = { has_character_flag = is_displaced_prince } }
			log = "Displaced royal: [raiding_adventurer.GetBestName] settled after buying the province [selling_province.GetName] to the price of [transfer_money.GetValue] ducats, from [Root.GetBestName]."
		}
		if = {
			limit = {
				NOT = { FROM = { has_character_flag = no_haggling } }
			}
			log = "Adventurer [raiding_adventurer.GetBestName]([raiding_adventurer.GetID]) settled after buying the province [selling_province.GetName] to the haggle price of [transfer_money.GetValue] ducats."
		}
		if = {
			limit = {
				FROM = { has_character_flag = no_haggling }
			}
			log = "Adventurer [raiding_adventurer.GetBestName]([raiding_adventurer.GetID]) was given the province of [selling_province.GetName] after [raiding_adventurer.GetSheHe] couldn't afford to buy it."
		}

		# Chance to convert to your faith (not a displaced prince of Cathay)
		hidden_effect = {
			if = {
				limit = {
					event_target:raiding_adventurer = {
						NOR = {
							has_character_flag = is_displaced_prince
							religion = ROOT
						}
					}
				}
				event_target:raiding_adventurer = { character_event = { id = advneg.20 } }
			}
		}

		#Make country safe
		hidden_effect = {
			any_realm_province = {
				add_province_modifier = {
					name = settled_adventurer
					years = 10
				}
			}
			event_target:raiding_adventurer = {
				any_unit = {
					set_can_toggle_looting = yes
					set_looting = no
				}
				set_looter_hostility_days = {
					who = ROOT
					days = 0
				}
			}
			set_looter_hostility_days = {
				who = event_target:raiding_adventurer
				days = 0
			}
		}

		ai_chance = {
			factor = 50
			modifier = { trait = craven factor = 2 }
			modifier = { trait = greedy factor = 1.3 }
			modifier = { event_target:raiding_adventurer = { realm_levy_diff = { who = ROOT value = 1500 } } factor = 1.3 }
			modifier = { event_target:raiding_adventurer = { realm_levy_diff = { who = ROOT value = 3000 } } factor = 1.5 }
			modifier = { event_target:raiding_adventurer = { realm_levy_diff = { who = ROOT value = 5000 } } factor = 1.5 }
		}
	}

	#Decline
	option = {
		name = EVTOPTDadvneg5
		custom_tooltip = { text = EVTOPTDadvneg3_tt }

		trigger = {
			event_target:adventurer_target = {
				NOT = {
					has_character_flag = peaceful_negotiation
				}
			}
		}

		#Adventurer declares war
		raiding_adventurer_war_declaration_effect = yes

		ai_chance = {
			factor = 50
			modifier = { realm_levy_diff = { who = event_target:raiding_adventurer value = 1000 } factor = 1.3 }
			modifier = { realm_levy_diff = { who = event_target:raiding_adventurer value = 2500 } factor = 1.3 }
			modifier = { trait = brave factor = 2 }
			modifier = { trait = wroth factor = 1.5 }
			modifier = { trait = ambitious factor = 1.5 }
			modifier = { event_target:raiding_adventurer = { NOT = { has_character_flag = haggle_price } } factor = 5 }
			modifier = { event_target:selling_province = { holder_scope = { character = ROOT } } factor = 1.5 }
		}
	}

	#Decline - peaceful
	option = {
		name = EVTOPTEadvneg5
		custom_tooltip = { text = EVTOPTEadvneg3_tt }

		trigger = {
			event_target:adventurer_target = {
				has_character_flag = peaceful_negotiation
			}
		}

		clr_character_flag = peaceful_negotiation

		ai_chance = {
			factor = 50
			modifier = { realm_levy_diff = { who = event_target:raiding_adventurer value = 1000 } factor = 1.3 }
			modifier = { realm_levy_diff = { who = event_target:raiding_adventurer value = 2500 } factor = 1.3 }
			modifier = { trait = brave factor = 2 }
			modifier = { trait = wroth factor = 1.5 }
			modifier = { trait = ambitious factor = 1.5 }
			modifier = { event_target:raiding_adventurer = { NOT = { has_character_flag = haggle_price } } factor = 5 }
			modifier = { event_target:selling_province = { holder_scope = { character = ROOT } } factor = 1.5 }
		}
	}

	after = {
		hidden_effect = {
			clr_character_flag = adventurer_unsafe_duchy_war
			clr_character_flag = adventurer_safe_duchy_war
			clr_character_flag = negotiating_with_adventurer
			event_target:raiding_adventurer = { clr_character_flag = haggle_price }
		}
	}
}

# advneg.20 The settled raider's faith
# Hidden. Four times in five (more if cynical, content or craven) the new
# vassal and their landless courtiers take the liege's faith, if it fits
# their race; a player liege hears of it (advneg.21).
# Scope: ROOT is the settled raider.
character_event = {
	id = advneg.20
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		NOT = { liege = { religion = ROOT } }
	}

	immediate = {
		random_list = {
			80 = {
				modifier = { trait = cynical factor = 5 }
				modifier = { trait = content factor = 5 }
				modifier = { trait = craven factor = 5 }
				modifier = { factor = 0 liege = { NOT = { faith_fits_race_of_prev_trigger = yes } } }
				liege = {
					ROOT = { religion = PREV }
					if = { limit = { ai = no } letter_event = { id = advneg.21 days = 7 } }
				}
				any_courtier = {
					limit = {
						is_landed = no
						ROOT = { faith_fits_race_of_prev_trigger = yes }
					}
					religion = ROOT
				}
			}
			20 = {
				modifier = { trait = zealous factor = 5 }
				# Does not convert
			}
		}
	}
}
