###############################################################################
# The Mists of Albion: the lost fleets, the guides and the Ogham stones
# Merged in during the Nov 12 deep pass (each keeps its own header below):
#   The Fimir (namespace fimir, was WHG_fimir_events)
#
# Namespace: albion (was DR_misc.1002 - 1003 and DR_misc.103 in
#   DR_misc_events.txt)
# Faction: Generic
# Serves: every ruler who sends a fleet towards Albion, and the tribes,
#   Truthsayers and Fimir of the island itself
#
# Albion lies hidden in the mists of the Great Ocean, off the western coast
# of Norsca. The Old Ones raised the Ogham stones across the island, and the
# stones weave the mists that hide it. While the mists hang (global flag
# mists_of_albion, set at game start unless the game rule
# warhammer_mists_of_albion is off), any army that sails into the misty sea
# zones is lost to the last man: the ships drift on, the crews are gone.
# The tribes of Albion and the Fimir of its fens know the way through the
# outer mists, but nobody passes the innermost one. An outsider may find a
# guide: a Truthsayer, servant of the Old Ones, leads those who would keep
# the stones, and a Dark Emissary leads those who serve the Dark Masters.
# Either may be a fraud. The tribes find the wrecks of those who sailed
# without one.
#
# The Ogham stones are Albion's places of power (active_place_of_power_man
# and corrupt_place_of_power_man, set in WHG_maintenance_events.txt). The
# corrupt_pop and restore_pop title decisions (wh_magic_decisions.txt)
# topple and raise them. When six of them stand dark, the mists fail. They
# also fail in 2305, or when the doom of the world runs very high.
#
# Lost in the mists                    albion.1 - albion.2
# The mists lift                       albion.9 - albion.10
# A guide through the mists            albion.20 - albion.21
# The Ogham stones                     albion.30 - albion.33
# Wrecks on the shore                  albion.40
#
# Callers: albion.1 from on_unit_entering_province (nb_on_actions.txt);
#   albion.9 from on_yearly_pulse and albion.40 from the yearly list
#   albion_wrecks (wh_on_actions.txt; the old lifting event had no caller,
#   so the mists never lifted); albion.20 - 21 from the
#   albion_seek_mist_guide decision (wh_albion_decisions.txt); albion.30 -
#   31 from corrupt_pop and restore_pop (wh_magic_decisions.txt).
# Scripted triggers: common/scripted_triggers/wh_albion_triggers.txt
# Scripted effects: common/scripted_effects/wh_albion_effects.txt
# Event modifiers: common/event_modifiers/wh_albion_modifiers.txt
# Saved: global flags mists_of_albion and mists_of_albion_cleared;
#   character flags pleasestopalbionwarning (no more warnings),
#   albion_seeking_guide, albion_stone_oath (swore to a Truthsayer) and
#   albion_wreck_found (cooldown); character variable albion_dark_stones
#   (the count of dark stones, for the texts).
# Loc: localisation/00_WHG_albion_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

# albion.1 Lost in the mists
# Hidden. A unit enters a misty sea zone. In the innermost mist (1868)
# every army of the commander and his liege is lost. In the outer mists
# they are lost unless the commander is of Albion or the Fimir, or he or
# his liege has a guide (albion_mist_pilot). The commander, if he lives,
# comes back shaken. A player owner, and a player liege, hear of it
# (albion.2).
# Scope: ROOT is a commander whose unit entered a province.
character_event = {
	id = albion.1
	hide_window = yes

	is_triggered_only = yes # on_unit_entering_province

	has_global_flag = mists_of_albion

	trigger = {
		location = { albion_mist_province_trigger = yes }
		OR = {
			location = { albion_inner_mist_province_trigger = yes }
			AND = {
				albion_native_trigger = no
				albion_mist_guided_trigger = no
			}
		}
	}

	immediate = {
		if = {
			limit = {
				OR = {
					albion_native_trigger = yes
					albion_mist_guided_trigger = yes
				}
			}
			any_army = {
				limit = { location = { albion_inner_mist_province_trigger = yes } }
				troops = -1.00
			}
			liege = {
				any_army = {
					limit = { location = { albion_inner_mist_province_trigger = yes } }
					troops = -1.00
				}
			}
		}
		else = {
			any_army = {
				limit = { location = { albion_mist_province_trigger = yes } }
				troops = -1.00
			}
			liege = {
				any_army = {
					limit = { location = { albion_mist_province_trigger = yes } }
					troops = -1.00
				}
			}
		}
		if = {
			limit = { NOT = { trait = stressed } }
			random = {
				chance = 50
				add_trait = stressed
			}
		}
		save_event_target_as = albion_lost_commander
		if = {
			limit = {
				ai = no
				NOT = { has_character_flag = pleasestopalbionwarning }
			}
			character_event = { id = albion.2 }
		}
		liege = {
			if = {
				limit = {
					NOT = { character = ROOT }
					ai = no
					NOT = { has_character_flag = pleasestopalbionwarning }
				}
				character_event = { id = albion.2 }
			}
		}
	}
}

# albion.20 A guide from the mists
# The guide sought with albion_seek_mist_guide arrives: a Truthsayer for
# those who would keep the stones, a Dark Emissary for the servants of the
# Dark Masters. The Truthsayer asks an oath never to defile the stones; the
# Dark Emissary asks a life. Either way the lord's fleets may pass the
# outer mists for five years (albion_mist_pilot).
# Scope: ROOT is the lord who sought the guide.
character_event = {
	id = albion.20
	desc = {
		text = EVTDESCalbion20_truthsayer
		trigger = { albion_dark_path_trigger = no }
	}
	desc = {
		text = EVTDESCalbion20_emissary
		trigger = { albion_dark_path_trigger = yes }
	}
	picture = GFX_evt_pagan

	is_triggered_only = yes # albion_seek_mist_guide

	immediate = {
		clr_character_flag = albion_seeking_guide
	}

	# Truthsayer: swear the oath
	option = {
		name = EVTOPTAalbion20
		trigger = { albion_dark_path_trigger = no }
		set_character_flag = albion_stone_oath
		add_character_modifier = { name = albion_mist_pilot years = 5 }
		ai_chance = { factor = 90 }
	}
	# Truthsayer: refuse the oath; the guide leaves
	option = {
		name = EVTOPTBalbion20
		trigger = { albion_dark_path_trigger = no }
		prestige = 25
		ai_chance = { factor = 10 }
	}
	# Dark Emissary: give a prisoner to the Dark Masters
	option = {
		name = EVTOPTCalbion20
		trigger = {
			albion_dark_path_trigger = yes
			any_prisoner = { is_ruler = no }
		}
		random_prisoner = {
			limit = { is_ruler = no }
			death = { death_reason = death_sacrificed killer = ROOT }
		}
		piety = 25
		add_character_modifier = { name = albion_mist_pilot years = 5 }
		ai_chance = { factor = 70 }
	}
	# Dark Emissary: give one of your own warriors
	option = {
		name = EVTOPTDalbion20
		trigger = { albion_dark_path_trigger = yes }
		add_character_modifier = { name = albion_mist_pilot years = 5 }
		add_character_modifier = { name = albion_blood_price years = 1 }
		ai_chance = { factor = 30 }
	}
}

# albion.30 The stone falls
# A servant of the Dark Masters has toppled one of Albion's Ogham stones
# (corrupt_pop). The dark stones are counted; at six the mists fail. A lord
# who swore to a Truthsayer is cursed for it (albion.33), and the tribes of
# Albion who play hear of it (albion.32).
# Scope: ROOT is the lord who toppled the stone. Event target albion_stone
# is the stone's province.
character_event = {
	id = albion.30
	desc = EVTDESCalbion30
	picture = GFX_evt_beastmen_herdstone

	is_triggered_only = yes # corrupt_pop

	immediate = {
		albion_count_dark_stones_effect = yes
		any_player = {
			limit = {
				NOT = { character = ROOT }
				albion_native_trigger = yes
				albion_dark_path_trigger = no
			}
			character_event = { id = albion.32 }
		}
	}

	option = {
		name = EVTOPTAalbion30
		if = {
			limit = { has_character_flag = albion_stone_oath }
			clr_character_flag = albion_stone_oath
			character_event = { id = albion.33 days = 7 random = 7 }
		}
		if = {
			limit = {
				has_global_flag = mists_of_albion
				albion_mists_broken_trigger = yes
			}
			custom_tooltip = { text = albion_mists_will_fail_tt }
			hidden_effect = { albion_mists_lift_effect = yes }
		}
	}
}

# albion.40 A wreck on the shore
# While the mists hang, a lord of Albion with a coast finds the wreck of a
# ship that sailed in without a guide: empty, its cargo whole. He may keep
# the cargo, give the dead to the stones, or, if he serves the Dark
# Masters, give them to the fens. At most once in five years.
# Scope: ROOT is a ruler of Albion. Fired by the yearly list albion_wrecks
# (wh_on_actions).
character_event = {
	id = albion.40
	desc = EVTDESCalbion40
	picture = GFX_evt_ship_storm

	is_triggered_only = yes # on_yearly_pulse, albion_wrecks

	only_playable = yes
	has_global_flag = mists_of_albion

	trigger = {
		albion_native_trigger = yes
		OR = {
			NOT = { has_character_flag = albion_wreck_found }
			had_character_flag = { flag = albion_wreck_found days = 1825 }
		}
		capital_scope = {
			port = yes
			region = world_albion
		}
	}

	immediate = {
		clr_character_flag = albion_wreck_found
		set_character_flag = albion_wreck_found
	}

	# Keep the cargo
	option = {
		name = EVTOPTAalbion40
		scaled_wealth = 0.5
		ai_chance = {
			factor = 50
			modifier = { factor = 2 trait = greedy }
		}
	}
	# Give the dead to the stones
	option = {
		name = EVTOPTBalbion40
		trigger = { albion_dark_path_trigger = no }
		piety = 50
		ai_chance = {
			factor = 50
			modifier = { factor = 2 trait = zealous }
		}
	}
	# Give them to the fens, for the Dark Masters
	option = {
		name = EVTOPTCalbion40
		trigger = { albion_dark_path_trigger = yes }
		piety = 40
		prestige = 20
		ai_chance = { factor = 50 }
	}
}

# The Fimir
# Namespace: fimir (was wh_fimir_events.txt)
# Faction: Chaos
# Serves: the Fimir of the fens and mists (culture fimir)
#
# The Fimir are a cyclopean people of the marshes, born of Chaos. Their
# only females are the rare Meargh, witch-queens who rule each clan with
# sorcery; the males (Fimm warriors, Fimm nobles, Dirach sorcerers and the
# lowly Shearl) can only father young on women carried off from the races
# of Men.
#
# The Meargh's power                    fimir.1
# A Fimir mate                          fimir.2
#
# Callers: on_adulthood in wh_on_actions.txt (fimir.1), wedlock.21
#   in WHG_on_action_events.txt when a marriage to a Fimir is undone
#   (fimir.2).
# Saved: event target fimir_raided_realm.
# Loc: localisation/00_WHG_albion_events.csv
###############################################################################

namespace = fimir

# fimir.1 The Meargh's power
# A Meargh comes of age with the witch-sight of her line: she becomes a
# mage of a random lore.
# Scope: ROOT is the Meargh.
character_event = {
	id = fimir.1
	hide_window = yes

	is_triggered_only = yes # on_adulthood
	culture_group = fimir_group

	trigger = {
		trait = fimir_maergh
	}

	immediate = {
		random_lore_mage_effect = yes
	}
}

# fimir.2 A Fimir mate
# A Fimir ruler whose marriage to another people was undone looks for a
# new mate. A male Fimir carries off a woman from a neighbouring realm of
# Men; a Meargh takes a Fimm warrior, a Fimm noble or a Dirach sorcerer as
# her mate. The better the mate, the more it costs her.
# Scope: ROOT is the Fimir ruler.
character_event = {
	id = fimir.2
	desc = {
		text = EVTDESCfimir2_meargh
		trigger = { is_woman_trigger = yes }
	}
	desc = {
		text = EVTDESCfimir2_male
		trigger = { is_woman_trigger = no }
	}
	picture = GFX_evt_cave_lair

	is_triggered_only = yes # intermarriage.002

	trigger = {
		trait = creature_fimir
		is_married = no
	}

	immediate = {
		random_neighbor_independent_ruler = {
			limit = { trait = creature_human }
			save_event_target_as = fimir_raided_realm
		}
	}

	# Carry off a woman of Men
	option = {
		name = EVTOPTAfimir2
		trigger = {
			is_woman_trigger = no
			event_target:fimir_raided_realm = { always = yes }
		}
		event_target:fimir_raided_realm = {
			create_character = {
				age = 22
				female = yes
				random_traits = yes
				dynasty = none
				religion = PREV
				culture = PREV
				trait = creature_human
			}
			new_character = {
				move_character = ROOT
				add_spouse = ROOT
			}
			opinion = { who = ROOT modifier = opinion_furious }
		}
		prestige = 50
	}

	# A Fimm warrior
	option = {
		name = EVTOPTBfimir2
		trigger = { is_woman_trigger = yes }
		create_random_soldier = {
			trait = creature_fimir
			random_traits = yes
			dynasty = none
			martial = 8
		}
		new_character = {
			remove_trait = fimir_noble
			remove_trait = fimir_shearl
			remove_trait = fimir_dirach
			add_trait = fimir_fimm
			add_spouse_matrilineal = ROOT
		}
		prestige = -100
	}

	# A Fimm noble
	option = {
		name = EVTOPTCfimir2
		trigger = {
			is_woman_trigger = yes
			prestige = 500
		}
		create_random_steward = {
			trait = creature_fimir
			random_traits = yes
			stewardship = 8
			martial = 8
		}
		new_character = {
			remove_trait = fimir_fimm
			remove_trait = fimir_shearl
			remove_trait = fimir_dirach
			add_trait = fimir_noble
			add_spouse_matrilineal = ROOT
		}
		prestige = -500
		ai_chance = {
			factor = 1
			modifier = { factor = 0 NOT = { prestige = 1000 } }
		}
	}

	# A Dirach sorcerer
	option = {
		name = EVTOPTDfimir2
		trigger = {
			is_woman_trigger = yes
			prestige = 250
			piety = 250
		}
		create_random_priest = {
			trait = creature_fimir
			random_traits = yes
			stewardship = 8
			learning = 8
		}
		new_character = {
			random_lore_mage_effect = yes
			remove_trait = fimir_fimm
			remove_trait = fimir_shearl
			remove_trait = fimir_noble
			add_trait = fimir_dirach
			add_spouse_matrilineal = ROOT
		}
		prestige = -250
		piety = -250
		ai_chance = {
			factor = 1
			modifier = { factor = 0 NOT = { piety = 500 } }
		}
	}

	# Stay unwed
	option = {
		name = EVTOPTEfimir2
		ai_chance = { factor = 0 }
	}
}
