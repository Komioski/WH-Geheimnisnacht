###############################################################################
# Alternate start events
# Namespace: altstart (was HF.1, HF.2 and HF.5 in HF_alternate_start_events.txt)
# Faction: Generic
#
# One-off set-up for the Shattered World and Random World starts, run by the
# first human player on a new game:
#   altstart.1  Great Conquerors: spawns 4, 8 or 12 conquerors in a Shattered
#               World, as the game rule asks
#   altstart.2  Shattered succession: gives every feudal and tribal realm the
#               succession law the game rule asks for
#   altstart.3  General clean-up: renamed mercenary bands and holy orders,
#               new coats of arms for renamed faiths' heads, a shipyard and
#               naval tech for every port so no one is stranded on an island,
#               and a chance of cognatic succession
#
# Fired from on_startup (common/on_actions/00_on_actions.txt).
# Saved: global flags conquerors_setup, succession_setup,
#   general_maintenance_performed; event targets renaming_target, coa_target
#   (set_random_coa_effect).
# Scripted triggers: common/scripted_triggers/wh_altstart_triggers.txt
# Scripted effects: common/scripted_effects/wh_altstart_effects.txt and
#   02_scripted_effects_alternate_start.txt (vanilla copy)
# Loc: the vanilla ALT_* keys (HolyFury.csv), plus ALT_MERC_INQUISITION and
#   ALT_MERC_RAIDERS in localisation/00_WHG_character_events.csv
# Not carried over: HF.3 (Great Holy Wars and the Hashshashin for random
#   faith names; its caller was already switched off and d_hashshashin is
#   not a mod title). The unicorn prophecy HF.6 - HF.11 moved to
#   WHG_character_events.txt as charfl.70 - 75.
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = altstart

# altstart.1 Great Conquerors
# Scope: the first human player at on_startup.
character_event = {
	id = altstart.1
	hide_window = yes

	is_triggered_only = yes

	only_playable = yes

	trigger = {
		is_shattered_world = yes
		is_save_game = no
		NOT = { has_global_flag = conquerors_setup }
		NOT = {
			has_alternate_start_setting = {
				setting = shattered_conquerors
			}
		}
	}

	immediate = {
		set_global_flag = conquerors_setup
		if = {
			limit = {
				has_alternate_start_setting = {
					setting = shattered_conquerors
					option = on_few
				}
			}
			altstart_four_conquerors_effect = yes
		}
		else_if = {
			limit = {
				has_alternate_start_setting = {
					setting = shattered_conquerors
					option = on_normal
				}
			}
			altstart_four_conquerors_effect = yes
			altstart_four_conquerors_effect = yes
		}
		else_if = {
			limit = {
				has_alternate_start_setting = {
					setting = shattered_conquerors
					option = on_many
				}
			}
			altstart_four_conquerors_effect = yes
			altstart_four_conquerors_effect = yes
			altstart_four_conquerors_effect = yes
		}
	}
}

# altstart.2 Shattered succession
# Every landed feudal or tribal ruler takes the succession law the
# shattered_succession rule asks for (altstart_succession_effect).
# Scope: the first human player at on_startup.
character_event = {
	id = altstart.2
	hide_window = yes

	is_triggered_only = yes

	only_playable = yes

	trigger = {
		is_shattered_world = yes
		is_save_game = no
		NOT = { has_global_flag = succession_setup }
		NOT = {
			has_alternate_start_setting = {
				setting = shattered_succession
				option = default_laws
			}
		}
	}

	immediate = {
		set_global_flag = succession_setup
		if = {
			limit = { altstart_succession_ruler_trigger = yes }
			altstart_succession_effect = yes
		}
		any_playable_ruler = {
			limit = { altstart_succession_ruler_trigger = yes }
			altstart_succession_effect = yes
		}
	}
}
