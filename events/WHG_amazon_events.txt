# amazon.0 Enatic law
# Yearly pulse: an Amazon ruler's primary title is given enatic succession, and
# vassals lose their grudge over the law change (amazon.1).
# Scope: ROOT is an Amazon ruler.
character_event = {
	id = amazon.0
	desc = EVTDESCamazon0
	picture = "GFX_evt_amazon"

	religion = amazon

	is_triggered_only = yes

	trigger = {
		trait = creature_amazon
		primary_title = { NOT = { has_law = enatic_succession } }
	}

	immediate = {
		primary_title = { add_law = enatic_succession }
	}

	option = {
		name = OK
		any_vassal = {
			limit = {
				religion = amazon
				trait = creature_amazon
				has_opinion_modifier = { who = ROOT modifier = succ_law_change }
			}
			# Ping event to clear negative opinion modifiers for succession law change
			character_event = { id = amazon.1 }
		}
	}
}

# amazon.1 Law change forgiven
# Clears the succession-law opinion maluses toward the queen.
# Scope: ROOT is a vassal, FROM the queen.
character_event = {
	id = amazon.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		OR = {
			has_opinion_modifier = { who = FROM modifier = succ_law_change }
			has_opinion_modifier = { who = FROM modifier = bad_succ_law_change }
			has_opinion_modifier = { who = FROM modifier = good_succ_law_change }
		}
	}

	immediate = {
		remove_opinion = { who = FROM modifier = good_succ_law_change }
		remove_opinion = { who = FROM modifier = bad_succ_law_change }
		remove_opinion = { who = FROM modifier = succ_law_change }
	}

	option = {
		name = OK
	}
}

# amazon.2 No man holds Amazon land
# Yearly pulse: a man holding an Amazon barony gives it to his wife, a woman
# of his court or a new Amazon, and is sacrificed.
# Scope: ROOT is the male baron.
character_event = {
	id = amazon.2
	hide_window = yes

	religion = amazon
	only_men = yes

	is_triggered_only = yes

	trigger = {
		is_feudal = no
		trait = creature_amazon
		num_of_baron_titles = 1
		lower_tier_than = COUNT
		NOT = { has_character_flag = amazon_exception }
	}

	immediate = {
		# Abdicate to spouse, if married
		if = {
			limit = { is_married = yes any_spouse = { is_female = yes } }
			random_spouse = { limit = { is_female = yes } ROOT = { abdicate_to = PREV } }
			amazon_sacrifice_effect = yes
		}
		# Otherwise, to female courtier
		else_if = {
			limit = { any_courtier = { is_female = yes } }
			random_courtier = { limit = { is_female = yes } ROOT = { abdicate_to = PREV } }
			amazon_sacrifice_effect = yes
		}
		else = {
			create_character = {
				female = yes
				random_traits = yes
				religion = amazon
				culture = amazon_main
				trait = creature_amazon
			}
			new_character = { ROOT = { abdicate_to = PREV } }
			amazon_sacrifice_effect = yes
		}
	}

	option = {
		name = OK
	}
}

# amazon.3 Men loose at court
# Bi-yearly pulse: free men at an Amazon court are sent to the male dorms
# (in_stock), or kept free at a cost in piety and decadence.
# Scope: ROOT is an Amazon ruler.
character_event = {
	id = amazon.3
	desc = EVTDESCamazon3 # One or more males are running around freely in your court. This cannot be allowed to stand; They must be sent to the guarded male dorms immediately and put to good use!
	picture = "GFX_evt_amazon_queen"

	religion = amazon
	only_women = yes

	is_triggered_only = yes

	trigger = {
		is_landed = yes
		trait = creature_amazon
		any_courtier = {
			age >= 13
			NOR = { trait = in_stock has_character_flag = amazon_exception is_female = yes }
		}
	}

	# Yes, immediately!
	option = {
		name = EVTOPTAamazon3
		ai_chance = { factor = 65 }
		any_courtier = {
			limit = {
				age >= 13
				NOR = { trait = in_stock has_character_flag = amazon_exception is_female = yes }
			}
			add_trait = in_stock
			if = { limit = { ai = no } character_event = { id = amazon.4 } }
		}
	}
	# No, let these ones keep roaming free
	option = {
		name = EVTOPTBamazon3
		ai_chance = { factor = 35 modifier = { factor = 2.5 trait = kind } }
		piety = -100
		decadence = 2
		any_courtier = {
			limit = {
				age >= 13
				NOR = { trait = in_stock has_character_flag = amazon_exception is_female = yes }
			}
			set_character_flag = amazon_exception
			religion = amazon
		}
	}

	after = {
		any_courtier = {
			if = {
				limit = { NOR = { trait = disinherited_amazon is_female = yes } }
				add_trait = disinherited_amazon
				set_character_flag = disinherited_by_amazon_laws
			}
		}
	}
}

# amazon.6 The spring fertility rite
# From the hold_fertility_ritual decision: the host opens the spring rite.
# Every grown sister of the court picks a captive (amazon.7) and the feast
# brings two rounds of flavour (amazon.15). Ends after 60 days (amazon.14).
# Scope: ROOT is the host.
character_event = {
	id = amazon.6
	desc = EVTDESCamazon6
	picture = GFX_evt_amazon_feast

	is_triggered_only = yes

	# Let the rite begin
	option = {
		name = EVTOPTAamazon6
		hidden_effect = {
			save_event_target_as = amazon_feast_host
			set_character_flag = holding_amazon_ritual
			diplomatic_immunity = yes
			long_character_event = { id = amazon.7 days = 20 }
			character_event = { id = amazon.15 days = 2 random = 3 }
			character_event = { id = amazon.15 days = 12 random = 10 }
			character_event = { id = amazon.14 days = 60 }
			any_courtier = {
				limit = {
					is_adult = yes
					is_female = yes
					prisoner = no
					OR = { NOR = { age = 45 num_of_children = 5 } trait = immortal }
				}
				set_character_flag = holding_amazon_ritual
				diplomatic_immunity = yes
				long_character_event = { id = amazon.7 days = 20 }
				character_event = { id = amazon.14 days = 60 }
			}
		}
	}
	# Maybe next year
	option = {
		name = EVTOPTBamazon6
		trigger = { ai = no }
	}
}

# off and the host loses face.

	desc = {
		text = EVTDESCamazon47
		trigger = { check_variable = { which = brawl_player value = 3 } }
	}
	desc = {
		text = EVTDESCamazon47b
		trigger = { NOT = { check_variable = { which = brawl_player value = 3 } } }
	}
	picture = GFX_evt_amazon_fighter

	is_triggered_only = yes

	# Let the blood spill
	option = {
		name = EVTOPTAamazon47
		trigger = { check_variable = { which = brawl_player value = 3 } }
		add_trait = questing
		hidden_effect = {
			save_event_target_as = amazon_tournament_host
			set_variable = { which = brawl_player value = 0 }
			any_realm_character = {
				limit = {
					has_character_flag = brawling_participant
					NOT = { has_character_flag = brawling_host }
				}
				long_character_event = { id = amazon.48 }
			}
			character_event = { id = amazon.49 }
		}
	}
	# A waste of time
	option = {
		name = EVTOPTBamazon47
		trigger = { NOT = { check_variable = { which = brawl_player value = 3 } } }
		prestige = -250
		hidden_effect = {
			set_variable = { which = brawl_player value = 0 }
			character_event = { id = amazon.43 }
			any_realm_character = {
				limit = {
					has_character_flag = brawling_participant
					NOT = { has_character_flag = brawling_host }
				}
				character_event = { id = amazon.52 days = 7 random = 10 }

	}

# Scope: ROOT is the huntress, FROM the queen.
character_event = {
	id = amazon.260
	desc = EVTDESCamazon260
	picture = GFX_evt_amazon_huntress

	is_triggered_only = yes

	option = {
		name = EVTOPTAamazon260
		decadence = 1
		remove_trait = questing
		random = { chance = 60 add_trait = wounded }
		random = { chance = 15 add_trait = maimed }
		hidden_effect = {
			amazon_hunt_clear_effect = yes
			random = {
				chance = 2
				if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
				else = { death = { death_reason = death_accident } }
			}
		}
	}
}

# amazon.261 Who brings down the bigger beast?
# The two huntresses are evenly matched and turn the hunt into a contest.
# Picks the story and the winner; the winner hears it first.
# Scope: ROOT is the huntress, FROM the queen.
character_event = {
	id = amazon.261
	desc = EVTDESCamazon261
	picture = GFX_evt_amazon_huntress

	is_triggered_only = yes

	option = {
		name = EVTOPTAamazon261
		hidden_effect = {
			FROM = { save_event_target_as = amazon_contestant_1 }
			save_event_target_as = amazon_contestant_2
			amazon_contest_story_effect = yes
			amazon_contest_decide_effect = yes
			event_target:amazon_contest_winner = { long_character_event = { id = amazon.262 } }
		}
	}
}
