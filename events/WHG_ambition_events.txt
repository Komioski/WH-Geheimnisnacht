# ambition.1 A vassal wants a council seat
# From the become-councillor ambition decision: promise to consider it,
# refuse, or flatter them away (diplomacy 15).
# Scope: ROOT is the liege, FROMFROM the aspiring vassal.
character_event = {
	id = ambition.1
	picture = GFX_evt_courtiers_talking
	border = GFX_event_normal_frame_diplomacy
	desc = EVTDESCambition1

	is_triggered_only = yes

	trigger = {
		prisoner = no
		is_adult = yes
		NOR = { trait = incapable is_inaccessible_trigger = yes }
	}

	immediate = {
		FROMFROM = { save_event_target_as = aspiring_councillor }
		ROOT = { save_event_target_as = target_liege }
	}

	option = {
		name = EVTOPTAambition1
		ai_chance = {
			factor = 30
			modifier = {
				factor = 3
				OR = { opinion = { who = FROMFROM value = 75 } is_friend = FROMFROM }
			}
			modifier = { factor = 0 is_rival = FROMFROM }
		}
		hidden_tooltip = { FROMFROM = { set_character_flag = council_membership_requested } }
		FROMFROM = { character_event = { id = ambition.4 days = 1 tooltip = EVTTOOLTIPambition4 } }
	}

	option = {
		name = EVTOPTBambition1
		ai_chance = {
			factor = 30
			modifier = {
				factor = 0
				OR = { opinion = { who = FROMFROM value = 75 } is_friend = FROMFROM }
			}
			modifier = { factor = 3 is_rival = FROMFROM }
			modifier = { factor = 0 diplomacy = 15 }
		}
		FROMFROM = {
			set_character_flag = council_membership_denied
			opinion = { modifier = opinion_very_disappointed who = ROOT years = 5 }
			character_event = { id = ambition.5 days = 1 tooltip = EVTTOOLTIPambition5 }
		}
	}

	option = {
		name = EVTOPTCambition1
		trigger = { diplomacy = 15 }
		ai_chance = {
			factor = 30
			modifier = {
				factor = 0
				OR = { opinion = { who = FROMFROM value = 75 } is_friend = FROMFROM }
			}
			modifier = { factor = 3 is_rival = FROMFROM }
		}
		FROMFROM = {
			set_character_flag = council_membership_denied
			opinion = { modifier = opinion_flattered who = ROOT years = 2 }
			character_event = { id = ambition.6 days = 1 tooltip = EVTTOOLTIPambition6 }
		}
		tooltip_info = diplomacy
	}
}

# ambition.2 Reminding the liege
# 90 days after a promise the vassal may press again; a councillor to
# replace is picked if no fitting seat is free.
# Scope: ROOT is the vassal.
character_event = {
	id = ambition.2
	picture = GFX_evt_courtiers_talking
	border = GFX_event_normal_frame_diplomacy
	desc = EVTDESCambition2

	is_triggered_only = yes

	trigger = {
		has_ambition = obj_become_council_member
		has_character_flag = council_membership_requested
		NOT = { has_character_flag = council_membership_denied }
		prisoner = no
		is_adult = yes
		NOR = { trait = incapable is_inaccessible_trigger = yes }
	}

	immediate = {
		liege = {
			# Only when no fitting seat is empty: pick a councillor to replace,
			# a rival or an unfriendly one first
			if = {
				limit = {
					NOT = {
						OR = {
							AND = {
								NOT = { job_chancellor = { always = yes } }
								ROOT = { can_hold_title = job_chancellor }
							}
							AND = {
								NOT = { job_marshal = { always = yes } }
								ROOT = { can_hold_title = job_marshal }
							}
							AND = {
								NOT = { job_treasurer = { always = yes } }
								ROOT = { can_hold_title = job_treasurer }
							}
							AND = {
								NOT = { job_spymaster = { always = yes } }
								ROOT = { can_hold_title = job_spymaster }
							}
							AND = {
								NOT = { job_spiritual = { always = yes } }
								ROOT = { can_hold_title = job_spiritual }
							}
						}
					}
				}
				random_vassal = {
					limit = {
						NOT = { character = ROOT }
						is_councillor = yes
						OR = {
							AND = {
								has_job_title = job_chancellor
								ROOT = { can_hold_title = job_chancellor }
							}
							AND = {
								has_job_title = job_marshal
								ROOT = { can_hold_title = job_marshal }
							}
							AND = {
								has_job_title = job_treasurer
								ROOT = { can_hold_title = job_treasurer }
							}
							AND = {
								has_job_title = job_spymaster
								ROOT = { can_hold_title = job_spymaster }
							}
							AND = {
								has_job_title = job_spiritual
								ROOT = { can_hold_title = job_spiritual }
							}
						}
					}
					preferred_limit = { is_rival = ROOT }
					preferred_limit = { NOT = { opinion = { who = ROOT value = 0 } } }
					preferred_limit = { NOT = { among_most_powerful_vassals = 5 } }
					save_event_target_as = councillor_target
				}
			}
		}
	}

	option = {
		name = EVTOPTAambition2
		ai_chance = { factor = 100 }
		liege = { character_event = { id = ambition.3 days = 3 tooltip = EVTTOOLTIPambition3 } }
	}

	option = {
		name = EVTOPTBambition2
		ai_chance = { factor = 0 }
	}
}

# ambition.3 The vassal presses
# Replace a councillor, give an empty seat, refuse, or set them against each
# other with made-up slander (intrigue 15).
# Scope: ROOT is the liege, FROM the vassal.
character_event = {
	id = ambition.3
	picture = GFX_evt_courtiers_talking
	border = GFX_event_normal_frame_diplomacy

	#if we have a councillor in mind
	desc = {
		text = EVTDESCambition3a
		trigger = { event_target:councillor_target = { always = yes } }
	}

	#if there is an open spot
	desc = {
		text = EVTDESCambition3b
		trigger = { NOT = { event_target:councillor_target = { always = yes } } }
	}

	is_triggered_only = yes

	trigger = {
		prisoner = no
		is_adult = yes
		NOR = { trait = incapable is_inaccessible_trigger = yes }
	}

	#We fire the current councillor and replace with new
	option = {
		name = EVTOPTAambition3
		trigger = { event_target:councillor_target = { always = yes } }
		ai_chance = {
			factor = 30
			modifier = { factor = 3 OR = { opinion = { who = FROM value = 75 } is_friend = FROM } }
			modifier = { factor = 0 is_rival = FROM }
		}
		event_target:councillor_target = {
			opinion = { modifier = opinion_fired_from_council who = ROOT years = 5 }
			if = {
				limit = { has_job_title = job_chancellor }
				remove_title = job_chancellor
				event_target:aspiring_councillor = {
					clr_character_flag = council_membership_requested
					give_job_title = job_chancellor
				}
			}
			if = {
				limit = { has_job_title = job_treasurer }
				remove_title = job_treasurer
				event_target:aspiring_councillor = {
					clr_character_flag = council_membership_requested
					give_job_title = job_treasurer
				}
			}
			if = {
				limit = { has_job_title = job_spiritual }
				remove_title = job_spiritual
				event_target:aspiring_councillor = {
					clr_character_flag = council_membership_requested
					give_job_title = job_spiritual
				}
			}
			if = {
				limit = { has_job_title = job_spymaster }
				remove_title = job_spymaster
				event_target:aspiring_councillor = {
					clr_character_flag = council_membership_requested
					give_job_title = job_spymaster
				}
			}
			if = {
				limit = { has_job_title = job_marshal }
				remove_title = job_marshal
				event_target:aspiring_councillor = {
					clr_character_flag = council_membership_requested
					give_job_title = job_marshal
				}
			}
		}
	}

	#We have an open spot and approve the request
	option = {
		name = EVTOPTBambition3
		trigger = { NOT = { event_target:councillor_target = { always = yes } } }
		ai_chance = {
			factor = 30
			modifier = { factor = 3 OR = { opinion = { who = FROM value = 75 } is_friend = FROM } }
			modifier = { factor = 0 is_rival = FROM }
		}
		if = {
			limit = {
				NOT = { job_chancellor = { always = yes } }
				FROM = { can_hold_title = job_chancellor }
			}
			event_target:aspiring_councillor = {
				clr_character_flag = council_membership_requested
				give_job_title = job_chancellor
			}
		}
		else_if = {
			limit = {
				NOT = { job_treasurer = { always = yes } }
				FROM = { can_hold_title = job_treasurer }
			}
			event_target:aspiring_councillor = {
				clr_character_flag = council_membership_requested
				give_job_title = job_treasurer
			}
		}
		else_if = {
			limit = {
				NOT = { job_spiritual = { always = yes } }
				FROM = { can_hold_title = job_spiritual }
			}
			event_target:aspiring_councillor = {
				clr_character_flag = council_membership_requested
				give_job_title = job_spiritual
			}
		}
		else_if = {
			limit = {
				NOT = { job_spymaster = { always = yes } }
				FROM = { can_hold_title = job_spymaster }
			}
			event_target:aspiring_councillor = {
				clr_character_flag = council_membership_requested
				give_job_title = job_spymaster
			}
		}
		else_if = {
			limit = {
				NOT = { job_marshal = { always = yes } }
				FROM = { can_hold_title = job_marshal }
			}
			event_target:aspiring_councillor = {
				clr_character_flag = council_membership_requested
				give_job_title = job_marshal
			}
		}
	}

	#We decline the request
	option = {
		name = EVTOPTCambition3
		ai_chance = {
			factor = 70
			modifier = { factor = 0 OR = { opinion = { who = FROM value = 75 } is_friend = FROM } }
			modifier = { factor = 3 is_rival = FROM }
			modifier = { factor = 0 intrigue = 15 }
		}
		FROM = {
			set_character_flag = council_membership_denied
			opinion = { modifier = opinion_very_disappointed who = ROOT years = 5 }
			character_event = { id = ambition.5 days = 1 }
		}
	}

	#We play them against each other
	option = {
		name = EVTOPTDambition3
		trigger = { event_target:councillor_target = { always = yes } intrigue = 15 }
		ai_chance = {
			factor = 50
			modifier = { factor = 0 OR = { opinion = { who = FROM value = 75 } is_friend = FROM } }
			modifier = { factor = 3 ROOT = { is_rival = FROM } }
		}
		tooltip_info = intrigue
		event_target:aspiring_councillor = {
			set_character_flag = council_membership_denied
			character_event = { id = ambition.7 days = 1 tooltip = EVTTOOLTIPambition7 }
		}
	}
}

# ambition.10 Land for a son or daughter
# on_new_holder: marks the land-for-son/daughter ambition fulfilled when the
# child gets land.
# Scope: ROOT is the new holder, FROMFROM the old holder; hidden.
character_event = {
	id = ambition.10
	hide_window = yes

	is_triggered_only = yes
	only_capable = yes

	trigger = {
		is_heir = no
		NOT = { trait = incapable }
		OR = {
			AND = { ROOT = { is_female = no } father = { has_ambition = obj_land_for_son } }
			AND = { ROOT = { is_female = no } mother = { has_ambition = obj_land_for_son } }
			AND = { ROOT = { is_woman_trigger = yes } father = { has_ambition = obj_land_for_daughter } }
			AND = { ROOT = { is_woman_trigger = yes } mother = { has_ambition = obj_land_for_daughter } }
		}
	}

	immediate = {
		father = {
			if = {
				limit = {
					has_ambition = obj_land_for_son
					ROOT = { is_landed = yes is_female = no } #Might be making him a landless mercenary.
				}
				set_character_flag = land_for_son_successful
				opinion = { modifier = opinion_ambition_fulfilled who = FROMFROM years = 5 }
			}
		}
		mother = {
			if = {
				limit = {
					has_ambition = obj_land_for_son
					ROOT = { is_landed = yes is_female = no } #Might be making him a landless mercenary.
				}
				set_character_flag = land_for_son_successful
				opinion = { modifier = opinion_ambition_fulfilled who = FROMFROM years = 5 }
			}
		}
		father = {
			if = {
				limit = {
					has_ambition = obj_land_for_daughter
					ROOT = { is_landed = yes is_woman_trigger = yes } #Might be making him a landless mercenary.
				}
				set_character_flag = land_for_daughter_successful
				opinion = { modifier = opinion_ambition_fulfilled who = FROMFROM years = 5 }
			}
		}
		mother = {
			if = {
				limit = {
					has_ambition = obj_land_for_daughter
					ROOT = { is_landed = yes is_woman_trigger = yes } #Might be making him a landless mercenary.
				}
				set_character_flag = land_for_daughter_successful
				opinion = { modifier = opinion_ambition_fulfilled who = FROMFROM years = 5 }
			}
		}
	}
}

# ambition.12 Land for my children
# From the ambition decision: grant a county with gold, refuse, or grant it
# for a favour.

	id = ambition.12
	desc = { trigger = { FROMFROM = { has_ambition = obj_land_for_son } } text = EVTDESCambition12 }
	desc = {
		trigger = { FROMFROM = { has_ambition = obj_land_for_daughter } }
		text = EVTDESCambition12_b
	}
	picture = {
		trigger = { FROMFROM = { has_ambition = obj_land_for_son } }
		picture = GFX_evt_son_asking_father
	}
	picture = {
		trigger = { FROMFROM = { has_ambition = obj_land_for_daughter } }

	}

		}

			if = {
				limit = { has_ambition = obj_land_for_son }

					limit = {
						is_female = no
						is_landed = no
						is_adult = yes
						is_heir = no

				}
			}
			if = {
				limit = { has_ambition = obj_land_for_daughter }
				random_child = {
					limit = {
						is_woman_trigger = yes
						is_landed = no
						is_adult = yes
						is_heir = no
						prisoner = no
						NOR = { trait = incapable is_ascetic_trigger = yes }
						liege = { character = PREVPREV }
					}
					save_event_target_as = chosen_son
				}
			}
		}

	option = {
		name = {
			text = EVTOPTAambition12
			trigger = { FROMFROM = { has_ambition = obj_land_for_son } }
		}
		name = {
			text = EVTOPTAambition12_b

		}
		ai_chance = { factor = 20 }
		wealth = 200

		FROMFROM = { character_event = { id = ambition.13 } }
	}

	option = {
		name = EVTOPTBambition12
		ai_chance = { factor = 80 }
		FROMFROM = { character_event = { id = ambition.14 } }
	}

	option = {
		trigger = { NOT = { FROMFROM = { owes_favor_to = ROOT } } }
		name = EVTOPTCambition12
		ai_chance = { factor = 20 }
		event_target:chosen_son = { grant_title = event_target:requested_province }
		add_favor = FROMFROM
		hidden_tooltip = { FROMFROM = { character_event = { id = ambition.15 } } }
	}
}

# ambition.20 Change council position
# From the change-position decision: records the councillor's current seat.
# Scope: ROOT is the councillor; hidden, fires ambition.21 for the liege.
character_event = {
	id = ambition.20
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		if = { limit = { has_job_title = job_chancellor } set_character_flag = is_chancellor }
		if = { limit = { has_job_title = job_marshal } set_character_flag = is_marshal }
		if = { limit = { has_job_title = job_treasurer } set_character_flag = is_treasurer }
		if = { limit = { has_job_title = job_spymaster } set_character_flag = is_spymaster }
		if = { limit = { has_job_title = job_spiritual } set_character_flag = is_spiritual }
		if = {
			limit = { has_minor_title = title_councilmember_king }
			set_character_flag = is_councilmember_king
		}
		if = {
			limit = { has_minor_title = title_councilmember_emperor }
			set_character_flag = is_councilmember_emperor
		}
		save_event_target_as = moving_councillor
		liege = { character_event = { id = ambition.21 } }
	}
}

# ambition.21 Choosing a new seat
# Pick a seat (chancellor, marshal, treasurer, or advisor); swaps with the
# holder if taken.
# Scope: ROOT is the liege; moving_councillor the councillor.
character_event = {

# ambition.200 A war is won by its attacker
# Marks the war ambitions of the winner. Most are judged by the loser, since
# the war's casus belli can no longer be read here.
# Scope: ROOT is the defender (lost), FROM the attacker (won).
character_event = {
	id = ambition.200
	hide_window = yes

	is_triggered_only = yes # on_war_ended_victory

	trigger = {
		FROM = {
			OR = {
				has_ambition = obj_take_more_land
				has_ambition = obj_win_a_condotta
				has_ambition = obj_win_a_war_of_conquest
				has_ambition = obj_serve_the_great_plan
				has_ambition = obj_conquer_for_myrmidia
				has_ambition = obj_beat_da_umies
				has_ambition = obj_walk_the_wild_hunt
				has_ambition = obj_lead_the_tribes_south
				has_ambition = obj_make_a_great_waaagh
				has_ambition = obj_humble_a_rival_emperor
				has_ambition = obj_avenge_the_sundering
				has_ambition = obj_destroy_the_great_betrayers
				has_ambition = obj_take_a_thousand_slaves
			}
		}
	}

	immediate = {
		FROM = {
			if = {
				limit = {
					has_ambition = obj_take_more_land
					ROOT = { higher_tier_than = BARON }
				}
				set_character_flag = amb_land_taken
			}
			else_if = {
				limit = {
					has_ambition = obj_win_a_condotta
					ROOT = { NOT = { culture_group = tilean_group } }
				}
				set_character_flag = amb_condotta
			}
			else_if = {
				limit = {
					has_ambition = obj_win_a_war_of_conquest
					ROOT = { higher_tier_than = COUNT }
				}
				set_character_flag = amb_conquest_won
			}
			else_if = {
				limit = {
					has_ambition = obj_take_a_thousand_slaves
					ROOT = { NOT = { culture_group = druchii_group } }
				}
				change_variable = { which = amb_slave_raids value = 1 }
			}
			else_if = {
				limit = {
					has_ambition = obj_make_a_great_waaagh
					ROOT = { higher_tier_than = DUKE }
				}
				change_variable = { which = amb_waaagh_kingdoms value = 1 }
			}
			else_if = {
				limit = {
					has_ambition = obj_lead_the_tribes_south
					ROOT = { capital_scope = { region = world_old_world_empire } }
				}
				set_character_flag = amb_tribes_south
			}
			else = {
				ROOT = { save_event_target_as = amb_war_loser }
				WHG_ambition_war_won_effect = yes
			}
		}
	}
}

	hide_window = yes

	is_triggered_only = yes # on_assassination

	trigger = {
		FROM = {
			OR = {
				has_ambition = obj_sneak_a_good_stab
				has_ambition = obj_kill_a_man_thing
				has_ambition = obj_weave_the_great_scheme
				has_ambition = obj_betray_a_rival_clan
				has_ambition = obj_climb_over_others
			}
		}
	}

	immediate = {
		FROM = {
			if = {
				limit = { has_ambition = obj_sneak_a_good_stab }
				set_character_flag = amb_good_stab
			}
			else_if = {
				limit = {
					has_ambition = obj_kill_a_man_thing
					ROOT = { higher_tier_than = DUKE }
				}
				set_character_flag = amb_man_thing_killed
			}
			else_if = {
				limit = { has_ambition = obj_weave_the_great_scheme }
				change_variable = { which = amb_schemes value = 1 }
			}
			else_if = {
				limit = {
					has_ambition = obj_betray_a_rival_clan
					ROOT = {
						trait = creature_skaven
						is_ruler = yes
					}
				}
				set_character_flag = amb_clan_betrayed
			}
			else_if = {
				limit = {
					has_ambition = obj_climb_over_others
					ROOT = { trait = creature_skaven }
				}
				set_character_flag = amb_climbed
			}
		}
	}
}

# ambition.207 A holding is finished
# A temple for Build a temple; a castle or a city for Settle the wilds.
# Scope: ROOT is the builder, FROM the new holding.
character_event = {
	id = ambition.207
	hide_window = yes

	is_triggered_only = yes # on_settlement_construction_completed

	trigger = {
		OR = {
			has_ambition = obj_build_a_temple
			has_ambition = obj_settle_the_wilds
		}
	}

	immediate = {
		if = {
			limit = {
				has_ambition = obj_build_a_temple
				FROM = { holding_type = temple }
			}
			set_character_flag = amb_temple_built
		}
		else_if = {
			limit = {
				has_ambition = obj_settle_the_wilds
				FROM = {
					OR = {
						holding_type = castle
						holding_type = city
					}
				}
			}
			set_character_flag = amb_wilds_settled
		}
	}
}

# ambition.208 A holding is looted
# Marks the raiding ambitions of the looter: a Dawi hold, the Empire and
# Bretonnia, a holding of men, a temple-city, the great cities of men.
# Scope: ROOT is the looter, FROM the looted holding.
character_event = {
	id = ambition.208
	hide_window = yes

	is_triggered_only = yes # on_settlement_looted

	trigger = {
		OR = {
			has_ambition = obj_loot_a_stunty_hold
			has_ambition = obj_raid_the_southlands
			has_ambition = obj_raze_the_cities_of_men
			has_ambition = obj_find_a_temple_city
			has_ambition = obj_bring_the_great_ascendancy
		}
	}

	immediate = {
		if = {
			limit = {
				has_ambition = obj_loot_a_stunty_hold
				FROM = { holder_scope = { culture_group = dwarf_group } }
			}
			set_character_flag = amb_stunty_looted
		}
		else_if = {
			limit = {
				has_ambition = obj_raid_the_southlands
				FROM = {
					location = {
						OR = {
							region = world_old_world_empire
							region = world_old_world_bretonnia
						}
					}
				}
			}
			change_variable = { which = amb_south_raids value = 1 }
		}
		else_if = {
			limit = {
				has_ambition = obj_raze_the_cities_of_men
				FROM = { holder_scope = { is_living_human_trigger = yes } }
			}
			change_variable = { which = amb_men_razed value = 1 }
		}
		else_if = {
			limit = {
				has_ambition = obj_find_a_temple_city
				FROM = { location = { any_province_holding = { has_building = ct_templecity_1 } } }
			}
			set_character_flag = amb_temple_city
		}
		else_if = {
			limit = { has_ambition = obj_bring_the_great_ascendancy }
			if = {
				limit = { FROM = { de_jure_liege_or_above = c_altdorf } }
				set_character_flag = amb_asc_altdorf
			}
			else_if = {
				limit = { FROM = { de_jure_liege_or_above = c_couronne } }
				set_character_flag = amb_asc_couronne
			}
			else_if = {
				limit = { FROM = { de_jure_liege_or_above = c_kislev } }
				set_character_flag = amb_asc_kislev
			}
			else_if = {
				limit = { FROM = { de_jure_liege_or_above = c_middenheim } }
				set_character_flag = amb_asc_middenheim
			}
			else_if = {
				limit = { FROM = { de_jure_liege_or_above = c_nuln } }
				set_character_flag = amb_asc_nuln
			}
		}
	}
}

# ambition.210 An outbreak starts
# Brew a new plague counts an outbreak in the realm of a foe at war with the
# holder; Spread the Grandfather's gift counts one in another realm that
# borders the holder's or is at war with them.
# Scope: ROOT is the province.
province_event = {
	id = ambition.210
	hide_window = yes

	is_triggered_only = yes # on_outbreak

	trigger = { has_owner = yes }

	immediate = {
		owner = {
			top_liege = { save_event_target_as = amb_outbreak_realm }
		}
		any_playable_ruler = {
			limit = {
				OR = {
					has_ambition = obj_brew_a_new_plague
					has_ambition = obj_spread_the_gift
				}
				NOT = { same_realm = event_target:amb_outbreak_realm }
			}
			save_event_target_as = amb_outbreak_witness
			if = {
				limit = {
					has_ambition = obj_brew_a_new_plague
					war_with = event_target:amb_outbreak_realm
				}
				set_character_flag = amb_plague_brewed
			}
			else_if = {
				limit = {
					has_ambition = obj_spread_the_gift
					OR = {
						war_with = event_target:amb_outbreak_realm
						ROOT = {
							any_neighbor_province = {
								owner = { same_realm = event_target:amb_outbreak_witness }
							}
						}
					}
				}
				change_variable = { which = amb_gifts value = 1 }
			}
		}
	}
}

# ambition.212 A holy war fails
# A Black Crusade breaks: the defender may have held Break the Black Crusade,
# and the lords of Praag and Erengrad who held Hold the bulwark kept them.
# Scope: FROM is the head of the faith that called it; new_character the
#   defender.
character_event = {
	id = ambition.212
	hide_window = yes

	is_triggered_only = yes # on_crusade_failure

	trigger = {
		FROM = { religion_group = chaos_gods_group }
	}

	immediate = {
		new_character = {
			if = {
				limit = { has_ambition = obj_break_the_black_crusade }
				set_character_flag = amb_crusade_broken
			}
		}
		any_playable_ruler = {
			limit = {
				has_ambition = obj_hold_the_bulwark
				OR = {
					has_landed_title = c_praag
					has_landed_title = c_erengrad
				}
			}
			set_character_flag = amb_bulwark_held
		}
	}
}

# ambition.213 A murder is found out
# The close kin of the victim may swear vengeance (Avenge my kin): they are
# flagged and the killer saved as their persistent target amb_kin_killer.
# Scope: ROOT is the victim, FROM the killer.
character_event = {
	id = ambition.213
	hide_window = yes

	is_triggered_only = yes # on_assassination_disc

	trigger = {
		FROM = { is_alive = yes }
	}

	immediate = {
		any_close_relative = {
			limit = {
				ai = yes
				is_adult = yes
				NOT = { character = FROM }
				NOT = { has_character_flag = amb_kin_murdered }
			}
			set_character_flag = amb_kin_murdered
			save_persistent_event_target = { name = amb_kin_killer scope = FROM }
		}
	}
}

# ambition.215 Living with the Rot
# Every half year while Embrace rot is held: the first time Nurgle's Rot is
# found, Grandfather Nurgle visits (ambition.455); half a year and more
# later, if the Rot is still there, the ambition is done. Losing the Rot
# starts the year over.
# Scope: ROOT is the holder.
character_event = {
	id = ambition.215
	hide_window = yes

	is_triggered_only = yes # obj_embrace_rot, and itself

	trigger = { has_ambition = obj_embrace_rot }

	immediate = {
		if = {
			limit = { trait = has_nurgles_rot }
			if = {
				limit = { has_character_flag = amb_rot_caught }
				set_character_flag = amb_rot_embraced
			}
			else = {
				set_character_flag = amb_rot_caught
				character_event = { id = ambition.455 days = 7 random = 30 }
			}
		}
		else = {
			clr_character_flag = amb_rot_caught
		}
		character_event = { id = ambition.215 days = 365 }
	}
}
