###############################################################################
# Ancestor offerings
# Namespace: offering (was TOG.7000 - TOG.7054 in
#   oldgods_ancestor_worship_events.txt)
# Faction: Old World faiths
# Serves: the dawi and Norse dawi, Nehekharans, the Old Faith of Albion, the
#   Ungol and Norscan gods, Akghan and every faith that venerates its saints
#
# Once in ten years a lord goes to the shrine of the forefathers and asks for
# their aid: health, bonds, war, wealth or wisdom. The ancestors want
# something in return: the lord's own blood and flesh, a life (a prisoner, or
# a courtier when the dungeon is empty), a great feast or a small carved idol.
# The dawi never shed blood at the ancestor shrine; they fill it with gold.
# The greater the offering, the likelier the ancestors answer, and answer
# well; a mean offering may anger them. Months later the answer comes: a
# boon, a bane, or silence. Once in a life a lord who gave blood, a life or
# gold may find an idol of the forefathers on the shrine.
#
# Where the rite is held and who leads it speaks in the voice of the faith
# (WHG_old_gods_voices.txt).
#
# The offering                         offering.1 - offering.3
# The answer                           offering.10 - offering.15
#   health 10, bonds 11, war 12, wealth 13, wisdom 14; offering.15 is the
#   blessing of health driving out a sickness.
#
# Fired from: the ancestor_worship decision in minor_decisions.txt
#   (offering.1); offering.15 from blessing_of_health in
#   00_scripted_effects.txt. Shared triggers and effects are in
#   WHG_old_gods_triggers.txt and WHG_old_gods_effects.txt.
# Saved: event targets offering_prisoner, offering_courtier, offering_lover,
#   offering_friend, offering_rival, offering_commander, offering_sage,
#   offering_marshal, offering_treasurer; flags offering_aim_*,
#   offering_tier_*, offering_gold, offering_wound_* (cleared by
#   offering_clear_flags_effect), offering_boon_* (cleared by the answer),
#   offering_sage_*, old_gods_blessing, took_ancestor_idol (once per life).
###############################################################################

namespace = offering

		text = EVTDESCoffering3_life

	}

		text = EVTDESCoffering3_feast
	}
	desc = {
		trigger = { has_character_flag = offering_tier_token }
		text = EVTDESCoffering3_token
	}
	picture = GFX_evt_tribal_shrine
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes
	hide_from = yes

	option = {
		name = EVTOPTAoffering3
		clr_character_flag = do_not_disturb
		if = {
			limit = { has_character_flag = offering_wound_eye }
			add_trait = one_eyed
		}
		else_if = {
			limit = { has_character_flag = offering_wound_hand }
			add_trait = one_handed
		}
		else_if = {
			limit = { has_character_flag = offering_wound_leg }
			add_trait = one_legged
		}
		else_if = {
			limit = { has_character_flag = offering_wound_maimed }
			add_trait = maimed
		}
		if = {
			limit = {
				OR = {
					has_character_flag = offering_wound_eye
					has_character_flag = offering_wound_hand
					has_character_flag = offering_wound_leg
				}
			}
			add_trait = severely_injured
			resolve_severely_injured_effect = yes
		}
		# Once in a life, an idol of the forefathers is left on the shrine.
		if = {
			limit = {
				OR = {
					has_character_flag = offering_tier_blood
					has_character_flag = offering_tier_life
				}
				NOT = { has_character_flag = took_ancestor_idol }
				NOT = { any_artifact = { has_artifact_flag = ancestor_idol } }
			}
			random = {
				chance = 5
				set_character_flag = took_ancestor_idol
				add_artifact = ancestor_idol
			}
		}
		hidden_tooltip = { offering_answer_effect = yes }
	}
}
