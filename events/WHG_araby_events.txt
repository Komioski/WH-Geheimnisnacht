###############################################################################
# Araby events
# Namespace: holyfast (was 88500 - 88522 in the vanilla ramadan_events, never
#   copied into the mod)
# Faction: Araby & Southlands
# Serves: the faithful of Ormazd and the Kitabid
#
# The Fast of the Prophets, the vanilla Ramadan moved to Araby: a ruler of
# Ormazd or the Kitabid keeps a month of fasting by daylight before the
# Holy Flame. Three times during the month a trial of the fast comes
# (holyfast.4 - 10, 13); a ruler with a weakness may break the fast in
# secret (holyfast.30 - 34); war or illness ends it early (holyfast.11 -
# 12). At the end of the month the ruler feasts the vassals (holyfast.20 -
# 24) and gives the customary gifts to the household (holyfast.25 - 26).
# The vanilla mean-time-to-happen checks (war, illness, sins) are now
# scheduled rolls run by holyfast.2 and holyfast.3, so nothing polls.
#
# The fast begins               holyfast.1
# The rolls (hidden)            holyfast.2 - holyfast.3
# Trials of the fast            holyfast.4 - holyfast.10, holyfast.13
# The fast broken off           holyfast.11 - holyfast.12
# The fast-breaking feast       holyfast.20 - holyfast.24
# Gifts for the household       holyfast.25 - holyfast.26
# Secret sins                   holyfast.30 - holyfast.34
# The fast ends                 holyfast.40
#
# Saved: character flags holy_fast (also read by the hajj, on_hajj.txt and
#   go_on_hajj in decisions/minor_decisions.txt), holy_fast_<prisoners|
#   charity|prayers|ill|djinn|gluttonous|lustful|drunkard|greedy|cynical>,
#   attending_holy_fast_feast; character modifier observed_holy_fast.
#
# Callers:
#   decisions/WHG_araby_decisions.txt (observe_holy_fast: 1).
#
# Also in this file (each part keeps its own header and namespace):
#   Part 2: The Night of a Thousand Lamps (namespace lamps)
#   Part 3: Elephants (namespace elephant)
#   Part 4: Wonders of the Ivory Road (namespace souk)
#   Part 5: The pilgrimage to Ind (namespace ind)
#   Part 6: The Southlands, storytelling and the rains (namespace southland)
###############################################################################

namespace = holyfast

# From observe_holy_fast. Schedules the three trials, two sin checks, the
# feast, the gifts and the end of the month.

	desc = EVTDESCholyfast1
	picture = GFX_evt_imam

	is_triggered_only = yes

		name = EVTOPTAholyfast1

			character_event = { id = holyfast.2 days = 3 }
			character_event = { id = holyfast.3 days = 7 }
			character_event = { id = holyfast.2 days = 10 }
			character_event = { id = holyfast.2 days = 17 }
			character_event = { id = holyfast.3 days = 20 }
			character_event = { id = holyfast.20 days = 26 }
			character_event = { id = holyfast.25 days = 31 }
			character_event = { id = holyfast.26 days = 31 }
			character_event = { id = holyfast.40 days = 35 }
		}
	}
}

# holyfast.2 A day of the fast (hidden)
# Run three times by holyfast.1. War or illness ends the fast; otherwise
# one trial of the fast comes, if any fits.
# Scope: the fasting ruler.
character_event = {
	id = holyfast.2
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_character_flag = holy_fast
	}

	immediate = {
		if = { limit = { war = yes } character_event = { id = holyfast.11 } }
		else_if = {
			limit = {
				NOT = { has_character_flag = holy_fast_ill }
				OR = { is_ill = yes has_symptom_trigger = yes }
			}
			character_event = { id = holyfast.12 }
		}
		else = {
			random_list = {
				10 = {
					trigger = {
						holyfast_prisoners_to_free_trigger = yes
						NOT = { has_character_flag = holy_fast_prisoners }
					}
					character_event = { id = holyfast.4 }
				}
				10 = {
					trigger = {
						scaled_wealth = 0.2
						NOT = { has_character_flag = holy_fast_charity }
					}
					character_event = { id = holyfast.5 }
				}
				10 = {
					trigger = { NOT = { has_character_flag = holy_fast_prayers } }
					character_event = { id = holyfast.6 }
				}
				10 = {
					trigger = {
						NOT = { has_character_flag = holy_fast_ill }
						OR = { age = 50 is_weak_trigger = yes }
					}
					character_event = { id = holyfast.7 }
				}
				20 = {
					trigger = {
						has_character_flag = holy_fast_ill
						OR = { is_ill = yes has_symptom_trigger = yes }
						NOT = { trait = infirm }
					}
					character_event = { id = holyfast.8 }
				}
				4 = {
					trigger = { NOR = { trait = gluttonous trait = temperate } }
					character_event = { id = holyfast.9 }
				}
				4 = {
					trigger = { NOR = { trait = greedy trait = charitable } }
					character_event = { id = holyfast.10 }
				}
				2 = {
					trigger = { NOT = { has_character_flag = holy_fast_djinn } }
					character_event = { id = holyfast.13 }
				}
				10 = { } # a quiet day
			}
		}
	}
}

# holyfast.3 A weakness of the flesh (hidden)
# Run twice by holyfast.1. A ruler with a sinful trait may break the fast
# in secret, at most once per sin each month.
# Scope: the fasting ruler.
character_event = {
	id = holyfast.3
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_character_flag = holy_fast
		war = no
		prisoner = no
		is_incapable = no
	}

	immediate = {
		random_list = {
			10 = {
				trigger = { trait = gluttonous NOT = { has_character_flag = holy_fast_gluttonous } }
				character_event = { id = holyfast.30 }
			}
			10 = {
				trigger = {
					OR = { trait = lustful trait = hedonist }
					NOT = { has_character_flag = holy_fast_lustful }
				}
				character_event = { id = holyfast.31 }
			}
			10 = {
				trigger = { trait = drunkard NOT = { has_character_flag = holy_fast_drunkard } }
				character_event = { id = holyfast.32 }
			}
			10 = {
				trigger = {
					trait = greedy
					NOT = { trait = just }
					NOT = { has_character_flag = holy_fast_greedy }
				}
				character_event = { id = holyfast.33 }
			}
			10 = {
				trigger = { trait = cynical NOT = { has_character_flag = holy_fast_cynical } }
				character_event = { id = holyfast.34 }
			}
			15 = { } # the flesh holds firm
		}
	}
}

# roi_various_events, never copied into the mod), moved to Araby. Once per
# game a Sultan of Araby honours a human ruler with a white elephant, a
# gift no one can refuse and few can afford: it must be stabled like a
# prince, its dung is sold as a cure in the markets, it wrecks its stable
# and runs loose through the streets, and if it dies at the ruler's hand
# the Sultan must never find out. And a ruler of Araby or the Southlands
# may have a war elephant run amok in the capital, once per lifetime.
#   elephant.1      The Sultan's gift is chosen (hidden)
#   elephant.2-7    The white elephant
#   elephant.8-10   Getting rid of the white elephant
#   elephant.20-24  The elephant runs amok
# Saved: global flag white_elephant_given; character flags
#   gave_white_elephant (the giver), white_elephant_fine_stable,
#   elephant_amok_seen (once per lifetime); event target elephant_giver.
# Callers: wh_on_actions on_five_year_pulse elephants list (1, 20).
###############################################################################

namespace = elephant

# elephant.1 The Sultan's gift (hidden)
# Scope: a human player ruler of duke rank or above. Fired by the
# elephants five-year list.
character_event = {
	id = elephant.1
	hide_window = yes

	is_triggered_only = yes

	only_playable = yes
	ai = no
	only_capable = yes
	prisoner = no

	trigger = {
		NOT = { has_global_flag = white_elephant_given }
		higher_tier_than = COUNT
		trait = creature_human
		any_playable_ruler = { elephant_giver_trigger = yes }
	}

	immediate = {
		set_global_flag = white_elephant_given
		random_playable_ruler = {
			limit = { elephant_giver_trigger = yes }
			set_character_flag = gave_white_elephant
			save_event_target_as = elephant_giver
		}
		character_event = { id = elephant.2 }
	}
}

# elephant.2 A gift from the Sultan
# Scope: the ruler; event target elephant_giver is the Sultan.

	id = elephant.2
	desc = EVTDESCelephant2

	option = { # Accept with thanks
		name = EVTOPTAelephant2
		add_friend = event_target:elephant_giver

	}
	option = { # Refuse it
		name = EVTOPTBelephant2
		elephant_offend_giver_effect = yes
	}
}

# Part 5: The pilgrimage to Ind (namespace ind)
# Faction: Araby & the Southlands
# Serves: human rulers of Ormazd and of the gods of Cathay
#
# The vanilla Indian pilgrimage (MNM.8000 - MNM.8093 in
# indian_pilgrim_events, never copied into the mod), moved off the map to
# the Kingdom of Ind, the land of a thousand gods at the end of the Spice
# Route. The pilgrim chooses a holy place (the Temple of the Thousand Arms,
# the Tiger Shrine in the jungle, or the burning ghats of the Sacred River),
# meets two troubles on the road (monsoon, drought, jungle, a bandit rajah,
# a man-eating tiger, the poor, fellow travellers, fever, a splendid city,
# news from home), finds what the holy place holds for him, and comes home
# after about nine months. Once in twenty years. The rare rewards, once per
# lifetime each: a herd of war elephants from a rajah, or the holy place's
# own gift (an artifact or a blessing).
#   ind.1       Choosing the holy place
#   ind.10-19   The road
#   ind.30      Arrival (narrative)
#   ind.40-48   At the holy place
#   ind.49      A rajah's gift (once per lifetime)
#   ind.51-53   The holy place's gift (once per lifetime): the Scripture of
#               the Thousand Arms, the Mantle of the Tiger Shrine, or the
#               blessing of the Sacred River
#   ind.50      Home again
# Saved: character flags ind_pilgrimage_done (twenty years),
#   ind_dest_<temple|tiger|ghats>, ind_good_pilgrimage,
#   ind_elephants_given and ind_gift_given (once per lifetime each),
#   do_not_disturb; character modifier blessed_by_the_sacred_river;
#   artifacts ind_thousand_arms_scripture, ind_tiger_mantle
#   (common/artifacts/36_artifacts_ind.txt); traits
#   on_indian_pilgrimage (while away), indian_pilgrim (after a good
#   pilgrimage); unit earmark ind_war_elephants.
# Callers: decisions/WHG_araby_decisions.txt (go_on_ind_pilgrimage: 1).
# Support: ind_pilgrimage_faith_trigger (wh_araby_triggers),
#   ind_set_out_effect, ind_clear_flags_effect (wh_araby_effects).
###############################################################################

namespace = ind
