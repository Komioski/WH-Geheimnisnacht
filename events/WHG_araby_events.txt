###############################################################################
# Araby events
# Namespace: holyfast (was 88500 - 88522 in the vanilla ramadan_events, never
#   copied into the mod)
# Faction: Araby & Southlands
# Serves: the faithful of Ormazd and the Kitabid
#
# The Fast of the Prophets, the vanilla Ramadan moved to Araby: a ruler of
# Ormazd or the Kitabid keeps a month of fasting by daylight before the
# Holy Flame. Three times during the month a trial of the fast comes
# (holyfast.4 - 10, 13); a ruler with a weakness may break the fast in
# secret (holyfast.30 - 34); war or illness ends it early (holyfast.11 -
# 12). At the end of the month the ruler feasts the vassals (holyfast.20 -
# 24) and gives the customary gifts to the household (holyfast.25 - 26).
# The vanilla mean-time-to-happen checks (war, illness, sins) are now
# scheduled rolls run by holyfast.2 and holyfast.3, so nothing polls.
#
# The fast begins               holyfast.1
# The rolls (hidden)            holyfast.2 - holyfast.3
# Trials of the fast            holyfast.4 - holyfast.10, holyfast.13
# The fast broken off           holyfast.11 - holyfast.12
# The fast-breaking feast       holyfast.20 - holyfast.24
# Gifts for the household       holyfast.25 - holyfast.26
# Secret sins                   holyfast.30 - holyfast.34
# The fast ends                 holyfast.40
#
# Saved: character flags holy_fast (also read by the hajj, on_hajj.txt and
#   go_on_hajj in decisions/minor_decisions.txt), holy_fast_<prisoners|
#   charity|prayers|ill|djinn|gluttonous|lustful|drunkard|greedy|cynical>,
#   attending_holy_fast_feast; character modifier observed_holy_fast.
#
# Callers:
#   decisions/WHG_araby_decisions.txt (observe_holy_fast: 1).
#
# Also in this file (each part keeps its own header and namespace):
#   Part 2: The Night of a Thousand Lamps (namespace lamps)
#   Part 3: Elephants (namespace elephant)
#   Part 4: Wonders of the Ivory Road (namespace souk)
#   Part 5: The pilgrimage to Ind (namespace ind)
#   Part 6: The Southlands, storytelling and the rains (namespace southland)
#   Part 7: The schools of time (namespace sect)
#   Part 8: The pilgrimage to Martek (namespace hajj)
#   Part 9: The holy dogs (namespace holypet)
#   Part 10: The judgement of the sun (namespace sunjudge)
#   Part 11: The Califate of Ormazd (namespace califate)
#   Part 12: Polo of the horse lords (namespace polo)
#   Part 13: The Parade of the Idols (namespace idols)
###############################################################################

namespace = holyfast

# From observe_holy_fast. Schedules the three trials, two sin checks, the
# feast, the gifts and the end of the month.

	title = EVTTITLEholyfast
	desc = EVTDESCholyfast1
	picture = GFX_evt_imam
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPTAholyfast1
		hidden_tooltip = {

			character_event = { id = holyfast.3 days = 7 }
			character_event = { id = holyfast.2 days = 10 }
			character_event = { id = holyfast.2 days = 17 }
			character_event = { id = holyfast.3 days = 20 }
			character_event = { id = holyfast.20 days = 26 }
			character_event = { id = holyfast.25 days = 31 }
			character_event = { id = holyfast.26 days = 31 }
			character_event = { id = holyfast.40 days = 35 }
		}
	}
}

# holyfast.2 A day of the fast (hidden)
# Run three times by holyfast.1. War or illness ends the fast; otherwise
# one trial of the fast comes, if any fits.
# Scope: the fasting ruler.
character_event = {
	id = holyfast.2
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_character_flag = holy_fast
	}

	immediate = {
		if = { limit = { war = yes } character_event = { id = holyfast.11 } }
		else_if = {
			limit = {
				NOT = { has_character_flag = holy_fast_ill }
				OR = { is_ill = yes has_symptom_trigger = yes }
			}
			character_event = { id = holyfast.12 }
		}
		else = {
			random_list = {
				10 = {
					trigger = {
						holyfast_prisoners_to_free_trigger = yes
						NOT = { has_character_flag = holy_fast_prisoners }
					}
					character_event = { id = holyfast.4 }
				}
				10 = {
					trigger = {
						scaled_wealth = 0.2
						NOT = { has_character_flag = holy_fast_charity }
					}
					character_event = { id = holyfast.5 }
				}
				10 = {
					trigger = { NOT = { has_character_flag = holy_fast_prayers } }
					character_event = { id = holyfast.6 }
				}
				10 = {
					trigger = {
						NOT = { has_character_flag = holy_fast_ill }
						OR = { age = 50 is_weak_trigger = yes }
					}
					character_event = { id = holyfast.7 }
				}
				20 = {
					trigger = {
						has_character_flag = holy_fast_ill
						OR = { is_ill = yes has_symptom_trigger = yes }
						NOT = { trait = infirm }
					}
					character_event = { id = holyfast.8 }
				}
				4 = {
					trigger = { NOR = { trait = gluttonous trait = temperate } }
					character_event = { id = holyfast.9 }
				}
				4 = {
					trigger = { NOR = { trait = greedy trait = charitable } }
					character_event = { id = holyfast.10 }
				}
				2 = {
					trigger = { NOT = { has_character_flag = holy_fast_djinn } }
					character_event = { id = holyfast.13 }
				}
				10 = { } # a quiet day
			}
		}
	}
}

# holyfast.3 A weakness of the flesh (hidden)
# Run twice by holyfast.1. A ruler with a sinful trait may break the fast
# in secret, at most once per sin each month.
# Scope: the fasting ruler.
character_event = {
	id = holyfast.3
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_character_flag = holy_fast
		war = no
		prisoner = no
		is_incapable = no
	}

	immediate = {
		random_list = {
			10 = {
				trigger = { trait = gluttonous NOT = { has_character_flag = holy_fast_gluttonous } }
				character_event = { id = holyfast.30 }
			}
			10 = {
				trigger = {
					OR = { trait = lustful trait = hedonist }
					NOT = { has_character_flag = holy_fast_lustful }
				}
				character_event = { id = holyfast.31 }
			}
			10 = {
				trigger = { trait = drunkard NOT = { has_character_flag = holy_fast_drunkard } }
				character_event = { id = holyfast.32 }
			}
			10 = {
				trigger = {
					trait = greedy
					NOT = { trait = just }
					NOT = { has_character_flag = holy_fast_greedy }
				}
				character_event = { id = holyfast.33 }
			}
			10 = {
				trigger = { trait = cynical NOT = { has_character_flag = holy_fast_cynical } }
				character_event = { id = holyfast.34 }
			}
			15 = { } # the flesh holds firm
		}
	}
}

# Part 2: The Night of a Thousand Lamps (namespace lamps)
# Faction: Araby & Southlands
# Serves: the faithful of Ormazd and the Kitabid
#
# The vanilla Diwali feast (RoI.1000 - RoI.1999 in roi_feast_events, never
# copied into the mod), moved to the fire-temples of Araby: in autumn a
# ruler of Ormazd lights a thousand lamps for the Holy Flame and feasts the
# vassals for a month (hold_thousand_lamps_festival). Gifts go to the
# spouse or a lover, and the souk sends its entertainers: a snake charmer,
# a conjurer with a flying carpet, a troupe of monkeys from the Southlands,
# and sometimes a dancer of uncanny beauty who is gone by morning. A djinn?
# The vanilla Kali Puja sacrifices are not carried over.
#   lamps.1-3     Invitations
#   lamps.5-6     Gifts for a spouse or lover, and how they are taken
#   lamps.10-11   The feast begins; a guest arrives
#   lamps.15      The lighting of the lamps at the fire-temple
#   lamps.20-23   The feast ends, remembered or regretted
#   lamps.30-36   The snake charmer
#   lamps.40-42   The dancer who vanishes
#   lamps.50-54   The conjurer
#   lamps.60      The monkeys
#
# Saved: character flags lamps_begun, attending_lamps, lamps_bad (a feast
#   to forget), lamps_<fine|small|no>_gift_to_<spouse|lover>,
#   do_not_disturb (cleared at the end); character modifier
#   holding_thousand_lamps.
# Callers: decisions/WHG_araby_decisions.txt (hold_thousand_lamps_festival: 1).
###############################################################################

namespace = lamps

# lamps.6 The gift is taken
# Scope: the giver.
character_event = {
	id = lamps.6
	desc = {
		trigger = { has_character_flag = lamps_fine_gift_to_lover }
		text = EVTDESClamps6_lover
	}
	desc = {
		trigger = {
			NOT = { has_character_flag = lamps_fine_gift_to_lover }
			has_character_flag = lamps_fine_gift_to_spouse
		}
		text = EVTDESClamps6_fine
	}
	desc = {
		trigger = { has_character_flag = lamps_small_gift_to_spouse }
		text = EVTDESClamps6_small
	}
	desc = {
		trigger = {
			NOT = { has_character_flag = lamps_fine_gift_to_lover }
			has_character_flag = lamps_no_gift_to_spouse
		}
		text = EVTDESClamps6_none
	}
	picture = GFX_evt_feast
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPTAlamps6
		if = {
			limit = { has_character_flag = lamps_fine_gift_to_spouse }
			spouse = {
				opinion = { who = ROOT modifier = opinion_grateful years = 1 }
				random = { chance = 30 if = { limit = { is_no_females_race = no ROOT = { is_no_females_race = no } } add_lover = ROOT } }
			}
		}
		else_if = {
			limit = { has_character_flag = lamps_small_gift_to_spouse }
			spouse = { opinion = { who = ROOT modifier = opinion_happy years = 1 } }
		}
		if = {
			limit = { has_character_flag = lamps_fine_gift_to_lover }
			random_lover = {
				limit = { NOT = { is_married = ROOT } }
				opinion = { who = ROOT modifier = opinion_grateful years = 1 }
			}
			spouse = { opinion = { who = ROOT modifier = opinion_unhappy years = 2 } }
		}
		else_if = {
			limit = { has_character_flag = lamps_no_gift_to_spouse }
			spouse = { opinion = { who = ROOT modifier = spouse_ignores_me years = 1 } }
		}
		clr_character_flag = lamps_fine_gift_to_spouse
		clr_character_flag = lamps_small_gift_to_spouse
		clr_character_flag = lamps_no_gift_to_spouse
		clr_character_flag = lamps_fine_gift_to_lover
	}
}

# Serves: the faithful of the southern faiths (Ormazd, the Kitabids, the
#   Atruhayid and the Akghan)
#
# The Scriptures say that the djinn of the dark and the servants of the
# Great Conspirator prowl around every house at night. A pious southern
# ruler takes in one of the holy dogs that keep them away: a masterless
# street dog that guards the capital or the house, a hedgehog (the
# prickly-backed dog) or an otter (the dog of the water). Each lives some
# years and is mourned.
#

# Keeping the capital's dog in place   holypet.6 - holypet.7
# The pet dies                         holypet.8 - holypet.10
#

#   on_new_holder: 7; on_character_convert_religion: 6).
# Saved: character modifiers pet_masterless_dog, pet_house_dog,
#   pet_hedgehog, pet_otter; province modifier masterless_dog on the
#   capital.
# Dropped: ZP.5, an on_startup pass over every character that turned
#   pre-2.8 pet traits into modifiers; no save of this mod has them.
###############################################################################

# holypet.7 The capital's dog follows the capital
# When a ruler gains a title, the masterless dog's blessing moves to the
# ruler's capital, or leaves a province whose ruler keeps no such dog.
# Scope: ROOT is the new holder.

	id = holypet.7
	hide_window = yes

	is_triggered_only = yes # on_new_holder

	only_rulers = yes

	immediate = {
		any_demesne_province = {
			if = {
				limit = {
					has_province_modifier = masterless_dog
					ROOT = {
						OR = {
							NOT = { has_character_modifier = pet_masterless_dog }
							NOT = { capital_scope = { province = PREVPREV } }
						}
					}
				}
				remove_province_modifier = masterless_dog
			}
			if = {
				limit = {
					NOT = { has_province_modifier = masterless_dog }
					ROOT = {
						has_character_modifier = pet_masterless_dog

# Scope: ROOT is a participant (idols_rites_effect).
character_event = {
	id = idols.20
	title = EVTTITLE_DR_atruhayid_1530
	desc = EVTDESC_DR_atruhayid_1530
	picture = GFX_evt_questing_araby
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # idols_rites_effect

	capable_only = yes

	option = { # Bear an idol
		name = EVTOPTA_DR_atruhayid_1530
		trigger = { piety = 300 }
		piety = 30
		hidden_tooltip = {
			random_list = {
				10 = { character_event = { id = idols.21 } } # Drops the idol
				10 = { character_event = { id = idols.22 } } # Possessed
				10 = { character_event = { id = idols.23 } } # Zealous
				10 = { character_event = { id = idols.24 } } # Stronger
				280 = { }
			}
		}
	}
	option = { # Watch from the shade
		name = EVTOPTB_DR_atruhayid_1530
		piety = 10
	}
}

# idols.21 The idol falls
# An old, weak or sick bearer drops the idol and is crushed beneath it: lost
# prestige and piety, and a chance the strain leaves them infirm (health.25).
# Scope: ROOT is the bearer (idols.20).
character_event = {
	id = idols.21
	desc = EVTDESC_DR_atruhayid_1531
	picture = GFX_evt_questing_araby
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # idols.20

	trigger = {
		OR = {
			age = 50
			trait = weak
			trait = infirm
			is_ill_trigger = yes
		}
	}

	option = {
		name = EVTOPTA_DR_atruhayid_1531
		prestige = -100
		piety = -100
		hidden_tooltip = {
			random = {
				chance = 20
				character_event = { id = health.25 }
			}
		}
	}
}

# idols.26 Offerings from the pious
# Give a quarter of the purse (100 piety, may become charitable), a little
# (10 piety, a charitable soul may lose it), or nothing (lose 50 piety, may
# turn greedy).
# Scope: ROOT is a participant (idols_rites_effect).
character_event = {
	id = idols.26
	title = EVTTITLE_DR_atruhayid_1550
	desc = EVTDESC_DR_atruhayid_1550
	picture = GFX_evt_questing_araby
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # idols_rites_effect

	capable_only = yes

	option = { # Give a lot
		name = EVTOPTA_DR_atruhayid_1550
		trigger = { NOT = { trait = greedy } }
		piety = 100
		scaled_wealth = -0.25
		random = {
			chance = 50
			add_trait = charitable
		}
		ai_chance = {
			factor = 1
			modifier = { factor = 2 trait = zealous }
			modifier = { factor = 2 trait = charitable }
		}
	}
	option = { # Give some
		name = EVTOPTB_DR_atruhayid_1550
		piety = 10
		scaled_wealth = -0.05
		if = {
			limit = { trait = charitable }
			random = {
				chance = 25
				remove_trait = charitable
			}
		}
		ai_chance = { factor = 2 }
	}
	option = { # Give nothing
		name = EVTOPTC_DR_atruhayid_1550
		trigger = { NOT = { trait = charitable } }
		piety = -50
		random = {
			chance = 25
			add_trait = greedy
		}
		ai_chance = {
			factor = 1
			modifier = { factor = 3 trait = greedy }
			modifier = { factor = 2 trait = cynical }
		}
	}
}

# idols.30 The parade ends
# Hidden. The host is told (idols.31), and every vassal who came goes home
# pleased (idols.32).
# Scope: ROOT is the host (idols.10).
character_event = {
	id = idols.30
	hide_window = yes

	is_triggered_only = yes # idols.10

	immediate = {
		character_event = { id = idols.31 }
		any_vassal = {
			limit = { has_character_flag = attending_atruhayid_feast }
			character_event = { id = idols.32 }
		}
	}
}
