###############################################################################
# The armoury
# Namespace: armoury (was ARMS.1 - ARMS.9138 in wh_armoury_events.txt)
# Faction: Generic
# Serves: every ruler who employs a smith (decision employ_a_smith)
#
# A ruler sends for a smith. A weaponsmith or armoursmith comes to court,
# hears what the ruler wants (a sword, a scimitar, a lance, an axe, a mace,
# a bow, a suit of armour, or among greenskins a choppa, an 'ard axe, a
# club, a stabba or 'ard armour) and names a price. The work may be done,
# run into trouble with the materials (and get better for more gold), or
# be stolen; while the smith works he may teach, train a ward, take a lover
# or hurt himself. The greater lords of the Asur and the Druchii may ask
# for their people's great armours, dwarf lords for a runesmith's gromril
# or runes, and every ruler but the beasts for a goldsmith's crown jewels.
# Ogres, greenskins and skaven pick through the battlefields instead. After
# a won battle a commander may bring home an enemy banner.
#
# The weaponsmith and armoursmith      armoury.1 - armoury.24
# Salvage from the battlefields        armoury.30
# The runesmith                        armoury.40 - armoury.44
# The goldsmith                        armoury.50 - armoury.62
# In the smithy                        armoury.70 - armoury.75
# Banners of the fallen                armoury.80
#
# Callers: the decision employ_a_smith in decisions/mnm_society_decisions.txt
#   (armoury.1), craft_master_rune in decisions/wh_dawi_decisions.txt
#   (armoury.44), on_battle_won in common/on_actions/WHG_war_on_actions.txt
#   (armoury.80). The master of the realm's Armoury is
#   events/WHG_wonder_events.txt, wonder.130. Effects in
#   common/scripted_effects/WHG_war_effects.txt, triggers in
#   common/scripted_triggers/WHG_war_triggers.txt; the smiths' artifacts
#   and their give_ effects and have_ triggers in
#   common/scripted_effects/WHG_legend_effects.txt,
#   common/scripted_triggers/WHG_legend_triggers.txt and
#   common/scripted_triggers/WHG_war_triggers.txt.
# Saved: event targets smith, smith_employer, my_helpful_marshal,
#   neighbor_province, the_armory, the_wonder, smith_steal_target,
#   smith_event_target, sex_partner, combatant_1, combatant_2; flags
#   weaponsmith, armorsmith, gromrilsmith, runecraft, the order tiers
#   amazing_ / expensive_ / good_ / cheap_ weapons and jewels, the outcomes
#   smith_ and goldsmith_ success / fail / steal and smith_duel, the weapon
#   flags weapon_*, armor_set, ard_armour, the elven armour flags, the rune
#   flags malice_rune / honor_rune / might_rune, preparing_crafting,
#   got_terrible_sword, got_terrible_armor, terrible_sword_coming,
#   terrible_armor_coming, battle_banner_found and smith_event_armoury72
#   - 75; artifact flag stolen_by_smith.
# Dropped: the capital ransack chain ARMS.9899 - 9911 (no caller left; the
#   sack of a capital is artifact.100 - 104 in WHG_artifact_events.txt) and
#   the second goldsmith exit ARMS.9110, the same as ARMS.9127 (armoury.12).
###############################################################################

# armoury.2 A smith is found
# The marshal (or the boyz) found a smith nearby. Invite him, send him away
# (and let the court's own smiths try, once in a lifetime for a sword and
# once for armour: armoury.3), or have the realm's Armoury do the work
# (wonder.130).
# Scope: ROOT is the ruler.
character_event = {
	id = armoury.2
	desc = {
		trigger = {
			OR = { trait = creature_orc trait = creature_goblin trait = creature_hobgoblin }
		}
		text = EVTDESCarmoury2A
	}
	desc = {
		trigger = {
			job_marshal = { always = yes }
			NOR = { trait = creature_orc trait = creature_goblin trait = creature_hobgoblin }
		}
		text = EVTDESCarmoury2B
	}
	desc = {
		trigger = {
			NOT = { job_marshal = { always = yes } }
			NOR = { trait = creature_orc trait = creature_goblin trait = creature_hobgoblin }
		}
		text = EVTDESCarmoury2C
	}
	picture = {
		trigger = {
			OR = { trait = creature_orc trait = creature_goblin trait = creature_hobgoblin }
		}
		picture = GFX_evt_greenskinhuts
	}
	picture = GFX_evt_weaponsmith
	border = GFX_event_normal_frame_economy

	hide_from = yes
	is_triggered_only = yes # armoury.1

	trigger = {
		OR = { has_dlc = "Mystics" has_dlc = "Jade Dragon" }
	}

	immediate = {
		hidden_effect = {
			job_marshal = { save_event_target_as = my_helpful_marshal }
			armoury_create_smith_effect = yes
			event_target:smith = {
				if = {
					limit = { ROOT = { has_character_flag = armorsmith } }
					give_nickname = nick_the_armorsmith
				}
				else = { give_nickname = nick_the_weaponsmith }
				if = {
					limit = { ROOT = { government = greenskin_government } }
					random_list = {
						25 = { culture = goblin_main }
						25 = { culture = orc_main }
						25 = { culture = goblin_night }
						25 = { culture = ROOT }
					}
				}
				random_list = { # a weaponsmith's second chance at the craft
					15 = { change_stewardship = 1 }
					25 = { change_stewardship = 3 }
					20 = { artisan_craftsmen_bloodline_increase_score = yes change_stewardship = 5 }
					10 = { artisan_craftsmen_bloodline_increase_score = yes change_stewardship = 7 }
				}
				armoury_smith_talent_effect = yes
			}
		}
		# The province named in the text
		random_province = {
			limit = { owner = { is_within_diplo_range = ROOT } }
			save_event_target_as = neighbor_province
		}
		random_demesne_province = {
			limit = { any_neighbor_province = { NOT = { owner = { character = ROOT } } } }
			random_neighbor_province = {
				limit = { NOT = { owner = { character = ROOT } } }
				save_event_target_as = neighbor_province
			}
		}
		# The realm's Armoury
		random_realm_wonder_upgrade = {
			limit = { has_wonder_upgrade = upgrade_armory is_active = yes }
			save_event_target_as = the_armory
			wonder = { save_event_target_as = the_wonder }
		}
	}

	option = { # Invite them
		name = {
			trigger = { NOR = { culture_group = orc_group culture_group = goblin_group } }
			text = EVTOPTA_MNM_9115
		}
		name = {
			trigger = { OR = { culture_group = orc_group culture_group = goblin_group } }
			text = EVTOPTA_ARMS_10
		}
		custom_tooltip = { text = EVTOPTA_MNM_9115_TT }
		if = {
			limit = { event_target:my_helpful_marshal = { always = yes } }
			event_target:my_helpful_marshal = { show_portrait = yes }
			reverse_opinion = {
				who = event_target:my_helpful_marshal
				modifier = opinion_was_praised
			}
		}
		hidden_effect = {
			save_event_target_as = smith_employer
			event_target:smith = { character_event = { id = armoury.4 days = 7 } }
		}
	}

	option = { # Send them away
		name = EVTOPTB_MNM_9115
		custom_tooltip = { text = EVTOPTB_MNM_9115_TT }
		hidden_effect = { event_target:smith = { leave_court_and_die_effect = yes } }
		if = {
			limit = {
				has_character_flag = weaponsmith
				NOR = { has_artifact = terrible_sword has_character_flag = got_terrible_sword }
			}
			set_character_flag = got_terrible_sword
			set_character_flag = terrible_sword_coming
			character_event = { id = armoury.3 days = 7 }
		}
		else_if = {
			limit = {
				has_character_flag = armorsmith
				NOR = { has_artifact = terrible_armor has_character_flag = got_terrible_armor }
			}
			set_character_flag = got_terrible_armor
			set_character_flag = terrible_armor_coming
			character_event = { id = armoury.3 days = 7 }
		}
		clr_character_flag = weaponsmith
		clr_character_flag = armorsmith
		ai_chance = { factor = 0 }
	}

	option = { # The Armoury will do it
		name = EVTOPTBarmoury2
		trigger = { # greenskins use their own smiths
			event_target:the_armory = { always = yes }
			NOR = { culture_group = orc_group culture_group = goblin_group }
		}
		tooltip_info_custom = EVTOPTC_MNM_9115_TT
		custom_tooltip = { text = EVTOPTB_MNM_9115_TT }
		hidden_effect = {
			event_target:smith = { leave_court_and_die_effect = yes }
			# The Armoury's own master
			armoury_create_smith_effect = yes
			event_target:smith = {
				if = {
					limit = { ROOT = { has_character_flag = armorsmith } }
					give_nickname = nick_the_armorsmith
				}
				else = { give_nickname = nick_the_weaponsmith }
			}
		}
		character_event = { id = wonder.130 days = 7 } # the armoury's master smith (WHG_wonder_events)
		ai_chance = { factor = 100 }
	}
}

# armoury.3 The court's own smiths have a go
# The smith was sent away, so the court's own smiths try. Once in a
# lifetime for a sword and once for armour.
# Scope: ROOT is the ruler.
character_event = {
	id = armoury.3
	desc = { trigger = { has_character_flag = terrible_sword_coming } text = MNM_9999_DESC }
	desc = { trigger = { has_character_flag = terrible_armor_coming } text = MNM_9999_DESC_armor }
	picture = GFX_evt_weaponsmith
	border = GFX_event_normal_frame_economy

	hide_from = yes
	hide_new = yes
	is_triggered_only = yes # armoury.2

	trigger = {
		OR = {
			has_character_flag = terrible_sword_coming
			has_character_flag = terrible_armor_coming
		}
	}

	option = {
		name = EVTOPTAarmoury3
		if = {
			limit = { has_character_flag = terrible_sword_coming }
			add_artifact = terrible_sword
		}
		else = { add_artifact = terrible_armor }
		clr_character_flag = terrible_sword_coming
		clr_character_flag = terrible_armor_coming
	}
}

# armoury.4 The smith comes to court
# Hidden. A week after the invitation the smith is at court: a runesmith
# hears what is to be made of gromril (armoury.41), any other what kind of
# weapon (armoury.5).
# Scope: ROOT is the smith, FROM the employer.
character_event = {
	id = armoury.4
	hide_window = yes
	is_triggered_only = yes # armoury.2, armoury.40

	immediate = {
		if = {
			limit = { FROM = { has_character_flag = gromrilsmith } }
			event_target:smith_employer = { character_event = { id = armoury.41 } }
		}
		else = { event_target:smith_employer = { character_event = { id = armoury.5 } } }
	}
}

# armoury.80 A banner from the field
# After a won battle a commander brings home a fine banner taken from the
# enemy dead. Not once in a lifetime: a commander may win several over a
# long career, but at most one every ten years and never while holding three
# banners. Never for the Tomb Kings or daemons. The just and patient are likelier to
# be brought it, the deceitful less.
# Scope: ROOT is a commander on the winning side.
character_event = {
	id = armoury.80
	desc = EVTDESCarmoury80
	picture = GFX_evt_battle
	border = GFX_event_normal_frame_war

	hide_from = yes
	is_triggered_only = yes # on_battle_won

	trigger = {
		is_alive = yes
		OR = { # One banner every ten years at most
			NOT = { has_character_flag = battle_banner_found }
			had_character_flag = { flag = battle_banner_found years = 10 }
		}
		NOT = { trait = creature_nehekharan }
		NOT = { culture_group = daemonic_group }
		NOT = { any_artifact = { count = 3 has_artifact_flag = banner } }
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 1.5 trait = patient }
		modifier = { factor = 1.5 trait = just }
		modifier = { factor = 1.5 trait = diligent }
		modifier = { factor = 0.2 trait = deceitful }
		modifier = { factor = 0.2 trait = arbitrary }
		modifier = { factor = 2 has_character_modifier = good_battle_omen }
		modifier = { factor = 0.9 has_character_modifier = bad_battle_omen }
	}

	option = {
		name = EXCELLENT
		clr_character_flag = battle_banner_found # restart the ten years
		set_character_flag = battle_banner_found
		random_list = {
			10 = { add_artifact = rampagers_standard }
			10 = { add_artifact = rangers_standard }
			10 = { add_artifact = razor_standard }
			10 = { add_artifact = wailing_banner }
			10 = { add_artifact = banner_of_swiftness }
			10 = { add_artifact = lichebone_pennant }
			10 = { add_artifact = standard_of_discipline }
			10 = { add_artifact = banner_of_eternal_flame }
			10 = { add_artifact = scarecrow_banner }
			10 = { add_artifact = gleaming_pennant }
			3 = { trigger = { religion_group = chaos_gods_group } add_artifact = ruinous_standard }
			10 = {
				trigger = { trait = creature_beastman }
				add_artifact = beastmen_artifact_beast_banner
			}
		}
	}

	after = {
		remove_scrying_omen_modifiers_effect = yes
	}
}
