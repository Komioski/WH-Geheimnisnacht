###############################################################################
# Lost treasures
# Namespace: artifact (was MNM.9000 - 9035, MNM.9835 - 9841 and
#   MNM.9899 - 9911 in mnm_artifacts_events.txt)
# Faction: Generic
# Serves: every ruler and his council: the relic hunt and the treasures
#   taken when a capital is sacked
#
# A councillor brings word of a lost treasure: a common rumor of a hero's
# remains, a rare one of a treasure of the ruler's faith, a great one of a
# legend. The ruler sends a councillor after it, pays for the search out of
# his taxes and is asked for diplomats, soldiers, stewards, spies, scholars
# or gold along the way. The hunter's skill decides how fast he follows the
# trail, and he may steal, blunder, find a better rumor or die on the road.
# What he brings home belongs to the ruler's faith: Sigmar's swords, the
# Lady's chalices, Clan Skryre's engines, the crowns of Nehekhara, the
# daemon blades of the Dark Gods (artifact_hunt_treasure_effect). A
# treasure held by anyone alive is never found twice; when nothing is left,
# the trail ends at a hoard of gold. When a capital falls, the victor may
# carry off a treasure of its lord or his court, or burn a forgery of one
# of his own.
#
# The relic hunt: the rumor           artifact.1 - artifact.3
# The relic hunt: the search          artifact.10 - artifact.31
# Treasures of the sack               artifact.100 - artifact.104
#
# Callers: the yearly pulse (artifact.1), the decision search_for_artifact
#   in decisions/mnm_society_decisions.txt (artifact.10), on_death,
#   on_become_imprisoned_any_reason and the councillor jobs in
#   common/job_titles/00_job_titles.txt (artifact.13), on_loot_settlement
#   (artifact.100) and on_siege_over_winner (artifact.101). Triggers in
#   common/scripted_triggers/WHG_artifact_triggers.txt, effects in
#   common/scripted_effects/WHG_artifact_effects.txt, the hero's title of
#   the relics in GetArtifactSource and GetArtifactSourceCap
#   (localisation/customizable_localisation/00_customizable_localisation.txt).
# Saved: event targets rumor_councillor, relic_ruler, relic_hunter,
#   spy_target, spy_title, looter, target_location, target_title,
#   original_item_owner; variables relic_score and artifact_raid_quality;
#   flags relic_hunter,
#   they_asked_for_your_help, artifact_rumor_heard, artifact_hunt_gold,
#   artifact_hunt_lost_shown, looted_by_raiding_holding,
#   artifact_raid_looting, artifact_raid_heathen_relic,
#   looted_artifact_destroy_duplicate and the artifact flags
#   looted_artifact_tooltip_1 and looted_artifact_tooltip_2.
# Dropped: the 47 return events for the relics of the real world (the
#   bones of Saint Peter, the Seal of Solomon, the Sampo and the rest, none
#   of which exist in this mod) and the find tombola that sent them, the
#   Seal of Solomon and Blessed Mantle pulses (MNM.9815, 9816), the smith
#   chains MNM.9100 - 9138 (dead copies of ARMS.9100 - 9138 in
#   wh_armoury_events.txt), the hidden "nothing found" branch that kept the
#   AI from rumors (now a lower weight), and the seven return events for
#   the hero's relics, merged into artifact.11. The four pings for a lost
#   hunter (MNM.9030, 9033, 9034) are artifact.13, the progress letters
#   (MNM.9031, 9035) artifact.30 and the five quality tiers of the sack
#   (MNM.9902 - 9911) artifact.103 and artifact.104.
###############################################################################

# artifact.1 A rumor of treasure
# Hidden. Once in a ruler's life a trusted councillor hears of a lost
# treasure: a great rumor (one in ten), a rare one (three in ten) or a
# common one. Counts and barons only hear of it when played.
# Scope: a ruler.
character_event = {
	id = artifact.1
	hide_window = yes
	is_triggered_only = yes

	only_playable = yes
	only_capable = yes
	prisoner = no
	min_age = 16

	trigger = {
		has_dlc = "Mystics"
		NOT = { has_character_flag = artifact_rumor_heard }
		OR = {
			ai = no
			higher_tier_than = COUNT
		}
		is_landed = yes
		mercenary = no
		holy_order = no
		artifact_hunt_rumor_trigger = no
		any_courtier_or_vassal = { artifact_rumor_councillor_trigger = yes }
	}

	weight_multiplier = {
		days = 1
		modifier = { factor = 0.25 ai = yes }
		modifier = { factor = 2 has_ambition = obj_create_treasury }
		modifier = { factor = 1.5 trait = zealous }
	}

	immediate = {
		set_character_flag = artifact_rumor_heard
		save_event_target_as = relic_ruler
		random_courtier_or_vassal = {
			limit = { artifact_rumor_councillor_trigger = yes }
			save_event_target_as = rumor_councillor
		}
		random_list = {
			10 = {
				modifier = { factor = 0.7 ai = yes }
				add_character_modifier = { name = relic_rumor_major years = 20 }
			}
			30 = {
				add_character_modifier = { name = relic_rumor_medium years = 20 }
			}
			60 = {
				add_character_modifier = { name = relic_rumor_minor years = 10 }
			}
		}
		set_variable = { which = relic_score value = 0 }
		event_target:rumor_councillor = { character_event = { id = artifact.2 } }
	}
}

# artifact.102 The hidden treasures

	id = artifact.102
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		FROM = {
			OR = {
				NOT = { has_character_flag = artifact_raid_looting }
				had_character_flag = { flag = artifact_raid_looting days = 30 }

		}
	}

	immediate = {
		save_event_target_as = original_item_owner
		random_list = {
			100 = {
				modifier = {
					factor = 0
					NOT = {
						any_artifact = {
							quality == 1
							NOT = { artifact_type_owned_by = FROM }
						}
					}
				}
				FROM = { set_variable = { which = artifact_raid_quality value = 1 } }
			}
			70 = {
				modifier = {
					factor = 0
					NOT = {
						any_artifact = {
							quality == 2
							NOT = { artifact_type_owned_by = FROM }
						}
					}
				}
				FROM = { set_variable = { which = artifact_raid_quality value = 2 } }
			}
			45 = {
				modifier = { factor = 0 NOT = { any_artifact = { quality == 3 } } }
				FROM = { set_variable = { which = artifact_raid_quality value = 3 } }
			}

				modifier = { factor = 0 NOT = { any_artifact = { quality == 4 } } }
				FROM = { set_variable = { which = artifact_raid_quality value = 4 } }
			}
			10 = {
				modifier = { factor = 0 NOT = { any_artifact = { quality == 5 } } }
				FROM = { set_variable = { which = artifact_raid_quality value = 5 } }
			}
			6000 = { # Nothing found
				modifier = { # Raiders plunder every corner
					factor = 0.2
					FROM = {
