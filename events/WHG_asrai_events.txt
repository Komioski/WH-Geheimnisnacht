# The Asrai and Athel Loren
# Namespace: asrai (was asrai in S_wood_elves_events.txt)
# Faction: Wood Elves
# Serves: Wood Elves and the spirits of Athel Loren
#
# The Avatars of Isha and Kurnous sleep through the winter in the Oak of
# Ages. The spirits of the forest punish rulers who do not honour it, join
# the courts of the faithful, and sometimes want a mortal for themselves.
# Fallen Asrai return as tree kin, the Fair Folk steal human children, the
# Wild Hunt scourges the lands of the lesser races, and the forest itself
# fights any army that besieges a glade.
#
# The Avatars and the winter sleep         asrai.1 - asrai.6
# The anger of the forest                  asrai.10 - asrai.21
# Appeasing the spirits                    asrai.25 - asrai.27
# Tree kin                                 asrai.30 - asrai.32
# Slaying the sacred                       asrai.35 - asrai.36
# Fairy-bred children                      asrai.40 - asrai.43
# The Wild Hunt                            asrai.45 - asrai.48
# Spirits at court                         asrai.50 - asrai.54
# Calling the forest to war                asrai.60 - asrai.65
# Athel Loren protects its own             asrai.70 - asrai.73
#
# Callers:
#   decisions/wh_asrai_decisions.txt: enter_oak_of_ages and
#     spouseenter_oak_of_ages (1), appease_spirits (25), summon_spirits and
#     summon_spirits_e (60), asrai_kidnap (40), asrai_join_war (73).
#   decisions/wh_magic_decisions.txt: the treeman spell (65).
#   events/WHG_battle_events.txt: an Asrai falls in battle (30, from battle.10).
#   common/on_actions/wh_on_actions.txt: yearly asrai_restless_spirits (10),
#     bi-yearly (6, 51), five-yearly (50), decade asrai_spirit_departures
#     (53), on_death (35), on_adulthood (41), on_siege_over_winner (45),
#     on_siege_pulse (72), on_become_imprisoned_any_reason (70).
# Support: common/scripted_effects/wh_asrai_effects.txt,
#   common/scripted_triggers/wh_asrai_triggers.txt
# Loc: localisation/00_WHG_asrai_events.csv
#
# Also in this file (each part keeps its own header and namespace):
#   Spellsinger events (namespace spellsinger, was WHG_spellsinger_events)

# asrai.1 The Oak of Ages
# The ruler of Athel Loren (or their spouse) enters the Oak of Ages and
# comes out as the Avatar of Isha or of Kurnous. The decision has already
# given the trait.
# Scope: ROOT is the new Avatar (enter_oak_of_ages, spouseenter_oak_of_ages).
long_character_event = {
	id = asrai.1
	title = EVTTITLEasrai1
	desc = { text = EVTDESCasrai1 trigger = { is_woman_trigger = yes } }
	desc = { text = EVTDESCasrai1b trigger = { is_woman_trigger = no } }
	picture = GFX_evt_asrai_tree

	is_triggered_only = yes # enter_oak_of_ages, spouseenter_oak_of_ages

	option = { # Avatar of Isha
		name = EVTOPTAasrai1
		trigger = { is_woman_trigger = yes }
		add_trait = avatar_isha
		leave_society = yes
		hidden_effect = {
			if = { limit = { spouse = { trait = avatar_kurnous } } add_trait = chaste }
			if = { limit = { NOT = { trait = asrai_spellsinger } } add_trait = asrai_spellsinger }
			random_list = {
				30 = { add_trait = patient }
				30 = { add_trait = just }
				30 = { add_trait = proud }
				10 = { }
			}
			remove_trait = asur_mage
			magic_lvl_increase_effect = yes
			enable_magic_xp_effect = yes
			asrai_avatar_purify_effect = yes
		}
	}
	option = { # Avatar of Kurnous
		name = EVTOPTBasrai1
		trigger = { is_female = no }
		add_trait = avatar_kurnous
		leave_society = yes
		hidden_effect = {
			if = { limit = { spouse = { trait = avatar_isha } } add_trait = chaste }
			random_list = {
				30 = { add_trait = wroth }
				30 = { add_trait = zealous }
				30 = { add_trait = proud }
				10 = { }
			}
			asrai_avatar_purify_effect = yes
		}
	}
}

# asrai.2 The winter sleep
# Each autumn the Avatars and the forest's own fall asleep until spring
# (asrai.3). Once in a hundred winters, more often when Isha is gone, the
# Avatar of Kurnous wakes in a frenzy instead (asrai.4). Winter also weakens
# the forest's armies across Athel Loren.
# Scope: an Avatar, tree kin or forest spirit (MTTH, every September).
character_event = {
	id = asrai.2
	desc = { text = EVTDESCasrai2 trigger = { trait = avatar_isha } }
	desc = { text = EVTDESCasrai2b trigger = { trait = avatar_kurnous } }
	desc = { text = EVTDESCasrai2c trigger = { asrai_forest_being_trigger = yes } }
	picture = GFX_evt_elf_winter

	religion_group = cadai_group

	trigger = {
		month = 8
		NOT = { trait = resting }
		OR = { trait = avatar_isha trait = avatar_kurnous asrai_forest_being_trigger = yes }
	}

	mean_time_to_happen = {
		months = 1
	}

	immediate = {
		add_trait = resting
		if = {
			limit = { has_character_flag = kurnous_frenzy }
			remove_trait = lunatic_3
			set_variable = { which = "lunatic_counter" value = 0 }
			clr_character_flag = kurnous_frenzy
		}
		if = {
			limit = { has_character_flag = spirit_frenzy }
			random = {
				chance = 30
				remove_trait = lunatic_3
				set_variable = { which = "lunatic_counter" value = 0 }
				clr_character_flag = spirit_frenzy
			}
		}
		e_athel_loren = {
			holder_scope = {
				if = {
					limit = {
						capital_scope = { NOT = { has_province_modifier = athel_loren_winter } }
					}
					any_realm_province = {
						add_province_modifier = { name = athel_loren_winter duration = 120 }
					}
				}
			}
		}
	}

	option = { # Isha sleeps
		name = EVTOPTAasrai2
		trigger = { trait = avatar_isha }
		hidden_effect = { character_event = { id = asrai.3 days = 120 random = 30 } }
	}
	option = { # Kurnous sleeps
		name = EVTOPTBasrai2
		trigger = { trait = avatar_kurnous }
		hidden_effect = {
			random_list = {
				99 = { character_event = { id = asrai.3 days = 120 random = 30 } }
				1 = {
					modifier = { factor = 10 NOT = { avatar_isha = { is_alive = yes } } }
					character_event = { id = asrai.4 days = 120 random = 30 }
				}
			}
		}
	}
	option = { # The forest sleeps
		name = EVTOPTCasrai2
		trigger = { asrai_forest_being_trigger = yes }
		hidden_effect = { character_event = { id = asrai.3 days = 120 random = 30 } }
	}
}

						limit = { asrai_forest_being_trigger = yes }

					}

# asrai.10 The anger of the forest (hidden)
# Scope: a ruler in Athel Loren (yearly, asrai_restless_spirits).
character_event = {
	id = asrai.10
	hide_window = yes

	is_triggered_only = yes # on_yearly_pulse
	only_rulers = yes

	trigger = {
		higher_tier_than = BARON
		is_tribal = yes # A Bretonnian who clears the forest is left in peace
		NOT = { trait = morghur }
		capital_scope = { empire = { title = e_athel_loren } }
		OR = { NOT = { religion = asrai_cadai } NOT = { piety = 0 } }
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 0.8 ai = yes }
		modifier = { factor = 1.33 trait = creature_human }
		modifier = { factor = 2 trait = creature_dwarf }
		modifier = { factor = 4 religion_group = necromantic_group }
		modifier = { factor = 1.33 trait = creature_orc }
		modifier = { factor = 2 trait = creature_goblin }
		modifier = { factor = 2 trait = creature_hobgoblin }
		modifier = { factor = 10 religion_group = chaos_gods_group }
		modifier = { factor = 1.33 trait = stressed }
		modifier = { factor = 1.33 trait = depressed }
		modifier = { factor = 0.25 trait = creature_treekin }
		modifier = { factor = 1.33 any_courtier = { asrai_forest_being_trigger = yes } }
		modifier = { factor = 1.33 religion = asrai_cadai religion_authority = 0.25 }
		modifier = { factor = 1.33 religion = asrai_cadai religion_authority = 0.50 }
		modifier = { factor = 0.2 trait = avatar_kurnous }
		modifier = { factor = 0.25 trait = avatar_isha }
		modifier = { factor = 1.33 NOT = { avatar_kurnous = { is_alive = yes } } }
		modifier = { factor = 1.33 avatar_kurnous = { trait = lunatic_3 } }
		modifier = { factor = 1.33 NOT = { avatar_isha = { is_alive = yes } } }
	}

	immediate = {
		random_list = {
			10 = { character_event = { id = asrai.11 } } # A building vanishes
			10 = { # The spouse vanishes
				trigger = { spouse = { is_alive = yes is_ruler = no } }
				spouse = { save_event_target_as = asrai_spirit_victim }
				character_event = { id = asrai.12 }
			}
			10 = { # The spouse is scarred
				trigger = { spouse = { is_alive = yes } }
				spouse = { save_event_target_as = asrai_spirit_victim }
				character_event = { id = asrai.13 }
			}
			10 = { # A child vanishes
				trigger = { any_child = { is_ruler = no } }
				random_child = {
					limit = { is_ruler = no }
					save_event_target_as = asrai_spirit_victim
				}
				character_event = { id = asrai.14 }
			}
			10 = { # Wounded
				trigger = { NOT = { trait = wounded } }
				character_event = { id = asrai.15 }
			}
			5 = { # Scarred
				trigger = { NOT = { trait = creature_treekin } }
				character_event = { id = asrai.16 }
			}
			10 = { # Maimed
				trigger = { NOT = { trait = incapable } }
				modifier = { factor = 1.25 trait = wounded }
				character_event = { id = asrai.17 }
			}
			5 = { # Lost in the forest
				modifier = { factor = 0.1 trait = avatar_isha }
				modifier = { factor = 0.1 trait = avatar_kurnous }
				modifier = { factor = 1.25 trait = incapable }
				character_event = { id = asrai.18 }
			}
			10 = { # A lover vanishes
				trigger = { any_lover = { is_ruler = no } }
				random_lover = {
					limit = { is_ruler = no }
					save_event_target_as = asrai_spirit_victim
				}
				character_event = { id = asrai.19 }
			}
			15 = { # A courtier vanishes
				trigger = {
					NOT = { trait = avatar_isha }
					NOT = { trait = avatar_kurnous }
					any_courtier = {
						asrai_mortal_courtier_trigger = yes
						NOT = { dynasty = ROOT }
						NOT = { is_lover = ROOT }
					}
				}
				random_courtier = {
					limit = {
						asrai_mortal_courtier_trigger = yes
						NOT = { dynasty = ROOT }
						NOT = { is_lover = ROOT }
					}
					save_event_target_as = asrai_spirit_victim
				}
				character_event = { id = asrai.20 }
			}
			5 = { # Driven mad
				modifier = { factor = 0.5 trait = creature_elf }
				modifier = { factor = 0.75 trait = creature_treekin }
				character_event = { id = asrai.21 }
			}
		}
	}
}
