# The Hashishin and Clan Eshin
# Namespace: assassins (was MNM.6000 - 6990 and SKASOC.1 - 30 in
#   mnm_assassins_events.txt)
# Faction: Araby & Southlands
# Serves: the two murderous societies, the Hashishin (the_assassins) and
#   Clan Eshin (eshin_society) of the Skaven, both open to men only
#
# The Hashishin of Jebel Ezna kill for Ormazd from the shadows of Araby;
# Clan Eshin's gutter runners kill for the Horned Rat and the Council of
# Thirteen. Both lend gold and collect it, recruit through feasts, oaths
# and a blood price, send their members after persecutors, upstarts, Chaos
# cultists and debtors, ease their nerves with hashish or warpstone snuff,
# threaten rulers, mark men for death, train apprentices and watch each
# other: rivals, partners in murder and blackmailers among the brothers.
# Every text speaks in its order's voice (WHG_assassins_voices.txt).
#
# Loans and gifts                      assassins.1 - assassins.11
# Recruitment                          assassins.20 - assassins.41
# Murder plots                         assassins.100 - assassins.120
# Missions                             assassins.199 - assassins.225
# The apprentice                       assassins.248 - assassins.261
# Visions                              assassins.300 - assassins.320
# Powers                               assassins.340 - assassins.345
# Among the brothers                   assassins.399 - assassins.422
# The order's lords                    assassins.490 - assassins.491
#
# Callers: the decisions in decisions/mnm_society_decisions.txt
#   (assassins.1, 5, 20, 22, 250, 300, 340, 342, 343), society.12 in
#   WHG_society_events.txt (assassins.20, 22, 225, 260, 406), on_death
#   (assassins.3, 345), the society pulse (assassins.199, 399, 410, 420),
#   on_county_religion_change (assassins.206), on_siege_over_winner
#   (assassins.222), on_new_holder, on_new_holder_inheritance and
#   on_new_holder_usurpation (assassins.490, 491), and
#   events/WHG_hashishin_events.txt (assassins.490). Triggers in
#   common/scripted_triggers/WHG_assassins_triggers.txt, effects in
#   common/scripted_effects/WHG_assassins_effects.txt, voices in
#   localisation/customizable_localisation/WHG_assassins_voices.txt, death
#   reasons death_murder_eshin, death_murder_unknown_eshin and
#   death_warpstone_overdose in common/death/00_death.txt.
# Saved: event targets grand_master, order_lord, in_debt, child_offer,
#   castle_offer, assassins_recruiter, assassins_recruitee,
#   assassins_inductor, assassins_sacrifice, inductor_rendezvous,
#   plot_target, assassin, assassin_plot_owner, assassins_plotter,
#   lover_of_target, assassins_target, assassins_aggressor,
#   owes_the_assassins, my_quest_target, conversion_province,
#   conversion_province_owner, faithful_province, faithful_defender,
#   infidel_holding, infidel_holding_owner, infidel_holding_province,
#   fire_town, fire_town_province, threaten_dear_relation, drug_new_lover,
#   drug_cheater_lover, drug_cheater_spouse, drug_plotter_reveal,
#   drug_plotter_target, assassin_protege, assassin_quest_target,
#   friend_or_foe, slacker_buddy, conversationalist,
#   assassin_blackmail_target, assassins_lender; variable assassins_debt;
#   flags assassins_debt_eshin, assassins_refused_to_pay_debt,
#   assassins_joining_eshin, assassins_protege_eshin, the mission reasons
#   assassins_target_* and the marks assassins_target_marked_for_death and
#   eshin_target_marked_for_death.
# Dropped: the random religion names of alternate starts (Araby always
#   follows Ormazd), the long commented-out MNM.6230, the Cult of Pleasure
#   branch of the mission tombola (that cult has its own missions), and the
#   SKASOC copies of the tombola and debt collection, merged into
#   assassins.199 and assassins.217.

# assassins.1 A loan from the order
# Hidden. The member asked the order for a loan (assassins_borrow_money);
# the Grand Master sends the answer.
# Scope: ROOT is the Grand Master, FROM the member.
character_event = {
	id = assassins.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		save_event_target_as = grand_master
		FROM = { letter_event = { id = assassins.2 } }
	}
}

# assassins.2 The Grand Master's purse
# The Grand Master grants a loan of 50, 250 or 500 gold, the larger sums for
# a member with standing in the order. An AI Grand Master who holds the
# order's title pays it out of his own treasury. The debt falls due in about
# four years (assassins.10).
# Scope: ROOT is the member, event_target:grand_master the Grand Master.
letter_event = {
	id = assassins.2
	desc = EVTDESCassassins2
	border = GFX_event_letter_frame_religion
	is_triggered_only = yes

	immediate = {
		set_variable = { which = assassins_debt value = 0 }
		add_character_modifier = { name = assassins_debt duration = -1 }
		set_character_flag = block_payback
		if = {
			limit = { society_member_of = eshin_society }
			set_character_flag = assassins_debt_eshin
		}
		else = {
			clr_character_flag = assassins_debt_eshin
		}
	}

	# 50 gold
	option = {
		trigger = {
			event_target:grand_master = {
				OR = {
					assassins_order_bank_trigger = no
					wealth = 50
				}
			}
		}
		name = EVTOPTAassassins2
		wealth = 50
		hidden_effect = {
			change_variable = { which = assassins_debt value = 50 }
			character_event = { id = assassins.10 days = 1000 random = 1000 }
			event_target:grand_master = {
				if = {
					limit = { assassins_order_bank_trigger = yes }
					wealth = -50
				}
			}
		}
		clr_character_flag = block_payback
		ai_chance = {
			factor = 1
			modifier = { factor = 0 monthly_income = 10 }
		}
	}

	# 250 gold
	option = {
		trigger = {
			event_target:grand_master = {
				OR = {
					assassins_order_bank_trigger = no
					wealth = 250
				}
			}
			has_society_currency_minor_trigger = yes
		}
		name = EVTOPTBassassins2
		wealth = 250
		detract_society_currency_minor_effect = yes
		hidden_effect = {
			change_variable = { which = assassins_debt value = 250 }
			character_event = { id = assassins.10 days = 1000 random = 1000 }
			event_target:grand_master = {
				if = {
					limit = { assassins_order_bank_trigger = yes }
					wealth = -250
				}
			}
		}
		clr_character_flag = block_payback
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0
				OR = {
					monthly_income = 20
					NOT = { monthly_income = 10 }
				}
			}
		}
	}

	# 500 gold
	option = {
		trigger = {
			event_target:grand_master = {
				OR = {
					assassins_order_bank_trigger = no
					wealth = 500
				}
			}
			has_society_currency_medium_trigger = yes
		}
		name = EVTOPTCassassins2
		wealth = 500
		detract_society_currency_medium_effect = yes
		hidden_effect = {
			change_variable = { which = assassins_debt value = 500 }
			character_event = { id = assassins.10 days = 1000 random = 1000 }
			event_target:grand_master = {
				if = {
					limit = { assassins_order_bank_trigger = yes }
					wealth = -500
				}
			}
		}
		clr_character_flag = block_payback
		ai_chance = {
			factor = 1
			modifier = { factor = 0 NOT = { monthly_income = 20 } }
		}
	}

	# No loan after all
	option = {
		name = EVTOPTDassassins2
		hidden_effect = {
			remove_character_modifier = assassins_debt
			clr_character_flag = assassins_debt_eshin
		}
		clr_character_flag = block_payback
		ai_chance = { factor = 0 }
	}
}

# assassins.3 The debt passes to the heir
# Hidden, on death. The heir takes on the debt, and the order asks him for
# it soon after.
# Scope: ROOT is the dead debtor.
character_event = {
	id = assassins.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_character_modifier = assassins_debt
		current_heir = { always = yes }
	}

	immediate = {
		current_heir = {
			add_character_modifier = { name = assassins_debt duration = -1 }
			set_variable = { which = assassins_debt which = ROOT }
			if = {
				limit = { ROOT = { has_character_flag = assassins_debt_eshin } }
				set_character_flag = assassins_debt_eshin
			}
			if = {
				limit = { ROOT = { has_character_flag = assassins_refused_to_pay_debt } }
				set_character_flag = assassins_refused_to_pay_debt
			}
			character_event = { id = assassins.10 days = 50 }
		}
	}
}

# assassins.10 The debt falls due
# Hidden. The Grand Master of the order that lent the money (Clan Eshin if
# the flag assassins_debt_eshin is set, else the Hashishin) asks for it.
# Cleared by assassins_clear_debt_effect. If the order has no Grand Master,
# the debt waits another year.
# Scope: ROOT is the debtor.
character_event = {
	id = assassins.10
	hide_window = yes
	is_triggered_only = yes

	trigger = { has_character_modifier = assassins_debt }

	immediate = {
		save_event_target_as = in_debt
		if = {
			limit = { has_character_flag = assassins_debt_eshin }
			random_society_member = {
				limit = { society_member_of = eshin_society is_society_grandmaster = yes }
				save_event_target_as = grand_master
			}
		}
		else = {
			random_society_member = {
				limit = { society_member_of = the_assassins is_society_grandmaster = yes }
				save_event_target_as = grand_master
			}
		}
		if = {
			limit = { event_target:grand_master = { always = yes } }
			event_target:grand_master = { character_event = { id = assassins.11 } }
		}
		else = {
			character_event = { id = assassins.10 days = 365 }
		}
	}
}

# assassins.7 The order wants its gold
# Pay back the loan, or, if it was large and the order has a lord, give him
# a son of yours (250 gold) or let him build a castle in your demesne (500
# gold). A debtor who refuses becomes the target of the order's debt
# collectors (assassins.207).
# Scope: ROOT is the debtor, FROM the Grand Master.
letter_event = {
	id = assassins.7
	border = GFX_event_letter_frame_religion

	desc = {
		trigger = {
			OR = {
				is_variable_equal = { which = assassins_debt value = 50 }
				AND = {
					is_variable_equal = { which = assassins_debt value = 250 }
					NAND = {
						event_target:child_offer = { always = yes }
						event_target:order_lord = { always = yes }
					}
				}
				AND = {
					is_variable_equal = { which = assassins_debt value = 500 }
					NAND = {
						event_target:castle_offer = { always = yes }
						event_target:order_lord = { always = yes }
					}
				}
			}
		}
		text = EVTDESCassassins7_a
	}
	desc = {
		trigger = {
			is_variable_equal = { which = assassins_debt value = 250 }
			event_target:child_offer = { always = yes }
			event_target:order_lord = { always = yes }
		}
		text = EVTDESCassassins7_b
	}
	desc = {
		trigger = {
			is_variable_equal = { which = assassins_debt value = 500 }
			event_target:castle_offer = { always = yes }
			event_target:order_lord = { always = yes }
		}
		text = EVTDESCassassins7_c
	}

	is_triggered_only = yes

	trigger = {
		has_character_modifier = assassins_debt
		check_variable = { which = assassins_debt value = 50 }
	}

	immediate = {
		FROM = { save_event_target_as = grand_master }
		random_independent_ruler = {
			limit = {
				OR = {
					AND = {
						ROOT = { has_character_flag = assassins_debt_eshin }
						has_landed_title = k_eshin
						society_member_of = eshin_society
					}
					AND = {
						ROOT = { NOT = { has_character_flag = assassins_debt_eshin } }
						has_landed_title = d_hashishin
						society_member_of = the_assassins
					}
				}
			}
			save_event_target_as = order_lord
		}
		random_child = {
			limit = {
				is_primary_heir = no
				is_adult = yes
				prisoner = no
				useless_character_trigger = no
				is_in_society = no
				OR = {
					AND = {
						ROOT = { has_character_flag = assassins_debt_eshin }
						can_join_society = eshin_society
					}
					AND = {
						ROOT = { NOT = { has_character_flag = assassins_debt_eshin } }
						can_join_society = the_assassins
					}
				}
			}
			save_event_target_as = child_offer
		}
		random_demesne_province = {
			limit = {
				is_capital = no
				has_empty_holding = yes
			}
			save_event_target_as = castle_offer
		}
	}

	# Pay back
	option = {
		name = EVTOPTAassassins7
		trigger = { assassins_can_pay_debt_trigger = yes }
		assassins_pay_debt_effect = yes
		assassins_clear_debt_effect = yes
	}

	# Give a son
	option = {
		trigger = {
			is_variable_equal = { which = assassins_debt value = 250 }
			event_target:child_offer = { always = yes }
			event_target:order_lord = { always = yes }
		}
		name = EVTOPTBassassins7
		custom_tooltip = { text = EVTOPTBassassins7_tooltip }
		event_target:child_offer = {
			move_character = event_target:order_lord
			if = {
				limit = { ROOT = { has_character_flag = assassins_debt_eshin } }
				join_society = eshin_society
			}
			else = {
				join_society = the_assassins
			}
			hidden_effect = {
				clear_wealth = yes
				end_inaccessibility_effect = yes
				add_trait = disinherited
			}
		}
		assassins_clear_debt_effect = yes
		ai_chance = {
			factor = 1
			modifier = { factor = 0 assassins_can_pay_debt_trigger = yes wealth = 600 }
		}
	}

	# Let them build a castle
	option = {
		trigger = {
			is_variable_equal = { which = assassins_debt value = 500 }
			event_target:castle_offer = { always = yes }
			event_target:order_lord = { always = yes }
			wealth = 100
		}
		name = EVTOPTCassassins7
		wealth = -100
		event_target:castle_offer = {
			build_holding = {
				type = castle
				holder = event_target:order_lord
			}
		}
		assassins_clear_debt_effect = yes
		ai_chance = {
			factor = 1
			modifier = { factor = 0 assassins_can_pay_debt_trigger = yes wealth = 1000 }
		}
	}

	# Refuse
	option = {
		name = EVTOPTDassassins7
		custom_tooltip = { text = EVTOPTDassassins7_tooltip }
		set_character_flag = assassins_refused_to_pay_debt
		ai_chance = {
			factor = 1
			modifier = { factor = 0.1 assassins_can_pay_debt_trigger = yes }
			modifier = { factor = 2 trait = greedy }
		}
	}
}

# assassins.20 Alms and fasting
# A Hashishin recruiter tests an outsider's faith over a late feast. Agree
# and the talks go on (assassins.21); scoff and they end. Recruits who
# already follow Ormazd, and every Clan Eshin recruit, skip to assassins.22.
# Scope: ROOT is the recruit, FROMFROM the recruiter (assassins_recruit, or
# society.12).
character_event = {
	id = assassins.20
	picture = GFX_evt_feast
	border = GFX_event_normal_frame_diplomacy

	desc = {
		trigger = { NOT = { religion_group = ormazd_gods_group } }
		text = EVTDESCassassins20_a
	}
	desc = {
		trigger = { religion_group = ormazd_gods_group }
		text = EVTDESCassassins20_b
	}

	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { event_target:assassins_recruiter = { always = yes } } }
			FROMFROM = { save_event_target_as = assassins_recruiter }
		}
		save_event_target_as = assassins_recruitee
		set_character_flag = ongoing_recruitment
	}

	# Noble deeds
	option = {
		name = {
			trigger = { NOT = { religion_group = ormazd_gods_group } }
			text = EVTOPTAassassins20_other
		}
		name = {
			trigger = { religion_group = ormazd_gods_group }
			text = EVTOPTAassassins20_faithful
		}
		reverse_opinion = {
			who = event_target:assassins_recruiter
			modifier = opinion_agreed_with_opinion
			years = 5
		}
		set_character_flag = approves_of_ormazd
		hidden_effect = { event_target:assassins_recruiter = { character_event = { id = assassins.21 days = 3 } } }
		ai_chance = {
			factor = 20
			modifier = {
				factor = 0
				trait = zealous
				NOT = { true_religion_group = ormazd_gods_group }
			}
			modifier = {
				factor = 0
				NOT = { true_religion = ormazd }
				OR = {
					controls_religion = yes
					capital_holding = { holding_type = temple }
				}
			}
			modifier = { factor = 1.5 trait = cynical }
			modifier = { factor = 1.5 trait = trusting }
			modifier = { factor = 1.5 is_dumb_trigger = yes }
			modifier = { factor = 1.5 true_religion_group = ormazd_gods_group }
			modifier = { factor = 100 has_character_flag = was_trained_as_assassin_protege }
		}
	}

	# Never
	option = {
		trigger = { NOT = { true_religion = ormazd } }
		name = {
			trigger = { NOT = { religion_group = ormazd_gods_group } }
			text = EVTOPTBassassins20_other
		}
		name = {
			trigger = { religion_group = ormazd_gods_group }
			text = EVTOPTBassassins20_faithful
		}
		reverse_opinion = {
			who = event_target:assassins_recruiter
			modifier = opinion_disagreed_with_opinion
			years = 5
		}
		set_character_flag = disapproves_of_ormazd
		hidden_effect = { event_target:assassins_recruiter = { character_event = { id = assassins.21 days = 3 } } }
		ai_chance = {
			factor = 80
			modifier = { factor = 1.5 has_pious_trait_trigger = yes }
			modifier = {
				factor = 2
				has_crusade_trait_trigger = yes
				NOT = { trait = mujahid }
			}
		}
	}
}

# assassins.199 A task from the Grand Master
# Hidden, every two years. Picks a mission for a member of either order: a
# killing (assassins.200), converting a county to Ormazd (Hashishin only,
# assassins.203), collecting a debt (assassins.214), seizing an enemy
# holding (assassins.220) or training a protege (assassins.248). AI members
# mostly skip the missions and gain or lose standing instead.
# Scope: ROOT is the member.
character_event = {
	id = assassins.199
	hide_window = yes

	is_triggered_only = yes
	is_in_society = yes
	only_capable = yes

	trigger = {
		is_member_of_any_murderous_society = yes
		has_any_quest = no
		NOT = { has_character_modifier = quest_cooldown_timer }
		is_society_grandmaster = no
		is_inaccessible_trigger = no
	}

	immediate = {
		assassins_save_grand_master_effect = yes

		random_list = {
			15 = { # A killing
				character_event = { id = assassins.200 }
			}
			20 = { # Convert a county to Ormazd
				modifier = { factor = 0 ai = yes }
				modifier = { factor = 0 NOT = { society_member_of = the_assassins } }
				modifier = { factor = 0 NOT = { religion = ormazd } }
				event_target:grand_master = { character_event = { id = assassins.203 } }
			}
			15 = { # Collect a debt
				modifier = {
					factor = 0
					NOT = {
						any_character = {
							is_offmap_ruler = no
							is_within_diplo_range = ROOT
							assassins_debtor_of_root_trigger = yes
						}
					}
				}
				event_target:grand_master = { character_event = { id = assassins.214 } }
			}
			10 = { # Seize an enemy holding
				modifier = { factor = 0 is_playable = no }
				modifier = { factor = 0 independent = no }
				modifier = { factor = 0 ai = yes }
				modifier = { factor = 0 assassins_true_believer_trigger = no }
				modifier = {
					factor = 0
					NOT = {
						any_realm_province = {
							any_neighbor_province = {
								owner = {
									top_liege = {
										NOT = { character = ROOT }
										NOT = { religion = ROOT }
										any_realm_province = {
											owner = { NOT = { religion = ROOT } }
											any_province_holding = {
												owner = { NOT = { religion = ROOT } }
												OR = {
													holding_type = temple
													holding_type = castle
												}
											}
										}
									}
								}
							}
						}
					}
				}
				event_target:grand_master = { character_event = { id = assassins.220 } }
			}
			5 = { # Train a protege
				event_target:grand_master = { character_event = { id = assassins.248 } }
			}
			50 = { # The AI skips the mission
				modifier = { factor = 0 ai = no }
				random_list = {
					70 = {
						modifier = { factor = 0.9 society_influence >= 30 }
						modifier = { factor = 0.9 society_influence >= 40 }
						modifier = { factor = 0.8 society_influence >= 50 }
						modifier = { factor = 0.7 society_influence >= 70 }
						modifier = { factor = 0.6 society_influence >= 80 }
						modifier = { factor = 0.5 society_influence >= 90 }
						add_mission_succeed_influence_effect = yes
						if = { # Lets landless members reach rank 2, where they can take part in plots
							limit = {
								society_rank < 2
								society_can_rank_up = yes
							}
							random = {
								chance = 50
								society_rank_up = 1
							}
						}
					}
					30 = {
						add_mission_fail_influence_effect = yes
					}
				}
			}
			125 = { # Nothing for the AI this time
				modifier = { factor = 0 ai = no }
			}
		}
	}
}
