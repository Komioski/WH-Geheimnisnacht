###############################################################################
# Asur debates, symposia and favours
# Namespace: asurdebate (was asurdebate in S_high_elves_debate_events.txt)
# Faction: High Elves
#
# The great quarrel of Ulthuan: should the Asur shut their gates on the
# lesser races, or ride out to fight beside them? Isolationist and
# interventionist princes argue it out in public debates, buy each other's
# support at private symposia, and call in old favours.
#
# Public debates                          asurdebate.1 - asurdebate.8
# Private symposia                        asurdebate.10 - asurdebate.18
# Public shaming at a symposium           asurdebate.20 - asurdebate.22
# Buying support among isolationists      asurdebate.30 - asurdebate.32
#
# Callers (decisions/wh_asur_decisions.txt):
#   asur_debate_decision (1), asur_force_stance_favour (8),
#   asur_meeting_decision (10), asur_buy_favour (30).
# All four are targeted decisions: the event's ROOT and FROM are the target
# and FROMFROM is the character who took the decision.
# Support: common/scripted_effects/wh_asur_debate_effects.txt,
#   common/scripted_triggers/wh_asur_debate_triggers.txt
# Loc: localisation/00_WHG_asur_debate_events.csv
###############################################################################

namespace = asurdebate

###############################################################################
# Public debates
# A prince challenges a rival of the other stance to a debate before the
# nobles of Ulthuan. The loser must admit the winner was right and change
# stance, or refuse and be scorned for it. If neither yields, they become
# rivals.
###############################################################################

# asurdebate.1 Challenged to a debate
# Scope: ROOT is the challenged prince, FROMFROM the challenger
# (asur_debate_decision).
character_event = {
	id = asurdebate.1
	desc = EVTDESCasurdebate1
	picture = GFX_evt_elf_pondering
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # asur_debate_decision

	immediate = {
		if = {
			limit = { NOT = { has_character_modifier = recent_debate_timer } }
			add_character_modifier = { name = recent_debate_timer duration = 365 hidden = yes }
		}
	}

	option = { # Accept
		name = EVTOPTAasurdebate1
		if = { limit = { trait = shy } random = { chance = 20 remove_trait = shy } }
		if = {
			limit = {
				FROMFROM = { has_opinion_modifier = { who = ROOT modifier = grievously_offended } }
			}
			FROMFROM = { remove_opinion = { who = ROOT modifier = grievously_offended } }
		}
		save_event_target_as = debate_opponent
		FROMFROM = {
			save_event_target_as = debate_challenger
			character_event = { id = asurdebate.2 days = 3 }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 2 trait = gregarious }
			modifier = { factor = 1.5 trait = proud }
			modifier = { factor = 1.5 trait = patient }
			modifier = { factor = 1.5 age = 800 }
			modifier = { factor = 0.75 trait = craven }
			modifier = { factor = 0.75 religion = cult_of_pleasure }
		}
	}
	option = { # Refuse
		name = EVTOPTBasurdebate1
		remove_character_modifier = recent_debate_timer
		FROMFROM = {
			remove_character_modifier = recent_debate_timer
			opinion = { modifier = opinion_fled_debate who = ROOT years = 100 }
			character_event = { id = asurdebate.3 days = 3 }
		}
		ai_chance = { factor = 1 }
	}
}

# asurdebate.2 The debate
# The nobles gather to hear both sides. The better speaker usually wins; two
# evenly matched or two wrathful princes usually end in a stalemate.
# Scope: ROOT is the challenger, FROM the challenged prince (asurdebate.1).
character_event = {
	id = asurdebate.2
	desc = EVTDESCasurdebate2
	picture = GFX_evt_elf_pondering
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # asurdebate.1

	option = {
		name = EVTOPTAasurdebate2
		opinion = { modifier = opinion_fought_debate who = FROM years = 150 }
		FROM = { opinion = { modifier = opinion_fought_debate who = ROOT years = 150 } }
		remove_character_modifier = recent_debate_victory
		FROM = { remove_character_modifier = recent_debate_victory }
		hidden_effect = {
			random_list = {
				75 = { # Stalemate
					modifier = {
						factor = 0.2
						OR = {
							attribute_diff = { character = FROM attribute = diplomacy value = 4 }
							FROM = {
								attribute_diff = {
									character = ROOT
									attribute = diplomacy
									value = 4
								}
							}
						}
					}
					modifier = { factor = 5 trait = wroth FROM = { trait = wroth } }
					if = {
						limit = {
							NOT = { num_of_rivals = 2 }
							FROM = { NOT = { num_of_rivals = 2 } }
							NOT = { is_rival = FROM }
						}
						add_rival = FROM
					}
					long_character_event = { id = asurdebate.6 }
					FROM = { long_character_event = { id = asurdebate.6 } }
				}
				200 = { # The challenger wins
					modifier = {
						factor = 0.5
						FROM = {
							attribute_diff = { character = ROOT attribute = diplomacy value = 2 }
						}
					}
					modifier = {
						factor = 0.5
						FROM = {
							attribute_diff = { character = ROOT attribute = diplomacy value = 4 }
						}
					}
					modifier = {
						factor = 0.5
						FROM = {
							attribute_diff = { character = ROOT attribute = diplomacy value = 6 }
						}
					}
					modifier = {
						factor = 0.5
						FROM = {
							attribute_diff = { character = ROOT attribute = diplomacy value = 8 }
						}
					}
					add_character_modifier = { name = recent_debate_victory months = 12 }
					long_character_event = { id = asurdebate.4 }
				}
				200 = { # The challenged prince wins
					modifier = {
						factor = 0.5
						attribute_diff = { character = FROM attribute = diplomacy value = 2 }
					}
					modifier = {
						factor = 0.5
						attribute_diff = { character = FROM attribute = diplomacy value = 4 }
					}
					modifier = {
						factor = 0.5
						attribute_diff = { character = FROM attribute = diplomacy value = 6 }
					}
					modifier = {
						factor = 0.5
						attribute_diff = { character = FROM attribute = diplomacy value = 8 }
					}
					FROM = {
						add_character_modifier = { name = recent_debate_victory months = 12 }
						long_character_event = { id = asurdebate.4 }
					}
				}
			}
		}
	}
}

# asurdebate.8 An old debt called in
# A prince the character owes a favour offers to forget it if the character
# takes up the prince's stance.
# Scope: ROOT is the debtor, FROMFROM the creditor
# (asur_force_stance_favour, which already cleared the favour).
character_event = {
	id = asurdebate.8
	desc = EVTDESCasurdebate8
	picture = GFX_evt_elf_pondering
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # asur_force_stance_favour

	option = { # Agree
		name = EVTOPTAasurdebate8
		if = {
			limit = { FROMFROM = { trait = interventionist_elf } }
			remove_trait = isolationist_elf
			add_trait = interventionist_elf
		}
		else_if = {
			limit = { FROMFROM = { trait = isolationist_elf } }
			remove_trait = interventionist_elf
			add_trait = isolationist_elf
		}
	}
	option = { # Refuse
		name = EVTOPTBasurdebate8
		trigger = { ai = no }
		prestige = -400
		opinion = { modifier = opinion_insulted who = FROMFROM years = 150 }
		FROMFROM = { opinion = { modifier = opinion_insulted who = ROOT years = 150 } }
	}
}

###############################################################################
# Private symposia
# An isolationist prince invites a peer to a symposium at his villa and,
# between the music and the wine, asks for something: gold, a favour, the
# release of an old debt, the guest's divorce or the guest's hand. The guest
# may grant it freely, name a price (gold, a favour or marriage), or refuse.
# Or the host may have invited the guest only to shame them in front of
# everyone (asurdebate.20).
# The host and guest are the event targets asurmeet_host and
# asurmeet_guest; what was asked is a flag on the host, the price a flag on
# the guest.
###############################################################################

# asurdebate.10 An invitation to a symposium
# Scope: ROOT is the guest, FROMFROM the host (asur_meeting_decision).
letter_event = {
	id = asurdebate.10
	desc = EVTDESCasurdebate10

	is_triggered_only = yes # asur_meeting_decision

	option = { # Attend
		name = EVTOPTAasurdebate10
		if = {
			limit = {
				FROMFROM = { has_opinion_modifier = { who = ROOT modifier = grievously_offended } }
			}
			FROMFROM = { remove_opinion = { who = ROOT modifier = grievously_offended } }
		}
		FROMFROM = { long_character_event = { id = asurdebate.12 days = 5 random = 5 } }
		ai_chance = { factor = 98 }
	}
	option = { # Decline
		name = EVTOPTBasurdebate10
		FROMFROM = {
			opinion = { modifier = opinion_fled_debate who = ROOT years = 100 }
			letter_event = { id = asurdebate.11 days = 4 random = 4 }
		}
		ai_chance = {
			factor = 2
			modifier = { factor = 0.25 NOT = { prestige = 250 } }
			modifier = { factor = 1.25 trait = shy }
			modifier = { factor = 1.5 trait = paranoid }
			modifier = { factor = 1.5 higher_tier_than = FROMFROM }
			modifier = { factor = 2 is_rival = FROMFROM }
			modifier = { factor = 0 is_friend = FROMFROM }
			modifier = { factor = 0 is_lover = FROMFROM }
			modifier = { factor = 2 NOT = { opinion = { who = FROMFROM value = -50 } } }
			modifier = { factor = 1.5 NOT = { opinion = { who = FROMFROM value = -20 } } }
		}
	}
}

# asurdebate.12 The symposium
# The host takes the guest aside and asks for something, or springs the
# trap of a public shaming.
# Scope: ROOT is the host, FROM the guest (asurdebate.10).
long_character_event = {
	id = asurdebate.12
	desc = EVTDESCasurdebate12
	picture = GFX_evt_asur_arguing
	border = GFX_event_long_frame_religion

	is_triggered_only = yes # asurdebate.10

	immediate = {
		save_event_target_as = asurmeet_host
		FROM = { save_event_target_as = asurmeet_guest }
		asurmeet_clear_flags_effect = yes
		FROM = { asurmeet_clear_flags_effect = yes }
	}

	option = { # A great sum
		name = EVTOPTAasurdebate12
		trigger = { FROM = { wealth = 500 } }
		set_character_flag = asurmeet_ask_gold_500
		FROM = { long_character_event = { id = asurdebate.13 } }
		ai_chance = {
			factor = 15
			modifier = { factor = 2 trait = greedy }
			modifier = { factor = 2 FROM = { wealth = 1500 } }
			modifier = { factor = 2 NOT = { wealth = 0 } }
		}
	}
	option = { # A considerable sum
		name = EVTOPTBasurdebate12
		trigger = { FROM = { wealth = 250 } }
		set_character_flag = asurmeet_ask_gold_250
		FROM = { long_character_event = { id = asurdebate.13 } }
		ai_chance = {
			factor = 15
			modifier = { factor = 1.5 trait = greedy }
			modifier = { factor = 2 NOT = { wealth = 0 } }
		}
	}
	option = { # A small token
		name = EVTOPTCasurdebate12
		trigger = { FROM = { wealth = 100 } }
		set_character_flag = asurmeet_ask_gold_100
		FROM = { long_character_event = { id = asurdebate.13 } }
		ai_chance = { factor = 10 modifier = { factor = 0.5 trait = proud } }
	}
	option = { # A favour
		name = EVTOPTDasurdebate12
		trigger = { NOT = { owes_favor_to = FROM } FROM = { NOT = { owes_favor_to = ROOT } } }
		set_character_flag = asurmeet_ask_favor
		FROM = { long_character_event = { id = asurdebate.13 } }
		ai_chance = {
			factor = 25
			modifier = { factor = 1.5 trait = ambitious }
			modifier = { factor = 1.75 trait = paranoid }
			modifier = { factor = 1.75 FROM = { trait = interventionist_elf } }
		}
	}
	option = { # Forget my debt
		name = EVTOPTEasurdebate12
		trigger = { owes_favor_to = FROM }
		set_character_flag = asurmeet_ask_clear
		FROM = { long_character_event = { id = asurdebate.13 } }
		ai_chance = {
			factor = 25
			modifier = { factor = 1.5 trait = greedy }
			modifier = { factor = 1.5 trait = proud }
		}
	}
	option = { # Leave your spouse
		name = EVTOPTFasurdebate12
		trigger = { FROM = { is_married = yes NOT = { any_spouse = { character = ROOT } } } }
		set_character_flag = asurmeet_ask_divorce
		FROM = { long_character_event = { id = asurdebate.13 } }
		ai_chance = {
			factor = 10
			modifier = { factor = 3 is_lover = FROM }
			modifier = { factor = 2 FROM = { any_spouse = { is_rival = ROOT } } }
			modifier = { factor = 1.5 trait = deceitful }
			modifier = { factor = 1.5 trait = cruel }
		}
	}
	option = { # Marry me
		name = EVTOPTGasurdebate12
		trigger = { FROM = { asurmeet_can_wed_trigger = yes } }
		set_character_flag = asurmeet_ask_marriage
		FROM = { long_character_event = { id = asurdebate.13 } }
		ai_chance = {
			factor = 25
			modifier = { factor = 3 is_lover = FROM }
			modifier = { factor = 1.5 FROM = { higher_tier_than = ROOT } }
			modifier = { factor = 1.5 trait = ambitious }
			modifier = { factor = 1.5 NOT = { num_of_children = 1 } }
			modifier = {
				factor = 1.5
				FROM = { OR = { trait = fair trait = pretty trait = beautiful } }
			}
			modifier = { factor = 0.5 FROM = { NOT = { religion = ROOT } } }
			modifier = { factor = 0.5 trait = celibate }
		}
	}
	option = { # Shame them before everyone
		name = EVTOPTHasurdebate12
		trigger = { OR = { ai = no is_rival = FROM } }
		if = { limit = { religion = FROM } piety = -50 }
		if = {
			limit = {
				OR = {
					AND = { trait = isolationist_elf FROM = { trait = isolationist_elf } }
					AND = { trait = interventionist_elf FROM = { trait = interventionist_elf } }
				}
			}
			prestige = -50
		}

			factor = 25
			modifier = { factor = 0 is_lover = FROM }

			modifier = { factor = 1.5 trait = cruel }
			modifier = { factor = 0.5 trait = kind }
			modifier = { factor = 2 NOT = { opinion = { who = FROM value = -50 } } }
		}
	}
}
