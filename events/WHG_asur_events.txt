# asur.1 Crossing the flames
# The would-be Phoenix King walks into the Flame of Asuryan. Most pass
# unharmed (asur.2); the tainted, the mutated and the possessed are often
# burned (asur.3) or consumed (asur.4).
# Scope: ROOT is the claimant (crown decision).
character_event = {
	id = asur.1
	desc = EVTDESCasur1
	picture = GFX_evt_phoenix

	is_triggered_only = yes

	option = {
		name = EVTOPTAasur1
		hidden_tooltip = {
			random_list = {
				#all is fine
				75 = {
					modifier = { factor = 0.75 religion = cult_of_pleasure }
					modifier = { factor = 0.75 religion_group = cytharai_group }
					modifier = { factor = 0.75 trait = asuryan_stain }
					modifier = { factor = 0.5 trait = lunatic_3 }
					modifier = { factor = 0.75 trait = lunatic_2 }
					modifier = { factor = 0.85 trait = lunatic_1 }
					modifier = { factor = 0.25 trait = possessed }
					modifier = { factor = 0.95 trait = craven }
					modifier = { factor = 0.95 trait = cruel }
					modifier = { factor = 0.95 trait = infirm }
					modifier = { factor = 0.5 society_member_of = cult_of_pleasure }
					modifier = { factor = 0.95 trait = interventionist_elf }
					modifier = { factor = 0.95 trait = slow }
					modifier = { factor = 0.95 trait = imbecile }
					modifier = { factor = 0.85 trait = inbred }
					modifier = { factor = 0.5 has_minor_mutation_trigger = yes }
					modifier = { factor = 0.5 has_major_mutation_trigger = yes }
					modifier = { factor = 0.25 has_massive_mutation_trigger = yes }
					character_event = { id = asur.2 }
				}
				#burned!
				20 = {
					modifier = { factor = 5 has_minor_mutation_trigger = yes }
					modifier = { factor = 8 has_major_mutation_trigger = yes }
					modifier = { factor = 12 has_massive_mutation_trigger = yes }
					modifier = { factor = 15 trait = possessed }
					character_event = { id = asur.3 }
				}
				#burned to death.
				5 = {
					modifier = { factor = 5 has_minor_mutation_trigger = yes }
					modifier = { factor = 8 has_major_mutation_trigger = yes }
					modifier = { factor = 12 has_massive_mutation_trigger = yes }
					modifier = { factor = 14 trait = possessed }
					character_event = { id = asur.4 }

			}
		}
	}

# asur.2 The flames accept
# The claimant becomes Phoenix King; a line once burned is redeemed.
# Scope: ROOT is the new Phoenix King (asur.1).
character_event = {
	id = asur.2
	desc = EVTDESCasur2
	picture = GFX_evt_phoenix_king
	sound = church_bells

	is_triggered_only = yes

	option = {
		name = EVTOPTAasur2
		prestige = 250
		add_trait = phoenix_king
		if = {
			limit = {
				any_owned_bloodline = {
					has_bloodline_flag = burnt_by_flames
					bloodline_is_active_for = PREV
				}
				NOT = { any_owned_bloodline = { has_bloodline_flag = burnt_by_flames_redeemed } }
			}
			# Apply redeemed by flames bloodline
			narrative_event = { id = aiclean.46 }
		}
		hidden_tooltip = {
			#inform vassals
			any_vassal = { character_event = { id = asur.5 } }
		}
	}
}
# asur.3 Burned
# Asuryan rejects the claimant: stained, burned and laid up until they
# recover (asur.8). Their living children carry the stain (asur.7).
# Scope: ROOT is the claimant (asur.1).
character_event = {
	id = asur.3
	desc = EVTDESCasur3
	picture = GFX_evt_malekith
	sound = church_bells

	is_triggered_only = yes

	option = {
		name = EVTOPTAasur3
		prestige = -250
		add_trait = asuryan_stain
		add_trait = burnt_by_asuryan
		add_trait = incapable #just for a while to recover
		# Apply burnt by flames bloodline
		character_event = { id = aiclean.45 }
		hidden_tooltip = {
			#inform vassals
			any_vassal = { character_event = { id = asur.6 } }
			#already born children acquire stain
			any_child = {
				limit = { religion_group = ROOT }
				character_event = { id = asur.7 days = 2 }
			}
			#recover from burns
			character_event = { id = asur.8 days = 30 random = 90 }
		}
	}
}
# asur.4 Consumed by the flames
# The claimant dies in the fire. The other Princes raise one of their own
# over the heir (asur.10). Those who cannot die are burned instead.
# Scope: ROOT is the claimant (asur.1).
character_event = {
	id = asur.4
	desc = EVTDESCasur4
	picture = GFX_evt_malekith
	sound = church_bells

	is_triggered_only = yes

	option = {
		name = EVTOPTAasur4
		prestige = -500
		add_trait = asuryan_stain
		add_trait = burnt_by_asuryan
		# Apply burnt by flames bloodline
		character_event = { id = aiclean.45 }
		hidden_tooltip = {
			any_child = {
				limit = { religion_group = ROOT }
				character_event = { id = asur.7 days = 2 }
			}
		}
		if = { # Too mighty to die: burned like asur.3 instead
			limit = { is_invincible_trigger = yes }
			antideath_effect = yes
			add_trait = incapable
			hidden_tooltip = {
				any_vassal = { character_event = { id = asur.6 } }
				character_event = { id = asur.8 days = 30 random = 90 }
			}
		}
		else = {
			hidden_tooltip = {
				any_vassal = { character_event = { id = asur.9 } }
				# The Princes choose one of their own over the heir
				random_vassal = {
					limit = { asur_phoenix_candidate_trigger = yes }
					character_event = { id = asur.10 days = 7 }
				}
			}
			death = { death_reason = death_flames_of_asuryan }
		}
	}
}

# When war is declared on the Everqueen, every independent elf ruler of the
# faith hears the call (asur.50). The interaction gives the attacker and

	hide_window = yes

		OR = { asur_everqueen_ruling_trigger = yes FROM = { asur_everqueen_ruling_trigger = yes } }
	}

	immediate = {
		if = {
			limit = { asur_everqueen_ruling_trigger = yes }
			save_event_target_as = asur_vale_defender
			FROM = { save_event_target_as = asur_vale_attacker }
		}
		else = {
			save_event_target_as = asur_vale_attacker
			FROM = { save_event_target_as = asur_vale_defender }
		}
		any_independent_ruler = {
			limit = {
				higher_tier_than = COUNT
				trait = creature_elf
				religion = cadai
				NOT = { character = event_target:asur_vale_defender }
				NOT = { character = event_target:asur_vale_attacker }
			}
			character_event = { id = asur.50 days = 15 }
		}
	}
}

# asur.70 Ulthuan begins to fade (hidden)

	id = asur.70
	hide_window = yes

	is_triggered_only = yes # on_startup
	only_playable = yes

	trigger = {
		is_save_game = no
		has_game_rule = { name = warhammer_ulthuan_depopulation value = default }
		has_landed_title = e_ulthuan
		NOT = { has_global_flag = ulthuan_pop_set }
	}

		character_event = { id = asur.71 }

# asur.71 The elven provinces (hidden)
# Marks every elven province of Ulthuan except Lothern, and unmarks those
# that lost their elves.
# Scope: ROOT is an independent ruler (on_five_year_pulse, asur.70).
character_event = {
	id = asur.71
	hide_window = yes

	is_triggered_only = yes # on_five_year_pulse, asur.70
	has_global_flag = ulthuan_pop_set
	only_playable = yes
	only_independent = yes

	trigger = {
		any_realm_province = {
			NOT = { province_id = 1492 }
			OR = {
				AND = {
					region = world_ulthuan
					culture_group = asur_group
					NOT = { has_province_modifier = ulthuan }
				}
				AND = { has_province_modifier = ulthuan NOT = { culture_group = asur_group } }
			}
		}
	}

	immediate = {
		any_realm_province = {
			limit = { NOT = { province_id = 1492 } } # Lothern: newcomers from abroad fill the gaps
			if = {
				limit = {
					region = world_ulthuan
					culture_group = asur_group
					NOT = { has_province_modifier = ulthuan }
				}
				add_province_modifier = { name = ulthuan duration = -1 }
			}
			else_if = {
				limit = { has_province_modifier = ulthuan NOT = { culture_group = asur_group } }
				remove_province_modifier = ulthuan
				remove_province_modifier = ulthuan_lightly_depopulated
				remove_province_modifier = ulthuan_moderately_depopulated
				remove_province_modifier = ulthuan_heavily_depopulated
				remove_province_modifier = ulthuan_severely_depopulated
			}
		}
	}
}
# asur.72 Ulthuan fades
# One of the ruler's elven provinces in Ulthuan falls a step; faster in war
# or under occupation.
# Scope: ROOT is an Asur ruler (on_decade_pulse random list).
character_event = {
	id = asur.72
	picture = "GFX_evt_elf_city"
	# Whether through war or just the slow passage of time, the population in this province has fallen. Such is the fate of all elves in Ulthuan, to slowly diminish, but they will never yield to their enemies even so.
	desc = {
		text = EVTDESCasur72
		trigger = {
			event_target:elven_province = {
				NOR = {
					has_province_modifier = ulthuan_moderately_depopulated
					has_province_modifier = ulthuan_heavily_depopulated
					has_province_modifier = ulthuan_severely_depopulated
				}
			}
		}
	}
	# Whether through war or just the slow passage of time, the population in this province is now moderately depopulated. Empty villages and mansions dot the country side, their former inhabitants now only a distant memory.
	desc = {
		text = EVTDESCasur72b
		trigger = {
			event_target:elven_province = { has_province_modifier = ulthuan_moderately_depopulated }
		}
	}
	# Whether through war or just the slow passage of time, the population in this province is now heavily depopulated. The Lords there struggle to maintain a semblance of civilization, and many areas are now devoid of elves entirely, given over to nature and the wild beasts.
	desc = {
		text = EVTDESCasur72c
		trigger = {
			event_target:elven_province = { has_province_modifier = ulthuan_heavily_depopulated }
		}
	}
	# Whether through war or just the slow passage of time, the population in this province is now severely depopulated. The major noble families still maintain their mansions and the immediate surrounds, but they have lost the ability to control much more than that.
	desc = {
		text = EVTDESCasur72d
		trigger = {
			event_target:elven_province = { has_province_modifier = ulthuan_severely_depopulated }
		}
	}

	has_global_flag = ulthuan_pop_set
	culture_group = asur_group
	only_rulers = yes

	is_triggered_only = yes

	# Select province to depopulate
	immediate = {
		random_demesne_province = {
			limit = {
				location = { region = world_ulthuan }
				has_province_modifier = ulthuan
				NOT = { has_province_modifier = ulthuan_severely_depopulated }
			}
			save_event_target_as = elven_province
		}
	}

	trigger = {
		has_game_rule = { name = warhammer_ulthuan_depopulation value = default }
		higher_tier_than = BARON
		any_demesne_province = {
			AND = {
				location = { region = world_ulthuan }
				has_province_modifier = ulthuan
				NOT = { has_province_modifier = ulthuan_severely_depopulated }
			}
		}
	}

	weight_multiplier = {
		modifier = { factor = 2 liege = { war = yes has_raised_levies = ROOT } }
		#if a province is occupied, the population decline is even faster
		modifier = {
			factor = 2
			any_demesne_title = {
				location = { has_province_modifier = ulthuan }
				tier = BARON
				NOT = { controlled_by = ROOT }
			}
		}
	}

	option = {
		name = EVTOPTAasur72
		event_target:elven_province = {
			trigger_switch = {
				on_trigger = has_province_modifier
				# heavily -> severly
				ulthuan_heavily_depopulated = {
					add_province_modifier = { name = ulthuan_severely_depopulated duration = -1 }
					remove_province_modifier = ulthuan_heavily_depopulated
				}
				# moderatly -> heavily
				ulthuan_moderately_depopulated = {
					add_province_modifier = { name = ulthuan_heavily_depopulated duration = -1 }
					remove_province_modifier = ulthuan_moderately_depopulated
				}
				# lightly -> moderatly
				ulthuan_lightly_depopulated = {
					add_province_modifier = { name = ulthuan_moderately_depopulated duration = -1 }
					remove_province_modifier = ulthuan_lightly_depopulated
				}
			}
			# none -> lightly
			if = {
				limit = {
					NOR = {
						has_province_modifier = ulthuan_lightly_depopulated
						has_province_modifier = ulthuan_moderately_depopulated
						has_province_modifier = ulthuan_heavily_depopulated
						has_province_modifier = ulthuan_severely_depopulated
					}
				}
				add_province_modifier = { name = ulthuan_lightly_depopulated duration = -1 }
			}
		}
	}
}

#
# The Phoenix Guard are the silent temple guard of Asuryan. They keep the
# Shrine of Asuryan on the Isle of Flame and are a holy order the Asur can
# hire (d_phoenix_guard, common/landed_titles/wh_holy_orders.txt). If the
# isle ever falls to a ruler outside the elven faiths, the Guard send a
# written demand to leave it. Refuse, and they rise to take it back.
#
# Replaces DoW_Elven_events.txt (HELF), which needed the Phoenix Guard to
# be a mercenary band and so never fired.
#
# Callers:
#   common/on_actions/wh_on_actions.txt: on_new_holder,
#     on_new_holder_inheritance, on_new_holder_usurpation (1).
###############################################################################

namespace = phoenix

# phoenix.1 The Shrine falls to an outsider (hidden)
# The title flag stops a second demand while one is pending.
# Scope: ROOT is the new holder, FROM the title (on_new_holder and kin).
character_event = {
	id = phoenix.1
	hide_window = yes

	is_triggered_only = yes # on_new_holder, on_new_holder_inheritance, on_new_holder_usurpation

	trigger = {
		FROM = { title = c_isle_of_flame NOT = { has_title_flag = phoenix_demand_sent } }
		NOT = { religion_group = cadai_group }
	}

	immediate = {
		FROM = { set_title_flag = phoenix_demand_sent }

	}

# phoenix.2 The silent demand
# A Phoenix Guard brings a written demand: leave the Isle of Flame. Yield,
# and the isle goes to the Phoenix King (or a warden of the shrine). Refuse,
# and the Guard rise in a religious revolt for the isle.
# Scope: the holder of the Isle of Flame (phoenix.1).
letter_event = {
	id = phoenix.2
	desc = EVTDESCphoenix2

	is_triggered_only = yes # phoenix.1

	trigger = {
		has_landed_title = c_isle_of_flame
		NOT = { religion_group = cadai_group }
	}

	immediate = {
		c_isle_of_flame = { clr_title_flag = phoenix_demand_sent }
	}

	option = { # Leave the isle
		name = EVTOPTAphoenix2
		prestige = -100
		if = {
			limit = {
				e_ulthuan = { has_holder = yes holder_scope = { religion_group = cadai_group } }
			}
			e_ulthuan = {
				holder_scope = {
					c_isle_of_flame = { grant_title_no_opinion = PREV }
					if = {
						limit = { ROOT = { has_landed_title = d_isle_of_flame } }
						d_isle_of_flame = { grant_title_no_opinion = PREV }
					}
				}
			}
		}
		else = {
			create_character = {
				random_traits = yes
				dynasty = random
				religion = cadai
				culture = chracian
				female = no
				age = 140
				trait = creature_elf
				trait = zealous
				attributes = { martial = 10 learning = 8 }
			}
			new_character = {
				c_isle_of_flame = { grant_title_no_opinion = PREV }
				if = {
					limit = { ROOT = { has_landed_title = d_isle_of_flame } }
					d_isle_of_flame = { grant_title_no_opinion = PREV }
				}
				set_defacto_liege = THIS
			}
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 3 trait = craven }
			modifier = { factor = 0.25 trait = brave }
			modifier = { factor = 0.25 higher_tier_than = DUKE }
		}
	}
	option = { # Refuse
		name = EVTOPTBphoenix2
		create_character = {
			random_traits = yes
			dynasty = random
			religion = cadai
			culture = chracian
			female = no
			age = 300
			trait = creature_elf
			trait = zealous
			trait = brave
			trait = skilled_tactician
			attributes = { martial = 14 }
		}
		new_character = {
			set_character_flag = religious_revolter
			create_title = {
				tier = DUKE
				landless = yes
				temporary = yes
				rebel = yes
				culture = chracian
				name = "PHOENIX_GUARD_RISING"
				holder = THIS
			}
			wealth = 200
			c_isle_of_flame = {
				location = {
					PREVPREV = {
						spawn_unit = {
							province = PREV
							owner = THIS
							leader = THIS
							troops = {
								pikemen = { 1500 1500 }
								heavy_infantry = { 1500 1500 }
								archers = { 500 500 }
							}
							attrition = 0.5
						}
					}
				}
			}
			ROOT = {

					reverse_war = {
						target = PREVPREV
						casus_belli = religious_revolt
