# Baptism
# Namespace: baptism (was HF.20000 - HF.20119 in HF_baptism_events;
#   baptism.N = old HF.20N, so HF.20000 is baptism.0)
# Faction: Faiths & doctrines
# Serves: Norscan rulers and their Old World or Chaos sponsors; Old World parents
#
# Mass conversion: a human Norscan ruler asks a human sponsor of an Old World
# faith (or of the Dark Gods, the "dark baptism") to bring the realm into the
# faith. Vassals choose to follow, pretend, or rise as True Believers; the
# sponsor sends priests, gold, temples and missionaries, may visit, and a
# sponsor's priest at court offers new laws, learning, a new name and warriors
# of the faith. Baptism of the heir: Old World parents have their first heir
# blessed by the head of the cult, a temple priest, or, in secret, the cult of
# the Dark Gods, and pick a godparent.
#
# The request and the sponsor's answer   baptism.0 - baptism.2, baptism.24,
#                                          baptism.97 (AI request)
# The great ceremony                     baptism.3 - baptism.8
# Vassals, courtiers, True Believers     baptism.9 - baptism.11,
#                                          baptism.25 - baptism.35
# Safety (death or conversion of either  baptism.12 - baptism.23, baptism.27,
#   side; on_actions)                      baptism.34, baptism.95, baptism.96
# Aftermath and the sponsor's help       baptism.36 - baptism.62
#   (trait changes 39-46, temple 47-51, gold 52-53, missionaries 54-62)
# The sponsor's visit                    baptism.63 - baptism.91
# Notices and forgiveness                baptism.90 - baptism.94, baptism.98,
#                                          baptism.99
# Baptism of the heir                    baptism.100 - baptism.108
#   (head of the cult 100, godparent 101/102, rite 103 - 105, the cult's
#   initiation 106/107, decision version 108)
# The sponsor's priest at court          baptism.110 - baptism.119
#
# Saved: event targets sponsoring_king, converting_king, baptizing_child,
#   scoped_pope, scoped_bishop, scoped_satanist (a cult elder), scoped_baptizer;
#   flags flag_sponsoring_baptism, flag_converting_baptism, flag_baptized_by_*.
# Support: common/scripted_triggers/wh_baptism_triggers.txt,
#   decisions/HF_baptism_decisions.txt, HFP_pregnancy_events (fires 100).
# Loc: localisation/00_WHG_baptism_events.csv
###############################################################################

letter_event = {
	id = baptism.0
	desc = EVTDESCbaptism0
	border = GFX_event_letter_frame_religion
	show_from_from = yes

	is_triggered_only = yes

	option = { #Accept request, start preparation for mass conversion
		name = EVTOPTAbaptism0
		custom_tooltip = { text = accept_baptism_request }
		hidden_tooltip = {
			set_character_flag = flag_sponsoring_baptism
			FROMFROM = { set_character_flag = flag_converting_baptism }
			save_event_target_as = sponsoring_king
			FROMFROM = { save_event_target_as = converting_king }
		}
		FROMFROM = { hidden_tooltip = { letter_event = { id = baptism.2 days = 5 random = 5 } } }
		ai_chance = {
			factor = 100 # Acceptance added in the decision itself.
		}
	}
	option = { #Deny request, end event chain
		name = EVTOPTBbaptism0
		custom_tooltip = { text = refuse_baptism_request }
		FROMFROM = { opinion = { modifier = opinion_denied_request years = 10 who = ROOT } }
		hidden_tooltip = {
			remove_opinion = { who = FROMFROM modifier = opinion_baptism_sponsor_target }
			FROMFROM = { remove_opinion = { who = ROOT modifier = opinion_baptism_sponsor_target } }
		}

		FROMFROM = { hidden_tooltip = { letter_event = { id = baptism.1 days = 5 random = 5 } } }
		ai_chance = { #Only for Multiplayer
			factor = 0
		}
	}
}
