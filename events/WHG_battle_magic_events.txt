###############################################################################
# Battle magic events
# Namespace: battlemagic (was magicengine in wh_new_battle_magic.txt)
# Faction: Generic
#
# The automatic magic of mages, Khorne's daemon summoners, the Red Duke and
# battle priests who lead troops. In battle, every five days each such
# commander spends a point of his magic pool to conjure spell units (and
# daemons or the dead) for his side; the units vanish when the battle ends.
# In a siege, from the second month on, each one wears down the defences
# with a province modifier based on his magic power, at the cost of a point
# of pool for three months.
#
# Battle                                 battlemagic.1 - battlemagic.10
# Siege                                  battlemagic.20 - battlemagic.25
#
# Callers (common/on_actions/wh_magic_on_actions.txt):
#   battlemagic.1 on on_combat_pulse; battlemagic.6 on on_startup,
#   on_yearly_pulse and on_chronicle_start; battlemagic.10 on the battle won
#   and lost on_actions and on_new_holder_inheritance; battlemagic.20 on
#   on_unit_entering_province, on_command_unit and on_command_sub_unit;
#   battlemagic.21 on on_unit_entering_province.
# Support: battle_magic_casting_effect and calc_magic_pool_effect in
#   common/scripted_effects/wh_magic_scripted_effects.txt, triggers in
#   common/scripted_triggers/wh_battle_magic_triggers.txt.
# Loc: localisation/00_WHG_battle_magic_events.csv
###############################################################################

namespace = battlemagic

###############################################################################
# Battle
###############################################################################

# battlemagic.1 The battle begins (hidden)
# The commander, and every casting flank commander of his realm in the same
# province, fills his magic pool and starts casting.
# Scope: ROOT is a commander in battle (on_combat_pulse).
character_event = {
	id = battlemagic.1
	hide_window = yes

	is_triggered_only = yes # on_combat_pulse

	trigger = {
		is_alive = yes
		NOT = { has_character_flag = started_battle_casting }
	}

	immediate = {
		if = {
			limit = { has_character_flag = battle_magic_system_flag in_command = yes }
			set_character_flag = started_battle_casting
			calc_magic_pool_effect = yes
			character_event = { id = battlemagic.2 }
			character_event = { id = battlemagic.3 }
			character_event = { id = battlemagic.4 }
		}
		location = {
			any_province_character = {
				limit = {
					has_character_flag = battle_magic_system_flag
					NOT = { has_character_flag = started_battle_casting }
					in_command = yes
					in_battle = yes
					same_realm = ROOT
				}
				set_character_flag = started_battle_casting
				calc_magic_pool_effect = yes
				character_event = { id = battlemagic.2 }
				character_event = { id = battlemagic.3 }
				character_event = { id = battlemagic.4 }
			}
		}
	}
}

# battlemagic.2 Casting (hidden)
# Every five days of battle the caster spends a point of his pool on spell
# units. The first time a player's mage casts, the player is told how
# battle magic works (battlemagic.5).
# Scope: ROOT is the caster (battlemagic.1, then itself).
character_event = {
	id = battlemagic.2
	hide_window = yes

	is_triggered_only = yes # battlemagic.1
	has_character_flag = battle_magic_system_flag

	trigger = {
		is_alive = yes
		in_battle = yes
		combat = { is_retreating = no }
	}

	immediate = {
		if = {
			limit = { ai = no NOT = { has_character_flag = battle_magic_explained } }
			set_character_flag = battle_magic_explained
			narrative_event = { id = battlemagic.5 }
		}
		else_if = {
			limit = {
				ai = yes
				liege = { ai = no NOT = { has_character_flag = battle_magic_explained } }
			}
			liege = {
				set_character_flag = battle_magic_explained
				narrative_event = { id = battlemagic.5 }
			}
		}
		battle_magic_casting_effect = yes
		# on_combat_pulse is too slow, so the chain times itself
		repeat_event = { id = battlemagic.2 days = 5 }
	}
}

# battlemagic.3 The battle is over (hidden)
# Checks every day until the caster is out of battle, then ends his
# casting and refills his pool for the next one.
# Scope: ROOT is the caster (battlemagic.1, then itself).
character_event = {
	id = battlemagic.3
	hide_window = yes

	is_triggered_only = yes # battlemagic.1
	has_character_flag = started_battle_casting

	trigger = {
		in_battle = no
	}

	fail_trigger_effect = { repeat_event = { id = battlemagic.3 days = 1 } }

	immediate = {
		clr_character_flag = started_battle_casting
		calc_magic_pool_effect = yes
	}
}

# battlemagic.4 Magic on the battlefield
# Tells a player that his commander has begun casting.
# Scope: ROOT is the caster (battlemagic.1).
character_event = {
	id = battlemagic.4
	desc = {
		text = EVTDESCbattlemagic4
		trigger = {
			NOT = { summons_daemons_in_battle_trigger = yes }
			NOT = { summons_undead_in_battle_trigger = yes }
		}
	}
	desc = {
		text = EVTDESCbattlemagic4b
		trigger = {
			summons_daemons_in_battle_trigger = yes
			NOT = { khorne_summoner_trigger = yes }
		}
	}
	desc = { text = EVTDESCbattlemagic4c trigger = { khorne_summoner_trigger = yes } }
	desc = {
		text = EVTDESCbattlemagic4d
		trigger = {
			summons_undead_in_battle_trigger = yes
			NOT = { summons_daemons_in_battle_trigger = yes }
		}
	}
	picture = GFX_evt_mage_battle_magic

	is_triggered_only = yes # battlemagic.1
	notification = yes
	ai = no

	option = {
		name = EVTOPTAbattlemagic4
	}
}

# battlemagic.6 Ready for war (hidden)
# Marks the characters who cast in battle and keeps their magic pool up to
# date for the trait text. A trained mage without a power level gets one.
# Scope: ROOT is any character (on_startup, on_yearly_pulse,
# on_chronicle_start).
character_event = {
	id = battlemagic.6
	hide_window = yes

	is_triggered_only = yes # on_startup, on_yearly_pulse, on_chronicle_start
	min_age = 16
	only_capable = yes

	trigger = {
		battle_magic_system_trigger = yes
	}

	immediate = {
		set_character_flag = battle_magic_system_flag
		calc_magic_pool_effect = yes
		if = {
			limit = { is_trained_mage_trigger = yes NOT = { has_magic_power_trigger = yes } }
			magic_power_random_effect = yes
		}
	}
}

# battlemagic.10 The spells fade (hidden)
# Once none of his casters is still fighting, the ruler's spell units
# vanish. Spell units are always owned by a ruler: the caster or his liege.
# Scope: ROOT is a ruler (battle won and lost on_actions,
# on_new_holder_inheritance).
character_event = {
	id = battlemagic.10
	hide_window = yes

	is_triggered_only = yes # battle on_actions, on_new_holder_inheritance
	only_rulers = yes

	trigger = {
		is_alive = yes
	}

	immediate = {
		if = {
			limit = {
				NOT = { AND = { has_character_flag = battle_magic_system_flag in_battle = yes } }
				NOT = {
					any_courtier = { has_character_flag = battle_magic_system_flag in_battle = yes }
				}
			}
			disband_event_forces = magical_spells
		}
		else = { repeat_event = { id = battlemagic.10 days = 5 } }
	}
}

# battlemagic.20 On the march (hidden)
# A caster who enters a province or takes command starts the siege check
# (battlemagic.22) the next day. One who no longer casts loses the mark.
# Scope: ROOT is a commander (on_unit_entering_province, on_command_unit,
# on_command_sub_unit).
character_event = {
	id = battlemagic.20
	hide_window = yes

	is_triggered_only = yes # unit and command on_actions
	has_character_flag = battle_magic_system_flag

	immediate = {
		if = {
			limit = { NOT = { battle_magic_system_trigger = yes } }
			clr_character_flag = battle_magic_system_flag
		}
		else = { character_event = { id = battlemagic.22 days = 1 } }
	}
}

# battlemagic.21 New ground (hidden)
# Entering a new province lifts the siege timer, so a new siege can start.
# Scope: ROOT is a commander (on_unit_entering_province).
character_event = {
	id = battlemagic.21
	hide_window = yes

	is_triggered_only = yes # on_unit_entering_province
	has_character_flag = battle_magic_system_flag

	trigger = {
		has_character_modifier = magic_siege_timer
	}

	immediate = {
		remove_character_modifier = magic_siege_timer
	}
}

# battlemagic.22 A siege may begin (hidden)
# Starts the monthly siege chain (battlemagic.23) for this province, unless
# one is already running here.
# Scope: ROOT is the caster (battlemagic.20).
character_event = {
	id = battlemagic.22
	hide_window = yes

	is_triggered_only = yes # battlemagic.20
	has_character_flag = battle_magic_system_flag

	trigger = {
		NOT = { has_character_modifier = magic_siege_timer }
		# No resetting the count by swapping commanders
		location = {
			NOR = {
				has_province_modifier = magic_siege1
				has_province_modifier = magic_siege2
				has_province_modifier = magic_siege3
				has_province_modifier = magic_siege4
			}
		}
	}

	immediate = {
		# The chain stops if the caster moves on
		location = { save_event_target_as = siege_location }
		add_character_modifier = { name = magic_siege_timer hidden = yes duration = 30 }
		character_event = { id = battlemagic.23 days = 1 }
	}
}

# battlemagic.23 Siege magic (hidden)
# Each month the caster stays in a hostile province, from the second month
# on, he wears down its defences: a siege modifier as strong as his magic
# power (Khorne's summoners and the Red Duke count as power 2), at the cost
# of a point of his battle magic pool for three months.
# Scope: ROOT is the caster (battlemagic.22, then itself).
character_event = {
	id = battlemagic.23
	hide_window = yes

	is_triggered_only = yes # battlemagic.22
	has_character_flag = battle_magic_system_flag

	trigger = {
		at_location = event_target:siege_location
		location = {
			NOT = { owner = { same_realm = ROOT } }
			# At war with the province's realm, or raiding it
			OR = {
				owner = { any_liege = { war_with = ROOT } }
				owner = { war_with = ROOT }
				owner = { ROOT = { any_liege = { war_with = PREVPREV } } }
				owner = { any_liege = { ROOT = { any_liege = { war_with = PREVPREV } } } }
			}
		}
	}

	immediate = {
		# Variables named local_ belong to this event chain only
		change_variable = { which = local_siegemonths value = 1 }
		if = {
			limit = {
				check_variable = { which = local_siegemonths value = 2 }
				battle_magic_system_trigger = yes
			}
			if = {
				limit = { NOT = { has_character_modifier = magic_siege_pool_reduction_timer } }
				character_event = { id = battlemagic.24 }
			}
			magic_xp_effect = yes
			add_character_modifier = {
				name = magic_siege_pool_reduction_timer
				hidden = yes
				duration = 90
			}
			calc_magic_pool_effect = yes
			set_variable = { which = local_siegepower which = cl }
			if = {
				limit = {
					OR = { khorne_summoner_trigger = yes summons_undead_in_battle_trigger = yes }
					NOT = { check_variable = { which = local_siegepower value = 2 } }
				}
				set_variable = { which = local_siegepower value = 2 }
			}
			if = {
				limit = { check_variable = { which = local_siegepower value = 4 } }
				location = {
					add_province_modifier = { name = magic_siege4 duration = 150 stacking = yes }
				}
			}
			else_if = {
				limit = { check_variable = { which = local_siegepower value = 3 } }
				location = {
					add_province_modifier = { name = magic_siege3 duration = 120 stacking = yes }
				}
			}
			else_if = {
				limit = { check_variable = { which = local_siegepower value = 2 } }
				location = {
					add_province_modifier = { name = magic_siege2 duration = 90 stacking = yes }
				}
			}
			else_if = {
				limit = { check_variable = { which = local_siegepower value = 1 } }
				location = {
					add_province_modifier = { name = magic_siege1 duration = 60 stacking = yes }
				}
			}
			if = {
				limit = { ai = no NOT = { has_character_flag = siege_magic_explained } }
				set_character_flag = siege_magic_explained
				narrative_event = { id = battlemagic.25 }
			}
			else_if = {
				limit = {
					ai = yes
					liege = { ai = no NOT = { has_character_flag = siege_magic_explained } }
				}
				liege = {
					set_character_flag = siege_magic_explained
					narrative_event = { id = battlemagic.25 }
				}
			}
		}
		repeat_event = { id = battlemagic.23 days = 30 }
	}
}
