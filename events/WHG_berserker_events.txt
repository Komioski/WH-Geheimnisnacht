###############################################################################
# Berserkers
# Namespace: berserk (was TOG.3000 - TOG.3009 in oldgods_various_events.txt)
# Faction: Norsca
# Serves: Norscans, Norse dwarfs, Ulricans, the men of Kislev and all who keep
#   the old Norscan gods
#
# In the press of battle a wrathful warrior may be taken by the rage: the
# wolf-spirit of Ulric, the bear of Ursun, the fury of Grimnir, the howling
# gods of the north. A berserker hews through scores of foes, takes a spear
# or an arrow without noticing, or is cut off and dragged down. At home the
# rage may rise against a tiresome courtier. One day the body can no longer
# bear the rage, and the berserker grows old.
#
# The rage awakens                     berserk.1
# In battle                            berserk.2 - berserk.5
# Too old for the rage                 berserk.6
# The rage at court                    berserk.10 - berserk.11
#
# Fired from: the battle tombola battle.1 in WHG_battle_events.txt (berserk.1
#   - berserk.5; event targets home_commander, enemy_commander, enemy_liege),
#   on_bi_yearly_pulse in 00_on_actions.txt (berserk.6, and berserk.10 in the
#   random list). Shared triggers are in WHG_old_gods_triggers.txt; the
#   berserker's spirit speaks in WHG_old_gods_voices.txt (GetBerserkerSpirit).
#   The Khornate berserkers of m_Khorne.txt (namespace berserker) are separate.
# Saved: event target berserk_courtier; flags berserk_rage_lost,
#   berserk_rage_held (cleared by berserk.11).
###############################################################################

namespace = berserk

###############################################################################
# The rage awakens
###############################################################################

# berserk.1 The rage awakens
# A wrathful or aggressive warrior of the berserker peoples, young enough for
# the rage. A coward is cured of it.
# Scope: ROOT is home_commander.
character_event = {
	id = berserk.1
	desc = EVTDESCberserk1
	picture = GFX_evt_viking_battle_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	trigger = {
		berserk_can_rage_trigger = yes
		NOT = { trait = berserker }
		OR = {
			trait = wroth
			trait = aggressive_leader
		}
	}

	option = {
		name = EVTOPTAberserk1
		if = {
			limit = { trait = craven }
			remove_trait = craven
		}
		add_trait = berserker
		allow_new_battle_events_effect = yes
	}
}

# berserk.2 Maimed in the rage
# Scope: ROOT is home_commander.
character_event = {
	id = berserk.2
	desc = EVTDESCberserk2
	picture = GFX_evt_viking_battle_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	trigger = {
		trait = berserker
		is_alive = yes
		NOT = { trait = maimed }
		NOT = { trait = one_legged }
	}

	option = {
		name = EVTOPTAberserk2
		if = {
			limit = { has_dlc = "Reapers" }
			add_trait = one_legged
			add_trait = severely_injured
			hidden_effect = {
				character_event = { id = maimed.3 } # Notify one-legged
				resolve_severely_injured_effect = yes
			}
		}
		else = {
			add_trait = maimed
			hidden_effect = { character_event = { id = 38281 } }
		}
		allow_new_battle_events_effect = yes
	}
}

# berserk.4 Cut off and dragged down
# The berserker dies, slain by the enemy commander when there is one. The
# home liege (battle.11) and the enemy liege or commander (battle.14) hear of
# it, as for any fallen commander.
# Scope: ROOT is home_commander.
character_event = {
	id = berserk.4
	desc = EVTDESCberserk4
	picture = GFX_evt_viking_battle_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	trigger = {
		trait = berserker
		is_alive = yes
		is_invincible_trigger = no
	}

	immediate = {
		hidden_effect = {
			liege = {
				if = {
					limit = { NOT = { character = ROOT } }
					save_event_target_as = home_liege
					character_event = { id = battle.11 }
				}
			}
			event_target:enemy_liege = {
				if = {
					limit = {
						NOR = {
							is_vassal_or_below = ROOT
							character = ROOT
						}
					}
					set_character_flag = temp_battle_kill_242
					character_event = { id = battle.14 }
				}
			}
			event_target:enemy_commander = {
				if = {
					limit = {
						NOR = {
							has_character_flag = temp_battle_kill_242
							is_vassal_or_below = ROOT
							character = ROOT
						}
					}
					character_event = { id = battle.14 }
				}
			}
			event_target:enemy_liege = { clr_character_flag = temp_battle_kill_242 }
		}
	}

	option = {
		name = EVTOPTAberserk4
		if = {
			limit = {
				event_target:enemy_commander = {
					is_alive = yes
					NOT = { character = ROOT }
				}
			}
			death = {
				death_reason = death_battle
				killer = event_target:enemy_commander
			}
		}
		else = {
			death = { death_reason = death_battle }
		}
	}
}

###############################################################################
# The rage at court
###############################################################################

# berserk.10 The rage rises
# A courtier who thinks little of the berserker drones on until the rage
# rises. The berserker fights it down, or not: patience and temperance help,
# wrath does not.
# Scope: ROOT is the berserker. From the on_bi_yearly_pulse random list.
character_event = {
	id = berserk.10
	desc = EVTDESCberserk10
	picture = GFX_evt_viking_throneroom_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	trigger = {
		trait = berserker
		prisoner = no
		is_inaccessible_trigger = no
		NOT = { trait = incapable }
		any_courtier = { berserk_tiresome_courtier_trigger = yes }
	}

	immediate = {
		random_courtier = {
			limit = { berserk_tiresome_courtier_trigger = yes }
			save_event_target_as = berserk_courtier
		}
	}

	option = {
		name = EVTOPTAberserk10
		random_list = {
			50 = {
				modifier = { factor = 1.5 trait = wroth }
				modifier = { factor = 1.5 trait = drunkard }
				modifier = { factor = 0.5 trait = patient }
				modifier = { factor = 0.5 trait = temperate }
				set_character_flag = berserk_rage_lost
				custom_tooltip = { text = EVTTOOLTIPberserk11a }
			}
			50 = {
				set_character_flag = berserk_rage_held
				custom_tooltip = { text = EVTTOOLTIPberserk11b }
			}
		}
		character_event = { id = berserk.11 days = 1 }
	}
}

# berserk.11 After the rage
# The courtier is beaten bloody and remembers it, or the table takes the
# blows and the gods see the berserker's will.
# Scope: ROOT is the berserker; berserk_courtier the courtier.
character_event = {
	id = berserk.11
	desc = {
		trigger = { has_character_flag = berserk_rage_lost }
		text = EVTDESCberserk11_lost
	}
	desc = {
		trigger = { has_character_flag = berserk_rage_held }
		text = EVTDESCberserk11_held
	}
	picture = GFX_evt_viking_throneroom_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	option = {
		name = EVTOPTAberserk11_lost
		trigger = { has_character_flag = berserk_rage_lost }
		event_target:berserk_courtier = {
			add_trait = wounded
			opinion = { modifier = opinion_went_berserk who = ROOT }
		}
	}
	option = {
		name = EVTOPTAberserk11_held
		trigger = { has_character_flag = berserk_rage_held }
		piety = 20
	}

	after = {
		clr_character_flag = berserk_rage_lost
		clr_character_flag = berserk_rage_held
	}
}
