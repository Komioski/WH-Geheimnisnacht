# The Black Crusade
# Namespace: bcrusade (was blckcs in wh_black_crusades.txt, and HFP.41000 -
#   HFP.41200 in HFP_crusade_events.txt: bcrusade.N = HFP.41NNN minus 40990,
#   except HFP.41200 -> bcrusade.25, HFP.41199 -> 190 and HFP.41143-41147
#   -> 151-154). Part 2 is HFP_crusade_flavour.txt: HFP.42100 -> 200,
#   42101 -> 201, 42001 -> 202, 42003 -> 203, 42002 -> 204, 42200 -> 210,
#   the Shepherds' Crusade HFP.42300-42330 -> the Penitent Host 300-316.
# Namespace: crusadefl (campaign flavour, HFP.422NN -> crusadefl.NN)
# Faction: Chaos
# Serves: the five Chaos faiths (Undivided, Khorne, Nurgle, Slaanesh,
#   Tzeentch), the only faiths in the mod that call a new-style crusade;
#   the Forces of Order who answer it
#
# The Chaos faiths have no religious head. A great lord takes the right to
# call a Black Crusade by decision (call_black_crusade in
# wh_chaos_decisions.txt), which makes his primary title the faith's head
# for a time. He has ten years to call it. The engine below runs the war:
# who gets the conquered land, pledges and broken pledges, the war chest,
# the defenders who rally to the target, and the warlord's letters to lords
# who will not march or will not fight. When it ends, that god may not call
# another for thirty years (a hundred after a low-doom crusade).
#
# Calling the Black Crusade               bcrusade.1 - bcrusade.6
# The chosen lord of the target lands     bcrusade.10 - bcrusade.12
# Preparation and launch                  bcrusade.13 - bcrusade.25
# Debts called in                         bcrusade.30 - bcrusade.32
# Pledging                                bcrusade.40 - bcrusade.45
# Breaking the pledge                     bcrusade.50 - bcrusade.55
# Letters, beneficiaries, favour          bcrusade.60 - bcrusade.67
# A beneficiary who stayed behind         bcrusade.70 - bcrusade.71
# The defenders gather                    bcrusade.80 - bcrusade.85
# The war chest                           bcrusade.90 - bcrusade.98
# After the war, the stance               bcrusade.100 - bcrusade.101
# Beneficiary lost notices                bcrusade.120 - bcrusade.128
# The chosen lord changes                 bcrusade.130 - bcrusade.131
# Lords who did not pledge                bcrusade.140 - bcrusade.141
# Lords who pledged and did not fight     bcrusade.150 - bcrusade.154
# Victory, dynasty and clean-up           bcrusade.170 - bcrusade.190
# Part 2
# Carving a dominion                      bcrusade.200 - bcrusade.204
# The monthly campaign roll               bcrusade.210
# On campaign (also the Errantry War and
#   the Reconquest)                       crusadefl.1 - crusadefl.15
# The Penitent Host                       bcrusade.300 - bcrusade.316
#
# Fired from: the crusade hooks in common/on_actions/00_on_actions.txt
#   (on_crusade_preparation_starts, _launches, _canceled, _monthly,
#   _target_changes, on_pledge/unpledge_crusade_participation), on_death,
#   on_new_holder, the battle and siege hooks; bcrusade.2 and .4 from
#   on_yearly_pulse in wh_on_actions.txt; the new_crusade CBs in
#   00_cb_types.txt; the crusade effects in
#   04_scripted_effects_crusades.txt.
# Saved: title flags temp_religion_head (the warlord's primary title),
#   <god>_crusade_timer and <god>_low_doom_crusade_timer on
#   k_cult_of_sigmar, crusade_inherently_chaos / crusade_fringe_chaos /
#   crusade_outside_chaos / crusaded_against_before on the target title;
#   global flag self_declared_black_crusade; character flags
#   must_declare_black_crusade, low_doom_black_crusade,
#   black_crusade_started, cant_target_for_blackcrusade,
#   forces_of_order_defender, previous_defender_against_black_crusade, the
#   crusade_*_stance and *_beneficiary_flag flags.
# Decisions: decisions/WHG_black_crusade_decisions.txt
# Scripted triggers: common/scripted_triggers/03_crusades.txt
# Scripted effects: common/scripted_effects/04_scripted_effects_crusades.txt,
#   wh_black_crusade_effects.txt
# Loc: localisation/00_WHG_black_crusade_events.csv; launch and stance
#   texts in 00_customizable_localisation.txt (GetCrusadeLaunchFirstPart)

# bcrusade.1 The call is made
# Ten years after a lord claimed the right to call a Black Crusade: if
# no war was declared, the claim lapses (bcrusade.2).
# Scope: ROOT is the lord who took the decision. Fired by the call
# decision in wh_chaos_decisions.txt, 3650 days later.
character_event = {
	id = bcrusade.1
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		has_character_flag = must_declare_black_crusade
		NOT = { any_war = { using_cb = new_crusade } }
	}

	immediate = {
		character_event = { id = bcrusade.2 }
	}
}

# bcrusade.2 The claim lapses
# Hidden. A lord who holds the right to call a Black Crusade (title flag
# temp_religion_head) loses it again, unless he claimed it less than ten
# years ago, is preparing a Black Crusade or is fighting one.
# Scope: ROOT is a Chaos ruler. Fired from on_yearly_pulse and on_death
# (wh_on_actions), by bcrusade.1, bcrusade.22 and the CBs when the war ends.
character_event = {
	id = bcrusade.2
	is_triggered_only = yes
	hide_window = yes

	religion_group = chaos_gods_group

	trigger = {
		any_demesne_title = { has_title_flag = temp_religion_head }
		NOR = {
			AND = {
				has_character_flag = must_declare_black_crusade
				NOT = { had_character_flag = { flag = must_declare_black_crusade days = 3650 } }
			}
			is_preparing_crusade = yes
			any_war = { using_cb = new_crusade }
		}
	}

	immediate = {
		any_demesne_title = {
			limit = { has_title_flag = temp_religion_head }
			set_religion_head = none
			ROOT = { clr_global_flag = self_declared_black_crusade }
		}
	}
}

# bcrusade.3 The cooldown begins
# Hidden. When a Black Crusade ends, that god may not call another for
# thirty years, or a hundred for a low-doom crusade. The timers are
# title flags on k_cult_of_sigmar.
# Scope: ROOT is the warlord who called it. Fired by bcrusade.22 and the
# new_crusade CBs in 00_cb_types.
character_event = {
	id = bcrusade.3
	is_triggered_only = yes
	hide_window = yes

	trigger = { }

	immediate = {
		if = { limit = { religion = chaos } k_cult_of_sigmar = { set_flag = chaos_crusade_timer } }
		else_if = {
			limit = { religion = tzeentch }
			k_cult_of_sigmar = { set_flag = tzeentch_crusade_timer }
		}
		else_if = {
			limit = { religion = nurgle }
			k_cult_of_sigmar = { set_flag = nurgle_crusade_timer }
		}
		else_if = {
			limit = { religion = slaanesh }
			k_cult_of_sigmar = { set_flag = slaanesh_crusade_timer }
		}
		else_if = {
			limit = { religion = khorne }
			k_cult_of_sigmar = { set_flag = khorne_crusade_timer }
		}
		#Low Doom Crusade time, if chaos is in real trouble then it may not be set
		if = {
			limit = { has_character_flag = low_doom_black_crusade core_chaos_realms_occupied = no }
			if = {
				limit = { religion = chaos }
				k_cult_of_sigmar = { set_flag = chaos_low_doom_crusade_timer }
			}
			else_if = {
				limit = { religion = tzeentch }
				k_cult_of_sigmar = { set_flag = tzeentch_low_doom_crusade_timer }
			}
			else_if = {
				limit = { religion = nurgle }
				k_cult_of_sigmar = { set_flag = nurgle_low_doom_crusade_timer }
			}
			else_if = {
				limit = { religion = slaanesh }
				k_cult_of_sigmar = { set_flag = slaanesh_low_doom_crusade_timer }
			}
			else_if = {
				limit = { religion = khorne }
				k_cult_of_sigmar = { set_flag = khorne_low_doom_crusade_timer }
			}
		}
	}
}

# bcrusade.4 The cooldown ends
# Hidden. Clears Black Crusade timers older than thirty (or a hundred)
# years. can_call_black_crusade also treats such old timers as expired,
# so a Black Crusade can come again even with no Grand Theogonist.
# Scope: ROOT is the Grand Theogonist. Fired from on_yearly_pulse.
character_event = {
	id = bcrusade.4
	is_triggered_only = yes
	hide_window = yes

	religion = sigmarite

	trigger = {
		controls_religion = yes
	}

	immediate = {
		k_cult_of_sigmar = {
			if = {
				limit = { had_title_flag = { flag = chaos_crusade_timer days = 10950 } }
				clr_title_flag = chaos_crusade_timer
			}
			else_if = {
				limit = { had_title_flag = { flag = tzeentch_crusade_timer days = 10950 } }
				clr_title_flag = tzeentch_crusade_timer
			}
			else_if = {
				limit = { had_title_flag = { flag = nurgle_crusade_timer days = 10950 } }
				clr_title_flag = nurgle_crusade_timer
			}
			else_if = {
				limit = { had_title_flag = { flag = slaanesh_crusade_timer days = 10950 } }
				clr_title_flag = slaanesh_crusade_timer
			}
			else_if = {
				limit = { had_title_flag = { flag = khorne_crusade_timer days = 10950 } }
				clr_title_flag = khorne_crusade_timer
			}

			#Low Doom Crusades

			if = {
				limit = { had_title_flag = { flag = chaos_low_doom_crusade_timer days = 36500 } }
				clr_title_flag = chaos_low_doom_crusade_timer
			}
			else_if = {
				limit = { had_title_flag = { flag = tzeentch_low_doom_crusade_timer days = 36500 } }
				clr_title_flag = tzeentch_low_doom_crusade_timer
			}
			else_if = {
				limit = { had_title_flag = { flag = nurgle_low_doom_crusade_timer days = 36500 } }
				clr_title_flag = nurgle_low_doom_crusade_timer
			}
			else_if = {
				limit = { had_title_flag = { flag = slaanesh_low_doom_crusade_timer days = 36500 } }
				clr_title_flag = slaanesh_low_doom_crusade_timer
			}
			else_if = {
				limit = { had_title_flag = { flag = khorne_low_doom_crusade_timer days = 36500 } }
				clr_title_flag = khorne_low_doom_crusade_timer
			}
		}
	}
}

# bcrusade.5 Other lords are off limits
# Hidden. While a Black Crusade runs, every other independent ruler is
# flagged so the call CB does not show against them.
# Scope: ROOT is the warlord. Fired from on_crusade_launches and
# on_crusade_monthly.
character_event = {
	id = bcrusade.5
	is_triggered_only = yes
	hide_window = yes

	trigger = { }

	immediate = {
		crusade_target_char = { save_event_target_as = crusade_target }
		any_independent_ruler = {
			limit = { NOT = { character = event_target:crusade_target } }
			set_character_flag = cant_target_for_blackcrusade
		}
	}
}

# bcrusade.10 The chosen lord
# Hidden. Picks the official recipient of the target lands: the faithful
# holder, his heir, or a claimant the warlord likes. With nobody fit,
# pledged lords fall back to their own beneficiaries, and the horde is
# told by letter (bcrusade.130, .131).
# Scope: ROOT is the warlord. Fired from on_crusade_preparation_starts and
# on_crusade_target_changes, and by bcrusade.11 and .12.
character_event = {
	id = bcrusade.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		uses_new_crusade = yes
		religion_group = chaos_gods_group
		OR = { is_preparing_crusade = yes any_war = { using_cb = new_crusade } }
	}

	immediate = {
		crusade_target_title = { save_event_target_as = current_crusade_target_title }
		official_crusade_recipient = { save_event_target_as = old_crusade_recipient }
		#There is a King fitting of the title
		if = {
			limit = {
				crusade_target_title = {
					owner = {
						religion = ROOT
						NOR = { trait = excommunicated trait = incapable is_dying = yes }
					}
				}
			}
			crusade_target_title = {
				owner = {
					save_event_target_as = new_crusade_recipient
					set_official_crusade_recipient = THIS
				}
			}
		}
		#There is an heir to the King fitting the title
		else_if = {
			limit = {
				crusade_target_title = {
					owner = {
						religion = ROOT
						NOR = { trait = excommunicated trait = incapable }
						is_dying = yes
						current_heir = { any_heir_title = { title = PREVPREVPREV } }
					}
				}
			}
			crusade_target_title = {
				owner = {
					current_heir = {
						save_event_target_as = new_crusade_recipient
						set_official_crusade_recipient = THIS
					}
				}
			}
		}
		#There is a Claimant chaos approves of
		else_if = {
			limit = {
				crusade_target_title = {
					any_claimant = {
						religion = ROOT
						NOR = { trait = excommunicated trait = incapable }
						ROOT = { opinion = { who = PREV value = 0 } }
						NOT = { is_dying = yes }
					}
				}
			}
			random_character = {
				limit = {
					is_alive = yes
					religion = ROOT
					crusade_target_title = { claimed_by = PREV }
					NOT = { trait = excommunicated }
					ROOT = { opinion = { who = PREV value = 1 } }
					NOT = { is_dying = yes }
				}
				preferred_limit = { ROOT = { opinion = { who = PREV value = 90 } } is_female = no }
				preferred_limit = { ROOT = { opinion = { who = PREV value = 75 } } is_female = no }
				preferred_limit = { ROOT = { opinion = { who = PREV value = 50 } } is_female = no }
				preferred_limit = { ROOT = { opinion = { who = PREV value = 25 } } is_female = no }
				preferred_limit = { is_female = no }
				preferred_limit = { ROOT = { opinion = { who = PREV value = 90 } } is_female = yes }
				preferred_limit = { ROOT = { opinion = { who = PREV value = 75 } } is_female = yes }
				preferred_limit = { ROOT = { opinion = { who = PREV value = 50 } } is_female = yes }
				preferred_limit = { ROOT = { opinion = { who = PREV value = 25 } } is_female = yes }
				save_event_target_as = new_crusade_recipient
				set_official_crusade_recipient = THIS
			}
		}
		if = {
			limit = { NOT = { event_target:new_crusade_recipient = { always = yes } } }
			set_official_crusade_recipient = NONE
			any_crusade_participant = {
				limit = { NOT = { has_character_flag = crusade_selfish_stance } holy_order = no }
				clr_character_flag = crusade_papal_stance
				set_character_flag = crusade_beneficiary_stance
				if = {
					limit = {
						any_owned_bloodline = {
							has_bloodline_flag = grand_crusader_bloodline
							bloodline_is_active_for = PREV
						}
					}
					set_crusade_pot_multiplier = 1.2
				}
				else = { set_crusade_pot_multiplier = 1 }
			}
		}
		#Sending out letters to the Crusaders
		if = {
			limit = {
				event_target:old_crusade_recipient = { always = yes }
				event_target:new_crusade_recipient = { always = yes }
				event_target:old_crusade_recipient = {
					NOT = { character = event_target:new_crusade_recipient }
				}
			}
			any_crusade_participant = { letter_event = { id = bcrusade.130 } }
		}
		else_if = {
			limit = {
				event_target:old_crusade_recipient = { always = yes }
				NOT = { event_target:new_crusade_recipient = { always = yes } }
			}
			any_crusade_participant = { letter_event = { id = bcrusade.131 } }
		}
	}
}

# bcrusade.15 The first pledges
# Hidden. AI rulers of the faith may pledge at the start, more often the
# zealous, the great and those whose lands lie in the target realm.
# Scope: ROOT is the warlord. Fired from on_crusade_preparation_starts.
character_event = {
	id = bcrusade.15
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		uses_new_crusade = yes
		holy_order = no
	}

	immediate = {
		crusade_target_title = {
			any_title = {
				limit = { de_jure_vassal_or_below = PREV }
				save_event_target_as = crusade_target_empire_liege
			}
		}
		any_playable_ruler = {
			limit = {
				age = 16
				religion = ROOT
				check_if_alternate_crusades_trigger = yes
				OR = {
					independent = yes
					higher_tier_than = COUNT
					OR = { holy_order = yes is_landed = yes }
				}
				rightful_religious_head_scope = { character = ROOT NOT = { war_with = PREV } }
				mercenary = no
				is_within_diplo_range = ROOT
			}
			random = {
				chance = 20
				modifier = { factor = 1.5 trait = zealous }
				modifier = { factor = 1.5 trait = holy_warrior }
				modifier = { factor = 1.25 ai_zeal = 0 }
				modifier = { factor = 1.5 ai_zeal = 10 }

# bcrusade.19 Black Crusade for the target
# The launch, seen by every ruler. The text says whether the target was
# once Chaos land (the Wastes, Norsca), on its fringe (the Darklands)
# or never held. Lords who promised the Forces of Order their help are
# called to the defence (bcrusade.25).
# Scope: ROOT is any ruler, FROM the warlord. Fired by bcrusade.18.
narrative_event = {
	id = bcrusade.19
	desc = EVTDESCbcrusade19
	title = EVTTITLEbcrusade19
	border = GFX_event_narrative_frame_religion
	picture = GFX_evt_chaos_everchosen
	is_triggered_only = yes

	trigger = {
		FROM = { is_preparing_crusade = yes }
	}

	immediate = {
		FROM = {
			crusade_target_title = { save_event_target_as = crusade_title }
			crusade_target_char = { save_event_target_as = crusade_character }
		}

		if = {
			limit = {
				OR = {
					event_target:crusade_title = { region = world_chaos_wastes }
					event_target:crusade_title = { region = world_norsca }
				}
			}
			event_target:crusade_title = { set_title_flag = crusade_inherently_chaos }
		}
		if = {
			limit = { OR = { event_target:crusade_title = { region = world_darklands } } }
			event_target:crusade_title = { set_title_flag = crusade_fringe_chaos }
		}

		if = {
			limit = {
				NOT = {
					event_target:crusade_title = { has_title_flag = crusade_inherently_chaos }
					event_target:crusade_title = { has_title_flag = crusade_fringe_chaos }
				}
			}
			event_target:crusade_title = { set_title_flag = crusade_outside_chaos }
		}
	}

	option = {
		name = EVTOPTAbcrusade19
		custom_tooltip = { text = EVTOPTAbcrusade19_TT }

		trigger = { religion_group = chaos_gods_group }
		if = { limit = { religion = FROM has_pledged_crusade_participation = yes } piety = 100 }
	}

	option = {
		name = EVTOPTBbcrusade19
		custom_tooltip = { text = EVTOPTAbcrusade19_TT }

		trigger = { NOT = { religion_group = chaos_gods_group } }
	}

	after = {
		crusade_target_title = { set_title_flag = crusaded_against_before }
		clr_character_flag = asked_to_pledge
		any_playable_ruler = {
			limit = {
				has_character_flag = forces_of_order_defender
				NOT = { religion_group = chaos_gods_group }
			}
			character_event = { id = bcrusade.25 days = 2 }
		}
	}
}

# bcrusade.25 Called to the defence
# Hidden. A lord who promised to fight the Black Crusade joins the
# defender's wars, with an alliance and good opinion both ways.
# Scope: ROOT is that lord. Fired by bcrusade.19, two days later.
character_event = {
	id = bcrusade.25
	is_triggered_only = yes
	hide_window = yes

	trigger = { }

	immediate = {
		if = {
			limit = {
				NOT = {
					has_opinion_modifier = {
						who = event_target:crusade_character
						name = opinion_fighting_against_black_crusade
					}
				}
			}
			opinion = {
				modifier = opinion_fighting_against_black_crusade
				who = event_target:crusade_character
			}
			reverse_opinion = {
				modifier = opinion_fighting_against_black_crusade
				who = event_target:crusade_character
			}
			add_alliance = { who = event_target:crusade_character years = 5 }
		}
		join_defender_wars = event_target:crusade_character
		clr_character_flag = forces_of_order_defender
		set_character_flag = previous_defender_against_black_crusade
	}
}

# bcrusade.92 A gift of gold
# Hidden. An AI ruler gives a year's income or more, by zeal.
# Scope: ROOT is the AI ruler. Fired by bcrusade.91.
character_event = {
	id = bcrusade.92
	hide_window = yes
	is_triggered_only = yes
	ai = yes

	trigger = {
		is_preparing_crusade = yes
	}

	immediate = {
		export_to_variable = { which = small_contribution value = monthly_income }
		multiply_variable = { which = small_contribution value = 12 }
		if = {
			limit = {
				check_variable = { which = small_contribution value < 10 }
				OR = { tier = COUNT tier = BARON }
			}
			set_variable = { which = small_contribution value = 10 }
		}
		else_if = {
			limit = { check_variable = { which = small_contribution value < 20 } tier = DUKE }
			set_variable = { which = small_contribution value = 20 }
		}
		else_if = {
			limit = { check_variable = { which = small_contribution value < 30 } tier = KING }
			set_variable = { which = small_contribution value = 30 }
		}
		else_if = {
			limit = { check_variable = { which = small_contribution value < 40 } tier = EMPEROR }
			set_variable = { which = small_contribution value = 40 }
		}
		set_variable = { which = medium_contribution which = small_contribution }
		multiply_variable = { which = medium_contribution value = 2 }
		set_variable = { which = large_contribution which = medium_contribution }
		multiply_variable = { which = large_contribution value = 2 }

		set_variable = { which = small_contribution_cost value = 0 }
		subtract_variable = { which = small_contribution_cost which = small_contribution value = 2 }
		set_variable = { which = medium_contribution_cost value = 0 }
		subtract_variable = {
			which = medium_contribution_cost
			which = medium_contribution
			value = 2
		}
		set_variable = { which = large_contribution_cost value = 0 }
		subtract_variable = { which = large_contribution_cost which = large_contribution value = 2 }

		set_variable = { which = small_piety_boon which = small_contribution }
		divide_variable = { which = small_piety_boon value = 2 }
		if = {
			limit = {
				check_variable = { which = small_piety_boon value < 5 }
				OR = { tier = COUNT tier = BARON }
			}
			set_variable = { which = small_piety_boon value = 5 }
		}
		else_if = {
			limit = { check_variable = { which = small_piety_boon value < 10 } tier = DUKE }
			set_variable = { which = small_piety_boon value = 10 }
		}
		else_if = {
			limit = { check_variable = { which = small_piety_boon value < 15 } tier = KING }
			set_variable = { which = small_piety_boon value = 15 }
		}
		else_if = {
			limit = { check_variable = { which = small_piety_boon value < 20 } tier = EMPEROR }
			set_variable = { which = small_piety_boon value = 20 }
		}
		set_variable = { which = medium_piety_boon which = medium_contribution }
		divide_variable = { which = medium_piety_boon value = 2 }
		set_variable = { which = large_piety_boon which = large_contribution }
		divide_variable = { which = large_piety_boon value = 2 }

		trigger_switch = {
			on_trigger = ai_zeal
			70 = {
				add_to_crusade_gold_pot = large_contribution
				wealth = large_contribution_cost
				piety = large_piety_boon
			}
			35 = {
				add_to_crusade_gold_pot = medium_contribution
				wealth = medium_contribution_cost
				piety = medium_piety_boon
			}
			0 = {
				add_to_crusade_gold_pot = small_contribution
				wealth = small_contribution_cost
				piety = small_piety_boon
			}
		}

		if = { limit = { NOT = { wealth = 1 } } clear_wealth = yes }
	}

	after = {
		set_variable = { which = small_contribution value = 0 }
		set_variable = { which = medium_contribution value = 0 }
		set_variable = { which = large_contribution value = 0 }
		set_variable = { which = small_contribution_cost value = 0 }
		set_variable = { which = medium_contribution_cost value = 0 }
		set_variable = { which = large_contribution_cost value = 0 }
		set_variable = { which = small_piety_boon value = 0 }
		set_variable = { which = medium_piety_boon value = 0 }
		set_variable = { which = large_piety_boon value = 0 }

		set_character_flag = supported_crusade
	}
}

# bcrusade.98 A gift to the war chest

	picture = GFX_evt_throne_room
	border = GFX_event_normal_frame_religion
	is_triggered_only = yes

	immediate = {
		# Variable Magic, essentially this:
		# A Small Contribution is one yearly income
		# A Medium Contribution is 2x Small
		# A Large Contribution is 2x Medium
		# The piety boon you get is half of what you give in wealth
		# If the small contribution is less than 50 gold, it will be set to 50 gold
		# All other values are based on the small contribution, they then update accordingly

		export_to_variable = { which = small_contribution value = monthly_income }
		multiply_variable = { which = small_contribution value = 12 }
		if = {
			limit = { check_variable = { which = small_contribution value < 50 } }
			set_variable = { which = small_contribution value = 50 }
		}
		set_variable = { which = medium_contribution which = small_contribution }
		multiply_variable = { which = medium_contribution value = 2 }
		set_variable = { which = large_contribution which = medium_contribution }
		multiply_variable = { which = large_contribution value = 2 }

		set_variable = { which = small_contribution_cost value = 0 }
		subtract_variable = { which = small_contribution_cost which = small_contribution value = 2 }
		set_variable = { which = medium_contribution_cost value = 0 }
		subtract_variable = {
			which = medium_contribution_cost
			which = medium_contribution
			value = 2
		}
		set_variable = { which = large_contribution_cost value = 0 }
		subtract_variable = { which = large_contribution_cost which = large_contribution value = 2 }

		set_variable = { which = small_piety_boon which = small_contribution }
		divide_variable = { which = small_piety_boon value = 2 }
		if = {
			limit = { check_variable = { which = small_piety_boon value < 25 } }
			set_variable = { which = small_piety_boon value = 25 }
		}
		set_variable = { which = medium_piety_boon which = medium_contribution }

			show_scope_change = no
			opinion = { who = ROOT modifier = opinion_grateful years = 3 }

# bcrusade.200 The duchy is ours
# The lord has taken the last stronghold of a duchy. Give it to his chosen
# lord (or a fitting kinsman) as a tributary realm, take it for himself
# (the other crusaders resent it), abdicate his old realm to his heir and
# rule it in person, or let it go for the prestige and gold the decision
# cost.
# Scope: ROOT is the lord, FROM the duchy. Fired by the decision
# dynamic_crusader_state (event targets duchy_region, crusader_lord).
character_event = {
	id = bcrusade.200
	desc = EVTDESCbcrusade200
	picture = GFX_evt_mountaintop_castle_hf
	border = GFX_event_normal_frame_religion
	portrait = event_target:current_crusade_beneficiary
	is_triggered_only = yes
	immediate = {
		# Saves the capital of the new Crusader state as an event target
		event_target:duchy_region = {
			capital_scope = { county = { save_event_target_as = crusade_state_capital } }
		}
		# Saves heir
		player_heir = { save_event_target_as = crusader_heir }
		# Check if you have a Beneficiary
		if = {
			limit = { crusade_beneficiary = { always = yes } }
			crusade_beneficiary = { save_event_target_as = current_crusade_beneficiary }
		}
		# Set up a backup Beneficiary like character if you don't have one
		else = {
			random_dynasty_member = {
				limit = {
					is_lowborn = no
					NOT = { any_heir_title = { always = yes } }
					NAND = { is_female = yes age = 40 }
					mercenary = no
					is_landed = no
					OR = { age = 16 AND = { age = 12 is_female = no } }
					NOR = { trait = incapable trait = monk trait = nun }
					NAND = {
						is_married = yes
						spouse = { OR = { is_landed = yes any_heir_title = { always = yes } } }
					}
					NAND = {
						is_betrothed = yes
						betrothed = { OR = { is_landed = yes any_heir_title = { always = yes } } }
					}
					NAND = {
						player_heir = { OR = { is_landed = yes any_heir_title = { always = yes } } }
					}
					NAND = { is_female = yes any_child = { is_landed = yes } }
				}
				save_event_target_as = current_crusade_beneficiary
			}
		}
	}

	option = { # I will find an appropriate ruler
		name = EVTOPTBbcrusade200

		add_character_modifier = { name = settled_crusader_state years = 15 }

		piety = 100
		if = {
			limit = { NOR = { trait = crusader_king trait = crusader_queen trait = crusader } }
			add_trait = crusader
		}

		custom_tooltip = {
			text = EVTOPTBbcrusade200_TT

			set_crusade_beneficiary = none

			event_target:duchy_region = {
				# Grant all the provinces to the new Crusader Lord/Lady
				any_direct_de_jure_vassal_title = {
					owner = { save_event_target_as = previous_owner }
					event_target:current_crusade_beneficiary = {
						grant_title_no_opinion = PREV
						set_defacto_liege = THIS
					}
					event_target:previous_owner = { add_claim = PREV }
					if = {
						limit = { event_target:previous_owner = { top_liege = { ai = no } } }
						event_target:previous_owner = { top_liege = { add_claim = PREVPREV } }
					}
				}
				# Check if the Duchy has an owner, receive the Duchy if nobody owns it
				if = {
					limit = { NOT = { owner = { always = yes } } }
					event_target:current_crusade_beneficiary = { grant_title_no_opinion = PREV }
				}
				# Make a titular Duchy if there already is a Duke/Duchess for the title, and get a claim on the real Duchy title
				else = {
					create_title = {
						tier = DUKE
						name = "crusade_state_duchy_name"
						landless = no
						culture = event_target:current_crusade_beneficiary
						holder = event_target:current_crusade_beneficiary
						custom_created = yes
						base_title = event_target:crusade_state_capital
					}

					new_title = {
						set_title_flag = temporary_crusade_title_flag
						adjective = "crusade_state_duchy_adjective"
					}

					event_target:current_crusade_beneficiary = { add_claim = PREV }
				}
			}

			event_target:current_crusade_beneficiary = {
				# Grab all remaining vassal titles held by characters of the wrong Religion
				any_vassal = {
					limit = { NOT = { religion = PREV } }
					any_demesne_title = {
						event_target:current_crusade_beneficiary = { grant_title = PREV }
					}
				}

				# Make Stewards for the cities, and grant them a city each
				any_demesne_title = {
					limit = { TIER = BARON holding_type = CITY }
					event_target:current_crusade_beneficiary = {
						create_random_steward = { random_traits = yes }
						new_character = {
							# Set Racial Trait
							character_event = { id = aiclean.31 }
							grant_title_no_opinion = PREVPREV
						}

# The Penitent Host
# (was the Shepherds' Crusade, HFP.42300 - HFP.42330)
#
# When a Black Crusade is beaten back, the faithful of the lands it
# struck do not all go home. In the years after, a doom-prophet stirs the
# commons of the Old World: flagellants, zealots, the ruined and the
# grieving. With no lord to lead them, they march north to carry the war
# into the lands that sent the horde. The lords of the Old World may feed
# them, join them or bar the road; on the way they hunt the Strigany and
# burn the priests who call them a rabble. If they win, their prophet
# crowns himself in the conquered kingdom.
#
# Vanilla fired the Shepherds' Crusade after a failed Catholic crusade and
# sent it to Iberia or Jerusalem; in the mod the call was commented out
# and the chain pointed at the Pope, so none of it could fire. Now it
# follows a failed Black Crusade, game rule shepherds_crusade, 15%, at most
# once every twenty years, for a defender of the Old World faiths.
#
# Setting out                             bcrusade.300 - bcrusade.304
# How it ends                             bcrusade.305 - bcrusade.307
# On the road                             bcrusade.310 - bcrusade.316
#
# Fired from: on_reverse_demand of new_crusade (00_cb_types.txt), the
#   shepherds_crusade_invasion CB (01_cb_types.txt), bcrusade.310 from
#   on_unit_entering_province (00_on_actions.txt).
# Saved: title flag penitent_host_target (the kingdom the failed horde
#   came from), global flag penitent_host_rising, province flag
#   shepherds_crusade_originator, character flag shepherds_leader.
###############################################################################

# bcrusade.300 The horde is broken
# Hidden. A Black Crusade has failed. Marks the kingdom the warlord came
# from and, in one to three years, a province of the defender's faith
# where a prophet will preach.
# Scope: ROOT is the defender, FROM the beaten warlord. Fired by
# on_reverse_demand of new_crusade.
character_event = {
	id = bcrusade.300
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		religion_group = old_world_group
		has_game_rule = { name = shepherds_crusade value = on }
		OR = {
			NOT = { has_global_flag = penitent_host_rising }
			had_global_flag = { flag = penitent_host_rising days = 7300 }
		}
		FROM = { capital_scope = { kingdom = { NOT = { religion_group = ROOT } } } }
		any_province = { religion = ROOT owner = { religion = ROOT } }
	}

	immediate = {
		clr_global_flag = penitent_host_rising
		set_global_flag = penitent_host_rising
		FROM = { capital_scope = { kingdom = { set_title_flag = penitent_host_target } } }
		random_province = {
			limit = { religion = ROOT owner = { religion = ROOT } }
			preferred_limit = {
				owner = { OR = { character = ROOT top_liege = { character = ROOT } } }
			}
			set_province_flag = shepherds_crusade_originator
			if = {
				limit = { has_global_flag = qa_testing }
				province_event = { id = bcrusade.301 days = 1 }
			}
			else = { province_event = { id = bcrusade.301 days = 365 random = 730 } }
		}
	}
}

# bcrusade.310 The host on the road
# Hidden. As the host enters a province, something happens there, at most
# once every fifty days per lord. In lands of the host's faith, locals join
# it, or the lord is asked for help (.311), to drive out the Strigany
# (.312), to bear its disorder (.313) or its deserters (.314), or it burns
# a priest who spoke against it (.315). In other lands it brings disorder
# (.316) or the downtrodden of its faith rise to join it.
# Scope: ROOT is a commander of the host. Fired from
# on_unit_entering_province.
character_event = {
	id = bcrusade.310
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_character_flag = shepherds_leader
		NOT = {
			location = {
				owner = {
					has_character_modifier = shepherds_crusade_cooldown
					has_character_flag = shepherds_crusade_do_not_disturb
				}
			}
		}
	}

	immediate = {
		location = { owner = { save_event_target_as = location_owner } }
		event_target:location_owner = { set_character_flag = shepherds_crusade_do_not_disturb }
		location = { save_event_target_as = current_location }
		if = {
			limit = { has_character_flag = shepherds_leader }
			save_event_target_as = shepherds_leader
		}
		else = { top_liege = { save_event_target_as = shepherds_leader } }

		if = {
			limit = { event_target:location_owner = { religion_group = ROOT } }
			spawn_unit = {
				province = ROOT
				home = ROOT
				owner = THIS
				match_character = event_target:location_owner
				match_mult = 0.1
				match_max = 1000
				match_min = 50
				troops = { light_infantry = { 100 100 } }
				attrition = 0
			}
			random_list = {
				# A random character joins
				2 = {
					create_random_soldier = {
						random_traits = yes
						culture = event_target:location_owner
						religion = ROOT
						dynasty = none
					}
					spawn_unit = {
						province = ROOT
						home = ROOT
						owner = THIS
						match_character = event_target:location_owner
						match_mult = 0.2
						match_max = 1000
						match_min = 200
						troops = { light_infantry = { 100 100 } }
						attrition = 0
					}
				}
				# Some locals decide to pick up their arms to join the Crusade
				2 = {
					spawn_unit = {
						province = ROOT
						home = ROOT
						owner = THIS
						troops = { light_infantry = { 100 100 } }
						attrition = 0
					}
				}
				# Dropping by and asking for help in their quest
				1 = { event_target:location_owner = { character_event = { id = bcrusade.311 } } }
				# Asking for the Strigany to be driven out
				1 = {
					trigger = {
						event_target:location_owner = {
							NOT = { has_character_modifier = expelled_strigany }
							independent = yes
						}
					}
					event_target:location_owner = { character_event = { id = bcrusade.312 } }
				}
				# Causing chaos among the local populace
				1 = { event_target:location_owner = { character_event = { id = bcrusade.313 } } }
				# Some of the peasants settle in the area
				1 = { event_target:location_owner = { character_event = { id = bcrusade.314 } } }
				# Burning the local priest
				1 = {
					trigger = {
						event_target:current_location = {
							any_province_holding = {
								holding_type = TEMPLE
								owner = { government = theocracy_government }
							}
						}
					}
					event_target:location_owner = { character_event = { id = bcrusade.315 } }
				}
			}
		}
		else_if = {
			limit = { event_target:location_owner = { NOT = { religion_group = ROOT } } }
			random_list = {
				# Nothing happens
				2 = { }
				# Chaos in the local province
				1 = { event_target:location_owner = { character_event = { id = bcrusade.316 } } }
				# The downtrodden of the host's faith rise to join it
				1 = {
					create_random_soldier = {
						random_traits = yes
						dynasty = none
						culture = event_target:location_owner
						religion = ROOT
					}

					}
				}
			}
		}
	}

	after = {

			owner = {
				add_character_modifier = {
					name = shepherds_crusade_cooldown
					hidden = yes
					days = 50
				}
			}
		}

	}
