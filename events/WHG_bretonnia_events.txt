# Part 1: The Knights of Bretonnia (namespace bretonnia)
# Faction: Bretonnia
# Serves: the knights of the Lady (knights_of_bretonnia), the King and his
#   realm
#
# A Bretonnian is born a noble, a peasant or one of the clergy; a priestess
# of the Lady may be a Damsel from the start. Knights rise from Knight
# Errant to Knight of the Realm, take a vow of the Quest and, if they
# prove themselves against greed, cowardice, lust and wrath and best the
# Green Knight, are blessed by the Lady as Grail Knights. Along the way they
# take the Virtues, ride on quests (trolls, an orc warband, Druchii
# raiders, a village beset by beastmen, a tithe for the Lady), and the
# King of Bretonnia holds the crown only as a Grail Knight. When the realm
# is attacked the Green Knight may ride to its aid.
#
#   bretonnia.2 - 3        Birth and class: the knight's rank, peasant,
#                          noble or clergy (on_startup, on_yearly_pulse,
#                          on_adulthood)
#   bretonnia.6 - 7        The Green Knight rides to the King's aid
#   bretonnia.8 - 10       The King must be a Grail Knight, and the crown
#   bretonnia.11           A knight of another faith takes the Grail Quest
#   bretonnia.100 - 109    The Virtues (decisions)
#   bretonnia.200 - 207    Joining and rising in the order
#   bretonnia.300 - 305    A quest is given (on_society_bi_yearly_pulse)
#   bretonnia.400 - 419    The quests
#   bretonnia.500 - 507    The Grail Quest: the trials and the Lady
#
# Fixed: the knight's death rolls had a break; a knight of another faith
#   who took the Grail Quest had two options with the same text, and a
#   Taalite who kept the old gods got the Virtue of Chivalry instead of
#   Taal's favour; a debug log; a duplicate check on the starting
#   Damsels; the real-world joke behind the tithe quest (it was
#   quest_ladys_onlyfans, with a phial of the Lady's bath water) is now
#   quest_lady_tithe, and the phial holds water from the Lady's lake, with
#   a gentler chance of sickness. The four yearly trials have the
#   is_in_society pre-trigger.
# Callers: wh_on_actions.txt, decisions/wh_bretonnian_decisions.txt
#   (100 - 109, 11, 203, 204, 418, 419, 504), the mutation effects (507).
###############################################################################

namespace = bretonnia

# bretonnia.2 A knight wears his rank: Knight Errant, Knight of the Realm,
# Questing Knight or Grail Knight.
character_event = {
	id = bretonnia.2
	hide_window = yes
	is_triggered_only = yes
	religion = grail
	trigger = {
		is_member_of_bretonnia_trigger = yes
	}
	immediate = {
		if = { limit = { society_rank == 1 } add_trait = bretonnian_knight_errant }
		if = {
			limit = { society_rank == 2 }
			add_character_modifier = { name = virtue_chivalry duration = -1 }
			add_trait = bretonnian_knight
		}
		if = {
			limit = { society_rank == 3 }
			add_trait = bretonnian_knight
			add_character_modifier = { name = virtue_chivalry duration = -1 }
			add_character_modifier = { name = virtue_quest duration = -1 }
			set_character_flag = virtue_quest
		}
		if = {
			limit = { society_rank == 4 }
			add_character_modifier = { name = virtue_chivalry duration = -1 }
			add_trait = grail_knight
		}
	}
}

character_event = {
	id = bretonnia.3
	hide_window = yes
	religion = grail
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { is_bretonnian = yes trait = peasant_leader has_class_trait = no }

			limit = { is_bretonnian = yes is_theocracy = yes }
			add_trait = clergy

			add_trait = noble
		}
		else_if = {
			limit = { is_bretonnian = yes is_republic = yes has_class_trait = no }
			add_trait = peasant
		}
		else_if = {
			limit = { is_bretonnian = yes is_landed = no has_class_trait = no }
			add_trait = peasant
		}
	}
}

character_event = {
	id = bretonnia.6
	hide_window = yes
	only_playable = yes
	war = yes
	is_triggered_only = yes
	religion = grail
	trigger = {
		has_landed_title = e_bretonnia
		is_primary_war_defender = yes
		NOT = { has_character_flag = green_knight_help }
	}
	immediate = {
		set_character_flag = green_knight_help
		create_random_soldier = {
			name = "The Green Knight"
			random_traits = yes
			dynasty = none
			religion = grail
			culture = old_breton
			trait = creature_human
			trait = brilliant_strategist
			female = no
			age = 30
		}
		new_character = {
			change_martial = 20
			add_trait = noble
			add_trait = green_knight
			add_trait = brave
			add_trait = holy_warrior
			set_character_flag = green_knight
			add_artifact = dolorous_blade
			add_artifact = shadow_steed
			unsafe_give_minor_title = title_commander
			character_event = { id = bretonnia.7 months = 6 random = 6 }
		}
	}
}

character_event = {
	id = bretonnia.8
	desc = EVTDESCbretonnia8
	picture = GFX_evt_questing_bretonnia
	only_playable = yes
	only_independent = yes
	religion_group = old_world_group
	is_triggered_only = yes
	trigger = {
		has_landed_title = e_bretonnia
		NOT = { trait = grail_knight }
	}
	immediate = { }
	option = {
		name = EVTOPTAbretonnia8
		religion = grail
		if = {
			limit = { society_rank == 1 is_member_of_bretonnia_trigger = yes }
			society_rank_up = 3
			character_event = { id = bretonnia.9 }
		}
		else_if = {
			limit = { society_rank == 2 is_member_of_bretonnia_trigger = yes }
			society_rank_up = 2
			character_event = { id = bretonnia.9 }
		}
		else_if = {
			limit = { society_rank == 3 is_member_of_bretonnia_trigger = yes }
			society_rank_up = 1
			character_event = { id = bretonnia.9 }
		}
		else_if = {
			limit = { can_join_society = knights_of_bretonnia }
			join_society = knights_of_bretonnia
			society_rank_up = 3
			character_event = { id = bretonnia.9 }
		}
		else = { abdicate = yes }
		ai_chance = { factor = 5 }
	}
	option = {
		name = EVTOPTBbretonnia8
		abdicate = yes
		add_character_modifier = { name = renounced_imperial_throne duration = -1 }
		ai_chance = { factor = 0 }
	}
}

# given one by a Damsel of the highest rank: the hunt for a man who works

#   bretonmagic.1          The Damsels' missions (on_society_bi_yearly_pulse)
#   bretonmagic.2 - 6      The hunt for a man who works magic
#   bretonmagic.7 - 18     A new Damsel: the quest, the offer, the lessons
#   bretonmagic.26         A new Fay Enchantress
#   bretonmagic.50         AI rank-ups (on_five_year_pulse)
#
# Rebuilt on the pattern of the Ice Witch Sisterhood (kislev.1 - 40 in
# WHG_kislev_events.txt), which this chain was copied from. Fixed:
# - The hunt never ended: the decision that confronts the quarry
#   (ice_witch_hunt, wh_witch_decisions.txt) was open to Ice Witches only;
#   a Damsel may take it now, and it fires bretonmagic.4.

#   checked religion_group = grail, which is a religion, so it never
#   matched.
# - A recruit who accepted never became a Damsel: her induction timer sat
#   in an else with no if. She also moved to a court that was never saved.
# - The lessons' death rolls killed pupils who could not die, and the
#   search for a recruit had a stray trigger in its effects.
# - The texts were the Ice Witches' (Priviet, Kislevite law, a
#   necromancer's lessons); the Damsels have their own now, and the missing
#   option texts (ACCEPT, DECLINE, Refuse) are filled.
# - A new Fay Enchantress is human.

#   and common/scripted_effects/WHG_bretonnia_effects.txt; the Ice Witches'
#   kislev_male_mage_trigger, kislev_save_matriarch_effect and
#   kislev_lesson_accident_effect.

# errant.9 The visit goes on (hidden)
# Picks the next scene. The knights stay for at most three scenes, then ride
# on; a recruit makes them leave sooner (was MTTH 40 days, x0.1 with a recruit).
# Scope: ROOT is the host.
character_event = {
	id = errant.9
	hide_window = yes
	is_triggered_only = yes # errant.1 - 5

	trigger = {
		has_character_flag = errant_visiting
	}

	immediate = {
		if = {
			limit = {
				OR = { prisoner = yes is_alive = no war = yes has_character_flag = errant_scene_3 }
			}
			character_event = { id = errant.6 }
		}
		else = {
			if = {
				limit = { has_character_flag = errant_scene_2 }
				set_character_flag = errant_scene_3
			}
			else_if = {
				limit = { has_character_flag = errant_scene_1 }
				set_character_flag = errant_scene_2
			}
			else = { set_character_flag = errant_scene_1 }
			random_list = {
				30 = { # Hurt in the lists
					trigger = { errant_first_scene_trigger = yes }
					character_event = { id = errant.2 }
				}
				30 = { # One knight shines
					trigger = { ai = no errant_first_scene_trigger = yes }
					character_event = { id = errant.3 }
				}
				20 = { # A courtier takes offence
					trigger = { any_courtier = { errant_offended_trigger = yes } }
					modifier = { factor = 0.2 has_character_flag = errant_recruited }
					modifier = { factor = 0.2 has_character_flag = errant_seduction }
					character_event = { id = errant.4 }
				}
				20 = { # A girl of the court is seduced
					trigger = {
						ai = no
						NOT = { has_character_flag = errant_seduction }
						any_courtier = { errant_seduced_trigger = yes }
					}
					modifier = { factor = 0.2 has_character_flag = errant_recruited }
					character_event = { id = errant.5 }
				}
				20 = { # They ride on
					modifier = { factor = 10 has_character_flag = errant_recruited }
					character_event = { id = errant.6 }
				}
			}
		}
	}
}
