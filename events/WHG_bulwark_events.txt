# bulwark.1 General Winter
# Hidden. A foreign commander marches into a Kislevite province in winter,
# at war with its lord: the province turns hostile with cold (supply and
# attrition) for three months, and he may fall ill. Kislevites, the peoples
# of the Wastes and Norsca, the dead and daemons do not feel it.
# Scope: ROOT is the commander entering the province.
character_event = {
	id = bulwark.1
	hide_window = yes

	is_triggered_only = yes # on_unit_entering_province
	min_age = 16
	prisoner = no

	trigger = {
		in_command = yes
		NOT = { is_sub_unit_leader = yes }
		bulwark_winter_immune_trigger = no
		location = {
			region = world_old_world_kislev
			is_winter = yes
			owner = {
				top_liege = {
					ROOT = { top_liege = { war_with = PREVPREV } }
				}
			}
		}
	}

	immediate = {
		location = {
			if = {
				limit = {
					NOR = {
						has_province_modifier = bulwark_kislev_winter
						has_province_modifier = bulwark_called_winter
					}
				}
				add_province_modifier = { name = bulwark_kislev_winter days = 90 }
			}
		}
		if = {
			limit = { NOT = { has_character_modifier = bulwark_frost_cd } }
			add_character_modifier = { name = bulwark_frost_cd years = 1 hidden = yes }
			random = {
				chance = 10
				add_trait = ill
			}
		}
	}
}

# bulwark.2 Call the Winter
# An Ice Witch calls an early winter on the enemies in her lord's Kislevite
# lands (decision bulwark_call_winter). The besieged and occupied provinces
# freeze for half a year. One time in three the cold comes home too: a
# famine at her lord's capital.
# Scope: ROOT is the Ice Witch.
character_event = {
	id = bulwark.2
	desc = EVTDESCbulwark2
	picture = GFX_evt_kislev_ice_witch_casting
	border = GFX_event_narrative_frame_war

	is_triggered_only = yes # bulwark_call_winter

	immediate = {
		top_liege = {
			any_realm_province = {
				limit = {
					region = world_old_world_kislev
					OR = { has_siege = yes is_occupied = yes }
				}
				remove_province_modifier = bulwark_kislev_winter
				add_province_modifier = { name = bulwark_called_winter days = 180 }
			}
		}
	}

	option = {
		name = EVTOPTAbulwark2
		random = {
			chance = 33
			top_liege = {
				capital_scope = { add_province_modifier = { name = bulwark_winter_famine years = 2 } }
			}
		}
		random = {
			chance = 20
			add_trait = ill
		}
	}
}

# The Spring Driving
# Every spring the Kurgan ride south over the Lynsk. A raid has a face: a
# chieftain (flag bulwark_nemesis, the lord's rival) who comes back year
# after year until he is killed.
###############################################################################

# bulwark.10 The Spring Driving comes
# Hidden. Picks the chieftain: the lord's old nemesis if he still lives,
# otherwise a new one from a Kurgan court (or a Chaos court of the Wastes).
# Scope: ROOT is a Kislevite lord with land in northern Kislev.
character_event = {
	id = bulwark.10
	hide_window = yes

	is_triggered_only = yes # on_yearly_pulse bulwark_spring
	only_rulers = yes
	min_age = 16
	prisoner = no

	trigger = {
		bulwark_northern_lord_trigger = yes
		NOT = { has_character_modifier = bulwark_spring_cd }
	}

	immediate = {
		add_character_modifier = { name = bulwark_spring_cd years = 3 hidden = yes }
		random_demesne_province = {
			limit = { region = world_old_world_kislev_northern }
			preferred_limit = { NOT = { has_province_modifier = frost_great_warding_mod } }
			save_event_target_as = spring_province
		}
		if = {
			limit = {
				any_rival = {
					has_character_flag = bulwark_nemesis
					is_alive = yes
					prisoner = no
				}
			}
			random_rival = {
				limit = {
					has_character_flag = bulwark_nemesis
					is_alive = yes
					prisoner = no
				}
				save_event_target_as = spring_chieftain
			}
		}
		else = {
			random_independent_ruler = {
				limit = {
					OR = {
						culture_group = kurgan_group
						AND = {
							religion_group = chaos_gods_group
							capital_scope = { region = world_chaos_wastes }
						}
					}
				}
				preferred_limit = { culture_group = kurgan_group }
				create_character = {
					random_traits = yes
					dynasty = random
					religion = THIS
					culture = THIS
					female = no
					age = 30
					attributes = { martial = 12 }
					trait = creature_human
					trait = brave
					trait = wroth
				}
				new_character = {
					set_character_flag = bulwark_nemesis
					set_character_flag = ai_flag_refuse_marriage
					add_rival = ROOT
					save_event_target_as = spring_chieftain
				}
			}
		}
		if = {
			limit = { event_target:spring_chieftain = { always = yes } }
			character_event = { id = bulwark.11 }
		}
	}
}

# bulwark.11 The Spring Driving
# The chieftain's riders come over the river into the lord's lands. Ride
# out to meet them (a fight, and a chance to kill him), hold the stanitsa
# walls (a Frost Ward helps; if it holds it becomes a Stanitsa of Heroes),
# or pay them to ride on. A burned stanitsa sends refugees south (bulwark.12).
# Scope: ROOT is the lord; event_target:spring_chieftain the chieftain,
# event_target:spring_province the province raided.
character_event = {
	id = bulwark.11
	desc = {
		trigger = { event_target:spring_chieftain = { has_character_flag = bulwark_nemesis_returned } }
		text = EVTDESCbulwark11b
	}
	desc = {
		trigger = { event_target:spring_chieftain = { NOT = { has_character_flag = bulwark_nemesis_returned } } }
		text = EVTDESCbulwark11
	}
	picture = GFX_evt_battle_kislev_gods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # bulwark.10

	immediate = {
		event_target:spring_chieftain = { set_character_flag = bulwark_nemesis_returned }
	}

	# Ride out
	option = {
		name = EVTOPTAbulwark11
		random_list = {
			55 = {
				modifier = { factor = 1.5 martial = 14 }
				modifier = { factor = 1.3 trait = brave }
				modifier = { factor = 1.3 society_member_of = circle_of_the_bear }
				modifier = { factor = 0.6 trait = craven }
				prestige = 75
				random_list = {
					35 = {
						modifier = { factor = 1.5 combat_rating = 40 }
						character_event = { id = bulwark.13 }
					}
					65 = {
						custom_tooltip = { text = bulwark_chieftain_escapes_tt }
					}
				}
			}
			45 = {
				prestige = -25
				random = { chance = 30 add_trait = wounded }
				event_target:spring_province = { add_province_modifier = { name = bulwark_stanitsa_burned years = 5 } }
				character_event = { id = bulwark.12 days = 30 }
			}
		}
		ai_chance = {
			factor = 50
			modifier = { factor = 2 trait = brave }
			modifier = { factor = 0.3 trait = craven }
		}
	}

	# Hold the walls
	option = {
		name = EVTOPTBbulwark11
		random_list = {
			50 = {
				modifier = {
					factor = 1.6
					event_target:spring_province = { has_province_modifier = kislevsoc_frost_ward_mod }
				}
				modifier = {
					factor = 1.6
					event_target:spring_province = { has_province_modifier = frost_great_warding_mod }
				}
				modifier = { factor = 1.3 stewardship = 14 }
				event_target:spring_province = { add_province_modifier = { name = bulwark_stanitsa_heroes years = 10 } }
				prestige = 50
			}
			50 = {
				event_target:spring_province = { add_province_modifier = { name = bulwark_stanitsa_burned years = 5 } }
				character_event = { id = bulwark.12 days = 30 }
			}
		}
		ai_chance = { factor = 35 }
	}

	# Pay them to ride on
	option = {
		name = EVTOPTCbulwark11
		trigger = { wealth = 75 }
		wealth = -75
		prestige = -50
		event_target:spring_chieftain = { prestige = 50 }
		ai_chance = {
			factor = 15
			modifier = { factor = 3 trait = craven }
			modifier = { factor = 2 trait = greedy }
		}
	}
}

# bulwark.13 The chieftain falls
# The lord cuts down the chieftain who led the Spring Driving. His skull-torc
# can be taken, once in a lifetime, and the Wastes are a little weaker.
# Scope: ROOT is the lord; event_target:spring_chieftain the chieftain.
character_event = {
	id = bulwark.13
	desc = EVTDESCbulwark13

	border = GFX_event_narrative_frame_war

	is_triggered_only = yes # bulwark.11

	immediate = {
		event_target:spring_chieftain = {
			clr_character_flag = bulwark_nemesis
			death = { death_reason = death_battle killer = ROOT }
		}
	}

	option = {
		name = EVTOPTAbulwark13
		prestige = 100
		reduce_doom_progress_tiny_effect = yes
		if = {
			limit = {
				NOT = { has_character_flag = bulwark_took_torc }
				NOT = { has_artifact = bulwark_chieftain_torc }
			}
			set_character_flag = bulwark_took_torc
			add_artifact = bulwark_chieftain_torc
		}
	}
}

# bulwark.20 The Motherland Calls
# Hidden. A Black Crusade falls on a Kislevite realm: its ruler and every
# Kislevite lord of the realm take heart (morale, levy reinforcement, less
# revolt) until the war ends. The first time the Tsardom itself is the
# target, the Oath of Karaz brings the Dwarfs (bulwark.25).
# Scope: ROOT is the defender, FROM the attacker.
character_event = {
	id = bulwark.20
	hide_window = yes

	is_triggered_only = yes # on_war_started
	only_rulers = yes

	trigger = {
		culture_group = kislevite_group
		FROM = { religion_group = chaos_gods_group }
		any_war = {
			using_cb = new_crusade
			attacker = { character = FROM }
			defender = { character = ROOT }
		}
	}

	immediate = {
		set_character_flag = bulwark_motherland
		add_character_modifier = { name = bulwark_motherland_calls years = 10 }
		any_realm_lord = {
			limit = { culture_group = kislevite_group }
			add_character_modifier = { name = bulwark_motherland_calls years = 10 }
		}
		character_event = { id = bulwark.21 }
		if = {
			limit = {
				has_landed_title = e_kislev
				NOT = { has_global_flag = bulwark_oath_of_karaz }
			}
			set_global_flag = bulwark_oath_of_karaz
			character_event = { id = bulwark.25 days = 10 }
		}
	}
}
