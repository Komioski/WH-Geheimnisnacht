###############################################################################
# Way of Life: the business focus
# Namespace: business (was WoL.10100 - WoL.10999 in wol_business_events.txt)
# Faction: Generic
#
# A ruler with the business focus keeps an eye on coin and trade. Once a
# year the focus pulse may bring a venture or a worry (business.1): a
# trading expedition to a foreign court that may open a trade route for
# thirty years, a great tower (which may one day lean, and draw visitors
# who want to meet the genius who built it), a hedge maze, a carpenters'
# guild, new coin for the realm, a burgher who wants land, a workshop to
# fund, or simply sleepless nights over the accounts.
#
# The partner of an expedition is event_target:business_partner, the
# steward event_target:steward; the tower's mason is
# event_target:stonemason, its priest event_target:tower_priest and its
# province, once it leans, event_target:leaning_tower_province.
#
# The hub                              business.1 - business.2
# The trading expedition               business.10 - business.35
# The great tower                      business.40 - business.58
# The hedge maze                       business.60 - business.66
# Guild, coin, land, workshop          business.70 - business.76
# A head for business                  business.80
#
# Fired from: on_focus_pulse in 00_on_actions.txt (business.1). Shared
#   triggers and effects are in WHG_focus_triggers.txt and
#   WHG_focus_effects.txt.
###############################################################################

namespace = business

# business.1 Hub: a venture or a worry
# Picks one of the business events; the big ventures need the trader
# modifier and no lifestyle trait yet, and come once a lifetime.
# Scope: the ruler.
character_event = {
	id = business.1
	hide_window = yes
	is_triggered_only = yes # on_focus_pulse

	only_playable = yes
	only_capable = yes
	prisoner = no

	trigger = {
		has_focus = focus_business
		in_command = no
		demesne_size = 1
		NOT = { trait = infirm }
		NOT = { has_character_flag = do_not_disturb }
	}

	immediate = {
		random_list = {
			15 = { # Sleepless nights
				modifier = { factor = 0 trait = stressed }
				character_event = { id = business.2 }
			}
			30 = { # A workshop to fund
				modifier = {
					factor = 0
					NOT = {
						any_demesne_province = {
							any_province_holding = { business_burgher_holding_trigger = yes }
							NOT = { has_province_modifier = artisan_workshop }
							NOT = { has_province_flag = workshop_construction }
						}
					}
				}
				random_demesne_province = {
					limit = {
						any_province_holding = { business_burgher_holding_trigger = yes }
						NOT = { has_province_modifier = artisan_workshop }
						NOT = { has_province_flag = workshop_construction }
					}
					set_province_flag = workshop_construction
					save_event_target_as = business_province
				}
				character_event = { id = business.74 }
			}
			30 = { # A burgher wants land
				modifier = { factor = 0 has_character_modifier = sell_land_cd }
				modifier = {
					factor = 0
					NOT = {
						capital_scope = {
							NOT = { has_province_modifier = sold_off_land }
							any_province_holding = {
								business_burgher_holding_trigger = yes
								holder_scope = { wealth = 150 }
							}
						}
					}
				}
				add_character_modifier = { name = sell_land_cd hidden = yes years = 5 }
				capital_scope = {
					random_province_holding = {
						limit = {
							business_burgher_holding_trigger = yes
							holder_scope = { wealth = 150 }
						}
						holder_scope = { save_event_target_as = business_burgher }
					}
				}
				character_event = { id = business.73 }
			}
			30 = { # New coin
				modifier = { factor = 0 has_character_modifier = wol_debase_timer }
				character_event = { id = business.71 }
			}
			30 = { # The carpenters' guild
				modifier = {
					factor = 0
					NOT = {
						any_demesne_province = {
							any_province_holding = { business_burgher_holding_trigger = yes }
							NOT = { has_province_modifier = carpenters_guild }
							NOT = { has_province_modifier = outlawed_guild }
						}
					}
				}
				random_demesne_province = {
					limit = {
						any_province_holding = { business_burgher_holding_trigger = yes }
						NOT = { has_province_modifier = carpenters_guild }
						NOT = { has_province_modifier = outlawed_guild }
					}
					save_event_target_as = business_province
				}
				long_character_event = { id = business.70 }
			}
			30 = { # The great tower
				modifier = {
					factor = 0
					OR = {
						is_tribal = yes
						is_nomadic = yes
						NOT = { has_character_modifier = trader }
						lifestyle_traits = 1
						has_character_flag = tower_start
						has_character_flag = tower_end
						has_character_flag = hedge_maze
					}
				}
				character_event = { id = business.40 }
			}
			30 = { # The hedge maze
				modifier = {
					factor = 0
					OR = {
						NOT = { has_character_modifier = trader }
						lifestyle_traits = 1
						has_character_flag = tower_start
						has_character_flag = hedge_maze
						has_character_flag = maze_end
					}
				}
				character_event = { id = business.60 }
			}
			30 = { # The trading expedition
				modifier = {
					factor = 0
					OR = {
						NOT = { job_treasurer = { always = yes } }
						has_character_flag = est_trade_route
						has_character_flag = trade_route_progress
						business_trading_race_trigger = no
						NOT = {
							any_independent_ruler = { business_trade_partner_trigger = yes }
						}
					}
				}
				character_event = { id = business.10 }
			}
			1 = { } # Nothing this year
		}
	}
}

# business.15 The priests want to come along
# They pay part of the cost. Word of the venture may reach a merchant
# prince nearby (business.17). A failed theft ends it here (business.20).
# Scope: the ruler.
character_event = {
	id = business.15
	desc = EVTDESCbusiness15
	picture = GFX_evt_bishop
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # business.13, business.14

	option = { # Welcome
		name = EVTOPTAbusiness15
		set_character_flag = trade_priest
		if = {
			limit = { yearly_income = 200 }
			wealth = 80
		}
		else_if = {
			limit = { yearly_income = 25 }
			scaled_wealth = 0.4
		}
		else = {
			wealth = 5
		}
		hidden_tooltip = {
			if = {
				limit = { has_character_flag = trade_bandits_fail }
				character_event = { id = business.20 days = 7 }
			}
			else = {
				random = {
					chance = 50
					random_independent_ruler = {
						limit = {
							NOT = { character = ROOT }
							is_within_diplo_range = ROOT
							is_merchant_republic = yes
						}
						character_event = { id = business.17 }
					}
				}
				character_event = { id = business.21 days = 14 }
			}
		}
		ai_chance = {
			factor = 2
			modifier = { factor = 2 trait = zealous }
			modifier = { factor = 0.25 trait = cynical }
		}
	}
	option = { # No
		name = EVTOPTBbusiness15
		any_vassal = {
			limit = { is_priest = yes }
			opinion = { modifier = opinion_disappointed who = ROOT years = 5 }
		}
		hidden_tooltip = {
			if = {
				limit = { has_character_flag = trade_bandits_fail }
				character_event = { id = business.20 days = 7 }
			}
			else = {
				character_event = { id = business.21 days = 7 }
			}
		}
		ai_chance = {
			factor = 1
			modifier = { factor = 3 has_character_flag = trade_bandits }
		}
	}
}
