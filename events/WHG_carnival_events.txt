# The Carnival of Chaos
#
# Namespace carnival
# Faction: Chaos
# Serves: human lords (duke and higher) of the Empire, Tilea, Estalia,
# Bretonnia and Kislev, and the Nurgle cultists among them.
#
# A Nurgle warband disguised as a travelling show rolls into a lord's lands,
# plays its pieces, infects the crowd and, at sunset, shows its true face.
# Lore: Mordheim: Empire in Flames, White Dwarf 119, Realm of Chaos: The Lost
# and the Damned. Only one Carnival roams the map at a time. It visits up to
# three realms, then vanishes until the next one.
#
#   carnival.1        Starter (hidden, on_decade_pulse random list)
#   carnival.2-3      Act I: the wagons arrive, an arrest that fails
#   carnival.10-15    Act II: omens in the night, the truth uncovered
#   carnival.20-23    Act III: the performance, the Master's offer, the patron
#   carnival.25       The camp raided before the show (truth seen early)
#   carnival.30-35    Act IV: sunset, victory, the ravaged town, the Master
#                     ascends to Daemon Prince (rare), the Master taken
#   carnival.40-42    Act V: the hunt for the Master
#   carnival.80-81    The Carnival moves on (hidden), news of where it went
#   carnival.90-91    The Carnival vanishes (hidden), word to its patron
#   carnival.92-93    Safety (hidden): the Master or host dies, the chain
#                     stalls
#
# State: global flag carnival_on_the_road; global targets carnival_master,
# carnival_town, carnival_host, carnival_watcher, carnival_patron and
# carnival_next_host (the patron's choice), and the troupe carnival_jester,
# carnival_brute and carnival_plaguebearer; global variable
# global_carnival_realms. Host flags are cleared by
# carnival_clear_host_effect.
#
# Scripted triggers: common/scripted_triggers/wh_carnival_triggers.txt
# Scripted effects: common/scripted_effects/wh_carnival_effects.txt
# Event modifiers: common/event_modifiers/wh_carnival_modifiers.txt
# On actions: common/on_actions
# Artifact: common/artifacts/35_artifacts_carnival.txt
# Loc: localisation/00_WHG_carnival_events.csv

# carnival.1 The Carnival takes to the road
# Hidden. Picks a rural town in the ruler's lands, creates the Carnival
# Master (a secret sorcerer of Nurgle in a fine mask) and his troupe (a
# Tainted One and a Brute, Nurglite humans) and sends the arrival.
# More likely when the doom is high and for players.
# Scope: ROOT is the ruler. Fired from the on_decade_pulse random list
# carnival_of_chaos.
character_event = {
	id = carnival.1
	hide_window = yes
	is_triggered_only = yes

	only_rulers = yes
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		NOT = { has_global_flag = carnival_on_the_road }
		carnival_host_trigger = yes
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 5 ai = no }
		modifier = { factor = 1.5 is_doom_medium_trigger = yes }
		modifier = { factor = 2 is_doom_high_trigger = yes }
		modifier = { factor = 3 is_doom_very_high_trigger = yes }
		modifier = {
			factor = 2
			any_realm_province = {
				OR = {
					disease = nurgles_rot
					disease = creeping_buboes
					disease = typhus
					disease = measles
				}
			}
		}
	}

	immediate = {
		set_global_flag = carnival_on_the_road
		set_variable = { which = global_carnival_realms value = 0 }
		random_realm_province = {
			limit = { carnival_region_trigger = yes is_capital = no }
			save_global_event_target_as = carnival_town
		}
		save_global_event_target_as = carnival_host
		create_character = {
			random_traits = no
			dynasty = none
			religion = ROOT
			culture = ROOT
			female = no
			age = 55
			attributes = { diplomacy = 16 martial = 8 stewardship = 4 intrigue = 16 learning = 14 }
			trait = gregarious
			trait = deceitful
			trait = cruel
			trait = content
			flag = ai_flag_refuse_conversion
		}
		new_character = {
			save_global_event_target_as = carnival_master
			set_character_flag = carnival_master
			set_character_flag = carnival_troupe
			add_trait = cannot_marry
			set_secret_religion = nurgle
			give_nickname = nick_the_carnival_master
		}
		carnival_create_troupe_effect = yes
		character_event = { id = carnival.2 days = 7 random = 14 }
	}
}

# carnival.10 The omens begin
# Hidden. Sends one omen the host has not had yet (carnival.11-14). After two
# omens the play comes (carnival.20). If the truth was seen, the camp can be
# raided before the show (carnival.25).
# Scope: ROOT is the host.
character_event = {
	id = carnival.10
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { carnival_master_free_trigger = no }
			character_event = { id = carnival.90 }
		}
		else_if = {
			limit = { has_character_flag = carnival_truth_seen }
			character_event = { id = carnival.25 days = 3 }
		}
		else_if = {
			limit = { check_variable = { which = carnival_omens value = 2 } }
			character_event = { id = carnival.20 days = 7 random = 7 }
		}
		else = {
			change_variable = { which = carnival_omens value = 1 }
			random_list = {
				25 = {
					modifier = { factor = 0 has_character_flag = carnival_omen_11 }
					character_event = { id = carnival.11 }
				}
				25 = {
					modifier = { factor = 0 has_character_flag = carnival_omen_12 }
					character_event = { id = carnival.12 }
				}
				25 = {
					modifier = { factor = 0 has_character_flag = carnival_omen_13 }
					character_event = { id = carnival.13 }
				}
				25 = {
					modifier = { factor = 0 has_character_flag = carnival_omen_14 }
					character_event = { id = carnival.14 }
				}
			}
		}
	}
}

# carnival.80 The Carnival moves on
# Hidden. Clears the old host's flags and sends the Carnival to the next
# realm: the patron's choice if there is one, else a lord two provinces
# away. After three realms, or if the Master is dead or taken, or no host is
# found, it ends (carnival.90).
# Scope: ROOT is the old host.
character_event = {
	id = carnival.80
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		carnival_clear_host_effect = yes
		change_variable = { which = global_carnival_realms value = 1 }
		if = {
			limit = {
				OR = {
					carnival_master_free_trigger = no
					check_variable = { which = global_carnival_realms value = 3 }
				}
			}
			character_event = { id = carnival.90 }
		}
		else = {
			# The patron's choice
			if = {
				limit = {
					event_target:carnival_next_host = { is_alive = yes carnival_host_trigger = yes }
				}
				event_target:carnival_next_host = {
					save_event_target_as = carnival_new_host
					random_realm_province = {
						limit = { carnival_region_trigger = yes is_capital = no }
						save_event_target_as = carnival_new_town
					}
				}
			}
			# Two provinces down the road
			else = {
				event_target:carnival_town = {
					random_neighbor_province = {
						limit = { carnival_region_trigger = yes }
						random_neighbor_province = {
							limit = {
								carnival_region_trigger = yes
								has_owner = yes
								owner = {
									OR = {
										AND = {
											higher_tier_than = COUNT
											carnival_host_trigger = yes
										}
										liege = { carnival_host_trigger = yes }
									}
								}
							}
							save_event_target_as = carnival_new_town
							owner = {
								if = {
									limit = { higher_tier_than = COUNT carnival_host_trigger = yes }
									save_event_target_as = carnival_new_host
								}
								else = { liege = { save_event_target_as = carnival_new_host } }
							}
						}
					}
				}
			}
			clear_global_event_target = carnival_next_host
			if = {
				limit = {
					event_target:carnival_new_host = { NOT = { character = ROOT } }
					event_target:carnival_new_town = { always = yes }
				}
				event_target:carnival_new_town = { save_global_event_target_as = carnival_town }
				event_target:carnival_new_host = {
					save_global_event_target_as = carnival_host
					event_target:carnival_master = { move_character = PREV }
					carnival_move_troupe_effect = yes
					character_event = { id = carnival.2 days = 60 random = 60 }
				}
				if = { limit = { ai = no } character_event = { id = carnival.81 days = 30 } }
				event_target:carnival_patron = {
					if = {
						limit = { ai = no is_alive = yes NOT = { character = ROOT } }
						character_event = { id = carnival.91 days = 60 }
					}
				}
			}
			else = { character_event = { id = carnival.90 } }
		}
	}
}

# carnival.92 A death on the road
# Hidden. If the Master dies, the chain ends a month later (carnival.90 on the
# host). If the host dies while the Carnival is in their lands, the Master
# moves on (carnival.80 from the Master).
# Scope: ROOT is the dying character. Fired from on_death.
character_event = {
	id = carnival.92
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = { has_character_flag = carnival_master has_character_flag = carnival_host }
		has_global_flag = carnival_on_the_road
	}

	immediate = {
		if = {
			limit = { has_character_flag = carnival_master }
			if = {
				limit = {
					event_target:carnival_host = { is_alive = yes NOT = { character = ROOT } }
				}
				event_target:carnival_host = { character_event = { id = carnival.90 days = 30 } }
			}
			else = {
				clr_global_flag = carnival_on_the_road
				set_variable = { which = global_carnival_realms value = 0 }
				clear_global_event_target = carnival_master
				clear_global_event_target = carnival_town
				clear_global_event_target = carnival_host
				clear_global_event_target = carnival_watcher
				clear_global_event_target = carnival_next_host
				clear_global_event_target = carnival_patron
			}
		}
		else_if = {
			limit = { carnival_master_free_trigger = yes }
			event_target:carnival_master = { character_event = { id = carnival.80 days = 30 } }
		}
	}
}
