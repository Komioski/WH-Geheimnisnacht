###############################################################################
# Way of Life: the carousing focus
# Namespace: carousing (was WoL.4000 - WoL.4801 in wol_carousing_events.txt)
# Faction: Generic
#
# A ruler invites friends from the realm to a private feast (the
# invite_to_party decision in way_of_life_decisions.txt). Four months
# later, if anyone accepted, the party runs for a month: a madman may eat a
# guest, lustful guests may share a bed, new friends are made, fists fly,
# a lunatic steals the show, a drunkard drinks too much, or two guests sit
# down to a game. The end of each party may make the host a carouser, a
# hedonist, a socializer or a gamer. The zealots of Araby's strict faiths
# refuse and condemn; Dawi and Slaaneshi hosts have their own feasts.
#
# Guests are the realm characters with the opinion_carousing modifier
# towards the host (carousing_guest_trigger); once the party starts they
# also carry the party_guest flag (carousing_at_party_trigger).
#
# The invitation                       carousing.1 - carousing.4
# The gathering                        carousing.10 - carousing.16
# The cannibal                         carousing.20 - carousing.21
# The fling                            carousing.25 - carousing.29
# A new friend                         carousing.30 - carousing.31
# The brawl                            carousing.35 - carousing.38
# The lunatic                          carousing.40 - carousing.42
# The drunkard                         carousing.45 - carousing.47
# The game                             carousing.50 - carousing.53
# The party is over                    carousing.60 - carousing.61
# The night court (vampire)            carousing.65 - carousing.66
# The feast of the dead (Tomb King)    carousing.67
# The revel of the Neverborn (daemon)  carousing.69
#
# Fired from: the invite_to_party decision (carousing.1, carousing.10)
#   and on_death in WHG_character_on_actions.txt (carousing.61). Shared triggers are in
#   WHG_character_triggers.txt.
###############################################################################

namespace = carousing

# carousing.1 An invitation to a feast
# The invitee accepts (carousing.2), declines (carousing.3) or, as a zealot
# of a strict faith of Araby, declines and condemns the host's decadence
# (carousing.4). The AI weighs its traits, health, wars and its liege.
# Scope: ROOT is the invitee, FROMFROM the host.
letter_event = {
	id = carousing.1
	desc = { trigger = { FROMFROM = { religion_group = dwarf_gods_group } } text = EVTDESCcarousing1b }
	desc = { trigger = { FROMFROM = { religion = slaanesh } } text = EVTDESCcarousing1c }
	desc = {
		trigger = {
			FROMFROM = {
				NOT = { religion_group = dwarf_gods_group }
				NOT = { religion = slaanesh }
			}
		}
		text = EVTDESCcarousing1
	}

	is_triggered_only = yes # invite_to_party decision

	option = { # Accept
		name = EVTOPTAcarousing1
		trigger = {
			NOT = { has_character_flag = do_not_disturb }
		}
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0
				carousing_temperate_faith_trigger = yes
				NOR = {
					trait = decadent
					trait = hedonist
					trait = drunkard
				}
			}
			modifier = { factor = 0 is_rival = FROMFROM }
			modifier = { factor = 0 war = yes }
			modifier = {
				factor = 0.1
				NOT = { opinion = { who = FROMFROM value = -25 } }
			}
			modifier = {
				factor = 5
				liege = { character = FROMFROM }
				is_ruler = no
			}
			modifier = {
				factor = 2
				liege = { character = FROMFROM }
				is_ruler = yes
			}
			modifier = { factor = 0.5 trait = stressed }
			modifier = { factor = 0.1 trait = depressed }
			modifier = { factor = 10 trait = drunkard }
			modifier = {
				factor = 0.1
				OR = {
					trait = leper
					trait = has_bubonic_plague
					trait = has_typhoid_fever
					trait = has_typhus
					trait = has_measles
					trait = has_small_pox
				}
			}
			modifier = { factor = 0.5 trait = celibate }
			modifier = { factor = 0.5 trait = chaste }
			modifier = { factor = 2 trait = lustful }
			modifier = { factor = 10 trait = hedonist }
			modifier = { factor = 0.2 trait = temperate }
			modifier = { factor = 2 trait = gluttonous }
			modifier = { factor = 2 trait = gregarious }
			modifier = { factor = 0.3 trait = shy }
			modifier = { factor = 0.5 trait = paranoid }
			modifier = { factor = 10 trait = decadent }
			modifier = { factor = 3 religion = slaanesh }
		}
		set_character_flag = do_not_disturb
		clr_character_flag = carousing_invite
		hidden_tooltip = { FROMFROM = { letter_event = { id = carousing.2 days = 7 } } }
	}
	option = { # Decline
		name = EVTOPTBcarousing1
		ai_chance = { factor = 1 }
		FROMFROM = {
			opinion = { modifier = opinion_declined_party who = ROOT years = 2 }
		}
		clr_character_flag = carousing_invite
		hidden_tooltip = { FROMFROM = { letter_event = { id = carousing.3 days = 7 } } }
	}
	option = { # Decline and condemn
		name = EVTOPTCcarousing1
		trigger = {
			carousing_temperate_faith_trigger = yes
			trait = zealous
		}
		ai_chance = {
			factor = 1
			modifier = {
				factor = 5
				OR = {
					trait = brave
					opinion = { who = FROMFROM value = -50 }
				}
			}
		}
		piety = 30
		prestige = 30
		FROMFROM = {
			opinion = { modifier = opinion_declined_party who = ROOT years = 2 }
			opinion = { modifier = opinion_angry who = ROOT years = 2 }
		}
		clr_character_flag = carousing_invite
		hidden_tooltip = { FROMFROM = { letter_event = { id = carousing.4 days = 7 } } }
	}
}

# carousing.10 The day of the feast
# Hidden. Four months after the first invitation: no guest cancels
# (carousing.13), war or incapacity cancels (carousing.11), otherwise the
# party starts (carousing.14).
# Scope: ROOT is the host.
character_event = {
	id = carousing.10
	hide_window = yes

	is_triggered_only = yes # invite_to_party decision

	immediate = {
		clr_character_flag = inviting_guests_to_party_timer
		if = {
			limit = {
				NOT = {
					top_liege = {
						any_realm_character = { carousing_guest_trigger = yes }
					}
				}
			}
			character_event = { id = carousing.13 }
		}
		else_if = {
			limit = {
				OR = {
					war = yes
					trait = incapable
				}
			}
			character_event = { id = carousing.11 }
		}
		else = {
			character_event = { id = carousing.14 }
		}
	}
}

# carousing.14 Let the feast begin
# The host and every guest (carousing.15) carouse for forty days. Something
# happens on day fifteen (carousing.16); the party ends on day thirty
# (carousing.60). Half the gatherings of a Slaaneshi host are orgies (flag
# party_orgy), and the orgy takes the place of the day-fifteen happening.
# Scope: ROOT is the host.
character_event = {
	id = carousing.14
	desc = { trigger = { religion_group = dwarf_gods_group } text = EVTDESCcarousing14b }
	desc = { trigger = { has_character_flag = party_orgy } text = EVTDESCcarousing14d }
	desc = {
		trigger = {
			religion = slaanesh
			NOT = { has_character_flag = party_orgy }
		}
		text = EVTDESCcarousing14c
	}
	desc = {
		trigger = {
			NOT = { religion_group = dwarf_gods_group }
			NOT = { religion = slaanesh }
		}
		text = EVTDESCcarousing14
	}
	picture = {
		trigger = { NOT = { religion = slaanesh } }
		picture = GFX_evt_feast
	}
	picture = {
		trigger = {
			religion = slaanesh
			NOT = { has_character_flag = party_orgy }
		}
		picture = GFX_evt_slaanesh_feast
	}
	picture = {
		trigger = { has_character_flag = party_orgy }
		picture = GFX_evt_slaanesh_party
	}
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # carousing.10

	hide_from = yes

	immediate = {
		set_character_flag = party_host
		if = { # Half the gatherings of Slaanesh are feasts, half are orgies
			limit = { religion = slaanesh }
			random = {
				chance = 50
				set_character_flag = party_orgy
			}
		}
	}

	option = { # Let's get it started
		name = EVTOPTAcarousing14
		remove_character_modifier = organizing_party
		add_character_modifier = { name = party_carousing days = 40 }
		hidden_tooltip = {
			top_liege = {
				any_realm_character = {
					limit = { carousing_guest_trigger = yes }
					character_event = { id = carousing.15 }
					character_event = { id = carousing.60 days = 30 }
				}
			}
			character_event = { id = carousing.16 days = 15 }
			character_event = { id = carousing.60 days = 30 }
		}
	}
}

# carousing.16 What happens at the feast
# Hidden. A Slaaneshi orgy (flag party_orgy) has its orgy (carousing.70).
# Otherwise picks one happening by the traits of the host and guests:
# the cannibal (a madman or a cannibal, and an AI at the table, 20), the
# fling (both lustful, 25), a new friend (30), a brawl (35), the lunatic
# (40), the drunkard (45) or the game (50). Unliving hosts often have their
# own night instead: a vampire's night court (65), a Tomb King's feast of the
# dead (67) or a daemon's revel (69).
# Scope: ROOT is the host.
character_event = {
	id = carousing.16
	hide_window = yes

	is_triggered_only = yes # carousing.14

	trigger = {
		top_liege = {
			any_realm_character = { carousing_at_party_trigger = yes }
		}
	}

	immediate = {
		if = {
			limit = { has_character_flag = party_orgy }
			character_event = { id = carousing.70 }
		}
		else = {
			random_list = {
				40 = { # The night court (a vampire host)
					trigger = {
						is_vampire_trigger = yes
						top_liege = {
							any_realm_character = { carousing_living_prey_trigger = yes }
						}
					}
					character_event = { id = carousing.65 }
				}
				40 = { # The feast of the dead (a Tomb King host)
					trigger = { focus_tomb_king_trigger = yes }
					character_event = { id = carousing.67 }
				}
				40 = { # The revel of the Neverborn (a daemon host)
					trigger = {
						focus_daemon_host_trigger = yes
						top_liege = {
							any_realm_character = { carousing_living_prey_trigger = yes }
						}
					}
					character_event = { id = carousing.69 }
				}
				2 = { # The cannibal
					trigger = {
						OR = {
							carousing_maniac_trigger = yes
							top_liege = {
								any_realm_character = {
									carousing_at_party_trigger = yes
									carousing_maniac_trigger = yes
								}
							}
						}
						OR = {
							ai = yes
							top_liege = {
								any_realm_character = {
									carousing_at_party_trigger = yes
									ai = yes
								}
							}
						}
					}
					modifier = {
						factor = 20
						OR = {
							has_character_modifier = cannibal
							trait = cannibal_trait
						}
					}
					modifier = { factor = 1.5 trait = gluttonous }
					character_event = { id = carousing.20 }
				}
				20 = { # The fling
					trigger = {
						carousing_lustful_trigger = yes
						top_liege = {
							any_realm_character = {
								carousing_at_party_trigger = yes
								carousing_lustful_trigger = yes
								carousing_lust_match_trigger = yes
							}
						}
					}
					character_event = { id = carousing.25 }
				}
				30 = { # A new friend
					trigger = {
						top_liege = {
							any_realm_character = {
								carousing_at_party_trigger = yes
								NOT = { is_friend = ROOT }
								NOT = { is_rival = ROOT }
							}
						}
					}
					character_event = { id = carousing.30 }
				}
				10 = { # The brawl
					trigger = {
						top_liege = {
							any_realm_character = {
								carousing_at_party_trigger = yes
								NOT = { is_rival = ROOT }
							}
						}
						OR = {
							carousing_brawler_trigger = yes
							top_liege = {
								any_realm_character = {
									carousing_at_party_trigger = yes
									carousing_brawler_trigger = yes
								}
							}
						}
					}
					character_event = { id = carousing.35 }
				}
				20 = { # The lunatic
					trigger = {
						OR = {
							is_lunatic_trigger = yes
							top_liege = {
								any_realm_character = {
									carousing_at_party_trigger = yes
									is_lunatic_trigger = yes
								}
							}
						}
					}
					character_event = { id = carousing.40 }
				}
				10 = { # The drunkard
					trigger = {
						OR = {
							trait = drunkard
							top_liege = {
								any_realm_character = {
									carousing_at_party_trigger = yes
									trait = drunkard
								}
							}
						}
					}
					character_event = { id = carousing.45 }
				}
				10 = { # The game
					modifier = {
						factor = 3
						OR = {
							trait = gamer
							top_liege = {
								any_realm_character = {
									carousing_at_party_trigger = yes
									trait = gamer
								}
							}
						}
					}
					top_liege = {
						if = {
							limit = {
								any_realm_character = {
									carousing_at_party_trigger = yes
									trait = gamer
								}
							}
							random_realm_character = {
								limit = {
									carousing_at_party_trigger = yes
									trait = gamer
								}
								character_event = { id = carousing.50 }
							}
						}
						else = {
							random_realm_character = {
								limit = { carousing_at_party_trigger = yes }
								character_event = { id = carousing.50 }
							}
						}
					}
				}
			}
		}
	}
}

# carousing.21 A sleeping guest
# Leave them be (a fright: craven, stressed, depressed), or eat: the host
# eats a guest, a guest eats the host. Either way the eater becomes a
# cannibal.
# Scope: ROOT is the would-be eater, FROM the host.
character_event = {
	id = carousing.21
	desc = EVTDESCcarousing21
	picture = GFX_evt_feast
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # carousing.20

	option = { # How peaceful
		name = EVTOPTAcarousing21
		ai_chance = { factor = 0 }
		if = {
			limit = {
				trait = craven
				trait = stressed
				NOT = { trait = depressed }
			}
			add_trait = depressed
		}
		else_if = {
			limit = {
				trait = craven
				NOT = { trait = stressed }
			}
			add_trait = stressed
		}
		else_if = {
			limit = { NOR = { trait = craven trait = brave trait = berserker } }
			add_trait = craven
		}
	}
	option = { # Eat (host)
		name = EVTOPTBcarousing21
		trigger = { has_character_flag = party_host }
		ai_chance = { factor = 1 }
		carousing_become_cannibal_effect = yes
		top_liege = {
			random_realm_character = {
				limit = { carousing_at_party_trigger = yes }
				if = {
					limit = { is_invincible_trigger = yes }
					antideath_effect = yes
				}
				else = {
					death = { death_reason = death_murder_unknown_butchered killer = ROOT }
				}
			}
		}
	}
	option = { # Eat (guest)
		name = EVTOPTBcarousing21
		trigger = { has_character_flag = party_guest }
		ai_chance = { factor = 1 }
		carousing_become_cannibal_effect = yes
		FROM = {
			if = {
				limit = { is_invincible_trigger = yes }
				antideath_effect = yes
			}
			else = {
				death = { death_reason = death_murder_unknown_butchered killer = ROOT }
			}
		}
	}
}

# carousing.71 The orgy
# A guest joins the revel (good times, perhaps the host's bed, a little of
# Slaanesh's favour, and for the lustful of other faiths a taste for more)
# or slips away (a zealous guest of another faith is disgusted with the
# host).
# Scope: ROOT is the guest, FROM the host.
character_event = {
	id = carousing.71
	desc = EVTDESCcarousing71
	picture = GFX_evt_slaanesh_party
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # carousing.70

	option = { # Join the revel
		name = EVTOPTAcarousing71
		add_character_modifier = { name = good_times duration = 730 }
		if = {
			limit = { religion = slaanesh }
			piety = 15
		}
		else_if = {
			limit = {
				NOR = { trait = lustful trait = chaste }
			}
			random = {
				chance = 20
				add_trait = lustful
			}
		}
		if = {
			limit = { FROM = { carousing_lust_match_trigger = yes } }
			random = {
				chance = 50
				hidden_tooltip = { unprotected_sex_effect = yes }
			}
		}
		ai_chance = {
			factor = 2
			modifier = { factor = 3 religion = slaanesh }
			modifier = { factor = 2 trait = lustful }
			modifier = { factor = 2 trait = hedonist }
			modifier = { factor = 0.1 trait = chaste }
			modifier = {
				factor = 0.2
				trait = zealous
				NOT = { religion = slaanesh }
			}
		}
	}
	option = { # Slip away
		name = EVTOPTBcarousing71
		if = {
			limit = {
				trait = zealous
				NOT = { religion = slaanesh }
			}
			piety = 10
			opinion = { modifier = opinion_disgusted who = FROM years = 5 }
		}
		ai_chance = {
			factor = 1
			modifier = { factor = 3 trait = chaste }
			modifier = { factor = 2 trait = shy }
		}
	}
}
