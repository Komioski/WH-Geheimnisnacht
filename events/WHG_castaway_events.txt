###############################################################################
# The castaway's saga
# Namespace: castaway
# Faction: Generic
#
# A rare, player-only story over three to eight years. A storm that feels
# wrong (or a mutiny) leaves a count or duke for drowned. The lands go to
# the heir, the castaway washes up on a small island, survives on it, meets
# the exiled sorcerer who raised the storm, cleans the island's hall of the
# thing that haunts it and its mother in the mere, and at last sees a sail:
# go home, stay as Lord of the Isle, or send for the family. All the while a
# grey stranger plays them at chess, one move per act, and the score decides
# how the story ends. Beowulf, The Tempest, Robinson Crusoe and the knight's
# game with Death, told in the voices of the island's waters.
#
# Rules: players only, once per game (global flag castaway_saga_begun);
# count or duke, seat on the coast; mortal Men, Dawi, elves and orcs; never
# one who has already played Death at chess (chess_with_death, which the
# saga also sets, so the chess chain in WHG_death_events never fires for
# them). The companion, the sorcerer and his daughter are generated. A lost
# game shortens life, it does not end it (Komi, 2026-11-12).
#
# The wreck                                castaway.1-3
# Washed ashore, presumed drowned          castaway.4-7
# Survival                                 castaway.10-12
# The sorcerer of the far shore            castaway.20-23
# The hall and the mere                    castaway.30-32
# A sail on the horizon                    castaway.40-44
# The last move                            castaway.50
# The dragon of the isle (a drawn game)    castaway.60-61
#
# Fired from: castaway.1 by naval.1 (a player leading an army at sea, about
# one sea tile in three hundred, WHG_naval_events.txt), seas.21 (a storm on
# a Set Out voyage) and seas.23 (a mutiny), WHG_sea_events.txt. The rest is
# the chain.
# Scripted triggers: common/scripted_triggers/WHG_sea_triggers.txt
# Scripted effects: common/scripted_effects/WHG_sea_effects.txt
# Loc: localisation/00_WHG_castaway_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = castaway

# castaway.1 The saga begins (hidden)
# Finds a free island for the castaway (castaway_pick_island_effect). With
# none the saga does not start. A mutiny goes to castaway.3, anything else
# to the storm and the stranger (castaway.2).
# Scope: ROOT is the castaway. The caller has checked
# castaway_candidate_trigger and set castaway_by_mutiny for a mutiny.
character_event = {
	id = castaway.1
	hide_window = yes

	is_triggered_only = yes # naval.1, seas.21, seas.23

	trigger = {
		castaway_candidate_trigger = yes
	}

	immediate = {
		if = {
			limit = { any_province = { castaway_island_trigger = yes } }
			set_global_flag = castaway_saga_begun
			set_character_flag = castaway_saga
			set_character_flag = chess_with_death
			castaway_pick_island_effect = yes
			if = {
				limit = { has_character_flag = castaway_by_mutiny }
				character_event = { id = castaway.3 }
			}
			else = {
				narrative_event = { id = castaway.2 }
			}
		}
		else = {
			clr_character_flag = castaway_by_mutiny
		}
	}
}

# castaway.2 The grey stranger
# The fleet breaks up in a storm that feels wrong. In the wreck (or on the
# mutineers' sandbar) a grey stranger sits at a board: "Play me, and I give
# you time." Accepting starts the game (variable castaway_game). Refusing
# is allowed: Death only watches, and the end is harsher. The stranger
# speaks to Dawi and elves in their own way.
# Scope: ROOT is the castaway.
narrative_event = {
	id = castaway.2
	title = EVTNAMEcastaway2
	desc = {
		trigger = {
			has_character_flag = castaway_by_mutiny
		}
		text = EVTDESCcastaway2_mutiny
	}
	desc = {
		trigger = {
			NOT = { has_character_flag = castaway_by_mutiny }
			trait = creature_dwarf
		}
		text = EVTDESCcastaway2_dawi
	}
	desc = {
		trigger = {
			NOT = { has_character_flag = castaway_by_mutiny }
			trait = creature_elf
		}
		text = EVTDESCcastaway2_elf
	}
	desc = {
		trigger = {
			NOT = { has_character_flag = castaway_by_mutiny }
			NOR = { trait = creature_dwarf trait = creature_elf }
		}
		text = EVTDESCcastaway2
	}
	picture = GFX_evt_death
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes # castaway.1, castaway.3

	option = { # Set up the pieces
		name = EVTOPTAcastaway2
		set_character_flag = castaway_playing
		set_variable = { which = castaway_game value = 0 }
		custom_tooltip = { text = castaway_game_begins_tt }
		character_event = { id = castaway.4 days = 3 }
	}
	option = { # I do not play with you
		name = EVTOPTBcastaway2
		set_character_flag = castaway_refused
		set_variable = { which = castaway_game value = 0 }
		custom_tooltip = { text = castaway_game_refused_tt }
		character_event = { id = castaway.4 days = 3 }
	}
}

# castaway.3 The sandbar
# For a sea captain the end comes from their own deck: the first mate takes
# the flagship (and a Black Ark goes back to the Witch King's reserve) and
# leaves the captain on a sandbar with a pistol, one shot and a bottle.
# The mutineer sails home with the news of a drowning. Then the stranger.
# Scope: ROOT is the castaway.
character_event = {
	id = castaway.3
	desc = EVTDESCcastaway3
	picture = GFX_evt_pirate_crew
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # castaway.1

	immediate = {
		create_character = {
			random_traits = yes
			dynasty = none
			culture = ROOT
			religion = ROOT
			female = no
			age = 35
			attributes = { martial = 9 intrigue = 9 }
			trait = ambitious
			trait = deceitful
		}
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			set_character_flag = castaway_mutineer
			save_event_target_as = castaway_mutineer
		}
	}

	option = { # Curse his name to the deep
		name = EVTOPTAcastaway3
		any_artifact = {
			limit = {
				OR = {
					artifact_type = small_flagship
					artifact_type = regular_flagship
					artifact_type = enormous_flagship
				}
			}
			unsafe_destroy_artifact = yes
		}
		if = {
			limit = { sea_has_black_ark_trigger = yes }
			sea_return_arks_effect = yes
		}
		narrative_event = { id = castaway.2 days = 2 }
	}
}

# castaway.7 The widow's veil comes off (hidden)
# Half the time the spouse stops waiting and the marriage to the drowned
# lord is ended, so they may wed again: Penelope does not always weave.
# A spouse with no women's roles in the race, or one already wed again, is
# left alone.
# Scope: ROOT is the spouse.
character_event = {
	id = castaway.7
	hide_window = yes

	is_triggered_only = yes # castaway.6

	trigger = {
		has_character_flag = castaway_spouse
		any_spouse = { has_character_flag = castaway_saga }
	}

	immediate = {
		random = {
			chance = 50
			modifier = { factor = 1.5 trait = lustful }
			modifier = { factor = 0.5 trait = chaste }
			modifier = { factor = 0.5 trait = patient }
			random_spouse = {
				limit = { has_character_flag = castaway_saga }
				save_event_target_as = castaway_lost_spouse
			}
			remove_spouse = event_target:castaway_lost_spouse
			set_character_flag = castaway_spouse_moved_on
		}
	}
}

# castaway.20 The sorcerer of the far shore
# On the far side of the island lives an exiled magister (off Ulthuan, an
# exiled mage of Saphery) with his daughter. He keeps an air spirit bound
# in a split pine, and the misshapen son of the witch who ruled the island
# before him serves him in chains. The twist: he raised the storm, to drag
# a ship to his shore. Ally with him (a tutor in magic, and his daughter's
# hand for a son or the companion), free the spirit (a blessing, and his
# anger), or side with the witch's son (an uprising against him).
# Scope: ROOT is the castaway.
character_event = {
	id = castaway.20
	desc = EVTDESCcastaway20
	picture = GFX_evt_wandering_mage
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes # castaway.12

	immediate = {
		castaway_make_magister_effect = yes
		any_courtier = {
			limit = { has_character_flag = castaway_companion }
			save_event_target_as = castaway_companion
		}
	}

	option = { # Ally with the sorcerer
		name = EVTOPTAcastaway20
		castaway_magister_tutor_effect = yes
		event_target:castaway_magister = {
			opinion = { who = ROOT modifier = opinion_grateful years = 20 }
		}
		castaway_good_move_effect = yes
		if = {
			limit = {
				OR = {
					any_child = { castaway_can_wed_daughter_trigger = yes }
					event_target:castaway_companion = { castaway_can_wed_daughter_trigger = yes }
				}
			}
			character_event = { id = castaway.21 days = 30 random = 30 }
		}
		else = {
			character_event = { id = castaway.30 days = 150 random = 120 }
		}
		ai_chance = { factor = 40 }
	}
	option = { # Split the pine and free the spirit
		name = EVTOPTBcastaway20
		add_character_modifier = { name = castaway_spirit_blessing years = 30 }
		event_target:castaway_magister = {
			opinion = { who = ROOT modifier = opinion_furious years = 20 }
		}

			chance = 30
			add_trait = wounded
		}
		castaway_good_move_effect = yes

		character_event = { id = castaway.30 days = 150 random = 120 }

	}
	option = { # Strike off the witch's son's chains

# castaway.50 The last move
# The grey stranger comes back for the end of the game, and the chess
# score decides. Five good moves or more: Death concedes (a long life).
# Two to four: a draw, Death says "later" (in old age a dragon wakes on
# the island, castaway.60). Fewer: Death wins, and takes years off the
# castaway's life. One who refused the game can draw at best.
# Scope: ROOT is the castaway.
narrative_event = {
	id = castaway.50
	title = EVTNAMEcastaway50
	desc = {
		trigger = { has_character_flag = castaway_end_win }
		text = EVTDESCcastaway50_win
	}
	desc = {
		trigger = { has_character_flag = castaway_end_draw }
		text = EVTDESCcastaway50_draw
	}
	desc = {
		trigger = { has_character_flag = castaway_end_loss }
		text = EVTDESCcastaway50_loss
	}
	picture = GFX_evt_death
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes # castaway.40, castaway.42, castaway.43

	immediate = {
		if = {
			limit = { has_character_flag = castaway_refused }
			if = {
				limit = { check_variable = { which = castaway_game value = 5 } }
				set_character_flag = castaway_end_draw
			}
			else = {
				set_character_flag = castaway_end_loss
			}
		}
		else_if = {
			limit = { check_variable = { which = castaway_game value = 5 } }
			set_character_flag = castaway_end_win
		}
		else_if = {
			limit = { check_variable = { which = castaway_game value = 2 } }
			set_character_flag = castaway_end_draw
		}
		else = {
			set_character_flag = castaway_end_loss
		}
	}

	option = { # Death tips over his king
		name = EVTOPTAcastaway50
		trigger = { has_character_flag = castaway_end_win }
		add_character_modifier = { name = castaway_death_conceded duration = -1 }
		prestige = 300
	}
	option = { # "Later."
		name = EVTOPTBcastaway50
		trigger = { has_character_flag = castaway_end_draw }
		prestige = 100
		custom_tooltip = { text = castaway_dragon_later_tt }
		character_event = { id = castaway.60 days = 5475 random = 3650 }
	}
	option = { # Checkmate
		name = EVTOPTCcastaway50
		trigger = { has_character_flag = castaway_end_loss }
		add_character_modifier = { name = castaway_deaths_due duration = -1 }
