###############################################################################
# Treasures of Cathay
# Namespace: cathayart (was JD.60000 - 60004 and JD.10079 - 10083 in
#   jd_artifact_events)
# Faction: Cathay & the East
#
# Envoys of the Celestial Court come west to trade treasures for eunuchs,
# concubines or gold; Cathayan guests leave parting gifts; and when a prince
# or princess of the Dragon House dies at a western court, Cathay sends its
# condolences, a keepsake of the dead, or the mourner finds their journal.
#
# The envoy                               cathayart.1 - cathayart.4
# The parting gift                        cathayart.5
# An imperial spouse dies                 cathayart.10 - cathayart.14
#
# Callers: on_yearly_pulse and on_death (common/on_actions), ivory.1
# (WHG_ivory_road_events.txt), the Celestial Court guests
# (WHG_cathay_diplomacy_events.txt).
# Loc: localisation/00_WHG_cathay_artifact_events.csv
#
# Moved: the false limbs (JD.60005 - 60006) to maimed.600 - 601 and the
# horse chancellor's barding (JD.60007 - 60008) to health.165 - 166 in
# WHG_health_events.txt; writing a book (JD.10060 - 10070) to
# WHG_book_events.txt.
###############################################################################

namespace = cathayart

# cathayart.1 An envoy of Cathay comes to trade
# Hidden. Rarely, and once in a lord's life, a Cathayan envoy of the Celestial
# Court comes to trade treasures; one time in ten a famed strategist. Never
# while a treasure of Cathay is held; less likely in hard times, likelier in a
# golden age or an expansionist reign.
# Scope: a lord.
character_event = {
	id = cathayart.1
	hide_window = yes
	is_triggered_only = yes # on_yearly_pulse random list

	only_capable = yes
	only_playable = yes
	has_dlc = "Jade Dragon"
	prisoner = no
	war = no
	min_age = 16

	trigger = {
		NOT = { offmap_cathay = { governor = { character = ROOT } } }
		is_inaccessible_trigger = no
		cathay_diplo_interaction_trigger = yes
		NOT = { has_character_flag = cathayart_envoy_came } # once in a life
		NOT = { any_artifact = { has_artifact_flag = cathayan } }
		offmap_cathay = { NOR = { has_status = cathay_civil_war has_status = cathay_plague } }
	}

	weight_multiplier = {
		days = 1
		modifier = { offmap_cathay = { has_status = cathay_unrest } factor = 0.6 }
		modifier = { offmap_cathay = { has_status = cathay_mongol_invasion } factor = 0.3 }
		modifier = { offmap_cathay = { has_status = cathay_jurchen_invasion } factor = 0.3 }
		modifier = { offmap_cathay = { has_status = cathay_famine } factor = 0.3 }
		modifier = { offmap_cathay = { has_status = cathay_golden_age } factor = 3 }
		modifier = { offmap_cathay = { has_policy = cathay_expansionist } factor = 2 }
	}

	immediate = {
		set_character_flag = cathayart_envoy_came
		offmap_cathay = {
			governor = {
				# A Cathayan of the court, whoever the Viceroy is
				random_list = {
					5 = {
						create_character = {
							age = 32
							culture = cathayan
							religion = draconite
							dynasty = actually_culture
							female = yes
							random_traits = yes
							historical = yes
							trait = creature_human
						}
					}
					95 = {
						create_character = {
							age = 32
							culture = cathayan
							religion = draconite
							dynasty = actually_culture
							female = no
							random_traits = yes
							historical = yes
							trait = creature_human
						}
					}
				}
				new_character = {
					set_character_flag = cathayan_envoy_artifacts
					random = {
						chance = 10
						remove_education_effect = yes
						add_trait = brilliant_strategist
						change_martial = 5
						set_character_flag = cathayan_envoy_warrior
					}
					character_event = { id = cathayart.2 }
				}
			}
		}
	}
}

# cathayart.11 The condolences arrive
# Hidden. If the Viceroy still has the keepsake, the mourner gets the letter
# (cathayart.12), the treasure (cathayart.13) or the journal (cathayart.14);
# else the letter.
# Scope: the mourner; FROM is the dead.
character_event = {
	id = cathayart.11
	hide_window = yes
	is_triggered_only = yes # cathayart.10
	has_dlc = "Jade Dragon"

	immediate = {
		if = { #for loc + artifact purposes
			limit = { event_target:the_protector_general = { is_alive = no } }
			offmap_cathay = { governor = { save_event_target_as = the_protector_general } }
		}

		if = { ### (1) If you are to receive a letter...
			limit = { has_character_flag = scenario_1 }
			character_event = { id = cathayart.12 }
		}

		if = { ### (2) A keepsake
			limit = { has_character_flag = scenario_2 }
			if = { # the Viceroy (or his heir) still has the keepsake
				limit = {
					event_target:the_protector_general = {
						any_artifact = { has_artifact_flag = royal_inheritance_gift@FROM }
					}
				}
				character_event = { id = cathayart.13 } #Continue and send scenario 2...!
			}
			else = {
				character_event = { id = cathayart.12 } # else, send scenario 1 instead...
			}
		}

		if = { ### (3) If you are to find a personal journal...
			limit = { has_character_flag = scenario_3 }
			if = { # the Viceroy (or his heir) still has the keepsake
				limit = {
					event_target:the_protector_general = {
						any_artifact = { has_artifact_flag = royal_inheritance_gift@FROM }
					}
				}
				character_event = { id = cathayart.14 } #Continue and send scenario 3...!
			}
			else = {
				character_event = { id = cathayart.12 } # else, send scenario 1 instead...
			}
		}

		### Fall-back clearing of flags (if at war with WP already)...
		if = {
			limit = { war_with = event_target:the_protector_general }
			if = {
				limit = { has_character_flag = scenario_2 }
				# no keepsake piles up with the Viceroy
				hidden_effect = { event_target:cathayan_artifact = { destroy_artifact = yes } }
			}
			if = {
				# the journal needs no peace
				limit = { NOT = { has_character_flag = scenario_3 } }
				clr_character_flag = mourner_is_spouse
				clr_character_flag = mourner_is_grown_child
				clr_character_flag = mourner_is_small_child
				clr_character_flag = scenario_1
				clr_character_flag = scenario_2
				clr_character_flag = scenario_3
			}
		}
	}
}
