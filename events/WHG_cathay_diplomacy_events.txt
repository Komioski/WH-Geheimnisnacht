###############################################################################
# The Celestial Court
# Namespace: celestial (was JD.10022, 10023, JD.20000 - 20020 and
#   JD.30000 - 30012 in jd_chinese_diplomacy_events)
# Faction: Cathay & the East
#
# The Jade Viceroy of the West keeps the Dragon Emperor's dealings with the
# realms beyond the Great Bastion. One of the Viceroy's councillors may come
# to a lord's court: a diplomat to talk politics, a general to see the troops,
# a treasurer to count taxes or make trade pacts, a priest of the Celestial
# Dragon to teach or to debate. How the host greets the guest, in Grand
# Cathay's tongue, in borrowed silks or in idle talk about the weather, wins or
# loses the Emperor's grace. A guest who dies or is jailed on the visit leaves
# an awkward letter to write.
#
# Merchants who cheat Cathay's traders on the roads east bring a stern letter
# from the Viceroy, and reining them in may anger the vassals whose cities pay
# for it. A Cathayan who has lived at a lord's court may write home in praise
# of the host, or storm off swearing the Emperor will hear of an insult.
#
# The councillor's visit                celestial.1 - celestial.20
# Dishonest traders                     celestial.30 - celestial.34
# A Cathayan guest's verdict            celestial.40 - celestial.42
#
# Callers: common/on_actions/00_on_actions.txt, the yearly random list
# (celestial.1, .30, .40), on_death and on_imprisoned (celestial.18).
# Grace effects and triggers: 00_scripted_effects.txt and
# 00_scripted_triggers.txt (vanilla Jade Dragon). The councillor's parting
# gift is JD.60004 in jd_artifact_events.txt.
# Loc: localisation/00_WHG_cathay_diplomacy_events.csv
###############################################################################

# celestial.1 A councillor sets out
# Hidden. One of the Jade Viceroy's Cathayan councillors picks a reason to
# visit a ruler at peace with the Viceroy: talks for the chancellor, the
# troops or military study for the marshal, taxes or trade for the steward,
# preaching or philosophy for the priest (the first of each pair for a
# tributary). Likelier in a golden age, under an expansionist policy, with a
# tributary and with more grace; rarer while Cathay suffers, and rare
# without grace.
# Scope: the host.
character_event = {
	id = celestial.1
	hide_window = yes
	is_triggered_only = yes # on_yearly_pulse random list

	has_dlc = "Jade Dragon"
	only_capable = yes
	prisoner = no
	war = no
	min_age = 16

	trigger = {
		NOT = { has_character_flag = cathayan_governor_councilor_guest }
		is_inaccessible_trigger = no
		cathay_diplo_interaction_trigger = yes
		offmap_cathay = {
			NOR = { has_status = cathay_civil_war has_status = cathay_plague }
			governor = {
				is_landed = yes
				NOT = { character = ROOT }
				OR = {
					job_chancellor = { celestial_councillor_trigger = yes }
					job_marshal = { celestial_councillor_trigger = yes }
					job_treasurer = { celestial_councillor_trigger = yes }
					job_spiritual = { celestial_councillor_trigger = yes }
				}
			}
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 0.6 offmap_cathay = { has_status = cathay_unrest } }
		modifier = {
			factor = 0.3
			offmap_cathay = {
				OR = {
					has_status = cathay_mongol_invasion
					has_status = cathay_jurchen_invasion
					has_status = cathay_famine
					has_status = cathay_chaos_invasion
					has_status = cathay_ogre_war
					has_status = cathay_skaven_war
				}
			}
		}
		modifier = { factor = 1.5 offmap_cathay = { has_status = cathay_golden_age } }
		modifier = { factor = 1.5 offmap_cathay = { has_policy = cathay_expansionist } }
		modifier = { factor = 1.5 has_grace_medium_trigger = yes }
		modifier = { factor = 1.5 has_grace_major_trigger = yes }
		modifier = { factor = 0.1 has_grace_trivial_trigger = no }
		modifier = {
			factor = 1.5
			is_tributary = { type = offmap suzerain = e_cathay_west_governor }
		}
	}

	immediate = {
		offmap_cathay = {
			governor = {
				save_event_target_as = target_governor
				random_list = {
					20 = {
						trigger = { job_chancellor = { celestial_councillor_trigger = yes } }
						job_chancellor = {
							set_character_flag = cg_diplomatic_visit
							character_event = { id = celestial.2 }
						}
					}
					20 = {
						trigger = { job_marshal = { celestial_councillor_trigger = yes } }
						job_marshal = {
							if = {
								limit = {
									ROOT = {
										is_tributary = {
											type = offmap
											suzerain = event_target:target_governor
										}
									}
								}
								set_character_flag = cg_oversee_troops
							}
							else = { set_character_flag = cg_study_military }
							character_event = { id = celestial.2 }
						}
					}
					20 = {
						trigger = { job_treasurer = { celestial_councillor_trigger = yes } }
						job_treasurer = {
							if = {
								limit = {
									ROOT = {
										is_tributary = {
											type = offmap
											suzerain = event_target:target_governor
										}
									}
								}
								set_character_flag = cg_oversee_tax_collection
							}
							else = { set_character_flag = cg_trade }
							character_event = { id = celestial.2 }
						}
					}
					20 = {
						trigger = { job_spiritual = { celestial_councillor_trigger = yes } }
						job_spiritual = {
							if = {
								limit = {
									ROOT = {
										is_tributary = {
											type = offmap
											suzerain = event_target:target_governor
										}
									}
								}
								set_character_flag = cg_preaching
							}
							else = { set_character_flag = cg_study_theology }
							character_event = { id = celestial.2 }
						}
					}
				}
			}
		}
	}
}

# or jailing can be answered (celestial.18), and sends the host the letter

# celestial.3 A letter from the Jade Viceroy's court
# The councillor asks to visit. Welcoming the guest wins a little grace; the
# visit comes in about two months (celestial.4), and the host is cleared for
# new guests a while after (celestial.20). Turning the guest away costs much
# grace and offends the councillor and the Viceroy.
# Scope: the host; FROM is the councillor.
letter_event = {
	id = celestial.3
	desc = {
		trigger = { is_tributary = { type = offmap suzerain = event_target:target_governor } }
		text = EVTDESCcelestial3_tributary
	}
	desc = {
		trigger = {
			NOT = { is_tributary = { type = offmap suzerain = event_target:target_governor } }
		}
		text = EVTDESCcelestial3
	}
	border = GFX_event_letter_frame_diplomacy

	is_triggered_only = yes # celestial.2

	has_dlc = "Jade Dragon"

	option = { # Welcome
		name = EVTOPTAcelestial3
		sound_effect = china_grace_gain
		custom_tooltip = { text = EVTOPTAcelestial3_tt }
		add_grace_trivial_effect = yes
		hidden_effect = {
			set_character_flag = cathayan_governor_councilor_guest
			event_target:cathayan_guest = {
				character_event = { id = celestial.4 days = 50 random = 20 }
			}
			character_event = { id = celestial.20 days = 160 }
		}
		ai_chance = { factor = 90 }
	}
	option = { # Turn them away
		name = EVTOPTBcelestial3
		custom_tooltip = { text = EVTOPTBcelestial3_tt }
		detract_grace_major_effect = yes
		event_target:cathayan_guest = {
			show_scope_change = no
			opinion = { who = ROOT modifier = opinion_offended years = 15 }
			liege = {
				show_scope_change = no
				opinion = { who = ROOT modifier = opinion_offended years = 15 }
			}
		}
		hidden_effect = {
			event_target:cathayan_guest = {
				clear_persistent_event_target = host_of_cathayan_diplomat
				celestial_clear_mission_effect = yes
			}
		}
		ai_chance = {
			factor = 10
			modifier = { factor = 3 trait = proud }
			modifier = {
				factor = 0
				is_tributary = { type = offmap suzerain = event_target:target_governor }
			}
		}
	}
}

# celestial.4 The visit
# Hidden. The guest arrives. If the host's own councillor for the same
# office is skilled, all goes well (celestial.5); otherwise the omen of
# falling stars (celestial.6), the greeting in Grand Cathay's tongue
# (celestial.7), the borrowed silks (celestial.12) or the talk about the
# weather (celestial.13). The tongue and the silks need a host of another
# culture than the guest.
# Scope: the councillor; FROM is the host.
character_event = {
	id = celestial.4
	hide_window = yes
	is_triggered_only = yes # celestial.3

	has_dlc = "Jade Dragon"
	prisoner = no

	trigger = {
		offmap_cathay = { governor = { is_liege_of = ROOT } }
		FROM = { is_alive = yes prisoner = no }
	}

	immediate = {
		FROM = {
			if = {
				limit = { ROOT = { has_job_title = job_chancellor } }
				job_chancellor = { save_event_target_as = corresponding_councilor }
			}
			else_if = {
				limit = { ROOT = { has_job_title = job_marshal } }
				job_marshal = { save_event_target_as = corresponding_councilor }
			}
			else_if = {
				limit = { ROOT = { has_job_title = job_treasurer } }
				job_treasurer = { save_event_target_as = corresponding_councilor }
			}
			else_if = {
				limit = { ROOT = { has_job_title = job_spiritual } }
				job_spiritual = { save_event_target_as = corresponding_councilor }
			}
			random_list = {
				30 = { # All is well
					trigger = { event_target:corresponding_councilor = { always = yes } }
					modifier = {
						factor = 0.5
						event_target:corresponding_councilor = {
							OR = {
								AND = { has_job_title = job_chancellor NOT = { diplomacy = 8 } }
								AND = { has_job_title = job_marshal NOT = { martial = 8 } }
								AND = { has_job_title = job_treasurer NOT = { stewardship = 8 } }
								AND = { has_job_title = job_spiritual NOT = { learning = 8 } }
							}
						}
					}
					modifier = {
						factor = 1.5
						event_target:corresponding_councilor = { culture_group = oriental_group }
					}
					character_event = { id = celestial.5 }
				}
				10 = { # Falling stars
					character_event = { id = celestial.6 }
				}
				10 = { # The greeting
					trigger = { NOT = { culture = ROOT } }
					modifier = { factor = 1.3 is_learned_trigger = yes }
					character_event = { id = celestial.7 }
				}
				10 = { # The borrowed silks
					trigger = { NOT = { culture = ROOT } }
					modifier = { factor = 1.3 is_dumb_trigger = yes }
					modifier = { factor = 1.3 trait = greedy }
					modifier = { factor = 1.3 trait = vain }
					character_event = { id = celestial.12 }
				}
				10 = { # The weather
					modifier = { factor = 1.3 is_smart_trigger = yes }
					modifier = { factor = 1.3 trait = gregarious }
					modifier = { factor = 1.5 trait = poet }
					modifier = { factor = 1.3 trait = socializer }
					character_event = { id = celestial.13 }
				}
			}
		}
		clear_persistent_event_target = host_of_cathayan_diplomat
	}
}

# celestial.8 Flawless
# The host learned the tongue and greeted the guest flawlessly: a major gain
# of grace and Student of Language, or Multilingual for a former student.
# Rarely (15%) the guest leaves a Cathayan treasure (JD.60004), once per
# life and only if none is held.

	id = celestial.8
	desc = EVTDESCcelestial8
	picture = GFX_evt_chinese_bureaucrat
	border = GFX_event_normal_frame_religion

	has_dlc = "Jade Dragon"

	trigger = {
		event_target:cathayan_guest = { is_alive = yes prisoner = no }
	}

		name = EVTOPTAcelestial8

			}
		}
	}
}
