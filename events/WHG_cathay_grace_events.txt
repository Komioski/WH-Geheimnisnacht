# Faction: Cathay & the East
#
# Lords within reach of Cathay earn the grace of the Dragon Throne with
# gifts, tribute, relief and horses, and spend it on the throne's favours:
# a physician, a peace deal, a scholar-bureaucrat, a master engineer who
# builds, a strategist who teaches, an imperial marriage, siege engineers,
# a war on a rival or a treasure. A Cathayan sent west who dies within five
# years is refunded, unless the lord killed them. A Cathay that shuts its
# gates can be forced open by war, by mastery of the roads east or by
# talks with the Jade Viceroy.
#
# Gaining grace                           cathaygrace.1 - cathaygrace.7
# The physician and the peace deal        cathaygrace.10 - cathaygrace.13
# The scholar-bureaucrat                  cathaygrace.20 - cathaygrace.23
# The master engineer                     cathaygrace.30 - cathaygrace.37
# The strategist                          cathaygrace.40 - cathaygrace.45
# The imperial marriage                   cathaygrace.50 - cathaygrace.54
# Engineers, relief, war, treasure, horses cathaygrace.60 - cathaygrace.72
# Opening Cathay                          cathaygrace.80 - cathaygrace.88
# A gift of Cathay dies                   cathaygrace.90 - cathaygrace.94
#
# Callers: the offmap decisions in decisions/WHG_cathay_decisions.txt and
# wh_orient_decisions.txt, the warp decisions in wh_chaos_decisions.txt,
# common/on_actions, title_master_engineer and the
# force_open_cathay_cb (01_cb_types).
# Shared code: cathaygrace_* in WHG_race_effects.txt and
# WHG_race_triggers.txt.

# cathaygrace.1 A gift for the throne
# Hidden. Passes the thanks of the throne to the giver (cathaygrace.2).
# Scope: the Jade Viceroy (or a warp herald); FROM is the offmap power,
# FROMFROM the lord.
character_event = {
	id = cathaygrace.1
	has_dlc = "Jade Dragon"
	hide_window = yes
	is_triggered_only = yes # gift decisions (wh_orient, wh_chaos)
	immediate = {
		FROMFROM = { letter_event = { id = cathaygrace.2 } }
	}
}

# cathaygrace.2 The gift is accepted
# Grace from Cathay, more if the Emperor likes coin, less if not. A gifted
# artifact earns no extra grace here, and a gift to a warp power none at all.
# Scope: the lord; FROM is the Jade Viceroy.
letter_event = {
	id = cathaygrace.2
	has_dlc = "Jade Dragon"
	desc = {
		trigger = { FROM = { is_offmap_governor = offmap_cathay } }
		text = EVTDESCcathaygrace2
	}
	desc = {
		trigger = { FROM = { NOT = { is_offmap_governor = offmap_cathay } } }
		text = EVTDESCcathaygrace2_warp
	}
	border = GFX_event_letter_frame_diplomacy

	is_triggered_only = yes # cathaygrace.1

	option = {
		name = EVTOPTAcathaygrace2

		sound_effect = china_grace_gain
		if = {
			limit = {
				NOT = { has_character_flag = jd_gifting_artifact }
				FROM = { is_offmap_governor = offmap_cathay }
			}
			if = {
				limit = {
					OR = {
						liked_by_offmap = { type = offmap_cathay context = wealth }
						liked_by_offmap = { type = offmap_cathay }
					}
					NOR = {
						disliked_by_offmap = { type = offmap_cathay }
						disliked_by_offmap = { type = offmap_cathay context = wealth }
					}
				}
				add_offmap_currency = { offmap = offmap_cathay value = 450 }
			}
			else_if = {
				limit = {
					OR = {
						disliked_by_offmap = { type = offmap_cathay context = wealth }
						disliked_by_offmap = { type = offmap_cathay }
					}
					NOR = {
						liked_by_offmap = { type = offmap_cathay }
						liked_by_offmap = { type = offmap_cathay context = wealth }
					}
				}
				add_offmap_currency = { offmap = offmap_cathay value = 150 }
			}
			else = { add_offmap_currency = { offmap = offmap_cathay value = 300 } }
		}
		clr_character_flag = jd_gifting_artifact
	}
}

# cathaygrace.7 War voids the contracts
# Hidden. A war between a lord and the Viceroy ends their trade contracts and
# horse supplies, and those of their vassals.
# Scope: ROOT is the defender; FROM is the attacker (on_war_started).
character_event = {
	id = cathaygrace.7
	has_dlc = "Jade Dragon"
	hide_window = yes

	is_triggered_only = yes # on_war_started

	trigger = {
		offmap_cathay = { governor = { OR = { character = ROOT character = FROM } } }
	}

	immediate = {
		remove_character_modifier = cathayan_imperial_trade_contract
		any_vassal = { remove_character_modifier = cathayan_imperial_trade_contract }
		FROM = {
			remove_character_modifier = cathayan_imperial_trade_contract
			any_vassal = { remove_character_modifier = cathayan_imperial_trade_contract }
		}
		remove_character_modifier = cathay_supply_horses
		any_vassal = { remove_character_modifier = cathay_supply_horses }
		FROM = {
			remove_character_modifier = cathay_supply_horses
			any_vassal = { remove_character_modifier = cathay_supply_horses }
		}
	}
}
