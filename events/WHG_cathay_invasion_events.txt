###############################################################################
# Cathay's wars in the west
# Namespace: cathayinv (was JD.10100 - 10202 and JD.60100 - 60403 in
#   jd_chinese_invasion_events)
# Faction: Cathay & the East
#
# Every year the Jade Viceroy, Cathay's governor in the west, may march on
# a lord within his reach: a small lord is asked to pay tribute and is
# attacked if he refuses, while an Expansionist Cathay takes the Dzhagna
# frontier or bends whole kingdoms to its will. A western empire that
# claims the Dragon's mandate (a rebel general's) is always first in line.
# When a house falls in Cathay, its exiles ride west: a displaced prince of
# the old house or a Man-chu warlord loots the steppe until he can carve
# out a duchy, and a rebel general of a lost civil war marches to found an
# empire of his own, with temple names for his heirs.
#
# The Viceroy marches                     cathayinv.1 - cathayinv.12
# Exiles from Cathay                      cathayinv.20 - cathayinv.27
# Silver for the steppe                   cathayinv.30 - cathayinv.31
# The rebel general                       cathayinv.40 - cathayinv.49
# The west invades Cathay                 cathayinv.50 - cathayinv.53
#
# Callers: common/on_actions/00_on_actions.txt: on_offmap_yearly_pulse (1,
# 10, 11), on_offmap_ruler_changed (12), on_offmap_governor_changed (20),
# on_new_holder, on_new_holder_inheritance and on_new_holder_usurpation
# (44), on_death (47); WHG_cathay_state_events.txt and
# WHG_cathay_state_effects.txt (22, 23, 40); common/cb_types (30, 31, 46,
# 48, 51 - 53); decisions/wh_chaos_decisions.txt (50).
# Shared code: cathayinv_* in WHG_cathay_invasion_effects.txt and
# WHG_cathay_invasion_triggers.txt.
# Loc: localisation/00_WHG_cathay_invasion_events.csv
#
# Dropped: the debug logs; JD.10114 and JD.10115 (never called); the
# monthly fallback JD.10121, whose flags nothing set any more (cathayinv.20
# hands an unspawned exile to the next Viceroy instead).
# Fixed: target searches of up to 18 passes now one pass with preferred
# limits; a Man-chu or Tong warlord is a Chaos-worshipping Hung, not a
# Cathayan; exiles and the rebel general's household are of the right
# race; the rebel general's empires take the fantasy dynasty names; the
# Viceroy's muster point is saved before war is declared; no break.
###############################################################################

namespace = cathayinv

# cathayinv.1 The Viceroy plans a war
# Hidden. Once a year a Cathay at ease may make war in the west: on a
# western pretender to the Dragon's mandate, on a lord it asks for tribute,
# or, when Expansionist, for land. A busy Cathay mostly stays home.
# Scope: the Jade Viceroy; FROM is Cathay.
character_event = {
	id = cathayinv.1
	hide_window = yes
	is_triggered_only = yes # on_offmap_yearly_pulse

	has_dlc = "Jade Dragon"

	trigger = {
		FROM = { is_offmap_tag = offmap_cathay }
		war = no
		NOT = { has_offmap_tmp_flag = launching_invasion_of_the_west }
		cathay_is_fine_trigger = yes
		NOR = {
			has_game_rule = { name = cathayan_invasions value = none }
			has_game_rule = { name = cathayan_invasions value = adventurers_only }
		}
	}

	immediate = {
		set_offmap_tmp_flag = launching_invasion_of_the_west
		save_event_target_as = the_protector_general
		if = { limit = { is_landed = yes } set_offmap_tmp_flag = WP_is_on_map }
		else = { set_offmap_tmp_flag = WP_is_not_on_map }
		if = {
			limit = {
				OR = {
					NOT = { has_offmap_tmp_flag = pretender_cathayan_empire_war_cooldown }
					had_offmap_tmp_flag = {
						flag = pretender_cathayan_empire_war_cooldown
						years = 150
					}
				}
				any_playable_ruler = { primary_title = { has_title_flag = rebel_general_title } }
			}
			set_offmap_tmp_flag = pretender_cathayan_empire_exists
		}
		random_list = {
			100 = { # Bring down the pretender
				trigger = { has_offmap_tmp_flag = pretender_cathayan_empire_exists }
				modifier = { factor = 2 has_offmap_tmp_flag = WP_is_on_map }
				modifier = { factor = 1.5 has_status = cathay_golden_age }
				random_playable_ruler = {
					limit = { primary_title = { has_title_flag = rebel_general_title } }
					save_event_target_as = cathayan_invasion_target
				}
				cathayinv_spawn_point_effect = yes
				war = {
					casus_belli = dismantle_pretender_cathay_cb
					target = event_target:cathayan_invasion_target
				}
				clr_offmap_tmp_flag = pretender_cathayan_empire_exists
				clr_offmap_tmp_flag = pretender_cathayan_empire_war_cooldown
				any_player = {
					limit = { has_offmap_news_enabled = offmap_cathay is_within_diplo_range = ROOT }
					narrative_event = { id = cathayinv.8 days = 1 }
				}
			}
			100 = { # Ask for tribute
				trigger = {
					NOT = { has_offmap_tmp_flag = pretender_cathayan_empire_exists }
					NOT = { has_offmap_tmp_flag = tributary_invasion_fired }
					OR = {
						AND = {
							NOT = { has_policy = cathay_expansionist }
							any_independent_ruler = {
								cathayinv_tributary_target_trigger = yes
								cathayinv_small_realm_trigger = yes
							}
						}
						AND = {
							has_policy = cathay_expansionist
							any_independent_ruler = {
								cathayinv_tributary_target_trigger = yes
								num_of_count_titles_in_realm = 15
							}
						}
					}
				}
				modifier = { factor = 1.5 has_offmap_tmp_flag = WP_is_not_on_map }
				modifier = {
					factor = 1.5
					OR = {
						cathay_is_stable_expansionist_trigger = yes
						cathay_is_golden_age_open_trigger = yes
					}
				}
				modifier = { # An Open Cathay with few tributaries wants more
					factor = 10
					OR = {
						AND = {
							cathay_is_stable_open_trigger = yes
							NOT = {
								any_playable_ruler = {
									count = 4
									is_tributary = { suzerain = ROOT }
								}
							}
						}
						AND = {
							cathay_is_golden_age_open_trigger = yes
							NOT = {
								any_playable_ruler = {
									count = 9
									is_tributary = { suzerain = ROOT }
								}
							}
						}
					}
				}
				modifier = { factor = 2 has_status = cathay_golden_age }
				random_list = {
					10 = { # A small realm
						trigger = {
							OR = {
								NOT = { has_policy = cathay_expansionist }
								NOT = {
									any_independent_ruler = {
										cathayinv_tributary_target_trigger = yes
										num_of_count_titles_in_realm = 15
									}
								}
							}
						}
						cathayinv_pick_small_tributary_effect = yes
					}
					10 = { # A large realm, only when Expansionist
						trigger = {
							has_policy = cathay_expansionist
							any_independent_ruler = {
								cathayinv_tributary_target_trigger = yes
								num_of_count_titles_in_realm = 15
							}
						}
						modifier = { factor = 10 cathay_is_golden_age_expansionist_trigger = yes }
						cathayinv_pick_big_tributary_effect = yes
					}
				}
				event_target:cathayan_invasion_target = { character_event = { id = cathayinv.2 } }
			}
			100 = { # Take land
				trigger = {
					NOT = { has_offmap_tmp_flag = pretender_cathayan_empire_exists }
					NOT = { has_offmap_tmp_flag = major_invasion_fired }
					NOT = { has_offmap_tmp_flag = attacked_small_neighbor }
					NOT = {
						has_game_rule = { name = cathayan_invasions value = no_major_invasions }
					}
					OR = {
						cathayinv_dzhagna_open_trigger = yes
						AND = {
							has_policy = cathay_expansionist
							any_independent_ruler = {
								OR = {
									cathayinv_medium_conquest_target_trigger = yes
									cathayinv_big_conquest_target_trigger = yes
								}
							}
						}
						AND = {
							has_offmap_tmp_flag = WP_is_on_map
							any_neighbor_independent_ruler = {
								valid_offmap_cathay_target_location = yes
								valid_offmap_cathay_target = yes
								higher_real_tier_than = BARON
								is_nomadic = no
							}
						}
					}
				}
				modifier = { factor = 1.5 has_status = cathay_golden_age }
				modifier = { factor = 3 has_policy = cathay_expansionist }
				modifier = { factor = 4 cathay_is_golden_age_expansionist_trigger = yes }
				modifier = { # A Cathay with no land in the west wants some
					factor = 2
					has_offmap_tmp_flag = WP_is_not_on_map
					OR = { has_status = cathay_golden_age has_policy = cathay_expansionist }
				}
				random_list = {
					100 = { # The Dzhagna frontier, whatever the policy
						trigger = { cathayinv_dzhagna_open_trigger = yes }
						d_dzhagna = {
							any_direct_de_jure_vassal_title = { add_claim = ROOT }
							random_direct_de_jure_vassal_title = {
								limit = {
									holder_scope = {
										top_liege = { valid_offmap_cathay_target = yes }
									}
								}
								holder_scope = {
									top_liege = { save_event_target_as = cathayan_invasion_target }
								}
							}
						}
						set_offmap_tmp_flag = should_claim_all_titles_war
					}
					50 = { # A middling realm, when Expansionist
						trigger = {
							has_policy = cathay_expansionist
							any_independent_ruler = {
								cathayinv_medium_conquest_target_trigger = yes
							}
						}
						if = { # a neighbour first
							limit = {
								has_offmap_tmp_flag = WP_is_on_map
								any_neighbor_independent_ruler = {
									cathayinv_medium_conquest_target_trigger = yes
								}
							}
							random_neighbor_independent_ruler = {
								limit = { cathayinv_medium_conquest_target_trigger = yes }
								save_event_target_as = cathayan_invasion_target
							}
						}
						else = {
							random_independent_ruler = {
								limit = { cathayinv_medium_conquest_target_trigger = yes }
								save_event_target_as = cathayan_invasion_target
							}
						}
						set_offmap_tmp_flag = start_submission_war
					}
					10 = { # A great realm, when Expansionist
						trigger = {
							has_policy = cathay_expansionist
							any_independent_ruler = { cathayinv_big_conquest_target_trigger = yes }
						}
						modifier = { factor = 2 has_status = cathay_golden_age }
						if = { # a neighbour first, one the Emperor dislikes best
							limit = {
								has_offmap_tmp_flag = WP_is_on_map
								any_neighbor_independent_ruler = {
									valid_offmap_cathay_target = yes
									is_nomadic = no
									num_of_count_titles_in_realm = 20
								}
							}
							random_neighbor_independent_ruler = {
								limit = {
									valid_offmap_cathay_target = yes
									is_nomadic = no
									num_of_count_titles_in_realm = 20
								}
								preferred_limit = { cathayinv_disliked_trigger = yes }
								save_event_target_as = cathayan_invasion_target
							}
						}
						else = {
							random_independent_ruler = {
								limit = { cathayinv_big_conquest_target_trigger = yes }
								preferred_limit = { cathayinv_disliked_trigger = yes }
								save_event_target_as = cathayan_invasion_target
							}
						}
						set_offmap_tmp_flag = start_submission_war
					}
					100 = { # A neighbour, small if possible
						trigger = {
							has_offmap_tmp_flag = WP_is_on_map
							any_neighbor_independent_ruler = {
								valid_offmap_cathay_target_location = yes
								valid_offmap_cathay_target = yes
								is_nomadic = no
							}
						}
						random_neighbor_independent_ruler = {
							limit = {
								valid_offmap_cathay_target_location = yes
								valid_offmap_cathay_target = yes
								is_nomadic = no
							}
							preferred_limit = { NOT = { num_of_count_titles_in_realm = 5 } }
							preferred_limit = { NOT = { num_of_count_titles_in_realm = 10 } }
							preferred_limit = { NOT = { num_of_count_titles_in_realm = 20 } }
							save_event_target_as = cathayan_invasion_target
						}
						set_offmap_tmp_flag = start_submission_war
						if = { # A small war: it does not count as a major one
							limit = {
								event_target:cathayan_invasion_target = {
									NOT = { num_of_count_titles_in_realm = 20 }
								}
							}
							set_offmap_tmp_flag = attacked_small_neighbor
						}
						else_if = { # Too big for a Cathay that is not Expansionist
							limit = { NOT = { has_policy = cathay_expansionist } }
							cathayinv_clear_plans_effect = yes
						}
					}
				}
				if = {
					limit = { has_offmap_tmp_flag = launching_invasion_of_the_west }
					character_event = { id = cathayinv.6 }
				}
			}
			400 = { # Cathay is busy with its own troubles
				modifier = { factor = 0 has_policy = cathay_expansionist }
				cathayinv_clear_plans_effect = yes
			}
		}
	}
}

# cathayinv.2 The Viceroy asks for tribute
# Kneel to the Dragon Throne as a tributary, or face war (cathayinv.3).
# A raider of Cathay gives up the raids when kneeling.
# Scope: the target; FROM is the Jade Viceroy.
character_event = {
	id = cathayinv.2
	desc = EVTDESCcathayinv2
	picture = GFX_evt_china_invades
	is_triggered_only = yes # cathayinv.1

	has_dlc = "Jade Dragon"

	immediate = {
		set_character_flag = asked_to_become_tributary
	}

	option = {
		name = {
			text = EVTOPTAcathayinv2
			trigger = { NOT = { has_character_modifier = cathay_raid_active } }
		}
		name = {
			text = EVTOPTAcathayinv2_raider
			trigger = { has_character_modifier = cathay_raid_active }
		}
		prestige = -100
		add_grace_minor_effect = yes
		if = {
			limit = { is_tributary = yes }
			custom_tooltip = {
				text = REMOVE_PREVIOUS_SUZERAIN
				hidden_effect = { any_suzerain = { ROOT = { remove_tributary = PREV } } }
			}
		}
		if = { limit = { has_character_modifier = cathay_raid_active } cathayraid_end_effect = yes }
		FROM = { prestige = 200 make_tributary = { who = ROOT tributary_type = offmap } }
		any_player = {
			limit = {
				has_offmap_news_enabled = offmap_cathay
				is_within_diplo_range = ROOT
				NOT = { character = ROOT }
			}
			narrative_event = { id = cathayinv.4 }
		}
		clr_character_flag = asked_to_become_tributary
		ai_chance = {
			factor = 20
			modifier = { factor = 1.5 trait = craven }
			modifier = { factor = 1.1 offmap_cathay = { has_policy = cathay_expansionist } }
		}
	}

	option = {
		name = EVTOPTBcathayinv2
		detract_grace_super_huge_effect = yes
		custom_tooltip = { text = EVTOPTBcathayinv2_tt }
		FROM = { character_event = { id = cathayinv.3 } }
		clr_character_flag = asked_to_become_tributary
		ai_chance = {
			factor = 10
			modifier = { factor = 1.1 trait = brave }
			modifier = { factor = 1.1 trait = ambitious }
			modifier = { factor = 1.1 trait = greedy }
			modifier = { # The bigger the realm, the prouder its lord
				factor = 1.5
				num_of_count_titles_in_realm = 10
			}
			modifier = { factor = 1.5 num_of_count_titles_in_realm = 15 }
			modifier = { factor = 1.5 num_of_count_titles_in_realm = 20 }
		}
	}
}

# cathayinv.6 The Viceroy makes war for land
# Hidden. Musters the army and declares war for the Dzhagna claims or for
# the target's submission.
# Scope: the Jade Viceroy.
character_event = {
	id = cathayinv.6
	hide_window = yes
	is_triggered_only = yes # cathayinv.1

	has_dlc = "Jade Dragon"

	immediate = {
		if = { # A war on a small neighbour does not count as major
			limit = { NOT = { has_offmap_tmp_flag = attacked_small_neighbor } }
			set_offmap_tmp_flag = major_invasion_fired
		}
		cathayinv_spawn_point_effect = yes
		if = {
			limit = { has_offmap_tmp_flag = should_claim_all_titles_war }
			if = {
				limit = {
					can_use_cb = {
						casus_belli = claim_all
						target = event_target:cathayan_invasion_target
						only_check_triggers = yes
					}
				}
				war = { casus_belli = claim_all target = event_target:cathayan_invasion_target }
			}
			else = {
				any_claim = {
					limit = {
						holder_scope = {
							OR = {
								character = event_target:cathayan_invasion_target
								top_liege = { character = event_target:cathayan_invasion_target }
							}
						}
					}
					ROOT = {
						war = {
							casus_belli = claim
							target = event_target:cathayan_invasion_target
							thirdparty_title = PREV
						}
					}
				}
			}
		}
		if = {
			limit = { has_offmap_tmp_flag = start_submission_war }
			war = { casus_belli = offmap_submission target = event_target:cathayan_invasion_target }
		}
		clr_offmap_tmp_flag = start_submission_war
		clr_offmap_tmp_flag = should_claim_all_titles_war
		any_player = {
			limit = { has_offmap_news_enabled = offmap_cathay is_within_diplo_range = ROOT }
			narrative_event = { id = cathayinv.7 days = 1 }
		}
	}
}

# cathayinv.43 The rebel general invades
# Hidden. Declares war and raises an army for the general at a road end:
# fewer troops than the Viceroy's, but commanders as good as a quarter of
# his own skill makes them. If the general is dead, the prey hears of it
# (cathayinv.45).
# Scope: the prey.
character_event = {
	id = cathayinv.43
	hide_window = yes
	is_triggered_only = yes # cathayinv.42

	has_dlc = "Jade Dragon"

	trigger = {
		event_target:rebel_general = { is_alive = yes }
	}

	fail_trigger_effect = { character_event = { id = cathayinv.45 } }

	immediate = {
		event_target:rebel_general = {
			war = { casus_belli = offmap_submission target = event_target:rebel_general_target }
			set_variable = { which = cathay_war_troop_quantity value = 10 }
			set_variable = { which = cathay_war_target_match_mult value = 1 }
			set_variable = { which = cathay_war_troop_quality value = 3 }
			set_variable = { which = cathay_war_minimum_commander_martial value = 14 }
			export_to_variable = { which = rebel_general_martial value = martial }
			divide_variable = { which = rebel_general_martial value = 4 }
			change_variable = {
				which = cathay_war_minimum_commander_martial
				which = rebel_general_martial
			}
			cathay_war_modify_variables_by_century_effect = yes
			export_to_variable = {
				which = cathay_temp_troop_calc
				value = realm_levies_plus_allies
				who = FROM
			}
			divide_variable = { which = cathay_temp_troop_calc value = 500 }
			multiply_variable = {
				which = cathay_temp_troop_calc
				which = cathay_war_target_match_mult
			}
			change_variable = { which = cathay_war_troop_quantity which = cathay_temp_troop_calc }
			any_courtier_or_vassal = {
				set_variable = { which = cathay_war_minimum_commander_martial which = PREV }
			}
			if = {
				limit = {
					NOT = {
						any_courtier_or_vassal = {
							count = 5
							martial = cathay_war_minimum_commander_martial
						}
					}
				}
				cathay_spawn_commander_effect = yes
				cathay_spawn_commander_effect = yes
				cathay_spawn_commander_effect = yes
				cathay_spawn_commander_effect = yes
				cathay_spawn_commander_effect = yes
			}
			cathayinv_road_end_effect = yes
			event_target:cathayinv_road_end = { save_event_target_as = spawn_province }
			cathay_war_spawn_troops_no_disband_effect = yes
		}
	}
}

# cathayinv.47 A Cathayan Emperor dies
# Hidden. A rebel general's empire with an heir not of the east shatters:
# its vassals go free and its counties to new lords. A Cathayan-governed
# realm of another kind lets a player or landless heir choose to keep the
# eastern ways (cathayinv.49); an AI heir lets them go.
# Scope: the dying character.
character_event = {
	id = cathayinv.47
	hide_window = yes
	is_triggered_only = yes # on_death

	has_dlc = "Jade Dragon"

	trigger = {
		government = cathayan_imperial_government
		current_heir = { NOR = { culture_group = oriental_group culture = tong culture = man_chu } }
	}

	immediate = {
		if = {
			limit = { primary_title = { has_title_flag = rebel_general_title } }
			set_character_flag = pretender_cathayan_empire_destroyed_inheritance
			any_player = {
				limit = {
					OR = {
						AND = {
							has_offmap_news_enabled = offmap_cathay
							is_within_diplo_range = ROOT
						}
						liege = { character = ROOT }
					}
				}
				narrative_event = { id = cathayinv.46 }
			}
			any_vassal = { set_defacto_liege = THIS }
			any_demesne_title = {
				limit = {
					higher_tier_than = COUNT
					NOT = { has_title_flag = pretender_cathayan_empire }
				}
				destroy_landed_title = THIS
			}
			any_demesne_title = {
				limit = { tier = COUNT }
				location = {
					create_character = {
						religion = THIS
						culture = THIS
						dynasty = actually_culture
						age = 33
						random_traits = yes
						female = no
					}
					new_character = {
						save_event_target_as = pretender_cathayan_empire_new_character
						PREVPREV = {
							usurp_title_plus_barony_if_unlanded = {
								target = event_target:pretender_cathayan_empire_new_character
								type = invasion
							}
						}
					}
				}
			}
			primary_title = {
				activate_title = { title = THIS status = no }
				destroy_landed_title = THIS
			}
		}
		else_if = {
			limit = { OR = { ai = no current_heir = { is_landed = no } } }
			current_heir = { set_character_flag = cathayan_imperial_government_preserve }
		}
		else = {
			primary_title = {
				clr_title_flag = pretender_cathayan_empire
				clr_title_flag = uses_temple_names_by_script
				clear_flags_with_prefix = temple_name_
				clear_flags_with_prefix = cathay_name_
				set_name = "[Root.Capital.GetName] Empire"
			}
		}
	}
}
