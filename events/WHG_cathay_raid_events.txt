###############################################################################
# Raiding Cathay
# Namespace: cathayraid (was JD.4000 - 4030 in jd_raiding_china and the
#   warning letter JD.3050 - 3051 in jd_grace_decision_events)
# Faction: Cathay & the East
#
# Lords who sit near the ends of the roads east (the Kurgan, the hobgoblin
# khans, the ogre tribes, the Dark Lands warlords) may send raiders over
# Cathay's western border instead of begging the Emperor for grace. The
# raid runs until it is called off, it costs grace every month and holds
# back part of the levies, and every year the raiders come home with loot:
# gold, prestige and people for hordes, now and then a captive, a Cathayan
# siege engineer, a pair of pandas or, once in a lifetime, a treasure of
# the Celestial Empire. A Cathay that starves, sickens or fights itself is
# easy prey; a Cathay in its golden age guards its border well.
#
# The raid                               cathayraid.1 - cathayraid.5
# Pandas                                 cathayraid.10
#
# Callers: the offmap decisions cathay_raid and cathay_stop_raid
# (decisions/WHG_cathay_decisions.txt, Cathay's hostile actions button),
# on_yearly_pulse (1) in common/on_actions/00_on_actions.txt.
# Shared code: cathayraid_* in WHG_cathay_raid_effects.txt and
# WHG_cathay_raid_triggers.txt, add_cathayan_artifact_effect in
# 00_scripted_effects.txt.
# Loc: localisation/00_WHG_cathay_raid_events.csv
#
# Fixed: the raid decisions were lost when the mod replaced the decisions
# folder, so raids could never start; the captives came from Ind and
# Nippon; the pandas modifier used modifier = instead of name =; the
# tapestry looked for the vanilla Silk Road, empty in the mod.
###############################################################################

namespace = cathayraid

###############################################################################
# The raid
###############################################################################

# cathayraid.1 The raiders' year
# Hidden. Once a year, a raiding lord's war bands come home. A lord who may
# no longer raid (a vassal now, landless, or forbidden to loot) calls them
# back (cathayraid.2); otherwise one outcome is picked and shown
# (cathayraid.3).
# Scope: the raiding lord.
character_event = {
	id = cathayraid.1
	hide_window = yes
	is_triggered_only = yes # on_yearly_pulse
	only_playable = yes

	has_dlc = "Jade Dragon"

	trigger = {
		has_character_modifier = cathay_raid_active
		offmap_cathay = { NOT = { governor = { war_with = ROOT } } }
	}

	immediate = {
		if = {
			limit = { OR = { independent = no is_allowed_to_loot = no is_landed = no } }
			cathayraid_end_effect = yes
			character_event = { id = cathayraid.2 }
		}
		else = { cathayraid_pick_outcome_effect = yes character_event = { id = cathayraid.3 } }
	}
}

# cathayraid.2 The raids are over
# The lord can no longer send raiders east. Raids may start again in five
# years.
# Scope: the former raider.
character_event = {
	id = cathayraid.2
	desc = EVTDESCcathayraid2
	picture = GFX_evt_china_stable
	border = GFX_event_normal_frame_war
	is_triggered_only = yes # cathayraid.1

	has_dlc = "Jade Dragon"

	notification = yes

	option = {
		name = EVTOPTAcathayraid2
	}
}

# cathayraid.3 The raiders return
# One of eight outcomes, set by cathayraid_pick_outcome_effect: plain loot,
# rich loot, a ransomed mandarin, a disaster, a captive, a siege engineer,
# a treasure of Cathay (once per lifetime) or a pair of pandas.
# Scope: the raiding lord.
character_event = {
	id = cathayraid.3
	picture = GFX_evt_china_unrest
	border = GFX_event_normal_frame_war
	is_triggered_only = yes # cathayraid.1

	has_dlc = "Jade Dragon"

	desc = { text = EVTDESCcathayraid3_loot trigger = { has_character_flag = cathayraid_loot } }
	desc = { text = EVTDESCcathayraid3_rich trigger = { has_character_flag = cathayraid_rich } }
	desc = { text = EVTDESCcathayraid3_ransom trigger = { has_character_flag = cathayraid_ransom } }
	desc = { text = EVTDESCcathayraid3_bad trigger = { has_character_flag = cathayraid_bad } }
	desc = {
		text = EVTDESCcathayraid3_captive
		trigger = { has_character_flag = cathayraid_captive }
	}
	desc = {
		text = EVTDESCcathayraid3_engineer
		trigger = { has_character_flag = cathayraid_engineer }
	}
	desc = {
		text = EVTDESCcathayraid3_treasure
		trigger = { has_character_flag = cathayraid_treasure }
	}
	desc = { text = EVTDESCcathayraid3_panda trigger = { has_character_flag = cathayraid_panda } }

	immediate = {
		if = { limit = { has_character_flag = cathayraid_captive } cathayraid_captive_effect = yes }
		else_if = {
			limit = { has_character_flag = cathayraid_engineer }
			cathayraid_engineer_effect = yes
		}
	}

	# Plain loot
	option = {
		name = EVTOPTAcathayraid3_loot
		trigger = { has_character_flag = cathayraid_loot }
		prestige = 50
		cathayraid_spoils_effect = yes
	}

	# Rich loot
	option = {
		name = EVTOPTAcathayraid3_rich
		trigger = { has_character_flag = cathayraid_rich }
		prestige = 150
		cathayraid_spoils_effect = yes
	}

	# A mandarin ransomed
	option = {
		name = EVTOPTAcathayraid3_ransom
		trigger = { has_character_flag = cathayraid_ransom }
		prestige = 300
		cathayraid_spoils_effect = yes
	}

	# Disaster
	option = {
		name = EVTOPTAcathayraid3_bad
		trigger = { has_character_flag = cathayraid_bad }
		prestige = -50
	}

	# A captive: kept at court
	option = {
		name = {
			text = EVTOPTAcathayraid3_captive
			trigger = { event_target:cathayraid_captive = { is_female = yes } }
		}
		name = {
			text = EVTOPTAcathayraid3_captive_hostage
			trigger = { event_target:cathayraid_captive = { is_female = no } }
		}
		trigger = { has_character_flag = cathayraid_captive }
		prestige = 75
		cathayraid_spoils_effect = yes
		event_target:cathayraid_captive = {
			show_portrait = yes
			opinion = { modifier = opinion_abducted who = ROOT years = 10 }
		}
	}

	# A captive: taken as a concubine
	option = {
		name = EVTOPTBcathayraid3_captive
		trigger = {
			has_character_flag = cathayraid_captive
			is_woman_trigger = no
			is_no_females_race = no
			has_concubinage = yes
			NOT = { num_of_consorts = 3 }
			event_target:cathayraid_captive = { is_woman_trigger = yes }
		}
		prestige = 75
		cathayraid_spoils_effect = yes
		event_target:cathayraid_captive = {
			show_portrait = yes
			opinion = { modifier = opinion_abducted who = ROOT years = 10 }
			ROOT = { show_scope_change = no add_consort = PREV }
		}
	}

	# The siege engineer: kept at court
	option = {
		name = EVTOPTAcathayraid3_engineer
		trigger = { has_character_flag = cathayraid_engineer }
		prestige = 75
		cathayraid_spoils_effect = yes
		event_target:cathayraid_engineer = {
			show_portrait = yes
			opinion = { modifier = opinion_abducted who = ROOT years = 10 }
		}
	}

	# The siege engineer: made a commander at once
	option = {
		name = EVTOPTBcathayraid3_engineer
		trigger = { has_character_flag = cathayraid_engineer }
		prestige = 75
		cathayraid_spoils_effect = yes
		event_target:cathayraid_engineer = {
			show_portrait = yes
			opinion = { modifier = opinion_abducted who = ROOT years = 10 }
		}
		custom_tooltip = {
			text = EVTOPTBcathayraid3_engineer_tt
			cathayraid_engineer_commander_effect = yes
		}
	}

	# A treasure of Cathay
	option = {
		name = EVTOPTAcathayraid3_treasure
		trigger = { has_character_flag = cathayraid_treasure }
		prestige = 75
		cathayraid_spoils_effect = yes
		set_character_flag = cathayraid_treasure_taken
		custom_tooltip = {
			text = EVTOPTAcathayraid3_treasure_tt
			add_cathayan_artifact_effect = yes
		}
	}

	# Pandas
	option = {
		name = EVTOPTAcathayraid3_panda
		trigger = { has_character_flag = cathayraid_panda }
		prestige = 200
		cathayraid_spoils_effect = yes
		add_character_modifier = { name = panda_pet duration = -1 }
		hidden_effect = { character_event = { id = cathayraid.10 days = 7000 random = 2000 } }
	}

	after = {
		cathayraid_clear_outcome_effect = yes
	}
}

# cathayraid.5 A warning from the Viceroy
# Letter sent when a lord starts raiding Cathay: the Jade Viceroy warns that
# the Dragon Throne will not suffer it for long.
# Scope: the new raider; event_target:cathayraid_viceroy is the Viceroy.
letter_event = {
	id = cathayraid.5
	desc = EVTDESCcathayraid5
	border = GFX_event_letter_frame_diplomacy
	is_triggered_only = yes # the decision cathay_raid

	has_dlc = "Jade Dragon"

	option = {
		name = EVTOPTAcathayraid5
	}
}

# cathayraid.10 The pandas die
# Twenty-odd years after they were brought west, the pandas die without
# young. A lord may mourn them hard.
# Scope: the owner.
character_event = {
	id = cathayraid.10
	desc = EVTDESCcathayraid10
	picture = GFX_evt_bad_news
	border = GFX_event_normal_frame_religion
	is_triggered_only = yes # cathayraid.3

	has_dlc = "Jade Dragon"

	trigger = {
		has_character_modifier = panda_pet
	}

	option = {
		name = EVTOPTAcathayraid10
		remove_character_modifier = panda_pet
		prestige = 5
	}

	option = {
		name = EVTOPTBcathayraid10
		remove_character_modifier = panda_pet
		if = {
			limit = { NOT = { trait = depressed } }
			random = {
				chance = 50
				add_trait = depressed
				hidden_effect = {
					character_event = { id = traitnote.288 } # notify depressed
				}
			}
		}
	}
}
