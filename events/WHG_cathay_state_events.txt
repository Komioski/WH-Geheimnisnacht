###############################################################################
# The state of Cathay
# Namespace: cathaystate (was JD.10005 - 10006 in jd_china_succession_events
#   and JD.50000 - 50401, JD.30504 in jd_chinese_status_and_policy_events)
# Faction: Cathay & the East
#
# How Grand Cathay fares, as the west hears of it. Every year the empire
# may settle down, fall into unrest, famine or plague, enter a golden age,
# be invaded by the Hobgoblin hordes or the Man-chu horsemen of the Hung,
# or tear itself apart in civil war (the Chaos, ogre and Skaven wars are in
# wh_cathay_events.txt). Each change sets the Emperor's policy (open,
# isolationist or expansionist) and the worth of the roads east, and lords
# within reach read the news.
#
# The Celestial Dragon Emperor does not die and is never overthrown by
# these wars: invaders are paid off or driven back. Only once the Dragon
# Throne has fallen (to Chaos, cathay_chaos_takes_throne_effect, or an
# invasion war from the west) can a hobgoblin khan, a Man-chu khagan or a
# rebel general crown a new house. The empire then takes a new name, every
# favour, timer and pact with the old house lapses, and the tributaries of
# the old house are set free.
#
# A new house on the Dragon Throne        cathaystate.1 - cathaystate.2
# A new Emperor and policy                cathaystate.10 - cathaystate.16
# Status news and the roads east          cathaystate.20 - cathaystate.26
# Peace, unrest, famine, plague, glory    cathaystate.30 - cathaystate.34
# The Hobgoblin hordes                    cathaystate.40 - cathaystate.44
# The Man-chu horsemen                    cathaystate.50 - cathaystate.54
# Civil war                               cathaystate.60 - cathaystate.65
# The old house goes home                 cathaystate.70
#
# Callers: common/on_actions/00_on_actions.txt: on_offmap_ruler_changed (1,
# 10), on_startup (2, 26, 70), on_offmap_policy_changed (15, 25),
# on_offmap_status_changed (20, 25) and the random events of
# on_offmap_yearly_pulse (30 - 34, 40, 42, 50, 52, 60, 62).
# Shared code: cathaystate_* in WHG_cathay_state_effects.txt and
# WHG_cathay_state_triggers.txt.
# Loc: localisation/00_WHG_cathay_state_events.csv
#
# Dropped: the sacking of the western governor (he is the on-map King of
# Naparaya, so he stays), the real-world dynasty names, the startup rename
# of Cathay to Song or Ming, the 1200 date checks (the game starts in 2010),
# the debug logs (JD.31000, JD.30500 - 30503), the Borjigin rules (their
# dynasty ids belong to a Talabecland house in the mod), the grant of the
# Viceroy's lands to a victorious khan, the Nipponese and Indish emperors,
# and the commented templates.
# Fixed: the roads east (the vanilla Silk Road is empty in the mod); the
# foreign-emperor checks read the offmap power instead of its ruler; the
# Man-chu winner was always a Cathayan.
###############################################################################

namespace = cathaystate

# cathaystate.1 A new house on the Dragon Throne
# Hidden. Names the empire after the new house, unless the war that set it
# up already chose a name (rename_override). Ends the peace pacts, grace
# timers and trade contracts of the old house, frees its tributaries
# (was_tributary_of_cathay, read by the news in
# jd_chinese_status_and_policy_events) and wipes the kowtow honours.
# Scope: the new ruler of Cathay; FROM is the old ruler, FROMFROM Cathay.
character_event = {
	id = cathaystate.1
	hide_window = yes
	is_triggered_only = yes # on_offmap_ruler_changed
	offmap = only

	has_dlc = "Jade Dragon"

	trigger = {
		is_offmap_tag = offmap_cathay
		NOT = { dynasty = FROM }
	}

	immediate = {
		offmap_cathay = {
			if = {
				limit = { NOT = { any_player = { dynasty = ROOT } } }
				clr_offmap_flag = cathay_invaded_player_dynasty
			}
			if = {
				limit = { NOT = { has_offmap_flag = rename_override } }
				cathaystate_new_empire_name_effect = yes
			}
		}

		# Pacts with the old house end
		any_playable_ruler = {
			limit = {
				has_character_modifier = peace_deal_with_cathay
				OR = {
					ai = yes
					NOT = { has_offmap_news_enabled = FROMFROM }
					NOT = { is_within_diplo_range = ROOT }
				}
			}
			remove_character_modifier = peace_deal_with_cathay
		}

		# Grace timers and favours of the old house lapse
		any_playable_ruler = {
			limit = {
				OR = {
					has_character_modifier = cathayan_grace_send_gift_timer
					has_character_modifier = cathayan_grace_send_eunuch_timer
					has_character_modifier = cathayan_grace_send_concubine_timer
					has_character_modifier = cathayan_grace_tributary_timer
					has_character_modifier = jd_cathay_boon_cd
					has_character_modifier = cathay_force_open_cooldown
					has_character_modifier = angered_cathay_modifier
					has_character_modifier = cathayan_imperial_trade_contract
					has_character_modifier = cathay_raid_grace_cd
				}
			}
			remove_character_modifier = cathayan_grace_send_gift_timer
			remove_character_modifier = cathayan_grace_send_eunuch_timer
			remove_character_modifier = cathayan_grace_send_concubine_timer
			remove_character_modifier = cathayan_grace_tributary_timer
			remove_character_modifier = jd_cathay_boon_cd
			remove_character_modifier = cathay_force_open_cooldown
			remove_character_modifier = angered_cathay_modifier
			remove_character_modifier = cathayan_imperial_trade_contract
			remove_character_modifier = cathay_raid_grace_cd
		}

		# The old house's tributaries go free
		governor = {
			any_tributary = {
				limit = { is_tributary = { type = offmap } }
				remove_tributary = PREV
				set_character_flag = was_tributary_of_cathay
			}
		}

		# Kowtow honours were owed to the old house
		any_playable_ruler = {
			limit = {
				OR = {
					trait = kow_tow_completed_tier_1
					trait = kow_tow_completed_tier_2
					trait = kow_tow_completed_tier_3
				}
			}
			remove_trait = kow_tow_completed_tier_1
			remove_trait = kow_tow_completed_tier_2
			remove_trait = kow_tow_completed_tier_3
			clr_character_flag = went_on_kow_tow_pilgrimage
		}
	}
}

# cathaystate.2 Cathay's name and government at game start
# Hidden. Makes sure Cathay and its western governor run the Celestial
# bureaucracy, and that Cathay under the Dragon Emperor is called Cathay.
# Runs once per game.
# Scope: a player.
character_event = {
	id = cathaystate.2
	hide_window = yes
	is_triggered_only = yes # on_startup

	has_dlc = "Jade Dragon"

	trigger = {
		ai = no
		NOT = { has_global_flag = cathaystate_startup_done }
	}

	immediate = {
		set_global_flag = cathaystate_startup_done
		offmap_cathay = {
			set_government_type = confucian_bureaucracy
			governor = { set_government_type = confucian_bureaucracy }
			if = {
				limit = {
					offmap_ruler = { trait = creature_celdragon }
					NOT = { has_offmap_name = cathay_cathay }
				}
				set_offmap_name = cathay_cathay
			}
		}
	}
}

# cathaystate.10 A new Emperor picks a policy
# Hidden. The new ruler of Cathay chooses to keep the empire open, close it
# or turn expansionist, swayed by its state, by trends the last reign or
# war left (cathay_isolationist_trend, cathay_expansionist_trend) and by
# past defeats. Lords within reach hear of the new reign (cathaystate.11).
# Scope: the new ruler; FROM is the old ruler, FROMFROM Cathay.
character_event = {
	id = cathaystate.10
	hide_window = yes
	is_triggered_only = yes # on_offmap_ruler_changed
	offmap = only

	has_dlc = "Jade Dragon"

	trigger = {
		is_offmap_tag = offmap_cathay
	}

	immediate = {
		FROMFROM = {
			set_offmap_flag = no_policy_news
			random_list = {
				100 = {
					modifier = { factor = 1.5 has_status = cathay_stable }
					modifier = {
						factor = 5
						has_policy = cathay_open
						NOT = { days_since_policy_change = 10950 } # 30 years
					}
					set_policy = cathay_open
				}
				20 = {
					trigger = {
						OR = {
							NOT = { has_offmap_tmp_flag = cathay_forced_open }
							had_offmap_tmp_flag = { flag = cathay_forced_open years = 50 }
						}
					}
					modifier = { factor = 10 has_offmap_flag = cathay_isolationist_trend }
					modifier = { factor = 2 cathay_is_suffering_trigger = yes }
					modifier = { factor = 0.5 has_status = cathay_golden_age }
					modifier = {
						factor = 5
						has_policy = cathay_isolationist
						NOT = { days_since_policy_change = 10950 } # 30 years
					}
					set_policy = cathay_isolationist
					clr_offmap_flag = cathay_isolationist_trend
					clr_offmap_tmp_flag = cathay_forced_open
				}
				10 = {
					trigger = {
						NOT = { has_offmap_flag = cathay_badly_beaten_3 }
						NOT = {
							check_variable = {
								which = global_cathay_change_to_expansionist_count
								value = 6
							}
						}
					}
					modifier = {
						factor = 3
						OR = {
							cathay_colony_in_danger_trigger = yes
							cathay_exile_at_large_trigger = yes
						}
					}
					modifier = { factor = 20 has_offmap_flag = cathay_expansionist_trend }
					modifier = { factor = 2 has_status = cathay_golden_age }
					modifier = { factor = 0.75 has_offmap_flag = cathay_badly_beaten_1 }
					modifier = { factor = 0.5 has_offmap_flag = cathay_badly_beaten_2 }
					modifier = {
						factor = 10
						has_policy = cathay_expansionist
						NOT = { days_since_policy_change = 14600 } # 40 years
					}
					modifier = {
						factor = 0.1
						has_policy = cathay_expansionist
						days_since_policy_change = 32850 # 90 years
					}
					modifier = {
						factor = 0.2
						check_variable = {
							which = global_cathay_change_to_expansionist_count
							value = 3
						}
					}
					if = {
						limit = { NOT = { has_policy = cathay_expansionist } }
						change_variable = {
							which = global_cathay_change_to_expansionist_count
							value = 1
						}
					}
					set_policy = cathay_expansionist
					clr_offmap_flag = cathay_expansionist_trend
				}
			}
		}
		if = {
			limit = { FROMFROM = { NOT = { has_offmap_flag = no_succession_news } } }
			any_player = {
				limit = { has_offmap_news_enabled = FROMFROM is_within_diplo_range = ROOT }
				narrative_event = { id = cathaystate.11 }
			}
		}
		FROMFROM = { clr_offmap_flag = no_policy_news clr_offmap_flag = no_succession_news }
	}
}

			is_offmap_tag = offmap_cathay

		factor = 1

# cathaystate.42 The Hobgoblin war ends
# Hidden. The longer the war, the likelier its end. Cathay wins (likelier
# after a golden age) or the hordes win (likelier after hardship, or when a
# hobgoblin khan of the map leads them). A victorious horde is paid off by
# the Dragon Emperor; once the Dragon Throne has fallen, the khan or his
# kin takes it. Exiles may flee west afterwards.
# Scope: the Jade Viceroy; FROM is Cathay.

	id = cathaystate.42
	hide_window = yes
	is_triggered_only = yes # on_offmap_yearly_pulse random

	has_dlc = "Jade Dragon"

	trigger = {
		FROM = { is_offmap_tag = offmap_cathay has_status = cathay_mongol_invasion }
	}

	weight_multiplier = {
		factor = 1
		modifier = {
			factor = 2
			FROM = { had_offmap_tmp_flag = { flag = cathay_mongol_invasion years = 2 } }
		}
		modifier = {
			factor = 2
			FROM = { had_offmap_tmp_flag = { flag = cathay_mongol_invasion years = 4 } }
		}
		modifier = {
			factor = 2
			FROM = { had_offmap_tmp_flag = { flag = cathay_mongol_invasion years = 8 } }
		}
		modifier = {
			factor = 5
			FROM = { had_offmap_tmp_flag = { flag = cathay_mongol_invasion years = 10 } }
		}
	}

	immediate = {
		random_independent_ruler = {
			limit = { cathaystate_hobgoblin_khan_trigger = yes }
			preferred_limit = { real_tier = EMPEROR }
			preferred_limit = { real_tier = KING }
			save_event_target_as = cathaystate_khan
		}
		random_list = {
			100 = { # Cathay wins
				modifier = { factor = 3 FROM = { prev_status = cathay_golden_age } }
				FROM = { cathaystate_invasion_beaten_effect = yes }
				any_player = {
					limit = { has_offmap_news_enabled = FROM is_within_diplo_range = ROOT }
					narrative_event = { id = cathaystate.43 }
				}
			}
			100 = { # The hordes win
				modifier = { factor = 3 FROM = { cathaystate_prev_hardship_trigger = yes } }
				modifier = { factor = 2 event_target:cathaystate_khan = { always = yes } }
				FROM = {
					cathaystate_invasion_won_effect = yes
					set_offmap_flag = cathay_expansionist_trend
				}
				if = {
					limit = { dragon_emperor_reigns_trigger = no }
					FROM = {
						set_offmap_flag = no_succession_news
						set_offmap_flag = cathaystate_heir_hobgoblin
					}
					cathaystate_new_emperor_effect = yes
				}
				event_target:cathaystate_khan = { cathaystate_khan_reward_effect = yes }
				any_player = {
					limit = { has_offmap_news_enabled = FROM is_within_diplo_range = ROOT }
					narrative_event = { id = cathaystate.44 }
				}
				FROM = { clr_offmap_flag = no_succession_news }
				cathaystate_exiles_flee_effect = yes
			}
		}
	}
}

# cathaystate.62 The civil war ends
# Hidden. After at least two years, likelier the longer it lasts. The
# loyalists win (always under the Dragon Emperor), or, once the Dragon
# Throne has fallen, a rebel general founds a new house (likelier against
# a foreign Emperor) or a pretender of the old house takes the throne.
# Rebel generals and exiles may head west afterwards.
# Scope: the Jade Viceroy; FROM is Cathay.
character_event = {
	id = cathaystate.62
	hide_window = yes
	is_triggered_only = yes # on_offmap_yearly_pulse random

	has_dlc = "Jade Dragon"

	trigger = {
		FROM = {
			is_offmap_tag = offmap_cathay
			has_status = cathay_civil_war
			had_offmap_tmp_flag = { flag = cathay_civil_war years = 2 }
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = {
			factor = 2
			FROM = { had_offmap_tmp_flag = { flag = cathay_civil_war years = 3 } }
		}
		modifier = {
			factor = 2
			FROM = { had_offmap_tmp_flag = { flag = cathay_civil_war years = 5 } }
		}
		modifier = {
			factor = 2
			FROM = { had_offmap_tmp_flag = { flag = cathay_civil_war years = 8 } }
		}
		modifier = {
			factor = 5
			FROM = { had_offmap_tmp_flag = { flag = cathay_civil_war years = 10 } }
		}
	}

	immediate = {
		FROM = { clr_offmap_flag = cathay_civil_war }
		random_list = {
			70 = { # The loyalists win
				FROM = {
					set_offmap_flag = no_status_news
					set_status = cathay_stable
					clr_offmap_flag = no_status_news
				}
				any_player = {
					limit = { has_offmap_news_enabled = FROM is_within_diplo_range = ROOT }
					narrative_event = { id = cathaystate.63 }
				}
				if = {
					limit = { NOT = { has_game_rule = { name = cathayan_invasions value = none } } }
					random_list = {
						50 = {
							character_event = { id = cathayinv.22 days = 15 } # a Man-chu warlord
							set_character_flag = spawning_a_jurchen_invader
						}
						50 = {
							trigger = {
								OR = {
									NOT = { has_global_flag = rebel_general_invasion_success }
									had_global_flag = {
										flag = rebel_general_invasion_success
										days = 54750
									}
								}
							}
							character_event = { id = cathayinv.40 days = 15 } # a rebel general
							set_character_flag = spawning_a_rebel_general
						}
					}
				}
			}
			100 = { # A rebel general founds a new house
				modifier = { factor = 0 dragon_emperor_reigns_trigger = yes }
				modifier = { factor = 2.5 FROM = { cathaystate_foreign_emperor_trigger = yes } }
				FROM = {
					set_offmap_flag = no_status_news
					random_list = {
						50 = { set_status = cathay_stable }
						50 = { set_status = cathay_unrest set_offmap_flag = cathay_had_unrest }
					}
					clr_offmap_flag = no_status_news
					set_offmap_flag = no_succession_news
					if = {
						limit = { cathaystate_foreign_emperor_trigger = yes }
						set_offmap_flag = foreign_dynasty_ousted
					}
				}
				cathaystate_new_emperor_effect = yes
				any_player = {
					limit = { has_offmap_news_enabled = FROM is_within_diplo_range = ROOT }
					narrative_event = { id = cathaystate.64 }
				}
				FROM = {
					clr_offmap_flag = foreign_dynasty_ousted
					clr_offmap_flag = no_succession_news
				}
				cathaystate_exiles_flee_effect = yes
			}
			70 = { # A pretender of the old house wins
				modifier = { factor = 0 dragon_emperor_reigns_trigger = yes }
				modifier = { factor = 0.5 FROM = { cathaystate_foreign_emperor_trigger = yes } }
				FROM = {
					set_offmap_flag = no_status_news
					set_status = cathay_stable
					clr_offmap_flag = no_status_news
					set_offmap_flag = no_succession_news
					offmap_ruler = {
						if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
						else = { death = { death_reason = death_offmap } }
					}
				}
				any_player = {
					limit = { has_offmap_news_enabled = FROM is_within_diplo_range = ROOT }
					narrative_event = { id = cathaystate.65 days = 1 }
				}
				if = {
					limit = { NOT = { has_game_rule = { name = cathayan_invasions value = none } } }
					random_list = {
						90 = {
							character_event = { id = cathayinv.22 days = 15 } # a Man-chu warlord
							set_character_flag = spawning_a_jurchen_invader
						}
						10 = { }
					}
				}
			}
		}
	}
}
