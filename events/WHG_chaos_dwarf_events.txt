###############################################################################
# Dawi Zharr: Hashut's furnaces, sorcerers, slaves and the arms trade
# Namespace: dawizharr (was "DR_chaos_dwarf" and "DR_order_dwarf" in
# DR_chaos_dwarfs_events.txt before the 2026 events refactor)
# Faction: Chaos Dwarfs
#
# The Chaos Dwarfs of the Dark Lands feed slaves and captives to the brazen
# furnaces of Hashut, the Father of Darkness, and he answers with gifts, curses
# or silence. Their Sorcerer-Prophets rule through his fire until his curse
# turns them to stone. Their raids fill the slave factories and levy pens of
# each capital, and their forges sell war machines to Chaos and greenskin
# warlords. The Dawi of the Karaks sell their own arms to the Old World through
# the same trade chain.
#
# Sections:
#   1-8    sacrifices to Hashut and the grand sacrifice
#   10-19  Hashut answers a grand sacrifice
#   20-23  Sorcerer-Prophets: ascension, the Sorcerer's Curse, bound K'daai
#   30-37  slaves: raids, the slave buildings, selling captives
#   40-45  the arms trade (Dawi Zharr and Dawi sellers)
#
# Callers:
#   dawizharr.1   sacrifice_to_hashut decision (wh_chaos_dwarf_decisions)
#   dawizharr.5   grand_sacrifice_hashut decision
#   dawizharr.20  on_five_year_pulse random list dawizharr_ascension
#   dawizharr.21  on_five_year_pulse random list dawizharr_curse, antideath_effect
#   dawizharr.22-23  daemon summoning in WHG_daemon_events (daemon.45, daemon.144)
#   dawizharr.30  on_siege_over_winner (wh_on_actions)
#   dawizharr.31  on_yearly_pulse (wh_on_actions)
#   dawizharr.33-34  on_decade_pulse random list dawizharr_decay
#   dawizharr.36  chaos_dwarf_sell_into_slavery decision
#   dawizharr.40  buy_from_chaos_dwarfs and buy_from_dwarfs decisions, chaoshorde.21
#   dawizharr.45  sell_stuff_chaos_dwarfs decision
#
# Shared code:
#   common/scripted_triggers/wh_dawizharr_triggers.txt
#   common/scripted_effects/wh_dawizharr_effects.txt
#   hashut_favor_* effects in common/scripted_effects/wh_scripted_effects.txt
#   artifacts dawizharr_bloodhashut and dawizharr_greatshield
#   (common/artifacts/20_artifacts_dawi_zharr.txt)
# Localisation: localisation/00_WHG_chaos_dwarf_events.csv
# Full list with callers: docs/events/event_database.csv

###############################################################################
# 1-8 Sacrifices to Hashut
# A prisoner can be thrown into the furnace at any time (sacrifice_to_hashut
# decision). Once every four years at least Hashut wants an offering: a
# prisoner, a slave building's worth of slaves, or the ruler's standing with
# vassals and priests (dawizharr.8). The grand sacrifice (dawizharr.5) costs
# gold and draws an answer from Hashut (dawizharr.10-19).
# The flag making_grand_sacrifice_hashut marks a grand rite in progress; the
# answer events clear it.
###############################################################################

# dawizharr.1 Into the furnace
# A prisoner is thrown into Hashut's furnace. A Dawi of the Ancestor Gods is the
# prize offering, then a captive lord, then anyone else. Outside a grand rite
# each offering adds a point of Hashut's favour; in a grand rite Hashut answers.
# Scope: ROOT is the sacrificing ruler, FROM the offering.
long_character_event = {
	id = dawizharr.1
	desc = {
		trigger = { NOT = { has_character_flag = making_grand_sacrifice_hashut } }
		text = EVTDESCdawizharr1
	}
	desc = {
		trigger = { has_character_flag = making_grand_sacrifice_hashut }
		text = EVTDESCdawizharr1b
	}
	picture = GFX_evt_altar_hashut

	is_triggered_only = yes

	immediate = {
		clr_character_flag = sacrificing_prisoner_hashut
	}

	option = { # A Dawi of the Ancestor Gods
		name = EVTOPTAdawizharr1
		trigger = { FROM = { trait = creature_dwarf } }
		piety = 150
		dawizharr_sacrifice_made_effect = yes
		hidden_effect = { FROM = { character_event = { id = dawizharr.2 } } }
		if = {
			limit = { has_character_flag = making_grand_sacrifice_hashut }
			piety = 100
			dawizharr_answer_prized_effect = yes
		}
		else = { hashut_favor_trivial_bonus_effect = yes }
	}
	option = { # A captive lord
		name = EVTOPTBdawizharr1
		trigger = { FROM = { is_ruler = yes NOT = { trait = creature_dwarf } } }
		piety = 100
		dawizharr_sacrifice_made_effect = yes
		hidden_effect = { FROM = { character_event = { id = dawizharr.2 } } }
		if = {
			limit = { has_character_flag = making_grand_sacrifice_hashut }
			dawizharr_answer_fair_effect = yes
		}
		else = { hashut_favor_trivial_bonus_effect = yes }
	}
	option = { # Anyone else
		name = EVTOPTCdawizharr1
		trigger = { FROM = { is_ruler = no NOT = { trait = creature_dwarf } } }
		piety = 50
		dawizharr_sacrifice_made_effect = yes
		hidden_effect = { FROM = { character_event = { id = dawizharr.2 } } }
		if = {
			limit = { has_character_flag = making_grand_sacrifice_hashut }
			dawizharr_answer_poor_effect = yes
		}
		else = { hashut_favor_trivial_bonus_effect = yes }
	}
}

# dawizharr.2 The offering burns
# The offering dies in the molten metal. Its dynasty hates the sacrificer, and
# killing close kin makes the sacrificer a kinslayer. The truly undying cheat
# the furnace.
# Scope: ROOT is the offering, FROM the sacrificing ruler.
character_event = {
	id = dawizharr.2
	desc = EVTDESCdawizharr2
	picture = GFX_evt_altar_hashut

	is_triggered_only = yes

	option = {
		name = EVTOPTAdawizharr2
		if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
		else = {
			any_dynasty_member = {
				limit = { NOT = { character = FROM } }
				opinion = { modifier = opinion_sacrificed_family who = FROM }
			}
			if = { limit = { is_close_relative = FROM } FROM = { add_trait = kinslayer } }
			death = { death_reason = death_sacrificed killer = FROM }
		}
	}
}

# dawizharr.5 The grand sacrifice
# The acolytes ready the ziggurat's furnaces. The ruler offers the slaves of a
# factory or of the levy pens (each costs a level of that building), or a
# prisoner. The rite costs 200 gold and Hashut will not hear another for about
# four years.
# Scope: ROOT is the ruler. Fired by decision grand_sacrifice_hashut.
long_character_event = {
	id = dawizharr.5
	desc = EVTDESCdawizharr5
	picture = GFX_evt_hashut_rune

	is_triggered_only = yes

	option = { # Slaves from the factory
		name = EVTOPTAdawizharr5
		trigger = { dawizharr_factory_trigger = yes }
		wealth = -200
		add_character_modifier = { name = recently_asked_hashut duration = 1500 }
		custom_tooltip = { text = dawizharr_factory_down_tt dawizharr_factory_down_effect = yes }
		character_event = { id = dawizharr.6 }
		ai_chance = { factor = 50 }
	}
	option = { # Slaves from the levy pens
		name = EVTOPTBdawizharr5
		trigger = { dawizharr_levy_trigger = yes }
		wealth = -200
		add_character_modifier = { name = recently_asked_hashut duration = 1500 }
		custom_tooltip = { text = dawizharr_levy_down_tt dawizharr_levy_down_effect = yes }
		character_event = { id = dawizharr.6 }
		ai_chance = { factor = 50 }
	}
	option = { # A prisoner
		name = EVTOPTCdawizharr5
		trigger = { num_of_prisoners = 1 }
		wealth = -200
		add_character_modifier = { name = recently_asked_hashut duration = 1500 }
		hidden_effect = {
			random_courtier = {
				limit = { prisoner = yes host = { character = ROOT } }
				character_event = { id = dawizharr.3 }
			}
		}
		ai_chance = { factor = 100 }
	}
	option = { # Call it off
		name = EVTOPTDdawizharr5
		piety = -100
		clr_character_flag = making_grand_sacrifice_hashut
		ai_chance = { factor = 0 }
	}
}

# 10-19 Hashut answers
# Rolled by the dawizharr_answer_<tier>_effect lists after a grand sacrifice.

# way to becoming a Sorcerer-Prophet) and clears making_grand_sacrifice_hashut.
###############################################################################

# dawizharr.10 Hashut offers gifts
# The brazen idol speaks and offers a choice: K'daai to burn the ruler's foes,
# a body like a Bull Centaur's, Hashut's fire on the land, or (for a sorcerer)
# deeper mastery of the Winds.
# Scope: ROOT is the ruler.
character_event = {
	id = dawizharr.10
	desc = EVTDESCdawizharr10
	picture = GFX_evt_hashut_rune

	is_triggered_only = yes

	immediate = {
		clr_character_flag = making_grand_sacrifice_hashut
		add_doom_progress_tiny_effect = yes
		change_variable = { which = hashut_favor value = 4 }
	}

	option = { # K'daai host
		name = EVTOPTAdawizharr10
		capital_scope = {
			ROOT = {
				spawn_unit = {
					province = PREV
					home = PREV
					owner = THIS
					attrition = 0.5
					cannot_inherit = yes
					earmark = demonic_host
					troops = { daemonic_host = { 350 350 } }
				}
				spawn_unit = {
					province = PREV
					home = PREV
					owner = THIS
					attrition = 0.5
					cannot_inherit = yes
					earmark = demonic_host
					troops = { daemonic_host = { 250 250 } }
				}
			}
		}
	}
	option = { # Strength
		name = EVTOPTBdawizharr10
		change_martial = 2
		if = { limit = { trait = weak } remove_trait = weak add_trait = hardy }
		else_if = { limit = { trait = fragile } remove_trait = fragile add_trait = hardy }
		else_if = { limit = { trait = soft } remove_trait = soft add_trait = tough }
		else_if = { limit = { trait = tough } remove_trait = tough add_trait = strong }
		else_if = { limit = { trait = hardy } remove_trait = hardy add_trait = strong }
		else_if = { limit = { NOT = { trait = strong } } add_trait = tough }
	}
	option = { # Fire on the land
		name = EVTOPTCdawizharr10
		trigger = { capital_scope = { NOT = { has_province_modifier = hashut_province_blessing } } }
		capital_scope = {
			add_province_modifier = { name = hashut_province_blessing duration = 1500 }
		}
	}
	option = { # Mastery of the Winds
		name = EVTOPTDdawizharr10
		trigger = { trait = chaos_dwarf_sorcerer }
		change_learning = 1
		magic_lvl_increase_effect = yes
		enable_magic_xp_effect = yes
	}
}

# dawizharr.20 A Sorcerer-Prophet of Hashut
# Hashut grants a devoted servant mastery of the Winds of Magic. About one
# chance in four per five years, more for the pious.
# Scope: ROOT is the character. on_five_year_pulse list dawizharr_ascension.
character_event = {
	id = dawizharr.20
	desc = EVTDESCdawizharr20
	picture = GFX_evt_chaos_dwarf_sorcerer

	is_triggered_only = yes
	capable_only = yes
	min_age = 16
	religion = hashut
	culture_group = dwarf_chaos_group

	trigger = {
		trait = creature_chaos_dwarf
		NOT = { trait = chaos_dwarf_sorcerer }
		OR = { trait = proud trait = cruel trait = greedy trait = wroth }
		check_variable = { which = hashut_favor value = 30 }
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 1.33 piety = 750 }
		modifier = { factor = 1.33 piety = 1500 }
		modifier = { factor = 2 piety = 2500 }
	}

	option = {
		name = EVTOPTAdawizharr20
		add_doom_progress_minor_effect = yes
		prestige = 750
		piety = 450
		add_trait = chaos_dwarf_sorcerer
		magic_lvl_increase_effect = yes
		enable_magic_xp_effect = yes
	}
}

# brings a Hellforged Greatshield for the buyer (dawizharr_greatshield).
# Scope: ROOT is the buyer, FROM the seller.
character_event = {
	id = dawizharr.44
	desc = {
		trigger = {
			has_character_flag = arms_order_artillery
			FROM = { NOT = { dawizharr_dawi_seller_trigger = yes } }
		}
		text = EVTDESCdawizharr44
	}
	desc = {
		trigger = {
			has_character_flag = arms_order_weapons
			FROM = { NOT = { dawizharr_dawi_seller_trigger = yes } }
		}
		text = EVTDESCdawizharr44b
	}
	desc = {
		trigger = {
			has_character_flag = arms_order_artillery
			FROM = { dawizharr_dawi_seller_trigger = yes }
		}
		text = EVTDESCdawizharr44c
	}
	desc = {
		trigger = {
			has_character_flag = arms_order_weapons
			FROM = { dawizharr_dawi_seller_trigger = yes }
		}
		text = EVTDESCdawizharr44d
	}
	picture = GFX_evt_chaos_dwarf_artillery_crew

	is_triggered_only = yes

	option = { # War machines
		name = EVTOPTAdawizharr44
		trigger = { has_character_flag = arms_order_artillery }
		custom_tooltip = {
			text = dawizharr_artillery_tt
			capital_scope = {
				ROOT = {
					spawn_unit = {
						province = PREV
						owner = THIS
						troops = { heavy_infantry = { 100 100 } artillery = { 50 50 } }
					}
				}
			}
		}
		if = {
			limit = { FROM = { dawizharr_dawi_seller_trigger = yes } }
			add_character_modifier = { name = recently_brought_from_order_dwarfs duration = 3650 }
		}
		else = {
			add_character_modifier = { name = recently_brought_from_chaos_dwarfs duration = 3650 }
		}
		dawizharr_arms_clear_buyer_effect = yes
	}
	option = { # Arms and armour
		name = EVTOPTBdawizharr44
		trigger = { has_character_flag = arms_order_weapons }
		if = {
			limit = { FROM = { dawizharr_dawi_seller_trigger = yes } }
			add_character_modifier = { name = recently_brought_from_order_dwarfs duration = 3650 }
			capital_scope = {
				add_province_modifier = { name = order_dwarf_weapons duration = 3650 }
			}
		}
		else = {
			add_character_modifier = { name = recently_brought_from_chaos_dwarfs duration = 3650 }
			capital_scope = {
				add_province_modifier = { name = chaos_dwarf_weapons duration = 3650 }
			}
			if = {
				limit = { NOT = { has_artifact = dawizharr_greatshield } }
				add_artifact = dawizharr_greatshield
			}
		}
		dawizharr_arms_clear_buyer_effect = yes
	}
}
