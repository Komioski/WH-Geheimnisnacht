# Chaos hordes: settlements, Karag Dum, sacking and break-up
# Namespace: chaoshorde (was "chaosr" in wh_chaos_razing_seige_events.txt
# before the 2026 events refactor)
# Faction: Chaos
#
# The life of a Chaos horde of the Wastes: the camps it raises, the fall of
# the dwarf hold of Karag Dum, what its warriors do to a captured holding,
# and how it shatters when its warlord dies.
#
# Sections:
#   1-9    a semi-nomadic settlement grows into a castle or a city
#   20-22  the fate of Karag Dum
#   30-42  sacking captured holdings
#   50-56  the horde shatters on the warlord's death
#
# Callers:
#   chaoshorde.2   establish_tribe decision (wh_chaos_decisions)
#   chaoshorde.20  on_siege_over_winner (wh_on_actions)
#   chaoshorde.30-32, 39, 42  on_siege_over_winner random list (wh_on_actions)
#   chaoshorde.50  on_death (wh_on_actions)
#
# Shared code:
#   common/scripted_triggers/wh_chaoshorde_triggers.txt
#   common/scripted_effects/wh_chaoshorde_effects.txt
#   the settlement progress effects in common/scripted_effects/wh_scripted_effects.txt

namespace = chaoshorde

###############################################################################
# 1-9 Semi-nomadic settlements
# A horde ruler of the Wastes plants a tribal camp under a chieftain
# (establish_tribe decision, then chaoshorde.2). Every few months chaoshorde.1
# rolls what happens to it; good and bad turns move a progress counter on the
# province. Once the counter reaches 6 the camp can become a castle or a city
# (chaoshorde.4).
#
# Event targets kept along the chain:
#   chaos_chief                the chieftain
#   chaos_settlement           the tribal holding
#   chaos_settlement_province  its province
###############################################################################

# chaoshorde.1 The settlement grows
# Hidden. Rolls the next turn in the settlement's life. If the camp is held
# above baron rank (no chieftain to govern it) it is warned for two years and
# then abandoned (chaoshorde.9).
# Scope: ROOT is the horde ruler.
character_event = {
	id = chaoshorde.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		event_target:chaos_settlement_province = { owner = { character = ROOT } }
		event_target:chaos_settlement = { holding_type = tribal num_of_buildings >= 1 }
	}

	fail_trigger_effect = {
		event_target:chaos_settlement_province = { chaoshorde_settlement_reset_effect = yes }
	}

	immediate = {
		if = {
			limit = {
				event_target:chaos_settlement = { holder_scope = { higher_tier_than = BARON } }
			}
			if = {
				limit = {
					event_target:chaos_settlement_province = {
						has_province_flag = absence_of_governor2
					}
				}
				character_event = { id = chaoshorde.9 }
			}
			else_if = {
				limit = {
					event_target:chaos_settlement_province = {
						has_province_flag = absence_of_governor1
					}
				}
				event_target:chaos_settlement_province = {
					clr_province_flag = absence_of_governor1
					set_province_flag = absence_of_governor2
				}
				repeat_event = { id = chaoshorde.1 days = 365 }
			}
			else = {
				event_target:chaos_settlement_province = {
					set_province_flag = absence_of_governor1
				}
				repeat_event = { id = chaoshorde.1 days = 365 }
			}
		}
		else = {
			random_list = {
				15 = { # Clansmen flock to it
					trigger = {
						NOT = {
							event_target:chaos_settlement_province = {
								has_province_flag = development_is_done
							}
						}
					}
					add_1_point_to_chaos_settlement_progress_effect = yes
					evaluate_chaos_settlement_progress_effect = yes
					character_event = { id = chaoshorde.6 days = 365 random = 30 }
				}
				5 = { # A young warrior raises a band
					trigger = {
						NOT = {
							event_target:chaos_settlement_province = {
								has_province_flag = development_is_done
							}
						}
					}
					lose_1_point_to_chaos_settlement_progress_effect = yes
					evaluate_chaos_settlement_progress_effect = yes
					character_event = { id = chaoshorde.7 days = 360 }
				}
				10 = { # The chieftain is murdered
					trigger = {
						event_target:chaos_settlement_province = {
							NOT = { has_province_flag = start_bracket }
							NOT = { has_province_flag = development_is_done }
						}
					}
					lose_1_point_to_chaos_settlement_progress_effect = yes
					evaluate_chaos_settlement_progress_effect = yes
					character_event = { id = chaoshorde.8 days = 300 }
				}
				10 = { # The Wastes wreck a building
					trigger = {
						NOT = {
							event_target:chaos_settlement_province = {
								has_province_flag = development_is_done
							}
						}
					}
					lose_1_point_to_chaos_settlement_progress_effect = yes
					evaluate_chaos_settlement_progress_effect = yes
					character_event = { id = chaoshorde.3 days = 120 }
				}
				5 = { # Quiet growth (early)
					trigger = {
						event_target:chaos_settlement_province = {
							has_province_flag = start_bracket
						}
					}
					random = {
						chance = 75
						add_1_point_to_chaos_settlement_progress_effect = yes
						evaluate_chaos_settlement_progress_effect = yes
					}
					repeat_event = { id = chaoshorde.1 days = 60 }
				}
				10 = { # Quiet growth (middle)
					trigger = {
						event_target:chaos_settlement_province = { has_province_flag = mid_bracket }
					}
					random = {
						chance = 85
						add_1_point_to_chaos_settlement_progress_effect = yes
						evaluate_chaos_settlement_progress_effect = yes
					}
					repeat_event = { id = chaoshorde.1 days = 120 }
				}
				10 = { # Quiet growth (late)
					trigger = {
						event_target:chaos_settlement_province = {
							has_province_flag = top_bracket
							NOT = { has_province_flag = development_is_done }
						}
					}
					random = {
						chance = 85
						add_1_point_to_chaos_settlement_progress_effect = yes
						evaluate_chaos_settlement_progress_effect = yes
					}
					repeat_event = { id = chaoshorde.1 days = 240 }
				}
				10 = { # Ready to be fortified
					trigger = {
						event_target:chaos_settlement_province = {
							has_province_flag = development_is_done
						}
					}
					character_event = { id = chaoshorde.4 days = 62 }
				}
			}
		}
	}

	option = {
		name = OK
	}
}

# chaoshorde.2 A settlement is founded
# The chieftain raises a tribal camp with its first buildings, and the
# settlement chain starts (chaoshorde.1).
# Scope: ROOT is the province.
province_event = {
	id = chaoshorde.2
	desc = EVTDESCchaoshorde2
	picture = GFX_evt_chaos_marauder
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # establish_tribe decision

	option = {
		name = EXCELLENT
		hidden_effect = {
			set_variable = { which = chaos_settlement_event_progress value = 0 }
			set_variable = { which = chaos_settlement_development value = 0 }
		}
		save_event_target_as = chaos_settlement_province
		set_province_flag = start_bracket
		any_province_holding = {
			limit = { holding_type = tribal }
			add_building = tb_steppe_hold_1
			add_building = tb_market_town_1
			add_building = tb_steppechaos_barracks_1
			if = {
				limit = { event_target:chaos_chief = { culture = ROOT } }
				add_building = tb_noblestable_1
				add_building = tb_steppe_hold_2
			}
			if = {
				limit = { event_target:chaos_chief = { culture_group = kurgan_group } }
				add_building = tb_steppekurgan_militia_1
			}
			else_if = {
				limit = { event_target:chaos_chief = { culture_group = hung_group } }
				add_building = tb_steppehung_militia_1
			}
			save_event_target_as = chaos_settlement
		}
		owner = { character_event = { id = chaoshorde.1 days = 30 } }
	}
}

# chaoshorde.4 The settlement stands
# The camp has outlasted the Wastes: turn it into a city (trade) or a castle
# (war), with walls, barracks and a market or town. If the province already
# has both, the camp stays a camp.
# Scope: ROOT is the horde ruler.
character_event = {
	id = chaoshorde.4
	desc = EVTDESCchaoshorde4
	picture = GFX_evt_questing_darklands
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes

	option = {
		name = EVTOPTAchaoshorde4 # A city
		trigger = { event_target:chaos_settlement_province = { has_city = no } }
		event_target:chaos_settlement = {
			convert_to = city
			add_building = ct_wall_1
			add_building = ct_wall_2
			add_building = ct_wall_q_1
			add_building = ct_training_grounds_1
			add_building = ct_training_grounds_2
			add_building = ct_barracks_1
			add_building = ct_barracks_2
			add_building = ct_marketplace_1
		}
		event_target:chaos_settlement_province = { chaoshorde_settlement_reset_effect = yes }
		prestige = 250
		ai_chance = { factor = 1 }
	}
	option = {
		name = EVTOPTBchaoshorde4 # A Dark Fortress
		trigger = { event_target:chaos_settlement_province = { has_castle = no } }
		event_target:chaos_settlement = {
			convert_to = castle
			add_building = ca_wall_1
			add_building = ca_wall_2
			add_building = ca_wall_q_1
			add_building = ca_militia_barracks_1
			add_building = ca_militia_barracks_2
			add_building = ca_steppe_barracks_1
			add_building = ca_steppe_barracks_2
			add_building = ca_town_1
		}
		event_target:chaos_settlement_province = { chaoshorde_settlement_reset_effect = yes }
		prestige = 250
		ai_chance = { factor = 1 }
	}
	option = {
		name = EVTOPTCchaoshorde4 # Keep the camp
		trigger = { event_target:chaos_settlement_province = { has_city = yes has_castle = yes } }
		event_target:chaos_settlement = { add_building = tb_steppe_hold_2 }
		event_target:chaos_settlement_province = { chaoshorde_settlement_reset_effect = yes }
		prestige = 100
	}
}

# chaoshorde.20 Karag Dum falls
# Hidden. When a siege in the duchy of Karag Dum ends, the winner's top
# liege decides the hold's fate (chaoshorde.21) if they now hold all of it.
# Fired from on_siege_over_winner.
# Scope: ROOT is the siege winner, FROM the taken holding.
character_event = {
	id = chaoshorde.20
	hide_window = yes
	is_triggered_only = yes # on_siege_over_winner

	trigger = {
		NOT = { has_global_flag = fate_of_karakdum }
		FROM = { de_jure_liege_or_above = d_karak_dum }
	}

	immediate = {
		top_liege = { narrative_event = { id = chaoshorde.21 days = 1 } }
	}
}

			limit = { religion = nurgle }

# 50-56 The horde shatters
# When an independent Chaos horde warlord dies, the realm's stability law
# decides how much breaks away. Low stability: every duchy outside the
# capital's that borders a foreign realm may break off under a new warlord
# (chaoshorde.51). Middling: every kingdom outside the capital's (chaoshorde.56).
# The heir keeps more the more prestige they have. Settled vassals go free
# either way. Vassals choosing independence is the vanilla nomad rule
# (GR.10/GR.11), which also runs for Chaos hordes.
###############################################################################

# chaoshorde.50 The warlord is dead
# Hidden. Starts the break-up for the heir.
# Fired from on_death.
# Scope: ROOT is the dead warlord.
character_event = {
	id = chaoshorde.50
	hide_window = yes
	is_triggered_only = yes # on_death

	only_playable = yes
	only_independent = yes
	religion_group = chaos_gods_group

	trigger = {
		is_nomadic = yes
		government = chaos_horde_government
		current_heir = { always = yes }
	}

	immediate = {
		current_heir = { save_event_target_as = nomadic_heir }
		if = {
			limit = { has_law = nomad_stability_law_1 }
			capital_scope = {
				duchy = { save_event_target_as = capital_duchy }
				kingdom = { save_event_target_as = capital_kingdom }
			}
			event_target:nomadic_heir = { character_event = { id = chaoshorde.56 days = 1 } }
			chaoshorde_free_settled_vassals_effect = yes
		}
		else_if = {
			limit = { has_law = nomad_stability_law_0 }
			capital_scope = { duchy = { save_event_target_as = capital_duchy } }
			event_target:nomadic_heir = { character_event = { id = chaoshorde.51 days = 1 } }
			chaoshorde_free_settled_vassals_effect = yes
		}
	}
}

			duchy = { NOT = { title = event_target:capital_duchy } }
			any_neighbor_province = { owner = { NOT = { same_realm = ROOT } } }
		}
	}

	fail_trigger_effect = { set_variable = { which = chaos_tribe_break_aways value = 0 } }

	immediate = {

			}
			duchy = { save_event_target_as = target_duchy }
			save_event_target_as = target_prov
		}
	}

	option = {
		name = EVTOPTAchaoshorde51 # Who is the boss here?
		prestige = 250
		event_target:target_prov = { province_event = { id = chaoshorde.52 } }

		event_target:target_prov = { province_event = { id = chaoshorde.52 } }

# chaoshorde.52 A warlord breaks away
# Hidden. A warlord of the local tribes (or a beastman) takes the province's
# county, and the heir's other counties in that duchy, as an independent
# ruler. Champions of the gods may go on to subjugate their neighbours
# (chaoshorde.53); others may bind one weaker warband (chaoshorde.54). Then
# the heir is asked again (chaoshorde.51 or 56).
# Scope: ROOT is the breakaway province.
province_event = {
	id = chaoshorde.52
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				event_target:target_duchy = { owner = { character = event_target:nomadic_heir } }
			}
			event_target:target_duchy = { destroy_landed_title = THIS }
		}
		chaoshorde_create_breakaway_warlord_effect = yes
		event_target:nomadic_heir = {
			opinion = { modifier = opinion_breakaway_nomad who = new_character years = 3 }
		}
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			ROOT = { county = { grant_title_no_opinion = { target = PREVPREV type = invasion } } }
			set_defacto_liege = THIS
			if = {
				limit = { culture_group = beastman_group }
				set_government_type = beastman_government
			}
			else_if = {
				limit = { is_tribal = yes }
				random_list = {
					75 = { set_government_type = chaos_horde_government }
					25 = {
						if = {
							limit = { event_target:target_duchy = { has_holder = no } }
							event_target:target_duchy = {
								grant_title_no_opinion = { target = PREV }
							}
						}
					}
				}
			}
			population = 4000
			if = {
				limit = { is_chaos_champion_trigger = yes }
				random_list = {
					35 = {
						set_character_flag = successful_chaos_warlord
						character_event = { id = chaoshorde.53 days = 30 }
						prestige = 750
						piety = 750
					}
					65 = { }
				}
			}
			random_list = {
				25 = { # A strong band (always for a rising champion)
					set_character_flag = successful_chaos_warlord
					character_event = { id = chaoshorde.54 days = 15 }
					population = 2000
					prestige = 800
					wealth = 120
					piety = 200
					create_retinue = chaosmarauders
					create_retinue = chaosmarauders
					create_retinue = chaosmarauderchampions
					create_retinue = chaosmarauderhorsemen
				}
				75 = { # A weak band that can be bound by a neighbour
					modifier = { factor = 0 has_character_flag = successful_chaos_warlord }
					set_character_flag = breakaway_chaos_warlord
					character_event = { id = chaoshorde.55 days = 50 }
					prestige = 500
					wealth = 80
					piety = 100
					create_retinue = chaosmarauders
					create_retinue = chaosmarauders
					create_retinue = chaosmarauderhorsemen
				}
			}
		}
		event_target:nomadic_heir = {
			any_realm_province = {
				limit = {
					NOT = { any_province_holding = { holding_type = nomad } }
					duchy = { title = event_target:target_duchy }
				}
				county = {
					new_character = { grant_title_no_opinion = { target = PREV type = invasion } }
				}
			}
		}
	}

	after = {
		event_target:nomadic_heir = {
			if = {
				limit = { event_target:capital_kingdom = { always = yes } }
				character_event = { id = chaoshorde.56 }
			}
			else = { character_event = { id = chaoshorde.51 } }
		}
	}
}

# not yet risen kneel to them.

	id = chaoshorde.53

	picture = GFX_evt_chaos_warband

	notification = yes

	trigger = {
		is_ruler = yes
		independent = yes
	}

	fail_trigger_effect = { clr_character_flag = successful_chaos_warlord }

		any_neighbor_independent_ruler = {
			limit = {
				has_character_flag = breakaway_chaos_warlord
				NOT = { has_character_flag = successful_chaos_warlord }

			set_defacto_liege = ROOT
		}
		clr_character_flag = successful_chaos_warlord
	}

	option = {
		name = EXCELLENT
	}
}
