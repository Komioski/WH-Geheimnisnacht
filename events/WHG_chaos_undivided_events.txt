# Chaos flavour events (this file also holds the Everchosen events, namespace
# undivided, further down; they came from DR_chaos_undivided_events.txt and,
# for the path to the mantle, BR_chaos_events.txt)
# Namespace: chaosfl (was "schaosfl" before the 2026 events refactor)
# Faction: Chaos
#
# Gifts and trials of the Chaos gods outside the cults: Khorne's blood
# tournament, where a lord's warriors fight to the death until one champion
# remains; the price Tzeentch asks for a daemon host; the diseases a servant
# of Nurgle bestows; the revels thrown for Slaanesh; the ascension of a
# daemon prince; and the doom that lifts when a chosen of Chaos dies.
#
# Khorne's blood tournament      chaosfl.1 - chaosfl.23
# Tzeentch's price               chaosfl.30 - chaosfl.31
# Nurgle's gifts                 chaosfl.40 - chaosfl.44
# Slaanesh's Revel of Excess     chaosfl.70 - chaosfl.74
# Daemon princes                 chaosfl.50
# Doom                           chaosfl.60
#
# Decisions in wh_chaos_decisions.txt start the tournament (chaosfl.1 and
# chaosfl.2), Tzeentch's sacrifice (chaosfl.30 or 31), Nurgle's gift
# (chaosfl.40), the Revel of Excess (chaosfl.70) and every daemon prince
# ascension (chaosfl.50).
# send_duel_result_tournament_effect in wh_scripted_effects.txt fires the bout
# results (chaosfl.12). common/on_actions/wh_on_actions.txt on_death fires
# chaosfl.60.
# Scripted effects: wh_chaosfl_effects.txt
# Artifacts: common/artifacts/27_artifacts_chaosfl.txt
# Loc: localisation/00_WHG_chaos_undivided_events.csv
# Full list with callers: docs/events/event_database.csv
#
# Also in this file (each part keeps its own header and namespace):
#   Dark gifts and the Eye of the Gods (namespace darkgift, was WHG_dark_gifts_events)
#   The Raven's Child (namespace ravenchild, was WHG_raven_child_events)
###############################################################################

# chaosfl.2 The challenge
# A Khornate vassal or courtier is called to fight. Accepting pleases Khorne
# and tells the host (chaosfl.3); refusing angers him (chaosfl.4).
# Scope: ROOT is the invited warrior, FROM the host.
letter_event = {
	id = chaosfl.2
	desc = EVTDESCchaosfl2

	is_triggered_only = yes

	option = { # I'll rip your heart out
		name = EVTOPTAchaosfl2
		add_offmap_currency = { offmap = khorne_warp value = 75 }
		set_character_flag = khornetournament_participant
		hidden_effect = { FROM = { letter_event = { id = chaosfl.3 days = 1 random = 1 } } }
		ai_chance = { factor = 70 }
	}
	option = { # Not worthy of my blade
		name = EVTOPTBchaosfl2
		add_offmap_currency = { offmap = khorne_warp value = -50 }
		hidden_effect = { FROM = { letter_event = { id = chaosfl.4 days = 1 random = 1 } } }
		ai_chance = {
			factor = 30
			modifier = { factor = 0.5 trait = ambitious }
			modifier = { factor = 0.5 trait = brave }
			modifier = { factor = 0.75 trait = wroth }
			modifier = { factor = 0.75 trait = proud }
			modifier = { factor = 2 trait = wounded }
			modifier = { factor = 2.5 trait = maimed }
			modifier = { factor = 2.5 trait = blinded }
			modifier = { factor = 2 trait = craven }
			modifier = { factor = 0.5 trait = champion_khorne }
			modifier = { factor = 0.15 trait = chosen_khorne }
			modifier = { factor = 0 trait = prince_khorne }
			modifier = { # Few have answered yet: harder to say no
				factor = 0.25
				NOT = { FROM = { check_variable = { which = "khtournament_player" value = 3 } } }
			}
		}
	}
}

# chaosfl.5 The plain of skulls
# With at least three challengers the tournament begins: everyone is called
# (chaosfl.6) and the host fights first (chaosfl.7). With fewer it is called
# off, at a cost in prestige and Khorne's favour (chaosfl.16).
# Scope: ROOT is the host.
long_character_event = {
	id = chaosfl.5
	picture = GFX_evt_chaos_champions

	desc = {
		trigger = { check_variable = { which = "khtournament_player" value = 3 } }
		text = EVTDESCchaosfl5
	}
	desc = {
		trigger = { NOT = { check_variable = { which = "khtournament_player" value = 3 } } }
		text = EVTDESCchaosfl5b
	}

	is_triggered_only = yes

	option = { # Blood and skulls!
		name = EVTOPTAchaosfl5
		trigger = { check_variable = { which = "khtournament_player" value = 3 } }
		add_trait = questing
		add_offmap_currency = { offmap = khorne_warp value = 50 }
		hidden_effect = {
			set_variable = { which = "khtournament_player" value = 0 }
			any_realm_character = {
				limit = {
					NOT = { has_character_flag = kh_tournament_host }
					has_character_flag = khornetournament_participant
				}
				long_character_event = { id = chaosfl.6 }
			}
			character_event = { id = chaosfl.7 }
		}
	}
	option = { # A waste of time
		name = EVTOPTBchaosfl5
		trigger = { NOT = { check_variable = { which = "khtournament_player" value = 3 } } }
		prestige = -250
		add_offmap_currency = { offmap = khorne_warp value = -25 }
		hidden_effect = {
			set_variable = { which = "khtournament_player" value = 0 }
			character_event = { id = chaosfl.17 }
			any_realm_character = {
				limit = {
					NOT = { has_character_flag = kh_tournament_host }
					has_character_flag = khornetournament_participant
				}
				character_event = { id = chaosfl.16 days = 7 random = 10 }
			}
		}
	}
}

# chaosfl.10 The bout
# Hidden. Sets up the duel (the opponent is combatant_1, the challenger
# combatant_2), with no draws, and picks one of eleven fight descriptions
# for the result events.
# Scope: ROOT is the challenger, FROM the opponent.
character_event = {
	id = chaosfl.10
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		save_event_target_as = combatant_2
		set_character_flag = flag_duel_no_draws
		random_list = {
			15 = { set_character_flag = kh_fight_1 FROM = { set_character_flag = kh_fight_1 } }
			15 = { set_character_flag = kh_fight_2 FROM = { set_character_flag = kh_fight_2 } }
			15 = { set_character_flag = kh_fight_3 FROM = { set_character_flag = kh_fight_3 } }
			15 = { set_character_flag = kh_fight_4 FROM = { set_character_flag = kh_fight_4 } }
			15 = { set_character_flag = kh_fight_5 FROM = { set_character_flag = kh_fight_5 } }
			15 = { set_character_flag = kh_fight_6 FROM = { set_character_flag = kh_fight_6 } }
			10 = { set_character_flag = kh_fight_7 FROM = { set_character_flag = kh_fight_7 } }
			10 = { set_character_flag = kh_fight_8 FROM = { set_character_flag = kh_fight_8 } }
			10 = { set_character_flag = kh_fight_9 FROM = { set_character_flag = kh_fight_9 } }
			10 = { set_character_flag = kh_fight_10 FROM = { set_character_flag = kh_fight_10 } }
			10 = { set_character_flag = kh_fight_11 FROM = { set_character_flag = kh_fight_11 } }
		}
		FROM = {
			save_event_target_as = combatant_1
			set_character_flag = flag_duel_no_draws
			character_event = { id = chaosfl.11 }
		}
	}
}

# chaosfl.13 Death in the pit
# The loser's last moments: they die by the winner's hand (unless
# invincible). The winner then faces the next challenger (chaosfl.14) or, if
# none is left, is crowned champion (chaosfl.15).
# Scope: ROOT is the loser, FROM the winner.
long_character_event = {
	id = chaosfl.13
	picture = GFX_evt_chaos_champions

	desc = { trigger = { FROM = { has_character_flag = kh_fight_1 } } text = EVTDESCchaosfl13 }
	desc = { trigger = { FROM = { has_character_flag = kh_fight_2 } } text = EVTDESCchaosfl13b }
	desc = { trigger = { FROM = { has_character_flag = kh_fight_3 } } text = EVTDESCchaosfl13c }
	desc = { trigger = { FROM = { has_character_flag = kh_fight_4 } } text = EVTDESCchaosfl13d }
	desc = { trigger = { FROM = { has_character_flag = kh_fight_5 } } text = EVTDESCchaosfl13e }
	desc = { trigger = { FROM = { has_character_flag = kh_fight_6 } } text = EVTDESCchaosfl13f }
	desc = { trigger = { FROM = { has_character_flag = kh_fight_7 } } text = EVTDESCchaosfl13g }
	desc = { trigger = { FROM = { has_character_flag = kh_fight_8 } } text = EVTDESCchaosfl13h }
	desc = { trigger = { FROM = { has_character_flag = kh_fight_9 } } text = EVTDESCchaosfl13i }
	desc = { trigger = { FROM = { has_character_flag = kh_fight_10 } } text = EVTDESCchaosfl13j }
	desc = { trigger = { FROM = { has_character_flag = kh_fight_11 } } text = EVTDESCchaosfl13k }

	is_triggered_only = yes

	immediate = {
		random = { chance = 50 add_trait = wounded }
		random = { chance = 5 add_trait = maimed }
	}

	option = {
		name = EVTOPTAchaosfl13
		add_offmap_currency = { offmap = khorne_warp value = -50 }
		clr_character_flag = khornetournament_participant
		set_character_flag = brawling_loser

		hidden_effect = {
			chaosfl_clear_fight_flags_effect = yes
			FROM = { chaosfl_clear_fight_flags_effect = yes }
			clear_event_target = combatant_1
			clear_event_target = combatant_2

			# Anyone left to fight? Checked before the loser dies.
			if = {
				limit = {
					FROM = {
						top_liege = {
							any_realm_character = {
								has_character_flag = khornetournament_participant
								NOT = { has_character_flag = brawling_loser }
								NOT = { character = ROOT }
								NOT = { character = FROM }
							}
						}
					}
				}
				FROM = { character_event = { id = chaosfl.14 } }
			}
			else = { FROM = { long_character_event = { id = chaosfl.15 } } }

			if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
			else = { death = { death_reason = death_duel killer = FROM } }
		}
	}
}

# 5 The horde swells
# Each year, while the horde is still a nomad horde on the march, more tribes
# join it: 80,000 people a year while it is below its population cap and
# under 350,000. The loop ends when the invading modifier runs out or the
# Everchosen settles.
# Scope: ROOT = the Everchosen.
character_event = {
	id = undivided.5
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				OR = {
					NOR = { government = nomadic_government government = chaos_horde_government }
					NOT = { has_character_modifier = invading_population_halt }
				}
			}
			remove_character_modifier = invading_population_halt
		}
		else = {
			if = {
				limit = { NOT = { population_factor = 1 } NOT = { population = 350000 } }
				population = 80000
			}
			repeat_event = { id = undivided.5 days = 365 }
		}
	}
}

				check_variable = { which = children_crusade_morale value = -3 }
				NOT = { check_variable = { which = children_crusade_morale value = 2 } }
			}
			add_character_modifier = { name = children_crusade_morale_3 duration = 1000 }
		}
		if = {
			limit = { check_variable = { which = children_crusade_morale value = 2 } }
			add_character_modifier = { name = children_crusade_morale_4 duration = 1000 }
		}
		set_variable = { which = children_crusade_progress value = 0 }
		set_variable = { which = children_crusade_forces value = 0 }
		set_variable = { which = children_crusade_heavy_troops value = 0 }
		set_variable = { which = children_crusade_morale value = 0 }

		c_kislev = {
			holder_scope = {
				event_target:childrens_crusade_leader = {
					unsafe_war = {
						target = PREV
						casus_belli = childrens_crusade
						thirdparty_title = PREVPREV
						tier = KING
					}
				}
			}
		}

		if = {
			limit = {
				persistent_event_target:childrens_crusade_sponsor_per = {
					has_character_flag = flag_childrens_crusade_sponsor_joins_war
				}
			}
			persistent_event_target:childrens_crusade_sponsor_per = {
				join_attacker_wars = ROOT
				join_defender_wars = ROOT
			}
		}
	}
