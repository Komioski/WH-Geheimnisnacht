# charfl.1 A giant's body settles
# A grown giant either trains the bulk into strength (charfl.3) or sinks
# under its weight (charfl.2). Learned and idle giants grow frail, warriors
# and tribesmen robust. Once per life.
# Scope: a giant, 16 to 50.
character_event = {
	id = charfl.1
	hide_window = yes

	is_triggered_only = yes

	min_age = 16
	max_age = 50
	capable_only = yes
	prisoner = no

	trigger = {
		trait = giant
		NOR = {
			trait = feeble
			trait = robust
			has_character_flag = giant_condition
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = {
			factor = 1.5
			OR = {
				trait = weak
				trait = strong
			}
		}
		modifier = {
			factor = 1.5
			age = 20
		}
		modifier = {
			factor = 1.5
			age = 30
		}
		modifier = {
			factor = 1.5
			age = 40
		}
	}

		set_character_flag = giant_condition
		random_list = {
			75 = { # Frail
				modifier = {
					factor = 0
					trait = strong
				}

					factor = 1.5

				}

					factor = 1.5
					trait = clubfooted
				}

					factor = 1.5

					}
				}

					factor = 1.5
					trait = poet
				}
				modifier = {
					factor = 1.5
					trait = slothful
				}

					factor = 1.5
					trait = shy

		trait = giant
		NOR = {
			trait = feeble
			trait = robust
		}
	}

	option = {
		name = EVTOPTAcharfl2
		trigger = {
			trait = stubborn
			NOT = { trait = slothful }
		}
		tooltip_info = stubborn
		random = {
			chance = 50

		}
		if = {
			limit = { NOT = { trait = diligent } }

	}
	option = {
		name = EVTOPTBcharfl2

# charfl.10 A relative begs a drunkard to stop
# A sober close relative smashes the cup. Trying to stop rolls charfl.11 a
# month later; lashing out disappoints them. Once per life.
# Scope: an adult drunkard.
character_event = {
	id = charfl.10
	desc = EVTDESCcharfl10
	picture = GFX_evt_drunk
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	only_capable = yes
	prisoner = no

	trigger = {
		is_adult = yes
		trait = drunkard
		NOT = { has_character_flag = relative_concerned_about_drinking }
		any_close_relative = {
			charfl_sober_relative_trigger = yes
		}
	}

	immediate = {
		set_character_flag = relative_concerned_about_drinking
		random_close_relative = {
			limit = { charfl_sober_relative_trigger = yes }
			save_event_target_as = concerned_relative
		}
	}

	option = { # Try to change
		name = EVTOPTAcharfl10
		custom_tooltip = { text = EVTOPTAcharfl10_TT }
		character_event = { id = charfl.11 days = 30 random = 30 }
		ai_chance = {
			factor = 50
			modifier = {
				factor = 2
				trait = temperate
			}
			modifier = {
				factor = 0.5
				trait = gluttonous
			}
			modifier = {
				factor = 2
				opinion = { who = event_target:concerned_relative value = 50 }
			}
		}
	}
	option = { # Lash out
		name = EVTOPTBcharfl10
		custom_tooltip = { text = EVTOPTBcharfl10_TT }
		event_target:concerned_relative = {
			show_scope_change = no
			opinion = {
				who = ROOT
				modifier = opinion_disappointed
				years = 5
			}
		}
		ai_chance = {
			factor = 50
			modifier = {
				factor = 2
				trait = wroth
			}
			modifier = {
				factor = 1.5
				trait = proud
			}
		}
	}
}

# charfl.11 Does the drunkard keep the promise?
# If the relative died meanwhile, charfl.12. Otherwise one of: sober and
# praised (13), sober but sick from withdrawal (14), or a relapse (15).
# Scope: the drunkard.
character_event = {
	id = charfl.11
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				event_target:concerned_relative = { is_alive = yes }
			}
			random_list = {
				33 = {
					modifier = {
						factor = 1.5
						trait = stubborn
					}
					modifier = {
						factor = 1.5
						trait = temperate
					}
					character_event = { id = charfl.13 days = 1 }
				}
				33 = {
					character_event = { id = charfl.14 days = 1 }
				}
				33 = {
					modifier = {
						factor = 1.5
						trait = gluttonous
					}
					character_event = { id = charfl.15 days = 1 }
				}
			}
		}
		else = {
			character_event = { id = charfl.12 days = 1 }
		}
	}
}

# charfl.25 A draught for the melancholy
# The court physician offers a grey draught. Drunk, it lifts the melancholy
# 40% of the time (more with a learned or trained physician) and makes the
# drinker ill otherwise. Once per life.
# Scope: a depressed adult ruler with a court physician.
character_event = {
	id = charfl.25
	desc = EVTDESCcharfl25
	picture = GFX_evt_bad_news
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	only_playable = yes
	only_capable = yes
	prisoner = no

	trigger = {
		is_adult = yes
		trait = depressed
		NOT = { has_character_flag = depression_potion_offer }
		any_courtier_or_vassal = { has_minor_title = title_court_physician }
	}

	immediate = {
		set_character_flag = depression_potion_offer
		random_courtier_or_vassal = {
			limit = { has_minor_title = title_court_physician }
			save_event_target_as = physician
		}
	}

	option = { # Drink it
		name = EVTOPTAcharfl25
		custom_tooltip = { text = EVTOPTAcharfl25_TT }
		hidden_effect = {
			random_list = {
				60 = {
					add_symptom_effect = yes
				}
				40 = {
					modifier = {
						factor = 1.25
						event_target:physician = { learning = 12 }
					}
					modifier = {
						factor = 1.25
						event_target:physician = { learning = 16 }
					}
					modifier = {
						factor = 1.25
						event_target:physician = { learning = 20 }
					}
					modifier = {
						factor = 1.5
						event_target:physician = { trait = physician }
					}
					remove_trait = depressed
					character_event = { id = 38314 days = 1 } # Notify lost depressed
				}
			}
		}
		ai_chance = {
			factor = 50
			modifier = {
				factor = 2
				trait = brave
			}
			modifier = {
				factor = 0.5
				trait = craven
			}
		}
	}
	option = { # Refuse
		name = EVTOPTBcharfl25
		event_target:physician = {
			show_scope_change = no
			opinion = {
				who = ROOT
				modifier = opinion_disappointed
				years = 5
			}
		}
		ai_chance = {
			factor = 50
			modifier = {
				factor = 2
				trait = paranoid
			}
		}
	}
}

# into the mod), moved to the human faiths of the Warhammer world. A ruler
# sends for a wise counsellor (seek_court_sage): a priest of
# Verena for the Old World faiths, a hermit of the Lady in Bretonnia, a
# wandering starets in Kislev, a scholar of the Prophets in Araby, a sage of
# the Celestial Court in Cathay. The sage chides the ruler's vices, eases a
# troubled mind, and now and then turns out to be a charlatan.
#   sage.1-3     The search: the sage is found (great, ordinary or a fraud)

#   pre-trigger of sage.10), court_sage, fake_court_sage,

# sage.10 The sage at court (hidden)
# Yearly. A charlatan may be found out; otherwise the sage eases a troubled
# mind or chides one of the ruler's vices, if there is one.
# Scope: the ruler. Fired by the on_yearly_pulse court_sage_counsel list.
character_event = {
	id = sage.10
	hide_window = yes

	is_triggered_only = yes

	only_playable = yes
	has_character_flag = found_court_sage
	only_capable = yes
	prisoner = no
	min_age = 16
	ai = no

	trigger = {
		any_courtier = { has_minor_title = title_court_sage }
	}

	immediate = {
		any_courtier = {
			limit = { has_minor_title = title_court_sage }
			save_event_target_as = court_sage
		}
		if = {
			limit = {
				event_target:court_sage = {
					has_character_flag = fake_court_sage
					NOT = { has_character_flag = fake_court_sage_suspected }
				}
				is_dumb_trigger = no
				NOT = { trait = inbred }
				NOT = { trait = trusting }
			}
			random = {
				chance = 30
				character_event = { id = sage.20 }
			}
		}
		else_if = {
			limit = {
				event_target:court_sage = { NOT = { has_character_flag = fake_court_sage } }
				OR = {
					trait = stressed
					trait = depressed
				}
			}
			if = {
				limit = { event_target:court_sage = { trait = genius } }
				narrative_event = { id = sage.30 }
			}
			else = {
				random = {
					chance = 50
					narrative_event = { id = sage.30 }
				}
			}
		}
		else_if = {
			limit = {
				event_target:court_sage = { NOT = { has_character_flag = fake_court_sage } }
				sage_has_vice_trigger = yes
			}
			random_list = {
				10 = { trigger = { trait = drunkard } set_character_flag = sage_vice_drunkard }
				10 = { trigger = { trait = hedonist } set_character_flag = sage_vice_hedonist }
				10 = { trigger = { trait = lustful } set_character_flag = sage_vice_lustful }
				10 = { trigger = { trait = gluttonous } set_character_flag = sage_vice_gluttonous }
				10 = { trigger = { trait = greedy } set_character_flag = sage_vice_greedy }
				10 = { trigger = { trait = envious } set_character_flag = sage_vice_envious }
				10 = { trigger = { trait = wroth } set_character_flag = sage_vice_wroth }
				10 = { trigger = { trait = proud } set_character_flag = sage_vice_proud }
				10 = { trigger = { trait = arbitrary } set_character_flag = sage_vice_arbitrary }
				10 = { trigger = { trait = cynical } set_character_flag = sage_vice_cynical }
				10 = { trigger = { trait = cruel } set_character_flag = sage_vice_cruel }
				10 = { trigger = { trait = paranoid } set_character_flag = sage_vice_paranoid }

# Faction: Generic
# Serves: human and dwarf player rulers of duke rank and above
#
# The vanilla cursed diamond (RoI.215 - RoI.249 in roi_various_events,
# never copied into the mod): once per game a diamond the size of a hen's
# egg comes to a ruler, said to have been pried from the eye of a
# serpent-idol in the jungles of Lustria. Set in a crown, kept in the
# treasury or given to the spouse, it brings thieves, accidents, murder,
# madness and the gem-lust, until the ruler sells it, drowns it in the
# river or loses it anyway, and someone must be blamed. The Dawi feel the
# gem-lust worst of all.
#   gem.1        The diamond comes to the ruler
#   gem.2-6      The crown: thieves, a panicked warhorse, a would-be murderer
#   gem.7-10     The treasury: the gem-lust, theft, sleepless nights
#   gem.11-14    The spouse: robbed, maddened or murdered
#   gem.20-26    Is it cursed? Sell it, drown it or keep it, and the blame
#   gem.27-28    The owner dies and the diamond vanishes
#
# Fired from: on_five_year_pulse charfl_flavour list (wh_on_actions.txt):
#   1; on_death (wh_on_actions.txt): 27.
# Artifacts: cursed_diamond, cursed_diamond_crown (common/artifacts/
#   00_artifacts.txt), renamed the Serpent's Eye in the loc.
# Saved: global flag serpents_eye_found; character flags serpents_eye_owner
#   (the on_death pre-trigger), serpents_eye_crown, serpents_eye_treasury,
#   serpents_eye_spouse, serpents_eye_lust, serpents_eye_sold,
#   serpents_eye_sold_crown; character modifier the_diamond_thief; event
#   target gem_blamed.
###############################################################################

# Part 4: The young champion (namespace champion)
# Faction: Generic
# Serves: human, dwarf, elf and greenskin player rulers of duke rank and above
#
# The vanilla Roland chain (CM.1050 - CM.1057 in cm_charlemagne_story_events,
# never copied into the mod), cut free of Charlemagne: once in a lifetime a
# bold young warrior of the ruler's own people comes to court looking for a
# lord worth fighting for. Taken in, he grows close to the ruler within the
# year, wins renown on campaign, and in the end may fall a hero's death on
# the battlefield, in victory or in defeat, and be mourned by the realm.
#   champion.1     The young champion arrives
#   champion.2-3   A friend at court
#   champion.4, 9  Renown on campaign
#   champion.5-8   A hero's death
# Saved: character flags champion_seen (the ruler, once per lifetime),
#   young_champion and champion_renowned (the champion).
# Callers: wh_on_actions on_five_year_pulse young_champion list (1);
#   00_on_actions on_battle_won (4, 5), on_battle_lost (6).
###############################################################################

namespace = champion

# champion.1 The young champion arrives
# Scope: the ruler. Fired by the young_champion five-year list.
narrative_event = {
	id = champion.1
	title = EVTTITLEchampion1
	desc = { trigger = { trait = creature_dwarf } text = EVTDESCchampion1_dwarf }
	desc = {
		trigger = { OR = { trait = creature_orc trait = creature_goblin } }
		text = EVTDESCchampion1_greenskin
	}
	desc = { trigger = { trait = creature_elf } text = EVTDESCchampion1_elf }
	desc = { trigger = { trait = creature_human } text = EVTDESCchampion1 }
	picture = GFX_evt_emissary
	border = GFX_event_narrative_frame_war

	is_triggered_only = yes

	only_playable = yes
	only_independent = yes
	ai = no
	only_capable = yes
	min_age = 16
	prisoner = no

	trigger = {
		NOT = { has_character_flag = champion_seen }
		higher_tier_than = COUNT
		OR = {
			trait = creature_human
			trait = creature_dwarf
			trait = creature_elf
			trait = creature_orc
			trait = creature_goblin
		}
	}

	immediate = {
		set_character_flag = champion_seen
	}

	option = { # Take him into my service
		name = EVTOPTAchampion1
		custom_tooltip = { text = EVTOPTAchampion1_TT }
		hidden_tooltip = {
			if = {
				limit = { trait = creature_dwarf }
				create_character = {
					random_traits = no
					dynasty = random
					religion = ROOT
					culture = ROOT
					female = no
					age = 40
					attributes = { diplomacy = 8 martial = 12 stewardship = 6 intrigue = 4 learning = 6 }
					trait = creature_dwarf
					trait = honest
					trait = brave
					trait = tough_soldier
					trait = strong
					trait = aggressive_leader
				}
			}
			else_if = {
				limit = { trait = creature_elf }
				create_character = {
					random_traits = no
					dynasty = random
					religion = ROOT
					culture = ROOT
					female = no
					age = 60
					attributes = { diplomacy = 9 martial = 12 stewardship = 6 intrigue = 6 learning = 8 }
					trait = creature_elf
					trait = honest
					trait = brave
					trait = tough_soldier
					trait = fair
					trait = aggressive_leader
				}
			}
			else_if = {
				limit = { OR = { trait = creature_orc trait = creature_goblin } }
				create_character = {
					random_traits = no
					dynasty = random
					religion = ROOT
					culture = ROOT
					female = no
					age = 18
					attributes = { diplomacy = 4 martial = 12 stewardship = 3 intrigue = 5 learning = 2 }
					trait = creature_orc
					trait = brave
					trait = tough_soldier
					trait = strong
					trait = aggressive_leader
				}
			}
			else = {
				create_character = {
					random_traits = no
					dynasty = random
					religion = ROOT
					culture = ROOT
					female = no
					age = 22
					attributes = { diplomacy = 10 martial = 12 stewardship = 7 intrigue = 5 learning = 7 }
					trait = creature_human
					trait = honest
					trait = brave
					trait = tough_soldier
					trait = strong
					trait = fair
					trait = aggressive_leader
				}
			}
			new_character = {
				set_character_flag = young_champion
				opinion = {
					modifier = opinion_loyal_servant
					who = ROOT
					years = 100
				}
				character_event = { id = champion.2 days = 350 random = 30 }
			}
		}
	}
	option = { # Send him on his way
		name = EVTOPTBchampion1
		prestige = -5
	}
}
