# charfl.1 A giant's body settles
# A grown giant either trains the bulk into strength (charfl.3) or sinks
# under its weight (charfl.2). Learned and idle giants grow frail, warriors
# and tribesmen robust. Once per life.
# Scope: a giant, 16 to 50.
character_event = {
	id = charfl.1
	hide_window = yes

	is_triggered_only = yes

	min_age = 16
	max_age = 50
	capable_only = yes
	prisoner = no

	trigger = {
		trait = giant
		NOR = {
			trait = feeble
			trait = robust
			has_character_flag = giant_condition
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = {
			factor = 1.5
			OR = {
				trait = weak
				trait = strong
			}
		}
		modifier = {
			factor = 1.5
			age = 20
		}
		modifier = {
			factor = 1.5
			age = 30
		}

			factor = 1.5
			age = 40
		}
	}

		set_character_flag = giant_condition

			75 = { # Frail

					factor = 0
					trait = strong

					factor = 1.5

				}

				}

				}

		}

		name = EVTOPTBcharfl2

# charfl.10 A relative begs a drunkard to stop
# A sober close relative smashes the cup. Trying to stop rolls charfl.11 a
# month later; lashing out disappoints them. Once per life.
# Scope: an adult drunkard.
character_event = {
	id = charfl.10
	desc = EVTDESCcharfl10
	picture = GFX_evt_drunk
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	only_capable = yes
	prisoner = no

	trigger = {
		is_adult = yes
		trait = drunkard
		NOT = { has_character_flag = relative_concerned_about_drinking }
		any_close_relative = {
			charfl_sober_relative_trigger = yes
		}
	}

	immediate = {
		set_character_flag = relative_concerned_about_drinking
		random_close_relative = {
			limit = { charfl_sober_relative_trigger = yes }
			save_event_target_as = concerned_relative
		}
	}

	option = { # Try to change
		name = EVTOPTAcharfl10
		custom_tooltip = { text = EVTOPTAcharfl10_TT }
		character_event = { id = charfl.11 days = 30 random = 30 }
		ai_chance = {
			factor = 50
			modifier = {
				factor = 2
				trait = temperate
			}
			modifier = {
				factor = 0.5
				trait = gluttonous
			}
			modifier = {
				factor = 2
				opinion = { who = event_target:concerned_relative value = 50 }
			}
		}
	}
	option = { # Lash out
		name = EVTOPTBcharfl10
		custom_tooltip = { text = EVTOPTBcharfl10_TT }
		event_target:concerned_relative = {
			show_scope_change = no
			opinion = {
				who = ROOT
				modifier = opinion_disappointed
				years = 5
			}
		}
		ai_chance = {
			factor = 50
			modifier = {
				factor = 2
				trait = wroth
			}
			modifier = {
				factor = 1.5
				trait = proud
			}
		}
	}
}

# charfl.11 Does the drunkard keep the promise?
# If the relative died meanwhile, charfl.12. Otherwise one of: sober and
# praised (13), sober but sick from withdrawal (14), or a relapse (15).
# Scope: the drunkard.
character_event = {
	id = charfl.11
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				event_target:concerned_relative = { is_alive = yes }
			}
			random_list = {
				33 = {
					modifier = {
						factor = 1.5
						trait = stubborn
					}
					modifier = {
						factor = 1.5
						trait = temperate
					}
					character_event = { id = charfl.13 days = 1 }
				}
				33 = {
					character_event = { id = charfl.14 days = 1 }
				}
				33 = {
					modifier = {
						factor = 1.5
						trait = gluttonous
					}
					character_event = { id = charfl.15 days = 1 }
				}
			}
		}
		else = {
			character_event = { id = charfl.12 days = 1 }
		}
	}
}

# charfl.25 A draught for the melancholy
# The court physician offers a grey draught. Drunk, it lifts the melancholy
# 40% of the time (more with a learned or trained physician) and makes the
# drinker ill otherwise. Once per life.
# Scope: a depressed adult ruler with a court physician.
character_event = {
	id = charfl.25
	desc = EVTDESCcharfl25
	picture = GFX_evt_bad_news
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	only_playable = yes
	only_capable = yes
	prisoner = no

	trigger = {
		is_adult = yes
		trait = depressed
		NOT = { has_character_flag = depression_potion_offer }
		any_courtier_or_vassal = { has_minor_title = title_court_physician }
	}

	immediate = {
		set_character_flag = depression_potion_offer
		random_courtier_or_vassal = {
			limit = { has_minor_title = title_court_physician }
			save_event_target_as = physician
		}
	}

	option = { # Drink it
		name = EVTOPTAcharfl25
		custom_tooltip = { text = EVTOPTAcharfl25_TT }
		hidden_effect = {
			random_list = {
				60 = {
					add_symptom_effect = yes
				}
				40 = {
					modifier = {
						factor = 1.25
						event_target:physician = { learning = 12 }
					}
					modifier = {
						factor = 1.25
						event_target:physician = { learning = 16 }
					}
					modifier = {
						factor = 1.25
						event_target:physician = { learning = 20 }
					}
					modifier = {
						factor = 1.5
						event_target:physician = { trait = physician }
					}
					remove_trait = depressed
					character_event = { id = 38314 days = 1 } # Notify lost depressed
				}
			}
		}
		ai_chance = {
			factor = 50
			modifier = {
				factor = 2
				trait = brave
			}
			modifier = {
				factor = 0.5
				trait = craven
			}
		}
	}
	option = { # Refuse
		name = EVTOPTBcharfl25
		event_target:physician = {
			show_scope_change = no
			opinion = {
				who = ROOT
				modifier = opinion_disappointed
				years = 5
			}
		}
		ai_chance = {
			factor = 50
			modifier = {
				factor = 2
				trait = paranoid
			}
		}
	}
}
