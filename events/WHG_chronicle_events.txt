###############################################################################
# The dynasty chronicle
# Namespace: chronicle (was CM.10000 - CM.10529 in chronicle_events.txt)
# Faction: Generic
#
# Hidden events that write the player's dynasty chronicle: inheritance,
# births and deaths, wars, holy wars, titles, marriages, prison, conversions,
# battles, great cities taken and lost, peace. When a year passes with
# nothing written, a filler entry tells of monsters, omens and strange
# happenings in the realm, Morrslieb and Geheimnisnacht among them.
#
# Warhammer rules: the great cities are Altdorf, Middenheim, Nuln,
# Marienburg, Kislev, Praag, Couronne, Karaz-a-Karak, Zharr-Naggrund,
# Lothern, Naggarond, Khemri and Hexoatl (chronicle_great_city_trigger,
# wh_chronicle_triggers.txt); the hooks on on_siege_won_leader and
# on_siege_over_winner are restored.
# The vanilla Aztec entries are gone.
#
# Start and hand-over        chronicle.1 - chronicle.2
# Life, war and holy war     chronicle.10 - chronicle.18
# Titles, marriage, prison,
#   faith, battles, peace    chronicle.20 - chronicle.40
# Filler entries             chronicle.50 - chronicle.86
# The Empire reborn          chronicle.90 - chronicle.91
#   (was chronicled_achievements.1 - .2 in achievement_events.txt)
# Renown counters            chronicle.95 - chronicle.98
#   (were the achievement counters 718 - 720)
#
# Fired from: the chronicle on_actions in 00_on_actions.txt (each event names
#   its own); chronicle.90 from on_create_title; chronicle.95 from
#   WHG_seduction_events.txt, chronicle.96 from WHG_intrigue_events.txt,
#   chronicle.97 from 01_scripted_effects_duels.txt.
# Saved: global flag chronicle_empire_reborn; character flags renown_seducer,
#   renown_slanderer, renown_duellist (+ renown_new_*); variables
#   renown_seducer_count, renown_slanderer_count, renown_duellist_count.
###############################################################################

namespace = chronicle

# chronicle.1 The chronicle begins
# Writes the opening entry when a new game starts.
# Scope: ROOT is the player.
character_event = {
	id = chronicle.1

	is_triggered_only = yes # on_chronicle_start

	hide_window = yes

	immediate = {
		chronicle = { entry = CHRONICLE_START picture = GFX_evt_monk }
	}
}

# chronicle.2 The chronicle changes hands
# Writes how the chronicle passed to the new player character: by inheritance (same or new culture and faith), to a landless heir, or to a ruler of a realm won in holy war.
# Scope: ROOT is the new holder, FROM the old one.
character_event = {
	id = chronicle.2

	is_triggered_only = yes # on_chronicle_owner_change

	hide_window = yes

	immediate = {
		if = {
			limit = {
				FROM = { is_dying = no }
				ROOT = {
					NOR = {
						trait = crusader_king
						trait = crusader_queen
						has_character_flag = beneficiary_check_potential_player_character
					}
				}
			}
			chronicle = {
				entry = CHRONICLE_PLAYER_INHERITANCE_LANDLESS
				portrait = [Root.GetID]
				picture = GFX_evt_emissary
			}
		}
		else_if = {
			limit = {
				FROM = { is_dying = no }
				ROOT = { OR = { trait = crusader_king trait = crusader_queen } }
			}
			chronicle = {
				entry = CHRONICLE_BECAME_RULER_OF_CRUSADER_KINGDOM
				portrait = [Root.GetID]
				picture = GFX_evt_crusaders
			}
		}
		else_if = {
			limit = { has_character_flag = beneficiary_check_potential_player_character }
			chronicle = {
				entry = CHRONICLE_BECAME_LORD_OF_CRUSADER_KINGDOM
				portrait = [Root.GetID]
				picture = GFX_evt_crusaders
			}
		}
		if = {
			limit = { FROM = { is_dying = yes culture = ROOT religion = ROOT } }
			chronicle = {
				entry = CHRONICLE_PLAYER_INHERITANCE_NORMAL
				portrait = [Root.GetID]
				picture = GFX_evt_coronation
			}
		}
		if = {
			limit = { FROM = { is_dying = yes culture = ROOT NOT = { religion = ROOT } } }
			chronicle = {
				entry = CHRONICLE_PLAYER_INHERITANCE_NEW_RELIGION
				portrait = [Root.GetID]
				picture = GFX_evt_coronation
			}
		}
		if = {
			limit = { FROM = { is_dying = yes NOT = { culture = ROOT } religion = ROOT } }
			chronicle = {
				entry = CHRONICLE_PLAYER_INHERITANCE_NEW_CULTURE
				portrait = [Root.GetID]
				picture = GFX_evt_coronation
			}
		}
		if = {
			limit = { FROM = { is_dying = yes NOT = { culture = ROOT } NOT = { religion = ROOT } } }
			chronicle = {
				entry = CHRONICLE_PLAYER_INHERITANCE_NEW_CULTURE_AND_RELIGION
				portrait = [Root.GetID]
				picture = GFX_evt_coronation
			}
		}
	}
}

# chronicle.10 A death
# Writes the death of a liege, spouse, parent, heir or faith head into the chronicles of the players it touches.
# Scope: ROOT is the dead character.
character_event = {
	id = chronicle.10

	hide_window = yes

	is_triggered_only = yes # on_death

	trigger = {
		any_player = {
			OR = {
				liege = { character = ROOT }
				any_spouse = { character = ROOT }
				is_child_of = ROOT
				current_heir = { character = ROOT }
				AND = { religion = ROOT ROOT = { controls_religion = yes } }
			}
		}
	}

	immediate = {
		any_vassal = {
			limit = {
				ai = no
				NOT = { OR = { father = { character = ROOT } mother = { character = ROOT } } }
				NOT = { is_primary_heir = ROOT } # Any vassal except primary heir which is handled in chronicle owner transfer event chronicle.2
			}
			chronicle = { entry = CHRONICLE_DEATH_OF_LIEGE portrait = [Root.GetID] }
		}
		any_spouse = {
			limit = {
				ai = no
				NOT = { is_primary_heir = ROOT } # Not used for primary heir which is handled in chronicle owner transfer event chronicle.2
			}
			chronicle = { entry = CHRONICLE_DEATH_OF_SPOUSE portrait = [Root.GetID] }
		}
		any_child = {
			limit = {
				ai = no
				ROOT = {
					NOT = { is_primary_heir = PREV } # Sent to any child except primary heir which is handled in chronicle owner transfer event chronicle.2
				}
			}
			chronicle = { entry = CHRONICLE_DEATH_OF_PARENT portrait = [Root.GetID] }
		}
		any_player = {
			limit = {
				current_heir = { character = ROOT }
				NOT = { is_primary_heir = ROOT } # Not used for primary heir which is handled in chronicle owner transfer event chronicle.2
			}
			chronicle = { entry = CHRONICLE_DEATH_OF_HEIR portrait = [Root.GetID] }
		}
		any_player = {
			limit = {
				religion_head = { character = ROOT }
				NOT = { is_primary_heir = ROOT } # Not used for primary heir which is handled in chronicle owner transfer event chronicle.2
				NOT = {
					OR = {
						father = { character = ROOT }
						mother = { character = ROOT }
						spouse = { character = ROOT }
					}
				}
			}
			chronicle = { entry = CHRONICLE_DEATH_OF_RELIGION_HEAD portrait = [Root.GetID] }
		}
	}
}

# chronicle.11 A birth
# A player had a child; waits two days for the name (chronicle.12).
# Scope: ROOT is the child.
character_event = {
	id = chronicle.11

	hide_window = yes

	is_triggered_only = yes # on_birth

	trigger = {
		OR = { father = { ai = no } mother = { ai = no } }
	}

	immediate = {
		character_event = { id = chronicle.12 days = 2 } # Delay to allow for naming of child.
	}
}

# chronicle.12 A birth, named
# Writes the firstborn child, or first son, into the parents' chronicles.
# Scope: ROOT is the child, FROM the child.
character_event = {
	id = chronicle.12

	hide_window = yes

	is_triggered_only = yes # chronicle.11

	trigger = {
		OR = { father = { ai = no } mother = { ai = no } }
	}

	immediate = {
		father = {
			if = {
				limit = {
					ai = no
					OR = {
						NOT = { num_of_children = 2 }
						AND = {
							FROM = { is_female = no }
							NOT = { any_child = { NOT = { character = FROM } is_female = no } }
						}
					}
				}
				chronicle = { entry = CHRONICLE_BIRTH_FATHER portrait = [Root.GetID] }
			}
		}
		mother = {
			if = {
				limit = {
					ai = no
					OR = {
						NOT = { num_of_children = 2 }
						AND = {
							FROM = { is_female = no }
							NOT = { any_child = { NOT = { character = FROM } is_female = no } }
						}
					}
				}
				chronicle = { entry = CHRONICLE_BIRTH_MOTHER portrait = [Root.GetID] }
			}
		}
	}
}

# chronicle.13 War declared
# Writes a war into the attacker's and defender's chronicles.
# Scope: ROOT is the defender, FROM the attacker.
character_event = {
	id = chronicle.13

	hide_window = yes

	is_triggered_only = yes # on_war_started

	trigger = {
		OR = { ai = no FROM = { ai = no } }
	}

	immediate = {
		FROM = {
			if = {
				limit = { ai = no }
				chronicle = { entry = CHRONICLE_WAR_DECLARED_ATTACKER portrait = [Prev.GetID] }
			}
		}
		if = {
			limit = { ai = no }
			chronicle = { entry = CHRONICLE_WAR_DECLARED_DEFENDER portrait = [From.GetID] }
		}
	}
}

# chronicle.16 Holy war declared

	id = chronicle.16

	hide_window = yes

	is_triggered_only = yes # on_crusade_creation

		any_player = {
			limit = { religion = ROOT }
			chronicle = { entry = CHRONICLE_FRIENDLY_CRUSADE_STARTED portrait = [Root.GetID] }
		}
		any_player = {
			limit = { new_character = { religion = PREV } }
			chronicle = { entry = CHRONICLE_UNFRIENDLY_CRUSADE_STARTED portrait = [Root.GetID] }
		}
		ROOT = {
			if = {

			}
		}
	}
}

# chronicle.17 Holy war won
# Scope: ROOT is the target title, FROM the taker, new_character the enemy.
character_event = {
	id = chronicle.17

	hide_window = yes

	is_triggered_only = yes # on_crusade_success

	immediate = {
		if = {
			limit = { FROM = { religion_head = { is_alive = yes } } }
			any_player = {
				limit = {
					OR = { religion = FROM new_character = { religion = PREV } }
					OR = { NOT = { character = FROM } FROM = { uses_new_crusade = yes } }
				}
				chronicle = { entry = CHRONICLE_CRUSADE_WON portrait = [From.RelHead.GetID] }
			}
			FROM = {
				if = {
					limit = { ai = no uses_new_crusade = no }
					chronicle = {
						entry = CHRONICLE_CRUSADE_WON_AND_I_GOT_THE_TITLE
						portrait = [From.GetID]
					}
				}
			}
		}
		if = {
			limit = {
				NOT = {
					FROM = {
						religion_head = { is_alive = yes } # Crusade ended inconclusively as the religious head no longer exist
					}
				}
			}
			any_player = {
				limit = { OR = { religion = FROM new_character = { religion = PREV } } }
				chronicle = { entry = CHRONICLE_CRUSADE_LOST_NO_HEAD }
			}
		}
	}
}

# chronicle.22 Marriage
# Scope: ROOT is the employer, FROM the player, new_character the spouse.
character_event = {
	id = chronicle.22

	hide_window = yes

	is_triggered_only = yes # on_marriage

	trigger = {
		FROM = { ai = no }
	}

	immediate = {
		if = {
			limit = {
				new_character = {
					mother_even_if_dead = { always = yes }
					OR = {
						father_even_if_dead = {
							PREV = { mother_even_if_dead = { higher_tier_than = PREVPREV } }
						}
						NOT = { father_even_if_dead = { always = yes } }
					}
				}
			}
			FROM = {
				chronicle = { entry = CHRONICLE_MARRIED_MOTHER_IMPORTANT portrait = [New.GetID] }
			}
		}
		if = {
			limit = {
				new_character = {
					father_even_if_dead = { always = yes }
					OR = {
						father_even_if_dead = {
							PREV = {
								mother_even_if_dead = { NOT = { higher_tier_than = PREVPREV } }
							}
						}
						NOT = { mother_even_if_dead = { always = yes } }
					}
				}
			}
			FROM = {
				chronicle = { entry = CHRONICLE_MARRIED_FATHER_IMPORTANT portrait = [New.GetID] }
			}
		}
		if = {
			limit = {
				new_character = {
					NOT = { father_even_if_dead = { always = yes } }
					NOT = { mother_even_if_dead = { always = yes } }
				}
			}
			FROM = { chronicle = { entry = CHRONICLE_MARRIED_NO_PARENT portrait = [New.GetID] } }
		}
	}
}

						liege = { primary_title = { tier = EMPEROR } }

				chronicle = { entry = CHRONICLE_MARRIED_MOTHER_IMPORTANT portrait = [New.GetID] }
			}
		}
		if = {
			limit = {
				new_character = {
					father_even_if_dead = { always = yes }
					OR = {
						father_even_if_dead = {
							PREV = {
								mother_even_if_dead = { NOT = { higher_tier_than = PREVPREV } }
							}
						}
						NOT = { mother_even_if_dead = { always = yes } }
					}
				}
			}
			any_player = {
				limit = { FROM = { is_primary_heir = PREV } }
				chronicle = { entry = CHRONICLE_MARRIED_FATHER_IMPORTANT portrait = [New.GetID] }
			}
		}
		if = {
			limit = {
				new_character = {
					NOT = { father_even_if_dead = { always = yes } }
					NOT = { mother_even_if_dead = { always = yes } }
				}
			}
			any_player = {
				limit = { FROM = { is_primary_heir = PREV } }

# chronicle.95 Renown: the seducer

# (chronicle.98). Was the Casanova achievement counter 718.

	id = chronicle.95
	hide_window = yes
	is_triggered_only = yes # WHG_seduction_events.txt, seduction success

	trigger = { NOT = { has_character_flag = renown_seducer } }

	immediate = {
		change_variable = { which = renown_seducer_count value = 1 }

		}
	}
}

# chronicle.98 A name in every tavern
# The ruler's renown as a seducer, slanderer or duellist spreads: prestige,
# the matching nickname if they have none, and a chronicle entry for a player.
# Scope: ROOT is the ruler, set to the flag just earned.
character_event = {
	id = chronicle.98
	desc = {
		trigger = { has_character_flag = renown_new_duellist }
		text = EVTDESCchronicle98_duel
	}
	desc = {
		trigger = { has_character_flag = renown_new_slanderer }
		text = EVTDESCchronicle98_slander
	}
	desc = {
		trigger = { has_character_flag = renown_new_seducer }
		text = EVTDESCchronicle98_seduce
	}
	picture = GFX_evt_courtiers_talking
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes # chronicle.95 - chronicle.97

	after = {
		clr_character_flag = renown_new_seducer
		clr_character_flag = renown_new_slanderer
		clr_character_flag = renown_new_duellist
	}

	option = {
		name = EVTOPTAchronicle98
		prestige = 100
		if = {
			limit = {
				has_character_flag = renown_new_duellist
			}
			if = {
				limit = { has_nickname = no }
				give_nickname = nick_the_blade
			}
			if = {
				limit = { ai = no }
				chronicle = { entry = CHRONICLE_RENOWN_DUEL picture = GFX_evt_duel }
			}
		}
		else_if = {
			limit = {
				has_character_flag = renown_new_slanderer
			}
			if = {
				limit = { has_nickname = no }
				give_nickname = nick_snaketongue
			}
			intrigue = 1
			if = {
				limit = { ai = no }
				chronicle = { entry = CHRONICLE_RENOWN_SLANDER picture = GFX_evt_whispers }
			}
		}
		else = {
			if = {
				limit = { has_nickname = no }
				give_nickname = nick_the_seducer
			}
			if = {
				limit = { ai = no }
				chronicle = { entry = CHRONICLE_RENOWN_SEDUCE picture = GFX_evt_lovers }
			}
		}
	}
}
