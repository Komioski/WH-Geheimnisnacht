# Colleges of Magic events
# Namespace: college (was COM in wh_college_of_magic_events.txt)
# Faction: Empire
#
# The Colleges of Magic: the eight Orders founded in Altdorf when an Emperor
# signs the Articles of Imperial Magic (found_the_imperial_collages_of_magic),
# and the five lesser colleges other Old World realms may found
# (found_new_collages_of_magic). A gifted applicant is tested and admitted,
# chooses the wind he will study, and is given a Magister of his Order as
# teacher. Members may buy the tools of their art from the college stores,
# seek a promotion through a superior's favour, choose how to make their name,
# and a Lord Magister may challenge his Patriarch to a duel of spells for the
# seat. AI Lord Magisters challenge player Patriarchs the same way.
#
# Joining                                college.1 - college.4
# AI members                             college.10
# Choosing a wind and a teacher          college.20 - college.26
# Making a name                          college.30
# Favouritism                            college.39 - college.42
# Duel for the Patriarch's seat          college.50 - college.56
# Founding of the colleges               college.60
# The college stores                     college.70 - college.73
# A house for the Order                  college.79 - college.82
# Flavour                                college.90 - college.94
#
# Callers:
#   common/on_actions/WHG_society_on_actions.txt: college.1 on
#     on_character_ask_to_join_society, college.10 on on_five_year_pulse,
#     college.26 on on_death,
#     college.90 on on_society_bi_yearly_pulse.
#   common/scripted_effects/wh_magic_scripted_effects.txt: join effects fire
#     college.20. common/scripted_effects/wh_college_effects.txt:
#     college_ai_challenge_effect fires college.55.
#   decisions/wh_colleges_of_magic_decisions.txt: founding decisions fire
#     college.60 and college.20, political_focus_college college.30,
#     favouritism_colleges college.39, college_equipment_requisition
#     college.70, request_college_house college.79.
#   decisions/mnm_society_decisions.txt: rank-up to Patriarch fires
#     college.50.
# Scripted effects: common/scripted_effects/wh_college_effects.txt
# Scripted triggers: common/scripted_triggers/wh_colleges_of_magic_scripted_triggers.txt
# Event modifiers: common/event_modifiers/wh_college_modifiers.txt
# Artifacts: common/artifacts/17_artifacts_magic.txt
# Loc: localisation/00_WHG_college_of_magic_events.csv

# college.10 AI mages join and rise
# On the five-year pulse an AI ruler who is a member may rise a rank (1 in 4,
# up to Lord Magister), trained AI mages at his court (and he himself) may
# join a college, AI Lord Magisters may challenge a player Patriarch of
# their wind (college_ai_challenge_effect), and AI members buy one item
# from the stores (college_ai_store_effect). Which colleges are near is
# worked out once per ruler (college_mark_near_effect).
# Scope: ROOT is the ruler.
character_event = {
	id = college.10
	hide_window = yes
	is_triggered_only = yes # on_five_year_pulse

	only_rulers = yes

	trigger = {
		any_college_active = yes
	}

	immediate = {
		if = {
			limit = {
				ai = yes
				prisoner = no
				NOT = { trait = incapable }
				is_college_of_magic_member = yes
				society_rank <= 3
				society_can_rank_up = yes
				is_trained_mage_trigger = yes
			}
			random = { chance = 25 society_rank_up = 1 }
		}
		if = {
			limit = {
				OR = {
					college_ai_candidate_trigger = yes
					any_courtier = { college_ai_candidate_trigger = yes }
				}
			}
			college_mark_near_effect = yes
			if = {
				limit = { college_ai_candidate_trigger = yes }
				save_event_target_as = college_candidate
				college_ai_join_effect = yes
			}
			any_courtier = {
				limit = { college_ai_candidate_trigger = yes }
				save_event_target_as = college_candidate
				college_ai_join_effect = yes
			}
			college_clear_near_effect = yes
		}
		if = {
			limit = { ai = yes is_college_of_magic_member = yes }
			college_ai_challenge_effect = yes
			college_ai_store_effect = yes
		}
		any_courtier = {
			limit = { ai = yes is_college_of_magic_member = yes society_rank = 2 }
			college_ai_challenge_effect = yes
			college_ai_store_effect = yes
		}
	}
}

# college.23 Finding a teacher
# Looks for a Magister (rank 3 or more) of the student's wind, preferring AI
# Magisters with fewer than three apprentices and plain Magisters over Lord
# Magisters. Sets found_teacher_in_college when one is found.
# Scope: ROOT is the student.
character_event = {
	id = college.23
	hide_window = yes
	is_triggered_only = yes # college.20 - college.22, college.26

	immediate = {
		society = {
			random_society_member = {
				limit = {
					prisoner = no
					is_incapacitated_trigger = no
					is_correct_teacher_for_root = yes
					society_rank = 3
					OR = {
						ai = yes
						NOT = {
							any_friend = {
								trait = mage_apprentice
								has_character_flag = mage_master@PREV
							}
						}
					}
				}
				preferred_limit = {
					ai = yes
					NOT = {
						any_friend = {
							count = 3
							trait = mage_apprentice
							has_character_flag = mage_master@PREV
						}
					}
				}
				preferred_limit = { ai = yes }
				preferred_limit = { NOT = { society_rank = 4 } }
				save_event_target_as = potential_teacher
			}
		}
		if = {
			limit = { event_target:potential_teacher = { always = yes } }
			set_character_flag = found_teacher_in_college
		}
		set_character_flag = learning_magic_in_college
		character_event = { id = college.24 }
	}
}

# college.24 Meeting the teacher
# The student meets his teacher, or learns that no Magister of his wind can
# take him and he must study alone. Studying alone is harder but stands out
# (prestige, and more college currency when he finishes). A player may also
# go back and pick another wind.
# Scope: ROOT is the student, event_target:potential_teacher the teacher.
character_event = {
	id = college.24
	desc = {
		text = EVTDESCcollege24
		trigger = {
			NOT = { has_character_flag = found_teacher_in_college }
			NOT = { has_character_flag = dead_teacher }
		}
	}
	desc = {
		text = EVTDESCcollege24b
		trigger = {
			has_character_flag = found_teacher_in_college
			NOT = { has_character_flag = dead_teacher }
		}
	}
	desc = {
		text = EVTDESCcollege24c
		trigger = {
			NOT = { has_character_flag = found_teacher_in_college }
			has_character_flag = dead_teacher
		}
	}
	desc = {
		text = EVTDESCcollege24d
		trigger = {
			has_character_flag = found_teacher_in_college
			has_character_flag = dead_teacher
		}
	}
	picture = GFX_evt_apprentice
	portrait = event_target:potential_teacher

	is_triggered_only = yes # college.23

	trigger = {
		NOT = { has_character_flag = college_member_learning_on_own }
	}

	option = { # Study under the teacher
		name = EVTOPTAcollege24
		trigger = { event_target:potential_teacher = { always = yes } }
		ai_chance = { factor = 95 }
		add_trait = mage_apprentice
		set_character_flag = mage_master@event_target:potential_teacher
		add_friend = event_target:potential_teacher
		save_event_target_as = mage_ward
		clr_character_flag = dead_teacher
		event_target:potential_teacher = {
			save_event_target_as = mage_tutor
			if = { limit = { ai = no } character_event = { id = college.25 } }
			character_event = { id = apprentice.10 days = 15 }
		}
	}
	option = { # Study alone
		name = EVTOPTBcollege24
		ai_chance = {
			factor = 5
			modifier = { factor = 0.1 NOT = { trait = proud } }
			modifier = { factor = 0.1 trait = humble }
			modifier = { factor = 2 is_smart_trigger = yes }
			modifier = { factor = 2 trait = magic_potential_3 }
			modifier = { factor = 2 trait = magic_potential_4 }
		}
		prestige = 100
		clr_character_flag = dead_teacher
		custom_tooltip = {
			text = EVTTOOLTIPecollege_learn_on_own
			set_character_flag = college_member_learning_on_own
			character_event = { id = apprentice.51 days = 10 }
		}
	}
	option = { # Pick another wind
		name = EVTOPTCcollege24
		trigger = { ai = no }
		ai_chance = { factor = 0 }
		clr_learning_flags_effect = yes
		clr_character_flag = learning_magic_in_college
		clear_event_target = potential_teacher
		character_event = { id = college.20 }
	}

	after = {
		clr_character_flag = found_teacher_in_college
	}
}

# college.40 A patron offers a promotion
# A Lord Magister (favouritism_colleges) offers to have a lesser member
# promoted for a favour. Accepting raises the member a rank and puts him in
# the patron's debt (or cancels a favour the patron owed him).
# Scope: ROOT is the member, FROM is the patron.
letter_event = {
	id = college.40
	desc = EVTDESCcollege40
	border = GFX_event_letter_frame_war

	is_triggered_only = yes # college.39

	option = { # Accept
		name = EVTOPTAcollege40
		ai_chance = {
			factor = 3
			modifier = { factor = 2 trait = ambitious }
			modifier = { factor = 0.2 trait = proud }
			modifier = { factor = 0.5 trait = honest }
		}
		if = { limit = { FROM = { owes_favor_to = ROOT } } remove_favor = FROM }
		else = { FROM = { add_favor = ROOT } }
		society_rank_up = 1
		FROM = { hidden_effect = { letter_event = { id = college.41 days = 1 } } }
	}
	option = { # Refuse
		name = EVTOPTBcollege40
		ai_chance = { factor = 1 }
		FROM = {
			change_society_currency = 300 # The full cost of favouritism_colleges
			hidden_effect = { letter_event = { id = college.42 days = 1 } }
		}
	}
}

# college.51 The duel is fought
# A duel of spells, not of swords: the odds follow the magic power, gift,
# learning and focus of both wizards, with the sitting Patriarch slightly
# favoured. The loser may be burned by a miscast.
# Scope: ROOT is the challenger, event_target:college_patriarch the Patriarch.
character_event = {
	id = college.51
	hide_window = yes
	is_triggered_only = yes # college.50

	immediate = {
		clr_character_flag = has_sent_request_to_rank_up
		random_list = {
			40 = { # Challenger wins
				modifier = { factor = 2 has_character_modifier = magic_power_4 }
				modifier = { factor = 1.5 has_character_modifier = magic_power_3 }
				modifier = { factor = 1.5 trait = magic_potential_4 }
				modifier = { factor = 1.25 learning = 18 }
				modifier = { factor = 1.25 owns_an_active_powerstone = yes }
				modifier = { factor = 0.75 is_ill_trigger = yes }
				modifier = {
					factor = 0.5
					event_target:college_patriarch = { has_character_modifier = magic_power_4 }
				}
				modifier = {
					factor = 0.75
					event_target:college_patriarch = { trait = magic_potential_4 }
				}
				modifier = { factor = 0.8 event_target:college_patriarch = { learning = 18 } }
