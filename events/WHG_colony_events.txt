###############################################################################
# Colonies
# Namespace: colony (was HColony.2 in WH_Colonies.txt and HighElfColony.1 in
#   wh_high_elf_colony_events.txt)
# Faction: Diplomacy & realms
# Serves: the Phoenix King, and the human, elf and Chaos Dwarf realms that
#   hold land far from home
#
# A ruler whose lords hold land far from the homeland can gather those lands
# into a colony under one of them, a tributary that pays a quarter of its
# income. The ruler picks the colony's lord (colony.1), then everyone
# touched hears of it (colony.2).
#
# The choice of lord                    colony.1
# The colony is founded                 colony.2
#
# Callers: the decisions create_new_colony and create_new_colony_2 - _7 in
#   decisions/dow_new_world.txt and create_new_colony in
#   decisions/wh_asur_decisions.txt (colony.1), create_colony in
#   common/scripted_effects/WHG_colony_effects.txt (colony.2).
# Triggers in common/scripted_triggers/WHG_colony_triggers.txt.
# Saved: event targets ColonyRuler1 - 3, NewColonyRuler; flags
#   NewColonyLord, LandUsurpedForColony, create_colony_empire,
#   create_colony_kingdom, RelativeGrantedTitleInColony@; variable
#   countys_given_away_value. All cleared by colony_cleanup.
# Loc: localisation/00_WHG_colony_events.csv
###############################################################################

namespace = colony

# colony.1 Choose the colony's lord
# The ruler picks one of up to three lords of their own race and faith from
# among those holding land outside the homeland, or calls it off.
# Scope: ROOT is the ruler.
narrative_event = {
	id = colony.1
	title = EVTNAMEcolony1
	desc = EVTDESCcolony1
	picture = {
		trigger = { trait = creature_elf }
		picture = GFX_evt_phoenix_king
	}
	picture = {
		trigger = { trait = creature_dwarf }
		picture = GFX_evt_dwarf_king
	}
	picture = {
		trigger = { trait = creature_chaos_dwarf }
		picture = GFX_evt_chaos_dwarfs
	}
	picture = {
		trigger = {
			NOR = {
				trait = creature_elf
				trait = creature_dwarf
				trait = creature_chaos_dwarf
			}
		}
		picture = GFX_evt_explore_ocean
	}
	border = GFX_event_narrative_frame_diplomacy

	is_triggered_only = yes # the colony decisions

	option = {
		name = EVTOPTAcolony1
		trigger = {
			event_target:ColonyRuler1 = { always = yes }
		}
		event_target:ColonyRuler1 = { save_event_target_as = NewColonyRuler }
		create_colony = yes
	}
	option = {
		name = EVTOPTBcolony1
		trigger = {
			event_target:ColonyRuler2 = { always = yes }
		}
		event_target:ColonyRuler2 = { save_event_target_as = NewColonyRuler }
		create_colony = yes
	}
	option = {
		name = EVTOPTCcolony1
		trigger = {
			event_target:ColonyRuler3 = { always = yes }
		}
		event_target:ColonyRuler3 = { save_event_target_as = NewColonyRuler }
		create_colony = yes
	}

	# Call it off
	option = {
		name = EVTOPTDcolony1
		colony_cleanup = yes
		any_realm_lord = { colony_cleanup = yes }
	}
}

# colony.2 The colony is founded
# Sent to the founder, the founder's lords, the new colony lord and that
# lord's vassals. The new colony lord gains prestige. Lords who lost land
# to the colony were paid for it in create_colony and are told so.
# Scope: ROOT is the one told; event target NewColonyRuler is the colony's
#   lord.
narrative_event = {
	id = colony.2
	title = EVTNAMEcolony2
	desc = {
		trigger = { character = event_target:NewColonyRuler }
		text = EVTDESCcolony2_new_ruler
	}
	desc = {
		trigger = {
			NOT = { character = event_target:NewColonyRuler }
			has_character_flag = NewColonyLord
		}
		text = EVTDESCcolony2_lord
	}
	desc = {
		trigger = {
			NOT = { character = event_target:NewColonyRuler }
			NOT = { has_character_flag = NewColonyLord }
			independent = no
		}
		text = EVTDESCcolony2_vassal
	}
	desc = {
		trigger = {
			NOT = { character = event_target:NewColonyRuler }
			NOT = { has_character_flag = NewColonyLord }
			independent = yes
		}
		text = EVTDESCcolony2_founder
	}
	picture = GFX_evt_explore_ocean
	border = GFX_event_narrative_frame_diplomacy
	portrait = event_target:NewColonyRuler

	is_triggered_only = yes # create_colony

	# The new colony lord
	option = {
		name = EVTOPTAcolony2_new_ruler
		trigger = { character = event_target:NewColonyRuler }
		prestige = 500
	}

	# Everyone else
	option = {
		name = EVTOPTAcolony2
		trigger = {
			NOT = { character = event_target:NewColonyRuler }
			NOT = { check_variable = { which = countys_given_away_value value = 1 } }
		}
	}

	# Lost land to the colony
	option = {
		name = EVTOPTBcolony2
		trigger = {
			NOT = { character = event_target:NewColonyRuler }
			check_variable = { which = countys_given_away_value value = 1 }
		}
		custom_tooltip = { text = colony_compensated_tt }
	}

	after = {
		colony_cleanup = yes
	}
}
