###############################################################################
# Coronation
# Namespace: coronation (was HF.20200 - HF.20420 in HF_coronation_events;
#   coronation.N = old HF.20N)
# Faction: Old World faiths
# Serves: kings and emperors of the Imperial cults (Sigmar, Ulric, Taal and
#   Rhya, Morr), plus the Tzar of Kislev for the Frost Crown
#
# A ruler of the Imperial cults is not truly king until a priest sets the
# crown on their head. The ruler picks who crowns them (any priest, a high
# priest, or the head of the cult), pays what the priest asks, sets a budget
# and holds a month of feasting full of guests, rivals, knights and
# sermons. Then the crowning itself, or a self-coronation scandal. An
# uncrowned ruler suffers restless peasants, scolding priests and nagging
# spouses. Crowned rulers receive homage from new vassals.
# The Emperor gets the Crown of Sigmar and the Tzar the Frost Crown; both
# follow their title, never the dead ruler's own heir. (Bretonnia's crown
# is handled by bretonnia.9/10 in WHG_bretonnia_events.)
#
# Choosing the priest and the price    coronation.200 - coronation.210,
#                                      coronation.212, coronation.217,
#                                      coronation.235 - coronation.241
# The cult's harder prices            coronation.211 (free the temple),
#                                      coronation.214 (war on the cast-out),
#                                      coronation.215 (depose the pretender),
#                                      coronation.218 (the target is told),
#                                      coronation.216, coronation.219,
#                                      coronation.221, coronation.222
#                                      (take back a holy site in ten years)
# Wars for a coronation               coronation.220, coronation.224,
#                                      coronation.227 - coronation.229,
#                                      coronation.418 - coronation.420
#   (CBs excommunicated_ruler in 00_cb_types, depose_cult_pretender in
#   wh_cb_types)
# Priest dies, ruler converts         coronation.231 - coronation.234,
#                                      coronation.408, coronation.409
# Budget, invitations, guests         coronation.300 - coronation.304
# The month of feasting               coronation.305 - coronation.365
# The crowning, self-coronation       coronation.366 - coronation.371
# Game start, uncrowned rulers        coronation.372 - coronation.388,
#                                      coronation.405, coronation.416,
#                                      coronation.417
# After the coronation                coronation.389 - coronation.392
# Homage of new vassals               coronation.393 - coronation.398,
#                                      coronation.410 - coronation.413
# The crowns                          coronation.399 - coronation.407,
#                                      coronation.414
#
# Saved: character flags flag_crowned_by_*, flag_*_for_coronation and
#   flag_holy_site_for_coronation, many
#   coronation progress flags; global flags crown_of_sigmar_made and
#   kislev_frost_crown_made; event targets coronation_ruler,
#   coronation_priest, crown_heir_empire, crown_heir_kislev,
#   excommunication_war_target, pretender_war_target, coronation_holy_site.
#
# Callers:
#   decisions/HF_coronation_decisions.txt (200),
#   common/on_actions/00_on_actions.txt: on_startup (372), on_yearly_pulse
#     (405), on_five_year_pulse (375; random list 399, 400), on_death
#     (227-229, 231, 374, 393, 406), on_new_holder and
#     on_new_holder_inheritance (414), on_new_holder_usurpation (414, 417),
#     on_character_convert_religion (408, 416);
#   common/cb_types/00_cb_types.txt and 01_cb_types.txt (220, 224);
#   events/HFP_religious_events.txt (234);
#   common/scripted_effects/00_scripted_effects.txt.
#   events/WHG_kislev_crown_events.txt calls coronation_regalia_effect.
# Support: common/scripted_triggers/wh_coronation_triggers.txt,
#   common/scripted_effects/wh_coronation_effects.txt,
#   common/artifacts/33_artifacts_crowns.txt
# Loc: localisation/00_WHG_coronation_events.csv

	id = coronation.414
	hide_window = yes

	only_rulers = yes

	is_triggered_only = yes

		OR = {
			AND = { FROM = { title = e_the_empire } FROMFROM = { has_artifact = crown_of_sigmar } }
			AND = { FROM = { title = e_kislev } FROMFROM = { has_artifact = kislev_frost_crown } }

				}
			}
		}
		if = {
			limit = { FROM = { title = e_kislev } FROMFROM = { has_artifact = kislev_frost_crown } }
			FROMFROM = {
