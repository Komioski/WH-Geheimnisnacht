# Council actions: what happens while a councillor works a job action
# Namespace: marshal (Part 1, was 938 - 942 and 20221 - 20421 in vanilla
#   job_marshal.txt), steward (Part 2)
# Faction: Generic
# Serves: every ruler with a council, and the councillors on it
#
# These events stay MTTH: the engine only checks them for a councillor doing
# the job action, and each one is listed in that action's events = { } in
# common/job_actions/00_job_actions.txt.
#
# Part 1: the Marshal
#   Advance military technology
#     An accident at the drill yard               marshal.1 - marshal.2
#     A lord shares his secrets                   marshal.10 - marshal.13
#   Train troops
#     Fresh levies                                marshal.20 - marshal.21
#     A lord holds back his levies                marshal.30 - marshal.32
#   Organize raid (tribal)
#     The raiders muster                          marshal.40 - marshal.43
#
# Part 2: the Steward (namespace steward, was 920 - 925 and 20200 - 20244 in
#   vanilla job_steward.txt)
#   Squeeze peasants
#     The tithe                                   steward.1 - steward.2
#     The peasants attack                         steward.10 - steward.14
#     Tax riots                                   steward.20 - steward.22
#   Oversee construction
#     A master builder                            steward.30 - steward.31
#     Sabotage                                    steward.40 - steward.45
#   Settle tribe (tribal)
#     The tribe settles                           steward.50 - steward.51
#     The settlers fight back                     steward.55 - steward.56
#   Build legend (tribal)
#     Warriors flock to the hall                  steward.60 - steward.62
#   Collect tribute (nomad)
#     Extra tribute                               steward.70 - steward.75
#
# Scripted triggers: common/scripted_triggers/WHG_council_action_triggers.txt
# Scripted effects: common/scripted_effects/WHG_council_action_effects.txt
# Localisation: localisation/00_WHG_council_action_events.csv
# Full list with callers: docs/events/event_database.csv

# marshal.1 An accident at the drill yard
# The marshal is hurt while testing new arms and drills: three times in four
# he is wounded, else maimed. A clumsy marshal is hurt more often. The liege
# hears of it by letter.
# Scope: ROOT is the marshal.
character_event = {
	id = marshal.1
	desc = EVTDESCmarshal1
	picture = GFX_evt_bloody_man

	min_age = 16
	capable_only = yes
	prisoner = no
	has_job_title = yes

	trigger = {
		has_job_action = action_advance_mil_tech
		council_action_fit_trigger = yes
		can_be_maimed_trigger = yes
		marshal_at_post_trigger = yes
	}

	mean_time_to_happen = {
		months = 240
		modifier = { factor = 0.75 NOT = { martial = 2 } }
		modifier = { factor = 0.75 NOT = { martial = 3 } }
		modifier = { factor = 0.75 NOT = { martial = 4 } }
		modifier = { factor = 0.75 NOT = { martial = 5 } }
		modifier = { factor = 1.25 martial = 10 }
		modifier = { factor = 1.25 martial = 11 }
		modifier = { factor = 1.25 martial = 12 }
		modifier = { factor = 1.25 martial = 13 }
	}

	immediate = {
		random_list = {
			75 = { set_character_flag = marshal_wounded }
			25 = { set_character_flag = marshal_maimed }
		}
	}

	option = { # Wounded
		name = EVTOPTAmarshal1
		trigger = { has_character_flag = marshal_wounded }
		add_trait = wounded
		liege = { letter_event = { id = marshal.2 tooltip = TOOLTIPmarshal2 } }
		clr_character_flag = marshal_wounded
	}
	option = { # Maimed
		name = EVTOPTBmarshal1
		trigger = { has_character_flag = marshal_maimed }
		add_maimed_trait_effect = yes
		liege = {
			set_character_flag = marshal_maimed_news
			letter_event = { id = marshal.2 tooltip = TOOLTIPmarshal2 }
		}
		clr_character_flag = marshal_maimed
	}
}

# marshal.13 The liege gains the secrets
# 50 military points. Once in a lifetime, if the liege has no such book yet,
# the lord's drill book itself may come with the letter (1 in 10).
# Scope: ROOT is the liege; FROM is the marshal, FROMFROM the lord.
letter_event = {
	id = marshal.13
	desc = EVTDESCmarshal13

	is_triggered_only = yes # marshal.12

	immediate = {
		if = {
			limit = {
				NOT = { has_character_flag = marshal_drill_book_given }
				NOT = { has_artifact = military_strategy_guide }
			}
			random = {
				chance = 10
				set_character_flag = marshal_drill_book_given
				set_character_flag = marshal_drill_book_now
			}
		}
	}

	option = { # A fine gift
		name = EVTOPTAmarshal13
		military_techpoints = 50
		FROMFROM = { reverse_opinion = { modifier = opinion_grateful who = ROOT years = 2 } }
		if = {
			limit = { has_character_flag = marshal_drill_book_now }
			add_artifact = military_strategy_guide
			clr_character_flag = marshal_drill_book_now
		}
	}
}

# marshal.40 The raiders muster
# Tribal job action. Once every ten years the marshal of a liege at peace
# raises a band of raiders in the capital. Counts see it less often, the
# raider bloodline more often.
# Scope: ROOT is the marshal.
character_event = {
	id = marshal.40
	desc = EVTDESCmarshal40
	picture = GFX_evt_battle
	border = GFX_event_normal_frame_war

	min_age = 16
	capable_only = yes
	prisoner = no
	has_job_title = yes

	trigger = {
		has_job_action = action_organize_raid
		council_action_fit_trigger = yes
		liege = {
			is_adult = yes
			prisoner = no
			war = no
			NOT = { trait = incapable }
			OR = {
				NOT = { has_character_flag = tribal_organize_raid_troops }
				had_character_flag = { flag = tribal_organize_raid_troops days = 3650 }
			}
		}
		location = { county = { is_occupied = no } ROOT = { liege = { is_capital = PREVPREV } } }
	}

	mean_time_to_happen = {
		months = 120
		modifier = { factor = 3.0 liege = { tier = COUNT } }
		modifier = {
			factor = 0.75
			liege = {
				any_owned_bloodline = {
					has_bloodline_flag = more_raiders_bloodline
					bloodline_is_active_for = PREV
				}
			}
		}
		modifier = { factor = 1.25 NOT = { martial = 2 } }
		modifier = { factor = 1.25 NOT = { martial = 3 } }
		modifier = { factor = 1.25 NOT = { martial = 4 } }
		modifier = { factor = 1.25 NOT = { martial = 5 } }
		modifier = { factor = 0.75 martial = 9 }
		modifier = { factor = 0.75 martial = 10 }
		modifier = { factor = 0.75 martial = 11 }
		modifier = { factor = 0.75 martial = 12 }
		modifier = { factor = 0.75 martial = 13 }
	}

	option = { # Lead them to the liege
		name = EVTOPTAmarshal40
		prestige = 50
		hidden_tooltip = { liege = { character_event = { id = marshal.41 } } }
	}
}

# marshal.41 The liege gets the raiders
# Three looter regiments in the capital (bigger with the raider bloodline).
# If they have not gone raiding after four months they disband (marshal.42,
# marshal.43).
# Scope: ROOT is the liege; FROM is the marshal.
character_event = {
	id = marshal.41
	desc = EVTDESCmarshal41
	picture = GFX_evt_battle
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # marshal.40

	immediate = {
		capital_scope = { marshal_raiders_spawn_effect = yes }
		clr_character_flag = tribal_organize_raid_troops
		set_character_flag = tribal_organize_raid_troops
		character_event = { id = marshal.42 days = 30 }
	}

	option = { # Well done
		name = EVTOPTAmarshal41
		opinion = { who = FROM modifier = opinion_grateful months = 60 }
	}
}

# marshal.42 Are the raiders out? (hidden)
# Checks every 30 days while the raiders sit at home.
# Scope: ROOT is the liege.
character_event = {
	id = marshal.42
	hide_window = yes
	is_triggered_only = yes # marshal.41, marshal.42

	immediate = {
		if = {
			limit = { has_earmarked_regiments_not_raiding = tribal_organize_raid }
			if = {
				limit = { had_character_flag = { flag = tribal_organize_raid_troops days = 122 } }
				character_event = { id = marshal.43 }
			}
			else = { character_event = { id = marshal.42 days = 30 } }
		}
	}
}
