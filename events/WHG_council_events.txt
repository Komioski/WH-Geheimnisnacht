# ze_lord_spiritual_events).

#   10-20    Chancellor: language lessons, bureaucratic abuse of office
#   100-123  Spymaster: councillor robs the coffers, blackmail, mad safety
#            measures against spies and vampires

#            monument, lessons in stewardship
#   300-350  Marshal: commander suggestion, war machine experiments
#   400-420  Marshal: stirs disloyalty, protection racket
#   500-540  Lord Spiritual: mission to the northern heathens, temple,
#            secret sympathies, selling favours, cowardly missionary
#
# Most chains start from the yearly pulse random_events in 00_on_actions.

# council.10 Chancellor offers language lessons
# Yearly pulse. A diligent or silver-tongued chancellor is tired of the
# ruler embarrassing the court in front of foreign envoys.
# Scope: ROOT is the ruler; hidden, fires council.11 for the chancellor.
character_event = {
	id = council.10
	hide_window = yes

	is_triggered_only = yes

	min_age = 16
	only_playable = yes
	only_capable = yes

	trigger = {
		NOR = {
			trait = charismatic_negotiator
			trait = grey_eminence
			trait = gregarious
			is_inaccessible_trigger = yes
			has_character_modifier = multilingual
		}
		prisoner = no
		job_chancellor = {
			is_alive = yes
			vassal_of = ROOT
			NOT = { is_inaccessible_trigger = yes }
			NOT = { character = ROOT }
			OR = { trait = diligent AND = { trait = gregarious diplomacy = 12 } }
		}
	}

	weight_multiplier = {
		days = 1
		modifier = { factor = 2 NOT = { diplomacy = 10 } }
		modifier = { factor = 1.5 trait = wroth }
	}

	immediate = {
		job_chancellor = { character_event = { id = council.11 } }
	}
}

	id = council.16
	desc = EVTDESCcouncil16

	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

		name = EVTOPTAcouncil16
		ai_chance = {
			factor = 20
			modifier = { factor = 2 event_target:language_teacher = { diplomacy = 12 } }
		}
		hidden_tooltip = { character_event = { id = council.15 days = 340 random = 150 } }

			chance = 15

				hidden_tooltip = { # Notify diligent
					character_event = { id = 38256 }
				}
			}
			if = {
				limit = { trait = slothful }
				remove_trait = slothful
				hidden_tooltip = { # Notify loss of slothful
					character_event = { id = 38298 }
				}
			}
		}
		random = {
			chance = 30
			if = {
				limit = { NOT = { trait = stressed } }
				add_trait = stressed
				hidden_tooltip = { # Notify stressed
					character_event = { id = 38282 }
				}
			}
		}
	}

	option = {
		name = EVTOPTBcouncil16
		ai_chance = {
			factor = 5
			modifier = { factor = 5 trait = slothful }
			modifier = { factor = 2 event_target:language_teacher = { trait = underhanded_rogue } }
			modifier = { factor = 3 event_target:language_teacher = { trait = naive_appeaser } }
		}
		clear_delayed_event = { id = council.13 }
		remove_character_modifier = student_of_language
		if = {
			limit = { NOT = { trait = slothful } }
			random = {
				chance = 15
				add_trait = slothful
				hidden_tooltip = {
					character_event = { id = 38251 } # Notify Slothful
				}
			}
		}
		ROOT = { custom_tooltip = { text = EVTTOOLTIPcouncil16 } }
		event_target:language_teacher = {
			opinion = { who = ROOT modifier = opinion_disappointed years = 5 }
			hidden_tooltip = { character_event = { id = council.17 } }
		}

# council.20 Bureaucratic abuse of office
# Yearly pulse. A peasant flees the chancellor's fee-and-queue racket;
# depose and imprison, or let it slide.
# Scope: ROOT is the ruler.
character_event = {
	id = council.20
	desc = EVTDESCcouncil20
	picture = GFX_evt_bad_news_female
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	min_age = 16
	only_playable = yes
	only_capable = yes

	trigger = {
		NOT = { is_inaccessible_trigger = yes }
		prisoner = no
		job_chancellor = {
			NOT = { opinion = { who = liege value = -74 } }
			NOR = { trait = honest is_inaccessible_trigger = yes }
			ai = yes
		}
	}

	weight_multiplier = {
		days = 1

		modifier = { factor = 2 job_chancellor = { trait = deceitful } }
		modifier = { factor = 2 job_chancellor = { trait = arbitrary } }
		modifier = { factor = 0.5 job_chancellor = { trait = craven } }
		modifier = { factor = 0.5 job_chancellor = { trait = slothful } }
	}

	immediate = {
		job_chancellor = { wealth = 100 set_character_flag = chancellor_abuse_of_office }
	}

	option = {
		name = EVTOPTAcouncil20 # Depose + get imprison reason
		job_chancellor = {
			reverse_opinion = { who = ROOT years = 3 modifier = abuse_of_office }
			remove_title = job_chancellor
			prestige = -50
			piety = -50
			imprison = ROOT
		}
	}
	option = {
		name = EVTOPTBcouncil20 # Turn a blind eye for now but get imprison reason
		capital_scope = { add_province_modifier = { modifier = excessive_bureaucracy months = 60 } }
		job_chancellor = {
			reverse_opinion = { who = ROOT years = 3 modifier = abuse_of_office }
			opinion = { who = ROOT years = 3 modifier = opinion_turned_a_blind_eye }
		}
	}
}
