# Court life: the courtiers of the human courts
# Namespace: court (Part 1, was humanflavor.0 - humanflavor.1 in
#   BR_random_flavor_events.txt)
# Faction: Generic
# Serves: human rulers of the Old World faiths (Part 1), player kings and
#   emperors of every race (Part 2)
#
# A courtier riding the borders with a patrol comes home a hero after
# cutting down a band of beastmen or bandits, or comes home alone after
# leaving the men to die. The ruler honours or envies the hero and shames,
# jails, banishes or sends the coward back out to redeem themselves.
# The Druchii have their own take on this (druchii.50 - druchii.51).
#
# Part 1: courtiers on patrol          court.1 - court.2
# Life at court                        court.10 - court.19
#   (was 73000 - 73009 in the vanilla court_events.txt)
# A seat on the council                court.20 - court.27
#   (was 73200 - 73245, five copies of one chain; now with Conclave too)
# Court upkeep                         court.30 - court.36
#   (was LT.61001 - 61008 and LT.10103 in LT_court_maintenance_events.txt)
# Part 2: the hero knight               heroknight.1 - heroknight.8
#   (namespace heroknight, was CM.2000 - CM.2030 in cm_various_events.txt)
# Part 3: eunuchs of the court        eunuch.1 - eunuch.15
#   (namespace eunuch, was LoR.800 - 805 and 885 - 890 in byzantine_events.txt)
#
# Fired from: on_bi_yearly_pulse random list in wh_on_actions.txt (1, 2);
#   on_yearly_pulse random_events in 00_on_actions.txt (10 - 19); the
#   on_yearly_pulse list court_council_petition in wh_on_actions.txt (20);
#   on_yearly_pulse events (30) and random_events (32, 36) in
#   00_on_actions.txt; the court exodus decision in LT_decisions.txt (34).
#   Parts 2 and 3 have their own callers.
# Saved: event targets court_hero, court_coward, knocked_up_maid,
#   grumpy_priest, court_petitioner, court_holder; character flags
#   court_fashion_wife, court_petition_open, court_petition_<job>,
#   asked_for_job.
# Scripted effects: common/scripted_effects/wh_court_effects.txt
# Custom loc: GetCourtPetitionJob in
#   localisation/customizable_localisation/wh_court_customizable_localisation.txt
# Loc: localisation/00_WHG_court_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = court

# court.1 A courtier wins a victory
# A brave and skilled courtier cuts down a band that cornered their patrol.
# Honour them, give them a Sword of Battle if they carry no blade of note,
# or sneer that you would have
# done as well.
# Scope: ROOT is the ruler, event_target:court_hero the courtier.
character_event = {
	id = court.1
	desc = EVTDESCcourt1
	picture = GFX_evt_crusaders
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # on_bi_yearly_pulse random list
	only_playable = yes
	religion_group = old_world_group
	prisoner = no

	trigger = {
		any_courtier = {
			is_adult = yes
			prisoner = no
			trait = creature_human
			martial = 12
			trait = brave
		}
	}

	immediate = {
		random_courtier = {
			limit = {
				is_adult = yes
				prisoner = no
				trait = creature_human
				martial = 12
				trait = brave
			}
			save_event_target_as = court_hero
		}
	}

	option = { # Honour them
		name = EVTOPTAcourt1
		event_target:court_hero = {
			prestige = 100
			opinion = { who = ROOT modifier = opinion_grateful months = 60 }
		}
		prestige = 25
		ai_chance = { factor = 60 }
	}
	option = { # A Sword of Battle for a fine warrior
		name = EVTOPTBcourt1
		trigger = {
			wealth = 50
			event_target:court_hero = {
				NOT = { any_artifact = { has_artifact_flag = combat_weapon } }
			}
		}
		wealth = -50
		event_target:court_hero = {
			prestige = 100
			opinion = { who = ROOT modifier = opinion_grateful months = 120 }
			add_artifact = sword_of_battle
		}
		prestige = 25
		ai_chance = {
			factor = 20
			modifier = { factor = 2 trait = generous }
			modifier = { factor = 0 trait = greedy }
		}
	}
	option = { # I could have done as well
		name = EVTOPTCcourt1
		trigger = { OR = { trait = proud trait = envious trait = cruel } }
		event_target:court_hero = {
			prestige = 25
			opinion = { who = ROOT modifier = opinion_disappointed months = 36 }
		}
		prestige = 10
		ai_chance = { factor = 20 }
	}
}

# court.2 A courtier flees
# A craven, weak, drunk or foolish courtier runs from an ambush and leaves
# the patrol to die. Shame, jail or banish them, or send them back out to
# win their name back.
# Scope: ROOT is the ruler, event_target:court_coward the courtier.
character_event = {
	id = court.2
	desc = EVTDESCcourt2
	picture = GFX_evt_crusaders
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # on_bi_yearly_pulse random list
	only_playable = yes
	religion_group = old_world_group
	prisoner = no

	trigger = {
		any_courtier = {
			is_adult = yes
			prisoner = no
			trait = creature_human
			OR = { trait = craven trait = misguided_warrior trait = drunkard trait = weak }
		}
	}

	immediate = {
		random_courtier = {
			limit = {
				is_adult = yes
				prisoner = no
				trait = creature_human
				OR = { trait = craven trait = misguided_warrior trait = drunkard trait = weak }
			}
			save_event_target_as = court_coward
		}
	}

	option = { # Ill-done
		name = EVTOPTAcourt2
		event_target:court_coward = { prestige = -100 }
		prestige = -25
		ai_chance = { factor = 20 modifier = { factor = 2 trait = kind } }
	}
	option = { # The dungeon
		name = EVTOPTBcourt2
		event_target:court_coward = { imprison = ROOT prestige = -100 }
		prestige = -10
		ai_chance = { factor = 30 modifier = { factor = 2 trait = wroth } }
	}
	option = { # Banish them
		name = EVTOPTCcourt2
		event_target:court_coward = { prestige = -100 banish = yes }
		ai_chance = { factor = 20 }
	}
	option = { # Back out there
		name = EVTOPTDcourt2
		custom_tooltip = { text = EVTTOOLTIPcourt2_d }
		event_target:court_coward = {
			random_list = {
				40 = {
					prestige = 50
					change_martial = 1
					if = { limit = { trait = craven } remove_trait = craven }
				}
				40 = { prestige = -100 }
				20 = { death = { death_reason = death_battle } }
			}
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 2 trait = brave }
			modifier = { factor = 2 trait = just }
		}
	}
}
