###############################################################################
# Court life: the courtiers of the human courts
# Namespace: court (Part 1, was humanflavor.0 - humanflavor.1 in
#   BR_random_flavor_events.txt)
# Faction: Generic
# Serves: human rulers of the Old World faiths (Part 1), player kings and
#   emperors of every race (Part 2)
#
# A courtier riding the borders with a patrol comes home a hero after
# cutting down a band of beastmen or bandits, or comes home alone after
# leaving the men to die. The ruler honours or envies the hero and shames,
# jails, banishes or sends the coward back out to redeem themselves.
# The Druchii have their own take on this (druchii.50 - druchii.51).
#
# Part 1: courtiers on patrol          court.1 - court.2
# Part 2: the hero knight               heroknight.1 - heroknight.8
#   (namespace heroknight, was CM.2000 - CM.2030 in cm_various_events.txt)
#
# Fired from: on_bi_yearly_pulse random list in wh_on_actions.txt (1, 2);
#   Part 2 has its own callers.
# Saved: event targets court_hero, court_coward.
# Loc: localisation/00_WHG_court_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = court

# court.1 A courtier wins a victory
# A brave and skilled courtier cuts down a band that cornered their patrol.
# Honour them, give them a Sword of Battle if they carry no blade of note,
# or sneer that you would have
# done as well.
# Scope: ROOT is the ruler, event_target:court_hero the courtier.
character_event = {
	id = court.1
	desc = EVTDESCcourt1
	picture = GFX_evt_crusaders
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # on_bi_yearly_pulse random list
	only_playable = yes
	religion_group = old_world_group
	prisoner = no

	trigger = {
		any_courtier = {
			is_adult = yes
			prisoner = no
			trait = creature_human
			martial = 12
			trait = brave
		}
	}

	immediate = {
		random_courtier = {
			limit = {
				is_adult = yes
				prisoner = no
				trait = creature_human
				martial = 12
				trait = brave
			}
			save_event_target_as = court_hero
		}
	}

	option = { # Honour them
		name = EVTOPTAcourt1
		event_target:court_hero = {
			prestige = 100
			opinion = { who = ROOT modifier = opinion_grateful months = 60 }
		}
		prestige = 25
		ai_chance = { factor = 60 }
	}
	option = { # A Sword of Battle for a fine warrior
		name = EVTOPTBcourt1
		trigger = {
			wealth = 50
			event_target:court_hero = { NOT = { any_artifact = { has_artifact_flag = combat_weapon } } }
		}
		wealth = -50
		event_target:court_hero = {
			prestige = 100
			opinion = { who = ROOT modifier = opinion_grateful months = 120 }
			add_artifact = sword_of_battle
		}
		prestige = 25
		ai_chance = {
			factor = 20
			modifier = { factor = 2 trait = generous }
			modifier = { factor = 0 trait = greedy }
		}
	}
	option = { # I could have done as well
		name = EVTOPTCcourt1
		trigger = { OR = { trait = proud trait = envious trait = cruel } }
		event_target:court_hero = {
			prestige = 25
			opinion = { who = ROOT modifier = opinion_disappointed months = 36 }
		}
		prestige = 10
		ai_chance = { factor = 20 }
	}
}

# court.2 A courtier flees
# A craven, weak, drunk or foolish courtier runs from an ambush and leaves
# the patrol to die. Shame, jail or banish them, or send them back out to
# win their name back.
# Scope: ROOT is the ruler, event_target:court_coward the courtier.
character_event = {
	id = court.2
	desc = EVTDESCcourt2
	picture = GFX_evt_crusaders
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # on_bi_yearly_pulse random list
	only_playable = yes
	religion_group = old_world_group
	prisoner = no

	trigger = {
		any_courtier = {
			is_adult = yes
			prisoner = no
			trait = creature_human
			OR = { trait = craven trait = misguided_warrior trait = drunkard trait = weak }
		}
	}

	immediate = {
		random_courtier = {
			limit = {
				is_adult = yes
				prisoner = no
				trait = creature_human
				OR = { trait = craven trait = misguided_warrior trait = drunkard trait = weak }
			}
			save_event_target_as = court_coward
		}
	}

	option = { # Ill-done
		name = EVTOPTAcourt2
		event_target:court_coward = { prestige = -100 }
		prestige = -25
		ai_chance = { factor = 20 modifier = { factor = 2 trait = kind } }
	}
	option = { # The dungeon
		name = EVTOPTBcourt2
		event_target:court_coward = { imprison = ROOT prestige = -100 }
		prestige = -10
		ai_chance = { factor = 30 modifier = { factor = 2 trait = wroth } }
	}
	option = { # Banish them
		name = EVTOPTCcourt2
		event_target:court_coward = { prestige = -100 banish = yes }
		ai_chance = { factor = 20 }
	}
	option = { # Back out there
		name = EVTOPTDcourt2
		custom_tooltip = { text = EVTTOOLTIPcourt2_d }
		event_target:court_coward = {
			random_list = {
				40 = {
					prestige = 50
					change_martial = 1
					if = { limit = { trait = craven } remove_trait = craven }
				}
				40 = { prestige = -100 }
				20 = {
					death = { death_reason = death_battle }
				}
			}
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 2 trait = brave }
			modifier = { factor = 2 trait = just }
		}
	}
}


###############################################################################
# Part 2: The hero knight (namespace heroknight)
# Faction: Generic
# Serves: player kings and emperors of every race
# (was CM.2000 - CM.2030 in the vanilla cm_various_events.txt; the guardian
#  chain had been switched off in the yearly pulse and is back)
#
# A battle-hardened champion comes to a king's gate asking to serve the
# house. Made guardian of a child, the hero may teach it courage, strength,
# swordplay or justice, once per child, or hurt it in the practice yard.
#
# The champion at the gate        heroknight.1
# The guardian's lessons          heroknight.2 - heroknight.8
#
# Callers: on_decade_pulse hero_knight_arrives list (1) and on_yearly_pulse
#   hero_knight_lessons list (2), wh_on_actions.txt.
# Saved: character flags hero_knight (also set by the bloodline heroes in
#   WHG_bloodline_events.txt), hero_knight_taught (on the child).
###############################################################################

namespace = heroknight

# heroknight.1 A champion at the gate
# A veteran warrior asks to serve a king or emperor. Taking him in gives a
# brave, strong, skilled commander loyal to the house.
# Scope: ROOT is the ruler.
character_event = {
	id = heroknight.1
	desc = EVTDESC_CM_2000
	picture = GFX_evt_emissary

	is_triggered_only = yes # on_decade_pulse hero_knight_arrives list
	only_playable = yes
	only_capable = yes
	min_age = 16
	prisoner = no
	ai = no

	trigger = {
		higher_tier_than = DUKE
		NOT = { any_courtier = { has_character_flag = hero_knight } }
	}

	option = { # Welcome!
		name = EVTOPTA_CM_2000
		create_character = {
			random_traits = no
			dynasty = random
			religion = ROOT
			culture = ROOT
			female = no
			age = 35
			trait = brave
			trait = strong
			trait = skilled_tactician
			attributes = { martial = 10 }
		}
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			set_character_flag = hero_knight
			opinion = { modifier = opinion_loyal_servant who = ROOT years = 100 }
		}
	}
	option = { # No room
		name = EVTOPTB_CM_2000
		prestige = 10
	}
}

# heroknight.2 The guardian's lessons
# Hidden. A hero knight guarding one of the ruler's children aged 9 or more
# teaches it one lesson (heroknight.3 - 6) or hurts it in practice
# (heroknight.7), once per child.
# Scope: ROOT is the ruler.
character_event = {
	id = heroknight.2
	hide_window = yes

	is_triggered_only = yes # on_yearly_pulse hero_knight_lessons list
	only_playable = yes
	ai = no

	trigger = {
		any_child = { heroknight_pupil_trigger = yes }
	}

	immediate = {
		random_child = {
			limit = { heroknight_pupil_trigger = yes }
			set_character_flag = hero_knight_taught
			save_event_target_as = heroknight_pupil
		}
		random_list = {
			10 = {
				modifier = { factor = 0 event_target:heroknight_pupil = { trait = brave } }
				modifier = { factor = 0 event_target:heroknight_pupil = { guardian = { trait = craven } } }
				character_event = { id = heroknight.3 }
			}
			10 = {
				modifier = { factor = 0 event_target:heroknight_pupil = { is_strong_trigger = yes } }
				character_event = { id = heroknight.4 }
			}
			10 = {
				modifier = { factor = 0 event_target:heroknight_pupil = { lifestyle_traits = 1 } }
				character_event = { id = heroknight.5 }
			}
			10 = {
				modifier = { factor = 0 event_target:heroknight_pupil = { trait = just } }
				modifier = { factor = 0 event_target:heroknight_pupil = { guardian = { trait = arbitrary } } }
				character_event = { id = heroknight.6 }
			}
			1 = {
				modifier = { factor = 0 event_target:heroknight_pupil = { trait = wounded } }
				modifier = { factor = 0 event_target:heroknight_pupil = { can_be_maimed_trigger = no } }
				character_event = { id = heroknight.7 }
			}
		}
	}
}

# heroknight.3 The child learns courage
# Scope: ROOT is the ruler, event_target:heroknight_pupil the child.
character_event = {
	id = heroknight.3
	desc = EVTDESCheroknight3

	option = {
		name = EVTOPTAheroknight3

		event_target:heroknight_pupil = { add_trait_brave_effect = yes }
	}
	option = {
		name = EVTOPTBheroknight3
		trigger = { trait = craven }
		event_target:heroknight_pupil = { add_trait_brave_effect = yes }
	}
}

# heroknight.7 Hurt in the practice yard
# The child falls hard at sword and lance practice: wounded, or now and then
# maimed (the notice is heroknight.8).
# Scope: ROOT is the ruler, event_target:heroknight_pupil the child.
character_event = {
	id = heroknight.7
	hide_window = yes

	is_triggered_only = yes # heroknight.2

	immediate = {
		event_target:heroknight_pupil = {
			random_list = {
				80 = { add_trait_wounded_effect = yes }
				20 = { add_maimed_trait_effect = yes }
			}
		}
		character_event = { id = heroknight.8 }
	}
}
