# Court offices: the honorary titles of the southern courts
# Namespace: judge (Part 1, was 105000 - 105044 in vanilla
#   muslim_honorary_titles_events.txt)
# Namespace: scribe (Part 2, was 105050 - 105069)
# Namespace: courtsong (Part 3, was 105075 - 105103)
# Namespace: taster (Part 4, was 105150 - 105253)
# Namespace: architect (Part 5, was 105200 - 105204)
# Namespace: fatwa (Part 6, was 88750 - 88754 in vanilla fatwa_events.txt)
# Faction: Araby & Southlands
# Serves: the Grand Judge and Court Calligrapher of the Arabyan courts, and
#   the Court Musician, Court Poet, Cupbearer, Food Taster and Chief
#   Architect of any court that grants them
#
# The Grand Judge keeps the Sacred Law of Ormazd: a wife may ask for a
# divorce, courtiers press the liege to fill an empty seat or to throw out an
# arbitrary judge, a drunkard faces eighty lashes and a lover-keeping vassal
# is called to account. The Court Calligrapher decorates a vassal's temple,
# writes a hilya of the Prophet al-Quyat for the Calif or the liege, or
# swindles the liege with the dream of a Golden Scripture. The Court
# Musician and Court Poet woo the liege's young wife at their peril. A
# Cupbearer or Food Taster who hates the liege may poison the cup. The Chief
# Architect asks for gold to mend the realm's buildings and may pocket it.
#
# Part 1: the Grand Judge
#   A wife asks for a divorce          judge.1 - judge.8
#   A judge is put forward             judge.10 - judge.15
#   The drunkard                       judge.30 - judge.32
#   The adulterer                      judge.40 - judge.49
# Part 2: the Court Calligrapher
#   The temple inscriptions            scribe.1 - scribe.6
#   The hilya                          scribe.10 - scribe.11
#   The Golden Scripture               scribe.20 - scribe.24
# Part 3: the Court Musician and Court Poet
#   The song                           courtsong.1 - courtsong.4
#   The poem                           courtsong.10 - courtsong.13
# Part 4: the Cupbearer and Food Taster
#   Poison at the table                taster.1 - taster.3, taster.10
# Part 5: the Chief Architect
#   The building plans                 architect.1 - architect.5
# Part 6: the fatwa
#   A fatwa against a decadent vassal  fatwa.1 - fatwa.5
#
# Fired from: on_five_year_pulse in wh_on_actions.txt (lists court_judge and
#   court_offices, which replace MTTH events of 200 - 500 months). judge.5
#   stays MTTH behind its flag, to wait out the iddah or a pregnancy.
#   Triggers in common/scripted_triggers/WHG_court_office_triggers.txt,
#   effects in common/scripted_effects/WHG_court_office_effects.txt.
# Saved: character modifiers qadi_divorce_timer, iddah_period,
#   suggested_qadi_timer, ignored_muslim_sins, mosque_calligrapher_timer,
#   beautiful_hilya, calligrapher_swindle_timer, musician_song_timer,
#   poet_verse_timer, cupbearer_poison_timer, food_taster_poison_timer and
#   architect_improvement_timer (the keys are vanilla's).
###############################################################################

# judge.1 A wife thinks of divorce
# A married woman of the Way of the One, whose husband is wrathful, cruel,
# arbitrary, greedy or hateful, thinks of asking the Grand Judge for a
# divorce. Once every three years at most.
# Scope: ROOT is the wife.
character_event = {
	id = judge.1
	desc = EVTDESCjudge1
	picture = GFX_evt_quarrel
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_five_year_pulse, list court_judge

	only_women = yes
	min_age = 16
	prisoner = no
	only_capable = yes
	religion_group = ormazd_gods_group
	is_married = yes

	trigger = {
		court_way_of_one_trigger = yes
		NOT = { has_minor_title = title_chief_qadi }
		NOT = { has_character_modifier = qadi_divorce_timer }
		NOT = { has_character_flag = qadi_granted_divorce }
		spouse = {
			OR = {
				trait = wroth
				trait = cruel
				trait = arbitrary
				trait = greedy
				NOT = { reverse_opinion = { who = ROOT value = 0 } }
			}
			NOT = { has_minor_title = title_chief_qadi }
		}
		liege = {
			any_courtier_or_vassal = {
				has_minor_title = title_chief_qadi
				NOT = { character = ROOT }
			}
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = {
			factor = 0.2
			trait = content
		}
		modifier = {
			factor = 2
			trait = honest
		}
		modifier = {
			factor = 0.5
			num_of_children = 2
		}
		modifier = {
			factor = 0.67
			num_of_children = 4
		}
		modifier = {
			factor = 0.75
			num_of_children = 6
		}
	}

	immediate = {
		add_character_modifier = {
			name = qadi_divorce_timer
			duration = 1095
			hidden = yes
		}
		save_event_target_as = divorcing_wife
		liege = {
			random_courtier_or_vassal = {
				limit = {
					has_minor_title = title_chief_qadi
					NOT = { character = ROOT }
				}
				save_event_target_as = court_judge
			}
		}
	}

	option = {
		name = EVTOPTAjudge1 # Go to the judge
		ai_chance = { factor = 40 }
		event_target:court_judge = {
			character_event = { id = judge.2 days = 1 random = 3 tooltip = EVTTOOLTIPjudge2 }
		}
	}
	option = {
		name = EVTOPTBjudge1 # Not now
		ai_chance = { factor = 60 }
		random = {
			chance = 20
			if = {
				limit = { NOT = { trait = depressed } }
				add_trait = depressed
				hidden_effect = { character_event = { id = traitnote.288 } } # Notify Depressed
			}
		}
	}
}

# judge.5 The iddah is over
# MTTH behind the qadi_granted_divorce flag: fires once a month has passed
# and the wife is not with child. She may still change her mind.
# Scope: ROOT is the wife.
character_event = {
	id = judge.5
	desc = EVTDESCjudge5
	picture = GFX_evt_quarrel
	border = GFX_event_normal_frame_religion

	has_character_flag = qadi_granted_divorce
	only_women = yes
	prisoner = no

	trigger = {
		is_pregnant = no
		had_character_flag = {
			flag = qadi_granted_divorce
			days = 30
		}
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		name = EVTOPTAjudge5 # Divorced
		remove_character_modifier = iddah_period
		clr_character_flag = qadi_granted_divorce
		if = {
			limit = { is_married = yes }
			spouse = { remove_spouse = ROOT }
		}
	}
	option = {
		name = EVTOPTBjudge5 # I stay
		trigger = { is_married = yes }
		remove_character_modifier = iddah_period
		clr_character_flag = qadi_granted_divorce

				years = 5
			}
			character_event = { id = judge.8 days = 1 random = 2 tooltip = EVTTOOLTIPjudge8_stays }
		}
	}
}

# judge.10 A Grand Judge is put forward
# A courtier or vassal of a southern king or emperor worries that the seat of
# the Grand Judge is empty, or that the judge in it is an arbitrary or lazy
# fool, and puts forward a courtier, a vassal or themselves. Once per
# lifetime; the liege hears such counsel at most every two years.
# Scope: ROOT is the courtier or vassal.
character_event = {
	id = judge.10
	desc = {
		trigger = { NOT = { has_character_flag = court_judge_replacing } }
		text = EVTDESCjudge10
	}
	desc = {
		trigger = { has_character_flag = court_judge_replacing }
		text = EVTDESCjudge10_replace
	}
	picture = GFX_evt_emissary
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_five_year_pulse, list court_judge

	min_age = 16
	prisoner = no
	only_capable = yes
	religion_group = ormazd_gods_group
	war = no

	trigger = {
		court_way_of_one_trigger = yes
		in_command = no
		NOT = { has_minor_title = title_chief_qadi }
		NOT = { has_character_flag = qadi_suggester }
		liege = {
			NOT = { character = ROOT }
			uses_decadence = yes
			prisoner = no
			religion_group = ormazd_gods_group
			NOT = { has_character_modifier = suggested_qadi_timer }
			primary_title = {
				OR = {
					tier = king
					tier = emperor
				}
			}
			OR = {
				NOT = { any_courtier_or_vassal = { has_minor_title = title_chief_qadi } }
				any_courtier_or_vassal = { court_judge_unfit_trigger = yes }
			}
		}
		# Someone fit to put forward. Replacing a judge takes a learned,
		# diligent candidate (court_judge_suggestion_trigger).
		OR = {
			AND = {
				liege = { NOT = { any_courtier_or_vassal = { has_minor_title = title_chief_qadi } } }
				OR = {
					court_judge_candidate_trigger = yes
					liege = {
						any_courtier_or_vassal = {
							NOT = { character = ROOT }
							court_judge_candidate_trigger = yes
							NOT = { any_rival = { character = ROOT } }
						}
					}
				}
			}
			AND = {
				NOT = { trait = content }
				liege = { any_courtier_or_vassal = { court_judge_unfit_trigger = yes } }
				OR = {
					AND = {
						court_judge_candidate_trigger = yes
						learning = 15
						NOT = { trait = arbitrary }
						NOT = { trait = slothful }
					}
					liege = {
						any_courtier_or_vassal = {
							NOT = { character = ROOT }
							court_judge_candidate_trigger = yes
							NOT = { any_rival = { character = ROOT } }
							learning = 15
							NOT = { trait = arbitrary }
							NOT = { trait = slothful }
						}
					}
				}
			}
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = {
			factor = 2
			trait = diligent
		}
		modifier = {
			factor = 2
			trait = ambitious
		}
		modifier = {
			factor = 0.5
			trait = slothful
		}
		modifier = {
			factor = 0.5
			trait = content
		}
	}

	immediate = {
		set_character_flag = qadi_suggester
		liege = {
			add_character_modifier = {
				name = suggested_qadi_timer
				duration = 730
				hidden = yes
			}
			if = {
				limit = { any_courtier_or_vassal = { court_judge_unfit_trigger = yes } }
				random_courtier_or_vassal = {
					limit = { court_judge_unfit_trigger = yes }
					save_event_target_as = old_judge
				}
				ROOT = { set_character_flag = court_judge_replacing }
			}
		}
	}

	option = {
		name = EVTOPTAjudge10 # Put forward a courtier
		trigger = {
			liege = {
				any_courtier = {
					NOT = { character = ROOT }
					court_judge_suggestion_trigger = yes
				}
			}
		}
		liege = {
			random_courtier = {
				limit = {
					NOT = { character = ROOT }
					court_judge_suggestion_trigger = yes
				}
				save_event_target_as = target_suggested_qadi
			}
			character_event = { id = judge.11 days = 2 random = 4 tooltip = EVTTOOLTIPjudge11 }
		}
	}
	option = {
		name = EVTOPTBjudge10 # Put myself forward
		trigger = {
			court_judge_candidate_trigger = yes
			OR = {
				NOT = { has_character_flag = court_judge_replacing }

			}

				}
				save_event_target_as = target_suggested_qadi
			}
			character_event = { id = judge.11 days = 2 random = 4 tooltip = EVTTOOLTIPjudge11 }

	}
	option = {
		name = EVTOPTDjudge10 # Hold my tongue
		clr_character_flag = court_judge_replacing
	}

# judge.11 The liege hears the counsel
# The liege appoints the candidate (dismissing the old judge, if any) or
# sends the courtier away.
# Scope: ROOT is the liege, FROM the courtier who spoke up.
character_event = {
	id = judge.11
	desc = {
		trigger = { FROM = { NOT = { has_character_flag = court_judge_replacing } } }
		text = EVTDESCjudge11
	}
	desc = {
		trigger = { FROM = { has_character_flag = court_judge_replacing } }
		text = EVTDESCjudge11_replace
	}
	picture = GFX_evt_emissary
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPTAjudge11 # Appoint them
		ai_chance = {
			factor = 40
			modifier = {
				factor = 1.5
				event_target:target_suggested_qadi = { learning = 15 }
			}
		}
		if = {
			limit = { FROM = { has_character_flag = court_judge_replacing } }
			event_target:old_judge = {
				remove_title = title_chief_qadi
				if = {
					limit = { has_opinion_modifier = { who = ROOT modifier = opinion_chief_qadi } }
					remove_opinion = { who = ROOT modifier = opinion_chief_qadi }
				}
				character_event = { id = judge.15 days = 1 tooltip = EVTTOOLTIPjudge15 }
			}
		}
		event_target:target_suggested_qadi = {
			give_minor_title = title_chief_qadi
			if = {
				limit = { character = FROM }
				character_event = { id = judge.14 days = 3 random = 3 tooltip = EVTTOOLTIPjudge14_yes }
			}
			else = {
				character_event = { id = judge.12 days = 3 random = 3 tooltip = EVTTOOLTIPjudge12 }
			}
		}
	}
	option = {
		name = EVTOPTBjudge11 # No
		ai_chance = { factor = 60 }
		FROM = {
			if = {
				limit = { character = event_target:target_suggested_qadi }
				opinion = {
					modifier = opinion_disappointed
					who = ROOT
					years = 5
				}
				character_event = { id = judge.14 days = 3 random = 3 tooltip = EVTTOOLTIPjudge14_no }
			}
			else = {
				character_event = { id = judge.13 days = 3 random = 3 tooltip = EVTTOOLTIPjudge13_no }
			}
		}
	}
}
