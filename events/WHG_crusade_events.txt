# Crusades: holy blood and the Errantry War against Araby
# Namespace: crusade (was HF.49020 - HF.49121 and HFP.49122 - HFP.49123 in
#   HF_crusade_events)
# Faction: Old World faiths
#   (the crusader bloodlines of any Great Crusade; the Errantry War is the
#   Bretonnian Knights of Antoch against Araby)
#
# Crusader blood: the knight who carries a Great Crusade twice, or kills
# a dozen foes in its battles, founds a holy bloodline.
#
# The Errantry War: once in a game, after 2100, the Knights of Antoch (a
# Grail order) turn on Araby. Old World rulers near the sands may ask the
# order to march on a Sultan, paying in piety, gold or land. The order
# fights for one Arabyan kingdom at a time (CB southern_crusade_war); when
# every county of it is held by the faithful, the war is won and the
# order becomes the Kingdom of Antoch. Lose the war twice and it is over.
# In 1449 the King of Bretonnia declared the first Errantry War against the
# Arabyans who had invaded Estalia; this is its heir.
#
# Crusader blood                    crusade.1 - crusade.2
# Broken pacts                      crusade.5 - crusade.6
# The Errantry War is called        crusade.10 - crusade.14
# Asking the knights for aid        crusade.15 - crusade.18
# The march                         crusade.19 - crusade.21
# Rewards, defeat and victory       crusade.22 - crusade.26
# On campaign                       crusade.30 - crusade.31
# The diverted Errantry War         crusade.50 - crusade.54
#
# The diverted Errantry War (the old Fourth Crusade, with the game rule on):
# when the knights first sail, a Tilean prince who built their fleet may
# demand his price. The master may pay, refuse, or settle the debt by
# sacking the prince's rival port, Miragliano, Remas or Luccini (CB
# diverted_errantry_war). The sacked city goes to the creditor, its gold
# to the knights, and the master's house earns a black bloodline. Then
# the fleet sails on for Araby.
#
# Saved: title flags northern_crusade_target (on the order: it has a
#   kingdom to conquer) and northern_crusade_active_target (on that
#   kingdom); global flags northern_crusades_active, northern_crusade_success,
#   northern_crusade_failure, diverted_errantry_done; persistent target
#   teutonic_target on the order (the kingdom). Character flags
#   diverted_errantry_attacker, diverted_errantry_target,
#   diverted_errantry_creditor; title flag diverted_errantry_port. Event
#   targets: northern_crusade_asker,
#   northern_crusade_potential_target, northern_crusade_target_character,
#   northern_crusade_kingdom_title, northern_crusade_duchy_title,
#   diverted_port, diverted_port_title, diverted_target, diverted_creditor.
#
# Callers:
#   common/on_actions/00_on_actions.txt: on_yearly_pulse (10, 26, 30),
#     on_battle_won_leader, on_major_battle_won_leader, on_siege_won_leader,
#     on_unit_entering_province (31), on_crusade_launches (5).
#   common/cb_types/00_cb_types.txt: new_crusade (1).
#   common/cb_types/ka_cb_types.txt southern_crusade_war and
#     01_cb_types.txt northern_crusade_war (22, 24, via the effects 23, 25);
#     ka_cb_types.txt diverted_errantry_war (53, 54).
#   common/scripted_effects/04_scripted_effects_crusades.txt (2).
#   decisions/HFP_holy_order_decisions.txt: ask_teutonic_for_help_teutonic,
#     ask_teutonic_for_help_pagan (15).
# Support: common/scripted_effects/wh_crusade_effects.txt,
#   common/scripted_triggers/wh_crusade_triggers.txt
# Loc: localisation/00_WHG_crusade_events.csv
#
# Also in this file (each part keeps its own header and namespace):
#   The Reconquest of the South (namespace reconquest, was WHG_reconquest_events)
#   Holy orders (namespace holyorder, was WHG_holy_order_events)
###############################################################################

# crusade.1 Grand Crusader
# The knight who did most for a second Great Crusade founds a holy line.
# Scope: ROOT is that knight (new_crusade CB on_success).
narrative_event = {
	id = crusade.1
	title = EVTNAMEcrusade1
	desc = EVTDESCcrusade1
	picture = GFX_evt_knight_kneeling
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes
	has_dlc = "Holy Fury"

	trigger = {
		NOT = {
			any_owned_bloodline = {
				has_bloodline_flag = crusader_bloodline
				founder = { character = ROOT }
			}
		}
	}

	immediate = {
		sound_effect = bloodline_added
	}

	option = {
		name = EVTOPTAcrusade1
		if = { limit = { is_female = no } create_bloodline = { type = grand_crusader } }
		else = { create_bloodline = { type = grand_crusader inheritance = matrilineal } }
		set_bloodline_founder_religion_flag_effect = yes
	}
}

# crusade.2 Commander of the Crusade
# A knight's twelfth kill on a Great Crusade: he cuts down the enemy
# commander and founds a holy line.
# Scope: ROOT is the knight; event_target enemy_commander
#   (crusader_commander_counter_effect).
narrative_event = {
	id = crusade.2
	title = EVTNAMEcrusade2
	desc = EVTDESCcrusade2
	picture = GFX_evt_melee
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes
	has_dlc = "Holy Fury"

	trigger = {
		NOT = {
			any_owned_bloodline = {
				has_bloodline_flag = crusader_bloodline
				founder = { character = ROOT }
			}
		}
	}

	immediate = {
		event_target:enemy_commander = { any_controlled_unit = { morale = -1 } }
		sound_effect = bloodline_added
	}

	option = {
		name = EVTOPTAcrusade2
		event_target:enemy_commander = {
			if = {
				limit = { ai = yes } # A player's commander lives
				if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
				else = { death = { death_reason = death_battle killer = ROOT } }
			}
		}
		if = { limit = { is_female = no } create_bloodline = { type = crusader_commander } }
		else = { create_bloodline = { type = crusader_commander inheritance = matrilineal } }
		set_bloodline_founder_religion_flag_effect = yes
	}
}

# A Great Crusade launches: every crusader who has a non-aggression pact
# with its target breaks it.

		any_crusade_participant = {

# Scope: ROOT is the crusader (crusade.5).

	id = crusade.6
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_alive = yes

	}

		crusade_target_char = { save_event_target_as = sad_nap_target }
		reverse_opinion = {

			modifier = broken_non_aggression_pact
		}

		add_character_modifier = { name = broken_non_aggression_pact_char_modifier years = 10 }
	}
}

# crusade.13 The knights sail
# Five to ten years after the call, the order picks a target if none
# has been asked of it (crusade.50 first, which may divert the fleet).
# Scope: Couronne (crusade.12, crusade.14, crusade.51, the effect
#   crusade_diverted_cleanup_effect).
province_event = {
	id = crusade.13
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = northern_crusades_active
		crusade_south_target_set_trigger = no
	}

	immediate = {
		if = {
			limit = { is_title_active = d_knights_of_antoch }
			d_knights_of_antoch = { holder_scope = { character_event = { id = crusade.50 } } }
		}
		else_if = {
			limit = { is_title_active = k_knights_of_antoch }
			k_knights_of_antoch = { holder_scope = { character_event = { id = crusade.50 } } }
		}
	}
}

# crusade.14 The order picks its own target
# The largest Sultan near the Old World. If no one has asked the order for
# aid by now, it chooses itself; with no target, try again in a year.
# Scope: ROOT is the master of the order (crusade.50, crusade.51).
character_event = {
	id = crusade.14
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				crusade_south_target_set_trigger = no
				any_independent_ruler = { crusade_south_target_ruler_trigger = yes }
			}
			random_independent_ruler = {
				limit = { crusade_south_target_ruler_trigger = yes }
				preferred_limit = { realm_size = 30 }
				preferred_limit = { realm_size = 20 }
				preferred_limit = { realm_size = 10 }
				random_demesne_title = {
					limit = { higher_tier_than = COUNT crusade_south_region_trigger = yes }
					kingdom = {
						set_title_flag = northern_crusade_active_target
						save_event_target_as = northern_crusade_kingdom_title
					}
				}
				save_event_target_as = northern_crusade_target_character
			}
			character_event = { id = crusade.19 }
		}
		else_if = {
			limit = { crusade_south_target_set_trigger = no }
			810 = { province_event = { id = crusade.13 days = 365 } }
		}
	}
}

# crusade.15 The order weighs a plea
# A ruler of the faith names a Sultan for the knights to march on. The
# master's price depends on his opinion of the asker: prayers, gold or land.
# Scope: ROOT is the master of the order; event targets
#   northern_crusade_asker and northern_crusade_potential_target
#   (ask_teutonic_for_help_teutonic, ask_teutonic_for_help_pagan).
character_event = {
	id = crusade.15
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { crusade_south_target_set_trigger = no }
			event_target:northern_crusade_potential_target = {
				random_realm_title = {
					limit = { crusade_south_region_trigger = yes }
					kingdom = { save_event_target_as = northern_crusade_kingdom_title }
				}
			}
			event_target:northern_crusade_asker = {
				any_landed_title = {
					limit = {
						tier = COUNT
						location = { crusade_south_region_trigger = yes }
						kingdom = { title = event_target:northern_crusade_kingdom_title }
						owner = {
							OR = { any_liege = { character = PREVPREVPREV } character = PREVPREV }
						}
					}
					location = { duchy = { save_event_target_as = northern_crusade_duchy_title } }
				}
			}
		}
		else = {
			event_target:northern_crusade_potential_target = {
				random_realm_title = {
					limit = {
						tier = COUNT
						kingdom = { has_title_flag = northern_crusade_active_target }
					}
					kingdom = { save_event_target_as = northern_crusade_kingdom_title }
				}
			}
			event_target:northern_crusade_asker = {
				any_landed_title = {
					limit = {
						tier = COUNT
						kingdom = { has_title_flag = northern_crusade_active_target }
						owner = {
							OR = { any_liege = { character = PREVPREVPREV } character = PREVPREV }
						}
					}
					location = { duchy = { save_event_target_as = northern_crusade_duchy_title } }
				}
			}
		}

		random_list = {
			# Asks only for prayers and piety
			10 = {
				trigger = { opinion = { who = event_target:northern_crusade_asker value = 1 } }
				modifier = {
					factor = 1.5
					opinion = { who = event_target:northern_crusade_asker value = 25 }
				}
				modifier = {
					factor = 1.5
					opinion = { who = event_target:northern_crusade_asker value = 50 }
				}
				modifier = {
					factor = 1.5
					opinion = { who = event_target:northern_crusade_asker value = 75 }
				}
				modifier = {
					factor = 1.5
					opinion = { who = event_target:northern_crusade_asker value = 90 }
				}
				modifier = {
					factor = 0.5
					NOT = { opinion = { who = event_target:northern_crusade_asker value = 10 } }
				}
				modifier = { factor = 0.5 OR = { trait = ambitious trait = greedy } }
				modifier = { factor = 1.5 OR = { trait = humble trait = content } }
				modifier = { factor = 0.1 is_rival = event_target:northern_crusade_asker }
				event_target:northern_crusade_asker = { letter_event = { id = crusade.16 } }
			}
			# Asks for gold
			10 = { event_target:northern_crusade_asker = { letter_event = { id = crusade.17 } } }
			# Asks for land
			10 = {
				trigger = {
					NOT = { opinion = { who = event_target:northern_crusade_asker value = 60 } }
					event_target:northern_crusade_asker = {
						any_landed_title = {
							kingdom = { title = event_target:northern_crusade_kingdom_title }
							owner = {
								OR = {
									any_liege = { character = PREVPREVPREV }
									character = PREVPREV
								}
							}
						}
						any_demesne_title = {
							tier = COUNT
							location = {
								duchy = {
									NOT = { title = event_target:northern_crusade_duchy_title }
								}
							}
						}
					}
				}
				modifier = {
					factor = 1.5
					NOT = { opinion = { who = event_target:northern_crusade_asker value = -10 } }
				}
				modifier = {
					factor = 1.5
					NOT = { opinion = { who = event_target:northern_crusade_asker value = 0 } }
				}
				modifier = {
					factor = 1.5
					NOT = { opinion = { who = event_target:northern_crusade_asker value = 15 } }
				}
				modifier = {
					factor = 0.5
					opinion = { who = event_target:northern_crusade_asker value = 25 }
				}
				modifier = {
					factor = 0.5
					opinion = { who = event_target:northern_crusade_asker value = 50 }
				}
				modifier = {
					factor = 0.5
					opinion = { who = event_target:northern_crusade_asker value = 75 }
				}
				modifier = { factor = 1.5 OR = { trait = ambitious trait = greedy } }
				modifier = { factor = 0.5 OR = { trait = content trait = humble } }
				modifier = { factor = 3 is_rival = event_target:northern_crusade_asker }
				event_target:northern_crusade_asker = { letter_event = { id = crusade.18 } }
			}
		}
	}
}

# crusade.17 The knights ask for gold
# Scope: ROOT is the asker; FROM is the master of the order (crusade.15).
letter_event = {
	id = crusade.17
	desc = EVTDESCcrusade17

	is_triggered_only = yes

	immediate = {
		random_list = {
			10 = {
				trigger = { FROM = { opinion = { who = ROOT value = 1 } } }
				modifier = { factor = 1.5 FROM = { opinion = { who = ROOT value = 25 } } }
				modifier = { factor = 1.5 FROM = { opinion = { who = ROOT value = 50 } } }
				modifier = { factor = 1.5 FROM = { opinion = { who = ROOT value = 75 } } }
				modifier = { factor = 0.5 FROM = { NOT = { opinion = { who = ROOT value = 10 } } } }
				modifier = { factor = 0.5 FROM = { OR = { trait = ambitious trait = greedy } } }
				modifier = { factor = 1.5 FROM = { OR = { trait = humble trait = content } } }
				modifier = { factor = 0.1 FROM = { is_rival = ROOT } }
				set_character_flag = northern_crusade_small_donation
			}
			10 = { set_character_flag = northern_crusade_medium_donation }
			10 = {
				modifier = { factor = 1.5 FROM = { NOT = { opinion = { who = ROOT value = 0 } } } }
				modifier = { factor = 1.5 FROM = { NOT = { opinion = { who = ROOT value = 15 } } } }
				modifier = { factor = 0.5 FROM = { opinion = { who = ROOT value = 25 } } }
				modifier = { factor = 0.5 FROM = { opinion = { who = ROOT value = 50 } } }
				modifier = { factor = 1.5 FROM = { OR = { trait = ambitious trait = greedy } } }
				modifier = { factor = 0.5 FROM = { OR = { trait = content trait = humble } } }
				modifier = { factor = 3 FROM = { is_rival = ROOT } }
				set_character_flag = northern_crusade_large_donation
			}
		}
	}

	option = {
		name = EVTOPTAcrusade17

			limit = { has_character_flag = northern_crusade_small_donation }

# crusade.19 The knights march
# The order declares war on the target for the kingdom (or, on later wars,
# for the kingdom it is already conquering), gets a fresh army and calls its
# allies.
# Scope: ROOT is the master of the order (crusade.14, the effect
#   crusade_south_call_order_effect).
character_event = {
	id = crusade.19

# The Reconquest of the South
# Namespace: reconquest (was HF.49200 - HF.49482 in HF_reconquista_events)
# Faction: Old World faiths
#   (Estalian and Tilean lords against whoever else holds Estalia and Tilea:
#   the Sultans of Araby first, but also greenskins, the dead, Skaven)
#
# When the Sultan Jaffar's armies landed in Estalia in 1448, the knights of
# the Old World drove them back into the sea. Whenever Estalia or Tilea falls
# again to a ruler outside the Old World faiths, a southern lord may swear to
# take it back: he prepares for a year (decision prepare_reconquista), raising
# troops, ships and commanders, gathering allies and sellswords, then wars on
# the occupiers land by land (CBs reconquista_astarios ... _monte_castello,
# one for each land of Estalia and Tilea). An adventurer may rise to do it
# for a southern king who will not. When no foreign ruler holds land in the
# south, the reconquest is over for good. The vanilla beneficiary clean-ups
# (old reconquest.4-6) are gone: no CB here names a beneficiary.
#
# Clean up                          reconquest.1 - reconquest.9
# Allies                            reconquest.10 - reconquest.13
# Holy blood                        reconquest.20 - reconquest.21
# Sellswords                        reconquest.30 - reconquest.35
# Preparations                      reconquest.40 - reconquest.49
# News                              reconquest.50 - reconquest.52
# The adventurer                    reconquest.60 - reconquest.63
# On campaign                       reconquest.70 - reconquest.71
# AI cooldown                       reconquest.80 - reconquest.82

# reconquest.1 The reconquest ends
# Clears every reconquest flag and disbands the levies and ships.
# Scope: ROOT is the reconquistador (the CBs, reconquest.2, 3, 8).
character_event = {
	id = reconquest.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		clr_character_flag = currently_reconquering_iberia
		clr_global_flag = active_reconquista
		any_character = {
			limit = {
				OR = {
					has_character_flag = promised_reconquista_ally
					has_character_flag = reconquista_adventurer
					has_character_flag = reconquista_sponsor
				}
			}
			clr_character_flag = promised_reconquista_ally
			clr_character_flag = reconquista_adventurer
			clr_character_flag = reconquista_sponsor
		}
		disband_event_forces = conscripted_reconquista_ships
		disband_event_forces = reconquista_event_troops_rabble
	}
}

# reconquest.8 The moment passes
# Two years after the oath and still no war: a player's reconquest lapses,
# the AI marches on the nearest target.
# Scope: ROOT is the reconquistador (prepare_reconquista, two years on).
character_event = {
	id = reconquest.8
	desc = EVTDESCreconquest8
	picture = GFX_evt_bad_news
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	trigger = {
		has_character_modifier = reconquista_preparation_left
	}

	immediate = {
		if = {
			limit = { ai = no }
			remove_character_modifier = reconquista_preparation_left
			disband_event_forces = reconquista_event_troops
			character_event = { id = reconquest.1 }
		}
		else = {
			set_character_flag = ai_reconquista_preparation_time_run_out
			character_event = { id = reconquest.49 }
		}
	}

	option = {
		name = EVTOPTAreconquest8
	}
}

# reconquest.33 Gold for your sword
# A side at war offers a lord gold to fight for it.
# Scope: ROOT is the lord; FROMFROM is the side at war
#   (reconquest_hire_sellsword_defender, reconquest_hire_sellsword_attacker).
letter_event = {
	id = reconquest.33
	desc = EVTDESCreconquest33
	border = GFX_event_letter_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPTAreconquest33
		if = {
			limit = { real_tier = COUNT }
			transfer_scaled_wealth = { from = FROMFROM value = 0.25 }
		}
		else_if = {
			limit = { real_tier = DUKE }
			transfer_scaled_wealth = { from = FROMFROM value = 0.5 }
		}
		else_if = {
			limit = { real_tier = KING }
			transfer_scaled_wealth = { from = FROMFROM value = 0.75 }
		}
		else = { transfer_scaled_wealth = { from = FROMFROM value = 1 } }
		custom_tooltip = {
			text = join_reconquista_war_other_TT
			FROMFROM = {
				any_war = {
					limit = { reconquest_cb_trigger = yes }
					if = {
						limit = { any_defender = { character = PREVPREV } }
						ROOT = { join_defender_wars = PREV }
					}
					else = { ROOT = { join_attacker_wars = PREV } }
				}
			}
		}
		reconquest_sellsword_greed_effect = yes
		FROMFROM = { letter_event = { id = reconquest.34 days = 3 } }
	}

	option = {
		name = EVTOPTBreconquest33
		FROMFROM = { letter_event = { id = reconquest.35 days = 3 } }
	}
}

# reconquest.40 The year of preparation
# Every month or so while preparing (every few days for an adventurer), one
# of the preparation events; at the end of the year, reconquest.49.
# Scope: ROOT is the reconquistador (prepare_reconquista, reconquest.61).
character_event = {
	id = reconquest.40
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_character_modifier = reconquista_preparation_duration
		not_reconquista_check_trigger = yes
		has_global_flag = active_reconquista
	}

	fail_trigger_effect = {
		if = {
			limit = { not_reconquista_check_trigger = yes has_global_flag = active_reconquista }
			character_event = { id = reconquest.49 }
		}
	}

	immediate = {
		random_list = {
			5 = { # Raise troops
				character_event = { id = reconquest.41 }
			}
			4 = { # The head of the faith
				trigger = {
					NOR = {
						has_character_modifier = reconquista_pope_aid_event_cooldown
						has_character_flag = reconquista_adventurer
					}
					rightful_religious_head_scope = { always = yes }
				}
				character_event = { id = reconquest.42 }
			}
			2 = { # A captain
				trigger = {
					NOT = { has_character_modifier = reconquista_commander_event_cooldown }
				}
				character_event = { id = reconquest.43 }
			}
			2 = { # Ships
				trigger = { NOT = { has_character_modifier = reconquista_boat_event_cooldown } }
				character_event = { id = reconquest.44 }
			}
		}
	}

	after = {
		if = {
			limit = { has_character_flag = reconquista_adventurer }
			character_event = { id = reconquest.40 days = 2 random = 5 }
		}
		else = { character_event = { id = reconquest.40 days = 25 random = 20 } }
	}
}

	id = reconquest.43
	desc = EVTDESCreconquest43
	picture = GFX_evt_dueling_knights_hf
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	immediate = {
		hidden_tooltip = {
			create_character = {
				dynasty = random
				culture = ROOT
				religion = ROOT
				health = 5
				attributes = { martial = 4 diplomacy = 0 stewardship = 2 intrigue = 0 learning = 0 }
			}
			new_character = {
				character_event = { id = aiclean.31 } # Racial trait
				save_event_target_as = new_commander
				opinion = { modifier = opinion_loyal_servant who = ROOT }
				add_trait = brilliant_strategist
				random_list = {
					6 = { } # Of the lord's own people
					2 = { culture = norteno }
					2 = { culture = avilian }
					2 = { culture = galice }
					2 = { culture = obregonese }
					2 = { culture = comarcano }
					2 = { culture = astur }
					2 = { culture = montanese }
					2 = { culture = tigarrese }
					2 = { culture = reman }
					2 = { culture = umbrian }
					2 = { culture = oltremari }
					2 = { culture = tobaran }
					2 = { culture = lucian }
					2 = { culture = miraglian }
				}
				random_list = {
					1 = { add_trait = strong }
					1 = { add_trait = robust }
				}
				random_list = {
					1 = { add_trait = strategist }
					1 = { add_trait = duelist }
					1 = { add_trait = hunter }
				}
				random_list = {
					1 = { add_trait = patient }
					1 = { add_trait = diligent }
				}
				random_list = {
					1 = { add_trait = proud }
					1 = { add_trait = envious }
				}
				random = { chance = 40 add_trait = ambitious }
				random = { chance = 40 add_trait = brave }
				random = { chance = 40 add_trait = zealous }
				add_random_commander_trait_effect = yes
			}
		}
	}

	option = { # Welcome to the court
		name = EVTOPTAreconquest43
	}

	option = { # Make him a commander
		name = EVTOPTBreconquest43
		trigger = {
			OR = {
				AND = {
					tier = COUNT
					NOT = {
						any_realm_character = {
							count = 2
							liege = { character = ROOT }
							has_minor_title = title_commander

						}

					NOT = {
						any_realm_character = {
							count = 8
							liege = { character = ROOT }
							has_minor_title = title_commander
						}
					}
				}
			}

		name = EVTOPTCreconquest43

###############################################################################
# Order fortresses (holyorder.20 - holyorder.24)
# The vanilla Templar, Hospitaller and Teutonic castle chains (8500 - 8799
# in knights_templar_events, knights_of_st_john_events and
# teutonic_order_events, never copied into the mod), opened to every holy
# order: once in twenty years a Grand Master asks a lord of his own faith
# for leave to raise a chapter fortress in a frontier province that borders
# the lands of another faith. A lord who grants it wins piety and the
# order's thanks, and the order builds a castle in the province; once in a
# lifetime the order lends him a lance of its knights. A lord who refuses
# angers the order and the head of his faith, and one who swears never to
# give land to an order is never asked again.
#   holyorder.20  A Grand Master looks for a frontier (hidden)
#   holyorder.21  The order asks for a fortress
#   holyorder.22  The fortress is raised (hidden, province)
#   holyorder.23  The fortress stands
#   holyorder.24  A lance of the order (once per lifetime)
# Saved: character flags order_fortress_asked (on the master, twenty
#   years), denied_order_fortress (on the lord), no_land_to_religious_orders
#   (vanilla flag), order_fortress_lance_lent (once per lifetime); event
#   targets order_master, order_fortress_province; unit earmark
#   order_fortress_lance.
# Callers: common/on_actions/wh_on_actions.txt on_five_year_pulse (20).
###############################################################################

# holyorder.20 A Grand Master looks for a frontier (hidden)
# Scope: the master of a holy order. Fired by the five-year pulse.
character_event = {
	id = holyorder.20
	hide_window = yes

	is_triggered_only = yes

	only_rulers = yes
	prisoner = no

	trigger = {
		holy_order = yes
		is_adult = yes
		NOT = { had_character_flag = { flag = order_fortress_asked days = 7300 } }
	}

	immediate = {
		save_event_target_as = order_master
		random_province = {
			limit = {
				has_empty_holding = yes
				owner = {
					NOT = { character = event_target:order_master }
					religion = event_target:order_master
					higher_tier_than = BARON
					prisoner = no
					is_adult = yes
					NOT = { has_character_flag = denied_order_fortress }
					NOT = { has_character_flag = no_land_to_religious_orders }
					top_liege = { NOT = { has_character_flag = no_land_to_religious_orders } }
					NOT = { war_with = event_target:order_master }
				}
				any_neighbor_province = {
					has_owner = yes
					owner = { NOT = { religion = event_target:order_master } }
				}
			}
			save_event_target_as = order_fortress_province
			ROOT = {
				clr_character_flag = order_fortress_asked
				set_character_flag = order_fortress_asked
			}
			owner = { letter_event = { id = holyorder.21 } }
		}
	}
}
