###############################################################################
# Culture conversion events
# Namespace: culture (was the vanilla ids 55000 - 55100 in
# culture_conversion_events.txt; the Kislevite melting pot 55002 - 55006 moved
# to WHG_kislev_events.txt as kislev.150 - 154)
# Faction: All
#
# Provinces slowly take up the culture of their lords: from neighbours of
# the lord's culture, or by settlers from the homeland at a conquered
# capital. Vassals resent a liege who goes native, and a liege hears when a
# tribal vassal settles down.
#
# Province conversion: the five-year roll  culture.1
# The people take up their lord's ways      culture.2
# Settlers from the homeland                culture.5
# A liege goes native                       culture.10
# A tribe settles down                      culture.20
#
# Fired from common/on_actions/wh_on_actions.txt: culture.1 on the five-year
# pulse. decisions/conversion_decisions.txt (convert_to_local_culture) fires
# culture.10; decisions/realm_decisions.txt (convert_to_feudalism_vassal)
# fires culture.20.
# Scripted triggers: common/scripted_triggers/wh_culture_triggers.txt
# Loc: localisation/00_WHG_culture_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = culture

###############################################################################
# Province conversion
###############################################################################

# culture.1 The five-year roll
# Hidden. Every five years each province a ruler holds directly may take up
# the ruler's culture: about 5% per roll (once in a century, as the old
# 1,200-month MTTH), faster with a good steward, a conquest culture, recent
# depopulation, a cultural saint or a culture wonder. The capital may draw
# settlers instead when no neighbour shares the ruler's culture. Replaces two
# province MTTH events that every province checked every month.
# Scope: ROOT is a ruler.
character_event = {
	id = culture.1
	hide_window = yes

	only_rulers = yes

	is_triggered_only = yes # on_five_year_pulse

	trigger = {
		culture_fixed_culture_trigger = no
		top_liege = { culture = ROOT }
		OR = {
			any_demesne_province = { culture_assimilates_trigger = yes }
			capital_scope = { culture_settlers_trigger = yes }
		}
	}

	immediate = {
		any_demesne_province = {
			limit = { culture_assimilates_trigger = yes }
			random = {
				chance = 5
				modifier = { factor = 0.67 ROOT = { NOT = { stewardship = 2 } } }
				modifier = { factor = 0.67 ROOT = { NOT = { stewardship = 3 } } }
				modifier = { factor = 0.67 ROOT = { NOT = { stewardship = 4 } } }
				modifier = { factor = 0.67 ROOT = { NOT = { stewardship = 5 } } }
				modifier = { factor = 1.33 ROOT = { stewardship = 9 } }
				modifier = { factor = 1.33 ROOT = { stewardship = 11 } }
				modifier = { factor = 1.33 ROOT = { stewardship = 13 } }
				modifier = { factor = 1.33 ROOT = { stewardship = 15 } }
				modifier = {
					factor = 1.05
					ROOT = {
						job_treasurer = {
							NOT = { stewardship = 10 }
							has_job_action = action_administer_realm
						}
					}
				}
				modifier = {
					factor = 1.33
					ROOT = {
						job_treasurer = {
							stewardship = 10
							has_job_action = action_administer_realm
						}
					}
				}
				modifier = { factor = 1.33 has_province_flag = recent_depopulation_1 }
				modifier = { factor = 2 has_province_flag = recent_depopulation_2 }
				modifier = { factor = 4 has_province_flag = recent_depopulation_3 }
				modifier = { factor = 0.33 culture_slower_conversion_trigger = yes }
				modifier = { factor = 1.33 culture_conquest_culture_trigger = yes }
				modifier = {
					factor = 1.67
					ROOT = {
						any_owned_bloodline = {
							has_bloodline_flag = cultural_saintly_bloodline
							founder = { culture = ROOT }
						}
					}
				}
				modifier = {
					factor = 1.33
					ROOT = {
						top_liege = {
							any_realm_wonder_upgrade = {
								has_wonder_upgrade_flag = culture_conversion_faster
								is_active = yes
								wonder = { original_wonder_owner = { culture_group = ROOT } }
							}
						}
					}
				}
				province_event = { id = culture.2 }
			}
		}
		capital_scope = {
			if = {
				limit = { culture_settlers_trigger = yes }
				random = {
					chance = 5
					modifier = { factor = 0.67 ROOT = { NOT = { stewardship = 2 } } }
					modifier = { factor = 0.67 ROOT = { NOT = { stewardship = 3 } } }
					modifier = { factor = 0.67 ROOT = { NOT = { stewardship = 4 } } }
					modifier = { factor = 0.67 ROOT = { NOT = { stewardship = 5 } } }
					modifier = { factor = 1.33 ROOT = { stewardship = 9 } }
					modifier = { factor = 1.33 ROOT = { stewardship = 11 } }
					modifier = { factor = 1.33 ROOT = { stewardship = 13 } }
					modifier = { factor = 1.33 ROOT = { stewardship = 15 } }
					modifier = {
						factor = 1.05
						ROOT = {
							job_treasurer = {
								NOT = { stewardship = 10 }
								has_job_action = action_administer_realm
							}
						}
					}
					modifier = {
						factor = 1.33
						ROOT = {
							job_treasurer = {
								stewardship = 10
								has_job_action = action_administer_realm
							}
						}
					}
					modifier = { factor = 1.33 has_province_flag = recent_depopulation_1 }
					modifier = { factor = 2 has_province_flag = recent_depopulation_2 }
					modifier = { factor = 4 has_province_flag = recent_depopulation_3 }
					modifier = { factor = 0.33 culture_slower_conversion_trigger = yes }
					modifier = {
						factor = 2 # the steppe peoples move whole clans
						ROOT = {
							OR = {
								culture_group = hung_group
								culture_group = kurgan_group
								culture = ungol
							}
						}
					}
					province_event = { id = culture.5 }
				}
			}
		}
	}
}

# culture.2 The people take up their lord's ways
# Scope: ROOT is the province, FROM its owner.
province_event = {
	id = culture.2
	desc = EVTDESCculture2
	picture = "GFX_evt_culture_change"

	is_triggered_only = yes # culture.1

	option = {
		name = EVTOPTAculture2
		culture = FROM
		if = {
			limit = { has_province_modifier = nomad_agitation }
			remove_province_modifier = nomad_agitation
		}
	}
}

# culture.5 Settlers from the homeland
# Settlers of the ruler's culture and faith fill the conquered capital.
# Scope: ROOT is the capital, FROM its owner.
province_event = {
	id = culture.5
	desc = EVTDESCculture5
	picture = "GFX_evt_culture_change"

	is_triggered_only = yes # culture.1

	option = {
		name = EVTOPTAculture5
		culture = FROM
		if = { limit = { NOT = { religion = FROM } } religion = FROM }
		if = {
			limit = { has_province_modifier = nomad_agitation }
			remove_province_modifier = nomad_agitation
		}
	}
}

###############################################################################
# Rulers and their ways
###############################################################################

# culture.10 A liege goes native
# A vassal hears that their liege took up the culture of a foreign capital
# (decision convert_to_local_culture). The opinion is set by the decision.
# Scope: ROOT is the vassal, FROM the liege.
character_event = {
	id = culture.10
	desc = EVTDESCculture10
	picture = "GFX_evt_scandal"

	is_triggered_only = yes # convert_to_local_culture

	option = {
		name = EVTOPTAculture10
		tooltip = { opinion = { who = FROM modifier = opinion_adopted_foreign_ways years = 10 } }
	}
}

# culture.20 A tribe settles down
# Scope: ROOT is the liege, FROM the vassal who gave up tribal rule.
character_event = {
	id = culture.20
	desc = EVTDESCculture20
	picture = "GFX_evt_courier"

	notification = yes

	is_triggered_only = yes # convert_to_feudalism_vassal

	option = {
		name = OK
	}
}
