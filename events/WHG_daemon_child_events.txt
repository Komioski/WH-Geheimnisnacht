###############################################################################
# The daemon child
# Namespace: daemonchild (was SoA.3000 - SoA.3092 in soa_various_events.txt,
#   the vanilla Demon Child chain)
# Faction: Chaos
# Serves: players of the Old World, Araby, Stromfels and Norsca whose child
#   the Dark Gods mark; the Chaos cults (Monks and Mystics)
#
# Once in a game the Dark Gods mark a young child at a player's court. The
# parent sees the signs and may ask the court chaplain, who may calm them,
# accuse the mother of witchcraft or call for her abdication. As the child
# grows it tries to murder siblings who stand before it, a witch nanny of
# the cult comes to raise it and may die for it, and wounds, blindness,
# castration, madness, plague and prison do not hold it. At adulthood its
# true nature shows: a genius, possessed, with the voice of a greater
# daemon, it leads a Chaos cult or warband; at war three witches ride to its
# side with armies. Cultists rejoice when it comes of age and mourn when it
# dies. The Dark Gods can also send such a child to an unholy mother
# (chaoscult.173 in WHG_chaos_cult_events). The vanilla Sunset Invasion
# branch (the child sails west and returns as an Aztec emperor) is gone.
#
# A child is marked                    daemonchild.1 - daemonchild.3
# The chaplain and the accusation      daemonchild.4 - daemonchild.17
# The child grows                      daemonchild.19 - daemonchild.21
# Murder in the nursery                daemonchild.30 - daemonchild.41
# Nothing holds it                     daemonchild.50 - daemonchild.61
# The witches                          daemonchild.65 - daemonchild.78
# The cult                             daemonchild.90 - daemonchild.92
#
# Callers: wh_on_actions.txt (on_yearly_pulse daemon_child list: 1),
#   00_on_actions.txt (on_adulthood: 20, on_death: 91, the yearly pulse
#   recovery list: 50 - 53, 60).
# Saved: global flag demon_child_born (once a game); character flags
#   demon_child_non_pagan, demon_child_non_pagan_got_nanny,
#   demon_child_non_pagan_cultists, witch_nanny, demonspawn_will_join_*;
#   character modifiers demon_child_non_pagan, suspected_demon_spawn,
#   suspected_witch, known_witch, voice_of_greater_daemon, demon_horns.
###############################################################################

namespace = daemonchild

# daemonchild.1 A child is marked
# Once a game, a young child of a player of the Old World, Araby, Stromfels or
# Norsca is touched by the Dark Gods (daemonchild.2). More likely for the
# possessed, mad, cruel and impaling, and four times for devil worshippers. Was
# MTTH 500 years.
# Scope: ROOT is the parent. From on_yearly_pulse (daemon_child list).
character_event = {
	id = daemonchild.1
	hide_window = yes

	is_triggered_only = yes
	only_playable = yes

	trigger = {
		OR = {
			has_dlc = "Sons of Abraham"
			has_dlc = "Mystics"
		}
		OR = {
			religion_group = old_world_group
			religion_group = ormazd_gods_group
			religion_group = stromfels_group
			religion = norscan_gods
			religion = norscan_gods_reformed
		}
		any_child = {
			NOT = { age = 3 }
			is_ill = no
			NOT = { trait = inbred }
			mother = {
				always = yes
			}
		}
		NOT = { has_global_flag = demon_child_born }
	}

	weight_multiplier = {
		factor = 1
		modifier = {
			factor = 2
			trait = possessed
		}
		modifier = {
			factor = 1.33
			OR = {
				trait = lunatic_1
				trait = lunatic_2
				trait = lunatic_3
			}
		}
		modifier = {
			factor = 1.33
			trait = cruel
		}
		modifier = {
			factor = 1.33
			trait = impaler
		}
		modifier = {
			factor = 4
			is_devil_or_warband_worshiper_trigger = yes
		}
	}

	immediate = {
		random_child = {
			limit = {
				NOT = { age = 3 }
				is_ill = no
				NOT = { trait = inbred }
			}
			character_event = { id = daemonchild.2 }
		}
	}
}

# daemonchild.2 The marked child
# Hidden: marks the child (flag and hidden modifier demon_child_non_pagan, global
# flag demon_child_born), tells the parent (daemonchild.3) and starts the
# child's antics (daemonchild.19).
# Scope: ROOT is the child, FROM the parent (or the child itself from the
#   orgy birth event in birth_events.txt, then the mother hears). From
#   daemonchild.1.
character_event = {
	id = daemonchild.2
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		set_character_flag = demon_child_non_pagan
		set_global_flag = demon_child_born
		add_character_modifier = {
			name = demon_child_non_pagan
			hidden = yes
			days = -1
		}
		remove_trait = sickly
		if = {
			limit = { FROM = { character = ROOT } } # fired by the child's own birth event
			mother = { narrative_event = { id = daemonchild.3 } }
		}
		else = {
			FROM = { narrative_event = { id = daemonchild.3 } }
		}
		character_event = { id = daemonchild.19 days = 90 random = 90 } # Demon child tombola
	}
}
