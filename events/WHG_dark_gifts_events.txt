###############################################################################
# Dark gifts and the Eye of the Gods
# Namespace: darkgift (was "RS_offmapK", "RS_offmapN", "RS_offmapS",
# "RS_offmapT" and "RS_offmapU" in wh_offmap_<god>_complete.txt before the
# 2026 events refactor; first written by Childeri "Red Skald")
# Faction: Chaos
#
# Servants of the Dark Gods spend their favour (the gods' offmap currency) on
# gifts: a lesser daemon to serve at court, an artifact from the god's
# armoury, or a mortal champion sworn to the god. And every mortal who bears
# a god's mark lives under the Eye of the Gods. Every few years the patron
# weighs its servant and answers with wrath, scorn, a test, favour or glory.
#
# Sections:
#   1-4    a lesser daemon servant (Bloodletter, Plaguebearer, Daemonette,
#          Flamer)
#   10-14  a dark gift from the god's armoury (Khorne, Nurgle, Slaanesh,
#          Tzeentch, Undivided)
#   20-24  a mortal champion sent by the god (same order)
#   30-35  the Eye of the Gods judges a marked aspirant, Champion or Chosen
#
# Callers:
#   darkgift.1-4    warp_servant_<god> and warp_servant_<god>_undivided
#                   decisions (wh_chaos_decisions)
#   darkgift.10-14  warp_artifact_<god> and warp_artifact_<god>_undivided
#                   decisions; darkgift_artifact_effect (darkgift.35)
#   darkgift.20-24  warp_champion_<god> and the Undivided champion decision
#   darkgift.30     on_five_year_pulse random list darkgift_eye (wh_on_actions)
#   darkgift.31-35  darkgift.30
#
# Shared code:
#   common/scripted_triggers/wh_darkgift_triggers.txt (patron, judged, pleases)
#   common/scripted_effects/wh_darkgift_effects.txt (bond, favour, outcomes)
#   localisation/customizable_localisation/wh_darkgift_customizable_localisation.txt
#   voice_<god> modifiers (common/event_modifiers/wh_offmap_event_modifiers.txt)
#   mutations_table_<god> and chaos_spawn_effect (wh_mutation_scripted_effects)
###############################################################################

namespace = darkgift

###############################################################################
# 1-4 Lesser daemon servants
# The god sends one of its lesser daemons to serve the buyer at court. In
# every decision the decision's ROOT is the offmap holder; these events fire
# in the buyer's scope, so ROOT here is the buyer.
###############################################################################

# darkgift.1 A Bloodletter of Khorne
# A Bloodletter answers the call: a warrior and nothing else.
# Scope: ROOT is the buyer.
character_event = {
	id = darkgift.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		create_character = {
			age = 588
			dynasty = actually_culture
			religion = khorne
			culture = bloodletter_khorne
			female = no
			health = 20
			random_traits = no
			attributes = { martial = 10 diplomacy = 4 stewardship = 5 intrigue = 3 learning = 3 }
		}
		new_character = {
			darkgift_servant_effect = yes
			add_trait = khorne_bloodletter
			add_trait = khorne_demon
			add_trait = wroth
			random_list = {
				60 = { add_trait = tough_soldier }
				30 = { add_trait = skilled_tactician }
				10 = { add_trait = brilliant_strategist }
			}
			random_list = {
				# Butcher
				35 = {
					add_trait = impaler
					add_trait = brave
					random_list = {
						50 = { add_trait = cruel }
						50 = { add_trait = strong }
					}
				}
				# Duellist, who wants worthy skulls
				35 = {
					add_trait = duelist
					add_trait = honest
					random_list = {
						50 = { add_trait = proud }
						50 = { add_trait = tough }
					}
				}
				# Warlord
				30 = {
					add_trait = strategist
					add_trait = ambitious
					random_list = {
						50 = { add_trait = aggressive_leader }
						50 = { add_trait = unyielding_leader }
					}
				}
			}
		}
	}
}

# 30-35 The Eye of the Gods
# Every mortal who bears a god's mark, or the rank of Champion or Chosen,
# lives under the Eye of the Gods. On about every second five-year pulse the
# patron weighs its servant (darkgift.30) and answers:
#   31 wrath: cast down, maimed, taken into the Realm of Chaos, or spawned
#   32 scorn: lost prestige, piety, favour, or a wound; an offering can
#      buy the scorn off
#   33 a test: the temper reshaped, a mutation, a little prestige or piety,
#      or silence
#   34 favour: body, piety, prestige, life, the god's voice, the god's boon,
#      a way of life, or magic (never from Khorne)
#   35 glory: an aspirant is named Champion; a Champion or Chosen receives a
#      dark gift (darkgift.10-14)
# Piety, prestige and living the way the patron likes or hates move the
# odds (darkgift_pleases_patron_trigger, darkgift_offends_patron_trigger).
# Chosen stand under a harsher gaze. Promotion to Chosen and Daemon Prince
# stays with the decisions.
# Restored: the old RS_offmap<god>.4-6 judgements and their outcomes were
# never called, had broken weights and no text.
###############################################################################

# darkgift.30 The Eye of the Gods (hidden)
# Rolls the patron's answer.
# Scope: ROOT is the judged character. on_five_year_pulse list darkgift_eye.
character_event = {
	id = darkgift.30
	hide_window = yes

	is_triggered_only = yes
	capable_only = yes
	prisoner = no
	min_age = 16
	religion_group = chaos_gods_group

	trigger = {
		darkgift_judged_trigger = yes
		is_inaccessible_trigger = no
	}

	immediate = {
		random_list = {
			10 = { # wrath
				modifier = { factor = 0.5 darkgift_aspirant_trigger = yes }
				modifier = { factor = 1.5 is_chaos_chosen_trigger = yes }
				modifier = { factor = 1.5 darkgift_offends_patron_trigger = yes }
				modifier = { factor = 0.5 darkgift_pleases_patron_trigger = yes }
				modifier = { factor = 2 NOT = { piety = 0 } }
				modifier = { factor = 1.5 NOT = { piety = -500 } }
				modifier = { factor = 0.5 piety = 1000 }
				character_event = { id = darkgift.31 }
			}
			20 = { # scorn
				modifier = { factor = 1.5 darkgift_offends_patron_trigger = yes }
				modifier = { factor = 0.67 darkgift_pleases_patron_trigger = yes }
				modifier = { factor = 1.5 NOT = { piety = 0 } }
				modifier = { factor = 1.5 NOT = { prestige = 0 } }
				character_event = { id = darkgift.32 }
			}
			30 = { # a test
				modifier = { factor = 0.7 is_chaos_chosen_trigger = yes }
				character_event = { id = darkgift.33 }
			}
