###############################################################################
# Council of Elders events
# Namespace: dawielder
# Faction: Dwarfs
#
# The clan elders of the Dawi (dawi_elder_council): joining, AI recruits,
# the rank powers, and the life of a council. Kings, thanes and Longbeards
# sit together: the head of every clan speaks for it, and every Dawi of
# great age and beard has a seat by right.
#
# Lore: Dawi defer to the old. At 150 a Dawi becomes an Elder of the clan
# and sits on its Elder Council, which weighs any matter touching the clan
# and judges breaches of Dwarf law; past 200 he is a Great Beard, past 400 a
# Living Ancestor. The elders keep the clan's Book of Grudges and its Book
# of Debt, settle which Full Beard takes an apprentice, give nicknames,
# agree the date of the Day of Remembrance, fine those guilty of crimes
# against property, and can end a blood-feud when the guilty pay
# compensation. The king picks the judges of a tribunal, and the accused
# swear under oath to Grungni and Valaya. A hold's Elder Loremaster keeps
# the clan names.
# Sources: WFRP 4E Dwarf Player's Guide (Social Structure and Stages of Life
# p.29-32, Book of Debt and Honour Debts p.35, Dwarf law and feuds p.37,
# nicknames and clan names p.54, holy days p.138).
#
# Joining and AI recruits                dawielder.1 - dawielder.6
# Rank powers                            dawielder.10 - dawielder.21
# Life of the council                    dawielder.30 - dawielder.35
#
# Callers:
#   on_actions (wh_on_actions.txt): dawielder.1
#     on_character_ask_to_join_society, dawielder.5 five-year pulse, the
#     dawielder_council_life bi-yearly random list (dawielder.30 - .35).
#   decisions/WHG_dawi_elder_decisions.txt: dawielder.10, 12, 13, 14, 15,
#     16, 18, 20, 21.
# Scripted triggers: common/scripted_triggers/wh_dawi_elder_triggers.txt
# Event modifiers: common/event_modifiers/wh_dawi_elder_modifiers.txt
# Grudge effects: common/scripted_effects/wh_dawi_effects.txt
# Loc: localisation/00_WHG_dawi_elder_events.csv
###############################################################################

# dawielder.12 A name is given
# A Dawi may not choose his own nickname; the elders give it, and it must be
# true. An unflattering one sticks for life.
# Scope: ROOT is the elder. event_target:dawielder_named is the Dawi named.

	id = dawielder.12
	desc = EVTDESCdawielder12
	picture = GFX_evt_dwarf_king_seated
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # dawielder_bestow_nickname

	immediate = {
		if = {
			limit = { any_vassal = { dawielder_name_candidate_trigger = yes } }
			random_vassal = {
				limit = { dawielder_name_candidate_trigger = yes }
				save_event_target_as = dawielder_named
			}
		}
		else_if = {
			limit = { any_courtier = { dawielder_name_candidate_trigger = yes } }
			random_courtier = {
				limit = { dawielder_name_candidate_trigger = yes }
				save_event_target_as = dawielder_named
			}
		}
		else = {
			liege = {
				random_courtier = {
					limit = { dawielder_name_candidate_trigger = yes }
					save_event_target_as = dawielder_named
				}
			}
		}
	}

	option = { # A name of his deeds
		name = EVTOPTAdawielder12

			random_list = {
				20 = { give_nickname = nick_ironhelm }
				20 = { give_nickname = nick_hammerfist }

				20 = { give_nickname = nick_irongrip }
				20 = { give_nickname = nick_sharpedge }
			}
			opinion = { who = ROOT modifier = opinion_grateful years = 10 }
		}
		prestige = 25
		ai_chance = {
			factor = 10
			modifier = { factor = 0.1 event_target:dawielder_named = { is_rival = ROOT } }

	option = { # A name of his wisdom
		name = EVTOPTBdawielder12
		event_target:dawielder_named = {
			if = { limit = { trait = longbeard } give_nickname = nick_greybeard }

				random_list = {
					50 = { give_nickname = nick_strongwill }
					50 = { give_nickname = nick_oath_holder }
				}
			}

		}

		ai_chance = {
			factor = 10
			modifier = { factor = 0.1 event_target:dawielder_named = { is_rival = ROOT } }
		}

		name = EVTOPTCdawielder12

# dawielder.14 The Book of Debt
# The elders keep the clan's ledger of debts, and an ancestor's unpaid debt
# falls on his descendants until it is paid in full. Paying it also lifts
# the shame of a father who died with grudges open (dawi.34) one step.
# Scope: ROOT is the elder. event_target:dawielder_lord owns the debts.
character_event = {
	id = dawielder.14
	desc = EVTDESCdawielder14
	picture = GFX_evt_dwarf_grudge_book
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # dawielder_book_of_debt

	immediate = {
		if = { limit = { is_ruler = yes } save_event_target_as = dawielder_lord }
		else = { liege = { save_event_target_as = dawielder_lord } }
	}

	option = { # Call in what is owed to the clan
		name = EVTOPTAdawielder14
		event_target:dawielder_lord = {
			wealth = 75
			random_vassal = {
				limit = { trait = creature_dwarf }
				opinion = { who = PREV modifier = opinion_dislike years = 5 }
			}
		}
		ai_chance = {
			factor = 1
			modifier = { factor = 3 trait = greedy }
			modifier = { factor = 2 event_target:dawielder_lord = { NOT = { wealth = 50 } } }
		}
	}
	option = { # Pay what the clan owes, to the last penny
		name = EVTOPTBdawielder14
		trigger = { event_target:dawielder_lord = { wealth = 75 } }
		event_target:dawielder_lord = {
			wealth = -75
			prestige = 100
			add_character_modifier = { name = dawielder_ledger_clean years = 10 }
			if = {
				limit = { has_character_modifier = dawi_far_from_ancestors_3 }
				remove_character_modifier = dawi_far_from_ancestors_3
				add_character_modifier = { name = dawi_far_from_ancestors_2 duration = -1 }
			}
			else_if = {
				limit = { has_character_modifier = dawi_far_from_ancestors_2 }
				remove_character_modifier = dawi_far_from_ancestors_2
				add_character_modifier = { name = dawi_far_from_ancestors_1 duration = -1 }
			}
			else_if = {
				limit = { has_character_modifier = dawi_far_from_ancestors_1 }
				remove_character_modifier = dawi_far_from_ancestors_1
			}
		}
		ai_chance = {
			factor = 2
			modifier = {
				factor = 3
				event_target:dawielder_lord = { dawielder_far_from_ancestors_trigger = yes }
			}
			modifier = { factor = 1.5 trait = just }
		}
	}
	option = { # Check the tallies and leave the ledger as it stands
		name = EVTOPTCdawielder14
		change_society_currency = 50
		ai_chance = { factor = 1 }
	}
}

# dawielder.16 A feud between clans
# Two Dawi lords are at feud. When a Dawi of high standing steps in, the
# guilty side pays the injured one and the grudge is struck out. Only
# ordinary grudges; a Blood Grudge is not the council's to end.
# Scope: ROOT is the elder. event_target:dawielder_feud_holder holds the
# grudge, event_target:grudge_offender owes it (the name
# dawi_settle_grudge_effect expects).
character_event = {
	id = dawielder.16
	desc = EVTDESCdawielder16
	picture = GFX_evt_dwarf_grudge_book
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # dawielder_end_a_feud

	immediate = {
		random_playable_ruler = {
			limit = { dawielder_feud_holder_trigger = yes }
			save_event_target_as = dawielder_feud_holder
			random_playable_ruler = {
				limit = { dawielder_feud_offender_trigger = yes }
				save_event_target_as = grudge_offender
			}
		}
	}

	option = { # Weigh the wrong and set the price
		name = EVTOPTAdawielder16
		custom_tooltip = { text = dawielder_feud_price_tt }
		hidden_effect = {
			event_target:grudge_offender = {
				if = {
					limit = { wealth = 100 }
					wealth = -100
					event_target:dawielder_feud_holder = { wealth = 100 }
				}
				else = { prestige = -100 }
			}
		}
		event_target:dawielder_feud_holder = {
			dawi_settle_grudge_effect = yes
			opinion = { who = ROOT modifier = opinion_grateful years = 10 }
		}
		prestige = 75
		hidden_effect = {
			event_target:dawielder_feud_holder = { character_event = { id = dawielder.17 } }
			event_target:grudge_offender = { character_event = { id = dawielder.17 } }
		}
		ai_chance = { factor = 3 }
	}
	option = { # Sit them down over a barrel and let them drink it out
		name = EVTOPTBdawielder16
		random_list = {
			50 = {
				custom_tooltip = { text = dawielder_feud_drunk_tt }
				event_target:dawielder_feud_holder = { dawi_settle_grudge_effect = yes }
				prestige = 50
				hidden_effect = {
					event_target:dawielder_feud_holder = { character_event = { id = dawielder.19 } }
					event_target:grudge_offender = { character_event = { id = dawielder.19 } }
				}
			}
			50 = { custom_tooltip = { text = dawielder_feud_brawl_tt } prestige = 10 }
		}
		ai_chance = { factor = 1 modifier = { factor = 2 trait = gregarious } }
	}
	option = { # It is for their clans to end
		name = EVTOPTCdawielder16
		change_society_currency = 100
		ai_chance = { factor = 0.5 }
	}
}
