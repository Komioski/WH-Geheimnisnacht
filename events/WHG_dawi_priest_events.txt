###############################################################################
# Priesthood of the Ancestors events
# Namespace: dawipriest
# Faction: Dwarfs
#
# The priests of the Ancestor Gods (ancestor_priesthood): joining, AI
# recruits, the rank powers, and the life of a temple. A priest of rank 2
# takes a calling: Hearth Priest of Valaya (healing, hearth and guests),
# Forge Priest of Grungni and Smednir (the works of the ancestors) or Doom
# Priest of Grimnir (the warrior cults, and witness to the Slayers' oaths).
#
# Lore: Dawi priests are not ranked under one head; each hold's temple runs
# itself, and all its priests come from the tradition-minded Runesmith clans.
# The High Priests of Grungni and Valaya sit on the king's council. The faith
# is an oral one: tales of the ancestors told for hours on holy days, with a
# respectful gulp of ale between them. Gazul's priests bury the dead.
# Sources: WFRP 4E Dwarf Player's Guide (Doom, Forge and Hearth Priests
# p.64-67, the Priesthood and its temples p.134-140).
#
# Joining and AI recruits                dawipriest.1 - dawipriest.6
# Rank powers                            dawipriest.10 - dawipriest.22
# Life of the temple                     dawipriest.30 - dawipriest.33
#
# Callers:
#   on_actions (wh_on_actions.txt): dawipriest.1
#     on_character_ask_to_join_society, dawipriest.4
#     on_character_leave_society, dawipriest.5 five-year pulse, the
#     dawipriest_temple_life bi-yearly random list (dawipriest.30 - .33).
#   decisions/WHG_dawi_priest_decisions.txt: dawipriest.10, 12, 13, 15, 16,
#     17, 20, 22.
# Scripted triggers: common/scripted_triggers/wh_dawi_priest_triggers.txt
# Event modifiers: common/event_modifiers/wh_dawi_priest_modifiers.txt
# Loc: localisation/00_WHG_dawi_priest_events.csv
###############################################################################

namespace = dawipriest

###############################################################################

# dawipriest.1 Asking to join the priesthood
# Scope: ROOT is the applicant, FROM is the society.
character_event = {
	id = dawipriest.1
	hide_window = yes
	is_triggered_only = yes # on_character_ask_to_join_society

	trigger = {
		FROM = { leader = { society_member_of = ancestor_priesthood } }
	}

	immediate = {
		FROM = { leader = { character_event = { id = dawipriest.2 } } }
	}
}

# dawipriest.2 Welcome letter ping
# Scope: ROOT is the eldest High Priest, FROM is the applicant.
character_event = {
	id = dawipriest.2
	hide_window = yes
	is_triggered_only = yes # dawipriest.1

	immediate = {
		FROM = { letter_event = { id = dawipriest.3 } }
	}
}

# dawipriest.3 Welcome to the temple
# Scope: ROOT is the applicant, FROM is the eldest High Priest.
letter_event = {
	id = dawipriest.3
	desc = EVTDESCdawipriest3

	is_triggered_only = yes # dawipriest.2

	option = {
		name = EVTOPTAdawipriest3

	}
}

# dawipriest.4 Leaving the priesthood
# A priest who leaves gives up his calling.
# Scope: ROOT is the former priest, FROM is the society.
character_event = {
	id = dawipriest.4
	hide_window = yes
	is_triggered_only = yes # on_character_leave_society

	trigger = {
		dawipriest_has_calling_trigger = yes
	}

	immediate = {
		remove_character_modifier = dawipriest_hearth_calling
		remove_character_modifier = dawipriest_forge_calling
		remove_character_modifier = dawipriest_doom_calling
	}
}

# dawipriest.5 AI recruits
# Every five years a Dawi ruler's devout AI courtiers (and the ruler) may
# enter the temple.
# Scope: ROOT is a ruler.
character_event = {
	id = dawipriest.5
	hide_window = yes
	is_triggered_only = yes # on_five_year_pulse
	only_rulers = yes
	religion_group = dwarf_gods_group

	immediate = {
		any_courtier = {
			limit = { dawipriest_ai_recruit_trigger = yes }
			character_event = { id = dawipriest.6 }
		}
		if = {
			limit = { dawipriest_ai_recruit_trigger = yes }
			character_event = { id = dawipriest.6 }
		}
	}
}

# dawipriest.6 An AI recruit enters the temple (or not)
# Scope: ROOT is the candidate.
character_event = {
	id = dawipriest.6
	hide_window = yes
	is_triggered_only = yes # dawipriest.5

	trigger = {
		NOT = { is_society_rank_full = { society = ancestor_priesthood rank = 1 } }
	}

	immediate = {
		random_list = {
			90 = { }
			10 = {
				modifier = { factor = 3 trait = theologian }
				modifier = { factor = 3 is_priest = yes }
				modifier = { factor = 2 trait = zealous }
				modifier = { factor = 0.5 is_ruler = yes }
				join_society = ancestor_priesthood
			}
		}
	}
}

# dawipriest.12 Taking a calling
# Scope: ROOT is the priest.
character_event = {
	id = dawipriest.12
	desc = EVTDESCdawipriest12
	picture = GFX_evt_dwarf_patron_gods
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # dawipriest_take_a_calling

	option = { # A Hearth Priest of Valaya
		name = EVTOPTAdawipriest12
		add_character_modifier = { name = dawipriest_hearth_calling duration = -1 }
		if = {
			limit = { dawipriest_has_patron_trigger = no }
			add_character_modifier = { name = follower_of_valaya duration = -1 }
		}
		ai_chance = {
			factor = 10
			modifier = { factor = 3 has_character_modifier = follower_of_valaya }
			modifier = { factor = 2 is_female = yes }
			modifier = { factor = 1.5 trait = kind }
			modifier = { factor = 1.5 diplomacy = 12 }
		}
	}
	option = { # A Forge Priest of Grungni and Smednir
		name = EVTOPTBdawipriest12
		add_character_modifier = { name = dawipriest_forge_calling duration = -1 }
		if = {
			limit = { dawipriest_has_patron_trigger = no }
			add_character_modifier = { name = follower_of_grungni duration = -1 }
		}
		ai_chance = {
			factor = 10
			modifier = { factor = 3 has_character_modifier = follower_of_grungni }
			modifier = { factor = 3 has_character_modifier = follower_of_thungni }
			modifier = { factor = 1.5 trait = diligent }
			modifier = { factor = 1.5 stewardship = 12 }
		}
	}
	option = { # A Doom Priest of Grimnir
		name = EVTOPTCdawipriest12
		add_character_modifier = { name = dawipriest_doom_calling duration = -1 }
		if = {
			limit = { dawipriest_has_patron_trigger = no }
			add_character_modifier = { name = follower_of_grimnir duration = -1 }
		}
		ai_chance = {
			factor = 10
			modifier = { factor = 3 has_character_modifier = follower_of_grimnir }
			modifier = { factor = 1.5 trait = wroth }
			modifier = { factor = 1.5 trait = brave }
			modifier = { factor = 1.5 martial = 12 }
		}
	}
}

# dawipriest.13 A holy day
# Each calling keeps its own: Valaya's Mitterfruhl, Grungni's First Quaff, a
# vigil at Grimnir's war shrine. A priest without a calling keeps the Day of
# Remembrance.
# Scope: ROOT is the priest.
character_event = {
	id = dawipriest.13
	desc = {
		trigger = { has_character_modifier = dawipriest_hearth_calling }
		text = EVTDESCdawipriest13a
	}
	desc = {
		trigger = { has_character_modifier = dawipriest_forge_calling }
		text = EVTDESCdawipriest13b
	}
	desc = {
		trigger = { has_character_modifier = dawipriest_doom_calling }
		text = EVTDESCdawipriest13c
	}
	desc = { trigger = { dawipriest_has_calling_trigger = no } text = EVTDESCdawipriest13d }
	picture = GFX_evt_feast
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # dawipriest_holy_day

	option = { # Lead the rites
		name = EVTOPTAdawipriest13
		piety = 75
		if = {
			limit = { is_ruler = yes }
			add_character_modifier = { name = dawipriest_holy_day_kept years = 5 }
		}
		else_if = {
			limit = { liege = { religion_group = dwarf_gods_group } }
			liege = {
				add_character_modifier = { name = dawipriest_holy_day_kept years = 5 }
				opinion = { who = ROOT modifier = opinion_grateful years = 5 }
			}
		}
		if = {
			limit = { has_character_modifier = dawipriest_forge_calling }
			change_society_currency = 50
		}
		else_if = { limit = { has_character_modifier = dawipriest_hearth_calling } health = 0.5 }
		else_if = { limit = { has_character_modifier = dawipriest_doom_calling } prestige = 75 }
	}
}

# dawipriest.15 Before the king's council
# Scope: ROOT is the priest. The counsel goes to ROOT if he rules, or else
# to his liege.
character_event = {
	id = dawipriest.15
	desc = EVTDESCdawipriest15
	picture = GFX_evt_dwarf_king_seated
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # dawipriest_counsel_the_king

	immediate = {
		if = { limit = { is_ruler = yes } save_event_target_as = dawipriest_counselled }
		else = { liege = { save_event_target_as = dawipriest_counselled } }
	}

	option = { # Speak for peace among the clans
		name = EVTOPTAdawipriest15
		event_target:dawipriest_counselled = {
			add_character_modifier = { name = dawipriest_counsel_peace years = 5 }
		}
		if = {
			limit = { NOT = { character = event_target:dawipriest_counselled } }
			event_target:dawipriest_counselled = {
				opinion = { who = ROOT modifier = opinion_grateful years = 5 }
			}
		}
		ai_chance = { factor = 1 modifier = { factor = 2 diplomacy = 12 } }
	}
	option = { # Speak the old law, as the ancestors wrote it
		name = EVTOPTBdawipriest15
		event_target:dawipriest_counselled = {
			add_character_modifier = { name = dawipriest_counsel_law years = 5 }
		}
		piety = 25
		ai_chance = { factor = 1 modifier = { factor = 2 trait = just } }
	}
}

# dawipriest.17 Tending the wounded
# The priest heals himself or a wounded or sick member of the court; a
# Hearth Priest may heal two.
# Scope: ROOT is the priest.
character_event = {
	id = dawipriest.17
	desc = EVTDESCdawipriest17
	picture = GFX_evt_dwarf_temple
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # dawipriest_tend_the_wounded

	option = {
		name = EVTOPTAdawipriest17
		if = {
			limit = { OR = { trait = wounded trait = ill } }
			remove_trait = wounded
			remove_trait = ill
		}
		else = {
			random_courtier = {
				limit = { OR = { trait = wounded trait = ill } }
				remove_trait = wounded
				remove_trait = ill
				opinion = { who = ROOT modifier = opinion_grateful years = 10 }
			}
		}
		if = {
			limit = { has_character_modifier = dawipriest_hearth_calling }
			random_courtier = {
				limit = { OR = { trait = wounded trait = ill } }
				remove_trait = wounded
				remove_trait = ill
				opinion = { who = ROOT modifier = opinion_grateful years = 10 }
			}
		}
		piety = 25
	}
}

# dawipriest.20 The master rune is finished
# Three years' work: the master rune of the priest's god, once in a
# lifetime, and only if the priest does not already hold it.
# Scope: ROOT is the High Priest.
character_event = {
	id = dawipriest.20
	desc = EVTDESCdawipriest20
	picture = GFX_evt_dwarf_patron_gods
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # dawipriest_master_rune

	option = {
		name = EVTOPTAdawipriest20
		remove_character_modifier = dawipriest_inscribing
		if = {
			limit = {
				has_character_modifier = dawipriest_hearth_calling
				NOT = { has_artifact = master_rune_of_valaya }
			}
			add_artifact = master_rune_of_valaya
		}
		else_if = {
			limit = {
				has_character_modifier = dawipriest_forge_calling
				NOT = { has_artifact = master_rune_of_grungi }
			}
			add_artifact = master_rune_of_grungi
		}
		else_if = {
			limit = {
				has_character_modifier = dawipriest_doom_calling
				NOT = { has_artifact = master_rune_of_grimnir }
			}
			add_artifact = master_rune_of_grimnir
		}
		else = { piety = 250 prestige = 250 }
	}
}

# dawipriest.33 An ancestor's work brought to the temple
# A damaged piece from a fallen hold is brought for restoring. A Forge
# Priest can mend it; once in a lifetime a mended piece may stay with him.
# Scope: ROOT is the priest.
character_event = {
	id = dawipriest.33
	desc = EVTDESCdawipriest33
	picture = GFX_evt_dwarf_seated_book
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_bi_yearly_pulse, dawipriest_temple_life
	religion_group = dwarf_gods_group

	trigger = {
		dawipriest_pulse_trigger = yes
	}

	option = { # Restore it, stroke by stroke
		name = EVTOPTAdawipriest33
		trigger = { has_character_modifier = dawipriest_forge_calling }
		piety = 50
		change_society_currency = 50
		if = {
			limit = {
				NOT = { has_character_flag = dawipriest_relic_kept }
				NOT = { any_artifact = { has_artifact_flag = armor } }
			}
			random = {
				chance = 10
				set_character_flag = dawipriest_relic_kept
				add_artifact = dawi_artifact_gromril_armour
				custom_tooltip = { text = dawipriest_relic_kept_tt }
			}

	}
	option = { # Set it in the Deep of Renown for all to honour
		name = EVTOPTBdawipriest33
		piety = 25
		prestige = 25
	}
