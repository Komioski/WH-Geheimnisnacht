###############################################################################
# Death events: the end of days during the great plagues, and the quest for
# eternal life
#
# Part 1: The end of days (namespace eod)
# Faction: Generic
# Serves: any independent player ruler whose realm the Nurgle plagues
# (Nurgle's Rot, the Creeping Buboes) or the Red Pox have reached.
#
# Also in this file (each part keeps its own header and namespace):
#   Part 2: The quest for eternal life (namespace eternal)
#
# When the great plagues reach the realm, one of three chains runs (once
# per two years):
#   eod.1            Starter (on_yearly_pulse random list)
#   eod.2-21         Doomsayers, Morrslieb swallows the sun
#   eod.30-47        The Ashen Rider, a false prophet claiming to be the
#                    god reborn, who may raise a holy revolt
#   eod.50-62        The restless dead: rumours that the plague dead walk
#   eod.70           Clears the end_of_days flag
#
# Was events/rip_eod_events.txt. eod.1-21 are old RIP.14000-14020 (minus
# 13999), the Ashen Rider RIP.15099-15116 and the restless dead
# RIP.22000-22020, in order. eod.1 was gated on the vanilla bubonic plague,
# which the mod does not have.
###############################################################################

namespace = eod

# eod.1 End of days starts
# Hidden. When the Nurgle plagues or the Red Pox reach the realm, picks one of
# three chains: doomsayers (eod.2), the Ashen Rider (eod.30) or the restless
# dead (eod.50). Clears after two years (eod.70).
# Scope: ROOT is an independent player ruler. Fired from the on_yearly_pulse
# random list (was gated on the bubonic plague, which the mod does not have).
character_event = {
	id = eod.1
	hide_window = yes
	is_triggered_only = yes

	only_playable = yes
	min_age = 16
	capable_only = yes
	ai = no
	prisoner = no
	only_independent = yes

	trigger = {
		any_realm_province = {
			OR = { disease = nurgles_rot disease = creeping_buboes disease = typhus }
		}
		host = { character = ROOT }

		NOT = { has_character_flag = end_of_days }
	}

	immediate = {
		random_list = {
			20 = {
				narrative_event = { id = eod.2 } # Doomsayers
			}
			20 = {
				character_event = { id = eod.30 } # The Ashen Rider
			}
			20 = {
				narrative_event = { id = eod.50 } # The restless dead
			}
		}

		set_character_flag = end_of_days

		character_event = { id = eod.70 days = 730 }
	}
}

# eod.2 Doomsayers
# Doomsayers draw crowds in the capital. The ruler blames a rival, despairs,
# or lets it run its course. Greedy councillors may offer a scheme (eod.3).
# Scope: ROOT is the ruler. Fired by eod.1.
narrative_event = {
	id = eod.2
	picture = GFX_evt_bring_out_your_dead
	title = EVTNAMEeod2
	desc = EVTDESCeod2
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes

	immediate = {
		capital_scope = { save_event_target_as = capital }
	}

	option = {
		name = EVTOPTeod2_A
		hidden_tooltip = {
			random_list = {
				100 = {
					modifier = {
						factor = 0
						NOR = {
							job_chancellor = { OR = { trait = greedy trait = deceitful } }
							job_marshal = { OR = { trait = greedy trait = deceitful } }
							job_treasurer = { OR = { trait = greedy trait = deceitful } }
							job_spymaster = { OR = { trait = greedy trait = deceitful } }
							job_spiritual = { OR = { trait = greedy trait = deceitful } }
						}
					}
					job_chancellor = {
						if = {
							limit = { OR = { trait = greedy trait = deceitful } }
							save_event_target_as = greedy_councillor
							ROOT = { set_character_flag = greedy_chancellor }
							character_event = { id = eod.4 }
							break = yes
						}
					}
					job_marshal = {
						if = {
							limit = { OR = { trait = greedy trait = deceitful } }
							save_event_target_as = greedy_councillor
							ROOT = { set_character_flag = greedy_marshal }
							character_event = { id = eod.4 }
							break = yes
						}
					}
					job_treasurer = {
						if = {
							limit = { OR = { trait = greedy trait = deceitful } }
							save_event_target_as = greedy_councillor
							ROOT = { set_character_flag = greedy_treasurer }
							character_event = { id = eod.4 }
							break = yes
						}
					}
					job_spymaster = {
						if = {
							limit = { OR = { trait = greedy trait = deceitful } }
							save_event_target_as = greedy_councillor
							ROOT = { set_character_flag = greedy_spymaster }
							character_event = { id = eod.4 }
							break = yes
						}
					}
					job_spiritual = {
						if = {
							limit = { OR = { trait = greedy trait = deceitful } }
							save_event_target_as = greedy_councillor
							ROOT = { set_character_flag = greedy_spiritual }
							character_event = { id = eod.4 }
						}
					}
				}
				#Blame a rival
				100 = {
					modifier = { factor = 0 NOT = { any_rival = { at_location = ROOT } } }
					random_rival = {
						limit = { at_location = ROOT }
						character_event = { id = eod.6 }
					}
				}
				#Become depressed
				25 = {
					modifier = {
						factor = 0
						NOR = {
							trait = stressed
							trait = paranoid
							trait = craven
							is_weak_trigger = yes
						}
					}
					character_event = { id = eod.5 }
				}
				#nothing special happens
				10 = { character_event = { id = eod.11 days = 20 random = 20 } }
			}
		}
	}
}

	id = eod.32
	desc = EVTDESCeod32
	picture = GFX_evt_end_of_days
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes
	hide_from = yes

	immediate = {
		if = {
			limit = { any_friend = { is_adult = yes } }

			random_friend = {
				limit = { is_adult = yes }
				save_event_target_as = EoD_worried_one
				set_character_flag = worried_friend
			}
		}

		if = {
			limit = {
				NOT = { any_friend = { is_adult = yes } }
				any_playable_ruler = { ai = yes is_adult = yes NOT = { is_rival = ROOT } }
			}

			random_independent_ruler = {
				limit = {
					ai = yes
					is_adult = yes
					is_landed = yes
					NOT = { is_rival = ROOT }
					NOT = { distance = { value = 200 who = ROOT } }
				}
				save_event_target_as = EoD_worried_one
				set_character_flag = worried_ruler
			}
		}
	}

	#Make fun of horses
	option = {
		name = EVTOPTeod32_A

		add_character_modifier = { name = gave_funny_speech duration = 200 }

		hidden_tooltip = { event_target:EoD_worried_one = { character_event = { id = eod.33 } } }
	}

	#Give a speech of warning
	option = {
		name = EVTOPTeod32_B

		add_character_modifier = { name = gave_pious_speech duration = 200 }

		hidden_tooltip = { event_target:EoD_worried_one = { character_event = { id = eod.33 } } }
	}

	#Give a scary speech
	option = {
		name = EVTOPTeod32_C

		add_character_modifier = { name = gave_scary_speech duration = 200 }

		hidden_tooltip = {
			event_target:EoD_worried_one = {
				clr_character_flag = worried_friend
				clr_character_flag = worried_ruler
				clear_event_target = EoD_worried_one
			}

			if = {
				limit = { any_rival = { is_adult = yes } }

				random_rival = {
					limit = { is_adult = yes }
					save_event_target_as = EoD_worried_one
					set_character_flag = worried_rival
				}
			}

			if = {
				limit = {
					NOT = { any_rival = { is_adult = yes } }
					any_playable_ruler = { ai = yes is_adult = yes }
				}

				random_independent_ruler = {
					limit = {
						ai = yes
						is_adult = yes
						is_landed = yes
						NOT = { distance = { value = 200 where = ROOT } }
					}
					save_event_target_as = EoD_worried_one
					set_character_flag = worried_ruler
				}
			}
			event_target:EoD_worried_one = { character_event = { id = eod.33 } }
		}
	}

# eod.36 The Ashen Rider unmasked
# The prophet is brought before the ruler: bribe, imprison or execute them.
# Scope: ROOT is the ruler. Fired by eod.35.
character_event = {
	id = eod.36
	desc = EVTDESCeod36
	picture = GFX_evt_shadowy_cabal
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	immediate = {
		random_demesne_title = {
			limit = { tier = COUNT can_be_given_away = yes location = { is_capital = no } }
			save_event_target_as = EoD_some_province
		}
	}

	#BRIBE prophet
	option = {
		name = EVTOPTeod36_A
		custom_tooltip = { text = EVTOPTeod36_A_TOOLTIP }

		if = {
			limit = { NOT = { trait = cynical } }
			random = {
				chance = 25
				add_trait = cynical
				hidden_tooltip = { character_event = { id = 38258 } } #Notify Cynical
			}
		}

		hidden_tooltip = {
			any_realm_province = {
				limit = { owner = { character = ROOT } }
				remove_province_modifier = religious_tension
			}
			event_target:EoD_doomsday_prophet = {
				remove_opinion = { modifier = opinion_evil_tyrant who = ROOT }
			}
			random_list = {
				33 = {
					#prophet abdicates
					letter_event = { id = eod.37 days = 50 }
				}
				33 = {
					#prophet kill themselves
					character_event = { id = eod.45 days = 30 }
				}
				33 = {
					#prophet is readjusted to society
					character_event = { id = eod.44 days = 50 }
				}
			}
		}

		event_target:EoD_doomsday_prophet = {
			grant_title = event_target:EoD_some_province
			set_defacto_liege = ROOT
			disband_event_forces = EoD_uprising_troops
		}
	}

	#IMPRISON prophet
	option = {
		name = EVTOPTeod36_B
		custom_tooltip = { text = EVTOPTeod36_B_TOOLTIP }
		Prestige = 50

		hidden_tooltip = {
			event_target:EoD_doomsday_prophet = {
				imprison = ROOT
				disband_event_forces = EoD_uprising_troops
			}
			random_list = {
				#they escape from prison
				50 = { character_event = { id = eod.42 days = 20 } }
				#they kill themselves
				50 = {
					set_character_flag = imprisoned_white_rider
					character_event = { id = eod.45 days = 20 }
				}
			}
			any_realm_province = {
				limit = { owner = { character = ROOT } }
				remove_province_modifier = religious_tension
			}
		}
	}

	#EXECUTE prophet
	option = {
		name = EVTOPTeod36_C
		custom_tooltip = { text = EVTOPTeod36_C_TOOLTIP }
		Piety = 50

		hidden_tooltip = {
			random_list = {
				#they die
				40 = {
					event_target:EoD_doomsday_prophet = {
						death = { death_reason = death_execution_burning KILLER = FROM }
					}
				}

				#they escape execution
				60 = {
					#botched execution
					character_event = { id = eod.40 }

					event_target:EoD_doomsday_prophet = {
						give_nickname = nick_the_survivor

						add_character_modifier = { name = escaped_execution duration = -1 }
						leave_court_effect = yes
					}
				}
			}
			any_realm_province = {
				limit = { owner = { character = ROOT } }
				remove_province_modifier = religious_tension

				add_province_modifier = { name = "ordered_our_prophet_executed" duration = 365 }
			}
			disband_event_forces = EoD_uprising_troops
		}
	}
