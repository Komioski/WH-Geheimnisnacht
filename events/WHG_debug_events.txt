###############################################################################
# Debug and testing events: console only, never fired by the game
# Namespace: debug (was HH_Debug in HH_Testing_Events.txt; debug.N is the
#   old HH_Debug.N)
# Faction: Generic
# Serves: modders testing magic, doom, Chaos favour, colleges and cleanup
#
# Run from the console with: event debug.N   (ROOT is the player's character)
#
# debug.1   make ROOT a mage with a random lore (sets up the magic power
#           modifiers and levelling; use this, not the console trait)
# debug.2   add a massive amount of doom
# debug.3   remove a massive amount of doom
# debug.4   hand every Norscan king and count title to the Tsar of Kislev
#           (war and succession tests)
# debug.5   add 50000 favour with each of the four Chaos gods
# debug.7   give ROOT's faith heir designation and women as temple holders
# debug.8   kill every idle old lowborn with no ties to a player
#           (save-size and lag tests)
# debug.9   found the Imperial Colleges of Magic in ROOT's capital
# debug.10  found a new college of magic in ROOT's capital
# debug.11  kill every courtier of ROOT who is not kin, spouse, friend,
#           lover or ruler
#
# Scripted effects: random_lore_mage_effect, add/reduce_doom_progress_massive_
#   effect, college_found_imperial_effect, college_found_dynamic_effect
#   (wh_college_effects.txt)
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = debug

# debug.1 Make ROOT a mage
# Battle magic and levelling need the modifiers random_lore_mage_effect sets
# up; races that cannot use magic may give odd results.
# Scope: ROOT is the player's character.
character_event = {
	id = debug.1
	hide_window = yes
	is_triggered_only = yes # console

	immediate = {
		random_lore_mage_effect = yes
	}
}

# debug.2 Add doom
# Scope: ROOT is the player's character.
character_event = {
	id = debug.2
	hide_window = yes
	is_triggered_only = yes # console

	immediate = {
		add_doom_progress_massive_effect = yes
	}
}

# debug.3 Remove doom
# Scope: ROOT is the player's character.
character_event = {
	id = debug.3
	hide_window = yes
	is_triggered_only = yes # console

	immediate = {
		reduce_doom_progress_massive_effect = yes
	}
}

# debug.4 Norsca to Kislev
# Every king and count title in Norsca goes to the holder of e_kislev.
# Scope: ROOT is the player's character.
character_event = {
	id = debug.4
	hide_window = yes
	is_triggered_only = yes # console

	trigger = {
		e_kislev = { has_holder = yes }
	}

	immediate = {
		e_kislev = { holder_scope = { save_event_target_as = kislev } }
		any_title = {
			limit = { OR = { tier = KING tier = COUNT } region = world_norsca }
			grant_title = event_target:kislev
		}
	}
}

# debug.5 Favour with all four Chaos gods
# Scope: ROOT is the player's character.
character_event = {
	id = debug.5
	hide_window = yes
	is_triggered_only = yes # console

	immediate = {
		add_offmap_currency = { offmap = khorne_warp value = 50000 }
		add_offmap_currency = { offmap = nurgle_warp value = 50000 }
		add_offmap_currency = { offmap = tzeentch_warp value = 50000 }
		add_offmap_currency = { offmap = slaanesh_warp value = 50000 }
	}
}

# debug.7 Faith flags
# ROOT's faith designates heirs and lets women hold temples.
# Scope: ROOT is the player's character.
character_event = {
	id = debug.7
	hide_window = yes
	is_triggered_only = yes # console

	immediate = {
		religion_scope = { set_heir_designation = yes set_female_temple_holders = yes }
	}
}

# debug.8 Clear idle old lowborn
# Kills every lowborn of 50 or more with no children, no title, no job, no
# society and no tie to a player. For save-size and lag tests.
# Scope: ROOT is the player's character.
character_event = {
	id = debug.8
	hide_window = yes
	is_triggered_only = yes # console

	immediate = {
		any_character = {
			limit = {
				is_lowborn = yes
				age = 50
				has_children = no
				is_in_society = no
				is_chaos_champion_chosen_or_prince_trigger = no
				NOR = {
					is_married = yes
					is_landed = yes
					has_job_title = yes
					has_minor_title = yes
					AND = { NOT = { dynasty = none } num_of_dynasty_members = 2 }
					any_lover = { ai = no }
					any_close_relative = { ai = no }
					any_friend = { ai = no }
					any_rival = { ai = no }
					any_consort = { ai = no }
					num_of_claims = 1
					has_character_flag = eternal_life_mystic
					trait = horse
					has_character_modifier = in_seclusion
					has_character_modifier = elder_long_life_modifier
				}
			}
			death = { death_reason = death_murder_unknown }
		}
	}
}

# debug.11 Clear the court
# Kills every courtier who is not ROOT's kin, spouse, consort, friend, lover,
# parent or a ruler.
# Scope: ROOT is the player's character.
character_event = {
	id = debug.11
	hide_window = yes
	is_triggered_only = yes # console

	immediate = {
		any_courtier = {
			limit = {
				NOR = {
					dynasty = ROOT
					is_married = ROOT
					is_consort = ROOT
					is_friend = ROOT
					is_lover = ROOT
					is_ruler = yes
					is_mother = ROOT
					is_father = ROOT
				}
			}
			death = { death_reason = death_murder_unknown }
		}
	}
}
