# Decadence
# Namespace: decadence (was 91000 - 91379 in vanilla decadence_events.txt)
# Faction: Araby & Southlands
# Serves: the faiths that use decadence: Ormazd and the Kitabids of Araby,
#   the Old Ones of the Lizardmen and the Amazon sisterhoods
#
# A dynasty grows soft. Its members fall into decadence, its ruler's vices
# and virtues push the dynasty's decadence up or down, and once it runs too
# high a hard-living clan of the dynasty's own people rides against it and
# its vassals may break away. A dynasty head can try to talk, argue, scare
# or call in a favour to make a decadent kinsman reform, and may declare
# takfir on one who will not.
#
# The invasion                         decadence.1
# Vices and virtues                    decadence.10 - decadence.45
# After the invasion                   decadence.50 - decadence.51
# A member grows decadent              decadence.60
# Reform or takfir                     decadence.100 - decadence.129
#
# Fired from: on_yearly_pulse in wh_on_actions.txt (lists decadence_invasion
#   and decadence_fall, which replace MTTH 360 and 70 months); the decision
#   convince_to_straighten_up in dynasty_decisions.txt (decadence.100); the
#   decadence_invasion CB in 00_cb_types.txt (decadence.51). The vice and
#   virtue events stay MTTH behind ai = no.
# Saved: character flags decadence_<trait> (once per life),
#   straightened_up, negotiated_to_drop_decadence; event target
#   takfir_target.
###############################################################################

# decadence.60 A dynasty member grows decadent
# A man (or Amazon) of a decadence faith in a dynasty of three or more,
# below a king, falls into decadence: he gains the Decadent trait and player
# rulers of his dynasty and faith group are told. Vices, idleness and a vast
# realm make it likelier; piety, learning and diligence less likely.
# Scope: ROOT is the dynasty member.
character_event = {
	id = decadence.60
	desc = EVTDESCdecadence60
	picture = GFX_evt_sultan

	is_triggered_only = yes # on_yearly_pulse, list decadence_fall

	min_age = 16
	only_capable = yes
	prisoner = no

	trigger = {
		# Men, and the women of the Amazon sisterhoods
		OR = {
			is_woman_trigger = no
			religion = amazon
		}
		num_of_dynasty_members = 3 # Does not trigger for tiny dynasties
		uses_decadence = yes
		holy_order = no
		is_offmap_ruler = no
		lower_tier_than = KING
		NOR = {
			has_character_flag = invasion_cathay_claimant
			trait = decadent
			trait = zealous
			trait = mujahid
			trait = adventurer
			has_character_flag = straightened_up
		}
		biggest_realm_size_relative = {
			NOT = { character = PREV }
			realm_size = 5
			same_realm = ROOT
		}
	}

	weight_multiplier = {
		factor = 1

		modifier = {
			factor = 2
			trait = indulgent_wastrel
