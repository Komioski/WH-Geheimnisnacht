# alliance.10 A spouse's visit
# Yearly pulse. The spouse of a friendly-faith neighbour visits; the ruler
# may propose a non-aggression pact.
# Scope: ROOT is the ruler.
character_event = {
	id = alliance.10
	picture = GFX_evt_feast
	border = GFX_event_normal_frame_diplomacy
	desc = EVTDESCalliance10

	only_rulers = yes
	only_capable = yes
	prisoner = no
	only_independent = yes

	is_triggered_only = yes

	trigger = {
		mercenary = no
		primary_title = { is_landless_type_title = no }
		controls_religion = no
		holy_order = no
		NOT = { is_inaccessible_trigger = yes }
		NOR = {
			trait = creature_orc
			trait = creature_goblin
			trait = creature_hobgoblin
			trait = creature_beastman
		}
		spouse = { is_no_females_race = no }
		any_independent_ruler = { alliance_spouse_visit_target_trigger = yes }
	}

	immediate = {
		ROOT = { capital_scope = { save_event_target_as = capital } }
		random_independent_ruler = {
			limit = { alliance_spouse_visit_target_trigger = yes }
			spouse = { save_event_target_as = target_spouse }
			save_event_target_as = target_ruler
		}
	}

	option = {
		name = EVTOPTAalliance10
		ai_chance = { factor = 90 }
		show_portrait = event_target:target_ruler
		custom_tooltip = {
			text = EVTOPTAalliance10_tooltip
			hidden_tooltip = {
				event_target:target_ruler = { letter_event = { id = alliance.11 days = 3 } }
			}
		}
	}

	option = {
		name = EVTOPTBalliance10
		ai_chance = { factor = 10 }
	}
}

# alliance.20 A boorish guest
# Yearly pulse. A loud petty noble claiming to come from an ally or pact
# partner insults the court.
# Scope: ROOT is the ruler.
character_event = {
	id = alliance.20
	picture = GFX_evt_council
	border = GFX_event_normal_frame_diplomacy
	desc = EVTDESCalliance20

	only_rulers = yes
	only_capable = yes
	prisoner = no
	only_independent = yes

	is_triggered_only = yes

	trigger = {
		mercenary = no
		is_tributary = no
		NOT = { is_inaccessible_trigger = yes }
		OR = {
			any_non_aggression_pact_character = {
				mercenary = no
				is_tributary = no
				independent = yes
				is_within_diplo_range = ROOT
			}
			any_allied_character = {
				mercenary = no
				is_tributary = no
				independent = yes
				is_within_diplo_range = ROOT
			}
		}
	}

	immediate = {
		ROOT = { capital_scope = { save_event_target_as = capital } }
		if = {
			limit = {
				any_allied_character = {
					mercenary = no
					is_tributary = no
					is_within_diplo_range = ROOT
					independent = yes
				}
			}
			random_allied_character = {
				limit = {
					mercenary = no
					is_within_diplo_range = ROOT
					independent = yes
					is_tributary = no
				}
				save_event_target_as = target_ruler
				capital_scope = { save_event_target_as = target_ruler_capital }
			}
		}
		else_if = {
			limit = {
				any_non_aggression_pact_character = {
					mercenary = no
					is_tributary = no
					is_within_diplo_range = ROOT
					independent = yes
				}
			}
			random_non_aggression_pact_character = {
				limit = {
					mercenary = no
					is_within_diplo_range = ROOT
					independent = yes
					is_tributary = no
				}
				save_event_target_as = target_ruler
				capital_scope = { save_event_target_as = target_ruler_capital }
			}
		}
	}
	#write a formal complaint to ruler
	option = {
		name = EVTOPTAalliance20
		ai_chance = { factor = 90 }
		prestige = 50
		custom_tooltip = {
			text = EVTOPTAalliance20_tooltip
			hidden_tooltip = {
				event_target:target_ruler = { letter_event = { id = alliance.21 days = 3 } }
			}
		}
	}
	#ignore your courtiers complaints
	option = {
		name = EVTOPTBalliance20
		ai_chance = { factor = 10 }
		custom_tooltip = {
			text = EVTOPTBalliance20_tooltip
			hidden_tooltip = {
				any_courtier = {
					opinion = { modifier = opinion_disappointed who = ROOT months = 24 }
				}
			}
		}
	}
	#take care of him
	option = {
		trigger = { intrigue = 15 }
		name = EVTOPTCalliance20
		custom_tooltip = { text = EVTOPTCalliance20_tooltip }
		tooltip_info = intrigue
	}
}

# alliance.100 An idle ally
# Yearly pulse. An ally has not fought in four years of war
# (alliance_idle_ally_trigger): break the alliance or forgive.
# Scope: ROOT is the ruler.
character_event = {
	id = alliance.100
	desc = EVTDESCalliance100
	picture = GFX_evt_large_army
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	only_playable = yes
	only_capable = yes
	prisoner = no
	war = yes

	trigger = {
		OR = { is_primary_war_attacker = yes is_primary_war_defender = yes }
		any_allied_character = { alliance_idle_ally_trigger = yes }
	}

	immediate = {
		random_allied_character = {
			limit = { alliance_idle_ally_trigger = yes }
			save_event_target_as = ally_not_active_in_war
		}
	}

	option = {
		name = EVTOPTAalliance100 # Send angry letter and break alliance
		event_target:ally_not_active_in_war = {
			reverse_opinion = { who = ROOT modifier = ally_inactive_in_war years = 5 }
			ROOT = { if = { limit = { is_allied_with = PREV } break_alliance = PREV } }
			custom_tooltip = { text = alliance100_letter_sent }
			hidden_tooltip = { letter_event = { id = alliance.101 } }
		}
	}
	option = {
		name = EVTOPTBalliance100 # Forgive and forget
		event_target:ally_not_active_in_war = {
			reverse_opinion = { who = ROOT modifier = opinion_disappointed years = 3 }
		}
	}
}
