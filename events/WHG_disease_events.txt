###############################################################################
# Disease events: who is to blame for an epidemic, the dancing plague, and
# the Great Plague news
#
# Part 1: Blame for the epidemic (namespace blame)
# Faction: Generic
# Serves: any ruler of a settled people whose capital the sickness has
# emptied; not greenskins, Skaven, ogres, Lizardmen or the dead.
#
# Also in this file (each part keeps its own header and namespace):
#   Part 2: The dancing plague (namespace dancing)
#   Part 3: The Great Plague (namespace plague)
#
# When an epidemic has emptied a ruler's capital, the people look for someone
# to blame. One of four chains runs, once in a ruler's life:
#   blame.1          Starter (yearly random list blame_epidemic)
#   blame.10-20      The cats
#   blame.30-35      The Strigany
#   blame.40-58      A witch hunt, led by a court witch hunter, a witch
#                    hunter of the court or the court chaplain
#   blame.60-68      The impious ruler (restored; vanilla's was Hindu only
#                    and commented out)
#
# Was events/rip_epidemic_events.txt (RIP.13000-13508). blame.N for the
# cats is old RIP.130NN + 9, the Strigany are RIP.13400-13405, the witch
# hunt RIP.13198-13235 and the impious ruler RIP.13500-13508, in order.
###############################################################################

namespace = blame

# blame.1 The people look for someone to blame
# A ruler whose capital the epidemic has depopulated faces one of four blame
# chains: the cats, the Strigany (only with Strigany in the realm), a witch
# hunt (needs a chaplain or a witch hunter at court) or the ruler's own
# impiety (temple faiths with no temple in the capital). Once in a life.
# Scope: ROOT is the ruler. Fired from on_yearly_pulse, random list
# blame_epidemic.
character_event = {
	id = blame.1
	hide_window = yes

	is_triggered_only = yes # on_yearly_pulse (blame_epidemic)

	only_playable = yes
	only_independent = yes

	trigger = {
		host = { character = ROOT }
		NOT = { has_character_flag = epidemic_blame_game }
		blame_epidemic_ruler_trigger = yes
		capital_scope = {
			OR = {
				has_province_modifier = depopulated_1
				has_province_modifier = depopulated_2
				has_province_modifier = depopulated_3
			}
		}
	}

	immediate = {
		set_character_flag = epidemic_blame_game
		random_list = {
			10 = { # The cats
				character_event = { id = blame.10 }
			}
			10 = { # The Strigany
				modifier = { factor = 0 blame_strigany_present_trigger = no }
				character_event = { id = blame.30 }
			}
			10 = { # A witch hunt
				modifier = { factor = 0 blame_witch_hunt_possible_trigger = no }
				modifier = { factor = 2 any_courtier = { whunter_chapter_member_trigger = yes } }
				character_event = { id = blame.40 }
			}
			10 = { # The impious ruler
				modifier = { factor = 0 blame_impious_ruler_possible_trigger = no }
				character_event = { id = blame.60 }
			}
		}
	}
}

# blame.10 The cats are blamed
# Since the epidemic came the town is full of cats, and the people are sure
# they spread it. The ruler can pay to have some examined.
# Scope: ROOT is the ruler.
character_event = {
	id = blame.10
	desc = EVTDESCblame10
	picture = GFX_evt_depopulated_town
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	immediate = {
		capital_scope = { save_event_target_as = capital }
	}

	option = { # Gather some cats
		name = EVTOPTAblame10
		custom_tooltip = { text = EVTOPTAblame10_TT }
		if = { limit = { has_character_modifier = in_seclusion } scaled_wealth = -0.4 }
		else = { scaled_wealth = -0.1 }
		hidden_tooltip = {
			character_event = { id = blame.11 days = 20 random = 5 }
			character_event = { id = blame.12 days = 90 random = 30 }
		}
	}
	option = { # The cats are not to blame
		name = EVTOPTBblame10
		hidden_tooltip = { character_event = { id = blame.12 days = 90 random = 30 } }
	}
}

# blame.11 The cats seem healthy
# The examiner finds the cats calm and mostly healthy. Keeping some to watch
# may win the ruler a pet.
# Scope: ROOT is the ruler.
character_event = {
	id = blame.11
	desc = EVTDESCblame11
	picture = GFX_evt_cat
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	immediate = {
		set_character_flag = cats_examined
		if = {
			limit = { any_courtier_or_vassal = { has_minor_title = title_court_physician } }
			random_courtier_or_vassal = {
				limit = { has_minor_title = title_court_physician }
				save_event_target_as = cat_examiner
			}
		}
		else_if = {
			limit = { job_marshal = { always = yes } }
			job_marshal = { save_event_target_as = cat_examiner }
		}
		else = { random_courtier = { save_event_target_as = cat_examiner } }
	}

	option = { # Let them go
		name = EVTOPTAblame11
	}
	option = { # Keep some for observation
		name = EVTOPTBblame11
		scaled_wealth = -0.2
		set_character_flag = cats_observed
		hidden_tooltip = {
			random_list = {
				70 = { character_event = { id = blame.15 days = 30 random = 10 } }
				30 = { }
			}
		}
	}
}

			random_list = {
				20 = { character_event = { id = blame.20 days = 50 random = 20 } }

			}
		}
	}
	option = { # The cats are not the cause
		name = EVTOPTBblame12
		tooltip_info = diplomacy
		trigger = { diplomacy = 15 has_character_flag = cats_examined }
		custom_tooltip = { text = EVTOPTBblame12_TT }
		hidden_tooltip = {
			random_list = {
				20 = { character_event = { id = blame.20 days = 50 random = 20 } }
				80 = { character_event = { id = blame.19 days = 50 random = 20 } }
			}
		}
	}
	option = { # The cats seem mostly healthy
		name = EVTOPTCblame12
		trigger = {
			has_character_flag = cats_examined
			NOR = { diplomacy = 15 AND = { intrigue = 15 has_character_flag = cats_observed } }
		}
		custom_tooltip = { text = EVTOPTCblame12_TT }
		hidden_tooltip = {
			random_list = {
				50 = { character_event = { id = blame.20 days = 50 random = 20 } }
				50 = { character_event = { id = blame.19 days = 50 random = 20 } }
			}
		}
	}

		name = EVTOPTDblame12
		if = { limit = { has_character_modifier = pet_cat } remove_character_modifier = pet_cat }
		capital_scope = { add_province_modifier = { name = no_more_cats years = 10 } }
		hidden_tooltip = {
			random_list = {
				30 = { }
				70 = {
					random_courtier = {
						limit = {
							is_adult = yes
							NOR = { is_rival = ROOT is_friend = ROOT owes_favor_to = ROOT }
						}

					}
				}
			}
		}
	}
	option = { # This is absurd, go home
		name = EVTOPTEblame12
		capital_scope = { add_province_modifier = { name = epidemic_blame_1 years = 10 } }
		hidden_tooltip = {
			random_list = {
				20 = { character_event = { id = blame.19 days = 50 random = 20 } }
				80 = { character_event = { id = blame.20 days = 50 random = 20 } }
			}
		}
	}
}

# blame.32 The ruler decides
# Expelling the Strigany pays, but costs prestige and makes the ruler
# arbitrary; sparing them may lead to a purge (blame.33).
# Scope: ROOT is the ruler.
character_event = {
	id = blame.32
	picture = GFX_evt_suspicious_noble
	border = GFX_event_normal_frame_religion

	desc = { text = EVTDESCblame32 trigger = { has_character_flag = religion_head_says_stop } }
	desc = { text = EVTDESCblame32b trigger = { has_character_flag = religion_head_says_go } }
	desc = {
		text = EVTDESCblame32c
		trigger = {
			NOR = {
				has_character_flag = religion_head_says_stop
				has_character_flag = religion_head_says_go
			}
		}
	}

	is_triggered_only = yes

	option = { # The Strigany are not to blame
		name = EVTOPTAblame32
		clr_character_flag = religion_head_says_stop
		clr_character_flag = religion_head_says_go
		character_event = { id = blame.33 days = 30 }
	}
	option = { # Expel them
		name = EVTOPTBblame32
		clr_character_flag = religion_head_says_stop
		clr_character_flag = religion_head_says_go
		any_realm_province = {
			limit = { owner = { character = ROOT } }
			remove_province_modifier = epidemic_blame_1
		}
		prestige = -100
		scaled_wealth = 2
		if = { limit = { NOT = { trait = arbitrary } } add_trait = arbitrary }
		add_character_modifier = { name = expelled_strigany duration = 3650 inherit = yes }
		custom_tooltip = { text = blame_strigany_expelled_tt }
		hidden_tooltip = {
			expel_strigany_effect = yes
			character_event = { id = SoA.105 } # Strigany Expelled notice
		}
		any_friend = { limit = { culture_group = strigany_group } remove_friend = ROOT }
	}
}

# blame.33 A purge is coming
# Despite the ruler's word, a mob means to purge the Strigany in one of the
# ruler's cities.
# Scope: ROOT is the ruler.
character_event = {
	id = blame.33
	picture = GFX_evt_impalement
	border = GFX_event_normal_frame_religion

	desc = { text = EVTDESCblame33 trigger = { has_character_flag = pogrom_target_selected } }
	desc = {
		text = EVTDESCblame33b
		trigger = { NOT = { has_character_flag = pogrom_target_selected } }
	}

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				any_demesne_province = {
					NOT = { has_province_flag = pogrom_target }
					any_province_holding = { holding_type = city }
				}
			}
			random_demesne_province = {
				limit = {
					NOT = { has_province_flag = pogrom_target }
					any_province_holding = { holding_type = city }
				}
				random_province_holding = {
					limit = { holding_type = city }
					save_event_target_as = pogrom_city_target
				}
				set_province_flag = pogrom_target
				save_event_target_as = pogrom_province_target
			}
			set_character_flag = pogrom_target_selected
		}
	}

	option = { # Send troops to stop them
		name = EVTOPTAblame33
		if = {
			limit = { has_character_flag = pogrom_target_selected }
			event_target:pogrom_province_target = {
				remove_province_modifier = epidemic_blame_1
				add_province_modifier = { name = epidemic_blame_2 duration = 3650 }
			}
		}
		clr_character_flag = pogrom_target_selected
		clear_event_target = pogrom_city_target
	}
	option = { # Let them do it
		name = EVTOPTBblame33
		prestige = -100
		clr_character_flag = pogrom_target_selected
		clear_event_target = pogrom_city_target
	}
}

#   plague.1      Starter (on_outbreak)
#   plague.2      Word to the stricken realm
#   plague.3      Word to rulers far away
#   plague.4      Word to rulers nearby

#   plague.6      The plague reaches a border (disease on_province_infection)
#   plague.7      The plague approaches
#   plague.10     A depopulated province recovers (on_province_major_modifier)
#   plague.20     Yearly check for ravaged regions (on_yearly_pulse)
#   plague.21-32  A region is ravaged
#   plague.40     The plague is over (on_decade_pulse)
#
# Was events/rip_outbreak_events.txt: RIP.10000-10006 are plague.1-7 (the
# old RIP.10001 also picked the receivers, now plague.1 does), RIP.10030 is
# plague.10, RIP.10012 is plague.20, RIP.10015-10024 are plague.21-32 (the

# plague.1 A Great Plague breaks out
# The first outbreak of Nurgle's Rot or the Creeping Buboes since the last
# one ended is told to the owner, every player ruler near and far, and the
# owner's lieges.
# Scope: ROOT is the province. Fired from on_outbreak.
province_event = {
	id = plague.1
	hide_window = yes

	is_triggered_only = yes # on_outbreak

	trigger = {
		OR = { disease = nurgles_rot disease = creeping_buboes }
		NOT = { has_global_flag = great_plague_broadcasted }
	}

	immediate = {
		set_global_flag = great_plague_broadcasted
		if = {
			limit = { disease = nurgles_rot }
			set_global_flag = great_plague_rot
			clr_global_flag = great_plague_buboes
		}
		else = { set_global_flag = great_plague_buboes clr_global_flag = great_plague_rot }
		save_event_target_as = plague_outbreak_location
		owner = { save_event_target_as = plague_outbreak_ruler narrative_event = { id = plague.2 } }
		any_playable_ruler = {
			limit = { ai = no NOT = { character = event_target:plague_outbreak_ruler } }
			if = {
				limit = { is_liege_or_above = event_target:plague_outbreak_ruler }
				narrative_event = { id = plague.5 days = 2 random = 2 }
			}
			else_if = {
				limit = {
					distance = { where = event_target:plague_outbreak_location value = 1000 }
				}
				narrative_event = { id = plague.3 days = 6 random = 3 }
			}
			else = { narrative_event = { id = plague.4 days = 3 random = 3 } }
		}
	}
}

# plague.10 A depopulated province recovers
# A province with no sickness slowly fills again. Capitals, ports, trade
# posts, good stewards, many holdings and hospitals help; war, sieges,
# winter and deep depopulation hold it back.
# Scope: ROOT is the province. Fired from on_province_major_modifier.
province_event = {
	id = plague.10
	hide_window = yes

	is_triggered_only = yes # on_province_major_modifier

	trigger = {
		OR = {
			has_province_modifier = depopulated_1
			has_province_modifier = depopulated_2
			has_province_modifier = depopulated_3
		}
		NOT = { has_disease = yes }
	}

	immediate = {
		random_list = {
			30 = { # Recovers a step
				modifier = { factor = 1.5 is_capital = yes }
				modifier = { factor = 1.3 port = yes }
				modifier = { factor = 1.3 has_trade_post = yes }
				modifier = { factor = 1.3 holder_scope = { stewardship = 20 } }
				modifier = { factor = 1.2 holder_scope = { stewardship = 15 } }
				modifier = { factor = 1.1 holder_scope = { stewardship = 10 } }
				modifier = { factor = 1.1 num_of_settlements = 2 }
				modifier = { factor = 1.1 num_of_settlements = 3 }
				modifier = { factor = 1.1 num_of_settlements = 4 }
				modifier = { factor = 1.1 num_of_settlements = 5 }
				modifier = { factor = 1.1 num_of_settlements = 6 }
				modifier = { factor = 1.1 num_of_settlements = 7 }
				modifier = { factor = 1.1 hospital_level = 1 }
				modifier = { factor = 1.1 hospital_level = 2 }
				modifier = { factor = 1.1 hospital_level = 3 }
				modifier = { factor = 1.1 hospital_level = 4 }
				modifier = { factor = 1.1 hospital_level = 5 }
				modifier = { factor = 1.1 hospital_level = 6 }
				decrease_depopulation_effect = yes
			}
			70 = { # Nothing happens
				modifier = {
					factor = 3
					any_province_holding = { OR = { is_occupied = yes has_siege = yes } }
				}
				modifier = { factor = 1.2 is_winter = yes }
				modifier = { factor = 1.5 has_province_modifier = depopulated_2 }
				modifier = { factor = 2 has_province_modifier = depopulated_3 }
			}
		}
	}
}
