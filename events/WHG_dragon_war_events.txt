###############################################################################
# The Dragon's wars
# Namespace: dragonwar (was JD.30020 - 30053 and the news JD.10106, 10107 and
#   10131 in jd_chinese_war_events)
# Faction: Cathay & the East
#
# When anyone goes to war with Cathay's western governor, or the governor
# goes to war, the Dragon Throne sends imperial regiments and commanders
# down the roads that lead east: the Skull Road to Nuur, the Road to the
# East to Chimarak, the Ivory Road to Vargenspeak or the Spice Route to
# Skrapland, whichever ends nearest the enemy. If that host is destroyed
# within a year and the war goes on, a second, smaller host follows once.
# When the war ends, players in range hear the news from Cathay.
#
# Reinforcements                       dragonwar.1 - dragonwar.5
# End of the war                       dragonwar.10 - dragonwar.13
# News from Cathay                     dragonwar.20 - dragonwar.22
#
# Callers: common/on_actions/00_on_actions.txt, on_war_started (1, 2),
# on_war_ended_whitepeace and on_war_ended_invalid (10), on_war_ended_victory
# (11) and on_war_ended_defeat (12). An invasion of Cathay sets its own troop
# numbers first (cathay_invasion_declare_effect in wh_offmap_effects.txt).
# Shared code: cathay_war_* and cathay_spawn_commander_effect in
# 00_scripted_effects.txt, dragonwar_* in WHG_dragon_war_effects.txt. The
# century scaling of the vanilla effect always reads the latest era with the
# mod's dates, so Cathay sends its full might.
# Loc: localisation/00_WHG_dragon_war_events.csv
#
# Dropped: the Bay of Bengal fleet branch (the vanilla India region does not
# exist in the mod, and its spawn province is Aussenstadt in the Empire), the
# Jiuquan spawn (now Dragon Gate in Ulthuan), and the never-set
# cathay_sent_war_reinforcements flag. Saved: JD.30047 (the second host had
# no caller) is the second-wave text of dragonwar.4.
###############################################################################

namespace = dragonwar

# dragonwar.1 Cathay goes to war
# Hidden, on_war_started. The governor is the attacker: the defender's
# kowtow journey ends, and the governor raises the host (dragonwar.3).
# Scope: ROOT is the defender, FROM the governor.
character_event = {
	id = dragonwar.1
	hide_window = yes
	is_triggered_only = yes # on_war_started

	has_dlc = "Jade Dragon"

	trigger = {
		offmap_cathay = { governor = { character = FROM } }
	}

	immediate = {
		save_event_target_as = cathay_war_opponent
		if = {
			limit = { trait = kow_tow_travels }
			set_character_flag = kow_tow_abort_war
			character_event = { id = JD.32299 }
		}
		FROM = {
			save_event_target_as = cathay_war_governor
			save_event_target_as = cathay_war_aggressor
			character_event = { id = dragonwar.3 } # FROM is now the defender
		}
	}
}

# dragonwar.2 Cathay is attacked
# Hidden, on_war_started. The governor is the defender: the attacker's
# kowtow journey ends and his kowtow honours are stripped, and the governor
# raises the host (dragonwar.3).
# Scope: ROOT is the governor, FROM the attacker.
character_event = {
	id = dragonwar.2
	hide_window = yes
	is_triggered_only = yes # on_war_started

	has_dlc = "Jade Dragon"

	trigger = {
		offmap_cathay = { governor = { character = ROOT } }
	}

	immediate = {
		save_event_target_as = cathay_war_governor
		save_event_target_as = cathay_war_aggressor
		FROM = {
			save_event_target_as = cathay_war_opponent
			if = {
				limit = { trait = kow_tow_travels }
				set_character_flag = kow_tow_abort_war
				character_event = { id = JD.32299 }
			}
			if = {
				limit = {
					OR = {
						trait = kow_tow_completed_tier_1
						trait = kow_tow_completed_tier_2
						trait = kow_tow_completed_tier_3
					}
				}
				remove_trait = kow_tow_completed_tier_1
				remove_trait = kow_tow_completed_tier_2
				remove_trait = kow_tow_completed_tier_3
			}
		}
		repeat_event = { id = dragonwar.3 }
	}
}

# dragonwar.3 The Dragon sends a host
# Hidden. Works out the host: 8 regiments of quality 4 by default (or what
# an invasion set), changed by Cathay's status and the era, plus enough to
# match the enemy's levies and allies, minus the governor's own levies,
# capped at 350 regiments. Makes up to three commanders if the court lacks
# able ones, picks the road end nearest the enemy, spawns the host and warns
# the enemy (dragonwar.4). If the governor already has the levies, no host
# comes. The first host schedules the second-wave check (dragonwar.5).
# Scope: ROOT is the governor, FROM the enemy.
character_event = {
	id = dragonwar.3
	hide_window = yes
	is_triggered_only = yes # dragonwar.1, dragonwar.2, dragonwar.5

	has_dlc = "Jade Dragon"

	immediate = {
		offmap_cathay = {
			if = {
				limit = { NOT = { has_offmap_flag = cathay_entered_war } }
				set_offmap_flag = cathay_entered_war
			}
		}

		if = { # An invasion already set the numbers
			limit = { offmap_cathay = { has_offmap_tmp_flag = cathay_war_variables_set } }
			offmap_cathay = { clr_offmap_tmp_flag = cathay_war_variables_set }
		}
		else = {
			set_variable = { which = cathay_war_troop_quantity value = 8 } # regiments of about 500
			set_variable = { which = cathay_war_target_match_mult value = 0.95 }
			set_variable = { which = cathay_war_troop_quality value = 4 } # 1 - 7
			set_variable = { which = cathay_war_minimum_commander_martial value = 14 }
		}
		if = { # The second host is half the size
			limit = { offmap_cathay = { has_offmap_flag = cathay_second_wave } }
			set_variable = { which = cathay_war_troop_quantity value = 4 }
		}

		cathay_war_modify_variables_by_status_effect = yes
		cathay_war_modify_variables_by_century_effect = yes

		# Match the enemy's levies and allies
		export_to_variable = {
			which = cathay_temp_troop_calc
			value = realm_levies_plus_allies
			who = FROM
		}
		divide_variable = { which = cathay_temp_troop_calc value = 500 }
		multiply_variable = { which = cathay_temp_troop_calc which = cathay_war_target_match_mult }
		change_variable = { which = cathay_war_troop_quantity which = cathay_temp_troop_calc }

		# Every 780 of the governor's own levies means 500 fewer from the east
		export_to_variable = { which = cathay_temp_troop_calc value = realm_levies }
		subtract_variable = { which = cathay_temp_troop_calc value = 7500 }
		divide_variable = { which = cathay_temp_troop_calc value = 780 }
		subtract_variable = { which = cathay_war_troop_quantity which = cathay_temp_troop_calc }

		any_courtier_or_vassal = {
			set_variable = { which = cathay_war_minimum_commander_martial which = PREV }
		}
		if = {
			limit = {
				NOT = {
					any_courtier_or_vassal = {
						count = 3
						martial = cathay_war_minimum_commander_martial
					}
				}
			}
			cathay_spawn_commander_effect = yes
			cathay_spawn_commander_effect = yes
			cathay_spawn_commander_effect = yes
		}

		if = {
			limit = { check_variable = { which = cathay_war_troop_quantity value = 1 } }
			if = { # Cap at 175,000 men
				limit = { check_variable = { which = cathay_war_troop_quantity value > 350 } }
				set_variable = { which = cathay_war_troop_quantity value = 350 }
			}
			if = {
				limit = {
					check_variable = { which = cathay_war_troop_quantity_override value > 0 }
					check_variable = {
						which = cathay_war_troop_quantity_override
						which = cathay_war_troop_quantity
					}
				}
				set_variable = {
					which = cathay_war_troop_quantity
					which = cathay_war_troop_quantity_override
				}
				set_variable = { which = cathay_war_troop_quantity_override value = 0 }
			}
			FROM = { capital_scope = { save_event_target_as = cathay_war_front } }
			dragonwar_pick_road_effect = yes
			cathay_war_spawn_troops_effect = yes
			FROM = { character_event = { id = dragonwar.4 } }
			if = {
				limit = { offmap_cathay = { NOT = { has_offmap_flag = cathay_second_wave } } }
				FROM = { character_event = { id = dragonwar.5 days = 365 } }
			}
		}
	}
}

# dragonwar.4 The Dragon's host
# The enemy hears that an imperial host has come down the road from the
# east, or, the second time, that the Dragon has sent another.
# Scope: ROOT is the enemy, FROM the governor.
character_event = {
	id = dragonwar.4
	desc = {
		trigger = { offmap_cathay = { NOT = { has_offmap_flag = cathay_second_wave } } }
		text = EVTDESCdragonwar4
	}
	desc = {
		trigger = { offmap_cathay = { has_offmap_flag = cathay_second_wave } }
		text = EVTDESCdragonwar4_again
		picture = GFX_evt_china_rebel_general
	}
	picture = GFX_evt_china_invades
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # dragonwar.3

	has_dlc = "Jade Dragon"

	option = {
		name = {
			trigger = { offmap_cathay = { NOT = { has_offmap_flag = cathay_second_wave } } }
			text = EVTOPTAdragonwar4
		}
		name = {
			trigger = { offmap_cathay = { has_offmap_flag = cathay_second_wave } }
			text = EVTOPTAdragonwar4_again
		}
		custom_tooltip = { text = EVTTOOLTIPdragonwar4 }
	}
}

# dragonwar.5 The second host
# Hidden, a year after the first host. If the war with the same enemy goes
# on, the Dragon sends a second, smaller host, once per war. The levy check
# of dragonwar.3 keeps it home while the first host still stands.
# Scope: ROOT is the enemy, FROM the governor.
character_event = {
	id = dragonwar.5
	hide_window = yes
	is_triggered_only = yes # dragonwar.3

	has_dlc = "Jade Dragon"

	trigger = {
		offmap_cathay = {
			governor = { character = FROM }
			NOT = { has_offmap_flag = cathay_second_wave }
		}
		war_with = FROM
	}

	immediate = {
		offmap_cathay = { set_offmap_flag = cathay_second_wave }
		FROM = { character_event = { id = dragonwar.3 } }
	}
}

# dragonwar.10 White peace
# Hidden, on_war_ended_whitepeace and on_war_ended_invalid. If the governor
# fought, players in range hear of the white peace (dragonwar.22) unless the
# war was a minor one, and the war flags are cleared (dragonwar.13).
# Scope: ROOT is the defender, FROM the attacker.
character_event = {
	id = dragonwar.10
	hide_window = yes
	is_triggered_only = yes # on_war_ended_whitepeace, on_war_ended_invalid

	has_dlc = "Jade Dragon"

	trigger = {
		offmap_cathay = { governor = { OR = { character = ROOT character = FROM } } }
	}

	immediate = {
		character_event = { id = dragonwar.13 days = 1 }
		if = {
			limit = { offmap_cathay = { governor = { character = ROOT } } }
			offmap_cathay = { set_offmap_tmp_flag = cathay_was_defender_but_white_peace }
			save_event_target_as = the_protector_general
			FROM = { save_event_target_as = the_aggressor }
		}
		else = {
			offmap_cathay = { set_offmap_tmp_flag = cathay_was_aggressor_but_white_peace }
			FROM = { save_event_target_as = the_protector_general }
			save_event_target_as = the_defender
		}
		if = {
			limit = { using_unimportant_cb = no }
			any_player = {
				limit = {
					has_offmap_news_enabled = offmap_cathay
					is_within_diplo_range = event_target:the_protector_general
				}
				narrative_event = { id = dragonwar.22 days = 1 }
			}
		}
	}
}

# dragonwar.13 The war flags
# Hidden, a day after a war ends. Once the governor is at peace, the war and
# second-host flags are cleared.
# Scope: ROOT is either side of the war.
character_event = {
	id = dragonwar.13
	hide_window = yes
	is_triggered_only = yes # dragonwar.10, dragonwar.11, dragonwar.12

	has_dlc = "Jade Dragon"

	immediate = {
		if = {
			limit = { offmap_cathay = { governor = { war = no } } }
			offmap_cathay = {
				clr_offmap_flag = cathay_entered_war
				clr_offmap_flag = cathay_second_wave
			}
		}
	}
}

# dragonwar.20 News: the Dragon retreats
# Cathay lost a war it started or one it was defending. The winner hears it
# in their own words.
# Scope: ROOT is a player.
narrative_event = {
	id = dragonwar.20
	title = NEWS_FROM_CATHAY
	desc = {
		trigger = { offmap_cathay = { has_offmap_tmp_flag = cathay_lost_offensive_war } }
		text = EVTDESCdragonwar20_attack
	}
	desc = {
		trigger = { offmap_cathay = { has_offmap_tmp_flag = cathay_lost_defensive_war } }
		text = EVTDESCdragonwar20_defence
	}
	picture = GFX_evt_china_civil_war
	portrait = offmap_cathay
	window = EventWindowOffmap
	background = GFX_event_window_news_from_china

	is_triggered_only = yes # dragonwar_news_effect (dragonwar.11, dragonwar.12)

	has_dlc = "Jade Dragon"

	trigger = {
		NOT = { offmap_cathay = { has_offmap_flag = no_war_news } }
	}

	fail_trigger_effect = { offmap_cathay = { clr_offmap_flag = no_war_news } }

	option = {
		name = { trigger = { character = event_target:the_winner } text = EVTOPTAdragonwar20 }
		name = {
			trigger = { NOT = { character = event_target:the_winner } }
			text = EVTOPTAdragonwar20_other
		}
		custom_tooltip = { text = EVTTOOLTIPdragonwar20 }
		show_portrait = event_target:the_winner
	}

	after = {
		hidden_effect = {
			offmap_cathay = {
				clr_offmap_tmp_flag = cathay_lost_offensive_war
				clr_offmap_tmp_flag = cathay_lost_defensive_war
			}
		}
	}
}
