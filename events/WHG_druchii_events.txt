# The invasion of Ulthuan
# prepare_ulthuan_invasion gives an independent Druchii (or Asur) ruler two
# years to gather a host against whoever holds Ulthuan. Every month or so
# troops, ships or a famous commander may join (druchii_invasion_muster_effect).
# If no war has started by the end, the host breaks up (druchii.3).
###############################################################################

# druchii.1 The muster goes on (hidden)
# On the first call it warns the ruler of Ulthuan (druchii.2), so that his
# event has the invader as FROM.
# Scope: ROOT is the invader; FROM is the ruler of Ulthuan on the first
# call (prepare_ulthuan_invasion), then the invader himself
# (druchii_invasion_muster_effect).
character_event = {
	id = druchii.1
	hide_window = yes

	is_triggered_only = yes # prepare_ulthuan_invasion, druchii_invasion_muster_effect

	trigger = {
		has_character_modifier = preparing_ulthuan_invasion
	}

	immediate = {
		if = { # First call, from the decision: warn the ruler of Ulthuan
			limit = { NOT = { FROM = { character = ROOT } } }
			FROM = { character_event = { id = druchii.2 days = 12 random = 30 } }
		}
		druchii_invasion_muster_effect = yes
	}
}

# druchii.23 A sorcerer of Naggaroth (hidden)
# A gifted Druchii man under the Witch King is found out, about once in
# ten years. The stronger his power, the sooner; a man with only a faint
# gift may go unnoticed. The lord of Naggaroth judges him (druchii.24).
# Scope: ROOT is the man (five-year list druchii_sorcerers).
character_event = {
	id = druchii.23
	hide_window = yes

	is_triggered_only = yes # on_five_year_pulse
	only_men = yes
	only_capable = yes
	prisoner = no
	min_age = 16
	culture_group = druchii_group

	trigger = {
		OR = { hidden_or_open_magic_potential_trigger = yes has_magic_power_trigger = yes }
		NOT = { trait = in_hiding }
		NOT = { trait = malekith }
		NOT = { trait = hollowed }
		independent = no
		top_liege = { trait = malekith }
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 0.33 trait = magic_potential_1 }
		modifier = { factor = 0.5 trait = magic_potential_2 }
		modifier = { factor = 0.6 trait = magic_potential_3 }
		modifier = { factor = 0.67 trait = magic_potential_4 }
		modifier = { factor = 0.8 has_magic_power_trigger = yes }

		modifier = { factor = 3 magic_power_4_threshold_trigger = yes }
		modifier = { factor = 0.5 trait = deceitful }
	}

	immediate = {
		top_liege = { character_event = { id = druchii.24 } }
	}

###############################################################################
# The fall of Naggaroth
# Naggaroth is held together by Malekith alone. When it passes to anyone
# else, the empire, the Black Guard and (if the new lord is no Asur) a
# Druchii-held Ulthuan break apart. A ruler who later gathers enough of the
# Land of Chill can restore it by decision (druchii.29), and it is never
# torn apart again.
###############################################################################

# druchii.27 Naggaroth breaks apart (hidden)
# Scope: ROOT is the new holder of a title (on_new_holder).
character_event = {
	id = druchii.27
	hide_window = yes

	is_triggered_only = yes # on_new_holder
	only_rulers = yes

	trigger = {
		NOT = { has_global_flag = malekith_lost_naggaroth }
		e_naggaroth = { holder_scope = { NOT = { trait = malekith } } }
	}

	immediate = {
		set_global_flag = malekith_lost_naggaroth
		e_naggaroth = {
			holder_scope = {
				save_event_target_as = naggaroth_new_holder
				set_character_flag = new_holder_lost_naggaroth
			}
			set_government_type = druchii_government
			destroy_landed_title = THIS
		}
		e_ulthuan = {
			holder_scope = {
				if = {
					limit = {
						character = event_target:naggaroth_new_holder
						NOT = { culture_group = asur_group }
					}
					e_ulthuan = { destroy_landed_title = THIS }
				}
			}
		}
		d_blackguard = { set_government_type = druchii_government destroy_landed_title = THIS }
		narrative_event = { id = druchii.28 }
	}
}
