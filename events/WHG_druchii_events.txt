# The invasion of Ulthuan
# prepare_ulthuan_invasion gives an independent Druchii (or Asur) ruler two
# years to gather a host against whoever holds Ulthuan. Every month or so
# troops, ships or a famous commander may join (druchii_invasion_muster_effect).
# If no war has started by the end, the host breaks up (druchii.3).
###############################################################################

# druchii.1 The muster goes on (hidden)
# On the first call it warns the ruler of Ulthuan (druchii.2), so that his
# event has the invader as FROM.
# Scope: ROOT is the invader; FROM is the ruler of Ulthuan on the first
# call (prepare_ulthuan_invasion), then the invader himself
# (druchii_invasion_muster_effect).
character_event = {
	id = druchii.1
	hide_window = yes

	is_triggered_only = yes # prepare_ulthuan_invasion, druchii_invasion_muster_effect

	trigger = {
		has_character_modifier = preparing_ulthuan_invasion
	}

	immediate = {
		if = { # First call, from the decision: warn the ruler of Ulthuan
			limit = { NOT = { FROM = { character = ROOT } } }
			FROM = { character_event = { id = druchii.2 days = 12 random = 30 } }
		}
		druchii_invasion_muster_effect = yes
	}
}

# druchii.23 A sorcerer of Naggaroth (hidden)
# A gifted Druchii man under the Witch King is found out, about once in
# ten years. The stronger his power, the sooner; a man with only a faint
# gift may go unnoticed. The lord of Naggaroth judges him (druchii.24).
# Scope: ROOT is the man (five-year list druchii_sorcerers).
character_event = {
	id = druchii.23
	hide_window = yes

	is_triggered_only = yes # on_five_year_pulse
	only_men = yes
	only_capable = yes
	prisoner = no
	min_age = 16
	culture_group = druchii_group

	trigger = {
		OR = { hidden_or_open_magic_potential_trigger = yes has_magic_power_trigger = yes }

		top_liege = { trait = malekith }

		factor = 1

		modifier = { factor = 0.67 trait = magic_potential_4 }

		top_liege = { character_event = { id = druchii.24 } }

###############################################################################
# The fall of Naggaroth
# Naggaroth is held together by Malekith alone. When it passes to anyone
# else, the empire, the Black Guard and (if the new lord is no Asur) a
# Druchii-held Ulthuan break apart. A ruler who later gathers enough of the
# Land of Chill can restore it by decision (druchii.29), and it is never
# torn apart again.
###############################################################################

# druchii.27 Naggaroth breaks apart (hidden)
# Scope: ROOT is the new holder of a title (on_new_holder).
character_event = {
	id = druchii.27
	hide_window = yes

	is_triggered_only = yes # on_new_holder
	only_rulers = yes

	trigger = {
		NOT = { has_global_flag = malekith_lost_naggaroth }
		e_naggaroth = { holder_scope = { NOT = { trait = malekith } } }
	}

	immediate = {
		set_global_flag = malekith_lost_naggaroth
		e_naggaroth = {
			holder_scope = {
				save_event_target_as = naggaroth_new_holder
				set_character_flag = new_holder_lost_naggaroth
			}
			set_government_type = druchii_government
			destroy_landed_title = THIS
		}
		e_ulthuan = {
			holder_scope = {
				if = {
					limit = {
						character = event_target:naggaroth_new_holder
						NOT = { culture_group = asur_group }
					}
					e_ulthuan = { destroy_landed_title = THIS }
				}
			}
		}
		d_blackguard = { set_government_type = druchii_government destroy_landed_title = THIS }
		narrative_event = { id = druchii.28 }
	}
}

	}

	border = GFX_event_narrative_frame_war

	option = { # Malekith

		trigger = { trait = malekith }
		prestige = -2000
	}

		name = EVTOPTBdruchii28

		name = EVTOPTCdruchii28

		name = EVTOPTEdruchii28

# Namespace: drufeast (was druchii_feast in BR_druchi_feast.txt, a reflavour
# of the vanilla Kali Puja feast)
# Faction: Dark Elves
#
# A Druchii lord holds a month-long feast for his vassals (holdafeast
# decision). He sends the invitations, buys slaves as gifts for his spouse
# or lover, chooses how lavish his table will be and what bloody
# entertainment to hire. At the feast the guests eat, plot, duel, drink and
# seduce, a spectacle may go wrong, and at the end the host honours Khaine
# with a sacrifice. Each guest goes home with a good or a bad memory of it.
#
# Invitations and preparations            drufeast.1 - drufeast.8
# The feast                               drufeast.10 - drufeast.19
# Intrigues at the feast                  drufeast.20 - drufeast.39
# Spectacles                              drufeast.40 - drufeast.52
# The rite of Khaine                      drufeast.60 - drufeast.71
#
# The feast runs on character flags: the host has the modifier
# holding_large_druchi_feast, every guest the flag attending_druchii_feast,
# and the event target feast_host is saved down every chain. Intrigues are
# rolled once per guest when they arrive (drufeast_intrigue_roll_effect),
# not by MTTH.
#
# Callers:
#   decisions/wh_druchii_decisions.txt: holdafeast (1).
# Support: common/scripted_effects/wh_druchii_feast_effects.txt
# Loc: localisation/00_WHG_druchii_events.csv
###############################################################################

###############################################################################
# Druchii sorceress events
# Namespace: sorceress (was druchii.1 - 5 and darkelfmagic in
# BR_dark_elf_magic_events.txt)
# Faction: Dark Elves
#
# By the Witch King's decree every Druchii woman with the gift must give
# herself up to the Convent of Sorceresses in Ghrond. A gifted, unlanded
# woman under Malekith is summoned; a player may try to hide her gift, but
# she is usually found out. In Ghrond she spends ten years of trials. If
# her learning has reached 14 she becomes a Sorceress, a bride of the Witch
# King; if not she is hollowed and sent to his court as a mindless servant.
#
# The summons                             sorceress.1 - sorceress.4
# The trials of Ghrond                    sorceress.5 - sorceress.8
#
# Callers:
#   common/on_actions/wh_on_actions.txt: sorceress.1 on on_bi_yearly_pulse;
#     sorceress.2 on the bi-yearly list druchii (also fired by
#     sorceress.1).
# Support: common/scripted_effects/wh_sorceress_effects.txt
# Loc: localisation/00_WHG_druchii_events.csv
###############################################################################

namespace = sorceress

###############################################################################
# The summons
###############################################################################

# sorceress.1 The Witch King's decree
# A gifted, unlanded Druchii woman of sixteen or more under Malekith, while
# Ghrond is held by a Sorceress of his empire, is summoned there. A player
# may hide her gift instead; she is found out three times in ten
# (sorceress.2), less often if she is clever at lies, more if her gift is
# strong.
# Scope: ROOT is the woman (on_bi_yearly_pulse).
character_event = {
	id = sorceress.1
	title = EVTTITLEsorceress1
	desc = EVTDESCsorceress1
	picture = GFX_evt_sorceress

	is_triggered_only = yes # on_bi_yearly_pulse
	culture_group = druchii_group
	only_women = yes
	min_age = 16
	prisoner = no
	only_capable = yes

	trigger = {
		is_landed = no
		magic_potential_trigger = yes
		NOT = { is_trained_mage_trigger = yes }
		NOT = { trait = mage_apprentice }
		top_liege = { trait = malekith }
		sorceress_ghrond_ready_trigger = yes
	}

	option = { # Go
		name = EVTOPTAsorceress1
		sorceress_to_ghrond_effect = yes
	}
	option = { # Hide it
		name = EVTOPTBsorceress1
		trigger = { ai = no }
		hide_magic_effect = yes
		hidden_effect = {
			random_list = {
				70 = {
					modifier = { factor = 0.45 has_character_flag = magic_potential_flag_4 }
					modifier = { factor = 0.75 has_character_flag = magic_potential_flag_3 }
					modifier = { factor = 0.95 has_character_flag = magic_potential_flag_2 }
					modifier = { factor = 1.25 has_character_flag = magic_potential_flag_1 }
					modifier = { factor = 1.75 trait = deceitful }
					modifier = { factor = 1.5 trait = paranoid }
					modifier = { factor = 0.75 trait = trusting }
					modifier = { factor = 0.5 trait = honest }
				}
				30 = { character_event = { id = sorceress.2 days = 200 random = 400 } }
			}
		}
	}
}

# sorceress.5 A year in Ghrond
# Each year of the ten the apprentice learns cunning (and learning) or
# picks up a vice; the clever and gifted have fewer bad years. After ten
# years she faces her final trial (sorceress.6).
# Scope: ROOT is the apprentice (sorceress_to_ghrond_effect, then itself).
character_event = {
	id = sorceress.5
	title = EVTTITLEsorceress3
	desc = EVTDESCsorceress5
	picture = GFX_evt_sorceress

	is_triggered_only = yes # sorceress_to_ghrond_effect, sorceress.5

	trigger = {
		trait = mage_apprentice
	}

	option = {
		name = EVTOPTAsorceress5
		prestige = 25
		hidden_effect = {
			diplomatic_immunity = yes
			if = {
				limit = { check_variable = { which = sorceresses_studies value = 10 } }
				set_variable = { which = sorceresses_studies value = 0 }
				character_event = { id = sorceress.6 }
			}
			else = {
				change_variable = { which = sorceresses_studies value = 1 }
				character_event = { id = sorceress.5 days = 365 }
				random_list = {
					50 = { # A good year
						change_learning = 1
						random_list = {
							10 = { add_trait = deceitful }
							10 = { add_trait = ambitious }
							10 = { add_trait = cruel }
							10 = { add_trait = homosexual }
							10 = { religion = atharti }
						}
					}
					50 = { # A bad year
						modifier = { factor = 0.5 trait = genius }
						modifier = { factor = 0.65 trait = brilliant }
						modifier = { factor = 0.75 trait = smart }
						modifier = { factor = 0.85 trait = quick }
						modifier = { factor = 0.95 trait = bright }
						modifier = { factor = 1.25 trait = dim }
						modifier = { factor = 1.35 trait = dumb }
						modifier = { factor = 1.5 trait = slow }
						modifier = { factor = 1.75 trait = idiot }
						modifier = { factor = 2 trait = imbecile }
						modifier = { factor = 0.95 trait = magic_potential_2 }
						modifier = { factor = 0.85 trait = magic_potential_3 }
						modifier = { factor = 0.65 trait = magic_potential_4 }
						modifier = { factor = 1.25 trait = slothful }
						modifier = { factor = 1.35 trait = content }
						modifier = { factor = 1.25 trait = drunkard }
						modifier = { factor = 1.25 trait = addict }
						modifier = { factor = 1.25 trait = gluttonous }
						modifier = { factor = 1.25 trait = lustful }
						random_list = {
							10 = { add_trait = craven }
							10 = { add_trait = slothful }
							10 = { add_trait = paranoid }
							10 = { add_trait = lustful }
							5 = { add_trait = addict }
							5 = {
								change_variable = { which = "lunatic_counter" value = 1 }
								character_event = { id = mutate.5 } # Lunacy counter
							}
						}
					}
				}
			}
		}
	}
}

# sorceress.6 The final trial
# With learning 14 or more she becomes a Sorceress of the Convent, with
# the Lore of Dark Magic. Otherwise she is hollowed and sent to the Witch
# King's court as a servant. Her parents hear either way.
# Scope: ROOT is the apprentice (sorceress.5).
character_event = {
	id = sorceress.6
	title = EVTTITLEsorceress3
	desc = { text = EVTDESCsorceress6 trigger = { learning = 14 NOT = { trait = incapable } } }
	desc = {
		text = EVTDESCsorceress6b
		trigger = { OR = { NOT = { learning = 14 } trait = incapable } }
	}
	picture = GFX_evt_sorceress

	is_triggered_only = yes # sorceress.5

	option = { # Sorceress
		name = EVTOPTAsorceress6
		trigger = { learning = 14 NOT = { trait = incapable } }
		remove_trait = mage_apprentice
		add_trait = druchii_sorceress
		add_trait = lore_dark_magic
		prestige = 100
		piety = 100
		hidden_effect = {
			remove_trait = druchii_witch_elf
			diplomatic_immunity = no
			enable_magic_xp_effect = yes
			magic_power_random_effect = yes
			father = { character_event = { id = sorceress.7 } }
			mother = { character_event = { id = sorceress.7 } }
		}
	}
	option = { # Hollowed
		name = EVTOPTBsorceress6
		trigger = { OR = { NOT = { learning = 14 } trait = incapable } }
		gift_sever_effect = yes
		prestige = -100
		piety = -100
		e_naggaroth = { holder_scope = { ROOT = { move_character = PREV } } }
		hidden_effect = {
			remove_trait = druchii_witch_elf
			diplomatic_immunity = no
			father = { character_event = { id = sorceress.8 } }
			mother = { character_event = { id = sorceress.8 } }
		}
	}
}
