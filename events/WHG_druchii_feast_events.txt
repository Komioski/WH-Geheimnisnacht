###############################################################################
# The Feast of Khaine
# Namespace: drufeast (was druchii_feast in BR_druchi_feast.txt, a reflavour
# of the vanilla Kali Puja feast)
# Faction: Dark Elves
#
# A Druchii lord holds a month-long feast for his vassals (holdafeast
# decision). He sends the invitations, buys slaves as gifts for his spouse
# or lover, chooses how lavish his table will be and what bloody
# entertainment to hire. At the feast the guests eat, plot, duel, drink and
# seduce, a spectacle may go wrong, and at the end the host honours Khaine
# with a sacrifice. Each guest goes home with a good or a bad memory of it.
#
# Invitations and preparations            drufeast.1 - drufeast.8
# The feast                               drufeast.10 - drufeast.19
# Intrigues at the feast                  drufeast.20 - drufeast.39
# Spectacles                              drufeast.40 - drufeast.52
# The rite of Khaine                      drufeast.60 - drufeast.71
#
# The feast runs on character flags: the host has the modifier
# holding_large_druchi_feast, every guest the flag attending_druchii_feast,
# and the event target feast_host is saved down every chain. Intrigues are
# rolled once per guest when they arrive (drufeast_intrigue_roll_effect),
# not by MTTH.
#
# Callers:
#   decisions/wh_druchii_decisions.txt: holdafeast (1).
# Support: common/scripted_effects/wh_druchii_feast_effects.txt
# Loc: localisation/00_WHG_druchii_feast_events.csv
###############################################################################

namespace = drufeast

# drufeast.1 The invitations
# The host invites every vassal, picks a gift for spouse or lover, sets his
# table and hires entertainment. The feast begins twelve days later.
# Scope: ROOT is the host (holdafeast).
character_event = {
	id = drufeast.1
	desc = EVTDESCdrufeast1
	picture = GFX_evt_emissary

	is_triggered_only = yes # holdafeast

	option = {
		name = EVTOPTAdrufeast1
		custom_tooltip = { text = drufeast_invitations_tt }
		hidden_effect = {
			save_event_target_as = feast_host
			any_vassal = {
				limit = {
					prisoner = no
					war = no
					in_command = no
					NOT = { trait = incapable }
					age = 16
					NOT = { has_character_flag = do_not_disturb }
				}
				letter_event = { id = drufeast.2 }
			}
			if = { limit = { is_married = yes } character_event = { id = drufeast.4 } }
			else_if = { limit = { has_lover = yes } character_event = { id = drufeast.5 } }
			character_event = { id = drufeast.7 days = 1 }
			character_event = { id = drufeast.10 days = 12 }
		}
	}
}

# drufeast.2 An invitation from the liege
# Scope: ROOT is a vassal, FROM the host (drufeast.1).
letter_event = {
	id = drufeast.2
	desc = EVTDESCdrufeast2
	picture = GFX_evt_emissary

	is_triggered_only = yes # drufeast.1

	option = { # Attend
		name = EVTOPTAdrufeast2
		set_character_flag = do_not_disturb
		set_character_flag = drufeast_dnd
		if = {
			limit = { trait = in_hiding }
			remove_trait = in_hiding
			add_character_modifier = { name = went_out_of_hiding_timer duration = 180 hidden = yes }
		}
		hidden_effect = {
			character_event = { id = drufeast.11 days = 12 }
			character_event = { id = drufeast.19 days = 42 } # Safety cleanup
			if = { limit = { is_married = yes } character_event = { id = drufeast.4 } }
			else_if = { limit = { has_lover = yes } character_event = { id = drufeast.5 } }
		}
		ai_chance = {
			factor = 2
			modifier = { factor = 5 has_focus = focus_carousing }
			modifier = { factor = 0.1 trait = in_hiding }
			modifier = { factor = 2 opinion = { who = FROM value = -30 } }
			modifier = { factor = 5 opinion = { who = FROM value = -15 } }
			modifier = { factor = 5 opinion = { who = FROM value = 0 } }
		}
	}
	option = { # Refuse
		name = EVTOPTBdrufeast2
		FROM = { letter_event = { id = drufeast.3 days = 3 } }
		ai_chance = { factor = 1 modifier = { factor = 0 opinion = { who = FROM value = 25 } } }
	}
}

# drufeast.4 A slave for the spouse
# Tradition says a Druchii brings a slave as a gift for the spouse at a
# feast. A fine Asur (or Dawi) slave, a human wretch, nothing, or a gift
# for a lover instead. The spouse reacts at the feast (drufeast.6).
# Scope: ROOT is the host or a guest (drufeast.1, drufeast.2).
character_event = {
	id = drufeast.4
	desc = EVTDESCdrufeast4
	picture = GFX_evt_feast

	is_triggered_only = yes # drufeast.1, drufeast.2

	option = { # A fine slave
		name = EVTOPTAdrufeast4
		trigger = { spouse = { is_female = yes } }
		scaled_wealth = -0.2
		set_character_flag = druchii_feast_fine_gift_to_spouse
		hidden_effect = { character_event = { id = drufeast.6 days = 30 } }
	}
	option = { # A fine slave
		name = EVTOPTBdrufeast4
		trigger = { spouse = { is_female = no } }
		scaled_wealth = -0.2
		set_character_flag = druchii_feast_fine_gift_to_spouse
		hidden_effect = { character_event = { id = drufeast.6 days = 30 } }
	}
	option = { # A human wretch
		name = EVTOPTCdrufeast4
		scaled_wealth = -0.05
		set_character_flag = druchii_feast_small_gift_to_spouse
		hidden_effect = { character_event = { id = drufeast.6 days = 30 } }
	}
	option = { # Nothing
		name = EVTOPTDdrufeast4
		trigger = { NOT = { trait = kind } }
		set_character_flag = druchii_feast_no_gift_to_spouse
		hidden_effect = { character_event = { id = drufeast.6 days = 30 } }
	}
	option = { # The lover instead
		name = EVTOPTEdrufeast4
		trigger = { has_lover = yes spouse = { NOT = { is_lover = ROOT } } }
		set_character_flag = druchii_feast_no_gift_to_spouse
		hidden_effect = { character_event = { id = drufeast.5 } }
	}
}

# drufeast.6 The giving of gifts
# Spouse and lover react to their gifts. A spouse who sees the lover get
# the better slave is furious.
# Scope: ROOT is the giver (drufeast.4, drufeast.5).
character_event = {
	id = drufeast.6
	desc = EVTDESCdrufeast6
	picture = GFX_evt_feast

	is_triggered_only = yes # drufeast.4, drufeast.5

	option = {
		name = EVTOPTAdrufeast6
		if = {
			limit = { has_character_flag = druchii_feast_fine_gift_to_spouse }
			spouse = {
				opinion = { modifier = opinion_very_grateful who = ROOT years = 1 }
				random = { chance = 30 add_lover = ROOT }
			}
		}
		else_if = {
			limit = { has_character_flag = druchii_feast_small_gift_to_spouse }
			spouse = { opinion = { modifier = opinion_grateful who = ROOT years = 1 } }
		}
		else_if = {
			limit = { has_character_flag = druchii_feast_no_gift_to_spouse }
			spouse = { opinion = { modifier = opinion_insulted who = ROOT years = 1 } }
		}
		if = {
			limit = { has_character_flag = druchii_feast_fine_gift_to_lover }
			random_lover = { opinion = { modifier = opinion_very_grateful who = ROOT years = 1 } }
			spouse = { opinion = { modifier = opinion_furious who = ROOT years = 1 } }
		}
		else_if = {
			limit = { has_character_flag = druchii_feast_small_gift_to_lover }
			random_lover = { opinion = { modifier = opinion_grateful who = ROOT years = 1 } }
			spouse = { opinion = { modifier = opinion_insulted who = ROOT years = 1 } }
		}
		else_if = {
			limit = { has_character_flag = druchii_feast_no_gift_to_lover }
			random_lover = { opinion = { modifier = opinion_disappointed who = ROOT years = 1 } }
		}
		clr_character_flag = druchii_feast_fine_gift_to_spouse
		clr_character_flag = druchii_feast_small_gift_to_spouse
		clr_character_flag = druchii_feast_no_gift_to_spouse
		clr_character_flag = druchii_feast_fine_gift_to_lover
		clr_character_flag = druchii_feast_small_gift_to_lover
		clr_character_flag = druchii_feast_no_gift_to_lover
	}
}

# drufeast.10 The feast begins
# The host's courtiers join in. A month later the feast ends (drufeast.15).
# Scope: ROOT is the host (drufeast.1).
character_event = {
	id = drufeast.10
	desc = EVTDESCdrufeast10
	picture = GFX_evt_feast

	is_triggered_only = yes # drufeast.1

	option = {
		name = EVTOPTAdrufeast10
		set_character_flag = druchii_feast_begun
		hidden_effect = {
			save_event_target_as = feast_host
			character_event = { id = drufeast.14 days = 3 random = 5 } # The show
			character_event = { id = drufeast.60 days = 25 } # The rite of Khaine
			character_event = { id = drufeast.15 days = 30 } # The end
			character_event = { id = drufeast.19 days = 31 } # Cleanup
			random_list = { # A spectacle by the beastmaster or a sorceress
				10 = {
					modifier = { factor = 5 has_character_flag = feast_beastmaster }
					character_event = { id = drufeast.40 days = 10 random = 5 }
				}
				10 = { character_event = { id = drufeast.47 days = 10 random = 5 } }
				8 = { # Harpies from the crags, only in Naggaroth
					modifier = { factor = 0 NOT = { capital_scope = { region = world_new_world_naggaroth } } }
					character_event = { id = drufeast.52 days = 10 random = 10 }
				}
				60 = { }
			}
			any_courtier = {
				limit = { is_adult = yes is_ruler = no prisoner = no NOT = { trait = incapable } }
				set_character_flag = attending_druchii_feast
				drufeast_intrigue_roll_effect = yes
				character_event = { id = drufeast.19 days = 31 } # Cleanup
			}
		}
	}
}
