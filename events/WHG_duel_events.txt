###############################################################################
# Duels
# Namespace: duel (was HFP.10095 - HFP.10197, HF.10095 - HF.10107 and
#   HF.16000 - HF.16002 in hfp_duel_events.txt)
# Faction: Generic
# Serves: every duelling race; the warrior societies, greenskin head-bashing,
#   Khornates and Bretonnian knights have their own rules
#
# The Holy Fury duel engine. A challenge (duel.1, or duel.40 for a duel over
# claims) is accepted, fought dirty or refused; the fight itself is scored by
# the effects in 01_scripted_effects_duels.txt (from duel.10), which send one
# of the outcomes below. The scars of a duel may turn into rivalry, friendship
# or love. A craven lord who keeps refusing shames his vassals.
#
# The challenge                    duel.1 - duel.6
# The fight                        duel.10
# Death                            duel.11 - duel.14
# Wounds and flight                duel.20 - duel.26
# After the fight                  duel.30 - duel.39
# Claim duels                      duel.40 - duel.45
# The craven liege                 duel.50 - duel.52
# Housekeeping                     duel.90
#
# Callers: duel.1 from the duel decisions (decisions/hf_warrior_lodge_,
#   wh_greenskin_decisions) and many events and effects; duel.10 from the
#   society, monster-hunt and battle duels; the outcomes from
#   send_duel_result_event_effect; duel.40 from the claim duel decision.
# Saved: event targets combatant_1, combatant_2, other_combatant; flags
#   mock_duel, deadly_duel@, claim_duel@, punish_declining, took_duel_trophy
#   (once per lifetime); modifiers recent_duel_timer,
#   declined_prestige_duel_timer, has_recently_declined_duel_cooldown.
###############################################################################

namespace = duel

# duel.1 A challenge
# The target of a duel hears the challenge: an ordinary duel, a society
# sparring bout, a deadly society duel, a regent's rival or a greenskin
# head-bashing. He may accept, fight dirty (intrigue 12) or refuse. Refusing
# when it counts costs prestige and may make him craven, and a lord of Khorne
# loses piety for it. Warrior priests of Sigmar, Ulric, Myrmidia and Solkan
# accept gladly; only priests of Shallya and Morr never fight. Khornates and
# Bretonnian knights almost never refuse.
# Scope: ROOT is the challenged (combatant_2), combatant_1 the challenger.
#   Fired from the duel decisions and most duel callers in the mod.
character_event = {
	id = duel.1
	border = GFX_event_normal_frame_war

	picture = {
		picture = GFX_evt_armory
		trigger = {
			NOR = { trait = creature_orc trait = creature_goblin trait = creature_hobgoblin }
		}
	}
	picture = {
		picture = GFX_evt_orcthrone
		trigger = {
			OR = { trait = creature_orc trait = creature_goblin trait = creature_hobgoblin }
		}
	}
	desc = {
		trigger = {
			event_target:combatant_1 = { has_character_flag = greenskin_bash_head_duel } #set in Greenskin Decisions
		}
		text = EVTDESCduel1_bash_duel
	}
	desc = {
		trigger = {
			event_target:combatant_1 = {
				NOR = {
					has_character_flag = mock_duel
					has_character_flag = greenskin_bash_head_duel
					has_character_flag = deadly_duel@ROOT
					has_character_flag = flag_courtier_challenging_regent
				}
			}
		}
		text = EVTDESCduel1_default
	}
	desc = {
		trigger = {
			event_target:combatant_1 = { has_character_flag = mock_duel } #set in lodge.5
		}
		text = EVTDESCduel1_mock_duel
	}
	desc = {
		trigger = {
			event_target:combatant_1 = { has_character_flag = deadly_duel@ROOT } #set in lodge.226
		}
		text = EVTDESCduel1_deadly_duel
	}
	desc = {
		trigger = {
			event_target:combatant_1 = { has_character_flag = flag_courtier_challenging_regent } #set in regency_events
		}
		text = EVTDESCduel1_courtier_regent_duel
	}

	is_triggered_only = yes

	immediate = {
		if = {
			limit = { event_target:combatant_1 = { has_character_flag = mock_duel } }
			set_character_flag = mock_duel #makes sure outcome results works correctly for both parties...
		}
		if = {
			limit = {
				AND = {
					NOT = { has_character_modifier = declined_prestige_duel_timer }
					NOT = {
						AND = {
							is_female = no
							event_target:combatant_1 = { is_woman_trigger = yes }
							top_liege = { is_nomadic = no }
						}
					}
				}
			}
			add_character_modifier = {
				modifier = declined_prestige_duel_timer
				duration = 1095
				hidden = yes
			}
			set_character_flag = punish_declining
		}
		if = {
			limit = { NOT = { has_character_modifier = recent_duel_timer } }
			add_character_modifier = { modifier = recent_duel_timer hidden = yes months = 3 }
		}
		event_target:combatant_1 = { #combatant_1/attacker
			if = {
				limit = { NOT = { has_character_modifier = recent_duel_timer } }
				#special circumstances, like initiation duel, etc
				add_character_modifier = { #OUTCOMMENT FOR TESTING PURPOSES
					modifier = recent_duel_timer
					hidden = yes
					months = 3
				}
			}
		}

		if = { #If the challenger did this for a mission...
			limit = {
				event_target:combatant_1 = {
					OR = {
						has_quest = quest_warband_duel_honor
						has_quest = quest_chaos_cult_duel_honor
					}
					quest_target = { character = event_target:combatant_2 }
				}
			}
			event_target:combatant_1 = { set_character_flag = mock_duel }
			set_character_flag = mock_duel #makes sure outcome results works correctly for both parties...
		}
	}

	option = { #Challenge accepted
		name = EVTOPTAduel1
		clr_character_flag = flag_duel_no_backsies
		ai_chance = {
			factor = 50
			modifier = { factor = 2 trait = proud }
			modifier = { factor = 0.05 trait = craven }
			modifier = { factor = 0.3 religion = shallyan }
			modifier = {
				is_priest = yes
				OR = { religion = shallyan religion = morrian }
				factor = 0
			}
			modifier = {
				is_priest = yes
				OR = {
					religion = sigmarite
					religion = ulrican
					religion = myrmidian
					religion = solkanite
				}
				factor = 1.5
			}
			modifier = { religion = khorne factor = 3 }
			modifier = {
				OR = {
					trait = bretonnian_knight
					trait = bretonnian_knight_errant
					trait = questing
					trait = grail_knight
				}
				factor = 2
			}
			modifier = {
				is_female = no
				event_target:combatant_1 = { is_woman_trigger = yes }
				top_liege = { is_nomadic = no }
				factor = 0.025
			}
		}
		if = { limit = { trait = craven } random = { chance = 20 remove_trait = craven } }
		pacifists_lose_piety_effect = yes
		if = { #If they previously offended you, this rectifies that.
			limit = {
				event_target:combatant_1 = {
					has_opinion_modifier = { who = ROOT modifier = grievously_offended }
				}
			}
			event_target:combatant_1 = {
				remove_opinion = { who = ROOT modifier = grievously_offended }
			}
		}
		if = {
			limit = { #Warriors like a warrior who acts like one...
				OR = {
					is_member_of_any_warband_trigger = yes
					is_member_of_any_warrior_lodge_trigger = yes
				}
			}
			add_legend_progress_trivial_effect = yes
		}
		event_target:combatant_1 = { character_event = { id = duel.2 days = 3 } }
		set_character_flag = add_duel_xp
		clr_character_flag = punish_declining
	}

	option = { # [INTRIGUE OPTION]
		name = {
			text = EVTOPTduel_intrigue #I will do something.... special.
		}
		tooltip_info = intrigue

		custom_tooltip = { text = EVTOPTduel_intrigue_TT } #you will use any means necessary...

		set_character_flag = defender_using_poison@event_target:combatant_1

		if = { limit = { trait = honest } remove_trait = honest }
		else_if = { limit = { NOT = { trait = deceitful } } add_trait = deceitful }
		else_if = { limit = { trait = brave } remove_trait = brave }
		else_if = { limit = { NOT = { trait = craven } } add_trait = craven }

		if = { #There is a chance the poison fails...
			limit = {
				OR = {
					event_target:combatant_1 = {
						job_spymaster = {
							opinion = { who = PREV value = 15 } #if they like their employer
							intrigue = ROOT #and they can match your intrigue
						}
					}
					event_target:combatant_1 = { intrigue = ROOT } #they alone can match your intrigue
				}
			}
			set_character_flag = poison_can_fail_against@event_target:combatant_1
		}

		pacifists_lose_piety_effect = yes
		if = { #If they previously offended you, this rectifies that.
			limit = {
				event_target:combatant_1 = {
					has_opinion_modifier = { who = ROOT modifier = grievously_offended }
				}
			}
			event_target:combatant_1 = {
				remove_opinion = { who = ROOT modifier = grievously_offended }
			}
		}
		event_target:combatant_1 = { character_event = { id = duel.2 days = 3 } }
		set_character_flag = add_duel_xp
		clr_character_flag = punish_declining

		ai_chance = { factor = 20 dishonorable_ai_chance_increase_score = yes }

		trigger = { intrigue = 12 NOT = { has_character_flag = mock_duel } has_dlc = "Holy Fury" }
	}

	option = { #No, that sounds dangerous
		name = EVTOPTBduel1
		ai_chance = {
			factor = 10
			modifier = { trait = craven factor = 1.5 }
			modifier = {
				is_priest = yes
				NOR = {
					religion = sigmarite
					religion = ulrican
					religion = myrmidian
					religion = solkanite
					religion = khorne
				}
				factor = 5
			}
			modifier = { religion = khorne factor = 0 }
			modifier = {
				OR = {
					trait = bretonnian_knight
					trait = bretonnian_knight_errant
					trait = questing
					trait = grail_knight
				}
				factor = 0.1
			}
			modifier = { event_target:combatant_1 = { has_character_flag = mock_duel } factor = 0 }
			modifier = { trait = green_knight factor = 0 }
			modifier = {
				event_target:combatant_1 = {
					OR = {
						has_quest = quest_warband_duel_honor
						has_quest = quest_warband_duel_deadly
						has_quest = quest_chaos_cult_duel_honor
						has_quest = quest_chaos_cult_duel_deadly
					}
					quest_target = { character = event_target:combatant_2 }
				}
				factor = 0
			}
		}
		if = { #Khorne despises the lord who will not shed blood
			limit = { religion = khorne }
			piety = -50
		}
		if = {
			limit = { has_character_flag = punish_declining }
			if = {
				limit = { NOR = { trait = craven trait = brave trait = berserker } }
				random = {
					chance = 30
					add_trait = craven
					hidden_tooltip = {
						character_event = {
							id = traitnote.257 #Notify Craven
						}
					}
				}
			}
			if = { limit = { trait = brave } remove_trait = brave }
			prestige = -50
			add_character_modifier = {
				modifier = declined_prestige_duel_timer
				hidden = yes
				duration = 1095
			}
		}
		hidden_tooltip = { remove_character_modifier = recent_duel_timer }
		event_target:combatant_1 = { #the attacker/combatant_1
			hidden_tooltip = {
				remove_character_modifier = recent_duel_timer
				if = {
					limit = { NOT = { has_character_modifier = duels_declined } }
					opinion = { modifier = opinion_fled_duel who = ROOT years = 100 }
				}
				else = {
					opinion = {
						modifier = opinion_declined_my_duel #For visuals
						who = ROOT
						years = 100
					}
				}
				character_event = { id = duel.3 days = 3 } #Duel rejected...
			}
		}
		increase_cowardly_status_effect = yes #tracks duel cowardice
		clr_character_flag = punish_declining

		if = { #For characters who were recruited by their parents...
			limit = {
				has_character_flag = is_being_recruited_to_warband_by_parent
				event_target:recruiting_parent = { is_alive = yes }
			}
			custom_tooltip = { text = notify_parent_recruiter_tt } #Your parent will be notified.
			set_character_flag = failed_initiation
			set_character_flag = chickened_out
			event_target:recruiting_parent = { character_event = { id = lodge.13 } }
			clr_character_flag = society_join_block
		}

		trigger = {
			NOT = { event_target:duel_target = { character = event_target:hero_character } } #if set-up failed spectacularly, and it's a case of MP, the player leader will be unable to turn down the duel just for the hell of it
			NOT = { has_character_flag = flag_duel_no_backsies }
		}
	}
}

	is_triggered_only = yes
	border = GFX_event_normal_frame_war

	picture = {
		picture = GFX_evt_armory
		trigger = {
			NOR = { trait = creature_orc trait = creature_goblin trait = creature_hobgoblin }
		}
	}
	picture = {
		picture = GFX_evt_greenskinanimosity
		trigger = {
			OR = { trait = creature_orc trait = creature_goblin trait = creature_hobgoblin }

			}
		}
	}

		picture = GFX_evt_orcthrone
		trigger = {
			OR = { trait = creature_orc trait = creature_goblin trait = creature_hobgoblin }

	}

		picture = GFX_evt_teaching_moment
		text = EVTDESCduel2_mock_duel_spouse #Initation duel against spouse

	desc = {
		trigger = {
			has_character_flag = mock_duel #set in lodge.5
			NOT = { any_spouse = { character = event_target:combatant_2 } }
			NOT = { has_quest = quest_warband_duel_honor }
			NOT = { has_quest = quest_chaos_cult_duel_honor }
			any_owned_bloodline = { has_bloodline_flag = bloodline_legendary_warrior }
		}

		text = EVTDESCduel2_bloodline #Initiation duel as special bloodline owner
	}

		name = {
			text = EVTOPTAduel2 #Let's fight!
			trigger = {
				NOT = { has_character_flag = mock_duel }
				NOT = { has_character_flag = target_was_threatened }
			}
		}
		name = {
			text = EVTOPTAduel2_mock # A [GetDuelType], huh?
			trigger = {
				has_character_flag = mock_duel
				NOT = { has_character_flag = target_was_threatened }
			}
		}
		name = {
			text = EVTOPTBduel2 # That's what I thought...
			trigger = { has_character_flag = target_was_threatened }
		}
		if = {
			limit = { has_character_flag = mock_duel }
			custom_tooltip = { text = EVTOPTAduel2_TT } #the duel will take place within the next few days.

			limit = { #Warriors like a warrior who acts like one...

					is_member_of_any_warrior_lodge_trigger = yes

			add_legend_progress_trivial_effect = yes
		}

			character_event = { id = duel.6 }
		}

			hidden_effect = {
				set_character_flag = add_duel_xp
				reset_duel_timers_effect = yes

	}

		name = {
			text = EVTOPTduel_intrigue #I will do something.... special.

		if = { limit = { trait = honest } remove_trait = honest }

		else_if = { limit = { trait = brave } remove_trait = brave }
		else_if = { limit = { NOT = { trait = craven } } add_trait = craven }

		if = { #There is a chance the poison fails...
			limit = {
				OR = {
					FROM = {
						job_spymaster = {
							opinion = { who = PREV value = 15 } #if they like their employer
							intrigue = ROOT #and they can match your intrigue
						}
					}
					FROM = { intrigue = ROOT } #they alone can match your intrigue
				}
			}
			set_character_flag = poison_can_fail_against@event_target:combatant_2
		}

		#FALLBACK:
		if = { #If FROM is no longer alive, clear relevant flags, etc...
			limit = { FROM = { NOT = { is_alive = yes } } }
			character_event = { id = duel.6 }
		}
		else = {
			hidden_effect = {
				set_character_flag = add_duel_xp
				reset_duel_timers_effect = yes
				character_event = { id = duel.10 } #Send hidden set-up of duel (from there: result)
			}
		}
		ai_chance = { factor = 20 dishonorable_ai_chance_increase_score = yes }
		trigger = { intrigue = 12 NOT = { has_character_flag = mock_duel } has_dlc = "Holy Fury" }
	}

# duel.3 The challenge is refused
# The challenger hears the refusal. A schemer (intrigue 15, the Elusive
# Shadow or a schemer) may try to corner the coward; if it works he must
# fight (duel.4), if not the schemer is caught (duel.5).
# Scope: ROOT is the challenger, FROM the challenged. Fired by duel.1 and
#   duel.40.
character_event = {
	id = duel.3
	is_triggered_only = yes
	desc = EVTDESCduel3
	picture = GFX_evt_bad_news
	border = GFX_event_normal_frame_war
	title = getDuelTitle

	immediate = {
		# To block spamming the decision when rejected...
		FROM = {
			add_character_modifier = {
				name = has_recently_declined_duel_cooldown
				hidden = yes
				months = 6
			}
		}
		if = { #only if intrigue character...
			limit = {
				NOT = { has_character_flag = mock_duel }
				OR = { intrigue = 15 trait = elusive_shadow trait = schemer }
			}
			random_list = {
				20 = { #success
					modifier = {
						has_bottom_tier_education_trait_trigger = no
						FROM = { has_bottom_tier_education_trait_trigger = yes }
						factor = 2
					}
					modifier = {
						trait = elusive_shadow
						FROM = { NOT = { trait = elusive_shadow } }
						factor = 2
					}
					modifier = { intrigue = 15 FROM = { NOT = { intrigue = 15 } } factor = 2 }
					modifier = { trait = schemer FROM = { NOT = { trait = schemer } } factor = 2 }
					modifier = { intrigue = FROM factor = 2 }
					modifier = {
						is_smart_trigger = yes
						FROM = { is_smart_trigger = no }
						factor = 2
					}

					set_character_flag = likely_to_succeed
				}
				10 = { #failure
					modifier = {
						has_bottom_tier_education_trait_trigger = yes
						FROM = { has_bottom_tier_education_trait_trigger = no }
						factor = 2
					}
					modifier = {
						NOT = { trait = elusive_shadow }
						FROM = { trait = elusive_shadow }
						factor = 2
					}
					modifier = { NOT = { intrigue = 15 } FROM = { intrigue = 15 } factor = 2 }
					modifier = { NOT = { trait = schemer } FROM = { trait = schemer } factor = 2 }
					modifier = { NOT = { intrigue = FROM } factor = 2 }
					modifier = { is_dumb_trigger = yes FROM = { is_dumb_trigger = no } factor = 2 }
					set_character_flag = likely_to_fail
				}
			}
		}
	}

	option = { #"Corner"-option
		name = EVTOPTAduel3 # You won't get away that easily! (Only for INTRIGUE characters)
		tooltip_info = intrigue
		custom_tooltip = { text = EVTOPTAduel3_TT }

		random_list = {
			10 = { #success! (sends event to target)
				additive_modifier = { value = 60 has_character_flag = likely_to_succeed }
				custom_tooltip = { text = you_succeeding_to_corner_tt }
				hidden_effect = { FROM = { character_event = { id = duel.4 } } }
			}
			20 = { #failure... (sends event to self)
				additive_modifier = { value = 60 has_character_flag = likely_to_fail }
				custom_tooltip = { text = you_failing_to_corner_tt }
				hidden_effect = { character_event = { id = duel.5 } }
			}
		}

# duel.50 Like liege, like vassal
# A vassal hears that his craven liege keeps refusing duels: be craven
# too (duel.51) or be ashamed of him (duel.52).
# Scope: ROOT is the vassal, FROM the craven liege. Fired by
#   spread_cowardice_trait_effect.
character_event = {
	id = duel.50
	picture = GFX_evt_camaraderie
	border = GFX_event_normal_frame_war
	is_triggered_only = yes

	portrait = event_target:craven_ruler

	trigger = {
		NOT = { trait = craven }
		is_inaccessible_or_incapable_trigger = no
		FROM = {
			is_alive = yes
			has_character_modifier = duels_declined
			trait = craven
			is_incapable = no
		}
		NAND = { #one part should be AI, for ease of choices
			ai = no
			FROM = { ai = no }
		}
	}

	immediate = {
		FROM = { save_event_target_as = craven_ruler }
	}

	desc = { text = EVTDESCduel50_a trigger = { event_target:craven_ruler = { dynasty = ROOT } } }
	desc = {
		text = EVTDESCduel50_b
		trigger = { event_target:craven_ruler = { NOT = { dynasty = ROOT } } }
	}

	option = { # Yes, you can't be too careful
		name = EVTOPTAduel50
		if = {
			limit = { ai = no } #otherwise, liege is player, and traits are added later
			if = { limit = { trait = brave } remove_trait = brave }
			else_if = { limit = { NOT = { trait = craven } } add_trait = craven }
		}

		event_target:craven_ruler = { character_event = { id = duel.51 days = 1 } }

		ai_chance = {
			factor = 50
			modifier = { factor = 1.5 trait = trusting }
			modifier = { factor = 1.5 trait = weak }
			modifier = { factor = 1.5 trait = feeble }
			modifier = { factor = 1.5 opinion = { who = event_target:craven_ruler value = 50 } }
			modifier = { factor = 1.5 opinion = { who = event_target:craven_ruler value = 70 } }
			modifier = { factor = 1.5 opinion = { who = event_target:craven_ruler value = 80 } }
			modifier = { factor = 1.5 is_child_of = event_target:craven_ruler }
		}
	}
	option = { # No, I am ashamed of my liege
		name = EVTOPTBduel50

		tooltip = { #added for real in follow up event
			opinion = {
				modifier = opinion_renowned_chicken
				who = event_target:craven_ruler
				years = 20
			}
			reverse_opinion = {
				modifier = opinion_disrespectful_subject
				who = event_target:craven_ruler
				years = 20
			}
		}
		event_target:craven_ruler = { character_event = { id = duel.52 days = 1 } }

		ai_chance = {
			factor = 50
			modifier = { factor = 1.5 trait = wroth }
			modifier = { factor = 1.5 trait = ambitious }
			modifier = { factor = 1.5 trait = strong }
			modifier = { factor = 1.5 trait = robust }
			modifier = {
				factor = 1.5
				NOT = { opinion = { who = event_target:craven_ruler value = 25 } }
			}
			modifier = {
				factor = 1.5
				NOT = { opinion = { who = event_target:craven_ruler value = 0 } }
			}
			modifier = {
				factor = 1.5
				NOT = { opinion = { who = event_target:craven_ruler value = -25 } }
			}
		}
	}
}
