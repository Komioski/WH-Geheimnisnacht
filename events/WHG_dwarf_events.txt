# Dawi events
# Namespace: dawi (was DR_dwarfs, GR_dwarf, GR_dwarf_reunification and
# SJDwarfevents in four files)
# Faction: Dwarfs
#
# The life of the Dawi of the Karaz Ankor: the Book of Grudges and the
# settling of grudges, the Slayers who seek a death to wipe out their shame,
# the warriors of the hold, delving the old tunnels, the Council of Kings
# that names a High King, and the daily life of a hold: ale and brewmasters,
# masons, the tavern, beardlings and the warrior paths.
#
# Lore: every Dwarf king keeps a Dammaz Kron, a Book of Grudges, for every
# wrong done to his hold or clan; a grudge is struck out only when amends
# are made or vengeance taken, and an unsettled grudge passes to the heirs.
# The greenskins, the Skaven and the Chaos Dwarfs are already written down as
# whole races in the High King's Great Book. A Dawi who dishonours himself
# goes to the temple of Grimnir for the Slayer ritual (Drengidumar): his head
# shaved to an orange crest, his shame tattooed on his skin, and no more
# armour, wealth or clan until he finds a death worthy of Grimnir. Karak
# Kadrin, home of the Shrine of Grimnir, is ruled by the Slayer Kings. An
# oathbreaker who will not take the Slayer's path is Unbaraki, the worst
# thing a Dwarf can be.
# Sources: WFRP 4E Dwarf Player's Guide (Book of Grudges p.32-33, Slayers in
# Karaz Ankor p.23, Dwarf Slayer p.59, Grimnir p.139, Khazalid p.70-73,
# coming of age p.31, food and ale p.35, crafts and stonemasons p.57,
# Brewer p.62, Hammerer p.68, Ironbreaker p.70, the Day of Grudgement
# p.84, Reckoners p.57, Dwarf-friends p.32).
#

# Unsettled grudges, death and inheritance dawi.30 - dawi.35
# The Slayer's path                        dawi.40 - dawi.43

# Grudges
# A grudge is an opinion modifier for its kind of wrong (wh_dawi_triggers.txt),
# held by the wronged Dawi and the kin who take it up. Ordinary grudges are
# settled by restitution, Blood Grudges only by death or the Slayer oath.
# dawi_enter_<kind>_grudge_effect writes one; dawi_settle_grudge_effect strikes
# out the ordinary ones and dawi_settle_blood_grudge_effect all of them. All
# read the offender from
# event_target:grudge_offender.
###############################################################################

# dawi.1 The clan's book is read to a new adult
# A young Dawi of a clan takes up the grudges his parents hold.
# Scope: ROOT is the new adult.
character_event = {
	id = dawi.1
	desc = EVTDESCdawi1
	picture = GFX_evt_dwarf_grudge_book
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_adulthood
	religion_group = dwarf_gods_group

	trigger = {
		dawi_grudge_keeper_trigger = yes
		OR = {
			father_even_if_dead = {
				dynasty = ROOT
				any_opinion_modifier_target = { is_alive = yes dawi_grudge_by_prev_trigger = yes }
			}
			mother_even_if_dead = {
				dynasty = ROOT
				any_opinion_modifier_target = { is_alive = yes dawi_grudge_by_prev_trigger = yes }
			}
		}
	}

	option = {
		name = EVTOPTAdawi1
		custom_tooltip = {
			text = dawi_inherit_grudges_tt
			hidden_effect = {
				father_even_if_dead = {
					limit = { dynasty = ROOT }
					any_opinion_modifier_target = {
						limit = {
							is_alive = yes
							NOT = { character = ROOT }
							dawi_grudge_by_prev_trigger = yes
						}
						dawi_inherit_grudges_effect = yes
					}
				}
				mother_even_if_dead = {
					limit = { dynasty = ROOT }
					any_opinion_modifier_target = {
						limit = {
							is_alive = yes
							NOT = { character = ROOT }
							dawi_grudge_by_prev_trigger = yes
						}
						dawi_inherit_grudges_effect = yes
					}
				}
			}
		}
	}
}

# dawi.6 The grudge war is won
# The winner of a settle_grudge war takes amends from event_target:
# grudge_offender (the enemy ruler or one of his vassals): his treasury, his
# kin as hostages, or his head. Gold or hostages strike out an ordinary
# grudge; a Blood Grudge takes his head or, from a Dawi, the Slayer oath
# (Dwarf Player's Guide p.86).
# Scope: ROOT is the Dawi who won the war.
character_event = {
	id = dawi.6
	desc = EVTDESCdawi6
	picture = GFX_evt_dwarf_grudge_war
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # settle_grudge CB on_success

	trigger = {
		event_target:grudge_offender = { is_alive = yes }
	}

	option = { # Empty his treasury
		name = EVTOPTAdawi6
		trigger = { event_target:grudge_offender = { dawi_blood_grudge_against_trigger = no } }
		event_target:grudge_offender = { transfer_scaled_wealth = { to = ROOT value = 4 } }
		dawi_settle_grudge_effect = yes
		ai_chance = {
			factor = 50
			modifier = { factor = 3 trait = greedy }
			modifier = { factor = 2 trait = content }
			modifier = { factor = 2 trait = kind }
		}
	}
	option = { # Take his kin as hostages
		name = EVTOPTBdawi6
		trigger = { event_target:grudge_offender = { dawi_blood_grudge_against_trigger = no } }
		event_target:grudge_offender = {
			any_courtier = { limit = { dynasty = event_target:grudge_offender } imprison = ROOT }
		}
		dawi_settle_grudge_effect = yes
		ai_chance = {
			factor = 30
			modifier = { factor = 3 trait = patient }
			modifier = { factor = 2 trait = deceitful }
		}
	}
	option = { # The Slayer oath, for a Dawi with a Blood Grudge on him
		name = EVTOPTDdawi6
		trigger = {
			event_target:grudge_offender = {
				dawi_blood_grudge_against_trigger = yes
				trait = creature_dwarf
				religion_group = dwarf_gods_group
				NOR = { trait = dwarf_slayer trait = dwarf_slayer_king }
			}
		}
		dawi_settle_blood_grudge_effect = yes
		event_target:grudge_offender = { dawi_take_slayer_oath_effect = yes }
		ai_chance = {
			factor = 20
			modifier = { factor = 3 trait = kind }
			modifier = { factor = 2 trait = zealous }
			modifier = { factor = 2 dynasty = event_target:grudge_offender }
		}
	}
	option = { # His head
		name = EVTOPTCdawi6
		dawi_settle_blood_grudge_effect = yes
		event_target:grudge_offender = {
			if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
			else = { death = { death_reason = death_execution killer = ROOT } }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 4 trait = wroth }
			modifier = { factor = 2 trait = cruel }
			modifier = { factor = 2 trait = proud }
			modifier = { factor = 0.2 trait = kind }
		}
	}
}

# dawi.13 Oathgold demanded
# The divorcer pays (more the higher his rank) or refuses.
# Scope: ROOT is the divorcer, FROM the divorced Dawi.
character_event = {
	id = dawi.13
	desc = EVTDESCdawi13
	picture = GFX_evt_dwarf_angry_look
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes # dawi.12

	option = { # Pay
		name = EVTOPTAdawi13
		trigger = {
			OR = {
				AND = { is_ruler = no wealth = 50 }
				AND = { tier = BARON wealth = 100 }
				AND = { tier = COUNT wealth = 250 }
				AND = { tier = DUKE wealth = 500 }
				AND = { tier = KING wealth = 750 }
				AND = { tier = EMPEROR wealth = 1000 }
			}
		}
		if = { limit = { is_ruler = no } wealth = -50 FROM = { wealth = 50 } }
		else_if = { limit = { tier = BARON } wealth = -100 FROM = { wealth = 100 } }
		else_if = { limit = { tier = COUNT } wealth = -250 FROM = { wealth = 250 } }
		else_if = { limit = { tier = DUKE } wealth = -500 FROM = { wealth = 500 } }
		else_if = { limit = { tier = KING } wealth = -750 FROM = { wealth = 750 } }
		else = { wealth = -1000 FROM = { wealth = 1000 } }
		FROM = { character_event = { id = dawi.14 } }
		ai_chance = {
			factor = 55
			modifier = { factor = 1.5 trait = kind }
			modifier = { factor = 1.5 trait = charitable }
			modifier = { factor = 1.5 trait = humble }
			modifier = { factor = 1.5 trait = craven }
			modifier = { factor = 2 trait = creature_dwarf }
			modifier = { factor = 0.5 trait = greedy }
			modifier = { factor = 0.5 trait = proud }
			modifier = { factor = 0.5 is_rival = FROM }
			modifier = { factor = 1.5 opinion = { who = FROM value = 50 } }
		}
	}
	option = { # Refuse
		name = EVTOPTBdawi13
		custom_tooltip = { text = dawi_oathgold_refused_tt }
		FROM = { character_event = { id = dawi.15 } }
		ai_chance = {
			factor = 45
			modifier = { factor = 1.5 trait = proud }
			modifier = { factor = 1.5 trait = stubborn }
			modifier = { factor = 1.5 trait = greedy }
			modifier = { factor = 1.5 trait = ambitious }
		}
	}
}

# Demanding compensation
# A Dawi ruler may send a Reckoner to the offender to demand amends for a
# grudge (grudge_demand_compensation decision, or dawi.25 for the AI). What
# he may ask depends on the grudge (Dwarf Player's Guide p.86): a public
# apology for a slight; gold, or a hold from a Dawi, for any ordinary grudge;
# for a Blood Grudge only the offender's head, or a Dawi offender's Slayer
# oath. Amends strike out the grudge; a refusal leaves it in the book.
###############################################################################

# dawi.20 Sending the Reckoner
# Scope: ROOT is the Dawi ruler, event_target:grudge_offender the offender.

# dawi.30 The weight of the book
# A Dawi ruler's unsettled grudges weigh on his honour and on the regard of
# other Dawi: unfulfilled_grudges_1 - 5, by how many living foes the book
# names. An open Blood Grudge weighs one step more.

	id = dawi.30
	hide_window = yes

	only_rulers = yes

		trait = creature_dwarf
	}

		remove_character_modifier = unfulfilled_grudges_1
		remove_character_modifier = unfulfilled_grudges_2
		remove_character_modifier = unfulfilled_grudges_3
		remove_character_modifier = unfulfilled_grudges_4
		remove_character_modifier = unfulfilled_grudges_5
		if = {
			limit = {
				OR = {
					any_opinion_modifier_target = {
						count = 5
						is_alive = yes
						dawi_grudge_against_trigger = yes
					}
					AND = {
						any_opinion_modifier_target = {
							is_alive = yes
							dawi_blood_grudge_against_trigger = yes
						}
						any_opinion_modifier_target = {
							count = 4
							is_alive = yes
							dawi_grudge_against_trigger = yes
						}
					}
				}
			}
			add_character_modifier = { name = unfulfilled_grudges_5 duration = -1 }
		}
		else_if = {
			limit = {

					any_opinion_modifier_target = {

						is_alive = yes
						dawi_grudge_against_trigger = yes
					}

							dawi_blood_grudge_against_trigger = yes

						is_alive = yes
						dawi_grudge_against_trigger = yes
					}
					AND = {

# dawi.31 An offender dies (hidden)
# A character named in a Book of Grudges has died. For every Dawi ruler
# whose book names him: if the clan or its kin killed him, the grudge is
# avenged (dawi.32); otherwise it passes to his heir (dawi.33), a Blood Grudge
# as a war grudge (Dwarf Player's Guide p.85).
# Scope: ROOT is the dead offender.
character_event = {
	id = dawi.31
	hide_window = yes

	is_triggered_only = yes # on_death

	trigger = {
		has_character_flag = dawi_named_in_book
	}

	immediate = {
		if = {
			limit = { current_heir = { is_alive = yes dawi_race_grudge_trigger = no } }
			current_heir = { save_event_target_as = dawi_grudge_heir }
		}
		any_ruler = {
			limit = {
				trait = creature_dwarf
				religion_group = dwarf_gods_group
				dawi_holds_grudge_against_root_trigger = yes
			}
			if = {
				limit = { ROOT = { killer = { OR = { character = PREVPREV dynasty = PREVPREV } } } }
				if = {
					limit = { dawi_holds_blood_grudge_against_root_trigger = yes }
					set_character_flag = dawi_avenged_blood
				}
				character_event = { id = dawi.32 }
			}
			else_if = {
				limit = {
					event_target:dawi_grudge_heir = {
						NOT = { character = PREV }
						NOT = { dynasty = PREV }
					}
				}
				dawi_pass_grudges_to_heir_effect = yes
				character_event = { id = dawi.33 }
			}
		}
	}
}

# dawi.34 A Dawi dies with his book open (hidden)
# A Dawi ruler has died. His heir of the clan takes up every grudge in his
# book, and if the book still names living foes, the dead sits far from the
# ancestors' table and the heir carries the shame (dawi.35) until he settles
# one of them (Dwarf Player's Guide p.86).
# Scope: ROOT is the dead Dawi ruler.
character_event = {
	id = dawi.34
	hide_window = yes

	is_triggered_only = yes # on_death
	religion_group = dwarf_gods_group

	trigger = {
		dawi_grudge_keeper_trigger = yes
		is_ruler = yes
		current_heir = { is_alive = yes dynasty = ROOT trait = creature_dwarf }
		any_opinion_modifier_target = { is_alive = yes dawi_grudge_against_trigger = yes }
	}

	immediate = {
		current_heir = { save_event_target_as = dawi_book_heir }
		any_opinion_modifier_target = {
			limit = {
				is_alive = yes
				NOT = { character = event_target:dawi_book_heir }
				dawi_grudge_against_trigger = yes
			}
			dawi_bequeath_grudges_effect = yes
			set_character_flag = dawi_named_in_book
		}
		event_target:dawi_book_heir = {
			remove_character_modifier = dawi_far_from_ancestors_1
			remove_character_modifier = dawi_far_from_ancestors_2
			remove_character_modifier = dawi_far_from_ancestors_3
			if = {
				limit = {
					ROOT = {
						OR = {
							any_opinion_modifier_target = {
								count = 5
								is_alive = yes
								dawi_grudge_against_trigger = yes
							}
							any_opinion_modifier_target = {
								is_alive = yes
								dawi_blood_grudge_against_trigger = yes
							}
						}
					}
				}
				add_character_modifier = { name = dawi_far_from_ancestors_3 duration = 3650 }
			}
			else_if = {
				limit = {
					ROOT = {
						any_opinion_modifier_target = {
							count = 3
							is_alive = yes
							dawi_grudge_against_trigger = yes
						}
					}
				}
				add_character_modifier = { name = dawi_far_from_ancestors_2 duration = 3650 }
			}
			else = { add_character_modifier = { name = dawi_far_from_ancestors_1 duration = 3650 } }
			character_event = { id = dawi.35 }
		}
	}
}

# dawi_grudge_remembered_effect) and shamed for Blood Grudges he still
# leaves open. One grudge each Day may go into the Great Book itself, held
# by every Dawi ruler (dawi_great_book_effect, also used when Karag Dum
# falls in WHG_chaos_horde_events).

# Underway events
# Namespace: underway (was ungdrin)
# Faction: Dwarfs
#
# The Ungdrin Ankor, the Great Underway, was dug in the Golden Age to join
# every hold of the Karaz Ankor beneath the mountains. Earthquakes shattered
# it, burying whole stretches under rubble, and the Skaven and the Night
# Goblins took much of what was left. A hold that clears and shores up its
# stretch again opens it with a wonder upgrade (upgrade_underway_<hold>):
# its counties get the ungdrin_ankor movement modifier. A Train Station
# (upgrade_trains) lays rails for mining carriages through the hold's open
# stretch (ungdrin_trains). The stretch closes when the upgrade is destroyed
# or the wonder falls to non-Dawi, and opens again when the Dawi retake it.
# Holds on an open stretch also live with what comes up the tunnels.
# Lore: WFRP 4E Dwarf Player's Guide (the Underway built p.10, Ironbreakers
# p.70, the Underway shattered and Mining Carriages p.107).
#
# Opening and closing the Underway          underway.1 - underway.6
# Life on the Underway                      underway.20 - underway.24
#
# Callers:
#   on_actions (00_on_actions.txt): underway.1 on_wonder_upgrade_finish,
#     underway.4 on_wonder_upgrade_destroyed, underway.6
#     on_wonder_owner_change.
#   on_actions (wh_on_actions.txt): the underway_tunnels bi-yearly random
#     list (underway.20 - underway.24).
# Scripted triggers: common/scripted_triggers/wh_underway_triggers.txt
# Scripted effects: common/scripted_effects/wh_underway_effects.txt
# Loc: localisation/00_WHG_dwarf_events.csv
###############################################################################

namespace = underway

###############################################################################
# Opening and closing the Underway
# The segment of each upgrade, and which counties are lowland, are listed in
# wh_underway_effects.txt.
###############################################################################

# underway.1 An upgrade is finished
# An Underway upgrade opens its stretch (with rails, if the wonder has a
# Train Station); a Train Station lays rails through the open stretches.
# Scope: ROOT is the owner, FROM the upgrade, FROMFROM the wonder.
character_event = {
	id = underway.1
	hide_window = yes
	is_triggered_only = yes # on_wonder_upgrade_finish

	trigger = {
		FROM = { OR = { has_wonder_upgrade = upgrade_trains underway_segment_trigger = yes } }
	}

	immediate = {
		if = {
			limit = { FROM = { underway_segment_trigger = yes } }
			FROM = { underway_open_segment_effect = yes }
			if = {
				limit = { FROMFROM = { has_wonder_upgrade = upgrade_trains } }
				FROM = { underway_carriages_on_effect = yes }
			}
			character_event = { id = underway.2 }
		}
		else = {
			FROMFROM = { underway_carriages_on_effect = yes }
			if = {
				limit = { FROMFROM = { underway_segment_trigger = yes } }
				character_event = { id = underway.3 }
			}
		}
	}
}
