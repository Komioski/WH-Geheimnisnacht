###############################################################################
# Dawi events
# Namespace: dawi (was DR_dwarfs in DR_dwarfs_events.txt)
# Faction: Dwarfs
#
# The life of the Dawi of the Karaz Ankor: the Book of Grudges and the
# settling of grudges, and the Slayers who seek a death to wipe out their
# shame.
#
# Lore: every Dwarf king keeps a Dammaz Kron, a Book of Grudges, for every
# wrong done to his hold or clan; a grudge is struck out only when amends
# are made or vengeance taken, and an unsettled grudge passes to the heirs.
# The greenskins, the Skaven and the Chaos Dwarfs are already written down as
# whole races in the High King's Great Book. A Dawi who dishonours himself
# goes to the temple of Grimnir for the Slayer ritual (Drengidumar): his head
# shaved to an orange crest, his shame tattooed on his skin, and no more
# armour, wealth or clan until he finds a death worthy of Grimnir. Karak
# Kadrin, home of the Shrine of Grimnir, is ruled by the Slayer Kings. An
# oathbreaker who will not take the Slayer's path is Unbaraki, the worst
# thing a Dwarf can be.
# Sources: WFRP 4E Dwarf Player's Guide (Book of Grudges p.32-33, Slayers in
# Karaz Ankor p.23, Dwarf Slayer p.59, Grimnir p.139, Khazalid p.70-73).
#
# Grudges                                  dawi.1 - dawi.16
# Demanding compensation                   dawi.20 - dawi.25
# Unsettled grudges                        dawi.30
# The Slayer's path                        dawi.40 - dawi.43
# Slayers at large                         dawi.50 - dawi.60
#
# Fired from common/on_actions/wh_on_actions.txt: dawi.1 on_adulthood,
# dawi.2 on_war_ended_victory, dawi.4 on_war_ended_defeat, dawi.10 on_death,
# dawi.12 on_divorce, dawi.16 on
# CALL_ALLY_INTERACTION_DECLINE_EVENT, dawi.43 on_new_holder_inheritance, dawi.25 on the
# five-year pulse, dawi.30, dawi.40, dawi.42 and dawi.50 on the bi-yearly
# pulse. dawi.6 from the settle_grudge CB (cb_types_dwarf.txt); dawi.5 and
# dawi.20 from decisions/wh_dawi_decisions.txt; dawi.9 from plot_events,
# job_spymaster and on_action_events.
# Scripted triggers: common/scripted_triggers/wh_dawi_triggers.txt
# Scripted effects: common/scripted_effects/wh_dawi_effects.txt
# Loc: localisation/00_WHG_dwarf_events.csv
###############################################################################

namespace = dawi

# Grudges
# Every grudge is the opinion modifier opinion_dwarf_grudge held by the wronged
# Dawi and the kin who take it up. dawi_enter_grudge_effect writes it,
# dawi_settle_grudge_effect strikes it out; both read the offender from
# event_target:grudge_offender.
###############################################################################

# dawi.1 The clan's book is read to a new adult
# A young Dawi of a clan takes up the grudges his parents hold.
# Scope: ROOT is the new adult.
character_event = {
	id = dawi.1
	desc = EVTDESCdawi1
	picture = GFX_evt_dwarf_grudge_book
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_adulthood
	religion_group = dwarf_gods_group

	trigger = {
		dawi_grudge_keeper_trigger = yes
		OR = {
			father_even_if_dead = {
				dynasty = ROOT
				any_opinion_modifier_target = {
					is_alive = yes
					reverse_has_opinion_modifier = {
						who = PREVPREV
						modifier = opinion_dwarf_grudge
					}
				}
			}
			mother_even_if_dead = {
				dynasty = ROOT
				any_opinion_modifier_target = {
					is_alive = yes
					reverse_has_opinion_modifier = {
						who = PREVPREV
						modifier = opinion_dwarf_grudge
					}
				}
			}
		}
	}

	option = {
		name = EVTOPTAdawi1
		custom_tooltip = {
			text = dawi_inherit_grudges_tt
			hidden_effect = {
				father_even_if_dead = {
					limit = { dynasty = ROOT }
					any_opinion_modifier_target = {
						limit = {
							is_alive = yes
							NOT = { character = ROOT }
							reverse_has_opinion_modifier = {
								who = PREVPREV
								modifier = opinion_dwarf_grudge
							}
						}
						reverse_opinion = {
							who = ROOT
							modifier = opinion_dwarf_grudge
							months = 12000
						}
					}
				}
				mother_even_if_dead = {
					limit = { dynasty = ROOT }
					any_opinion_modifier_target = {
						limit = {
							is_alive = yes
							NOT = { character = ROOT }
							reverse_has_opinion_modifier = {
								who = PREVPREV
								modifier = opinion_dwarf_grudge
							}
						}
						reverse_opinion = {
							who = ROOT
							modifier = opinion_dwarf_grudge
							months = 12000
						}
					}
				}
			}
		}
	}
}

# dawi.6 The grudge war is won
# The winner of a settle_grudge war takes amends from event_target:
# grudge_offender (the enemy ruler or one of his vassals): his treasury, his
# kin as hostages, or his head. Any of them strikes out the grudge.
# Scope: ROOT is the Dawi who won the war.
character_event = {
	id = dawi.6
	desc = EVTDESCdawi6
	picture = GFX_evt_dwarf_grudge_war
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # settle_grudge CB on_success

	trigger = {
		event_target:grudge_offender = { is_alive = yes }
	}

	option = { # Empty his treasury
		name = EVTOPTAdawi6
		event_target:grudge_offender = { transfer_scaled_wealth = { to = ROOT value = 4 } }
		piety = 50
		dawi_settle_grudge_effect = yes
		ai_chance = {
			factor = 50
			modifier = { factor = 3 trait = greedy }
			modifier = { factor = 2 trait = content }
			modifier = { factor = 2 trait = kind }
		}
	}
	option = { # Take his kin as hostages
		name = EVTOPTBdawi6
		event_target:grudge_offender = {
			any_courtier = { limit = { dynasty = event_target:grudge_offender } imprison = ROOT }
		}
		piety = 50
		dawi_settle_grudge_effect = yes
		ai_chance = {
			factor = 30
			modifier = { factor = 3 trait = patient }
			modifier = { factor = 2 trait = deceitful }
		}
	}
	option = { # His head
		name = EVTOPTCdawi6
		piety = 100
		dawi_settle_grudge_effect = yes
		event_target:grudge_offender = {
			if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
			else = { death = { death_reason = death_execution killer = ROOT } }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 4 trait = wroth }
			modifier = { factor = 2 trait = cruel }
			modifier = { factor = 2 trait = proud }
			modifier = { factor = 0.2 trait = kind }
		}
	}
}

# dawi.13 Oathgold demanded
# The divorcer pays (more the higher his rank) or refuses.
# Scope: ROOT is the divorcer, FROM the divorced Dawi.
character_event = {
	id = dawi.13
	desc = EVTDESCdawi13
	picture = GFX_evt_dwarf_angry_look
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes # dawi.12

	option = { # Pay
		name = EVTOPTAdawi13
		trigger = {
			OR = {
				AND = { is_ruler = no wealth = 50 }
				AND = { tier = BARON wealth = 100 }
				AND = { tier = COUNT wealth = 250 }
				AND = { tier = DUKE wealth = 500 }
				AND = { tier = KING wealth = 750 }
				AND = { tier = EMPEROR wealth = 1000 }
			}
		}
		if = { limit = { is_ruler = no } wealth = -50 FROM = { wealth = 50 } }
		else_if = { limit = { tier = BARON } wealth = -100 FROM = { wealth = 100 } }
		else_if = { limit = { tier = COUNT } wealth = -250 FROM = { wealth = 250 } }
		else_if = { limit = { tier = DUKE } wealth = -500 FROM = { wealth = 500 } }
		else_if = { limit = { tier = KING } wealth = -750 FROM = { wealth = 750 } }
		else = { wealth = -1000 FROM = { wealth = 1000 } }
		FROM = { character_event = { id = dawi.14 } }
		ai_chance = {
			factor = 55
			modifier = { factor = 1.5 trait = kind }
			modifier = { factor = 1.5 trait = charitable }
			modifier = { factor = 1.5 trait = humble }
			modifier = { factor = 1.5 trait = craven }
			modifier = { factor = 2 trait = creature_dwarf }
			modifier = { factor = 0.5 trait = greedy }
			modifier = { factor = 0.5 trait = proud }
			modifier = { factor = 0.5 is_rival = FROM }
			modifier = { factor = 1.5 opinion = { who = FROM value = 50 } }
		}
	}
	option = { # Refuse
		name = EVTOPTBdawi13
		custom_tooltip = { text = dawi_oathgold_refused_tt }
		FROM = { character_event = { id = dawi.15 } }
		ai_chance = {
			factor = 45
			modifier = { factor = 1.5 trait = proud }
			modifier = { factor = 1.5 trait = stubborn }
			modifier = { factor = 1.5 trait = greedy }
			modifier = { factor = 1.5 trait = ambitious }
		}
	}
}

# dawi.50 The Slayer goes hunting
# Scope: ROOT is an unlanded Slayer.
character_event = {
	id = dawi.50
	hide_window = yes

	is_triggered_only = yes # on_bi_yearly_pulse
	religion_group = dwarf_gods_group
	prisoner = no
	only_capable = yes
	min_age = 16

	trigger = {
		trait = dwarf_slayer
		is_landed = no
		in_command = no
		NOT = { has_character_modifier = recent_slayer_work }
		liege = { NOT = { character = ROOT } }
	}

	immediate = {
		add_character_modifier = { name = recent_slayer_work duration = 365 hidden = yes }
		if = {
			limit = { liege = { any_demesne_province = { dawi_monster_province_trigger = yes } } }
			liege = {
				random_demesne_province = {
					limit = { dawi_monster_province_trigger = yes }
					save_event_target_as = slayer_hunting_ground
				}
			}
			random_list = {
				80 = {
					event_target:slayer_hunting_ground = { province_event = { id = dawi.51 } }
					character_event = { id = dawi.54 }
				}
				10 = {
					event_target:slayer_hunting_ground = { province_event = { id = dawi.51 } }
					add_trait_wounded_effect = yes
				}
				10 = {
					modifier = { factor = 0.5 martial = 15 }
					modifier = { factor = 0 is_invincible_trigger = yes }
					liege = { character_event = { id = dawi.53 } }
					death = { death_reason = death_battle }
				}
			}
		}
		else = { random = { chance = 50 liege = { character_event = { id = dawi.55 } } } }
	}
}
