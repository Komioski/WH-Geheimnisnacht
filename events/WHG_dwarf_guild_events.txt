# Dawi guild events: the Guild of Runesmiths and the Engineers Guild
#
# Part 1: Runesmith Guild events
# Namespace: runesmith (was "DAWI" in dwarven_societies_events.txt, itself a
# copy of the vanilla Hermetics society events with Verena text)
# Faction: Dwarfs
#
# The society events of the Guild of Runesmiths (runesmiths_society): joining,
# the quest tombola and its missions, the forge, the apprentice, the hunt for
# rune lore lost in fallen holds, the raid on a rival's forge, the Runemoot,
# gathering ore and salves, the paths of rune lore and the commissions the
# Guild hands out.
#
# Lore: the Guild of Runesmiths calls itself the Clan of Morgrim and claims
# its craft from Thungni, son of Grungni and Valaya, who taught it to seven
# apprentices (one of them, the Silent Sibling, lost to history). The lore is
# never written down and passes from master to apprentice by word of mouth.
# An apprentice swears his oaths in the leather apron his master gives him
# and is a runesmith once he has forged his own anvil; no runesmith works on
# another's anvil. The Rules of Three, Form, Jealousy and Pride bind every
# rune. The Anvil Forge is the heart of a hold and the anvil its mind; it
# burns all year except the two days before Zagazdeg. The Order of Hrungil
# hunts rogue runesmiths and stolen runic items. Much lore was lost when the
# holds fell in the Time of Woe. Sources: WFRP 4E Dwarf Player's Guide
# (Runesmithing, p.121-125), Lords of Stone and Steel (Karak Norn), the
# Lexicanum pages Runesmiths, Runelord, Anvil of Doom and Dwarf Rune.
#
# Joining and the tombola               runesmith.1 - runesmith.4
# The forge                             runesmith.10 - runesmith.28
# The apprentice                        runesmith.30 - runesmith.40
# Rune lore of the fallen holds         runesmith.50 - runesmith.60
# The forge raid                        runesmith.70 - runesmith.96
# Paths of rune lore                    runesmith.100 - runesmith.107
# The Runemoot                          runesmith.110 - runesmith.131
# Ore and salves                        runesmith.140 - runesmith.149
# Commissions and the rune schools      runesmith.160 - runesmith.167
# Study a Gromdal                       runesmith.170 - runesmith.173
#
# Fired from common/on_actions/wh_on_actions.txt: runesmith.1 on
# on_character_ask_to_join_society, runesmith.4 on the society bi-yearly
# pulse, runesmith.110 in its random list, runesmith.38 and 40 on death.
# Decisions: build_smithy (runesmith.10), dawi_acquire_ingredients_decision
# (runesmith.140), runesmith_quest_crafting (runesmith.163) and
# runemagic_aspect (runesmith.166) and runesmith_study_gromdal
# (runesmith.170) in decisions/wh_dawi_decisions.txt;

# runesmith.1 Asking to join the Guild
# A character asks to join the Guild. The highest-ranked member who is not a
# prisoner (or the leader, if all are) answers with the welcome letter.
# Scope: ROOT is the applicant, FROM is the society.
character_event = {
	id = runesmith.1
	hide_window = yes
	is_triggered_only = yes # on_character_ask_to_join_society

	trigger = {
		FROM = { leader = { society_member_of = runesmiths_society } }
	}

	immediate = {
		if = {
			limit = { FROM = { leader = { prisoner = yes } } }
			FROM = {
				if = {
					limit = { any_society_member = { society_rank == 4 prisoner = no } }
					random_society_member = {
						limit = { society_rank == 4 prisoner = no }
						save_event_target_as = runesmith_greeter
					}
				}
				else_if = {
					limit = { any_society_member = { society_rank == 3 prisoner = no } }
					random_society_member = {
						limit = { society_rank == 3 prisoner = no }
						save_event_target_as = runesmith_greeter
					}
				}
				else_if = {
					limit = { any_society_member = { society_rank == 2 prisoner = no } }
					random_society_member = {
						limit = { society_rank == 2 prisoner = no }
						save_event_target_as = runesmith_greeter
					}
				}
			}
		}
		if = {
			limit = { event_target:runesmith_greeter = { always = yes } }
			event_target:runesmith_greeter = { character_event = { id = runesmith.2 } }
		}
		else = { FROM = { leader = { character_event = { id = runesmith.2 } } } }
	}
}

# runesmith.4 Guild quest tombola
# Every two years a free member is handed a mission by another member or by
# his apprentice: a commission for a dwarf lord, an apprentice, a forge, a hunt
# for lost rune lore, ore and salves, or a raid on a rival's forge. The AI
# often skips the mission and just gains or loses influence.
# Scope: ROOT is the member.
character_event = {
	id = runesmith.4
	hide_window = yes
	is_triggered_only = yes # on_society_bi_yearly_pulse
	capable_only = yes
	is_in_society = yes

	trigger = {
		society_member_of = runesmiths_society
		has_any_quest = no
		NOT = { has_character_modifier = quest_cooldown_timer }
		is_inaccessible_trigger = no
		prisoner = no
		society = {
			any_society_member = {
				NOT = { character = ROOT }
				prisoner = no
				is_within_diplo_range = ROOT
				opinion = { who = ROOT value = 0 }
			}
		}
	}

	immediate = {
		# The mission giver
		random_list = {
			20 = { # Any member
				society = {
					any_society_member = {
						limit = {
							NOT = { character = ROOT }
							prisoner = no
							is_within_diplo_range = ROOT
							opinion = { who = ROOT value = 0 }
						}
						save_event_target_as = collaborator
					}
				}
			}
			40 = { # A member of the same realm
				trigger = {
					society = {
						any_society_member = {
							NOT = { character = ROOT }
							prisoner = no
							is_within_diplo_range = ROOT
							opinion = { who = ROOT value = 0 }
							same_realm = ROOT
						}
					}
				}
				society = {
					any_society_member = {
						limit = {
							NOT = { character = ROOT }
							prisoner = no
							is_within_diplo_range = ROOT
							opinion = { who = ROOT value = 0 }
							same_realm = ROOT
						}
						save_event_target_as = collaborator
					}
				}
			}
			40 = { # The apprentice
				trigger = {
					any_courtier_or_vassal = {
						has_minor_title = title_runesmith_apprentice
						prisoner = no
					}
				}
				random_courtier_or_vassal = {
					limit = { has_minor_title = title_runesmith_apprentice prisoner = no }
					save_event_target_as = collaborator
				}
			}
		}

		# The mission
		random_list = {
			25 = { # A commission for a dwarf lord
				modifier = { factor = 1.5 ai = yes }
				trigger = { lower_tier_than = KING age = 16 runesmith_has_client_trigger = yes }
				event_target:collaborator = { character_event = { id = runesmith.160 } }
			}
			20 = { # Find an apprentice
				trigger = {
					NOT = {
						any_courtier_or_vassal = { has_minor_title = title_runesmith_apprentice }
					}
					ai = no
					society_rank >= 2
				}
				event_target:collaborator = {
					character_event = { id = runesmith.30 days = 7 random = 7 }
				}
			}
			20 = { # Build a forge
				trigger = {
					NOT = { has_character_flag = runesmith_building_forge }
					NOT = { has_character_flag = runesmith_built_forge }
					OR = { ai = no society_rank == 4 }
					is_landed = yes
				}
				event_target:collaborator = {
					character_event = { id = runesmith.26 days = 7 random = 7 }
				}
			}
			15 = { # Rune lore of a fallen hold
				trigger = { is_landed = yes }
				event_target:collaborator = {
					character_event = { id = runesmith.50 days = 7 random = 7 }
				}
			}
			20 = { # Ore and salves
				trigger = {
					is_landed = yes
					NOT = { any_artifact = { has_artifact_flag = ingredient } }
					event_target:collaborator = { ai = yes }
				}
				event_target:collaborator = {
					character_event = { id = runesmith.141 days = 7 random = 7 }
				}
			}
			10 = { # Raid a rival's forge
				trigger = {
					society = {
						any_society_member = {
							is_within_diplo_range = ROOT
							NOR = {
								character = ROOT
								character = event_target:collaborator
								reverse_opinion = { who = event_target:collaborator value = 20 }
								reverse_opinion = { who = ROOT value = 20 }
								has_character_flag = infiltration_rival
								has_character_flag = infiltration_mutual_dislike
								has_character_flag = infiltration_rich
							}
						}
					}
				}
				event_target:collaborator = {
					character_event = { id = runesmith.70 days = 7 random = 7 }
				}
			}
			105 = { # The AI skips the mission
				modifier = { factor = 0 ai = no }
				random_list = {
					70 = {
						modifier = { factor = 0.9 society_influence >= 30 }
						modifier = { factor = 0.9 society_influence >= 40 }
						modifier = { factor = 0.8 society_influence >= 50 }
						modifier = { factor = 0.7 society_influence >= 70 }
						modifier = { factor = 0.6 society_influence >= 80 }
						modifier = { factor = 0.5 society_influence >= 90 }
						add_mission_succeed_influence_effect = yes
					}
					30 = { add_mission_fail_influence_effect = yes }
				}
			}
			175 = { # Nothing for the AI this time
				modifier = { factor = 0 ai = no }
			}
		}
	}
}

# The forge
# A runesmith builds his own forge in steps: where, how big, then the anvil,
# the furnace, the tools and the rune tablets in random order. Each choice adds
# to the variable runesmith_forge_level, mishaps take from it, and the total
# decides which of the five forge modifiers he ends with.
# Flags: runesmith_building_forge (while building), runesmith_built_forge
# (done, kept), runesmith_hidden_forge and runesmith_deep_forge (kept, make the
# forge harder to raid), runesmith_forge_anvil/furnace/tools/tablets (steps
# taken, cleared at the end).
###############################################################################

# runesmith.10 A forge of my own
# From the build_smithy decision. Begin the work or take the coin back.
# Scope: ROOT is the runesmith.
character_event = {
	id = runesmith.10
	desc = EVTDESCrunesmith10
	picture = GFX_evt_castle_construction
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # build_smithy decision

	option = { # Begin
		name = EVTOPTArunesmith10
		hidden_effect = {
			set_variable = { which = runesmith_forge_level value = 1 }
			character_event = { id = runesmith.11 days = 7 }
		}
	}
	option = { # Not now
		name = EVTOPTBrunesmith10
		clr_character_flag = runesmith_building_forge
		wealth = 50
	}
}

# runesmith.17 The forge is finished
# Clears the building flags and picks the finished forge by its level.
# Scope: ROOT is the runesmith.
character_event = {
	id = runesmith.17
	hide_window = yes
	is_triggered_only = yes # runesmith_forge_next_step_effect

	immediate = {
		set_character_flag = runesmith_built_forge
		clr_character_flag = runesmith_building_forge
		clr_character_flag = runesmith_forge_anvil
		clr_character_flag = runesmith_forge_furnace
		clr_character_flag = runesmith_forge_tools
		clr_character_flag = runesmith_forge_tablets
		if = {
			limit = { NOT = { check_variable = { which = runesmith_forge_level value = 6 } } }
			character_event = { id = runesmith.18 }

# Scope: ROOT is the rival, FROM is the member.

	id = runesmith.87
	desc = { trigger = { FROM = { has_character_flag = great_theft } } text = EVTDESCrunesmith87 }
	desc = { trigger = { FROM = { has_character_flag = good_theft } } text = EVTDESCrunesmith87b }
	desc = { trigger = { FROM = { has_character_flag = ok_theft } } text = EVTDESCrunesmith87c }
	picture = GFX_evt_burning_house
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes # runesmith.86
	hide_from = yes

	option = {
		name = CURSES
		if = {
			limit = { FROM = { has_character_flag = great_theft } }
			random_artifact = {
				limit = { has_artifact_flag = rune NOT = { has_artifact_flag = special_rune } }
				tooltip = { destroy_artifact = yes }
				hidden_effect = { transfer_artifact = { from = ROOT to = FROM } }
			}
		}
		if = {
			limit = {
				FROM = { OR = { has_character_flag = great_theft has_character_flag = good_theft } }
				any_artifact = { has_artifact_flag = hermetics has_artifact_flag = ingredient }
			}
			random_artifact = {
				limit = { has_artifact_flag = hermetics has_artifact_flag = ingredient }
				tooltip = { destroy_artifact = yes }
				hidden_effect = { transfer_artifact = { from = ROOT to = FROM } }
			}
		}
		if = {
			limit = { FROM = { has_character_flag = great_theft } }
			tooltip = { scaled_wealth = { value = -0.5 min = -50 } }
		}
		else_if = {
			limit = { FROM = { has_character_flag = good_theft } }
			tooltip = { scaled_wealth = { value = -0.25 min = -25 } }
		}
		else = { tooltip = { scaled_wealth = { value = -0.1 min = -10 } } }
		hidden_effect = {
			FROM = {
				clr_character_flag = great_theft
				clr_character_flag = good_theft
				clr_character_flag = ok_theft
			}
		}
	}
}

					event_target:infiltration_target = {
						has_character_flag = runesmith_built_forge
					}

					event_target:infiltration_target = {
						has_character_flag = runesmith_hidden_forge

	option = { # The ward
		name = EVTOPTBrunesmith88
		trigger = { has_character_flag = lab_explosion_death }

	after = {

			event_target:infiltration_target = { character_event = { id = runesmith.89 } }
			character_event = { id = runesmith.95 days = 40 }
		}
	}

# engineer.63 A manling pupil
# From engineer_dawi_teach_manlings. The ablest manling of the court (his
# own, or his liege's) learns the craft. The tithe goes to Zhufbar.
# Scope: ROOT is a Dawi engineer of the manling cities.
character_event = {
	id = engineer.63
	desc = EVTDESCengineer63
	picture = GFX_evt_dwarf_engineer
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # engineer_dawi_teach_manlings

	immediate = {
		if = {
			limit = { any_courtier = { engineer_manling_pupil_trigger = yes } }
			random_courtier = {
				limit = { engineer_manling_pupil_trigger = yes }
				preferred_limit = { learning = 14 }
				save_event_target_as = engineer_pupil
			}
		}
		else = {
			liege = {
				random_courtier = {
					limit = { engineer_manling_pupil_trigger = yes }
					preferred_limit = { learning = 14 }
					save_event_target_as = engineer_pupil
				}
			}
		}
		d_zhufbar = {
			holder_scope = { if = { limit = { NOT = { character = ROOT } } wealth = 25 } }
		}
	}

	option = { # The craft, and not one rune
		name = EVTOPTAengineer63
		event_target:engineer_pupil = {
			add_trait = engineer
			opinion = { who = ROOT modifier = opinion_grateful years = 10 }
		}
		add_society_currency_minor_effect = yes
	}
}
