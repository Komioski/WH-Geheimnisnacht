###############################################################################
# Elders
# Namespace: elder (was HFP.24001 - HFP.24025 in HFP_eldership_events;
#   elder.N = old HFP.240NN)
# Faction: Family & bloodlines
# Serves: realms under Eldership succession
#   (the Ropsmann hill-clans of Kislev, and faiths with Ancestor Worship or
#   the reformed Atruhayid custom; Albion and the Kislevites, once the
#   melting pot forms that culture (kislev.150), may adopt it)
#
# Under Eldership the old ones of the realm choose the next ruler, and they
# never let the present one forget it. A pleased Elder moves to court, asks
# a county for a worthy kinsman, frets about being forgotten, sings the
# ruler's praises and passes on old wisdom (rarely a Staff of the Elders).
# A displeased one walks out, pushes an idiot for a county or talks a rival
# into a claim. Kill or jail an Elder without cause and all the others hold
# it against you.
#
# The long life                     elder.1
# Moving in and walking out         elder.2 - elder.5
# A county for a worthy kinsman     elder.6 - elder.9
# A county for a fool               elder.10 - elder.13
# The lonely Elder                  elder.14 - elder.15
# A claim against the ruler         elder.16 - elder.18
# Killed or jailed Elders           elder.19, elder.24
# Praise and wisdom                 elder.20 - elder.23
# Eldership at game start           elder.25
#
# Saved: character flag flag_elder_moved_court; character modifier
#   elder_demand_cooldown on the ruler. Event targets scoped_host,
#   scoped_ruler, scoped_recipient_first/second/third,
#   scoped_potential_count, scoped_suitor, scoped_claim_title.
#
# Callers:
#   common/on_actions/00_on_actions.txt: on_startup (25), on_decade_pulse
#     (1; list eldership_succession 2, 4, 6, 10, 14, 16, 20, 22), on_death
#     (19), on_become_imprisoned_any_reason (24).
# Support: common/scripted_triggers/wh_elder_triggers.txt,
#   common/scripted_effects/wh_elder_effects.txt,
#   common/artifacts/32_artifacts_eldership.txt
# Loc: localisation/00_WHG_eldership_events.csv
###############################################################################

# elder.1 So much to live for
# Being an Elder keeps the old ones going.
# Scope: ROOT is the Elder (on_decade_pulse).
character_event = {
	id = elder.1
	desc = EVTDESCelder1
	picture = { trigger = { elder_southern_picture_trigger = no } picture = GFX_evt_elder_council }
	picture = {
		trigger = { elder_southern_picture_trigger = yes }
		picture = GFX_evt_west_african_ruler
	}
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes
	min_age = 40

	trigger = {
		NOT = { has_character_modifier = elder_long_life_modifier }
		elder_able_trigger = yes
		any_eldership_title = {
			ROOT = {
				OR = {
					has_elector_stance = { stance = elder_displeased title = PREV }
					has_elector_stance = { stance = elder_pleased title = PREV }
					has_elector_stance = { stance = elder_ecstatic title = PREV }
				}
			}
		}
	}

	option = {
		name = EVTOPTAelder1
		add_character_modifier = { name = elder_long_life_modifier years = 100 }
		if = { limit = { is_ruler = no } give_nickname = nick_the_elder }
	}
}

# elder.2 The Elder comes to court
# A pleased Elder who lives elsewhere wants to watch the ruler up close.
# Scope: ROOT is the Elder (on_decade_pulse list eldership_succession).
character_event = {
	id = elder.2
	hide_window = yes
	is_triggered_only = yes
	min_age = 40

	trigger = {
		NOT = { has_character_flag = flag_elder_moved_court }
		is_ruler = no
		is_married = no
		is_consort = no
		elder_able_trigger = yes
		any_eldership_title = {
			elder_root_pleased_trigger = yes
			holder_scope = {
				NOT = { is_liege_of = ROOT }
				NOT = { has_character_modifier = elder_demand_cooldown }
			}
		}
		NOT = { # Happy where they are: their liege is a ruler they like
			any_eldership_title = {
				elder_root_pleased_trigger = yes
				holder_scope = { is_liege_of = ROOT }
			}
		}
	}

	immediate = {
		random_eldership_title = {
			limit = {
				elder_root_pleased_trigger = yes
				holder_scope = {
					NOT = { is_liege_of = ROOT }
					NOT = { has_character_modifier = elder_demand_cooldown }
				}
			}
			holder_scope = { save_event_target_as = scoped_host }
		}
		set_character_flag = flag_elder_moved_court
		event_target:scoped_host = { character_event = { id = elder.3 } }
	}
}

			holder_scope = {
				is_liege_of = ROOT

				is_landed = yes

			preferred_limit = { reverse_opinion = { who = ROOT value = 40 } }

# A pleased Elder picks out a kinsman of the ruler for a spare county: the

			elder_root_pleased_trigger = yes

		hidden_effect = {
			random_eldership_title = {
				limit = {
					elder_root_pleased_trigger = yes

				}
				holder_scope = {
					save_event_target_as = scoped_ruler

							NOR = {
								is_rival = ROOT

								is_weak_trigger = yes
								is_evil_trigger = yes
							}
							OR = { is_smart_trigger = yes is_attractive_trigger = yes }
						}
						preferred_limit = {
							NOR = {
								is_rival = ROOT
								is_dumb_trigger = yes
								is_weak_trigger = yes
								is_evil_trigger = yes
							}
						}
						preferred_limit = { NOT = { is_rival = ROOT } }
						save_event_target_as = scoped_recipient_first
					}
					random_dynasty_member = { # The best liked
						limit = {
							elder_kin_candidate_trigger = yes
							NOT = { character = event_target:scoped_recipient_first }
						}
						preferred_limit = {
							OR = { is_lover = ROOT is_friend = ROOT is_married = ROOT }
						}
						preferred_limit = { reverse_opinion = { who = ROOT value = 80 } }

						preferred_limit = { reverse_opinion = { who = ROOT value = 40 } }
						preferred_limit = { reverse_opinion = { who = ROOT value = 20 } }
						save_event_target_as = scoped_recipient_second
					}
					random_dynasty_member = { # The most able
						limit = {
							elder_kin_candidate_trigger = yes
							NOR = {
								character = event_target:scoped_recipient_first
								character = event_target:scoped_recipient_second
							}
						}
						preferred_limit = {
							OR = { diplomacy = 18 intrigue = 18 stewardship = 18 learning = 18 }
						}
						preferred_limit = {
							OR = { diplomacy = 16 intrigue = 16 stewardship = 16 learning = 16 }
						}
						preferred_limit = {
							OR = { diplomacy = 14 intrigue = 14 stewardship = 14 learning = 14 }
						}
						preferred_limit = {
							OR = { diplomacy = 12 intrigue = 12 stewardship = 12 learning = 12 }
						}
						preferred_limit = {
							OR = { diplomacy = 10 intrigue = 10 stewardship = 10 learning = 10 }
						}
						preferred_limit = {
							OR = { diplomacy = 8 intrigue = 8 stewardship = 8 learning = 8 }
						}
						save_event_target_as = scoped_recipient_third
					}
				}
			}
		}
	}
