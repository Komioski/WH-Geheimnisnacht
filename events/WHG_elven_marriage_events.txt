###############################################################################
# Elven courtship, weddings and divorce
# Namespace: elfwed (was selfmarriage in S_elven_marriage_events.txt and
# sdivorce in S_elven_divorce_events.txt)
# Faction: High Elves
# Serves: Asur, Druchii, Asrai
#
# Elves do not marry in haste. A suitor courts the beloved with gifts and
# shows of talent until the courtship is warm enough to propose. The
# beloved (or, for a courtier, their liege) gives the final word, and the
# wedding follows the rites of their people. An Asur who is divorced may
# demand gold or land as the price of letting the spouse go.
#
# Courtship                               elfwed.1 - elfwed.6
# The proposal and the wedding            elfwed.7 - elfwed.15
# AI courtships                           elfwed.50 - elfwed.51
# Divorce among the Asur                  elfwed.60 - elfwed.70
#
# The courtship keeps its state on the suitor: the variable courting_score,
# the flag elfwed_gift_* for the gift on its way, and the flag
# asking_for_matrilineal. Both carry the trait courting (which forbids any
# other marriage), the hidden modifier courting_timer (a one-year cooldown
# on starting another) and elfwed_courtship_alive (five years, see elfwed.15). The event targets
# elfwed_suitor and elfwed_beloved are saved at the start and used down the
# whole chain. A courtship that stalls lapses on its own (elfwed.15).
#
# Callers:
#   decisions/wh_asur_decisions.txt: elven_courting_decision (1).
#   common/on_actions/WHG_character_on_actions.txt: bi-yearly (50), on_divorce (60).
#   events/WHG_on_action_events.txt: the race check on succession marriages, response.214 (70).
# Support: common/scripted_effects/wh_elven_marriage_effects.txt,
#   common/scripted_triggers/wh_elven_marriage_triggers.txt
# Loc: localisation/00_WHG_elven_marriage_events.csv
###############################################################################

namespace = elfwed

# Courtship
# The suitor picks a gift or a show of talent (elfwed.2), the beloved
# judges it months later (elfwed.3), and the suitor learns how it went
# (elfwed.4 or elfwed.5). Each success raises the courtship score; a
# failure may lower it, and at nothing the beloved is free again.
###############################################################################

# elfwed.1 The courtship begins (hidden)
# Scope: ROOT is the beloved, FROMFROM the suitor (elven_courting_decision).
character_event = {
	id = elfwed.1
	hide_window = yes

	is_triggered_only = yes # elven_courting_decision

	immediate = {
		save_event_target_as = elfwed_beloved
		FROMFROM = {
			save_event_target_as = elfwed_suitor
			add_trait = courting
			character_event = { id = elfwed.2 }
			add_character_modifier = { name = elfwed_courtship_alive hidden = yes duration = 1825 }
			character_event = { id = elfwed.15 days = 1825 }
		}
		add_character_modifier = { name = elfwed_courtship_alive hidden = yes duration = 1825 }
		character_event = { id = elfwed.15 days = 1825 }
	}
}

# elfwed.62 The price of a divorce
# The higher the divorcer's rank, the more they pay. Refusing costs prestige
# by the spouse's rank.
# Scope: ROOT is the divorcer, FROM the divorced spouse (elfwed.60).
character_event = {
	id = elfwed.62
	desc = EVTDESCelfwed62
	picture = GFX_evt_asur_arguing
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # elfwed.60

	option = { # Pay
		name = EVTOPTAelfwed62
		trigger = {
			OR = {
				AND = { is_ruler = no wealth = 50 }
				AND = { tier = BARON wealth = 100 }
				AND = { tier = COUNT wealth = 250 }
				AND = { tier = DUKE wealth = 500 }
				AND = { tier = KING wealth = 750 }
				AND = { tier = EMPEROR wealth = 1000 }
			}
		}
		if = { limit = { tier = EMPEROR } wealth = -1000 FROM = { wealth = 1000 } }
		else_if = { limit = { tier = KING } wealth = -750 FROM = { wealth = 750 } }
		else_if = { limit = { tier = DUKE } wealth = -500 FROM = { wealth = 500 } }
		else_if = { limit = { tier = COUNT } wealth = -250 FROM = { wealth = 250 } }
		else_if = { limit = { tier = BARON } wealth = -100 FROM = { wealth = 100 } }
		else = { wealth = -50 FROM = { wealth = 50 } }
		elfwed_divorce_settled_effect = yes
		hidden_effect = { FROM = { character_event = { id = elfwed.63 } } }
		ai_chance = {
			factor = 70
			modifier = { factor = 1.5 is_lover = FROM }
			modifier = { factor = 1.75 is_friend = FROM }
			modifier = { factor = 0.5 is_rival = FROM }
			modifier = {
				factor = 0.75
				OR = { trait = wroth trait = envious trait = cruel trait = proud }
			}
			modifier = {
				factor = 1.5
				OR = { trait = charitable trait = kind trait = humble trait = shy }
			}
			modifier = { factor = 0.5 trait = greedy }
		}
	}
	option = { # Refuse
		name = EVTOPTBelfwed62
		if = { limit = { FROM = { tier = EMPEROR } } prestige = -1000 }
		else_if = { limit = { FROM = { tier = KING } } prestige = -750 }
		else_if = { limit = { FROM = { tier = DUKE } } prestige = -500 }
		else_if = { limit = { FROM = { tier = COUNT } } prestige = -250 }
