###############################################################################
# Empire events
# Namespace: empire (was unification, EVTweaponry, DR_imperial and NARFLA in
# wh_empire_events.txt)
# Faction: Empire
#
# The Empire of Sigmar: a claimant crowned by acclaim who asks the Electors
# to kneel, the Hammer of Sigmar, the great cults' long feud and their peace
# in the face of a common foe, the Arch-Lectors, and the game start
# narrative for the claimants.
#
# Lore: since 1359 IC the Empire has had rival claimants and no crowned
# Emperor. Ottilia of Talabecland crowned herself with the Ulricans behind
# her and named Sigmar a false god, and each of the great cults has backed a
# claimant of its own ever since. In 1979 IC the Grand Theogonist refused to
# crown Magritta of Marienburg, and the Electors stopped electing anyone at
# all. Magnus the Pious ends the Age of Three Emperors after the Great War
# Against Chaos. Ghal Maraz, the hammer of Sigmar, goes with the crown.
#
# Reunification                          empire.1 - empire.7
# The great cults                        empire.10, empire.12 - empire.14
# The Arch-Lectors                       empire.20 - empire.21
# Game start                             empire.30 - empire.31
# The Runefangs                          empire.40 - empire.42
# The Knights Griffon                    empire.50 - empire.51
# Ottilia's Edict, the War of the Hammer  empire.60 - empire.68
# The Pantheon of the Old World           empire.70
# The seats of Sigmar                    empire.80 - empire.85
#
# Callers:
#   decisions/wh_empire_decisions.txt: claim_the_throne fires empire.1.
#   common/on_actions/WHG_character_on_actions.txt: empire.7 and empire.20 on
#     on_death, empire.30 on on_startup, empire.40 on on_new_holder,
#     on_new_holder_inheritance and on_new_holder_usurpation; empire.80 on
#     the same three.
#   common/religious_titles/00_religious_titles.txt: title_lector fires
#     empire.21. common/minor_titles/wh_minor_titles.txt: the Reiksmarshal
#     and the Emperor's Champion fire empire.42.
#   empire.2 fires empire.10 (one faith triumphant).
#   decisions/wh_empire_decisions.txt: found_knights_griffon fires empire.50;
#     renew_ottilian_edict fires empire.60, revoke_ottilian_edict empire.65.
#   common/cb_types/wh_cb_types.txt: war_of_the_hammer fires empire.65 on
#     success and empire.70 on failure. WHG_realm_on_actions: empire.66 on
#     on_new_holder_inheritance and on_new_holder_usurpation, empire.67 on
#     on_character_convert_religion.
#   common/cb_types/wh_cb_types.txt: the vampires' great war fires
#     empire.12 on success. common/cb_types/00_cb_types.txt: a Black
#     Crusade on the Empire fires empire.12 when it is declared.
# Scripted effects: common/scripted_effects/WHG_empire_effects.txt
# Scripted triggers: common/scripted_triggers/WHG_empire_triggers.txt
# Opinions: common/opinion_modifiers/WHG_empire_opinion_modifiers.txt
# Modifiers: common/event_modifiers/WHG_empire_event_modifiers.txt (the burned
#   temples, the Pantheon), common/religion_modifiers/wh_religion_modifiers.txt
# Saved (Edict): global flags ottilian_edict_active, ottilian_edict_ended,
#   ottilian_edict_renewed (once per game);
#   persistent target ottilian_edict_giver; character flags

#   ottilian_edict_lost_war; title flag ottilian_edict_title.

###############################################################################
# Reunification
# An Elector of great renown and piety claims the throne by acclaim
# (claim_the_throne). A week later he is crowned and given Ghal Maraz; a
# month later every independent Elector of the Empire must kneel or refuse.
# Those who kneel bring their realms into the Empire. Those who refuse give
# the Emperor claims on their crowns, and the gentler of their own vassals
# think less of them for it.
###############################################################################

# empire.1 Proclaim the claim?
# Scope: ROOT is the claimant.
character_event = {
	id = empire.1
	desc = EVTDESCempire1
	picture = GFX_evt_rival_rulers
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # claim_the_throne

	option = { # Proclaim it
		name = EVTOPTAempire1
		set_character_flag = claimed_empire
		custom_tooltip = { text = empire_claim_tt }
		hidden_effect = {
			character_event = { id = empire.2 days = 7 }
			any_independent_ruler = {
				limit = { empire_elector_to_ask_trigger = yes }
				character_event = { id = empire.3 days = 30 }
			}
		}
	}
	option = { # Not yet
		name = EVTOPTBempire1
		trigger = { ai = no }
	}
}

# empire.2 The coronation
# The claimant is crowned, the old rival crowns are put aside and the Grand
# Theogonist gives him Ghal Maraz, taken from whoever holds it.
# Scope: ROOT is the new Emperor.
character_event = {
	id = empire.2
	desc = EVTDESCempire2
	picture = GFX_evt_bishop
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # empire.1

	trigger = {
		NOT = { has_global_flag = united_empire }
		e_the_empire = { has_holder = no }
	}

	immediate = {
		set_global_flag = united_empire
		remove_trait = crowned_by_priest
		remove_trait = crowned_by_bishop
		remove_trait = crowned_by_pope
		remove_trait = crowned_by_myself
		remove_trait = crowned_by_priest_rd
		remove_trait = crowned_by_bishop_rd
		remove_trait = crowned_by_pope_rd
		remove_character_modifier = crowned_in_barn
		remove_character_modifier = extravagant_coronation
	}

	option = {
		name = EVTOPTAempire2
		create_bloodline = { type = magnus }
		e_the_empire = {
			grant_title = ROOT
			add_law = { law = succ_hre_elective cooldown = no opinion_effect = no }
			add_law = { law = true_cognatic_succession cooldown = no opinion_effect = no }
		}
		empire_destroy_contender_titles_effect = yes
		piety = 150
		empire_ghal_maraz_effect = yes
		hidden_effect = { narrative_event = { id = empire.10 days = 3 } }
	}
}

# empire.7 A pretender's heir inherits a weaker claim
# The strong claims of the claimants (history, a coronation by the head of
# a great cult) pass to the primary heir as a weak claim, which still
# makes claim_the_throne cheaper. A weak claim is not passed on again.
# Scope: ROOT is the dying claimant.
character_event = {
	id = empire.7
	hide_window = yes
	is_triggered_only = yes # on_death
	only_playable = yes

	trigger = {
		has_strong_claim = e_the_empire
		NOT = { has_global_flag = united_empire }
		e_the_empire = { has_holder = no }
		current_heir = { culture_group = imperial_group NOT = { has_claim = e_the_empire } }
	}

	immediate = {
		current_heir = { add_weak_claim = e_the_empire }
	}
}

###############################################################################
# The great cults
# The cults of Sigmar, Ulric, Taal, Morr and Manann have backed rival
# claimants since 1359 IC. When the vampires win a great war in the Empire,
# or a Black Crusade marches on it, the Grand Theogonist makes peace among
# them (empire.12): their faithful may marry one another again, and every
# ruler of the cults must accept the peace, or seek allies against the foe,
# or refuse it. The coronation of a new Emperor ends the schism for good
# (empire.10), with one cult set first among them.
###############################################################################

# empire.10 One faith triumphant
# The new Emperor mends the schism of 1359. One cult is set first among the
# great cults: Sigmar's, as in the lore. A player Emperor crowned by Ar-Ulric
# or the High Priest of Taal, or who keeps one of those faiths, raises that
# cult instead. The great cults may marry one another again, the feud ends
# for good (no peace of the cults is needed after it). The other cults keep
# their temples and their faithful, and no one is asked to convert.
# Scope: ROOT is the new Emperor.
narrative_event = {
	id = empire.10
	title = EVTTITLEempire10
	desc = { trigger = { has_global_flag = empire_triumph_ulric } text = EVTDESCempire10_ulric }
	desc = { trigger = { has_global_flag = empire_triumph_taal } text = EVTDESCempire10_taal }
	desc = { trigger = { has_global_flag = empire_triumph_sigmar } text = EVTDESCempire10_sigmar }
	picture = GFX_evt_icon
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes # empire.2

	trigger = {
		NOT = { has_global_flag = imperial_churches_unified }
	}

	immediate = {
		set_global_flag = imperial_churches_unified
		set_global_flag = imperial_churches_peace
		if = {
			limit = { ai = no has_character_flag = empire_crowned_by_ulrican }
			set_global_flag = empire_triumph_ulric
		}
		else_if = {
			limit = { ai = no has_character_flag = empire_crowned_by_taalite }
			set_global_flag = empire_triumph_taal
		}
		else_if = {
			limit = { ai = no has_character_flag = empire_crowned_by_sigmarite }
			set_global_flag = empire_triumph_sigmar
		}
		else_if = { limit = { ai = no religion = ulrican } set_global_flag = empire_triumph_ulric }
		else_if = { limit = { ai = no religion = taalite } set_global_flag = empire_triumph_taal }
		else = { set_global_flag = empire_triumph_sigmar }
	}

	option = {
		name = EVTOPTAempire10
		custom_tooltip = { text = empire_schism_mended_tt }
		hidden_effect = {
			sigmarite = {
				add_intermarry = ulrican
				add_intermarry = taalite
				add_intermarry = morrian
				add_intermarry = manannite
			}
			ulrican = {
				add_intermarry = sigmarite
				add_intermarry = taalite
				add_intermarry = morrian
				add_intermarry = manannite
			}
			taalite = {
				add_intermarry = sigmarite
				add_intermarry = ulrican
				add_intermarry = morrian
				add_intermarry = manannite
			}
			morrian = {
				add_intermarry = sigmarite
				add_intermarry = ulrican
				add_intermarry = taalite
				add_intermarry = manannite
			}
			manannite = {
				add_intermarry = sigmarite
				add_intermarry = ulrican
				add_intermarry = taalite
				add_intermarry = morrian
			}
		}
		if = {
			limit = { has_global_flag = empire_triumph_ulric }
			ulrican = { religion_authority = { modifier = faith_triumphant } }
		}
		else_if = {
			limit = { has_global_flag = empire_triumph_taal }
			taalite = { religion_authority = { modifier = faith_triumphant } }
		}
		else = {
			sigmarite = { religion_authority = { modifier = faith_triumphant } }
			if = { limit = { religion = sigmarite } give_nickname = nick_the_pious }
		}
	}
}

# empire.12 The Grand Theogonist makes peace among the cults
# Once a game. The cults' faithful may marry one another again, the high
# priests debate their dogma, and every ruler of the cults hears of the
# peace.
# Scope: ROOT is the Grand Theogonist.
character_event = {
	id = empire.12
	hide_window = yes
	is_triggered_only = yes # wh_cb_types (vampires), 00_cb_types (Black Crusade)

	trigger = {
		NOT = { has_global_flag = imperial_churches_peace }
		NOT = { has_global_flag = imperial_churches_unified }
	}

	immediate = {
		set_global_flag = imperial_churches_peace
		remove_trait = zealous
		sigmarite = { add_intermarry = taalite add_intermarry = ulrican }
		ulrican = { add_intermarry = taalite add_intermarry = sigmarite add_intermarry = manannite }
		taalite = { add_intermarry = ulrican add_intermarry = sigmarite }
		manannite = { add_intermarry = ulrican }
		religion_authority = { modifier = theology_debates }
		k_cult_of_taal_and_rhya = {
			holder_scope = { religion_authority = { modifier = theology_debates } }
		}
		k_cult_of_ulric = {
			holder_scope = { religion_authority = { modifier = theology_debates } }
		}
		k_cult_of_morr = { holder_scope = { religion_authority = { modifier = theology_debates } } }
		any_playable_ruler = {
			limit = { empire_imperial_cult_trigger = yes controls_religion = no }
			narrative_event = { id = empire.13 }
		}
	}
}

# empire.13 The peace of the cults
# With a common foe at the gates, a claimant may also seek allies: a rival
# claimant, a Dawi king, or the King of Bretonnia. In peace, accepting costs
# piety and refusing earns it.
# Scope: ROOT is a ruler of the cults, FROM is the Grand Theogonist.
narrative_event = {
	id = empire.13
	title = EVTTITLEempire13
	desc = {
		trigger = { k_cult_of_sigmar = { has_title_flag = vampire_invasion } }
		text = EVTDESCempire13a
	}
	desc = {
		trigger = {
			k_cult_of_sigmar = {
				has_title_flag = black_crusade_invasion
				NOT = { has_title_flag = vampire_invasion }
			}
		}
		text = EVTDESCempire13b
	}
	desc = { trigger = { empire_common_foe_trigger = no } text = EVTDESCempire13c }
	picture = GFX_evt_icon
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes # empire.12

	immediate = {
		if = {
			limit = { empire_holds_contender_crown_trigger = yes empire_common_foe_trigger = yes }
			random_independent_ruler = {
				limit = {
					empire_holds_contender_crown_trigger = yes
					NOT = { character = ROOT }
					NOT = { is_allied_with = ROOT }
				}
				save_event_target_as = imperial_ally_candidate
			}
			random_independent_ruler = {
				limit = {
					OR = {
						has_landed_title = k_the_vaults
						has_landed_title = k_grey_mountains
						has_landed_title = k_zhufbarak
						has_landed_title = k_kadrin_ankor
						has_landed_title = k_everpeak
					}
					NOT = { is_allied_with = ROOT }
				}
				save_event_target_as = imperial_ally_dwarf_candidate
			}
			e_bretonnia = {
				holder_scope = {
					if = {
						limit = { NOT = { is_allied_with = ROOT } }
						save_event_target_as = imperial_ally_bretonnia_candidate
					}
				}
			}
		}
	}

	option = { # Accept the peace
		name = EVTOPTAempire13
		if = {
			limit = { empire_common_foe_trigger = yes }
			capital_scope = { add_province_modifier = { name = troops_increases years = 3 } }
		}
		else = { piety = -100 }
		ai_chance = {
			factor = 50
			modifier = { factor = 5 any_liege = { religion = FROM } }
			modifier = { factor = 0.1 empire_holds_contender_crown_trigger = yes }
			modifier = { factor = 0.1 trait = zealous }
			modifier = { factor = 0.5 trait = content }
			modifier = { factor = 2 trait = kind }
			modifier = { factor = 2 trait = craven }
		}
	}
	option = { # Seek allies among the Empire and the Dawi
		name = EVTOPTBempire13
		trigger = {
			empire_holds_contender_crown_trigger = yes
			empire_common_foe_trigger = yes
			OR = {
				event_target:imperial_ally_candidate = { always = yes }
				event_target:imperial_ally_dwarf_candidate = { always = yes }
			}
		}
		capital_scope = { add_province_modifier = { name = troops_increases years = 3 } }
		prestige = -250
		piety = -100
		event_target:imperial_ally_candidate = {
			show_portrait = yes
			character_event = { id = empire.14 days = 30 }
		}
		event_target:imperial_ally_dwarf_candidate = {
			show_portrait = yes
			character_event = { id = empire.14 days = 7 }
		}
		ai_chance = {
			factor = 50
			modifier = { factor = 5 trait = cynical }
			modifier = { factor = 0.1 trait = zealous }
			modifier = { factor = 2 trait = brave }
			modifier = { factor = 2 trait = just }
			modifier = { factor = 0.7 trait = proud }
		}
	}
	option = { # Plead with Bretonnia
		name = EVTOPTCempire13
		trigger = {
			empire_holds_contender_crown_trigger = yes
			empire_common_foe_trigger = yes
			event_target:imperial_ally_bretonnia_candidate = { always = yes }
		}
		capital_scope = { add_province_modifier = { name = troops_increases years = 3 } }
		prestige = -500
		piety = -250
		event_target:imperial_ally_bretonnia_candidate = {
			show_portrait = yes
			character_event = { id = empire.14 days = 25 }
		}
		ai_chance = {
			factor = 30
			modifier = { factor = 5 trait = cynical }
			modifier = { factor = 0.1 trait = zealous }
			modifier = { factor = 0.7 trait = proud }
		}
	}
	option = { # Refuse the peace
		name = EVTOPTDempire13
		if = {
			limit = { empire_common_foe_trigger = yes }
			capital_scope = { add_province_modifier = { name = inspired_warriors years = 2 } }
			add_trait = zealous
			piety = -50
		}
		else = { piety = 250 }
		ai_chance = {
			factor = 50
			modifier = { factor = 2 trait = content }
			modifier = { factor = 5 trait = zealous }
			modifier = { # Middenland's Ulricans are the most stubborn of all
				factor = 8
				religion = ulrican
				capital_scope = { kingdom = { title = k_middenland } }
			}
		}
	}
}

			10 = { # Drunkard
