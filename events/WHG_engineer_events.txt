###############################################################################
# Engineers Guild events
# Namespace: engineer (was DR_engineers_guild in
# DR_engineers_guild_mnm_events.txt, dwarfengineering in
# GR_Dwarf_engineering_events.txt and dawimech in GR_study_mechanics_events.txt)
# Faction: Dwarfs
#
# The Engineers Guild (engineers_guild): joining, AI recruits, experiments,
# the rank powers' events, Zhufbar's call to gifted young Dawi, hiring an
# engineer for a court, and the founding of the Imperial Engineers School in
# Nuln, which opens the guild to Imperial humans.
#
# Lore: the Dwarf Engineers Guild is the most secretive of the Dawi craft
# guilds. Its clans honour Morgrim, the First Engineer, as well as Grungni.
# Its Elders distrust anything new, and each work is made start to finish in
# a crafts hall under a master's eye. Black powder, cannon, handguns, the
# gyrocopter and the steam borer are its great works. Zhufbar is its heart.
# The western holds and the Imperial Dwarf clans share some of its knowledge
# with manlings, most of all in Nuln, home of the Imperial Engineers School.
# Sources: WFRP 4E Dwarf Player's Guide (Steel and Fire, p.87-90; Dwarf
# Engineer career, p.56).
#
# Joining and AI recruits                engineer.1 - engineer.6
# Experiments and masterworks            engineer.10 - engineer.18
# Becoming an engineer                   engineer.19, engineer.24
# Zhufbar's call                         engineer.20 - engineer.22
# Hiring an engineer                     engineer.30 - engineer.31
# The Imperial Engineers School          engineer.40
#
# Fired from common/on_actions/wh_on_actions.txt: engineer.1 on
# on_character_ask_to_join_society, engineer.5 and engineer.20 on the
# five-year pulse, engineer.24 on the bi-yearly pulse. Decisions in decisions/WHG_engineer_decisions.txt.
# Scripted triggers: common/scripted_triggers/wh_engineer_triggers.txt
# Event modifiers: common/event_modifiers/wh_engineer_modifiers.txt
# Loc: localisation/00_WHG_engineer_events.csv
###############################################################################

# engineer.1 Asking to join the guild
# Scope: ROOT is the applicant, FROM is the society.
character_event = {
	id = engineer.1
	hide_window = yes
	is_triggered_only = yes # on_character_ask_to_join_society

	trigger = {
		FROM = { leader = { society_member_of = engineers_guild } }
	}

	immediate = {
		FROM = { leader = { character_event = { id = engineer.2 } } }
	}
}

# engineer.5 AI recruits
# Every five years a ruler's gifted AI courtiers (and the ruler) may join.
# Scope: ROOT is a ruler.
character_event = {
	id = engineer.5
	hide_window = yes
	is_triggered_only = yes # on_five_year_pulse
	only_rulers = yes

	trigger = {
		OR = {
			culture_group = dwarf_group
			culture_group = tilean_group
			culture_group = imperial_group
		}
	}

	immediate = {
		any_courtier = {
			limit = { engineer_ai_recruit_trigger = yes }
			character_event = { id = engineer.6 }
		}
		if = { limit = { engineer_ai_recruit_trigger = yes } character_event = { id = engineer.6 } }
	}
}

# engineer.6 An AI recruit joins (or not)
# Scope: ROOT is the candidate.
character_event = {
	id = engineer.6
	hide_window = yes
	is_triggered_only = yes # engineer.5

	immediate = {
		random_list = {
			95 = { }
			5 = {
				modifier = { factor = 2 higher_tier_than = BARON }
				modifier = { factor = 3 learning = 15 }
				modifier = { factor = 2 is_smart_trigger = yes }
				modifier = { factor = 3 trait = dwarf_engineer }
				modifier = { factor = 3 trait = engineer }
				modifier = { factor = 0.5 trait = content }
				join_society = engineers_guild
			}
		}
	}
}

###############################################################################
# Experiments and masterworks
# Rank 1 members can test a new device (engineer_experiment); it may blow up,
# work modestly, or work so well no one will use it in the inventor's
# lifetime. A Guildmaster's long masterwork (engineer_masterwork) rolls on
# better odds and can yield a guild masterwork artifact once in a lifetime.
###############################################################################

# engineer.10 A new device
# From the engineer_experiment decision. The roll is hidden.
# Scope: ROOT is the engineer.
character_event = {
	id = engineer.10
	desc = EVTDESCengineer10
	picture = GFX_evt_machines_of_war
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # engineer_experiment

	option = { # Field testing
		name = EVTOPTAengineer10
		custom_tooltip = {
			text = engineer_experiment_roll_tt
			hidden_effect = {
				random_list = {
					25 = { # Blows up
						modifier = { factor = 2 has_mental_disorder_trigger = yes }
						modifier = { factor = 0.5 trait = patient }
						modifier = { factor = 0.5 trait = diligent }
						modifier = { factor = 0.7 trait = paranoid }
						modifier = { factor = 0.9 society_rank == 2 }
						modifier = { factor = 0.8 society_rank == 3 }
						modifier = { factor = 0.7 society_rank == 4 }
						modifier = { factor = 0.8 learning = 15 }
						modifier = { factor = 0.7 trait = dwarf_engineer }
						modifier = { factor = 0.5 trait = dwarf_engineer_master }
						character_event = { id = engineer.11 days = 3 random = 4 }
					}
					15 = { # Won't work
						modifier = { factor = 0.7 learning = 15 }
						character_event = { id = engineer.12 days = 3 random = 4 }
					}
					50 = { # Useful
						modifier = { factor = 2 trait = content }
						modifier = { factor = 1.2 trait = craven }
						modifier = { factor = 0.8 trait = brave }
						modifier = { factor = 0.7 trait = ambitious }
						character_event = { id = engineer.13 days = 3 random = 4 }
					}
					10 = { # Ahead of its time
						modifier = { factor = 3 trait = genius }
						modifier = { factor = 1.5 trait = quick }
						modifier = { factor = 1.5 learning = 20 }
						modifier = { factor = 0.2 trait = depressed }
						character_event = { id = engineer.14 days = 3 random = 4 }
					}
				}
			}
		}
	}
}

# Zhufbar's call
# Every five years a gifted young unlanded Dawi may be called to the crafts
# halls of Zhufbar. He joins the guild; his parents are told, and so is the
# lord of Zhufbar, who may take him into his court.
###############################################################################

# engineer.20 The call to Zhufbar
# Scope: ROOT is the young Dawi.
character_event = {
	id = engineer.20
	desc = EVTDESCengineer20
	picture = GFX_evt_dwarf_heroes
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_five_year_pulse
	culture_group = dwarf_group
	is_in_society = no
	min_age = 16
	max_age = 60

	trigger = {
		is_landed = no
		NOT = { has_character_flag = engineer_called_to_zhufbar }
		OR = { is_smart_trigger = yes learning = 12 }
		can_join_society = engineers_guild
		d_zhufbar = { holder_scope = { culture_group = dwarf_group } }
	}

	immediate = {
		set_character_flag = engineer_called_to_zhufbar
	}

	option = { # Go
		name = EVTOPTAengineer20
		join_society = engineers_guild
		father = { character_event = { id = engineer.21 } }
		mother = { character_event = { id = engineer.21 } }
		d_zhufbar = {
			holder_scope = {
				if = {
					limit = { NOT = { character = ROOT } }
					character_event = { id = engineer.22 days = 7 }
				}
			}
		}
	}
	option = { # My clan needs me
		name = EVTOPTBengineer20
		ai_chance = { factor = 0.2 }
	}
}
