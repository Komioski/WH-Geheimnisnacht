###############################################################################
# Engineers Guild events
# Namespace: engineer (was DR_engineers_guild in
# DR_engineers_guild_mnm_events.txt, dwarfengineering in
# GR_Dwarf_engineering_events.txt and dawimech in GR_study_mechanics_events.txt)
# Faction: Dwarfs
#
# The Engineers Guild (engineers_guild): joining, AI recruits, experiments,
# the rank powers' events, Zhufbar's call to gifted young Dawi, hiring an
# engineer for a court, war machines built at a court's capital, and the
# founding of the Imperial Engineers School in Altdorf, which opens the guild
# to Imperial humans.
#
# Lore: the Dwarf Engineers Guild is the most secretive of the Dawi craft
# guilds. Its clans honour Morgrim, the First Engineer, as well as Grungni.
# Its Elders distrust anything new, and each work is made start to finish in
# a crafts hall under a master's eye. Black powder, cannon, handguns, the
# gyrocopter and the steam borer are its great works. Zhufbar is its heart.
# The western holds and the Imperial Dwarf clans share some of its knowledge
# with manlings, most of all the Dawi clans of Nuln. The Imperial Engineers
# School in Altdorf was founded by the Tilean genius Leonardo of Miragliano
# with the Prince of Altdorf's gold, and is forever envious of the guild. Its
# Steam Tanks are the Empire's greatest war machines; only a handful run.
# Sources: WFRP 4E Dwarf Player's Guide (Steel and Fire, p.87-90; Dwarf
# Engineer career, p.56), Arcane Journal: Empire of Man (Nuln, Steam Tank).
#
# The guild has three homes, and every member belongs to one branch (a
# character modifier, engineer.7): the Dawi, of the holds or of the manling
# cities; the Imperial Engineers School of Altdorf, with the Tileans; and
# the Imperial Gunnery School of Nuln, which the Dawi exiles founded and
# which still pays the guild its tithe (WFRP Companion p.99-102).
#
# Joining and AI recruits                engineer.1 - engineer.6
# The guild's branches                   engineer.7 - engineer.9
# Experiments and masterworks            engineer.10 - engineer.18
# Becoming an engineer                   engineer.19, engineer.24
# Zhufbar's call                         engineer.20 - engineer.22
# Hiring an engineer                     engineer.30 - engineer.32
# War machines                           engineer.33 - engineer.36
# The Imperial Engineers School          engineer.40
# Life in the branches                   engineer.50 - engineer.58
# The branch powers                      engineer.60 - engineer.73
#
# Fired from common/on_actions/wh_on_actions.txt: engineer.1 on
# on_character_ask_to_join_society, engineer.7 on on_character_join_society
# and the society bi-yearly pulse, engineer.9 on on_character_leave_society,
# engineer.5 and engineer.20 on the five-year pulse, engineer.24 on the
# bi-yearly pulse, engineer.50 - .58 on the society bi-yearly pulse list
# engineer_branch_life.
# Decisions in decisions/WHG_engineer_decisions.txt: the society powers
# (branch powers fire .60, .63, .65, .70), hire_engineer (.30),
# hire_imperial_engineer (.32), commission_war_machine (.33),
# found_imperial_engineer_school (.40). engineer.34 - .36 are delayed from
# .33.
# Scripted triggers: common/scripted_triggers/wh_engineer_triggers.txt
# Event modifiers: common/event_modifiers/wh_engineer_modifiers.txt
# Loc: localisation/00_WHG_engineer_events.csv

###############################################################################
# The guild's branches
# Every member belongs to one branch, a character modifier that lasts while
# he stays in the guild. Dawi go to the holds or, if they live among
# manlings, to the clans of the manling cities. Tileans go to the Engineers
# School. An Imperial manling picks the Engineers School of Altdorf or the
# Gunnery School of Nuln, whichever is open; a player chooses when both are.
###############################################################################

# engineer.7 A new member's branch
# Also on the society bi-yearly pulse, for members who joined at game start.
# Scope: ROOT is the member.
character_event = {
	id = engineer.7
	hide_window = yes
	is_triggered_only = yes # on_character_join_society, on_society_bi_yearly_pulse

	trigger = {
		society_member_of = engineers_guild
		engineer_has_branch_trigger = no
	}

	immediate = {
		if = {
			limit = { trait = creature_dwarf }
			if = {
				limit = { engineer_dawi_among_manlings_trigger = yes }
				add_character_modifier = { name = engineer_branch_manling_dawi duration = -1 }
			}
			else = { add_character_modifier = { name = engineer_branch_hold duration = -1 } }
		}
		else_if = {
			limit = { has_character_flag = engineer_from_gunnery }
			add_character_modifier = { name = engineer_branch_gunnery duration = -1 }
		}
		else_if = {
			limit = {
				OR = { has_character_flag = engineer_from_school culture_group = tilean_group }
			}
			add_character_modifier = { name = engineer_branch_school duration = -1 }
		}
		else_if = {
			limit = {
				engineer_gunnery_school_open_trigger = yes
				engineer_school_open_trigger = yes
			}
			character_event = { id = engineer.8 }
		}
		else_if = {
			limit = { engineer_school_open_trigger = yes }
			add_character_modifier = { name = engineer_branch_school duration = -1 }
		}
		else = { add_character_modifier = { name = engineer_branch_gunnery duration = -1 } }
		clr_character_flag = engineer_from_gunnery
		clr_character_flag = engineer_from_school
	}
}

					limit = { NOT = { character = ROOT } }

# engineer.32 A manling engineer for the court
# From hire_imperial_engineer. A Tilean court hires one of its own. Any

# open, and part of the fee goes to the lord of Altdorf, who keeps it.

	id = engineer.32
	desc = { trigger = { culture_group = tilean_group } text = EVTDESCengineer32a }

		trigger = {
			NOT = { culture_group = tilean_group }

			NOT = { culture_group = tilean_group }
			NOT = { has_global_flag = imperial_engineering_school }

	}

	border = GFX_event_normal_frame_economy

		if = {
			limit = { OR = { culture_group = tilean_group culture_group = imperial_group } }

				random_traits = yes
				culture = ROOT
				religion = ROOT
				female = no
				age = 35
				trait = creature_human
				trait = engineer
			}
		}
		else = {
			create_character = {
				random_traits = yes
				culture = reiklander
				religion = sigmarite
				female = no
				age = 35
				trait = creature_human
				trait = engineer
			}
		}

			save_event_target_as = engineer_hired
			move_character = ROOT
			change_stewardship = 3
			change_martial = 2
		}

			limit = {
				NOT = { culture_group = tilean_group }
				has_global_flag = imperial_engineering_school
			}
			event_target:engineer_hired = { set_character_flag = engineer_from_school }

				holder_scope = { if = { limit = { NOT = { character = ROOT } } wealth = 50 } }
			}
		}

			limit = { NOT = { culture_group = tilean_group } }

			309 = { owner = { if = { limit = { NOT = { character = ROOT } } wealth = 50 } } }
		}
