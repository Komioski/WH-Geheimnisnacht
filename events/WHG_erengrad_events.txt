# erengrad.1 The Convoy's oath
# Once a game: the captain of the Erengrad Convoy swears to the ruler who
# holds Erengrad and joins their court as Convoy Master. The mercenary
# company becomes the city's own fleet.
# Scope: ROOT holds c_erengrad.
character_event = {
	id = erengrad.1
	desc = EVTDESCerengrad1
	picture = GFX_evt_merchant_ship_at_sea_republic
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # kislev.201, erengrad_charter_convoy

	trigger = {
		has_landed_title = c_erengrad
		NOT = { has_global_flag = erengrad_convoy_chartered }
	}

	immediate = {
		set_global_flag = erengrad_convoy_chartered
		if = {
			limit = { d_erengrad_convoy = { has_holder = yes } }
			d_erengrad_convoy = {
				holder_scope = { save_event_target_as = erengrad_captain }
				destroy_landed_title = THIS
			}
		}
		else = {
			create_character = {
				random_traits = yes
				dynasty = random
				religion = kislev_gods
				culture = gospodar
				female = no
				age = 40
				attributes = { martial = 10 stewardship = 12 }
				trait = creature_human
				trait = thrifty_clerk
			}
			new_character = { save_event_target_as = erengrad_captain }
		}
		activate_title = { title = d_erengrad_convoy status = no }
		event_target:erengrad_captain = {
			move_character = ROOT
			give_minor_title = title_convoy_master
		}
	}

	option = {
		name = EVTOPTAerengrad1
		prestige = 50
	}
}

# erengrad.12 The Convoy sails
# Forty galleys put out from Erengrad. The player must put the Convoy Master
# in command and sail to the chosen port within three years. A blessing at
# the Temple of Manann (free while the temple is in the ruler's debt) makes
# storms kinder.
# Scope: ROOT holds Erengrad and the charter.
character_event = {
	id = erengrad.12
	desc = EVTDESCerengrad12
	picture = GFX_evt_merchant_ship_at_sea_republic
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # erengrad.11

	immediate = {
		set_character_flag = erengrad_on_run
		add_character_modifier = { name = erengrad_run_cd years = 3 hidden = yes }
		175 = {
			spawn_fleet = {
				province = closest
				owner = ROOT
				cannot_inherit = yes
				earmark = erengrad_convoy_run
				troops = { galleys = { 40 40 } }
			}
		}
		character_event = { id = erengrad.14 days = 30 }
	}

	# Bless the ships
	option = {
		name = EVTOPTAerengrad12
		trigger = {
			OR = {
				wealth = 25
				c_erengrad = { location = { has_province_modifier = erengrad_manann_favour } }
			}
		}
		if = {
			limit = { NOT = { c_erengrad = { location = { has_province_modifier = erengrad_manann_favour } } } }
			wealth = -25
		}
		set_character_flag = erengrad_manann_blessed
		piety = 10
	}

	# Sail
	option = {
		name = EVTOPTBerengrad12
	}
}

	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # naval.1, erengrad.14

	trigger = {
		has_character_flag = erengrad_on_run
	}

	immediate = {
		clr_character_flag = erengrad_on_run
	}

	option = {
		name = EVTOPTAerengrad13
		if = {
			limit = {
				erengrad_cargo_suits_port_trigger = yes
				NOT = { has_character_flag = erengrad_cargo_spoiled }
			}
			scaled_wealth = { value = 0.6 min = 150 }
		}
		else_if = {
			limit = { erengrad_cargo_suits_port_trigger = yes }
			scaled_wealth = { value = 0.3 min = 75 }
		}
		else_if = {
			limit = { NOT = { has_character_flag = erengrad_cargo_spoiled } }
			scaled_wealth = { value = 0.35 min = 90 }
		}
		else = {
			scaled_wealth = { value = 0.15 min = 40 }
		}
		if = {
			limit = {
				OR = {
					has_character_flag = erengrad_dest_sudenburg
					has_character_flag = erengrad_dest_ulthuan
				}
			}
			scaled_wealth = { value = 0.2 min = 50 }
		}
		prestige = 50
		if = {
			limit = { has_landed_title = c_erengrad }
			random = {
				chance = 25
				c_erengrad = {
					location = { add_province_modifier = { name = erengrad_trade_boom years = 5 } }
				}
			}
		}
		erengrad_sea_chart_effect = yes
		erengrad_end_run_effect = yes
	}
}

# erengrad.14 The Convoy at sea
# Hidden. Every thirty days while the Convoy is out: has the Convoy Master
# made port (erengrad.13)? After three years the run fails (erengrad.15).
# Scope: ROOT sent the Convoy.
character_event = {
	id = erengrad.14
	hide_window = yes

	is_triggered_only = yes # erengrad.12, erengrad.14

	trigger = {
		has_character_flag = erengrad_on_run
	}

	immediate = {
		if = {
			limit = { any_courtier = { erengrad_convoy_arrived_trigger = yes } }
			character_event = { id = erengrad.13 }
		}
		else_if = {
			limit = {
				OR = {
					had_character_flag = { flag = erengrad_on_run days = 1095 }
					NOT = { erengrad_has_convoy_master_trigger = yes }
				}
			}
			character_event = { id = erengrad.15 }
		}
		else = {
			character_event = { id = erengrad.14 days = 30 }
		}
	}
}

	only_rulers = yes

	trigger = { has_landed_title = c_erengrad }

	# Raise him
	option = {
		name = EVTOPTAerengrad20
		scaled_wealth = { value = 0.2 min = 25 }
		create_character = {
			random_traits = yes
			dynasty = random
			religion = ROOT
			culture = gospodar
			female = no
			age = 45
			attributes = { stewardship = 14 diplomacy = 8 }
			trait = creature_human
			trait = midas_touched
		}
		new_character = {
			set_character_flag = kislevsoc_druzhina_raised
			join_society = boyar_duma
			opinion = { who = ROOT modifier = opinion_erengrad_guild_favoured }
		}
		any_vassal = {
			limit = {
				society_member_of = boyar_duma
				society_rank >= 2
			}
			opinion = { who = ROOT modifier = opinion_kislevsoc_upstart_raised }
		}
		ai_chance = {
			factor = 50
			modifier = { factor = 2 trait = greedy }
		}
	}

	# Refuse
	option = {
		name = EVTOPTBerengrad20
		prestige = 15
		ai_chance = {
			factor = 50
			modifier = { factor = 2 trait = proud }
		}
	}

# erengrad.32 The Dwarf engineer
# A Dwarf engineer offers to improve the Compound's guns, for a price.
# Pay (a ruler with an Erengrad battery gets a better one; rarely an
# engraved swivel gun for the treasury), or send him back to the mountains.
# Scope: ROOT holds c_erengrad.
character_event = {
	id = erengrad.32
	desc = EVTDESCerengrad32
	picture = GFX_evt_dwarf_cannon
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # on_five_year_pulse erengrad_life

	only_rulers = yes

	trigger = { has_landed_title = c_erengrad }

	# Pay him
	option = {
		name = EVTOPTAerengrad32
		trigger = { wealth = 75 }
		wealth = -75
		if = {
			limit = { has_character_modifier = erengrad_battery }
			remove_character_modifier = erengrad_battery
			add_character_modifier = { name = erengrad_battery_improved duration = -1 }
		}
		else = {
			prestige = 50
		}
		if = {
			limit = {
				NOT = { has_character_flag = erengrad_gun_given }
				NOT = { has_artifact = erengrad_swivel_gun }
			}
			random = {
				chance = 10
				set_character_flag = erengrad_gun_given
				add_artifact = erengrad_swivel_gun
			}
		}
		ai_chance = { factor = 50 }
	}

	# Send him away
	option = {
		name = EVTOPTBerengrad32
		ai_chance = { factor = 50 }

# erengrad.50 Erengrad sacked
# Hidden. When Erengrad is won by war or usurpation, at most once in twenty
# years, the guilds flee to their manors and the city is sacked. The Duma
# or the holder can rebuild it (decision erengrad_rebuild, erengrad.51).
# Scope: ROOT is the new holder; FROM the title.
character_event = {
	id = erengrad.50
	hide_window = yes

	is_triggered_only = yes # on_new_holder_usurpation

	trigger = {
		FROM = { title = c_erengrad }
		OR = {
			NOT = { has_global_flag = erengrad_sacked_recently }
			had_global_flag = { flag = erengrad_sacked_recently days = 7300 }
		}
	}

	immediate = {
		clr_global_flag = erengrad_sacked_recently
		set_global_flag = erengrad_sacked_recently
		c_erengrad = {
			location = {
				remove_province_modifier = erengrad_rebuilt
				add_province_modifier = { name = erengrad_sacked years = 30 }
			}
		}
	}
}
