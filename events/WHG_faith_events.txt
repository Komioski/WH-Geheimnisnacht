# Faith: the heads of the faiths, priests, preachers and conversions
# (namespace faith)
# Namespace: faith (was 39000 - 39999 in religious_events.txt; faith.N = old
#   id minus 39000; the sadaqah reply, 31000, is faith.800)
# Faction: Faiths & doctrines
# Serves: the Old World faiths (Part 1 - 3), Araby (Part 4, 6), every faith
#   (Part 5, 6)
#
# Part 1: The head of the faith        faith.200 - 277
#   The head of an Old World faith funds rulers who fight heathens, offers
#   absolution to kinslayers, hears imprisoned priests, sells indulgences,
#   and frowns on heathen councillors and on priest vassals who love their
#   liege more than the faith. Myrmidia has her own versions (230, 260 - 264).
# Part 2: Statues of the Old World gods faith.300 - 304
#   Ranald, Morr, Shallya, Verena and Myrmidia: a young artist asks for a
#   patron.
# Part 3: The true knight of the Lady  faith.310 - 311
# Part 4: Heresy news and Ormazd        faith.400, 499 - 501
#   The heresy notice of WHG_heresy_events; the land of an Ormazd ruler slowly
#   turns to the faith.
# Part 5: Preachers at a foreign court faith.600 - 619
#   A Lord Spiritual sent to proselytize at the capital of an independent
#   ruler of another faith asks leave to preach (faith.618), may be thrown in
#   the dungeon, and if allowed may win over the ruler or a courtier, or come
#   to blows with a rival preacher (faith.619).
# Part 6: Conversion and charity       faith.700, 800
#
# Callers: 00_on_actions.txt (faith.230 in the yearly religious list,
#   faith.700 in on_vassal_accepts_religious_conversion), wh_on_actions.txt
#   (faith.499 in the yearly list ormazd_spreads), job action
#   action_inquisition (faith.618, 619), decisions buy_indulgence_for_sins
#   (WHG_old_world_faith_decisions.txt, faith.250) and sadaqah_saddka
#   (realm_decisions.txt, faith.800), WHG_heresy_events and
#   WHG_chaos_cult_events (faith.400), WHG_maintenance_events (faith.700).
#
# Revived: the preacher chain (600 - 617) lost its vanilla hooks and never
#   fired; two job action events now start it. The indulgence (250, 251)
#   lost its decision; buy_indulgence_for_sins is restored for the Old World
#   faiths. The priests handed back to their lieges (275) only ran for
#   Catholics; it runs for the Old World faiths now.
# Fixed: the Myrmidian jailer freed only an "orthodox" priest (230); the
#   heathen councillors were looked for in the vanilla Muslim and pagan
#   groups (240, 241); the statues only came to Myrmidians although they
#   show the whole pantheon; the true knight's fall (311) was a 5-day MTTH on
#   every man in the world and now needs the knight's flag; the Ormazd
#   conversion (500) was an MTTH on every province and is a yearly list on
#   Ormazd rulers now; the execution of a letter-writing priest (222) used
#   `break`.
# Dropped: the lift of an excommunication (39252, 39253), which nothing ever
#   fired (the penance events of WHG_old_world_faith_events do that job),
#   and the commented-out vanilla heresy, holy order, crusade, jihad and
#   antipope events.
# Saved: character flags money_from_the_pope, kinslaying_offered,
#   nag_pope_to_help_me, pope_dislikes_heathen_employed,
#   patriarch_dislikes_heathen_employed, ask_for_indulgence,
#   commisioned_icon, true_knight_of_the_lady, faith_preacher_came,
#   faith_preacher_allowed.
# Loc: localisation/00_WHG_faith_events.csv (the vanilla texts with the
#   Pope, the Church, Christendom, God, Allah and Heaven moved into the
#   setting; the statues and the true knight from 00_vanilla_replacements).
###############################################################################

# faith.200 The head of the faith funds a ruler fighting the heathen
character_event = {
	id = faith.200
	desc = EVTDESCfaith200
	picture = GFX_evt_battle
	border = GFX_event_normal_frame_religion

	min_age = 16
	only_playable = yes
	religion_group = old_world_group
	war = no

	trigger = {
		controls_religion = yes

		wealth = 500
		any_playable_ruler = {
			religion = ROOT
			NOT = { wealth = 500 }
			religion_head = { character = ROOT }
			NOT = { excommunicated_for = ROOT }
			any_current_enemy = { NOT = { religion_group = PREV } }
			NOT = { war_with = ROOT }
			reverse_opinion = { who = ROOT value = 0 }
			OR = {
				NOT = { has_character_flag = money_from_the_pope }
				OR = {
					had_character_flag = { flag = money_from_the_pope days = 730 }
					AND = {
						ai = yes
						had_character_flag = { flag = money_from_the_pope days = 365 }
						any_war = {
							using_cb = religious
							OR = {
								attacker = { character = PREVPREV NOT = { war_score = -15 } }
								defender = { character = PREVPREV war_score = 15 }
							}
						}
					}
				}
			}
		}
	}

	mean_time_to_happen = {
		years = 5
		modifier = { factor = 0.5 wealth = 1000 }
		modifier = { factor = 0.5 wealth = 2000 }
	}

	option = {
		name = EVTOPTAfaith200
		ai_chance = { factor = 100 }
		wealth = -200

		random_playable_ruler = {
			limit = {
				religion = ROOT
				NOT = { wealth = 500 }
				religion_head = { character = ROOT }
				NOT = { excommunicated_for = ROOT }
				any_current_enemy = { NOT = { religion_group = PREV } }
				NOT = { war_with = ROOT }
				reverse_opinion = { who = ROOT value = 0 }
			}

			letter_event = { id = faith.201 tooltip = EVTTOOLTIPfaith201 }
		}
	}

	option = {
		name = EVTOPTBfaith200
		ai_chance = { factor = 0 }
	}
}

# faith.700 A vassal accepts a demand to convert (on_vassal_accepts_religious_conversion)
character_event = {
	id = faith.700

	hide_window = yes

	is_triggered_only = yes

	trigger = {
		NOT = { religion = FROMFROM }
		OR = {
			AND = {
				is_ruler = no
				NOT = { trait = zealous }
				religion = FROM # Will not change religion if already different from the vassal's
			}
			character = FROM
		}
	}

	immediate = {
		if = {
			limit = { character = FROM }
			opinion = { who = FROMFROM modifier = opinion_demanded_conversion }
			if = { limit = { prisoner = yes host = { character = FROMFROM } } imprison = no }
		}
	}

	option = { # No sympathy for the old religion
		name = OK
		ai_chance = { factor = 100 }
		religion = FROMFROM
	}

	option = { # Sympathy for the old religion
		name = OK
		ai_chance = {
			factor = 100
			modifier = { factor = 0 OR = { trait = zealous NOT = { age = 12 } } }
			modifier = { factor = 10 trait = cynical age = 12 }
		}
		if = { limit = { religion_group = old_world_group } add_trait = sympathy_old_world_group }
		if = {
			limit = { religion_group = ormazd_gods_group }
			add_trait = sympathy_ormazd_gods_group
		}
		if = { limit = { religion_group = cadai_group } add_trait = sympathy_cadai }
		if = { limit = { religion_group = cytharai_group } add_trait = sympathy_cytharai_group }
		if = { limit = { religion_group = dwarf_gods_group } add_trait = sympathy_dwarf_gods_group }
		if = {
			limit = { religion_group = halfling_gods_group }
			add_trait = sympathy_halfling_gods_group
		}
		if = {
			limit = { religion_group = oriental_gods_group }
			add_trait = sympathy_oriental_gods_group
		}
		religion = FROMFROM
	}
}
