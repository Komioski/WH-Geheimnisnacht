# Way of Life: the family focus
# Namespace: family (was WoL.3000 - WoL.3060 in wol_family_events.txt)
# Faction: Generic
#
# A ruler with the family focus spends time with a spouse, siblings,
# parents and children, reconciles quarrelling kin and teaches the young
# ones to get along. Each success may make the ruler prouder, grow their
# diplomacy education, and lead to Patriarch or Matriarch and in the end
# the socializer or gamer trait (family.1 - family.4). The Dawi speak of
# their clan and its grudges.
#
# Lessons of the family                family.1 - family.4
# The spouse                           family.10
# Siblings                             family.20 - family.22
# Parents                              family.30 - family.32
# Children                             family.40 - family.42
# Reconciling kin                      family.50 - family.59
# The children's friendship            family.60 - family.61
# Part 2: Bastards (namespace bastard), see its header
#                                      bastard.1 - bastard.81
# Part 3: Courtly romance (namespace romance), see its header
#                                      romance.1 - romance.11
# Part 4: Dynasty requests (namespace dynreq), see its header
#                                      dynreq.1 - dynreq.12
# Part 5: Kin who ask for land or a match (namespace kinreq), see its
#   header                             kinreq.1 - kinreq.34
# Part 6: The disobedient daughter (namespace daughter), see its header
#                                      daughter.10 - daughter.302
#
# Fired from: on_focus_pulse in WHG_pulse_on_actions.txt (family.10, 20, 30, 40,
#   50, 60). Shared triggers and effects are in WHG_focus_triggers.txt and
#   WHG_focus_effects.txt.
###############################################################################

# family.1 Lessons of the family
# Hidden. Each family success may make the ruler proud or less humble, may
# raise the diplomacy education (family.2), and moves the ruler up from
# Patriarch or Matriarch to the family lifestyle trait (family.4).
# Scope: ROOT is the ruler.
character_event = {
	id = family.1
	hide_window = yes

	is_triggered_only = yes # family.10, 20, 30, 40, 52, 60

	trigger = {
		has_dlc = "Way of Life"
	}

	immediate = {
		if = {
			limit = {
				NOR = { trait = proud trait = humble }
				random = 85
			}
			add_trait = proud
			character_event = { id = traitnote.261 } # Notify proud
		}
		if = {
			limit = {
				trait = humble
				random = 50
			}
			remove_trait = humble
			character_event = { id = traitnote.312 } # Notify no longer humble
		}
		character_event = { id = family.2 days = 7 }
		if = {
			limit = {
				OR = {
					has_character_modifier = wol_patriarch_2
					has_character_modifier = wol_matriarch_2
				}
			}
			random = {
				chance = 5
				character_event = { id = family.4 days = 4 }
			}
		}
		else_if = {
			limit = { has_character_modifier = wol_patriarch_1 }
			random = {
				chance = 20
				remove_character_modifier = wol_patriarch_1
				add_character_modifier = { modifier = wol_patriarch_2 duration = -1 }
			}
		}
		else_if = {
			limit = { has_character_modifier = wol_matriarch_1 }
			random = {
				chance = 20
				remove_character_modifier = wol_matriarch_1
				add_character_modifier = { modifier = wol_matriarch_2 duration = -1 }
			}
		}
		else = {
			random = {
				chance = 50
				if = {
					limit = { is_woman_trigger = yes }
					add_character_modifier = { modifier = wol_matriarch_1 duration = -1 }
				}
				else = {
					add_character_modifier = { modifier = wol_patriarch_1 duration = -1 }
				}
			}
		}
	}
}

# family.2 A chance to grow
# Hidden. The diplomacy education may go up a step (10% from the lowest,
# 5% from the second, 1% from the third), shown by family.3.
# Scope: ROOT is the ruler.
character_event = {
	id = family.2
	hide_window = yes

	is_triggered_only = yes # family.1

	immediate = {
		if = {
			limit = { trait = naive_appeaser }
			random = { chance = 10 character_event = { id = family.3 } }
		}
		else_if = {
			limit = { trait = underhanded_rogue }
			random = { chance = 5 character_event = { id = family.3 } }
		}
		else_if = {
			limit = { trait = charismatic_negotiator }
			random = { chance = 1 character_event = { id = family.3 } }
		}
	}
}

# family.10 Love grows
# Time with a spouse raises love on both sides; when both are fond enough
# they become lovers (only for peoples that have women). Seducers, the
# kind and the fair have it more often, the cruel less.
# Scope: ROOT is the ruler; event_target:target_spouse the spouse.
character_event = {
	id = family.10
	desc = { trigger = { religion_group = dwarf_gods_group } text = EVTDESCfamily10b }
	desc = { trigger = { NOT = { religion_group = dwarf_gods_group } } text = EVTDESCfamily10 }
	picture = GFX_evt_family
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # on_focus_pulse

	prisoner = no
	only_capable = yes
	is_married = yes

	trigger = {
		has_focus = focus_family
		family_free_for_kin_trigger = yes
		any_spouse = { family_spouse_at_hand_trigger = yes }
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 1.33 has_character_modifier = wol_seducer_1 }
		modifier = { factor = 2 has_character_modifier = wol_seducer_2 }
		modifier = {
			factor = 4
			OR = {
				trait = seducer
				trait = seductress
			}
		}
		modifier = { factor = 0.66 trait = cruel }
		modifier = { factor = 1.33 trait = kind }
		modifier = { factor = 1.33 has_fair_trait_trigger = yes }
	}

	immediate = {
		random_spouse = {
			limit = { family_spouse_at_hand_trigger = yes }
			save_event_target_as = target_spouse
		}
		random = {
			chance = 50
			character_event = { id = family.1 }
		}
	}

	option = { # Marital bliss
		name = EVTOPTAfamily10
		if = {
			limit = {
				event_target:target_spouse = {
					NOT = { personal_opinion = { who = ROOT value = 40 } }
				}
			}
			event_target:target_spouse = {
				opinion = { who = ROOT modifier = opinion_marital_love }
			}
		}
		if = {
			limit = {
				NOT = { personal_opinion = { who = event_target:target_spouse value = 40 } }
			}
			opinion = { who = event_target:target_spouse modifier = opinion_marital_love }
		}
		if = {
			limit = {
				personal_opinion = { who = event_target:target_spouse value = 40 }
				reverse_personal_opinion = { who = event_target:target_spouse value = 40 }
				is_no_females_race = no
				event_target:target_spouse = { is_no_females_race = no }
			}
			add_lover = event_target:target_spouse
		}
	}
}

# Hidden. Picks a courtier who might welcome the liege's attention and sends

	is_triggered_only = yes
	only_playable = yes
	min_age = 16

		has_regent = no

		NOT = { trait = disfigured }
		NOT = { has_character_modifier = lustful_affair_timer }
		NOT = { spouse = { is_lover = ROOT } }
		OR = {
			is_woman_trigger = no
			NOT = { age = 40 }
		}
		OR = {
			is_married = no
			NOT = { trait = chaste }
		}
		NOT = { num_of_lovers = 5 }
		any_courtier = { romance_liege_admirer_trigger = yes }
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 2 has_focus = focus_seduction }
		modifier = { factor = 0.67 num_of_lovers = 1 }
		modifier = { factor = 0.5 num_of_lovers = 2 }
		modifier = { factor = 0.25 num_of_lovers = 3 }
		modifier = { factor = 0.2 num_of_lovers = 4 }
		modifier = { factor = 0.25 religion_group = ormazd_gods_group }
		modifier = { factor = 3 trait = homosexual } # Fewer partners to choose from
		modifier = { factor = 0.5 spouse = { NOT = { age = 30 } } }
		modifier = { factor = 2 spouse = { age = 45 } }
		modifier = { factor = 2 trait = hedonist }
		modifier = { factor = 2 trait = lustful }
		modifier = { factor = 2 has_fair_trait_trigger = yes }
		modifier = { factor = 0.5 trait = shy }
		modifier = { factor = 0.5 has_ugly_trait_trigger = yes }
		modifier = { factor = 0.1 trait = chaste }
	}

	immediate = {
		random_courtier = {
			limit = { romance_liege_admirer_trigger = yes }
			score_value = { # The more interesting courtiers come first
				value = 10
				interesting_courtier_selection_score = yes

# daughter.10 An unwanted marriage
# Hidden, on_marriage. A strong-willed daughter of a player is married to
# someone she does not love and becomes disobedient.
# Scope: ROOT is her parent, FROM the daughter.
character_event = {
	id = daughter.10

	hide_window = yes

	is_triggered_only = yes
	has_dlc = "Zeus"

	trigger = {
		FROM = {
			is_woman_trigger = yes
			NOT = { new_character = { is_lover = FROM } }
			daughter_strong_willed_trigger = yes
			OR = { father = { character = ROOT ai = no } mother = { character = ROOT ai = no } }
		}
	}

	weight_multiplier = {
		days = 1

		modifier = { factor = 3 OR = { trait = celibate trait = homosexual } }
	}

	immediate = {
		FROM = { set_character_flag = disobedient_daughter }
		character_event = { id = daughter.11 days = 2 }
	}
}

# daughter.11 Annul the marriage
# The daughter begs her parent to annul the match. Refuse (she stays
# unhappy, may take a lover or a vow) or annul it at a cost in piety and
# prestige.
# Scope: ROOT is the parent or her liege, FROMFROM the daughter.
character_event = {
	id = daughter.11
	title = EVTTITLEdaughter11
	desc = EVTDESCdaughter11
	picture = GFX_evt_daughter_asking_father
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	option = {
		name = EVTOPTAdaughter11 # Tell her she must accept it
		FROMFROM = {
			set_character_flag = disobedient_daughter_unhappy
			if = {
				limit = { has_lover = no NOT = { trait = celibate } }
				random_list = {
					10 = { add_trait = celibate }
					10 = {
						trigger = {
							ROOT = {
								any_courtier = {
									is_ill = no
									NOT = { trait = incapable }
									NOT = { trait = celibate }
									NOT = { has_ugly_trait_trigger = yes }
									age = 16
									NOT = { age = 50 }
									is_no_females_race = no
								}
							}
						}
						custom_tooltip = { text = disobedient_daughter_takes_lover }
						ROOT = {
							random_courtier = {
								limit = {
									is_ill = no
									NOT = { trait = incapable }
									NOT = { trait = celibate }
									NOT = { has_ugly_trait_trigger = yes }
									age = 16
									NOT = { age = 50 }
									is_no_females_race = no
								}
								if = { limit = { is_no_females_race = no FROMFROM = { is_no_females_race = no } } add_lover = FROMFROM }
							}
						}
					}
					10 = { }
				}
			}
			opinion = { who = ROOT modifier = opinion_angry years = 3 }
			spouse = { reverse_opinion = { who = FROMFROM modifier = opinion_hate years = 100 } }
		}
	}
	option = {
		name = EVTOPTBdaughter11 # Annul the marriage immediately
		piety = -50
		prestige = -100
		FROMFROM = {
			piety = -50
			prestige = -50
			spouse = {
				save_event_target_as = target_spurned_spouse
				employer = {
					opinion = { who = ROOT modifier = grievously_offended years = 5 }
					hidden_tooltip = { character_event = { id = daughter.12 } }
				}
			}
			remove_spouse = spouse
		}
	}
}

# daughter.30 A lover killed
# Hidden, on_death. A player executed, murdered, starved or sacrificed their
# daughter's lover.
# Scope: ROOT is the dead lover.
character_event = {
	id = daughter.30

	hide_window = yes

	is_triggered_only = yes

	only_men = yes
	has_dlc = "Zeus"

	trigger = {
		has_lover = yes
		OR = {
			death_execution_trigger = yes
			death_murder_known_trigger = yes
			death_reason = death_dungeon
			death_sacrificed_trigger = yes
		}
		any_lover = {
			OR = {
				father = { ROOT = { killer = { character = PREVPREV } } ai = no }
				mother = { ROOT = { killer = { character = PREVPREV } } ai = no }
			}
			is_woman_trigger = yes
			NAND = {
				ROOT = { death_sacrificed_trigger = yes }
				trait = zealous
				ROOT = { killer = { religion = PREVPREV } }
			}
			daughter_strong_willed_trigger = yes
		}
	}

	immediate = {
		random_lover = {
			limit = {
				is_woman_trigger = yes
				OR = {
					father = { ROOT = { killer = { character = PREVPREV } } ai = no }
					mother = { ROOT = { killer = { character = PREVPREV } } ai = no }
				}
				NAND = {
					ROOT = { death_sacrificed_trigger = yes }
					trait = zealous
					ROOT = { killer = { religion = PREVPREV } }
				}
				daughter_strong_willed_trigger = yes
			}
			save_event_target_as = target_daughter_killed_her_friend
		}
		random = { chance = 30 killer = { character_event = { id = daughter.31 days = 5 } } }
	}
}

# daughter.101 The unhappy daughter acts
# Hidden. Weighted pick: poison or duel her husband, elope abroad, found a
# mercenary band, take a lover, torment her spouse, turn heretic or die in
# prison.
# Scope: ROOT is the daughter.
character_event = {
	id = daughter.101

	hide_window = yes

	is_triggered_only = yes

	immediate = {
		random_list = {
			10 = { # Murders spouse in sleep
				modifier = { factor = 0 prisoner = yes }
				modifier = { factor = 0 OR = { is_married = no spouse = { is_lover = ROOT } } }
				modifier = { factor = 0 spouse = { ai = no } }
				modifier = { factor = 0 trait = kind }
				modifier = { factor = 2 trait = wroth }
				modifier = { factor = 2 trait = cruel }
				modifier = {
					factor = 0.5
					spouse = {
						attribute_diff = { attribute = intrigue character = PREV value = 4 }
					}
				}
				modifier = {
					factor = 0.5
					spouse = {
						attribute_diff = { attribute = intrigue character = PREV value = 8 }
					}
				}
				modifier = {
					factor = 2
					spouse = {
						attribute_diff = { attribute = intrigue character = PREV value = -4 }
					}
				}
				modifier = {
					factor = 2
					spouse = {
						attribute_diff = { attribute = intrigue character = PREV value = -8 }
					}
				}
				if = {
					limit = { father = { ai = no } }
					father = { character_event = { id = daughter.120 } }
				}
				else_if = {
					limit = { mother = { ai = no } }
					mother = { character_event = { id = daughter.120 } }
				}
			}
			2 = { # Kills spouse in single combat - only if she is more skilled
				modifier = { factor = 0 prisoner = yes }
				modifier = {
					factor = 0
					OR = {
						is_married = no
						spouse = { is_lover = ROOT }
						spouse = { combat_rating_diff = { who = ROOT value = 0 } }
					}
				}
				modifier = { factor = 0 spouse = { ai = no } }
				modifier = { factor = 0 trait = craven }
				modifier = {
					factor = 10
					spouse = { combat_rating_diff = { who = ROOT value = -40 } }
				}
				modifier = { factor = 1.5 trait = brave }
				if = {
					limit = { father = { ai = no is_liege_of = ROOT } }
					father = { character_event = { id = daughter.130 } }
				}
				else_if = {
					limit = { mother = { ai = no is_liege_of = ROOT } }
					mother = { character_event = { id = daughter.130 } }
				}
			}
			5 = { # Elope and settle abroad (divorce spouse, marry any unmarried lover)
				modifier = { factor = 0 prisoner = yes }
				modifier = { factor = 0 spouse = { ai = no } }
				modifier = { factor = 0 trait = craven }
				modifier = {
					factor = 0
					NOT = { any_independent_ruler = { religion_group = ROOT ai = yes } }
				}
				character_event = { id = daughter.140 }
			}
			5 = { # Elope and start pirate band (mercs, divorce spouse, marry any lover)
				modifier = { factor = 0 prisoner = yes }
				modifier = { factor = 0 spouse = { ai = no } }
				modifier = { factor = 0 trait = craven }
				modifier = { factor = 2 martial = 9 }
				modifier = { factor = 3 martial = 15 }
				if = {
					limit = { father = { ai = no is_liege_of = ROOT } }
					father = { character_event = { id = daughter.150 } }
				}
				else_if = {
					limit = { mother = { ai = no is_liege_of = ROOT } }
					mother = { character_event = { id = daughter.150 } }
				}
			}
			10 = { # Take new lover (remove any chaste trait)
				modifier = { factor = 0 prisoner = yes }
				modifier = {
					factor = 0
					NOT = {
						liege = {
							any_courtier = {
								ai = yes
								is_valid_romance_target = ROOT
								is_no_females_race = no
								prisoner = no
								NOT = { character = ROOT }
								NOT = { num_of_lovers = 5 }
							}
						}
					}
				}
				modifier = { factor = 0 num_of_lovers = 5 }
				modifier = { factor = 0 trait = celibate }
				modifier = { factor = 3 trait = lustful }
				modifier = { factor = 1.5 trait = gregarious }
				modifier = { factor = 1.5 has_fair_trait_trigger = yes }
				remove_trait = chaste
				liege = {
					random_courtier = {
						limit = {
							ai = yes
							is_valid_romance_target = ROOT is_no_females_race = no
							is_no_females_race = no
							prisoner = no
							NOT = { character = ROOT }
							NOT = { num_of_lovers = 5 }
						}
						end_inaccessibility_effect = yes
						if = { limit = { is_no_females_race = no ROOT = { is_no_females_race = no } } add_lover = ROOT }
					}
				}
			}
			10 = { # Make spouse stressed, depressed or insane...
				modifier = { factor = 0 prisoner = yes }
				modifier = { factor = 0 is_married = no }
				modifier = { factor = 0 spouse = { ai = no } }
				modifier = { factor = 0 trait = kind }
				modifier = { factor = 0.5 trait = humble }
				modifier = {
					factor = 2
					OR = { trait = lunatic_1 trait = lunatic_2 trait = lunatic_3 }
				}
				modifier = { factor = 2 trait = possessed }
				modifier = { factor = 2 trait = wroth }
				modifier = { factor = 2 trait = cruel }
				spouse = {
					if = {
						limit = { trait = depressed }
						add_trait = lunatic_1
						set_variable = { which = "lunatic_counter" value = 5 }
					}
					if = {
						limit = {
							NOR = { trait = lunatic_1 trait = lunatic_2 trait = lunatic_3 }
							trait = stressed
						}
						add_trait = depressed
					}
					if = {
						limit = {
							NOR = { trait = lunatic_1 trait = lunatic_2 trait = lunatic_3 }
							NOT = { trait = depressed }
						}
						add_trait = stressed
					}
				}
			}
			10 = { # Become a heretic, spit on parent's religion
				modifier = { factor = 0 NOT = { religion = FROM } }
				modifier = { factor = 0 religion_group = chaos_gods_group }
				modifier = { factor = 0 religion_group = greenskin_gods_group } # greenskins never leave their gods
				modifier = { factor = 0.5 trait = zealous }
				modifier = { factor = 2 trait = cynical }
				modifier = { factor = 2 trait = brave }
				character_event = { id = daughter.180 }
			}
			5 = { # If imprisoned, commit suicide in prison
				modifier = { factor = 0 prisoner = no }
				modifier = {
					factor = 0
					NOT = { trait = stressed }
					NOT = { trait = depressed }
					NOR = { trait = lunatic_1 trait = lunatic_2 trait = lunatic_3 }
				}
				modifier = { factor = 3 trait = depressed }
				modifier = {
					factor = 2
					OR = { trait = lunatic_1 trait = lunatic_2 trait = lunatic_3 }
				}
				character_event = { id = daughter.190 }
			}
			30 = { # No action
			}
		}
	}
}
