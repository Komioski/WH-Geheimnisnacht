###############################################################################
# Feasts
# Namespace: feast (was 72000 - 72999, 4160 and 4165 in the vanilla
#   feast_events.txt)
# Faction: Generic
# Serves: the lords who hold the Large Feast decision (minor_decisions.txt
#   feast_0): men, elves, dwarfs, halflings and the Tomb Kings' living
#   subjects, but not the Druchii, Norscans or Kislevites, who feast their
#   own way.
#
# A lord calls a feast. For a fortnight the steward haggles with butchers,
# troupes of Strigany jugglers and travelling fire-eaters ask for a place,
# the marshal rides out after a boar or a robber band, and the cellars are
# counted. Then the vassals ride in, or send their regrets. Over three nights
# a troupe performs, the food is praised or mocked, guests drink, quarrel,
# fight duels, make friends over shared grudges, flirt with each other's
# spouses, kill a servant, and lose their hearts to the wrong daughter. A
# guest may challenge the host to a drinking contest. On the twentieth day
# the guests ride home with a good word for their host.
#
# Planning                feast.1
# Troupes ask for a place feast.10
# Preparations            feast.20 - feast.23
# Invitations             feast.30 - feast.32
# The guests arrive       feast.40 - feast.45 (43 - 45 are the three nights)
# Food and shows          feast.50 - feast.61
# Friends at table        feast.80 - feast.85
# Duels                   feast.90 - feast.97
# Wine                    feast.100 - feast.105
# Robbers on the road     feast.110 - feast.111
# A servant killed        feast.115 - feast.117
# The jester              feast.120 - feast.122
# A claim on a guest      feast.125 - feast.129
# Affairs                 feast.130 - feast.137
# A daughter seduced      feast.140 - feast.148
# The end of the feast    feast.160 - feast.164, feast.199
# A drinking contest      feast.170
#
# Callers: decisions/minor_decisions.txt feast_0 (feast.1 at once, the safety
#   catch feast.199 after 150 days); on_war_started in WHG_war_on_actions.txt
#   (feast.164, which cancels a feast when its host goes to war).
# Was: 72109 -> 1; 72000, 72001, 72002, 72006, 72007, 72009 and 72120 (seven
#   copies of the same offer) -> 10; 72003 -> 20, 72004 -> 21, 72005 -> 22,
#   72008 -> 23; 72010 - 72012 -> 30 - 32; 72015 -> 40 (its three waves are
#   now 43 - 45), 72016 -> 41; 72020 - 72024 -> 50 - 54; 72063 - 72072,
#   4160, 72115, 72116 and 4165 (one show and one reaction per troupe) -> 60,
#   61; 72025 - 72028 -> 80 - 83, 72080, 72081 -> 84, 85; 72030 - 72037 ->
#   90 - 97; 72045 -> 100, 72052 -> 101, 72046 - 72049 -> 102 - 105;
#   72050, 72051 -> 110, 111; 72055 - 72057 -> 115 - 117; 72060 - 72062 ->
#   120 - 122; 72085 - 72089 -> 125 - 129; 72040 - 72042 -> 130 - 132,
#   72075 - 72077 -> 135 - 137; 72090 - 72100 -> 140 - 148 (the four
#   reaction events are now two); 72110 - 72113 -> 160 - 163; 72999 ->
#   199. The cancel was a 1-day MTTH; it now runs from
#   on_war_started and from each night.
# Saved: character flags holding the plan (lavishly_food, medium_food,
#   low_food, feast_<troupe>, feast_offered_<troupe>, feast_offer_now_<troupe>,
#   feast_show_<troupe>, sent_invitations, host_feast_started, castle_problems,
#   feast_boar_hunt, feast_drunkard_guest, feast_duel,
#   feast_drinking_contest, feast_planned) and on guests (coming_to_feast,
#   guest_feast_started, feast_duel_wounded, feast_duel_maimed,
#   feast_duel_killed), all cleared by feast_clear_host_effect and
#   feast_clear_guest_effect; event targets feast_steward, feast_marshal,
#   book_target, potential_cannibal, feast_friend, feast_duel_foe,
#   feast_flirt, feast_affair_woman, feast_lady, feast_daughter,
#   feast_daughter_parent, feast_seducer, feast_love_host, feast_jester,
#   feast_claim_title, feast_claim_holder, feast_rival_drinker.
# Triggers: common/scripted_triggers/WHG_feast_triggers.txt
# Effects: common/scripted_effects/WHG_feast_effects.txt
# Loc: localisation/00_WHG_feast_events.csv

# feast.1 A feast is called
# The host sets the date. Over the next fortnight one or two troupes ask for
# a place, the marshal may ride out after a boar or robbers, the steward may
# find the hall in want of repair, and then the food is chosen.
# Scope: the host. Fired by the feast_0 decision (minor_decisions.txt).
character_event = {
	id = feast.1
	desc = EVTDESCfeast1
	picture = GFX_evt_feast

	is_triggered_only = yes # feast_0 decision

	option = {
		name = EVTOPTAfeast1
		set_character_flag = feast_planned
		hidden_effect = {
			character_event = { id = feast.10 days = 5 }
			random_list = {
				70 = { character_event = { id = feast.10 days = 10 } }
				5 = {
					modifier = { factor = 0 NOT = { job_treasurer = { always = yes } } }
					character_event = { id = feast.21 days = 10 }
				}
				5 = {
					modifier = { factor = 0 NOT = { job_marshal = { always = yes } } }
					modifier = {
						factor = 0
						NOT = { location = { has_province_modifier = highway_robber_band } }
					}
					character_event = { id = feast.22 days = 10 }
				}
				30 = {
					modifier = { factor = 0 NOT = { job_marshal = { always = yes } } }
					character_event = { id = feast.23 days = 10 }
				}
			}
			character_event = { id = feast.20 days = 15 random = 2 }
		}
	}
}

# feast.10 A troupe asks for a place
# One troupe not yet heard this feast asks to perform: Strigany jugglers, a
# minstrel, musicians, fire-eaters, a conjurer, acrobats or dancers. Hired,
# they may perform on the first night (feast.43). In the lands of Sigmar a
# conjurer is a risk: he swears his tricks are sleight of hand, but the
# witch hunters do not always ask before they burn.
# Scope: the host. Fired by feast.1 (once or twice).
character_event = {
	id = feast.10
	desc = { trigger = { has_character_flag = feast_offer_now_jongleur } text = EVTDESCfeast10 }
	desc = { trigger = { has_character_flag = feast_offer_now_minstrel } text = EVTDESCfeast10b }
	desc = { trigger = { has_character_flag = feast_offer_now_musicians } text = EVTDESCfeast10c }
	desc = { trigger = { has_character_flag = feast_offer_now_fireeaters } text = EVTDESCfeast10d }
	desc = { trigger = { has_character_flag = feast_offer_now_conjurer } text = EVTDESCfeast10e }
	desc = { trigger = { has_character_flag = feast_offer_now_acrobats } text = EVTDESCfeast10f }
	desc = { trigger = { has_character_flag = feast_offer_now_dancers } text = EVTDESCfeast10g }
	picture = GFX_evt_jugglers

	is_triggered_only = yes # feast.1

	only_rulers = yes
	capable_only = yes
	prisoner = no
	war = no

	trigger = {
		has_character_modifier = holding_large_feast
		NOT = { has_character_flag = sent_invitations }
		NOT = { has_character_flag = host_feast_started }
	}

	immediate = {
		feast_clear_offer_now_effect = yes
		random_list = {
			10 = {
				modifier = { factor = 0 has_character_flag = feast_offered_jongleur }
				set_character_flag = feast_offer_now_jongleur
				set_character_flag = feast_offered_jongleur
			}
			10 = {
				modifier = { factor = 0 has_character_flag = feast_offered_minstrel }
				set_character_flag = feast_offer_now_minstrel
				set_character_flag = feast_offered_minstrel
			}
			10 = {
				modifier = { factor = 0 has_character_flag = feast_offered_musicians }
				set_character_flag = feast_offer_now_musicians
				set_character_flag = feast_offered_musicians
			}
			10 = {
				modifier = { factor = 0 has_character_flag = feast_offered_fireeaters }
				set_character_flag = feast_offer_now_fireeaters
				set_character_flag = feast_offered_fireeaters
			}
			10 = {
				modifier = { factor = 0 has_character_flag = feast_offered_conjurer }
				set_character_flag = feast_offer_now_conjurer
				set_character_flag = feast_offered_conjurer
			}
			10 = {
				modifier = { factor = 0 has_character_flag = feast_offered_acrobats }
				set_character_flag = feast_offer_now_acrobats
				set_character_flag = feast_offered_acrobats
			}
			10 = {
				modifier = { factor = 0 has_character_flag = feast_offered_dancers }
				set_character_flag = feast_offer_now_dancers
				set_character_flag = feast_offered_dancers
			}
		}
	}

	# Hire them
	option = {
		name = EVTOPTAfeast10
		wealth = -10
		if = {
			limit = { has_character_flag = feast_offer_now_jongleur }
			set_character_flag = feast_jongleur
		}
		else_if = {
			limit = { has_character_flag = feast_offer_now_minstrel }
			set_character_flag = feast_minstrel
		}
		else_if = {
			limit = { has_character_flag = feast_offer_now_musicians }
			set_character_flag = feast_musicians
		}
		else_if = {
			limit = { has_character_flag = feast_offer_now_fireeaters }
			set_character_flag = feast_fireeaters
		}
		else_if = {
			limit = { has_character_flag = feast_offer_now_conjurer }
			set_character_flag = feast_conjurer
			if = {
				limit = { religion_group = old_world_group }
				random = { chance = 15 custom_tooltip = { text = EVTTOOLTIPfeast10 } piety = -10 }
			}
		}
		else_if = {
			limit = { has_character_flag = feast_offer_now_acrobats }
			set_character_flag = feast_acrobats
		}
		else = { set_character_flag = feast_dancers }
		hidden_effect = { feast_clear_offer_now_effect = yes }
		ai_chance = {
			factor = 5
			modifier = { factor = 20 wealth = 30 }
			modifier = { factor = 0.1 NOT = { wealth = 30 } }
			modifier = {
				factor = 0.2
				has_character_flag = feast_offer_now_conjurer
				religion_group = old_world_group
				trait = zealous
			}
		}
	}

	# Send them on their way
	option = {
		name = EVTOPTBfeast10
		prestige = -5
		hidden_effect = { feast_clear_offer_now_effect = yes }
		ai_chance = {
			factor = 5
			modifier = { factor = 20 NOT = { wealth = 30 } }
			modifier = { factor = 0.1 wealth = 30 }
		}
	}

	# The conjurer: hand him to the witch hunters
	option = {
		name = EVTOPTCfeast10
		trigger = { has_character_flag = feast_offer_now_conjurer religion_group = old_world_group }
		piety = 10
		hidden_effect = { feast_clear_offer_now_effect = yes }
		ai_chance = {
			factor = 2
			modifier = { factor = 5 trait = zealous }
			modifier = { factor = 3 trait = cruel }
			modifier = { factor = 0.2 trait = cynical }
			modifier = { factor = 0.2 trait = kind }
		}
	}
}

# feast.20 The kitchens
# The host chooses how lavish the feast will be, and so how the guests will
# judge it. A treasured cookbook in the host's hands offers a fourth way; the
# book called the Omnivore teaches its readers a taste for the forbidden.
# Then the invitations go out (feast.30).
# Scope: the host. Fired by feast.1.
character_event = {
	id = feast.20
	desc = {
		trigger = { OR = { trait = creature_dwarf trait = creature_chaos_dwarf } }
		text = EVTDESCfeast20b
	}
	desc = { trigger = { trait = creature_halfling } text = EVTDESCfeast20c }
	desc = {
		trigger = {
			NOR = { trait = creature_dwarf trait = creature_chaos_dwarf trait = creature_halfling }
		}
		text = EVTDESCfeast20
	}
	picture = GFX_evt_feast

	is_triggered_only = yes # feast.1

	only_rulers = yes
	capable_only = yes
	prisoner = no
	war = no

	trigger = {
		has_character_modifier = holding_large_feast
		NOR = {
			has_character_flag = lavishly_food
			has_character_flag = medium_food
			has_character_flag = low_food
		}
	}

	immediate = {
		if = {
			limit = { any_artifact = { has_artifact_flag = cooking } }
			random_artifact = {
				limit = {
					has_artifact_flag = cooking
					NOT = { artifact_type = gastronomic_tonics }
					OR = {
						is_artifact_equipped = yes
						AND = {
							artifact_type = the_omnivore
							PREV = { NOT = { trait = cannibal_trait } }
						}
						AND = {
							artifact_type = the_omnivore
							PREV = {
								any_courtier = {
									NOT = { trait = cannibal_trait }
									is_adult = yes
									is_inaccessible_or_incapable_trigger = no
								}
							}
						}
					}
				}
				preferred_limit = { quality = 3 }
				preferred_limit = { quality = 2 }
				preferred_limit = { quality = 1 }
				save_event_target_as = book_target
			}
			if = {
				limit = {
					event_target:book_target = { artifact_type = the_omnivore }
					trait = cannibal_trait
				}
				random_courtier = {
					limit = {
						NOT = { trait = cannibal_trait }
						is_adult = yes
						is_inaccessible_or_incapable_trigger = no
					}
					preferred_limit = {
						has_education_stewardship_trigger = yes
						family_of_root_trigger = yes
						is_evil_trigger = yes
					}
					preferred_limit = {
						has_education_stewardship_trigger = yes
						family_of_root_trigger = yes
					}
					preferred_limit = { family_of_root_trigger = yes }
					preferred_limit = { has_education_stewardship_trigger = yes }
					save_event_target_as = potential_cannibal
				}
			}
		}
	}

	# Spare nothing
	option = {
		name = EVTOPTAfeast20
		trigger = { NOT = { trait = temperate } }
		set_character_flag = lavishly_food
		wealth = -20
		prestige = 10
		gain_weight_small_effect = yes
		weight_trait_check_effect = yes
		hidden_effect = { character_event = { id = feast.30 days = 2 random = 5 } }
		ai_chance = {
			factor = 30
			modifier = { factor = 0.5 trait = greedy }
			modifier = { factor = 3 trait = gluttonous }
			modifier = { factor = 2 primary_title = { higher_tier_than = DUKE } }
			modifier = { factor = 0.1 NOT = { wealth = 50 } }
		}
	}

	# A good table, no more
	option = {
		name = EVTOPTBfeast20
		set_character_flag = medium_food
		wealth = -10
		hidden_effect = { character_event = { id = feast.30 days = 2 random = 5 } }
		ai_chance = {
			factor = 30
			modifier = { factor = 2 trait = greedy }
			modifier = { factor = 2 primary_title = { tier = DUKE } }
		}
	}

	# Bread and small beer
	option = {
		name = EVTOPTCfeast20
		trigger = { NOT = { trait = gluttonous } }
		set_character_flag = low_food
		wealth = -5
		prestige = -5
		hidden_effect = { character_event = { id = feast.30 days = 2 random = 5 } }
		ai_chance = {
			factor = 30
			modifier = { factor = 2.5 trait = greedy }
			modifier = { factor = 2 primary_title = { tier = COUNT } }
			modifier = { factor = 3 NOT = { wealth = 20 } }
		}
	}

	# Cook from the book
	option = {
		name = { text = EVTOPTDfeast20 trigger = { event_target:book_target = { quality > 1 } } }
		name = { text = EVTOPTDfeast20b trigger = { event_target:book_target = { quality < 2 } } }
		trigger = { event_target:book_target = { always = yes } }
		tooltip_info_custom = available_because_books_tt
		if = {
			limit = { event_target:book_target = { quality < 2 } }
			set_character_flag = low_food
			prestige = -5
		}
		else_if = {
			limit = { event_target:book_target = { artifact_type = the_omnivore } }
			if = {
				limit = { NOT = { trait = cannibal_trait } }
				custom_tooltip = { text = EVTTOOLTIPfeast20 }
				add_trait = cannibal_trait
			}
			else_if = {
				limit = { event_target:potential_cannibal = { always = yes } }
				custom_tooltip = { text = EVTTOOLTIPfeast20b }
				event_target:potential_cannibal = {
					show_scope_change = no
					add_trait = cannibal_trait
				}
			}
			set_character_flag = lavishly_food
			wealth = -10
			prestige = 20
		}
		else_if = {
			limit = { event_target:book_target = { NOT = { quality = 3 } } }
			set_character_flag = medium_food
			wealth = -5
			prestige = 5
		}
		else = { set_character_flag = lavishly_food wealth = -10 prestige = 20 }
		hidden_effect = { character_event = { id = feast.30 days = 2 random = 5 } }
		ai_chance = {
			factor = 50
			modifier = {
				factor = 0
				event_target:book_target = { artifact_type = the_omnivore }
				NOT = { trait = cannibal_trait }
				NOT = { is_evil_trigger = yes }
			}
		}
	}
}

# feast.170 A drinking contest
# A guest, often a dwarf, slams a tankard down and challenges the host to
# match them cup for cup. The host may drink, decline, or quietly water the
# wine; dwarfs and drunkards hold their drink best.
# Scope: the host. Fired by feast.44.
character_event = {
	id = feast.170
	desc = {
		trigger = { event_target:feast_rival_drinker = { trait = creature_dwarf } }
		text = EVTDESCfeast170b
	}
	desc = {
		trigger = { event_target:feast_rival_drinker = { NOT = { trait = creature_dwarf } } }
		text = EVTDESCfeast170
	}
	picture = GFX_evt_drunk

	is_triggered_only = yes # feast.44

	capable_only = yes
	prisoner = no
	war = no

	trigger = {
		feast_host_trigger = yes
		is_adult = yes
		NOT = { has_character_flag = feast_drinking_contest }
		any_vassal = { feast_guest_trigger = yes is_adult = yes }
	}

	immediate = {
		set_character_flag = feast_drinking_contest
		random_vassal = {
			limit = { feast_guest_trigger = yes is_adult = yes }
			preferred_limit = { trait = creature_dwarf }
			preferred_limit = { trait = drunkard }
			preferred_limit = { trait = gluttonous }
			save_event_target_as = feast_rival_drinker
		}
	}

	# Fill the tankards
	option = {
		name = EVTOPTAfeast170
		random_list = {
			50 = {
				modifier = { factor = 1.5 trait = drunkard }
				modifier = { factor = 2 trait = creature_dwarf }
				modifier = { factor = 1.5 trait = gluttonous }
				modifier = { factor = 0.5 trait = temperate }
				modifier = {
					factor = 0.5
					event_target:feast_rival_drinker = { trait = creature_dwarf }
				}
				modifier = { factor = 0.7 event_target:feast_rival_drinker = { trait = drunkard } }
				custom_tooltip = { text = EVTTOOLTIPfeast170 }
				prestige = 20
				event_target:feast_rival_drinker = {
					opinion = { modifier = opinion_approves who = ROOT years = 5 }
				}
			}
			50 = {
				custom_tooltip = { text = EVTTOOLTIPfeast170b }
				prestige = -10
				event_target:feast_rival_drinker = { prestige = 10 }
				if = {
					limit = { NOT = { trait = drunkard } }
					random = {
						chance = 20
						add_trait = drunkard
						hidden_effect = { character_event = { id = traitnote.286 } } # Notify drunkard
					}
				}
			}
		}
		ai_chance = {
			factor = 10
			modifier = { factor = 2 trait = creature_dwarf }
			modifier = { factor = 0.2 trait = temperate }
		}
	}

	# I keep a clear head
	option = {
		name = EVTOPTBfeast170
		prestige = -5
		ai_chance = { factor = 3 modifier = { factor = 5 trait = temperate } }
	}

	# Have the cupbearer water my wine
	option = {
		name = EVTOPTCfeast170
		trigger = { intrigue = 12 }
		random_list = {
			70 = { custom_tooltip = { text = EVTTOOLTIPfeast170 } prestige = 15 }
			30 = {
				custom_tooltip = { text = EVTTOOLTIPfeast170c }
				prestige = -25
				event_target:feast_rival_drinker = {
					opinion = { modifier = opinion_dislike who = ROOT years = 10 }
				}
			}

			factor = 5
			modifier = { factor = 3 trait = deceitful }
			modifier = { factor = 0 trait = honest }
		}
