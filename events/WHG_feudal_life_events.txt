###############################################################################
# Feudal life
# Namespace: feudal (was 4000 - 4175 in the vanilla feudal_life_events.txt)
# Faction: Generic
# Serves: the counts and dukes of the settled peoples (not the greenskins,
#   Skaven, Lizardmen, Fimir, Tomb Kings or beastmen, who keep no such courts)
#
# The small business of a lordship: judging thieves, poachers, witches and
# blasphemous cooks, a dragon in the hills, a hostage taken for ransom; and
# its pleasures: hawking, banquets, markets, Bretonnian jousts and tourneys,
# village festivals, weddings, bards, troubadours, and a masque that ends in
# fire. A starving old man comes out of Athel Loren claiming to be the lord's
# dead father.
# Witch trials never come to Kislev, which honours its ice witches, or to
# Araby, which keeps court sorcerers. The joust and the tourney are for the
# knights of Bretonnia (the Lady, or the false Grail).
#
# Justice at court        feudal.1 - feudal.11
# Poachers                feudal.20 - feudal.25
# A courtier robbed       feudal.30 - feudal.34
# An unfaithful spouse    feudal.40 - feudal.44
# Pastimes                feudal.50 - feudal.64
# Songs and masques       feudal.70 - feudal.77
#
# Callers: on_yearly_pulse random_events in 00_on_actions.txt (1 - 7, 10,
#   11, 20, 30, 40, 50, 55, 60 - 64, 70, 75 - 77).
# Moved elsewhere long ago: the deer park (hunting_events 4075), the puppet
#   show and the bear (summer_fair_events 4105, 4170), the funeral
#   (now mourning.39 in WHG_death_events), the acrobats and the dancer (feast_events 4160,
#   4165).
# Saved: character flags hanged_poachers, ears_off, dead_poachers,
#   prisoner_in, heads_up, hands_off, guest_declined, guest_accepted,
#   guest_happy, liege_wroth, liege_gregarious; event target
#   invitation_pending.
# Triggers: common/scripted_triggers/WHG_feudal_life_triggers.txt
# Loc: localisation/00_WHG_feudal_life_events.csv
###############################################################################

namespace = feudal
