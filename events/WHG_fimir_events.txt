###############################################################################
# The Fimir
# Namespace: fimir (was wh_fimir_events.txt)
# Faction: Chaos
# Serves: the Fimir of the fens and mists (culture fimir)
#
# The Fimir are a cyclopean people of the marshes, born of Chaos. Their
# only females are the rare Meargh, witch-queens who rule each clan with
# sorcery; the males (Fimm warriors, Fimm nobles, Dirach sorcerers and the
# lowly Shearl) can only father young on women carried off from the races
# of Men.
#
# The Meargh's power                    fimir.1
# A Fimir mate                          fimir.2
#
# Callers: on_adulthood in wh_on_actions.txt (fimir.1), intermarriage.002
#   in S_intermarriage_events.txt when a marriage to a Fimir is undone
#   (fimir.2).
# Saved: event target fimir_raided_realm.
# Loc: localisation/00_WHG_fimir_events.csv
###############################################################################

namespace = fimir

# fimir.1 The Meargh's power
# A Meargh comes of age with the witch-sight of her line: she becomes a
# mage of a random lore.
# Scope: ROOT is the Meargh.
character_event = {
	id = fimir.1
	hide_window = yes

	is_triggered_only = yes # on_adulthood
	culture_group = fimir_group

	trigger = {
		trait = fimir_maergh
	}

	immediate = {
		random_lore_mage_effect = yes
	}
}

# fimir.2 A Fimir mate
# A Fimir ruler whose marriage to another people was undone looks for a
# new mate. A male Fimir carries off a woman from a neighbouring realm of
# Men; a Meargh takes a Fimm warrior, a Fimm noble or a Dirach sorcerer as
# her mate. The better the mate, the more it costs her.
# Scope: ROOT is the Fimir ruler.
character_event = {
	id = fimir.2
	desc = {
		text = EVTDESCfimir2_meargh
		trigger = { is_woman_trigger = yes }
	}
	desc = {
		text = EVTDESCfimir2_male
		trigger = { is_woman_trigger = no }
	}
	picture = GFX_evt_cave_lair

	is_triggered_only = yes # intermarriage.002

	trigger = {
		trait = creature_fimir
		is_married = no
	}

	immediate = {
		random_neighbor_independent_ruler = {
			limit = { trait = creature_human }
			save_event_target_as = fimir_raided_realm
		}
	}

	# Carry off a woman of Men
	option = {
		name = EVTOPTAfimir2
		trigger = {
			is_woman_trigger = no
			event_target:fimir_raided_realm = { always = yes }
		}
		event_target:fimir_raided_realm = {
			create_character = {
				female = yes
				random_traits = yes
				dynasty = none
				religion = PREV
				culture = PREV
				trait = creature_human
			}
			new_character = {
				move_character = ROOT
				add_spouse = ROOT
			}
			opinion = { who = ROOT modifier = opinion_furious }
		}
		prestige = 50
	}

	# A Fimm warrior
	option = {
		name = EVTOPTBfimir2
		trigger = { is_woman_trigger = yes }
		create_random_soldier = {
			trait = creature_fimir
			random_traits = yes
			dynasty = none
			martial = 8
		}
		new_character = {
			remove_trait = fimir_noble
			remove_trait = fimir_shearl
			remove_trait = fimir_dirach
			add_trait = fimir_fimm
			add_spouse_matrilineal = ROOT
		}
		prestige = -100
	}

	# A Fimm noble
	option = {
		name = EVTOPTCfimir2
		trigger = {
			is_woman_trigger = yes
			prestige = 500
		}
		create_random_steward = {
			trait = creature_fimir
			random_traits = yes
			stewardship = 8
			martial = 8
		}
		new_character = {
			remove_trait = fimir_fimm
			remove_trait = fimir_shearl
			remove_trait = fimir_dirach
			add_trait = fimir_noble
			add_spouse_matrilineal = ROOT
		}
		prestige = -500
		ai_chance = {
			factor = 1
			modifier = { factor = 0 NOT = { prestige = 1000 } }
		}
	}

	# A Dirach sorcerer
	option = {
		name = EVTOPTDfimir2
		trigger = {
			is_woman_trigger = yes
			prestige = 250
			piety = 250
		}
		create_random_priest = {
			trait = creature_fimir
			random_traits = yes
			stewardship = 8
			learning = 8
		}
		new_character = {
			random_lore_mage_effect = yes
			remove_trait = fimir_fimm
			remove_trait = fimir_shearl
			remove_trait = fimir_noble
			add_trait = fimir_dirach
			add_spouse_matrilineal = ROOT
		}
		prestige = -250
		piety = -250
		ai_chance = {
			factor = 1
			modifier = { factor = 0 NOT = { piety = 500 } }
		}
	}

	# Stay unwed
	option = {
		name = EVTOPTEfimir2
		ai_chance = { factor = 0 }
	}
}
