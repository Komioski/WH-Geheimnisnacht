# bond.1 A playmate
# Another child at court takes a liking or a dislike to ROOT. Without
# Conclave only; Conclave has its own childhood friendships.
# Scope: the child. Fired by on_yearly_childhood_pulse (friends_rivals).
character_event = {
	id = bond.1
	desc = EVTDESCbond1
	picture = GFX_evt_child_sword

	is_triggered_only = yes # on_yearly_childhood_pulse

	min_age = 4
	max_age = 14
	capable_only = yes
	prisoner = no
	lacks_dlc = "Conclave"

	trigger = {
		bond_child_room_trigger = yes
		host = {
			any_courtier = {
				bond_child_peer_trigger = yes
				is_abroad = no
				age = 4
				NOT = { age = 14 }
				NOT = { trait = craven }
				NOT = { trait = honest }
			}
		}
	}

	immediate = {
		host = {
			random_courtier = {
				limit = {
					bond_child_peer_trigger = yes
					is_abroad = no
					age = 4
					NOT = { age = 14 }
					NOT = { trait = craven }
					NOT = { trait = honest }
				}
				save_event_target_as = troublemaker_child
			}
		}
	}

	# Knock them down
	option = {
		name = EVTOPTAbond1
		event_target:troublemaker_child = {
			add_rival = ROOT
			character_event = { id = bond.901 days = 1 random = 2 tooltip = EVTTOOLTIPbond901 }
			if = {
				limit = { NOT = { num_of_traits = 5 } NOT = { trait = wroth } trait = greedy }
				add_trait = wroth
			}
			else_if = {
				limit = { NOT = { num_of_traits = 5 } NOT = { trait = greedy } trait = wroth }
				add_trait = greedy
			}
		}
	}

	# Share the sword
	option = {
		name = EVTOPTBbond1
		event_target:troublemaker_child = {
			add_friend = ROOT
			character_event = { id = bond.900 tooltip = EVTTOOLTIPbond900 }
		}
	}
}

# bond.2 The shy child
# A bold or kind child notices another child who would rather read than play,
# and is picked on for it. ROOT may teach them to stand up for themselves,
# lend them a wooden sword, or leave them be.
# Scope: the bold child. Fired by on_yearly_childhood_pulse (friends_rivals).
character_event = {
	id = bond.2
	desc = EVTDESCbond2
	picture = GFX_evt_child_talking

	is_triggered_only = yes # on_yearly_childhood_pulse

	min_age = 6
	max_age = 14
	prisoner = no

	trigger = {
		bond_child_room_trigger = yes
		OR = { trait = gregarious trait = charitable trait = brave }
		host = {
			any_courtier = { bond_child_peer_trigger = yes age = 6 NOT = { age = 15 } trait = shy }
		}
	}

	immediate = {
		host = {
			random_courtier = {
				limit = { bond_child_peer_trigger = yes age = 6 NOT = { age = 15 } trait = shy }
				save_event_target_as = shy_child
			}
		}
	}

	# Teach them to stand up to bullies
	option = {
		name = EVTOPTAbond2
		event_target:shy_child = {
			character_event = { id = bond.3 days = 2 random = 5 tooltip = EVTTOOLTIPbond3 }
		}
	}

	# Lend them my wooden sword
	option = {
		name = EVTOPTBbond2
		event_target:shy_child = {
			character_event = { id = bond.4 days = 2 random = 5 tooltip = EVTTOOLTIPbond4 }
		}
	}

	# Leave them to their books
	option = {
		name = EVTOPTCbond2
		prestige = 5
		if = {
			limit = { trait = gregarious NOT = { trait = charitable } }
			remove_trait = gregarious
		}
		else_if = {
			limit = { trait = charitable NOT = { trait = gregarious } }
			remove_trait = charitable
		}
	}
}

# bond.6 A scolding interrupted
# A lazy ward is about to be thrashed by their guardian when another ward
# speaks up for them.
# Scope: the lazy ward. Fired by on_yearly_childhood_pulse (friends_rivals).
character_event = {
	id = bond.6
	desc = EVTDESCbond6
	picture = GFX_evt_child_bully

	is_triggered_only = yes # on_yearly_childhood_pulse

	min_age = 11
	max_age = 16
	prisoner = no

	trigger = {
		has_guardian = yes
		trait = slothful
		bond_child_room_trigger = yes
		host = {
			any_courtier = {
				bond_open_trigger = yes
				NOT = { character = ROOT }
				same_guardian = ROOT
				NOT = { sibling = ROOT }
				NOT = { num_of_friends = 3 }
				NOT = { num_of_rivals = 3 }
				NOT = { is_friend = ROOT }
				NOT = { is_rival = ROOT }
				OR = { trait = charitable trait = kind trait = ambitious }
			}
		}
	}

	immediate = {
		host = {
			random_courtier = {
				limit = {
					bond_open_trigger = yes
					NOT = { character = ROOT }
					same_guardian = ROOT
					NOT = { sibling = ROOT }
					NOT = { num_of_friends = 3 }
					NOT = { num_of_rivals = 3 }
					NOT = { is_friend = ROOT }
					NOT = { is_rival = ROOT }
					OR = { trait = charitable trait = kind trait = ambitious }
				}
				save_event_target_as = helping_child
			}
		}
	}

	# Thank them
	option = {
		name = EVTOPTAbond6
		event_target:helping_child = {
			add_friend = ROOT
			character_event = { id = bond.7 days = 1 random = 3 tooltip = EVTTOOLTIPbond7 }
		}
	}

	# I did not need their help
	option = {
		name = EVTOPTBbond6
		event_target:helping_child = {
			reverse_opinion = { modifier = opinion_irritated who = ROOT years = 100 }
			set_character_flag = wrongly_interfered
			character_event = { id = bond.7 days = 1 random = 3 tooltip = EVTTOOLTIPbond7b }
		}
	}

	# Own up to my laziness
	option = {
		name = EVTOPTCbond6
		if = { limit = { NOT = { num_of_traits = 5 } NOT = { trait = honest } } add_trait = honest }
	}
}

# bond.10 A wingman at court
# Hidden. In a ruler's household, a young man who is fond of women and of
# company finds another who cannot speak two words to a lady (bond.11).
# Scope: the ruler. Fired by on_yearly_pulse random_events.
character_event = {
	id = bond.10
	hide_window = yes

	is_triggered_only = yes # on_yearly_pulse

	only_rulers = yes

	trigger = {
		is_inaccessible_trigger = no
		OR = { bond_wingman_trigger = yes any_courtier = { bond_wingman_trigger = yes } }
		OR = { bond_shy_suitor_trigger = yes any_courtier = { bond_shy_suitor_trigger = yes } }
	}

	immediate = {
		if = { limit = { bond_wingman_trigger = yes } save_event_target_as = lustful_courtier }
		else = {
			random_courtier = {
				limit = { bond_wingman_trigger = yes }
				save_event_target_as = lustful_courtier
			}
		}
		if = {
			limit = {
				bond_shy_suitor_trigger = yes
				NOT = { character = event_target:lustful_courtier }
				NOT = { sibling = event_target:lustful_courtier }
				NOT = { is_friend = event_target:lustful_courtier }
				NOT = { is_rival = event_target:lustful_courtier }
			}
			save_event_target_as = shy_courtier
		}
		else = {
			random_courtier = {
				limit = {
					bond_shy_suitor_trigger = yes
					NOT = { character = event_target:lustful_courtier }
					at_location = event_target:lustful_courtier
					NOT = { sibling = event_target:lustful_courtier }
					NOT = { is_friend = event_target:lustful_courtier }
					NOT = { is_rival = event_target:lustful_courtier }
				}
				save_event_target_as = shy_courtier
			}
		}
		if = {
			limit = { event_target:shy_courtier = { always = yes } }
			event_target:lustful_courtier = { character_event = { id = bond.11 } }
		}
	}
}

	}

		}

		name = EVTOPTAfriend1

			factor = 10

		}

			limit = { event_target:friend_at_war = { is_primary_war_attacker = yes } }
			join_attacker_wars = event_target:friend_at_war
		}
		else = { join_defender_wars = event_target:friend_at_war }
		event_target:friend_at_war = {
			opinion = { modifier = opinion_very_grateful who = ROOT years = 5 }
			letter_event = { id = friend.2 days = 7 random = 14 tooltip = EVTTOOLTIPfriend2 }
		}
	}

	# Send gold instead
	option = {
		name = EVTOPTBfriend1
		ai_chance = {
			factor = 10
			modifier = { factor = 0.25 trait = greedy }
			modifier = { factor = 2 trait = charitable }
		}
		transfer_scaled_wealth = { to = event_target:friend_at_war value = 0.25 }
		event_target:friend_at_war = {
			opinion = { modifier = opinion_grateful who = ROOT years = 5 }
			letter_event = { id = friend.3 days = 7 random = 14 tooltip = EVTTOOLTIPfriend3 }
		}
	}

	# It is not my war
	option = {
		name = EVTOPTCfriend1
		ai_chance = { factor = 5 }

			chance = 25
			event_target:friend_at_war = {

					id = friend.4
					days = 90
					random = 30
					tooltip = EVTTOOLTIPfriend4
				}
			}
		}
	}

# rival.90 Revenge
# ROOT has sworn revenge on a rival. Slander them, have them killed, have
# them maimed, or let it go.
# Scope: the character. Fired by on_yearly_pulse (friends_rivals_adults).
character_event = {
	id = rival.90
	desc = EVTDESCrival90
	picture = GFX_evt_council

	is_triggered_only = yes # on_yearly_pulse friends_rivals_adults

	min_age = 16
	capable_only = yes
	prisoner = no
	rivals = yes

	trigger = {
		bond_open_trigger = yes
		in_command = no
		NOT = { has_character_modifier = revenge_seeker_timer }
		any_rival = { rival_revenge_trigger = yes }
	}

	immediate = {
		add_character_modifier = { name = revenge_seeker_timer duration = 1825 hidden = yes }
		random_rival = {
			limit = { rival_revenge_trigger = yes }
			save_event_target_as = revenge_target
		}
	}

	# Ruin their name
	option = {
		name = EVTOPTArival90
		trigger = { is_ruler = yes event_target:revenge_target = { is_ruler = yes } }
		ai_chance = { factor = 10 modifier = { factor = 2 intrigue = 15 } }
		rival_revenge_slander_effect = yes
	}

	# Hire blades to kill them
	option = {
		name = EVTOPTBrival90
		ai_chance = {
			factor = 5
			modifier = { factor = 3 is_maimed_trigger = yes }
			modifier = { factor = 2 trait = wroth }
			modifier = { factor = 0.1 trait = kind }
		}
		rival_revenge_blades_effect = yes
	}

	# Hire blades to cripple them
	option = {
		name = EVTOPTCrival90
		ai_chance = {
			factor = 5
			modifier = { factor = 3 trait = wounded }
			modifier = { factor = 2 trait = cruel }
		}
		set_character_flag = revenge_try_maim
		rival_revenge_blades_effect = yes
	}

	# Let it go
	option = {
		name = EVTOPTDrival90
		ai_chance = {
			factor = 5
			modifier = { factor = 3 trait = kind }
			modifier = { factor = 3 trait = content }
		}
		piety = 10
		remove_opinion = { who = event_target:revenge_target modifier = opinion_revenge_seeker }
	}
}

# rival.94 Revenge taken, or not
# Scope: the one who sought revenge. revenge_target was the target. Fired
# by rival_revenge_slander_effect and rival_revenge_blades_effect.
character_event = {
	id = rival.94
	desc = { trigger = { has_character_flag = revenge_result_slandered } text = EVTDESCrival94 }
	desc = {
		trigger = { has_character_flag = revenge_result_escaped }
		text = EVTDESCrival94_escaped
	}
	desc = {
		trigger = { has_character_flag = revenge_result_wounded }
		text = EVTDESCrival94_wounded
	}
	desc = { trigger = { has_character_flag = revenge_result_maimed } text = EVTDESCrival94_maimed }
	desc = { trigger = { has_character_flag = revenge_result_killed } text = EVTDESCrival94_killed }
	picture = GFX_evt_spymaster

	is_triggered_only = yes # rival_revenge_slander_effect, rival_revenge_blades_effect

	option = {
		name = EVTOPTArival94
		if = {
			limit = { has_character_flag = revenge_result_escaped }
			add_character_modifier = { name = poor_revenge_seeker duration = 365 }
		}
		else = {
			if = {
				limit = { has_character_flag = revenge_result_killed }
				add_character_modifier = { name = hot_revenge duration = 1825 }
			}
			else_if = {
				limit = {
					OR = {
						has_character_flag = revenge_result_maimed
						has_character_flag = revenge_result_slandered
					}
				}
				add_character_modifier = { name = hot_revenge duration = 1095 }
			}
			else = { add_character_modifier = { name = hot_revenge duration = 730 } }
			if = {
				limit = { has_character_flag = revenge_for_father }
				set_character_flag = avenged_father
				if = {
					limit = { event_target:revenge_target = { is_alive = yes } }
					remove_opinion = {
						who = event_target:revenge_target
						modifier = opinion_killed_father
					}
				}
			}
			else_if = {
				limit = { event_target:revenge_target = { is_alive = yes } }
				remove_opinion = {
					who = event_target:revenge_target
					modifier = opinion_revenge_seeker
				}
			}
		}
		rival_clear_revenge_effect = yes
	}
}
