# bond.1 A playmate
# Another child at court takes a liking or a dislike to ROOT. Without
# Conclave only; Conclave has its own childhood friendships.
# Scope: the child. Fired by on_yearly_childhood_pulse (friends_rivals).
character_event = {
	id = bond.1
	desc = EVTDESCbond1
	picture = GFX_evt_child_sword

	is_triggered_only = yes # on_yearly_childhood_pulse

	min_age = 4
	max_age = 14
	capable_only = yes
	prisoner = no
	lacks_dlc = "Conclave"

	trigger = {
		bond_child_room_trigger = yes
		host = {
			any_courtier = {
				bond_child_peer_trigger = yes
				is_abroad = no
				age = 4
				NOT = { age = 14 }
				NOT = { trait = craven }
				NOT = { trait = honest }
			}
		}
	}

	immediate = {
		host = {
			random_courtier = {
				limit = {
					bond_child_peer_trigger = yes
					is_abroad = no
					age = 4
					NOT = { age = 14 }
					NOT = { trait = craven }
					NOT = { trait = honest }
				}
				save_event_target_as = troublemaker_child
			}
		}
	}

	# Knock them down
	option = {
		name = EVTOPTAbond1
		event_target:troublemaker_child = {
			add_rival = ROOT
			character_event = { id = bond.901 days = 1 random = 2 tooltip = EVTTOOLTIPbond901 }
			if = {
				limit = { NOT = { num_of_traits = 5 } NOT = { trait = wroth } trait = greedy }
				add_trait = wroth
			}
			else_if = {
				limit = { NOT = { num_of_traits = 5 } NOT = { trait = greedy } trait = wroth }
				add_trait = greedy
			}
		}
	}

	# Share the sword
	option = {
		name = EVTOPTBbond1
		event_target:troublemaker_child = {
			add_friend = ROOT
			character_event = { id = bond.900 tooltip = EVTTOOLTIPbond900 }
		}
	}
}

# bond.2 The shy child
# A bold or kind child notices another child who would rather read than play,
# and is picked on for it. ROOT may teach them to stand up for themselves,
# lend them a wooden sword, or leave them be.
# Scope: the bold child. Fired by on_yearly_childhood_pulse (friends_rivals).
character_event = {
	id = bond.2
	desc = EVTDESCbond2
	picture = GFX_evt_child_talking

	is_triggered_only = yes # on_yearly_childhood_pulse

	min_age = 6
	max_age = 14
	prisoner = no

	trigger = {
		bond_child_room_trigger = yes
		OR = { trait = gregarious trait = charitable trait = brave }
		host = {
			any_courtier = { bond_child_peer_trigger = yes age = 6 NOT = { age = 15 } trait = shy }
		}
	}

	immediate = {
		host = {
			random_courtier = {
				limit = { bond_child_peer_trigger = yes age = 6 NOT = { age = 15 } trait = shy }
				save_event_target_as = shy_child
			}
		}
	}

	# Teach them to stand up to bullies
	option = {
		name = EVTOPTAbond2
		event_target:shy_child = {
			character_event = { id = bond.3 days = 2 random = 5 tooltip = EVTTOOLTIPbond3 }
		}
	}

	# Lend them my wooden sword
	option = {
		name = EVTOPTBbond2
		event_target:shy_child = {
			character_event = { id = bond.4 days = 2 random = 5 tooltip = EVTTOOLTIPbond4 }
		}
	}

	# Leave them to their books
	option = {
		name = EVTOPTCbond2
		prestige = 5
		if = {
			limit = { trait = gregarious NOT = { trait = charitable } }
			remove_trait = gregarious
		}
		else_if = {
			limit = { trait = charitable NOT = { trait = gregarious } }
			remove_trait = charitable
		}
	}
}

# bond.6 A scolding interrupted
# A lazy ward is about to be thrashed by their guardian when another ward
# speaks up for them.
# Scope: the lazy ward. Fired by on_yearly_childhood_pulse (friends_rivals).
character_event = {
	id = bond.6
	desc = EVTDESCbond6
	picture = GFX_evt_child_bully

	is_triggered_only = yes # on_yearly_childhood_pulse

	min_age = 11
	max_age = 16
	prisoner = no

	trigger = {
		has_guardian = yes
		trait = slothful
		bond_child_room_trigger = yes
		host = {
			any_courtier = {
				bond_open_trigger = yes
				NOT = { character = ROOT }
				same_guardian = ROOT
				NOT = { sibling = ROOT }
				NOT = { num_of_friends = 3 }
				NOT = { num_of_rivals = 3 }
				NOT = { is_friend = ROOT }
				NOT = { is_rival = ROOT }
				OR = { trait = charitable trait = kind trait = ambitious }
			}
		}
	}

	immediate = {
		host = {
			random_courtier = {
				limit = {
					bond_open_trigger = yes
					NOT = { character = ROOT }
					same_guardian = ROOT
					NOT = { sibling = ROOT }
					NOT = { num_of_friends = 3 }
					NOT = { num_of_rivals = 3 }
					NOT = { is_friend = ROOT }
					NOT = { is_rival = ROOT }
					OR = { trait = charitable trait = kind trait = ambitious }
				}
				save_event_target_as = helping_child
			}
		}
	}

	# Thank them
	option = {
		name = EVTOPTAbond6
		event_target:helping_child = {
			add_friend = ROOT
			character_event = { id = bond.7 days = 1 random = 3 tooltip = EVTTOOLTIPbond7 }
		}
	}

	# I did not need their help
	option = {
		name = EVTOPTBbond6
		event_target:helping_child = {
			reverse_opinion = { modifier = opinion_irritated who = ROOT years = 100 }
			set_character_flag = wrongly_interfered
			character_event = { id = bond.7 days = 1 random = 3 tooltip = EVTTOOLTIPbond7b }
		}
	}

	# Own up to my laziness
	option = {
		name = EVTOPTCbond6
		if = { limit = { NOT = { num_of_traits = 5 } NOT = { trait = honest } } add_trait = honest }
	}
}

# bond.10 A wingman at court
# Hidden. In a ruler's household, a young man who is fond of women and of
# company finds another who cannot speak two words to a lady (bond.11).
# Scope: the ruler. Fired by on_yearly_pulse random_events.
character_event = {
	id = bond.10
	hide_window = yes

	is_triggered_only = yes # on_yearly_pulse

	only_rulers = yes

	trigger = {
		is_inaccessible_trigger = no
		OR = { bond_wingman_trigger = yes any_courtier = { bond_wingman_trigger = yes } }
		OR = { bond_shy_suitor_trigger = yes any_courtier = { bond_shy_suitor_trigger = yes } }
	}

	immediate = {
		if = { limit = { bond_wingman_trigger = yes } save_event_target_as = lustful_courtier }
		else = {
			random_courtier = {
				limit = { bond_wingman_trigger = yes }
				save_event_target_as = lustful_courtier
			}
		}
		if = {
			limit = {
				bond_shy_suitor_trigger = yes
				NOT = { character = event_target:lustful_courtier }
				NOT = { sibling = event_target:lustful_courtier }
				NOT = { is_friend = event_target:lustful_courtier }
				NOT = { is_rival = event_target:lustful_courtier }
			}
			save_event_target_as = shy_courtier
		}
		else = {
			random_courtier = {
				limit = {
					bond_shy_suitor_trigger = yes
					NOT = { character = event_target:lustful_courtier }
					at_location = event_target:lustful_courtier
					NOT = { sibling = event_target:lustful_courtier }
					NOT = { is_friend = event_target:lustful_courtier }
					NOT = { is_rival = event_target:lustful_courtier }
				}
				save_event_target_as = shy_courtier
			}
		}
		if = {
			limit = { event_target:shy_courtier = { always = yes } }
			event_target:lustful_courtier = { character_event = { id = bond.11 } }
		}
	}
}

			opinion = { modifier = opinion_grateful who = ROOT years = 5 }

# friend.55 My friend shows me the way
# The friend's way is open to ROOT: the restraint of the celibate
# (lifestyle.62), the pleasures of the hedonist (lifestyle.61, or in Araby
# lifestyle.70), or the scholar's books (lifestyle.10).
# Scope: the one who asked. FROM is the friend. Fired by friend.53.
character_event = {
	id = friend.55
	desc = { trigger = { has_character_flag = friend_way_celibate } text = EVTDESCfriend55 }
	desc = {
		trigger = { has_character_flag = friend_way_hedonist }
		text = EVTDESCfriend55_hedonist
	}
	desc = { trigger = { has_character_flag = friend_way_scholar } text = EVTDESCfriend55_scholar }
	picture = GFX_evt_courtiers_talking

	is_triggered_only = yes # friend.53

	# Follow where they lead
	option = {
		name = EVTOPTAfriend55
		if = {
			limit = { has_character_flag = friend_way_celibate }
			character_event = { id = lifestyle.62 days = 7 random = 7 }
		}
		else_if = {
			limit = { has_character_flag = friend_way_hedonist lifestyle_araby_start_trigger = yes }
			character_event = { id = lifestyle.70 days = 7 random = 7 }
		}
		else_if = {
			limit = { has_character_flag = friend_way_hedonist }
			character_event = { id = lifestyle.61 days = 7 random = 7 }
		}
		else = { character_event = { id = lifestyle.10 days = 7 random = 7 } }
		clr_character_flag = friend_way_celibate
		clr_character_flag = friend_way_hedonist
		clr_character_flag = friend_way_scholar
	}

	# It is not for me after all
	option = {
		name = EVTOPTBfriend55
		prestige = 5
		clr_character_flag = friend_way_celibate
		clr_character_flag = friend_way_hedonist
		clr_character_flag = friend_way_scholar
	}
}

# friend.60 I do not know my friend any more
# A friend has grown cruel, false or wrathful, or ROOT has simply come to
# dislike them. End it, talk it out, or let it pass.
# Scope: a playable character. Fired by on_yearly_pulse
# (friends_rivals_adults).
character_event = {
	id = friend.60
	desc = EVTDESCfriend60
	picture = GFX_evt_courtiers_talking

	is_triggered_only = yes # on_yearly_pulse friends_rivals_adults

	only_playable = yes
	min_age = 16
	capable_only = yes
	prisoner = no
	war = no
	friends = yes

	trigger = {
		bond_open_trigger = yes
		in_command = no
		any_friend = { friend_estranged_trigger = yes }
	}

	weight_multiplier = {
		days = 1
		modifier = { factor = 0.25 trait = gregarious }
		modifier = { factor = 0.25 trait = trusting }
		modifier = { factor = 0.25 trait = socializer }
		modifier = { factor = 3 trait = paranoid }
		modifier = { factor = 3 trait = shy }
	}

	immediate = {
		random_friend = {
			limit = { friend_estranged_trigger = yes }
			save_event_target_as = estranged_friend
		}
	}

	# We are friends no more
	option = {
		name = EVTOPTAfriend60
		remove_friend = event_target:estranged_friend
		event_target:estranged_friend = {
			character_event = { id = friend.61 days = 3 tooltip = EVTTOOLTIPfriend61 }
		}
	}

	# We must talk
	option = {
		name = EVTOPTBfriend60
		event_target:estranged_friend = {
			character_event = { id = friend.62 days = 3 tooltip = EVTTOOLTIPfriend62 }
		}
	}

	# It will pass
	option = {
		name = EVTOPTCfriend60
		opinion = { modifier = opinion_glad who = event_target:estranged_friend years = 2 }
	}
}
