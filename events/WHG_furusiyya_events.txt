###############################################################################
# The Furusiyya
# Namespace: furusiyya (was 88250 - 88999 in vanilla furusiyya_events.txt)
# Faction: Araby & Southlands
# Serves: independent kings and emperors of the southern faiths (Ormazd,
#   the Kitabids, the Atruhayid and the Akghan)
#
# Once in a life a southern sovereign calls the warriors of the realm to the
# Furusiyya: two months of horsemanship, archery, wrestling, mock battles and
# chess. Guests may die, be maimed or wounded, find or lose their nerve,
# learn or forget the arts of war and rule, take to drink or race enchanted
# carpets over the grounds. The best scores are named first, second and
# third champion. The champions and the host lift their dynasty's
# decadence, and the first champion may ride home on a horse or camel from
# the host's stables.
#
# The host invites the realm           furusiyya.1 - furusiyya.3
# Start, war, the host's death         furusiyya.10 - furusiyya.14
# What happens to a guest              furusiyya.20 - furusiyya.50
# The host hears of the fallen         furusiyya.25 - furusiyya.27
# Scores and champions                 furusiyya.60 - furusiyya.69
# The end                              furusiyya.70 - furusiyya.76
#
# Fired from: the decision hold_a_furusiyya in realm_decisions.txt
#   (furusiyya.1, and furusiyya.14 as a 300-day safety catch); on_death in
#   WHG_character_on_actions.txt (furusiyya.12). Everything else is chained.
#   Triggers in common/scripted_triggers/WHG_east_triggers.txt,
#   effects in common/scripted_effects/WHG_east_effects.txt.
# Saved: event target furusiyya_host; host flags furusiyya_host,
#   furusiyya_running and furusiyya_no_msg_*; guest flags furusiyya_guest,
#   furusiyya_score_00 - 15, furusiyya_first, furusiyya_second and
#   furusiyya_third; furusiyya_prize_mount and furusiyya_carpet (once per
#   life); the vanilla modifier epic_tournament_furusiyya (100 years, so one
#   Furusiyya per life).
###############################################################################

namespace = furusiyya

###############################################################################
# The host invites the realm
###############################################################################

# furusiyya.1 Riders carry the invitations
# The host has taken the decision. Riders go out to every warrior of the
# realm (furusiyya.2), the host's kin hear of it (furusiyya.3) and the
# Furusiyya begins in 60 days (furusiyya.10). The host does not compete.
# Scope: ROOT is the host.
character_event = {
	id = furusiyya.1
	title = EVTNAMEfurusiyya
	desc = EVTDESCfurusiyya1
	picture = GFX_evt_moors

	is_triggered_only = yes # decision hold_a_furusiyya

	immediate = {
		set_character_flag = furusiyya_host
		save_event_target_as = furusiyya_host
	}

	option = {
		name = EVTOPTAfurusiyya1
		hidden_effect = {
			character_event = { id = furusiyya.10 days = 60 }
			any_realm_character = {
				limit = {
					NOT = { character = ROOT }
					furusiyya_can_compete_trigger = yes
				}
				character_event = { id = furusiyya.2 }
			}
			any_dynasty_member = {
				limit = {
					ai = no
					is_offmap_ruler = no
					uses_decadence = yes
					NOT = { character = ROOT }
				}
				character_event = { id = furusiyya.3 }
			}
		}
	}
}

# furusiyya.10 The Furusiyya begins
# At peace, the host opens the games: every guest gets a chance at one thing
# happening to them (furusiyya.20) and is scored on day 46 (furusiyya.60);
# the champions are named on day 48 (furusiyya.61). At war, the host calls it
# off, gets the gold back and may hold another one later. (Vanilla fired this
# only at peace, so a host at war lost the Furusiyya with no word.)
# Scope: ROOT is the host.
character_event = {
	id = furusiyya.10
	title = EVTNAMEfurusiyya
	desc = EVTDESCfurusiyya10
	picture = GFX_evt_feast
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # furusiyya.1, 60 days on

	trigger = { has_character_flag = furusiyya_host }

	immediate = { save_event_target_as = furusiyya_host }

	option = {
		name = EVTOPTAfurusiyya10
		trigger = { war = no }
		hidden_effect = {
			set_character_flag = furusiyya_running
			chronicle = {
				entry = CHRONICLE_FURUSIYYA
				picture = GFX_evt_joust
			}
			any_realm_character = {
				limit = { has_character_flag = furusiyya_guest }
				character_event = { id = furusiyya.20 days = 1 random = 15 }
				character_event = { id = furusiyya.60 days = 46 }
			}
			character_event = { id = furusiyya.61 days = 48 }
		}
	}
	option = {
		name = EVTOPTBfurusiyya10
		trigger = { war = yes }
		wealth = 200
		prestige = -100
		if = {
			limit = { uses_decadence = yes }
			decadence = 5
		}
		custom_tooltip = {
			text = EVTTOOLTIPfurusiyya10
			remove_character_modifier = epic_tournament_furusiyya
		}
		hidden_effect = {
			any_realm_character = {
				limit = { has_character_flag = furusiyya_guest }
				character_event = { id = furusiyya.11 }
			}
			furusiyya_clear_host_effect = yes
		}
	}
}

# furusiyya.14 Safety catch
# 300 days after the decision or an accepted invitation, clears whatever a
# broken chain left behind. The host's do_not_disturb is cleared only while
# the host flag still stands (vanilla cleared it on every guest too).
# Scope: ROOT is the host or a guest.
character_event = {
	id = furusiyya.14
	hide_window = yes

	is_triggered_only = yes # hold_a_furusiyya, furusiyya.2

	immediate = {
		furusiyya_clear_guest_effect = yes
		if = {
			limit = { has_character_flag = furusiyya_host }
			furusiyya_clear_host_effect = yes
		}
	}
}

			random_realm_character = {
				limit = {
					has_character_flag = furusiyya_guest
					has_character_flag = furusiyya_score_03
				}
				clr_character_flag = furusiyya_score_03
				set_character_flag = furusiyya_third
			}
		}
		else_if = {
			limit = {
				any_realm_character = {
					has_character_flag = furusiyya_guest
					has_character_flag = furusiyya_score_02
				}
			}
			random_realm_character = {
				limit = {
					has_character_flag = furusiyya_guest
					has_character_flag = furusiyya_score_02
				}
				clr_character_flag = furusiyya_score_02
				set_character_flag = furusiyya_third
			}
		}
		else_if = {
			limit = {
				any_realm_character = {
					has_character_flag = furusiyya_guest
					has_character_flag = furusiyya_score_01
				}
			}
			random_realm_character = {
				limit = {
					has_character_flag = furusiyya_guest
					has_character_flag = furusiyya_score_01
				}
				clr_character_flag = furusiyya_score_01
				set_character_flag = furusiyya_third
			}
		}
		else_if = {
			limit = {
				any_realm_character = {
					has_character_flag = furusiyya_guest
					has_character_flag = furusiyya_score_00
				}
			}
			random_realm_character = {
				limit = {
					has_character_flag = furusiyya_guest
					has_character_flag = furusiyya_score_00
				}
				clr_character_flag = furusiyya_score_00
				set_character_flag = furusiyya_third
			}
		}
		any_realm_character = {
			limit = {
				has_character_flag = furusiyya_guest
				OR = {
					has_character_flag = furusiyya_first
					has_character_flag = furusiyya_second
					has_character_flag = furusiyya_third
				}
			}
			character_event = { id = furusiyya.63 }
		}
		any_realm_character = {
			limit = { has_character_flag = furusiyya_guest }
			furusiyya_clear_score_effect = yes
		}
		character_event = { id = furusiyya.70 days = 6 }
	}
}

				limit = { has_character_flag = furusiyya_guest }

					limit = {

						}
					}
					random_list = {
