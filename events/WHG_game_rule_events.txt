# gamerule.1 A title without its lands
# on_five_year_pulse, rule de_jure_requirement = required: a duke or higher
# who holds none of a title's de jure counties loses the title and keeps a
# claim. The cults of Sigmar and Ulric are exempt.
# Scope: ROOT is the title holder.
character_event = {
	id = gamerule.1
	desc = EVTDESCgamerule1
	picture = GFX_evt_throne_room

	is_triggered_only = yes # on_five_year_pulse

	only_playable = yes
	war = no

	trigger = {
		has_game_rule = { name = de_jure_requirement value = required }
		is_merchant_republic = no
		any_demesne_title = { gamerule_landless_title_trigger = yes }
	}

	immediate = {
		random_demesne_title = {
			limit = { gamerule_landless_title_trigger = yes }
			save_event_target_as = title_to_destroy
		}
	}

	option = {
		name = EVTOPTAgamerule1
		event_target:title_to_destroy = { destroy_landed_title = THIS add_claim = ROOT }
	}
}

# gamerule.10 A khan dies
# on_death: when an independent nomad ruler dies, every vassal may stay or
# break away from the heir.
# Scope: ROOT is the dead khan.
character_event = {
	id = gamerule.10
	hide_window = yes

	is_triggered_only = yes # on_death

	only_playable = yes

	trigger = {
		has_dlc = "Horse Lords"
		independent = yes
		is_nomadic = yes
	}

	immediate = {
		current_heir = { save_event_target_as = nomadic_heir }
		any_vassal = { character_event = { id = gamerule.11 } }
	}
}

# gamerule.11 Stay or break away
# Scope: ROOT is the vassal, FROM the dead khan.
character_event = {
	id = gamerule.11
	desc = EVTDESCgamerule11
	picture = GFX_evt_horsemanship

	is_triggered_only = yes # gamerule.10

	only_playable = yes
	capable_only = yes
	prisoner = no
	min_age = 16

	trigger = {
		has_dlc = "Horse Lords"
		in_revolt = no
		independent = no
		liege = { independent = yes is_nomadic = yes in_revolt = no }
	}

	option = { # remain a vassal
		name = EVTOPTAgamerule11
		ai_chance = {
			factor = 100
			modifier = { factor = 2 opinion = { who = FROM value = 25 } }
			modifier = { factor = 2 opinion = { who = FROM value = 50 } }
			modifier = { factor = 2 opinion = { who = FROM value = 75 } }
			modifier = { factor = 2 opinion = { who = FROM value = 100 } }
			modifier = { factor = 2 liege = { NOT = { realm_size = 50 } } }
			modifier = { factor = 3 is_nomadic = yes }
			modifier = { factor = 10 liege = { trait = everchosen } }
			modifier = { factor = 4 trait = content }
		}
	}
	option = { # go independent
		name = EVTOPTBgamerule11
		if = {
			limit = { is_nomadic = yes }
			any_demesne_title = { limit = { tier = COUNT } add_claim = event_target:nomadic_heir }
		}
		else = {
			reverse_opinion = {
				who = event_target:nomadic_heir
				modifier = declared_independence_nomad
				years = 50
			}
			hidden_tooltip = {
				any_courtier_or_vassal = {
					limit = { dynasty = ROOT }
					reverse_opinion = {
						who = event_target:nomadic_heir
						modifier = declared_independence_nomad
						years = 50
					}
				}
			}
		}
		liege = {
			any_vassal = {
				limit = {
					in_revolt = yes
					any_war = {
						defender = { character = ROOT }
						attacker = { character = PREVPREV }
					}
				}
				set_defacto_liege = THIS
			}
		}
		set_defacto_liege = THIS
		hidden_tooltip = {
			event_target:nomadic_heir = { character_event = { id = gamerule.12 days = 1 } }
		}
		ai_chance = {
			factor = 50
			modifier = {
				factor = 0
				clan = yes
				OR = { has_blood_oath_with = FROM is_friend = FROM }
			}
			modifier = { factor = 2 clan = yes OR = { has_feud_with = FROM is_rival = FROM } }
			modifier = { factor = 2 NOT = { opinion = { who = FROM value = -25 } } }
			modifier = { factor = 2 NOT = { opinion = { who = FROM value = -50 } } }
			modifier = { factor = 2 NOT = { opinion = { who = FROM value = -75 } } }
			modifier = { factor = 2 NOT = { opinion = { who = FROM value = -100 } } }
			modifier = { factor = 2 trait = ambitious }
			modifier = { factor = 0.1 liege = { trait = everchosen } }
		}
	}
}

# The exclave purge
###############################################################################

# gamerule.20 The exclaves are cut loose
# on_death (and gamerule.25 for AI heirs who were at war): under the
# exclave_purge rule, the parts of an independent realm that are no longer
# connected to its capital break away. A realm at war waits: an AI heir is
# purged once the war is over, a player heir is spared. Holy orders and the
# Cult of Sigmar may hold land anywhere.
# Scope: ROOT is the independent ruler.
character_event = {
	id = gamerule.20
	hide_window = yes

	is_triggered_only = yes # on_death, gamerule.25

	only_independent = yes
	only_playable = yes

	trigger = {
		exclave_purge_on_trigger = yes
		holy_order = no
		NOT = { has_landed_title = k_cult_of_sigmar }
		capital_scope = {
			any_disconnected_province = {
				sub_realm = no
				NOR = { exclave_dejure_kept_trigger = yes exclave_connected_trigger = yes }
			}
		}
	}

	immediate = {
		if = {
			limit = { war = yes }
			if = {
				limit = { ai = yes player_heir = { ai = yes } }
				player_heir = { set_character_flag = delayed_cleanse_ai }
			}
		}
		else = { exclave_purge_effect = yes }
	}
}

# Generated families
###############################################################################

# gamerule.30 A family for the lonely
# on_startup, rule generate_families: landed rulers without a family get a
# spouse (except the peoples without women) and children, so the realms of
# the 2010 start have heirs.
# Scope: ROOT is the ruler.
character_event = {
	id = gamerule.30
	hide_window = yes

	is_triggered_only = yes # on_startup

	only_playable = yes
	min_age = 16

	trigger = {
		is_save_game = no
		is_landed = yes
		mercenary = no
		holy_order = no
		gamerule_family_rule_trigger = yes
		gamerule_family_people_trigger = yes
		OR = { is_married = no AND = { num_of_children < 1 spouse = { is_lowborn = yes } } }
		OR = { is_republic = yes is_feudal = yes is_nomadic = yes is_tribal = yes }
		NOR = {
			government = confucian_bureaucracy
			trait = eunuch
			trait = celibate
			has_nickname = nick_the_chaste
		}
		gamerule_family_lonely_trigger = yes
	}

	immediate = {
		# The parents of any new children
		if = {
			limit = { is_female = no }
			save_event_target_as = father_target
			spouse = { save_event_target_as = mother_target }
		}
		else = {
			save_event_target_as = mother_target
			spouse = { save_event_target_as = father_target }
		}
		# A spouse first
		if = {
			limit = { is_married = no can_marry = yes is_no_females_race = no }
			if = { limit = { is_female = no } gamerule_wife_effect = yes }
			else = { gamerule_husband_effect = yes }
		}
		# Then children
		if = { limit = { num_of_children < 1 } gamerule_children_effect = yes }
		# Historical children of a lord without a mother get the new wife
		if = {
			limit = { is_female = no is_married = yes }
			spouse = { save_event_target_as = new_mother }
			any_child = {
				limit = {
					NOR = { mother = { always = yes } trait = legit_bastard trait = bastard }
					event_target:new_mother = { could_be_parent_of = PREV }
				}
				set_mother = event_target:new_mother
			}
		}
		# Grown children who never got traits
		any_child = {
			limit = {
				age = 16
				NOR = {
					has_education_diplomacy_trigger = yes
					has_education_martial_trigger = yes
					has_education_learning_trigger = yes
					has_education_stewardship_trigger = yes
				}
			}
			add_random_education_trait = yes
			gamerule_copy_personality_effect = yes
			gamerule_copy_personality_effect = yes
			gamerule_copy_personality_effect = yes
		}
		any_child = {
			limit = { age = 10 NOT = { age = 16 } NOT = { personality_traits = 2 } }
			gamerule_copy_personality_effect = yes
			gamerule_copy_personality_effect = yes
		}
		any_child = {
			limit = { age = 6 NOT = { age = 10 } NOT = { personality_traits = 1 } }
			gamerule_copy_personality_effect = yes
		}
		recalc_succession = yes
	}
}
