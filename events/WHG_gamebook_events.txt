###############################################################################
# The game-book stories: grief in a time of plague, and courtly love
# Namespaces: grief, courtly (were GBC.1 - GBC.13 and GBC.199 - GBC.143 in
#   gbc_events.txt)
# Faction: Generic
# Serves: every lord and lady, above all the players
#
# Two short stories from the old game book. In the first a lord loses a
# friend or kinsman to one of the great plagues, and a kinsman with the dead
# one's owl helps them through their grief, until the plague comes for the
# kinsman too. In the second a newlywed lord falls for the husband or wife
# of another lord at the wedding feast, and courts them with poems, gifts and
# a hunt, until the jealous spouse must be dealt with.
# (Both stories were dead before the 2026 events refactor: the first waited
# for a Black Death flag the mod never sets, the second's wedding hook was
# commented out.)
#
# Grief brings the family together  grief.1 - grief.20
# Courtly love                      courtly.1 - courtly.44
#
# Callers: grief.1 from the on_yearly_pulse list (WHG_pulse_on_actions.txt);
#   courtly.1 from on_marriage (WHG_character_on_actions.txt).
# Saved: grief: character flags grief_mourning (once per lifetime),
#   grief_new_friend, grief_took_owl, grief_knows_friend_ill; event targets
#   grief_dead_one, grief_friend, grief_councillor.
#   courtly: character flags courtly_love (once per lifetime), courtly_lover
#   and the choice flags courtly_*; event targets courtly_lover,
#   courtly_rival.
# Scripted triggers: common/scripted_triggers/WHG_legend_triggers.txt
# Scripted effects: common/scripted_effects/WHG_legend_effects.txt
# Localisation: localisation/00_WHG_gamebook_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

# grief.1 The plague takes a friend
# on_yearly_pulse: one of the great plagues takes a friend or kinsman.
# Scope: ROOT is the mourner.
character_event = {
	id = grief.1
	hide_window = yes

	is_triggered_only = yes # on_yearly_pulse

	min_age = 16
	only_capable = yes
	prisoner = no

	trigger = {
		NOT = { has_character_flag = grief_mourning }
		OR = {
			any_friend = { grief_dying_one_trigger = yes }
			any_dynasty_member = { grief_dying_one_trigger = yes }
		}
	}

	immediate = {
		if = {
			limit = { any_friend = { grief_dying_one_trigger = yes } }
			random_friend = {
				limit = { grief_dying_one_trigger = yes }
				save_event_target_as = grief_dead_one
			}
		}
		else = {
			random_dynasty_member = {
				limit = { grief_dying_one_trigger = yes }
				save_event_target_as = grief_dead_one
			}
		}
		set_character_flag = grief_mourning
		event_target:grief_dead_one = { death = { death_reason = death_trait } }
		narrative_event = { id = grief.2 days = 10 }
	}
}

# grief.4 A kinsman steps forward
# Picks a kinsman (or a well-meaning courtier) who took in the dead one's owl.
# Scope: ROOT is the mourner.
character_event = {
	id = grief.4
	hide_window = yes

	is_triggered_only = yes # grief.3

	trigger = {
		OR = {
			any_dynasty_member = { grief_new_friend_trigger = yes }
			any_courtier = { grief_new_friend_trigger = yes opinion = { who = ROOT value = 10 } }
		}
	}

	immediate = {
		if = {
			limit = { any_dynasty_member = { grief_new_friend_trigger = yes } }
			any_dynasty_member = {
				limit = { grief_new_friend_trigger = yes }
				score_value = { value = 10 interesting_courtier_selection_score = yes }
				save_event_target_as = grief_friend
			}
		}
		else = {
			any_courtier = {
				limit = { grief_new_friend_trigger = yes opinion = { who = ROOT value = 10 } }
				score_value = { value = 10 interesting_courtier_selection_score = yes }
				save_event_target_as = grief_friend
			}
		}
		event_target:grief_friend = {
			set_character_flag = grief_new_friend
			character_event = { id = grief.5 }
		}
	}
}

# grief.8 A councillor who noticed
# Picks a councillor (or another courtier) who saw the two together. The
# kinsman starts to feel unwell.
# Scope: ROOT is the mourner.
character_event = {
	id = grief.8
	hide_window = yes

	is_triggered_only = yes # grief.7

	immediate = {
		if = {
			limit = {
				any_courtier = {
					is_councillor = yes
					NOT = { character = event_target:grief_friend }
					opinion = { who = ROOT value = 25 }
				}
			}
			any_courtier = {
				limit = {
					is_councillor = yes
					NOT = { character = event_target:grief_friend }
					opinion = { who = ROOT value = 25 }
				}
				score_value = { value = 10 interesting_courtier_selection_score = yes }
				save_event_target_as = grief_councillor
			}
		}
		else = {
			any_courtier = {
				limit = { is_adult = yes NOT = { character = event_target:grief_friend } }
				score_value = { value = 10 interesting_courtier_selection_score = yes }
				save_event_target_as = grief_councillor
			}
		}
		event_target:grief_friend = {
			if = {
				limit = { is_disease_immune_race = no }
				if = {
					limit = { has_dlc = "Reapers" }
					add_trait = fever
					set_character_flag = developing_illness
				}
				else = { add_trait = ill }
			}
		}
		if = {
			limit = { event_target:grief_councillor = { always = yes } }
			event_target:grief_councillor = { character_event = { id = grief.9 } }
		}
		else = { event_target:grief_friend = { character_event = { id = grief.11 days = 30 } } }
	}
}

# Courtly love
# A newlywed lord falls for the husband or wife of another lord at the
# wedding feast: the spouse of their liege, of a fellow vassal or of one of
# their own vassals.
###############################################################################

namespace = courtly

# courtly.1 A glance at the wedding feast
# on_marriage: a newlywed player may fall for someone else at their own
# wedding feast (3%).
# Scope: ROOT is the employer, FROM the newlywed.
character_event = {
	id = courtly.1
	hide_window = yes

	is_triggered_only = yes # on_marriage

	trigger = {
		FROM = {
			ai = no
			is_landed = yes
			is_adult = yes
			is_no_females_race = no
			block_general_event_trigger = no
			NOR = {
				any_lover = { always = yes }
				trait = celibate
				has_character_flag = courtly_love
			}
		}
	}

	immediate = {
		random = { chance = 3 FROM = { character_event = { id = courtly.2 days = 3 } } }
	}
}

# courtly.2 Who caught the eye?
# Picks the married one: the liege's spouse first, then the spouse of a
# fellow vassal, then of one of the newlywed's own vassals.
# Scope: ROOT is the newlywed.
character_event = {
	id = courtly.2
	hide_window = yes

	is_triggered_only = yes # courtly.1

	immediate = {
		if = {
			limit = {
				liege = {
					NOT = { character = ROOT }
					any_spouse = { courtly_love_target_trigger = yes }
				}
			}
			liege = {
				any_spouse = {
					limit = { courtly_love_target_trigger = yes }
					score_value = { value = 10 interesting_courtier_selection_score = yes }
					save_event_target_as = courtly_lover
				}
			}
		}
		else_if = {
			limit = {
				liege = {
					any_vassal = {
						NOT = { character = ROOT }
						any_spouse = { courtly_love_target_trigger = yes }
					}
				}
			}
			liege = {
				random_vassal = {
					limit = {
						NOT = { character = ROOT }
						any_spouse = { courtly_love_target_trigger = yes }
					}
					any_spouse = {
						limit = { courtly_love_target_trigger = yes }
						score_value = { value = 10 interesting_courtier_selection_score = yes }
						save_event_target_as = courtly_lover
					}
				}
			}
		}
		else_if = {
			limit = { any_vassal = { any_spouse = { courtly_love_target_trigger = yes } } }
			random_vassal = {
				limit = { any_spouse = { courtly_love_target_trigger = yes } }
				any_spouse = {
					limit = { courtly_love_target_trigger = yes }
					score_value = { value = 10 interesting_courtier_selection_score = yes }
					save_event_target_as = courtly_lover
				}
			}
		}
		if = {
			limit = { event_target:courtly_lover = { always = yes } }
			set_character_flag = courtly_love
			event_target:courtly_lover = {
				spouse = { save_event_target_as = courtly_rival }
				character_event = { id = courtly.3 }
			}
		}
	}
}

# courtly.34 The spouse is dealt with
# Rolls the newlywed's choice against the spouse.
# Scope: ROOT is the jealous spouse, FROM the newlywed.
character_event = {
	id = courtly.34
	hide_window = yes

	is_triggered_only = yes # courtly.33

	trigger = {
		block_general_event_trigger = no
	}

	immediate = {
		FROM = {
			if = {
				limit = { has_character_flag = courtly_duel }
				clr_character_flag = courtly_duel
				letter_event = { id = courtly.35 }
				random_list = {
					60 = {
						modifier = { factor = 1.2 combat_rating = 10 }
						modifier = { factor = 1.2 combat_rating = 20 }
						modifier = { factor = 1.2 combat_rating = 30 }
						modifier = { factor = 1.2 combat_rating = 40 }
						modifier = { factor = 1.2 combat_rating = 50 }
						modifier = { factor = 1.2 trait = duelist }
						modifier = { factor = 1.2 ROOT = { NOT = { combat_rating = 0 } } }
						modifier = { factor = 1.2 ROOT = { NOT = { combat_rating = -10 } } }
						modifier = { factor = 0 ROOT = { is_invincible_trigger = yes } }
						character_event = { id = courtly.36 days = 30 }
					}
					40 = {
						modifier = { factor = 1.2 NOT = { combat_rating = 0 } }
						modifier = { factor = 1.2 NOT = { combat_rating = -10 } }
						modifier = { factor = 1.2 ROOT = { combat_rating = 10 } }
						modifier = { factor = 1.2 ROOT = { combat_rating = 20 } }
						modifier = { factor = 1.2 ROOT = { combat_rating = 30 } }
						modifier = { factor = 1.2 ROOT = { combat_rating = 40 } }
						character_event = { id = courtly.37 days = 30 }
					}
				}
			}
			else_if = {
				limit = { has_character_flag = courtly_talk }
				clr_character_flag = courtly_talk
				random_list = {
					60 = {
						modifier = { factor = 1.2 diplomacy = 10 }
						modifier = { factor = 1.2 diplomacy = 13 }
						modifier = { factor = 1.2 diplomacy = 16 }
						modifier = { factor = 1.2 diplomacy = 19 }
						modifier = { factor = 1.2 diplomacy = 22 }
						modifier = { factor = 1.5 trait = socializer }
						character_event = { id = courtly.38 days = 60 }
					}
					40 = {
						modifier = { factor = 1.5 ROOT = { trait = wroth } }
						modifier = { factor = 1.5 ROOT = { trait = proud } }
						modifier = { factor = 1.5 ROOT = { trait = stubborn } }
						character_event = { id = courtly.39 days = 60 }
					}
				}
			}
			else = {
				clr_character_flag = courtly_murder
				random_list = {
					60 = {
						modifier = { factor = 1.2 intrigue = 10 }
						modifier = { factor = 1.2 intrigue = 13 }
						modifier = { factor = 1.2 intrigue = 16 }
						modifier = { factor = 1.2 intrigue = 19 }
						modifier = { factor = 1.2 intrigue = 22 }
						modifier = { factor = 1.5 trait = schemer }
						modifier = { factor = 0 ROOT = { is_invincible_trigger = yes } }
						character_event = { id = courtly.40 days = 90 }
					}
					40 = {
						modifier = { factor = 1.2 ROOT = { intrigue = 10 } }
						modifier = { factor = 1.2 ROOT = { intrigue = 13 } }
						modifier = { factor = 1.2 ROOT = { intrigue = 16 } }
						modifier = { factor = 1.2 ROOT = { intrigue = 19 } }
						modifier = { factor = 1.2 ROOT = { intrigue = 22 } }
						modifier = { factor = 1.5 ROOT = { trait = schemer } }
						character_event = { id = courtly.41 days = 90 }
					}
				}
			}
		}
	}
}
