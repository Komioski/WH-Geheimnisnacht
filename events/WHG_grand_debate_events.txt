###############################################################################
# The Grand Debate
# Namespace: debate (was MNM.1980 - 1998, MNM.5100 - 5106 and MNM.5201 - 5211
#   in mnm_grand_debate_events.txt)
# Faction: Old World faiths
# Serves: the Cult of Verena (its grandmaster hosts, every member may attend)
#
# Once a decade the grandmaster of the Cult of Verena may call the members to
# a Grand Debate at his court. For some weeks the learned of the Old World
# argue before the Goddess of Wisdom and Justice: verses and jokes, rivals
# won over, crushes, forged notes and spilt ink, letters home, sleepless
# nights. The best debater is crowned; the others go home wiser, or
# ridiculed.
#
# The debate                           debate.1 - debate.16
# At the debate                        debate.20 - debate.37
#
# Callers: the decision hermetic_grand_debate in
#   decisions/mnm_hermetics_decisions.txt (debate.1, debate.3). Triggers in
#   common/scripted_triggers/WHG_legend_triggers.txt, effects in
#   common/scripted_effects/WHG_legend_effects.txt, debate subjects in
#   localisation/customizable_localisation/WHG_debate_voices.txt.
# Saved: event targets debate_host, fellow_debater_target,
#   rival_debater_target, striking_debater_target, fellow_member_1,
#   fellow_member_2, fellow_member_target, inspired_letter_target,
#   sabotage_target; variables debate_score, debate_learning; flags
#   hosting_grand_debate, attending_grand_debate, grand_debate_begins,
#   grand_debate_has_ended, debate_winner, do_not_disturb, love_letters_flag,
#   pen_pal_flag, delighted_parent_flag, poetic_letter,
#   got_fake_note_planted, embarrassingly_misinformed, plagiarism_discovered;
#   global flags holding_grand_debate, held_grand_debate.
# Dropped: the score bands debate_score_00 - 20 (the winner is now simply the
#   best score) and the separate winner notice MNM.1988.
###############################################################################

# debate.1 The invitations
# The host sends out the invitations; the debate opens in two months
# (debate.4).
# Scope: ROOT is the host. From the decision hermetic_grand_debate.

	id = debate.1
	title = EVTTITLEdebate
	desc = EVTDESCdebate1
	picture = GFX_evt_gathering_of_minds
	is_triggered_only = yes

	option = {
		name = EVTOPTAdebate1
		hidden_effect = {
			set_character_flag = hosting_grand_debate
			character_event = { id = debate.4 days = 60 }
			society = {
				any_society_member = {
					limit = {
						is_adult = yes
						is_incapable = no

						NOT = { character = ROOT }
					}
					character_event = { id = debate.2 }
				}
			}
		}
	}
}

# debate.3 Safety catch
# Hidden. Clears the debate's flags if the chain broke.
# Scope: ROOT is the host or a member. From the decision and debate.2,
# 300 days on.
character_event = {
	id = debate.3
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		debate_leave_effect = yes
		clr_character_flag = grand_debate_has_ended

		}
	}

# debate.6 The score
# Hidden. An attendee's score: his learning, his way with people and words,
# and what befell him at the debate.
# Scope: ROOT is the attendee. From debate.4.
character_event = {
	id = debate.6
	hide_window = yes
	is_triggered_only = yes

	trigger = { has_character_flag = attending_grand_debate }

	immediate = {
		export_to_variable = { which = debate_learning value = learning }
		change_variable = { which = debate_score which = debate_learning }
		if = {
			limit = { trait = gregarious }
			change_variable = { which = debate_score value = 1 }
		}
		if = {
			limit = { trait = socializer }
			change_variable = { which = debate_score value = 1 }
		}
		if = {
			limit = { trait = shy }
			change_variable = { which = debate_score value = -1 }
		}
		if = {
			limit = { has_fair_trait_trigger = yes }
			change_variable = { which = debate_score value = 1 }
		}
		if = {
			limit = { trait = stutter }
			change_variable = { which = debate_score value = -2 }
		}
		if = {
			limit = { trait = lisp }
			change_variable = { which = debate_score value = -2 }
		}
		if = {
			limit = { is_incapable = yes }
			change_variable = { which = debate_score value = -100 }
		}
		random = { # A good or bad day
			chance = 50
			change_variable = { which = debate_score value = 2 }
		}
	}
}

# debate.30 Too many late nights
# He works himself to the bone. He can carry on (stressed), dose himself
# from his ingredients, or give up on winning.
# Scope: ROOT is the attendee. From debate.5.
character_event = {
	id = debate.30
	desc = EVTDESCdebate30
	picture = GFX_evt_secret_ceremony
	border = GFX_event_normal_frame_religion
	is_triggered_only = yes

	trigger = { NOT = { trait = stressed } }

	immediate = {
		select_random_hermetic_ingredient_herb_effect = yes
		if = {
			limit = { has_selected_ingredient_1_trigger = no }
			select_random_hermetic_ingredient_effect = yes
		}
	}

	option = {
		name = EVTOPTAdebate30
		custom_tooltip = { text = EVTTOOLTIPdebate_better }
		add_trait = stressed
	}
	option = {
		trigger = { has_selected_ingredient_1_trigger = yes }
		name = EVTOPTCdebate30
		custom_tooltip = { text = EVTTOOLTIPdebate30_ingredient }
		custom_tooltip = { text = EVTTOOLTIPdebate_better }
		random = {
			chance = 50
			add_trait = stressed
		}
		if = {
			limit = { selected_ingredient_is_sleep_inducing_trigger = no }
			random = {
				chance = 50
				add_character_modifier = {
					modifier = nauseous_from_potion
					years = 1
				}
			}
		}
		destroy_selected_ingredient_1_effect = yes
	}
	option = {
		name = EVTOPTBdebate30
		custom_tooltip = { text = EVTTOOLTIPdebate_worse }
		hidden_effect = { change_variable = { which = debate_score value = -2 } }
	}

	after = {
		remove_selection_of_ingredients_effect = yes
	}
}

# debate.32 A chance to sabotage
# A gifted rival leans on his notes. He can plant forged notes (debate.33),
# spill ink over the real ones (debate.34), or leave them be.
# Scope: ROOT is the attendee. From debate.5.
character_event = {
	id = debate.32
	desc = EVTDESCdebate32
	picture = GFX_evt_shady_meeting
	border = GFX_event_normal_frame_religion
	is_triggered_only = yes

	trigger = {
		society = { any_society_member = { debate_fellow_trigger = yes learning = ROOT } }
	}

	immediate = {
		society = {
			random_society_member = {
				limit = { debate_fellow_trigger = yes learning = ROOT }
				save_event_target_as = sabotage_target
			}
		}
	}

	option = {
		name = EVTOPTAdebate32
		custom_tooltip = { text = EVTTOOLTIPAdebate32 }
		event_target:sabotage_target = {
			set_character_flag = got_fake_note_planted
			character_event = { id = debate.33 }
		}
	}
	option = {
		name = EVTOPTBdebate32
		custom_tooltip = { text = EVTTOOLTIPBdebate32 }
		event_target:sabotage_target = {
			character_event = { id = debate.34 }
		}
	}
	option = {
		name = EVTOPTCdebate32
		if = {
			limit = { trait = deceitful }
			remove_trait = deceitful
		}
		else = {
			random = {
				chance = 35
				add_trait_kind_effect = yes
			}
		}
	}
}

# debate.33 Forgotten notes
# He finds notes in a corridor: real research, or forgeries planted by a
# rival. Using them is judged later (debate.36); the planter learns how it
# went (debate.37).
# Scope: ROOT is the attendee, FROM whoever planted them (or ROOT).
# From debate.5 and debate.32.
character_event = {
	id = debate.33
	desc = {
		trigger = { NOT = { has_character_flag = got_fake_note_planted } }
		text = EVTDESCdebate33
	}
	desc = {
		trigger = { has_character_flag = got_fake_note_planted }
		text = EVTDESCdebate33_forged
	}
	picture = GFX_evt_gathering_of_minds
	border = GFX_event_normal_frame_religion
	hide_from = yes
	is_triggered_only = yes

	option = {
		name = EVTOPTAdebate33
		custom_tooltip = {
			text = EVTTOOLTIPdebate33
			hidden_effect = {
				if = {
					limit = { has_character_flag = got_fake_note_planted }
					set_character_flag = embarrassingly_misinformed
				}
				character_event = { id = debate.36 days = 5 }
			}
		}
	}
	option = {
		trigger = {
			NOR = {
				has_character_flag = got_fake_note_planted
				intrigue = 18
			}
		}
		name = EVTOPTBdebate33
		prestige = 25
	}
	option = {
		trigger = {
			has_character_flag = got_fake_note_planted
			intrigue = 18
		}
		name = EVTOPTCdebate33
		tooltip_info = intrigue
		prestige = 25
	}

	after = {
		hidden_effect = {
			if = {
				limit = { has_character_flag = got_fake_note_planted }
				FROM = { character_event = { id = debate.37 days = 10 } }
			}
		}
		clr_character_flag = got_fake_note_planted
	}
}
