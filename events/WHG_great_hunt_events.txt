###############################################################################
# The Great Hunt
# Namespace: herdhunt (was TOG.100 - TOG.299 in oldgods_mongol_events.txt)
# Faction: Beastmen
# Serves: every beastman herd
#
# In late summer a Beastlord sounds the horns and the chieftains of the herd
# must come. For two months the warherd sweeps the deep forest in a great
# ring, driving stag, boar and the men of the woodland villages before it
# towards the herdstone. While the ring tightens a chieftain may be gored,
# two may fight to settle who stands nearer the herdstone (the Pretender's
# way), or a chieftain's ungors may break the line or kill before the
# Beastlord, who may beat him for it or let it pass. At the herdstone the
# Beastlord takes the first kill, then the chieftains take theirs. A young
# gor may outshine them all, or a sounder of razorgors may be caught in the
# ring: a runt to keep, or the whole sounder yoked for war. The captives
# left over are let run to tell the villages, or dragged to the herdstone
# for the Dark Gods. Then the herd feasts (the shared feast scenes in
# WHG_northern_rites_events.txt), and once in a life a Beastlord may keep a
# skull-totem of the hunt. If war comes, the hunt breaks up.
#
# A kept razorgor lives at its master's side for years: it awes visiting
# chieftains, mauls a courtier, pines in its pen, and dies one day.
#
# The hunt                             herdhunt.1 - herdhunt.2
# Trouble in the ring                  herdhunt.10 - herdhunt.18
# At the herdstone                     herdhunt.20 - herdhunt.28
# War breaks the hunt                  herdhunt.30
# The razorgor                         herdhunt.40 - herdhunt.44
# Housekeeping                         herdhunt.90 - herdhunt.91
#
# Fired from: the hold_nerge decision in minor_decisions.txt (herdhunt.1,
#   with the safety herdhunt.90). Shared triggers and effects are in
#   WHG_faith_triggers.txt and WHG_faith_effects.txt; the skull-totem is
#   in 41_artifacts_old_gods.txt. The feast scenes are rites.50 - rites.59
#   (flags rites_feast_host and rites_feast_guest).
# Saved: event targets herdhunt_brawler, herdhunt_rival, herdhunt_scorner,
#   herdhunt_victim; flags holding_nerge (decision), herdhunt_host,
#   herdhunt_hunter, herdhunt_incident, herdhunt_gap, herdhunt_early_kill,
#   herdhunt_finishing, herdhunt_found_champion, herdhunt_found_razorgors
#   (cleared by herdhunt_clear_flags_effect), herdhunt_razorgor and
#   herdhunt_razorgor_sullen (the pet), herdhunt_maul_* (cleared by
#   herdhunt.42), took_razorgor_herd and took_hunt_totem (once per life).
###############################################################################

# Every chieftain of the herd fit to hunt must come (herdhunt.2). The ring
# closes on the herdstone after about two months (herdhunt.20); trouble may
# break out three times on the way (herdhunt.10). War breaks it up
# (herdhunt.18, herdhunt.30).
# Scope: ROOT is the Beastlord (flag holding_nerge).

	id = herdhunt.1
	desc = EVTDESCherdhunt1

		name = EVTOPTAherdhunt1
		set_character_flag = herdhunt_host

				limit = { herdhunt_can_hunt_trigger = yes }

				character_event = { id = herdhunt.91 days = 150 } # Safety

			limit = { ai = no }

# herdhunt.10 The ring tightens
# Picks one chieftain who has had no trouble yet: gored (herdhunt.11), a
# fight with another chieftain (herdhunt.13), or a broken line or an early
# kill (herdhunt.15). Often nothing happens. War breaks the hunt (herdhunt.30).
# Scope: ROOT is the Beastlord (flag herdhunt_host).
character_event = {
	id = herdhunt.10

	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_character_flag = herdhunt_host
		NOT = { has_character_flag = herdhunt_finishing }
	}

	immediate = {
		if = {
			limit = { war = yes }
			character_event = { id = herdhunt.30 }
		}
		else = {
			random_list = {
				40 = { }
				20 = { # Gored
					modifier = {
						factor = 0
						NOT = {
							any_vassal = {
								herdhunt_free_hunter_trigger = yes
								NOT = { trait = wounded }
							}
						}
					}
					random_vassal = {
						limit = {
							herdhunt_free_hunter_trigger = yes
							NOT = { trait = wounded }
						}
						preferred_limit = { NOT = { martial = 8 } }
						set_character_flag = herdhunt_incident
						character_event = { id = herdhunt.11 }
					}
				}
				20 = { # The Pretender's way
					modifier = {
						factor = 0
						NOT = {
							any_vassal = {
								count = 2
								herdhunt_free_hunter_trigger = yes
							}
						}
					}
					random_vassal = {
						limit = { herdhunt_free_hunter_trigger = yes }
						preferred_limit = { trait = wroth }
						set_character_flag = herdhunt_incident
						save_event_target_as = herdhunt_brawler
					}
					random_vassal = {
						limit = {
							herdhunt_free_hunter_trigger = yes
							NOT = { character = event_target:herdhunt_brawler }
						}
						set_character_flag = herdhunt_incident
						save_event_target_as = herdhunt_rival
					}
					event_target:herdhunt_brawler = { character_event = { id = herdhunt.13 } }
					event_target:herdhunt_rival = { character_event = { id = herdhunt.13 } }
					character_event = { id = herdhunt.14 }
				}
				20 = { # A broken line or an early kill
					modifier = {
						factor = 0
						NOT = { any_vassal = { herdhunt_free_hunter_trigger = yes } }
					}
					random_vassal = {
						limit = { herdhunt_free_hunter_trigger = yes }
						preferred_limit = {
							OR = {
								trait = slothful
								trait = greedy
								trait = arbitrary
								trait = imbecile
								NOT = { martial = 6 }
							}
						}
						set_character_flag = herdhunt_incident
						random_list = {
							50 = {
								modifier = { factor = 1.5 trait = slothful }
								modifier = { factor = 1.5 NOT = { martial = 6 } }
								set_character_flag = herdhunt_gap
							}
							50 = {
								modifier = { factor = 1.5 trait = greedy }
								modifier = { factor = 1.5 trait = arbitrary }
								modifier = { factor = 0.5 trait = patient }
								set_character_flag = herdhunt_early_kill
							}
						}
						character_event = { id = herdhunt.15 }
					}
				}
			}
		}
	}
}

# (8%, once per life, none held).
# Scope: ROOT is the Beastlord (flag herdhunt_host) or a chieftain, FROM the
#   Beastlord.
character_event = {
	id = herdhunt.28
	desc = {
		trigger = { has_character_flag = herdhunt_host }
		text = EVTDESCherdhunt28_host
	}
	desc = {
		trigger = { NOT = { has_character_flag = herdhunt_host } }
		text = EVTDESCherdhunt28_hunter
	}
	picture = GFX_evt_beastman_chief
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	option = {
		name = EVTOPTAherdhunt28
		if = {
			limit = { has_character_flag = herdhunt_host }
			add_character_modifier = {
				name = morale_from_nerge
				duration = 730
			}
			add_character_modifier = {
				name = held_nerge_timer
				duration = 1460
				hidden = yes
			}
			if = {
				limit = {
					NOT = { has_character_flag = took_hunt_totem }
					NOT = { any_artifact = { has_artifact_flag = hunt_totem } }
				}
				random = {
					chance = 8
					set_character_flag = took_hunt_totem
					add_artifact = great_hunt_totem
				}
			}
			hidden_effect = {
				any_vassal = {
					limit = { has_character_flag = herdhunt_hunter }
					character_event = { id = herdhunt.28 }
				}
			}
		}
		herdhunt_clear_flags_effect = yes
	}
}

# herdhunt.40 A year with the razorgor
# Every two to three years: it awes visiting chieftains (herdhunt.41), mauls
# a courtier (herdhunt.42), pines in its pen (herdhunt.43) or dies
# (herdhunt.44, likelier once it has pined).
# Scope: ROOT is the owner (flag herdhunt_razorgor).
character_event = {
	id = herdhunt.40

	hide_window = yes
	is_triggered_only = yes

	trigger = { has_character_flag = herdhunt_razorgor }

	immediate = {
		random_list = {
			25 = { }
			25 = { character_event = { id = herdhunt.41 } }
			20 = {
				modifier = {
					factor = 0
					NOT = { any_courtier = { herdhunt_maul_victim_trigger = yes } }
				}
				random_courtier = {
					limit = { herdhunt_maul_victim_trigger = yes }
					save_event_target_as = herdhunt_victim
				}
				random_list = {
					50 = { set_character_flag = herdhunt_maul_wound }
					30 = { set_character_flag = herdhunt_maul_maim }
					20 = { set_character_flag = herdhunt_maul_kill }
				}
				character_event = { id = herdhunt.42 }
			}
			15 = {
				modifier = { factor = 0 has_character_flag = herdhunt_razorgor_sullen }
				character_event = { id = herdhunt.43 }
			}
			15 = {
				modifier = { factor = 3 has_character_flag = herdhunt_razorgor_sullen }
				character_event = { id = herdhunt.44 }
			}
		}
		character_event = { id = herdhunt.40 days = 730 random = 365 }
	}
}

# herdhunt.42 The razorgor mauls a courtier
# Wounded, maimed or killed (flag herdhunt_maul_* on the owner). The owner
# may have the beast butchered.
# Scope: ROOT is the owner; herdhunt_victim the courtier.
character_event = {
	id = herdhunt.42
	desc = {
		trigger = { has_character_flag = herdhunt_maul_wound }
		text = EVTDESCherdhunt42_wound
	}
	desc = {
		trigger = { has_character_flag = herdhunt_maul_maim }
		text = EVTDESCherdhunt42_maim
	}
	desc = {
		trigger = { has_character_flag = herdhunt_maul_kill }
		text = EVTDESCherdhunt42_kill
	}
	picture = GFX_evt_hunt
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	immediate = {
		event_target:herdhunt_victim = {
			if = {
				limit = { ROOT = { has_character_flag = herdhunt_maul_wound } }
				add_trait = wounded
			}
			else_if = {
				limit = { ROOT = { has_character_flag = herdhunt_maul_maim } }
				add_maimed_trait_effect = yes
			}
			else = {
				death = { death_reason = death_accident_boar }
			}
		}
	}

	option = { # It is a beast; it does as beasts do
		name = EVTOPTAherdhunt42
		ai_chance = { factor = 80 }
	}
	option = { # Butcher it
		name = EVTOPTBherdhunt42
		clr_character_flag = herdhunt_razorgor
		clr_character_flag = herdhunt_razorgor_sullen
		custom_tooltip = { text = herdhunt_razorgor_butchered_tt }
		prestige = -25
		ai_chance = {
			factor = 20
			modifier = { factor = 2 has_character_flag = herdhunt_maul_kill }
		}
	}

	after = {
		clr_character_flag = herdhunt_maul_wound
		clr_character_flag = herdhunt_maul_maim
		clr_character_flag = herdhunt_maul_kill
	}
}

# herdhunt.43 The razorgor pines
# Let it go back to the forest, or keep it penned (it may die of it,
# herdhunt.44).
# Scope: ROOT is the owner.

	id = herdhunt.43
	desc = EVTDESCherdhunt43
	picture = GFX_evt_beastmen_mist
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	option = { # Let it go
		name = EVTOPTAherdhunt43
		clr_character_flag = herdhunt_razorgor
		custom_tooltip = { text = herdhunt_razorgor_released_tt }
		prestige = -50
		add_trait_kind_effect = yes
		if = {
			limit = { trait = cruel }
			remove_trait = cruel
		}
		ai_chance = {
			factor = 30
			modifier = { factor = 2 trait = kind }
		}
	}
	option = { # Keep it penned
		name = EVTOPTBherdhunt43
		set_character_flag = herdhunt_razorgor_sullen
		custom_tooltip = { text = herdhunt_razorgor_penned_tt }
		random = {
			chance = 33
			add_trait_cruel_effect = yes
		}
		ai_chance = { factor = 70 }
	}
}
