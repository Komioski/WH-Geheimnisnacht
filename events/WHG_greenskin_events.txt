# greenskin.1 How big you grew
# An orc or goblin coming of age gets a size trait from martial skill:
# small below 10, average below 15, large below 25, huge from 25.
# Fired at 16, and every five years through greenskin.5 and greenskin.6
# for greenskins created as adults.
# Scope: ROOT is the greenskin.
character_event = {
	id = greenskin.1
	title = YOUR_SIZE
	desc = EVTDESCgreenskin1
	picture = GFX_evt_orcs
	is_triggered_only = yes # on_adulthood, greenskin.5, greenskin.6

	min_age = 16

	trigger = {
		OR = { culture_group = goblin_group culture_group = orc_group }
		NOR = {
			trait = orc_small
			trait = orc_average
			trait = orc_large
			trait = orc_huge
			trait = goblin_small
			trait = goblin_average
			trait = goblin_large
			trait = goblin_huge
		}
	}

	option = { # Arrrrgh!
		name = ARRRRGH
		if = {
			limit = { culture_group = orc_group }
			if = { limit = { martial = 25 } add_trait = orc_huge }
			else_if = { limit = { martial = 15 } add_trait = orc_large }
			else_if = { limit = { martial = 10 } add_trait = orc_average }
			else = { add_trait = orc_small }
		}
		else = {
			if = { limit = { martial = 25 } add_trait = goblin_huge }
			else_if = { limit = { martial = 15 } add_trait = goblin_large }
			else_if = { limit = { martial = 10 } add_trait = goblin_average }
			else = { add_trait = goblin_small }
		}
	}
}

# greenskin.10 Da Waaagh! in yer 'ead
# A landed greenskin of 30 or more who is pious (1000), a scholar, a mystic
# or a theocracy can take up the Lore of the Waaagh! Craven or content
# greenskins can only refuse. Theocracies may be refused by the gods (25%)
# or made full shamans (40%). Tells the father (greenskin.11) and the top
# liege (greenskin.12). One chance per character: refusing sets a flag.
# Scope: ROOT is the greenskin.
character_event = {
	id = greenskin.10
	title = MAGIC
	desc = EVTDESCgreenskin10
	picture = GFX_evt_orcs
	is_triggered_only = yes # on_five_year_pulse, random list greenskin_shamans

	religion_group = greenskin_gods_group
	min_age = 30

	trigger = {
		is_landed = yes
		OR = { culture_group = goblin_group culture_group = orc_group }
		OR = { piety = 1000 trait = scholar trait = mystic is_theocracy = yes }
		NOT = { trait = lore_waaagh }
		NOT = { has_character_flag = refused_shaman }
	}

	option = { # I will get da power!
		name = EVTOPTAgreenskin10
		trigger = { NOR = { trait = craven trait = content } }
		if = {
			limit = { is_theocracy = yes }
			random_list = {
				25 = { set_character_flag = refused_shaman }
				40 = { }
				40 = { add_trait = greenskin_shaman }
			}
		}
		if = {
			limit = { NOT = { has_character_flag = refused_shaman } }
			add_trait = lore_waaagh
			magic_lvl_increase_effect = yes
			enable_magic_xp_effect = yes
			hidden_effect = {
				top_liege = {
					if = {
						limit = {
							NOT = { character = ROOT }
							OR = { culture_group = orc_group culture_group = goblin_group }
						}
						character_event = { id = greenskin.12 days = 7 }
					}
				}
				if = {
					limit = { father = { is_alive = yes } }
					father = { character_event = { id = greenskin.11 } }
				}
			}
		}
	}

	option = { # Nah, I just likes to use my smasher!
		name = EVTOPTBgreenskin10
		set_character_flag = refused_shaman
	}
}

# greenskin.15 A shaman joins the warband
# At war, an orc ruler of 20 or more can take in a young shaman as a son of
# the dynasty (older shamans for older warbosses), or refuse out of fear.
# At most once every ten years.
# Scope: ROOT is the orc ruler.
character_event = {
	id = greenskin.15
	desc = EVTDESCgreenskin15
	picture = GFX_evt_orcs
	is_triggered_only = yes # on_bi_yearly_pulse

	min_age = 20
	war = yes
	prisoner = no

	trigger = {
		is_landed = yes
		culture_group = orc_group
		OR = {
			NOT = { has_character_flag = greenskin_took_shaman }
			had_character_flag = { flag = greenskin_took_shaman days = 3650 }
		}
	}

	immediate = {
		clr_character_flag = greenskin_took_shaman
		set_character_flag = greenskin_took_shaman
	}

	option = { # Tell 'im ta stay outta mah way
		name = EVTOPTAgreenskin15
		prestige = 50
		piety = 25
		if = {
			limit = { age >= 90 }
			create_random_priest = {
				random_traits = yes
				dynasty = ROOT
				culture = ROOT
				religion = ROOT
				female = no
				age = 82
			}
			new_character = {
				character_event = { id = aiclean.31 } # Racial trait
				change_learning = 8
				change_diplomacy = 6
				greenskin_new_shaman_effect = yes
			}
		}
		else_if = {
			limit = { age >= 60 }
			create_random_priest = {
				random_traits = yes
				dynasty = ROOT
				culture = ROOT
				religion = ROOT
				female = no
				age = 53
			}
			new_character = {
				character_event = { id = aiclean.31 } # Racial trait
				change_learning = 6
				change_diplomacy = 5
				greenskin_new_shaman_effect = yes
			}
		}
		else = {
			create_random_priest = {
				random_traits = yes
				dynasty = ROOT
				culture = ROOT
				religion = ROOT
				female = no
				age = 19
			}
			new_character = {
				character_event = { id = aiclean.31 } # Racial trait
				change_learning = 5
				change_diplomacy = 4
				greenskin_new_shaman_effect = yes
			}
		}
		recalc_succession = yes
	}

	option = { # Wot if he makes me 'ead splode?
		name = EVTOPTBgreenskin15
		prestige = -50
		random = { chance = 25 add_trait = craven }
	}
}

# warboss.35 Da big boss is dead
# When an independent greenskin king or emperor who was not weak dies, every

	id = warboss.35

		religion_group = greenskin_gods_group

		NOT = { has_character_modifier = weak_warboss }

		any_vassal = { character_event = { id = warboss.36 } }

		}

			limit = { government = greenskin_waagh_government }
			current_heir = {
				save_event_target_as = nomadic_heir
				character_event = { id = waaagh.50 days = 6 }
				if = {
					limit = {
						check_variable = { which = local_tyrrannical_prestige value >= 2000 }
					}
					divide_variable = { which = local_tyrrannical_prestige value = 2 }
					if = {
						limit = {
							check_variable = { which = local_tyrrannical_prestige value >= 2000 }
						}
						divide_variable = { which = local_tyrrannical_prestige value = 2 }
					}
				}
				any_sibling = { prestige = local_tyrrannical_prestige }
				prestige = local_tyrrannical_prestige
			}
		}
		else = {
			current_heir = {
				if = {
					limit = {
						check_variable = { which = local_tyrrannical_prestige value >= 2000 }
					}
					divide_variable = { which = local_tyrrannical_prestige value = 3 }

					}
				}
				any_sibling = { prestige = local_tyrrannical_prestige }

# warboss.60 A troublesome greenskin population
# When a greenskin takes over land, every province in his realm held by a
# non-greenskin gets the troublesome greenskin population modifier and
# loses its greenskin settlement modifiers.
# Scope: ROOT is the new holder.
character_event = {
	id = warboss.60
	desc = "EVTDESCwarboss60"
	picture = GFX_evt_orcs

	religion_group = greenskin_gods_group

	is_triggered_only = yes # on_new_holder, on_new_holder_inheritance, on_new_holder_usurpation

	trigger = {
		any_realm_province = {
			NOT = { has_province_modifier = greenskin_population_bad }
			holder_scope = {
				NOT = {
					religion_group = greenskin_gods_group
					OR = { culture_group = goblin_group culture_group = orc_group }
				}
			}
		}
	}

	option = {
		name = "EVTOPTAwarboss60"
		any_realm_province = {
			if = {
				limit = {
					NOT = { has_province_modifier = greenskin_population_bad }
					holder_scope = {
						NOT = {
							religion_group = greenskin_gods_group
							OR = { culture_group = goblin_group culture_group = orc_group }
						}
					}
				}
				add_province_modifier = { name = greenskin_population_bad duration = -1 }
				remove_province_modifier = large_greenskin_population
				remove_province_modifier = medium_greenskin_population
				remove_province_modifier = minimal_greenskin_population
			}
		}
	}
}

# warboss.61 Greenskins back in charge
# When a greenskin takes over land, provinces held by greenskins lose the
# troublesome greenskin population modifier.
# Scope: ROOT is the new holder.
character_event = {
	id = warboss.61
	hide_window = yes

	religion_group = greenskin_gods_group

	is_triggered_only = yes # on_new_holder, on_new_holder_inheritance, on_new_holder_usurpation

	trigger = {
		any_realm_province = {
			has_province_modifier = greenskin_population_bad
			holder_scope = { OR = { culture_group = goblin_group culture_group = orc_group } }
		}
	}

	immediate = {
		any_realm_province = {
			if = {
				limit = {
					has_province_modifier = greenskin_population_bad
					holder_scope = {
						OR = { culture_group = goblin_group culture_group = orc_group }
					}
				}
				remove_province_modifier = greenskin_population_bad
			}
		}
	}

	option = {
		name = "OK"
	}
}

# warboss.62 Settlers driven out
# When a non-greenskin takes a county that is not greenskin land, it loses
# its greenskin settlement modifiers. (Was a province event, which these
# character on_actions cannot fire.)
# Scope: ROOT is the new holder, FROM the county title.
character_event = {
	id = warboss.62
	hide_window = yes

	is_triggered_only = yes # on_new_holder, on_new_holder_inheritance, on_new_holder_usurpation

	only_rulers = yes

	trigger = {
		NOT = { religion_group = greenskin_gods_group }
		NOR = { culture_group = goblin_group culture_group = orc_group }
		FROM = {
			tier = count
			location = {
				NOR = { culture_group = goblin_group culture_group = orc_group }
				OR = {
					has_province_modifier = large_greenskin_population
					has_province_modifier = medium_greenskin_population
					has_province_modifier = minimal_greenskin_population
				}
			}
		}
	}

	immediate = {
		FROM = {
			location = {
				remove_province_modifier = large_greenskin_population
				remove_province_modifier = medium_greenskin_population
				remove_province_modifier = minimal_greenskin_population
			}
		}
	}
}

# warboss.70 Fund the expedition?
# The marshal asks for gold to clear greenskins out of a county. Hard in the
# Badlands and Darklands, in forests (forest goblins), mountains (night
# goblins), hills (hill goblins), jungles and against Black Orcs; easier
# with many commanders, big levies and a martial realm. Success
# (warboss.71) or failure (warboss.72) in about a month.
# Scope: ROOT is the liege, FROM the marshal, FROMFROM the county.
character_event = {
	id = warboss.70
	desc = EVTDESCwarboss70
	picture = GFX_evt_orcs_charging
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # decision cleanse_greenskin_presence

	option = {
		name = EVTOPTAwarboss70
		ai_chance = { factor = 1 }
		prestige = 100
		#Night Goblins
		if = {
			limit = {
				FROM = {
					location = {
						OR = { terrain = mountain terrain = impassable_mountains }
						culture = goblin_night
					}
				}
			}
			scaled_wealth = { value = -1.5 min = -40 max = -150 }
		}
		#Goblins
		else_if = {
			limit = {
				FROM = {
					location = { OR = { terrain = forest terrain = woods } culture = goblin_forest }
				}
			}
			scaled_wealth = { value = -1.5 min = -40 max = -150 }
		}
		#Black Orcs
		else_if = {
			limit = {
				FROM = {
					location = {
						OR = { region = world_darklands region = world_badlands }
						culture = orc_black
					}
				}
			}
			scaled_wealth = { value = -2 min = -50 max = -150 }
		}
		#Everything else
		else = { scaled_wealth = { value = -1 min = -40 max = -120 } }
		hidden_tooltip = {
			random_list = {
				75 = {
					modifier = {
						#Almost impossible to scrub greenskins out of the Badlands
						factor = 0.25
						FROM = { location = { region = world_badlands } }
					}
					modifier = {
						factor = 1.15
						FROM = { location = { region = world_badlands } }
						regional_percentage = {
							region = world_badlands
							percentage <= 0.35
							OR = { culture_group = orc_group culture_group = goblin_group }
						}
					}
					modifier = {
						#Almost impossible to scrub greenskins out of the Darklands
						factor = 0.25
						FROM = { location = { region = world_darklands } }
					}
					modifier = {
						factor = 1.15
						FROM = { location = { region = world_darklands } }
						regional_percentage = {
							region = world_darklands
							percentage <= 0.35
							OR = { culture_group = orc_group culture_group = goblin_group }
						}
					}
					modifier = {
						factor = 1.20
						has_assigned_minor_title = { title = title_commander count >= 4 }
					}
					modifier = { factor = 1.25 realm_levies >= 5000 }
					modifier = { factor = 1.15 realm_martial >= 20 }
					character_event = { id = warboss.71 days = 25 random = 25 }
				}
				25 = {
					modifier = {
						#Almost impossible to scrub greenskins out of the Badlands
						factor = 2
						FROM = { location = { region = world_badlands } }
					}
					modifier = {
						factor = 2
						FROM = { location = { region = world_badlands } }
						regional_percentage = {
							region = world_badlands
							percentage >= 0.50
							OR = { culture_group = orc_group culture_group = goblin_group }
						}
					}
					modifier = {
						#Almost impossible to scrub greenskins out of the darklands
						factor = 2
						FROM = { location = { region = world_darklands } }
					}
					modifier = {
						factor = 2
						FROM = { location = { region = world_darklands } }
						regional_percentage = {
							region = world_darklands
							percentage >= 0.50
							OR = { culture_group = orc_group culture_group = goblin_group }
						}
					}
					modifier = {
						#Almost impossible to scrub Forest Goblins out of Forests
						factor = 2
						FROM = {
							location = {
								OR = { terrain = forest terrain = woods }
								culture = goblin_forest
							}
						}
					}
					modifier = {
						#Almost impossible to scrub Night Goblins out of Mountains
						factor = 2
						FROM = {
							location = {
								OR = { terrain = mountain terrain = impassable_mountains }
								culture = goblin_night
							}
						}
					}
					modifier = {
						#Almost impossible to scrub Hill Goblins out of Hills
						factor = 2
						FROM = {
							location = {
								OR = { terrain = hills terrain = marsh terrain = mountain }
								culture = goblin_hill
							}
						}
					}
					modifier = {
						#Almost impossible to scrub greenskins out of jungle
						factor = 1.5
						FROM = { location = { OR = { terrain = jungle terrain = marsh } } }
					}
					modifier = {
						#Almost impossible to scrub black orcs
						factor = 2
						FROM = { location = { culture = orc_black } }
					}
					modifier = {
						factor = 1.20
						has_assigned_minor_title = { title = title_commander count <= 2 }
					}
					modifier = { factor = 1.25 realm_levies <= 2500 }
					modifier = { factor = 1.15 realm_martial < 20 }
					modifier = {
						factor = 0.75
						has_bloodline_flag = { gs_exterminator } #having the bloodline for clearing out greenskins makes you less likely to fail
					}

					character_event = { id = warboss.72 days = 25 random = 25 }
				}
			}
		}
	}
	option = {
		name = EVTOPTBwarboss70
		ai_chance = { factor = 0.1 }
	}
}

# warboss.71 Da greenskins are gone
# The greenskin modifiers are removed and the land takes the cleanser's
# culture and religion (Tomb Kings, Jade vampires, Blood Dragons and witch
# hunters use a fitting local culture instead).
# Scope: ROOT is the liege, FROMFROM the county.
character_event = {
	id = warboss.71
	desc = EVTDESCwarboss71
	picture = GFX_evt_greenskindeath
	border = GFX_event_normal_frame_war
	is_triggered_only = yes # warboss.70

	option = { # The land is ours again
		name = EVTOPTAwarboss71
		hidden_tooltip = {
			FROMFROM = {
				location = {
					remove_province_modifier = minimal_greenskin_population
					remove_province_modifier = medium_greenskin_population
					remove_province_modifier = large_greenskin_population
					remove_province_modifier = greenskin_population_bad
					remove_province_modifier = greenskins_wrecking_province
					remove_province_modifier = greenskins_animosity_province
				}
				refill_holding_levy = yes
			}
			# For the bloodline ambition of clearing out greenskins
			change_variable = { which = bloodline_exterminate_gs value = 1 }
			if = {
				limit = {
					check_variable = { which = bloodline_exterminate_gs value = 14 }
					has_ambition = obj_forge_bloodline
				}
				narrative_event = { id = HF.24258 days = 1 }
			}
			# Settle the land if greenskins still live there
			if = {
				limit = {
					FROMFROM = {
						location = {
							OR = { culture_group = orc_group culture_group = goblin_group }
						}
					}
				}
				if = {
					limit = { trait = creature_nehekharan }
					FROMFROM = { location = { culture = nehekka religion = atruhayid } }
				}
				else_if = {
					limit = { culture = living_jade }
					FROMFROM = { location = { culture = lizardman_lost religion = vampiric } }
				}
				else_if = {
					limit = {
						OR = { culture = blooddragons_culture culture = witchhunter_culture }
					}
					FROMFROM = { location = { warboss_cleansed_land_settlers_effect = yes } }
				}
				else = { FROMFROM = { location = { religion = ROOT culture = ROOT } } }
			}
		}
	}
}

# warboss.72 The greenskins hold
# The expedition fails. One time in ten (four in ten in the Badlands) the
# greenskins rise as a horde (warboss.50).
# Scope: ROOT is the liege, FROMFROM the county.
character_event = {
	id = warboss.72
	desc = EVTDESCwarboss72
	picture = GFX_evt_orcs
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # warboss.70

	option = {
		name = EVTOPTAwarboss72
		hidden_tooltip = {
			FROMFROM = {
				random = {
					chance = 10
					modifier = {
						#Try to scrub orcs out of the Badlands, have a bad time
						factor = 4
						FROMFROM = { location = { region = world_badlands } }
					}
					modifier = {
						factor = 0
						# Not if there is already an ongoing greenskin uprising for this province
						ROOT = {
							war = yes
							any_war = {
								defender = { character = PREV }
								using_cb = greenskin_uprising
							}
						}
					}
					modifier = {
						factor = 0
						# Not if there is already an ongoing greenskin uprising for this province
						top_liege = {
							war = yes
							any_war = {
								defender = { character = PREV }
								using_cb = greenskin_uprising
							}
						}
					}
					refill_holding_levy = yes #So that the rebellion is not odd-10 orcs
					province_event = { id = warboss.50 }
				}
			}
		}
	}
}
