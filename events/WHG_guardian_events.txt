###############################################################################
# The guardian and the ward: replies to a ward's leanings, guardianship,
# faith and culture lessons, and a child's questions
# Namespace: guardian (was 2000 - 2490 and 38100 - 38235 in the vanilla
#   guardian_events.txt)
# Faction: Generic
# Serves: every guardian and every child with one
#
# Part 1: Replies to a ward's leanings            guardian.10 - guardian.250
#   (guardian.N answers childtrait.N in WHG_childhood_events: the ward
#   leans to a vice or a virtue and the guardian encourages it, makes an
#   example of it, prays or reaches for the birch; childtrait.152 and .242,
#   the Araby versions, use guardian.140 and .240)
#
# Callers: childtrait.10 - childtrait.250 (WHG_childhood_events).
# Notify trait events (38150 - 38183, 38250 - 38283) are in
#   trait_notification.txt.
# Scripted triggers: common/scripted_triggers/WHG_guardian_triggers.txt
# Localisation: localisation/00_WHG_guardian_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = guardian

# guardian.10 The gluttonous ward
# Fired by childtrait.10: the guardian hears that the ward leans gluttonous.
# Scope: ROOT is the guardian, FROM the ward.
character_event = {
	id = guardian.10
	desc = EVTDESCguardian10
	picture = GFX_evt_child_play

	is_triggered_only = yes

	option = {
		name = EVTOPTAguardian10 # Encourage Gluttony
		ai_chance = {
			factor = 10
			modifier = { factor = 4 trait = gluttonous }
			modifier = { factor = 2 trait = slothful }
			modifier = { factor = 2 trait = hedonist }
			modifier = { factor = 0 trait = temperate }
		}

		if = { limit = { trait = temperate } prestige = -5 }
		if = { limit = { trait = gluttonous } prestige = 5 }
		FROM = {
			tooltip = { add_trait = gluttonous }
			opinion = { who = ROOT modifier = opinion_happy_with_guardian years = 2 }
		}
	}

	option = {
		name = EVTOPTBguardian10 # birch the ward
		trigger = { NOT = { trait = kind } }
		ai_chance = {
			factor = 10
			modifier = { factor = 5 trait = cruel }
			modifier = { factor = 5 trait = wroth }
			modifier = { factor = 0.1 trait = gluttonous }
		}

		pacifists_lose_piety_effect = yes
		reverse_opinion = { who = FROM modifier = opinion_sad years = 2 }
		random = {
			chance = 40
			FROM = {
				remove_trait = gluttonous
				lose_weight_medium_effect = yes
				weight_trait_check_effect = yes
				hidden_tooltip = { character_event = { id = childtrait.11 } }
			}
		}
	}

	option = {
		name = EVTOPTCguardian10 # Talk to Him about moderation.
		trigger = { NOT = { trait = gluttonous } }
		ai_chance = {
			factor = 10
			modifier = { factor = 20 trait = temperate }
			modifier = { factor = 10 trait = diligent }
			modifier = { factor = 10 trait = celibate }
		}

		random_list = {
			50 = {
				FROM = {
					add_trait = temperate
					lose_weight_medium_effect = yes
					weight_trait_check_effect = yes
					hidden_tooltip = {
						ROOT = { character_event = { id = 38176 } }
						character_event = { id = 38276 }
					}
				}
			}
			50 = {
				FROM = {
					remove_trait = gluttonous
					lose_weight_medium_effect = yes
					weight_trait_check_effect = yes
					hidden_tooltip = { character_event = { id = childtrait.11 } }
				}
			}
		}
	}
	option = {
		name = EVTOPTDguardian10 # pray to the gods for the ward
		trigger = { guardian_prays_trigger = yes }
		ai_chance = { factor = 2 modifier = { factor = 150 trait = zealous } }

		piety = 5
		random = {
			chance = 20
			FROM = {
				remove_trait = gluttonous
				lose_weight_medium_effect = yes
				weight_trait_check_effect = yes
				hidden_tooltip = { character_event = { id = childtrait.11 } }
			}
		}
	}
}
