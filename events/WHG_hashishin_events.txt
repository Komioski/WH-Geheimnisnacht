###############################################################################
# The Hashishin
# Namespace: hashishin (was 88000 - 88025 in vanilla
#   hashshashin_order_events.txt)
# Faction: Araby & Southlands
# Serves: the Hashishin, Ormazd's holy order of assassins (d_hashishin,
#   society the_assassins), and every southern ruler they watch
#
# The Hashishin hold the temple of Glak in Jebel Ezna and serve the Way of
# the One from the shadows. They can take a second fortress at Alq in Marib,
# go to ground when they lose their last one and come back when they win
# one again; if the order ever loses its Grand Master, they rise anew.
# They warn a decadent southern ruler, then kill a courtier, then the ruler.
# At war they kill the courtiers of their enemies and warn, then kill, a
# ruler who fights them. Now and then they take a contract on a courtier of
# a southern court. Whoever sacks their home may find a knife in the dark.
#
# Part 1: the order                     hashishin.1 - hashishin.4
# Part 2: the decadent ruler            hashishin.10 - hashishin.12
# Part 3: war with the Hashishin        hashishin.20 - hashishin.25
# Part 4: a contract                    hashishin.30
# Part 5: the sack of their home        hashishin.40 - hashishin.42
#
# Fired from: on_five_year_pulse (lists hashishin_return, hashishin_alq,
#   hashishin_contract) and on_yearly_pulse (lists hashishin_fortress,
#   hashishin_war) in common/on_actions; on_siege_won_leader in
#   WHG_war_on_actions.txt (hashishin.40). The warnings and killings of rulers
#   (hashishin.10 - hashishin.12, hashishin.24, hashishin.25) stay MTTH
#   behind ai = no or a warning flag.
#   Triggers in common/scripted_triggers/WHG_hashishin_triggers.txt.
# Saved: global flags assassins_founded (history), hashishin_alq,
#   hashishin_hidden; character flags decadence_warned,
#   decadence_courtier_killed, warned_by_assassins, hashishin_hunted;
#   event targets hashishin_master, hashishin_victim, hashishin_target.
# Dropped: vanilla's scatter and destruction (88002, 88004). With the
#   year = 1201 check always true in this mod, losing the temple of Glak
#   killed the Grand Master and ended the order for good, though it is a
#   holy order that can live without land. The Mongol strikes (88005 -
#   88009, commented out) are salvaged as Part 5.
###############################################################################

namespace = hashishin

# hashishin.1 The Hashishin rise again
# If the order has no Grand Master, a new one, Al-Mualim, rises with twelve
# followers and takes the order. Members of the Hashishin read it as a
# triumph. (Vanilla 88000 and 88003; the order exists from history.)
# Scope: ROOT is an independent ruler of Ormazd.
narrative_event = {
	id = hashishin.1
	title = EVTNAMEhashishin1
	desc = {
		trigger = { NOT = { society_member_of = the_assassins } }
		text = EVTDESChashishin1
	}
	desc = {
		trigger = { society_member_of = the_assassins }
		text = EVTDESChashishin1_member
	}
	picture = GFX_evt_assassination
	border = GFX_event_narrative_frame_intrigue

	major = yes
	major_trigger = {
		ai = no
		religion_group = ormazd_gods_group
	}
	hide_from = yes

	is_triggered_only = yes # on_five_year_pulse, list hashishin_return

	only_playable = yes
	only_independent = yes
	religion = ormazd

	trigger = {
		d_hashishin = { has_holder = no }
		NOT = { has_global_flag = assassins_destroyed }
		NOT = { has_alternate_start_parameter = { key = religion_names value = random } }
	}

	immediate = {
		set_global_flag = assassins_founded
		activate_title = { title = d_hashishin status = yes }
		create_character = {
			random_traits = no
			name = "Al-Mualim"
			dynasty = random
			religion = ormazd
			culture = cophic
			female = no
			age = 35
			attributes = {
				martial = 6
				diplomacy = 9
				stewardship = 8
				intrigue = 12
				learning = 12
			}
			health = 6
			fertility = 0.8
			trait = creature_human
			trait = elusive_shadow
			trait = patient
			trait = zealous
			trait = scholar
			trait = chaste
			trait = temperate
		}
		new_character = {
			wealth = 500
			d_hashishin = { grant_title = PREV }
			set_government_type = feudal_government
			if = {
				limit = { has_dlc = "Mystics" }
				character_event = { id = assassins.490 days = 1 }
			}
			create_character = { random_traits = yes dynasty = NONE religion = ormazd culture = cophic female = no age = 16 trait = creature_human }
			create_character = { random_traits = yes dynasty = NONE religion = ormazd culture = cophic female = no age = 18 trait = creature_human }
			create_character = { random_traits = yes dynasty = NONE religion = ormazd culture = cophic female = no age = 18 trait = creature_human }
			create_character = { random_traits = yes dynasty = NONE religion = ormazd culture = cophic female = no age = 20 trait = creature_human }
			create_character = { random_traits = yes dynasty = NONE religion = ormazd culture = cophic female = no age = 25 trait = creature_human }
			create_character = { random_traits = yes dynasty = NONE religion = ormazd culture = cophic female = no age = 26 trait = creature_human }
			create_character = { random_traits = yes dynasty = NONE religion = ormazd culture = cophic female = no age = 28 trait = creature_human }
			create_character = { random_traits = yes dynasty = NONE religion = ormazd culture = cophic female = yes age = 16 trait = creature_human }
			create_character = { random_traits = yes dynasty = NONE religion = ormazd culture = cophic female = yes age = 17 trait = creature_human }
			create_character = { random_traits = yes dynasty = NONE religion = ormazd culture = cophic female = yes age = 18 trait = creature_human }
			create_character = { random_traits = yes dynasty = NONE religion = ormazd culture = cophic female = yes age = 18 trait = creature_human }
			create_character = { random_traits = yes dynasty = NONE religion = ormazd culture = cophic female = yes age = 20 trait = creature_human }
		}
	}

	option = {
		name = {
			text = EVTOPTAhashishin1
			trigger = { NOT = { society_member_of = the_assassins } }
		}
		name = {
			text = EVTOPTAhashishin_member
			trigger = { society_member_of = the_assassins }
		}
	}
}

# hashishin.11 A courtier is killed as a message
# A warned ruler whose decadence is 70 or more loses a courtier.
# (Vanilla's text named FROM, which an MTTH event does not have.)
# Scope: ROOT is the ruler.
character_event = {
	id = hashishin.11
	desc = EVTDESChashishin11
	picture = GFX_evt_shadow
	border = GFX_event_normal_frame_intrigue

	ai = no
	only_playable = yes
	only_independent = yes
	min_age = 16
	prisoner = no
	only_capable = yes
	religion_group = ormazd_gods_group
	has_character_flag = decadence_warned
	has_global_flag = assassins_founded

	trigger = {
		uses_decadence = yes
		decadence = 70
		NOT = { has_character_flag = decadence_courtier_killed }
		hashishin_active_trigger = yes
		NOT = { society_member_of = the_assassins }
		any_courtier = { hashishin_victim_trigger = yes }
	}

	mean_time_to_happen = {
		months = 22
	}

	immediate = {
		d_hashishin = { holder_scope = { save_event_target_as = hashishin_master } }
		random_courtier = {
			limit = { hashishin_victim_trigger = yes }
			save_event_target_as = hashishin_victim
		}
	}

	option = {
		name = EVTOPTAhashishin11
		set_character_flag = decadence_courtier_killed
		event_target:hashishin_victim = {
			if = {
				limit = { is_invincible_trigger = yes }
				antideath_effect = yes
			}
			else = {
				death = {
					death_reason = death_hashshashin
					killer = event_target:hashishin_master
				}
			}
		}
	}
}

# hashishin.20 The Hashishin strike a court at war with them
# An independent southern ruler at war with the Grand Master loses a
# courtier (hashishin.21 - hashishin.23; the garden only when the ruler is
# at home). Was MTTH 880 months on every AI courtier of such a court.
# Scope: ROOT is the ruler.
character_event = {
	id = hashishin.20
	hide_window = yes

	is_triggered_only = yes # on_yearly_pulse, list hashishin_war

	only_playable = yes
	only_independent = yes
	prisoner = no
	war = yes
	religion_group = ormazd_gods_group

	trigger = {
		hashishin_active_trigger = yes
		d_hashishin = { holder_scope = { war_with = ROOT } }
		any_courtier = { hashishin_victim_trigger = yes }
	}

	immediate = {
		d_hashishin = { holder_scope = { save_event_target_as = hashishin_master } }
		random_courtier = {
			limit = { hashishin_victim_trigger = yes }
			save_event_target_as = hashishin_victim
		}
		if = {
			limit = { is_inaccessible_trigger = no }
			random_list = {
				33 = { character_event = { id = hashishin.21 } }
				33 = { character_event = { id = hashishin.22 } }
				33 = { character_event = { id = hashishin.23 } }
			}
		}
		else = {
			random_list = {
				50 = { character_event = { id = hashishin.21 } }
				50 = { character_event = { id = hashishin.22 } }
			}
		}
	}
}

# hashishin.30 Someone paid the Hashishin
# A courtier of a southern ruler who is not one of the order is killed
# (hashishin.21 - hashishin.23), the cruel, the possessed and impalers
# first. Was MTTH 5500 months on every AI courtier of those courts (and of
# Bretonnia's emperor).
# Scope: ROOT is the ruler.
character_event = {
	id = hashishin.30
	hide_window = yes

	is_triggered_only = yes # on_five_year_pulse, list hashishin_contract

	only_rulers = yes
	prisoner = no
	religion_group = ormazd_gods_group

	trigger = {
		hashishin_active_trigger = yes
		NOT = { has_landed_title = d_hashishin }
		NOT = { society_member_of = the_assassins }
		NOT = { d_hashishin = { holder_scope = { war_with = ROOT } } }
		any_courtier = { hashishin_victim_trigger = yes }
	}

	immediate = {
		d_hashishin = { holder_scope = { save_event_target_as = hashishin_master } }
		random_courtier = {
			limit = { hashishin_victim_trigger = yes }
			preferred_limit = {
				OR = {
					trait = cruel
					trait = possessed
					trait = impaler
				}
			}
			save_event_target_as = hashishin_victim
		}
		if = {
			limit = { is_inaccessible_trigger = no }
			random_list = {
				33 = { character_event = { id = hashishin.21 } }
				33 = { character_event = { id = hashishin.22 } }
				33 = { character_event = { id = hashishin.23 } }
			}
		}
		else = {
			random_list = {
				50 = { character_event = { id = hashishin.21 } }
				50 = { character_event = { id = hashishin.22 } }
			}
		}
	}
}

# per life for each ruler. (Salvaged from the Mongol strikes 88005 - 88009.)
# Scope: ROOT is the army's leader, FROM is the taken holding.
character_event = {
	id = hashishin.40
	hide_window = yes

	is_triggered_only = yes # on_siege_won_leader

	trigger = {
		hashishin_active_trigger = yes
		FROM = {
			OR = {
				location = {
					OR = {
						province_id = 1008 # Jebel Ezna
						province_id = 1003 # Marib
					}
				}
				holder_scope = { has_landed_title = d_hashishin }
			}
		}
		top_liege = {
			NOT = { religion_group = ormazd_gods_group }
			NOT = { has_character_flag = hashishin_hunted }
			NOT = { has_landed_title = d_hashishin }
		}
	}

	immediate = {
		d_hashishin = { holder_scope = { save_event_target_as = hashishin_master } }
		top_liege = {
			save_event_target_as = hashishin_target
			set_character_flag = hashishin_hunted
		}
		if = {
			limit = { event_target:hashishin_target = { ai = yes } }
			random_list = {
				40 = {
					event_target:hashishin_target = {
						if = {
							limit = { is_invincible_trigger = yes }
							antideath_effect = yes
						}
						else = {
							death = {
								death_reason = death_hashshashin
								killer = event_target:hashishin_master
							}
						}
					}
					event_target:hashishin_master = { narrative_event = { id = hashishin.41 days = 2 } }
				}
				60 = {
					event_target:hashishin_master = { narrative_event = { id = hashishin.42 days = 2 } }
				}
			}
		}
		else = {
			event_target:hashishin_master = { narrative_event = { id = hashishin.42 days = 2 } }
		}
	}
}
