###############################################################################
# Health events
#
# Namespace: health
# Faction: Generic
#
# The everyday wear on body and mind: chronic conditions, illness, wounds, old
# age and recovery, the voices a possessed character hears, the antics of mad
# rulers, the drowned dreams of the north, and Mittens the spymaster.
#
#   1-12     Chronic conditions: syphilis, leprosy, lunacy, possession, stress,
#            depression, and their cures
#   13-19    Illness: falling ill, recovering, sickly children, pneumonia,
#            lover's pox
#   20-21    Wounds: healing, and infection
#   25-26    Old age: infirmity and incapacity
#   30-35    Recovery from the old plagues (only without the Reapers DLC)
#   100-117  The voice from on high: a possessed character of a sane faith is
#            urged to virtue by a voice they take for their god. It is Tzeentch.
#   118-122  The twin-tailed omen that follows the voice from on high
#   123-141  The daemon's voice: a possessed Chaos worshipper or cultist is
#            urged to sin
#   150-158  Lunatic laws a mad ruler passes, and how the realm takes them
#   160-163  Random madness: a gopher maid, howling at Morrslieb, a horse
#            chancellor
#   170-178  Drowned dreams: a raving northerner follows dreams to the Sea of
#            Claws and meets the thing beneath
#   180-185  Mittens the spymaster
#   190-198  The singing sands: a raving southerner of Araby, Sudenburg,
#            Tilea, Estalia or Sartosa follows the desert's song to a city of
#            black glass
#   210-218  The dreaming temple: a raving Old World colonist in Lustria
#            follows the stars to a slann's dream
#
# Fired from:
#   on_yearly_pulse     health list (00_on_actions.txt): 1-14, 21, 25, 26;
#                       health_wounds list (wh_on_actions.txt): 20
#   on_bi_yearly_pulse  health_voices list (wh_on_actions.txt): 100-140
#   on_five_year_pulse  health_madness and health_drowned_dreams lists
#                       (wh_on_actions.txt): 150, 160-162, 170-173, 190-193,
#                       210-213
#   yearly pets list    180 (00_on_actions.txt)
#   MTTH                15-18, 30-35 (short recoveries, is_sick pre-trigger)
#   Elsewhere           19 (seduction and harem effects), 15 (assassins),
#                       20 (wound treatment), 25 (DR_misc), 14 (pregnancy,
#                       ancestor worship), 8 (S_chaos), 163 (lunatic events)
#
# Long-lived races roll less often on the odds of a mortal lifetime: see
# common/scripted_triggers/wh_lifespan_triggers.txt.
#
# Also in this file (each part keeps its own header and namespace):
#   Lunacy events (namespace lunacy, was WHG_lunacy_events)
###############################################################################

# health.1 Syphilis
# An unchaste character, or one whose spouse is infected, catches syphilis.
# Only without the Reapers DLC, which has its own disease system.
# Scope: ROOT is the character.
character_event = {
	id = health.1
	desc = EVTDESChealth1
	picture = GFX_evt_illness
	border = GFX_event_normal_frame_religion
	is_triggered_only = yes # on_yearly_pulse, health list

	min_age = 16
	capable_only = yes
	lacks_dlc = "Reapers"

	trigger = {
		health_mortal_body_trigger = yes
		NOR = { trait = syphilitic trait = celibate trait = infirm }
		OR = {
			NOT = { trait = chaste }
			spouse = { is_alive = yes is_abroad = no trait = syphilitic }
		}
	}

	weight_multiplier = {
		days = 1
		modifier = { factor = 0.9 kislev_northern_blood_trigger = yes } # Northern blood (WHG_bulwark)
		modifier = { factor = 10 spouse = { trait = syphilitic } }
		modifier = { factor = 2 in_command = yes }
		modifier = { factor = 2 trait = lustful }
		modifier = { factor = 1.7 trait = hedonist }
		modifier = { factor = 0.2 trait = chaste }
		modifier = { factor = 0.2 lifespan_dwarf_trigger = yes }
		modifier = { factor = 0.08 lifespan_elf_trigger = yes }
		modifier = { factor = 0.25 lifespan_other_long_lived_trigger = yes }
		modifier = { factor = 0.05 lifespan_hardy_race_trigger = yes }
	}

	option = { # Curse it
		name = EVTOPTAhealth1
		add_trait = syphilitic
		blessing_removes_disease_effect = yes
		deaths_gift_removes_disease_effect = yes
		nurgle_blessing_check_effect = yes
	}
}

# health.3 Leprosy
# A character in poor health catches leprosy. With Reapers, the court
# physician is told and the leper's treatment chain starts.
# Scope: ROOT is the character.
character_event = {
	id = health.3
	desc = EVTDESChealth3
	picture = GFX_evt_illness
	border = GFX_event_normal_frame_religion
	is_triggered_only = yes # on_yearly_pulse, health list

	min_age = 10
	capable_only = yes

	trigger = {
		health_mortal_body_trigger = yes
		NOT = { trait = leper }
		NOT = { health = 5 }
	}

	weight_multiplier = {
		days = 1
		modifier = { factor = 0.9 kislev_northern_blood_trigger = yes } # Northern blood (WHG_bulwark)
		modifier = { factor = 2 NOT = { health = 4 } }
		modifier = { factor = 2 in_command = yes }
		modifier = { factor = 2 any_courtier = { trait = leper } }
		modifier = { factor = 2 is_weak_trigger = yes }
		modifier = {
			factor = 2
			OR = { trait = infection trait = severely_injured trait = wounded }
		}
		modifier = { factor = 2 trait = charitable }
		modifier = { factor = 0.2 lifespan_dwarf_trigger = yes }
		modifier = { factor = 0.08 lifespan_elf_trigger = yes }
		modifier = { factor = 0.25 lifespan_other_long_lived_trigger = yes }
		modifier = { factor = 0.05 lifespan_hardy_race_trigger = yes }
	}

	option = { # Unclean
		name = EVTOPTAhealth3
		add_trait = leper
		nurgle_blessing_check_effect = yes
		if = {
			limit = { has_dlc = "Reapers" }
			set_character_flag = new_leper_character
			if = {
				limit = { is_playable = no }
				host = {
					random_courtier_or_vassal = {
						limit = {
							liege = { character = PREVPREV }
							has_minor_title = title_court_physician
						}
						character_event = { id = RIP.11031 days = 1 }
					}
				}
			}
			else = {
				random_courtier_or_vassal = {
					limit = { has_minor_title = title_court_physician }
					character_event = { id = RIP.11031 days = 1 }
				}
			}
			character_event = { id = RIP.11063 days = 20 }
		}
		hidden_tooltip = {
			random_list = {
				100 = { character_event = { id = RIP.5041 days = 365 random = 50 } }
				75 = { character_event = { id = RIP.5041 days = 1500 random = 300 } }
				30 = { character_event = { id = RIP.5041 days = 3000 random = 600 } }
			}
		}
	}
}

# health.5 Lunacy
# A character slips into the first stage of lunacy (lunatic_1, counter 5).
# Fever, prison, paranoia, zeal and a coma that ends in madness each have
# their own text. The lunacy events in WHG_health_events.txt carry on from here.
# Scope: ROOT is the character.
character_event = {
	id = health.5
	desc = {
		text = EVTDESChealth5
		trigger = {
			is_ill = no
			prisoner = no
			NOR = { trait = paranoid trait = zealous trait = incapable }
		}
	}
	desc = { text = EVTDESChealth5b trigger = { trait = paranoid NOT = { trait = incapable } } }
	desc = { text = EVTDESChealth5c trigger = { is_ill = yes NOT = { trait = incapable } } }
	desc = { text = EVTDESChealth5d trigger = { trait = zealous NOT = { trait = incapable } } }
	desc = { text = EVTDESChealth5e trigger = { prisoner = yes NOT = { trait = incapable } } }
	desc = { text = EVTDESChealth5f trigger = { trait = incapable } }
	picture = GFX_evt_lunatic
	border = GFX_event_normal_frame_religion
	is_triggered_only = yes # on_yearly_pulse, health list

	min_age = 10

	trigger = {
		is_lunatic_trigger = no
		lunacy_can_go_mad_trigger = yes
		OR = {
			# The senile and comatose old do not go mad (old by their race's measure); immortals get no pass
			lifespan_age_60_trigger = no
			immortal = yes
			NOT = { trait = incapable }
		}
	}

	weight_multiplier = {
		days = 1
		modifier = { factor = 0.9 kislev_northern_blood_trigger = yes } # Northern blood (WHG_bulwark)
		# Long-lived races: per-year odds scaled to their lifespan, and a little hardier
		modifier = { factor = 0.2 lifespan_dwarf_trigger = yes }
		modifier = { factor = 0.08 lifespan_elf_trigger = yes }
		modifier = { factor = 0.25 lifespan_other_long_lived_trigger = yes }
		modifier = { factor = 0.05 lifespan_hardy_race_trigger = yes }
		modifier = { # 12 years on average
			factor = 333
			OR = { trait = syphilitic trait = has_aztec_disease }
		}
		modifier = { # 24 years on average
			factor = 167
			prisoner = yes
			has_character_modifier = the_oubliette
		}
		modifier = { # 80 years on average
			factor = 50
			OR = { trait = has_typhoid_fever trait = has_typhus }
		}
		modifier = {
			factor = 10
			prisoner = yes
			NOT = { has_character_modifier = the_oubliette }
			NOT = { has_character_modifier = house_arrest }
		}
		modifier = { # 800 years on average
			factor = 5
			is_ill = yes
			NOR = {
				trait = has_typhoid_fever
				trait = has_typhus
				trait = syphilitic
				trait = has_aztec_disease
			}
		}
		modifier = { factor = 40 trait = rabies } # 1 year on average; rabies also counts as ill
		modifier = { factor = 40 trait = incapable } # A coma from a blow to the head ends in madness
		modifier = { factor = 40 trait = incapable immortal = yes } # Immortals wake from a coma madder still
		modifier = { factor = 4 trait = paranoid }
		modifier = { factor = 4 trait = stressed }
		modifier = { factor = 2 trait = depressed }
		modifier = { factor = 2 trait = zealous }
		modifier = { factor = 2 trait = inbred }
		modifier = { factor = 2 trait = imbecile }
		modifier = { factor = 2 trait = genius }
		modifier = { factor = 1.333 trait = drunkard }
	}

	option = {
		name = {
			text = EVTOPTAhealth5
			trigger = {
				is_ill = no
				prisoner = no
				NOR = { trait = paranoid trait = zealous trait = incapable }
			}
		}
		name = { text = EVTOPTAhealth5b trigger = { trait = paranoid NOT = { trait = incapable } } }
		name = { text = EVTOPTAhealth5c trigger = { is_ill = yes NOT = { trait = incapable } } }
		name = { text = EVTOPTAhealth5d trigger = { trait = zealous NOT = { trait = incapable } } }
		name = { text = EVTOPTAhealth5e trigger = { prisoner = yes NOT = { trait = incapable } } }
		name = { text = EVTOPTAhealth5f trigger = { trait = incapable } }
		add_trait = lunatic_1
		set_variable = { which = "lunatic_counter" value = 5 }
		if = { limit = { trait = depressed } remove_trait = depressed }
		if = { limit = { trait = stressed } remove_trait = stressed }
		if = { limit = { trait = incapable } remove_trait = incapable }
	}
}

# Namespace: lunacy (was "slunacy" before the 2026 events refactor)
# Faction: Generic
#
# A lunatic (lunatic_1 twitchy, lunatic_2 maniac, lunatic_3 raving) or the
# wearer of a madness artifact slowly loses their mind. Maniacs and raving
# lunatics catch one of nine madnesses (character modifiers), each with its own
# follow-up events. Old vampires go mad with age and catch vampire madnesses.
# Each madness event feeds the lunatic_counter variable through mutate.5.
#
# Umbrella and twitchy events   lunacy.1 - lunacy.9
# Madness onsets                lunacy.10 - lunacy.15
# Bloodthirsty                  lunacy.16 - lunacy.21
# Blind Virtue                  lunacy.22 - lunacy.24
# Fake Mage                     lunacy.25 - lunacy.26
# Despairing Laughter           lunacy.27 - lunacy.29
# Unfeeling                     lunacy.30 - lunacy.32, lunacy.35 - lunacy.42
# God Delusion                  lunacy.33 - lunacy.34
# Vampire ageing and umbrella   lunacy.50 - lunacy.51
# Vampire madness onsets        lunacy.52 - lunacy.54
# Everchild Syndrome            lunacy.60 - lunacy.67
# Exquisite Agonies             lunacy.70 - lunacy.82
# Heart of Melancholy           lunacy.85 - lunacy.88
# Recovery                      lunacy.90
#
# Fired from common/on_actions/wh_on_actions.txt: the bi-yearly list lunacy
# (lunacy.1) and lunacy.51 every two years, the five-year lists
# lunacy_<madness> and the decade lists lunacy_vampire_aging, lunacy_everchild
# and lunacy_hibernation.
# Scripted triggers: common/scripted_triggers/wh_lunacy_triggers.txt
# Scripted effects: common/scripted_effects/wh_lunacy_effects.txt
# Artifacts: common/artifacts/22_artifacts_lunacy.txt
# Loc: localisation/00_WHG_health_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = lunacy

# lunacy.1 Lunacy umbrella
# Every two years a mortal lunatic or madness artifact wearer may suffer a
# lunacy event: a twitchy quirk, or for maniacs and raving lunatics without a
# madness the onset of one. Doom, Morrslieb, mutations, Chaos cults and cursed
# lands make it more likely; a madness artifact alone makes it less likely.
# Scope: ROOT is the lunatic.
character_event = {
	id = lunacy.1
	hide_window = yes
	is_triggered_only = yes # on_bi_yearly_pulse, random list lunacy

	only_playable = yes

	trigger = {
		lunacy_can_go_mad_trigger = yes
		is_vampire_trigger = no
		OR = {
			trait = lunatic_1
			trait = lunatic_2
			trait = lunatic_3
			lunacy_madness_artifact_trigger = yes
		}
	}

	weight_multiplier = {
		factor = 1
		# Morrslieb and doom
		modifier = { factor = 1.25 has_global_flag = morrslieb_full }
		modifier = { factor = 0.5 is_doom_very_low_trigger = yes }
		modifier = { factor = 1.2 is_doom_low_trigger = yes }
		modifier = { factor = 1.5 is_doom_medium_trigger = yes }
		modifier = { factor = 1.75 is_doom_high_trigger = yes }
		modifier = { factor = 2 is_doom_very_high_trigger = yes }
		# A madness artifact alone is not true lunacy, and a healthy body resists it
		modifier = { factor = 0.6 NOR = { trait = lunatic_1 trait = lunatic_2 trait = lunatic_3 } }
		modifier = { factor = 0.9 lunacy_madness_artifact_trigger = yes health = 6 }
		modifier = { factor = 0.9 lunacy_madness_artifact_trigger = yes health = 8 }
		modifier = { factor = 0.9 lunacy_madness_artifact_trigger = yes health = 10 }
		# Race and body
		modifier = { factor = 0.9 kislev_northern_blood_trigger = yes } # Northern blood (WHG_bulwark)
		modifier = { factor = 0.75 trait = creature_dwarf }
		modifier = {
			factor = 0.75
			OR = {
				trait = creature_orc
				trait = creature_snotling
				trait = creature_goblin
				trait = creature_hobgoblin
			}
		}
		modifier = { factor = 2 trait = creature_fimir }
		modifier = { factor = 2 trait = creature_chaos_dwarf }
		modifier = { factor = 0.5 trait = creature_halfling }
		modifier = { factor = 1.5 has_minor_mutation_trigger = yes }
		modifier = { factor = 2 has_major_mutation_trigger = yes }
		modifier = { factor = 3 has_massive_mutation_trigger = yes }
		modifier = { factor = 1.25 trait = addict }
		modifier = { factor = 1.5 trait = lore_tzeentch }
		# The unschooled mind has no defence against the Winds of Magic
		modifier = {
			factor = 1.25
			NOR = {
				trait = magic_potential_1
				trait = magic_potential_2
				trait = magic_potential_3
				trait = magic_potential_4
			}
		}
		modifier = { factor = 1.25 has_magic_power_trigger = no }
		modifier = { factor = 2 has_character_flag = forsaken_flag }
		# Service of the Dark Gods
		modifier = { factor = 1.5 religion_group = chaos_gods_group }
		modifier = { factor = 1.25 society_member_of = cult_of_tzeentch }
		modifier = { factor = 1.5 society_member_of = cult_of_tzeentch2 }
		modifier = { factor = 1.75 society_member_of = cult_of_tzeentch3 }
		modifier = {
			factor = 1.25
			OR = {
				society_member_of = cult_of_khorne
				society_member_of = cult_of_slaanesh
				society_member_of = cult_of_nurgle
				society_member_of = cult_of_undivided
			}
		}
		modifier = {
			factor = 1.25
			OR = {
				society_member_of = cult_of_khorne2
				society_member_of = cult_of_slaanesh2
				society_member_of = cult_of_nurgle2
				society_member_of = cult_of_undivided2
			}
		}
		modifier = {
			factor = 1.25
			OR = {
				society_member_of = cult_of_khorne3
				society_member_of = cult_of_slaanesh3
				society_member_of = cult_of_nurgle3
				society_member_of = cult_of_undivided3
			}
		}
		# Cursed lands
		modifier = { factor = 1.5 location = { empire = { title = e_chaos_wastes } } }
		modifier = { factor = 1.25 location = { kingdom = { title = k_sylvania } } }
		modifier = { factor = 1.25 location = { kingdom = { title = k_mousillon } } }
		modifier = { factor = 1.25 location = { duchy = { title = d_geisterforst } } }
		modifier = { factor = 1.5 location = { has_province_modifier = warpdust_traces } }
		modifier = { factor = 0.5 ai = yes }
	}

	immediate = {
		random_list = {
			20 = { } # Nothing this time
			10 = { # Twitchy: whispers and shadows
				modifier = { factor = 0 lunacy_twitchy_trigger = no }
				modifier = { factor = 0 trait = paranoid }
				character_event = { id = lunacy.2 }
			}
			10 = { # Twitchy: unmoved by suffering
				modifier = { factor = 0 lunacy_twitchy_trigger = no }
				modifier = { factor = 0 trait = cruel }
				character_event = { id = lunacy.3 }
			}
			10 = { # Twitchy: too stressed
				modifier = { factor = 0 lunacy_twitchy_trigger = no }
				modifier = { factor = 0 NOT = { trait = stressed } }
				character_event = { id = lunacy.4 }
			}
			5 = { # Twitchy mage: dangerous experiment
				modifier = { factor = 0 lunacy_twitchy_trigger = no }
				modifier = { factor = 0 has_magic_power_trigger = no }
				character_event = { id = lunacy.5 }
			}
			5 = { # Twitchy: the drunkard feels poisoned
				modifier = { factor = 0 lunacy_twitchy_trigger = no }
				modifier = { factor = 0 NOT = { trait = drunkard } }
				character_event = { id = lunacy.9 }
			}
			10 = { # Maniac: Blind Virtue
				modifier = { factor = 0 NOT = { trait = lunatic_2 } }
				modifier = { factor = 0 lunacy_has_madness_trigger = yes }
				character_event = { id = lunacy.10 }
			}
			10 = { # Maniac: Fake Mage
				modifier = { factor = 0 NOT = { trait = lunatic_2 } }
				modifier = { factor = 0 lunacy_has_madness_trigger = yes }
				modifier = { factor = 0 has_magic_power_trigger = yes }
				character_event = { id = lunacy.11 }
			}
			10 = { # Maniac: Despairing Laughter
				modifier = { factor = 0 NOT = { trait = lunatic_2 } }
				modifier = { factor = 0 lunacy_has_madness_trigger = yes }
				character_event = { id = lunacy.12 }
			}
			10 = { # Maniac: Unfeeling
				modifier = { factor = 0 NOT = { trait = lunatic_2 } }
				modifier = { factor = 0 lunacy_has_madness_trigger = yes }
				character_event = { id = lunacy.13 }
			}
			10 = { # Raving: Bloodthirsty
				modifier = { factor = 0 NOT = { trait = lunatic_3 } }
				modifier = { factor = 0 lunacy_has_madness_trigger = yes }
				character_event = { id = lunacy.14 }
			}
			10 = { # Raving: God Delusion
				modifier = { factor = 0 NOT = { trait = lunatic_3 } }
				modifier = { factor = 0 lunacy_has_madness_trigger = yes }
				modifier = { factor = 0 is_ruler = no }
				character_event = { id = lunacy.15 }
			}
		}
	}
}

# lunacy.17 The victim dies
# The murder victim dies unless they cannot die. Their close kin, spouse and
# liege learn who did it (lunacy.18) and hate the murderer.
# Scope: ROOT is the victim, FROM the murderer.
character_event = {
	id = lunacy.17
	hide_window = yes
	is_triggered_only = yes # lunacy.16

	immediate = {
		if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
		else = {
			any_dynasty_member = {
				limit = { is_close_relative = ROOT NOT = { character = FROM } }
				opinion = { modifier = opinion_murderous_lunatic who = FROM years = 100 }
				character_event = { id = lunacy.18 }
			}
			any_spouse = {
				limit = { NOT = { character = FROM } }
				opinion = { modifier = opinion_murderous_lunatic who = FROM years = 100 }
				character_event = { id = lunacy.18 }
			}
			liege = {
				if = {
					limit = { NOT = { character = FROM } }
					opinion = { modifier = opinion_murderous_lunatic who = FROM years = 10 }
					character_event = { id = lunacy.18 }
				}
			}
			death = { death_reason = death_murder killer = FROM }
		}
	}
}

# A vampire with Heart of Melancholy withdraws into hibernation: gains the
# resting trait, and a ruler's capital takes hidden_hibernated_strigoi
# (vamp.60 clears it once they wake). Every few years they may wake
# (lunacy.88). About once in 30 years.

# Scope: ROOT is the vampire.

	id = lunacy.87
	desc = EVTDESClunacy87
	picture = GFX_evt_feral_vampire
	is_triggered_only = yes # on_decade_pulse, random list lunacy_hibernation

	min_age = 75
	capable_only = yes

	trigger = {
		has_character_modifier = heart_of_melancholy
		lunacy_can_act_trigger = yes
	}

	option = {
		name = EVTOPTAlunacy87
		hidden_tooltip = { character_event = { id = mutate.5 } } # Lunacy counter
		add_trait = resting
		if = {
			limit = { is_ruler = yes }
			capital_scope = {
				add_province_modifier = { name = hidden_hibernated_strigoi duration = -1 }
			}
		}
		hidden_tooltip = { character_event = { id = lunacy.88 days = 100 random = 900 } } # Hibernation clock
	}
}

# lunacy.90 The mind mends
# A sane character slowly sheds the strain that builds toward lunacy: the
# lunatic counter drops by 1. Without this the counter only ever grew, so a
# dwarf or elf living for centuries was all but sure to go mad. Long-lived and
# hardy races (lizardmen, greenskins) mend faster.
# Scope: ROOT is the character.
character_event = {
	id = lunacy.90
	hide_window = yes
	is_triggered_only = yes # on_five_year_pulse, random list lunacy_recovery

	trigger = {
		is_lunatic_trigger = no
		check_variable = { which = "lunatic_counter" value = 1 }
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 2 lifespan_dwarf_trigger = yes }
		modifier = { factor = 2 lifespan_elf_trigger = yes }
		modifier = { factor = 2 lifespan_other_long_lived_trigger = yes }
		modifier = { factor = 3 lifespan_hardy_race_trigger = yes }
	}

	immediate = {
		change_variable = { which = "lunatic_counter" value = -1 }
	}
}
