###############################################################################
# Fanatics and heresies
# Namespace: heresy (was SoA.4000 - SoA.4200 in vanilla soa_heresy_events.txt)
# Faction: Generic
# Serves: the Old World and Arabyan faiths, and the heads of faiths whose
#   heresies take hold
#
# Where a faith of the Old World or of Araby loses its grip, fanatics rise in
# a county: unrest for five years, and one county in the realm becomes the
# stronghold of the movement. When the fanatics follow a heresy and the top
# liege keeps the parent faith, the head of that faith may act: send a Holy
# Inquest of priest-investigators and armed guards (it may turn the county
# back to the true faith, or rile up the peasants), send an envoy who makes
# the top liege promise to clear them out within ten years (rewarded with
# gold, piety or a lifted excommunication if it is done), or send learned
# priests to argue the fanatics back into the fold. When a heresy grows so
# strong that it becomes the new orthodoxy, the world hears of it.
#
# Fanatics rise                        heresy.1 - heresy.2
# The head of the faith acts           heresy.10 - heresy.11
# The Holy Inquest                     heresy.20 - heresy.24
# The envoy and the promise            heresy.30 - heresy.35
# The dialogue                         heresy.40 - heresy.44
# The heresy triumphs                  heresy.50
#
# Fired from: on_five_year_pulse in wh_on_actions.txt (heresy.1, which
#   replaces two province MTTH events of 800 and 100 years),
#   on_county_religion_change (heresy.34) and on_heresy_takeover (heresy.50)
#   in 00_on_actions.txt. Triggers in
#   common/scripted_triggers/WHG_heresy_triggers.txt. The realm news is
#   event 39400 in religious_events.txt.
# Saved: province modifiers fanatic_uprising, heretic_stronghold,
#   active_inquisition, theological_dialogue and promised_to_clear_out_heresy;
#   character modifier promised_to_clear_out_heresy.
###############################################################################

namespace = heresy

###############################################################################
# Fanatics rise
###############################################################################

# heresy.1 Fanatics stir in the realm
# Every five years each independent ruler's counties may see fanatics rise
# (about 1 in 160 per county, more where the faith is weak or fanatics hold
# a neighbouring county; was MTTH 800 years), and the head of a faith may
# hear of heretic fanatics in a county (1 in 20, 1 in 60 for a player head;
# was MTTH 100 years).
# Scope: ROOT is an independent ruler. From on_five_year_pulse.
character_event = {
	id = heresy.1
	hide_window = yes

	is_triggered_only = yes
	only_independent = yes

	immediate = {
		any_realm_province = {
			limit = { heresy_fanatics_can_rise_trigger = yes }
			random = {
				chance = 1
				modifier = {
					factor = 0.625
				}
				modifier = {
					factor = 0.667
					religion_authority = 0.6
				}
				modifier = {
					factor = 0.667
					religion_authority = 0.7
				}
				modifier = {
					factor = 0.667
					religion_authority = 0.8
				}
				modifier = {
					factor = 2
					NOT = { religion_authority = 0.2 }
				}
				modifier = {
					factor = 1.333
					NOT = { religion_authority = 0.3 }
				}
				modifier = {
					factor = 1.333
					NOT = { religion_authority = 0.4 }
				}
				modifier = {
					factor = 1.333
					NOT = { religion_authority = 0.5 }
				}
				modifier = {
					factor = 2
					any_neighbor_province = {
						is_land = yes
						has_province_modifier = fanatic_uprising
					}
				}
				modifier = {
					factor = 4
					any_neighbor_province = {
						is_land = yes
						is_heresy_of = PREV
						has_province_modifier = heretic_stronghold
					}
				}
				province_event = { id = heresy.2 }
			}
		}
		any_realm_province = {
			limit = { heresy_head_worried_trigger = yes }
			random = {
				chance = 5
				modifier = {
					factor = 0.333 # Do not spam human players with this
					parent_religion_head = { ai = no }
				}
				province_event = { id = heresy.10 }
			}
		}
	}
}

# heresy.2 Fanatics rise
# Fanatics of the county's faith rise: five years of unrest, and if the realm
# has no stronghold of the faith yet, ten years of a fanatic stronghold. The
# owner loses piety and every liege hears of it (39400).
# Scope: ROOT is the county (province). From heresy.1.
province_event = {
	id = heresy.2
	desc = EVTDESCheresy2
	picture = GFX_evt_heretic
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	trigger = {
		heresy_fanatics_can_rise_trigger = yes
	}

	immediate = {
		add_province_modifier = {
			name = fanatic_uprising
			duration = 1825
		}
		# One fanatic county in the realm is the heart of the movement
		if = {
			limit = {
				owner = {
					top_liege = {
						NOT = {
							any_realm_province = {
								religion = ROOT
								has_province_modifier = heretic_stronghold
							}
						}
					}
				}
			}
			add_province_modifier = {
				name = heretic_stronghold
				duration = 3650
			}
		}
		owner = {
			if = {
				limit = { ai = no }
				chronicle = {
					entry = CHRONICLE_HERESY_SPREAD
					picture = GFX_evt_heretic
				}
			}
			any_liege = {
				character_event = { id = 39400 }
				if = {
					limit = { ai = no }
					chronicle = {
						entry = CHRONICLE_HERESY_SPREAD
						picture = GFX_evt_heretic
					}
				}
			}
		}
	}

	option = {
		name = EVTOPTAheresy2
		owner = { piety = -10 }
	}
}

# heresy.11 Heretic fanatics trouble the head of the faith
# The head may send a Holy Inquest (100 gold, heresy.20), an envoy to the
# county's top liege (heresy.30), learned priests for a dialogue (learning 7,
# 50 gold, heresy.40), or do nothing.
# Scope: ROOT is the head of the parent faith, FROM the county. From
#   heresy.10.
character_event = {
	id = heresy.11
	desc = EVTDESCheresy11
	picture = GFX_evt_heretic
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = { # Send a Holy Inquest
		name = EVTOPTAheresy11
		trigger = { wealth = 100 }
		ai_chance = {
			factor = 200
			modifier = {
				factor = 10
				trait = cruel
			}
			modifier = {
				factor = 3
				trait = zealous
			}
		}
		wealth = -100
		FROM = {
			province_event = { id = heresy.20 tooltip = EVTTOOLTIPheresy20 }
		}
	}

	option = { # Send an envoy to the top liege
		name = EVTOPTBheresy11
		trigger = {
			FROM = {
				owner = {
					top_liege = {
						NOT = { character = ROOT }
					}
				}
			}
		}
		ai_chance = { factor = 100 }
		FROM = {
			owner = {
				top_liege = {
					character_event = { id = heresy.30 days = 2 tooltip = EVTTOOLTIPheresy30 }
				}
			}
		}
	}

	option = { # Open a dialogue
		name = EVTOPTCheresy11
		tooltip_info = learning
		trigger = {
			learning = 7
			wealth = 50
		}
		ai_chance = { factor = 1000 }
		wealth = -50
		FROM = {
			province_event = { id = heresy.40 tooltip = EVTTOOLTIPheresy40 }
		}
	}

	option = { # Do nothing
		name = EVTOPTDheresy11
		ai_chance = { factor = 1 }
	}
}

# heresy.34 The promise is kept
# Hidden. When a promised county turns to the top liege's faith, the head of
# that faith hears of it (heresy.35).
# Scope: ROOT is the county (province). From on_county_religion_change.
province_event = {
	id = heresy.34
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_dlc = "Sons of Abraham"
		has_province_modifier = promised_to_clear_out_heresy
		owner = {
			top_liege = {
				religion = ROOT
				has_character_modifier = promised_to_clear_out_heresy
			}
		}
	}

	immediate = {
		remove_province_modifier = promised_to_clear_out_heresy
		religion_head = {
			character_event = { id = heresy.35 }
		}
	}
}

# heresy.35 The fanatics are gone
# The head of the faith rewards the top liege with 100 gold (heresy.31),
# 200 piety (heresy.32) or by lifting an excommunication (heresy.33). The
# top liege is grateful either way.
# Scope: ROOT is the head of the faith, FROM the county. From heresy.34.
character_event = {
	id = heresy.35
	desc = EVTDESCheresy35
	picture = GFX_evt_bishop
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	immediate = {
		FROM = {
			owner = {
				top_liege = {
					reverse_opinion = {
						who = ROOT
						modifier = opinion_grateful
						months = 120
					}
				}
			}
		}
	}

	option = { # Reward the ruler with gold
		name = EVTOPTAheresy35
		trigger = { wealth = 100 }
		ai_chance = { factor = 10 }
		wealth = -100
		FROM = {
			owner = {
				top_liege = {
					character_event = { id = heresy.31 tooltip = EVTTOOLTIPheresy31 }
				}
			}
		}
	}

	option = { # Proclaim the ruler's piety
		name = EVTOPTBheresy35
		ai_chance = { factor = 10 }
		FROM = {
			owner = {
				top_liege = {
					character_event = { id = heresy.32 tooltip = EVTTOOLTIPheresy32 }
				}
			}
		}
	}

	option = { # Lift the excommunication
		name = EVTOPTCheresy35
		trigger = {
			FROM = {
				owner = {
					top_liege = {
						excommunicated_for = ROOT
					}
				}
			}
		}
		ai_chance = { factor = 1000 }
		FROM = {
			owner = {
				top_liege = {
					character_event = { id = heresy.33 tooltip = EVTTOOLTIPheresy33 }
				}
			}
		}
	}
}

# heresy.50 A heresy becomes the new orthodoxy
# Everyone hears that a heresy has replaced its parent faith: its followers
# rejoice, the followers of the old faith despair, everyone else takes note.
# Scope: ROOT is any character; FROM and FROMFROM are temporary characters of
#   the new and the old faith. From on_heresy_takeover.
narrative_event = {
	id = heresy.50
	title = EVTNAMEheresy50
	desc = EVTDESCheresy50
	picture = GFX_evt_bishop
	border = GFX_event_narrative_frame_religion
	hide_from = yes

	major = yes

	is_triggered_only = yes

	trigger = {
		OR = {
			is_alternate_start = no
			total_years_played = 1
		}
	}

	option = {
		name = EVTOPTAheresy50
		trigger = { religion = FROM }
		if = {
			limit = { ai = no }
			chronicle = {
				entry = CHRONICLE_HERESY_TAKEOVER_PLAYER_IS_NEW
				portrait = [Root.GetID]
			}
		}
	}
	option = {
		name = EVTOPTBheresy50
		trigger = { religion = FROMFROM }
		if = {
			limit = { ai = no }
			chronicle = {
				entry = CHRONICLE_HERESY_TAKEOVER_PLAYER_IS_OLD
				portrait = [Root.GetID]
			}
		}
	}
	option = {
		name = EVTOPTCheresy50
		trigger = {
			NOT = { religion = FROM }
			NOT = { religion = FROMFROM }
		}
	}
}
