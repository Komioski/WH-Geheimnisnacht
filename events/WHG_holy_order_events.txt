###############################################################################
# Holy orders
# Namespace: holyorder (was HFP.44000 - HFP.44010 in HFP_holy_order_events)
# Faction: the Knights Mariner (Manann) and the Knights of the White Wolf
#   (Ulric), and any holy order serving a liege
#
# When a holy war wins an island, the Knights Mariner ask its new lord for
# it as a harbour against the corsairs; when it wins a northern duchy on
# the edge of the faith, the White Wolves ask for it as a march against
# Chaos. A generous lord is paid and thanked, and may receive a token of
# the order (rare). Any holy order that comes to despise its liege, or
# whose liege is cast out or of another faith, leaves its service.
#
# The Knights Mariner          holyorder.1 - holyorder.3
# The White Wolves             holyorder.5 - holyorder.7
# Leaving an unfit liege       holyorder.10 - holyorder.11
#
# Saved: character flags refused_to_give_island_to_knights,
#   refused_to_give_land_to_white_wolves; global flag
#   white_wolves_asked_for_land; event targets potential_title,
#   title_owner, holy_order_potential_title, liege_character,
#   kingdom_title, order_master.
#
# Callers:
#   common/scripted_effects/07_scripted_effects_holy_orders.txt
#     (check_holy_order_donation_effect: 1, 5),
#   common/on_actions/00_on_actions.txt: on_five_year_pulse (10).
# Support: common/scripted_triggers/wh_holy_order_triggers.txt,
#   common/artifacts/34_artifacts_holy_orders.txt
# Loc: localisation/00_WHG_holy_order_events.csv
###############################################################################

namespace = holyorder

# holyorder.1 The Knights Mariner want an island
# After a holy war wins an island, the Knights Mariner ask its new lord for
# it (check_holy_order_donation_effect).
# Scope: ROOT is the holder of d_knights_mariner; event targets
#   potential_title and title_owner.
character_event = {
	id = holyorder.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_alternate_start_parameter = { key = religion_names value = random } }
		wh_holds_mariner_island_trigger = no
	}

	immediate = {
		event_target:title_owner = { letter_event = { id = holyorder.2 } }
	}
}

# holyorder.2 A harbour for the Knights Mariner
# Scope: ROOT is the lord of the island; FROM is the Knights Mariner
#   (holyorder.1).
letter_event = {
	id = holyorder.2
	desc = EVTDESCholyorder2
	is_triggered_only = yes
	trigger = { }

	immediate = { }

	option = {
		name = EVTOPTAholyorder2
		piety = 100
		scaled_wealth = { value = 2 }
		if = { # A rare token of the order's thanks
			limit = { NOT = { has_artifact = mariner_sea_compass } }
			random = { chance = 20 add_artifact = mariner_sea_compass }
		}

		d_knights_mariner = {
			show_scope_change = no
			holder_scope = {
				show_scope_change = no
				opinion = { who = ROOT modifier = opinion_grateful years = 10 }
				opinion = {
					modifier = in_non_aggression_pact
					years = 10
					who = ROOT
					origin_description = EVTOPTAholyorder2
				}
			}
		}

		custom_tooltip = {
			text = EVTOPTAholyorder2_TT

			event_target:potential_title = {
				if = {
					limit = {
						owner = { OR = { character = ROOT any_liege = { character = ROOT } } }
						NOT = { tier = DUKE }
					}
					grant_title_no_opinion = FROM
				}
				else_if = {
					limit = {
						owner = {
							OR = { character = ROOT any_liege = { character = ROOT } }
							tier = DUKE
						}
					}
					destroy_landed_title = THIS
				}
				any_direct_de_jure_vassal_title = {
					limit = {
						OR = {
							owner = { character = ROOT }
							owner = { any_liege = { character = ROOT } }
						}
					}
					grant_title_no_opinion = FROM
					any_direct_de_jure_vassal_title = {
						limit = {
							OR = {
								owner = { character = ROOT }
								owner = { any_liege = { character = ROOT } }
							}
						}
						grant_title_no_opinion = FROM
					}
				}
			}
			FROM = {
				any_demesne_title = {
					limit = {
						NOR = {
							AND = { OR = { tier = BARON tier = COUNT } holding_type = CASTLE }
							title = d_knights_mariner
						}
					}
					if = {
						limit = { tier = BARON holding_type = CITY }
						create_random_steward = { random_traits = yes dynasty = random }
					}
					else_if = {
						limit = { tier = BARON holding_type = TEMPLE }
						create_random_priest = { random_traits = yes dynasty = random }
					}
					else = { create_random_soldier = { random_traits = yes dynasty = random } }
					grant_title_no_opinion = new_character
					new_character = {
						# Set Racial Trait
						character_event = { id = aiclean.31 }
					}
				}
			}

			event_target:potential_title = {
				capital_scope = {
					FROM = {
						if = {
							limit = {
								has_landed_title = PREV
								event_target:potential_title = { tier = DUKE }
							}
							capital = PREV
						}
						else = { capital = event_target:potential_title }
					}
				}
			}
			any_playable_ruler = {
				limit = { religion = FROM }
				narrative_event = { id = holyorder.3 }
			}
			ROOT = { narrative_event = { id = holyorder.3 } }
			FROM = {
				character_event = { id = HFP.41090 days = 1 }
				character_event = { id = HFP.41090 days = 3 }
			}
		}

		ai_chance = { factor = 1 }
	}

	option = {
		name = EVTOPTBholyorder2
		piety = -25

		set_character_flag = refused_to_give_island_to_knights

		ai_chance = { factor = 0 }
	}
}

# holyorder.10 The order leaves an unfit liege
# A holy order serving a liege it despises, or one cast out or of another
# faith, leaves that liege's service and hands back any land that is not
# the order's own.
# Scope: ROOT is the order's master (five-year pulse).
character_event = {
	id = holyorder.10
	hide_window = yes
	is_triggered_only = yes
	ai = yes
	only_rulers = yes

	trigger = {
		holy_order = yes
		independent = no
	}

	immediate = {
		liege = { save_event_target_as = liege_character }

		if = {
			limit = {
				OR = {
					NOT = { opinion = { who = liege value = -20 } }
					liege = { excommunicated_for = ROOT }
					liege = { NOT = { religion = ROOT } }
				}
			}
			opinion = {
				who = event_target:liege_character
				modifier = opinion_unfit_ruler
				years = 30
			}
			liege = { letter_event = { id = holyorder.11 } }
			any_demesne_title = {
				limit = {
					holy_order = no
					OR = {
						county = {
							owner = {
								OR = {
									character = event_target:liege_character
									event_target:liege_character = {
										any_vassal = {
											NOT = { character = ROOT }
											character = PREVPREV
										}
									}
								}
							}
						}
						duchy = {
							owner = {
								OR = {
									character = event_target:liege_character
									event_target:liege_character = {
										any_vassal = {
											NOT = { character = ROOT }
											character = PREVPREV
										}
									}
								}
							}
						}
						kingdom = {
							owner = {
								OR = {
									character = event_target:liege_character
									event_target:liege_character = {
										any_vassal = {
											NOT = { character = ROOT }
											character = PREVPREV
										}
									}
								}
							}
						}
						empire = {
							owner = {
								OR = {
									character = event_target:liege_character
									event_target:liege_character = {
										any_vassal = {
											not = { character = ROOT }
											character = PREVPREV
										}
									}
								}
							}
						}
					}
				}
				event_target:liege_character = { grant_title_no_opinion = PREV }
			}
			set_defacto_liege = ROOT
		}
	}
}

# holyorder.5 The White Wolves want land in the north
# After a holy war wins a northern duchy, the Knights of the White Wolf ask
# for it (check_holy_order_donation_effect). One request at a time.
# Scope: ROOT is the holder of d_knights_of_the_white_wolf; FROM is the
#   lord of the duchy.
character_event = {
	id = holyorder.5
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		NOT = { has_global_flag = white_wolves_asked_for_land }
	}

	immediate = {
		FROM = { letter_event = { id = holyorder.6 } }
		set_global_flag = white_wolves_asked_for_land
	}
}

# holyorder.6 Land for the White Wolves
# Scope: ROOT is the lord of the duchy; FROM is the White Wolves
#   (holyorder.5); event target holy_order_potential_title.
letter_event = {
	id = holyorder.6
	desc = EVTDESCholyorder6
	border = GFX_event_letter_frame_religion
	is_triggered_only = yes

	option = {
		name = EVTOPTAholyorder6

		piety = 100
		scaled_wealth = { value = 2 }
		if = { # A rare token of the order's thanks
			limit = { NOT = { has_artifact = white_wolf_pelt } }
			random = { chance = 20 add_artifact = white_wolf_pelt }
		}

		d_knights_of_the_white_wolf = {
			show_scope_change = no
			holder_scope = {
				show_scope_change = no
				opinion = { who = ROOT modifier = opinion_grateful years = 10 }
				opinion = {
					modifier = in_non_aggression_pact
					years = 10
					who = ROOT
					origin_description = EVTOPTAholyorder6
				}
			}
		}

		custom_tooltip = {
			text = EVTOPTAholyorder6_TT

			event_target:holy_order_potential_title = {
				# Destroy the Duchy title, if there is any
				if = {
					limit = {
						owner = { OR = { character = ROOT any_liege = { character = ROOT } } }
					}
					destroy_landed_title = THIS
				}
				# Give the White Wolves all the land you hold in the area
				any_direct_de_jure_vassal_title = {
					limit = {
						owner = { OR = { character = ROOT any_liege = { character = ROOT } } }
					}
					grant_title_no_opinion = FROM
					any_direct_de_jure_vassal_title = {
						limit = {
							owner = { OR = { character = ROOT any_liege = { character = ROOT } } }
						}
						grant_title_no_opinion = FROM
					}
				}
			}
			# Upgrade tribal areas into proper feudal areas
			FROM = { teutonic_order_upgrade_capital_effect = yes }
			# Give out any unfit land (see cities, temples) to new vassals
			FROM = { holy_order_distribute_titles_effect = yes }
			# Refilled holdings and remove province penalties
			FROM = {
				character_event = { id = HFP.41090 days = 1 }
				character_event = { id = HFP.41090 days = 3 }
			}
			# Making sure the White Wolves stay a proper Holy Order
			FROM = { set_government_type = order_government }
			# Sending out narrative event to all rulers of the same faith
			if = {
				limit = { FROM = { any_demesne_title = { tier = COUNT } } }
				any_playable_ruler = {
					limit = { religion = FROM }
					narrative_event = { id = holyorder.7 }
				}
				ROOT = { narrative_event = { id = holyorder.7 } }
			}
		}
		ai_chance = { factor = 1 }
	}
	option = {
		name = EVTOPTBholyorder6

		set_character_flag = refused_to_give_land_to_white_wolves
		ai_chance = { factor = 0 }
	}
	after = {
		clr_global_flag = white_wolves_asked_for_land
	}
}

# Scope: ROOT is any ruler of the faith; FROM is the donor (holyorder.6).

	id = holyorder.7

	}

}
