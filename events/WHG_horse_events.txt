###############################################################################
# The warhorse of a horse lord
# Namespace: horse (was HL.6000 - HL.6999 in hl_horse_events.txt and
#   hl_horse_events_2.txt)
# Faction: Generic
# Serves: the horse peoples of the steppe (Kurgan, Hung, Ungol) and every
#   other nomad ruler who rides
#
# A nomad ruler picks a warhorse from the herd and names it. It carries them
# for years until it dies of age, drowns at a river crossing or is given up
# to save a spouse. Among the way: a ward who cannot ride, a warrior whose
# mares died, a horse that bites, and a painter who wants to paint the
# horse. A follower of the Dark Gods may find a daemon-touched steed among
# the herd, a rider of the steppe a horse of rare breeding (once a life).
#
# The warhorse                         horse.1 - horse.4
# The ward who cannot ride             horse.10
# The river crossing                   horse.20 - horse.22
# The sick herd                        horse.30
# The bite                             horse.40 - horse.42
# The portrait                         horse.50
#
# Fired from: the choose_warhorse_nomad decision in minor_decisions.txt
#   (horse.1, horse.2) and the on_yearly_pulse random list in
#   00_on_actions.txt (horse.10, .20, .30, .40, .50). The horse's name is
#   [Root.GetWarhorseName] (WHG_horse_lords_voices.txt). Shared effects are
#   in WHG_horse_lords_effects.txt.
###############################################################################

namespace = horse

###############################################################################

###############################################################################

# horse.1 Choose a warhorse
# Name the horse. A rare mount may stand out from the herd once in a life
# (horse_rare_mount_effect).
# Scope: the rider.
long_character_event = {
	id = horse.1
	desc = EVTDESChorse1
	picture = GFX_evt_horse_friend
	border = GFX_event_long_frame_war

	is_triggered_only = yes # choose_warhorse_nomad

	immediate = {
		set_character_flag = horse_owner
		horse_rare_mount_effect = yes
	}

	option = { # Dragon
		name = EVTOPTAhorse1
		add_character_modifier = { name = nomad_warhorse duration = -1 }
		set_character_flag = nomad_warhorse_name_dragon
	}
	option = { # Sky
		name = EVTOPTBhorse1
		add_character_modifier = { name = nomad_warhorse duration = -1 }
		set_character_flag = nomad_warhorse_name_sky
	}
	option = { # Arrow
		name = EVTOPTChorse1
		add_character_modifier = { name = nomad_warhorse duration = -1 }
		set_character_flag = nomad_warhorse_name_arrow
	}
	option = { # Lightning
		name = EVTOPTDhorse1
		add_character_modifier = { name = nomad_warhorse duration = -1 }
		set_character_flag = nomad_warhorse_name_lightning
	}
}

# horse.3 Grief
# Some riders mourn their horse; a loving spouse, a friend or a kind
# courtier may share the grief.
# Scope: the rider.
character_event = {
	id = horse.3
	desc = EVTDESChorse3
	picture = GFX_evt_bad_news
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # horse.4

	trigger = {
		NOT = { has_character_modifier = nomad_warhorse }
	}

	option = { # Alone
		name = EVTOPTAhorse3
		random = {
			chance = 5
			add_trait = depressed
			hidden_tooltip = { character_event = { id = 38288 } } # Notify Depressed
		}
	}
	option = { # My spouse
		name = EVTOPTBhorse3
		trigger = {
			spouse = {
				opinion = { who = ROOT value = 60 }
				NOT = { trait = incapable }
			}
		}
		spouse = {
			random = {
				chance = 5
				add_trait = depressed
				hidden_tooltip = { character_event = { id = 38288 } } # Notify Depressed
			}
		}
	}
	option = { # A friend
		name = EVTOPTChorse3
		trigger = {
			OR = {
				any_friend = { NOT = { trait = incapable } }
				any_courtier = { horse_kind_mourner_trigger = yes }
			}
		}
		if = {
			limit = { any_friend = { NOT = { trait = incapable } } }
			random_friend = {
				limit = { NOT = { trait = incapable } }
				random = {
					chance = 5
					add_trait = depressed
					hidden_tooltip = { character_event = { id = 38288 } } # Notify Depressed
				}
			}
		}
		else = {
			random_courtier = {
				limit = { horse_kind_mourner_trigger = yes }
				random = { chance = 5 add_trait = depressed }
			}
		}
	}
}

# horse.4 Clean-up
# Hidden. The horse is gone: modifier, name and the pending death go, and a
# rider who did not give it up for a spouse may grieve (horse.3).
# Scope: the rider.
character_event = {
	id = horse.4
	hide_window = yes
	is_triggered_only = yes # horse.2, horse.21

	immediate = {
		remove_character_modifier = nomad_warhorse
		clr_character_flag = horse_owner
		clr_character_flag = nomad_warhorse_name_lightning
		clr_character_flag = nomad_warhorse_name_dragon
		clr_character_flag = nomad_warhorse_name_sky
		clr_character_flag = nomad_warhorse_name_arrow
		clear_delayed_event = { id = horse.2 }
		if = {
			limit = { NOT = { has_character_flag = nomad_warhorse_ignore_grief } }
			random = {
				chance = 10
				character_event = { id = horse.3 days = 150 }
			}
		}
		clr_character_flag = nomad_warhorse_ignore_grief
	}
}

# horse.10 Still on foot
# A nomad's seven-year-old ward still cannot ride. Teach them, or not.
# Scope: the guardian.
character_event = {
	id = horse.10
	desc = EVTDESChorse10
	picture = GFX_evt_child_sword_altaic
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # on_yearly_pulse

	only_playable = yes

	trigger = {
		is_nomadic = yes
		any_ward = {
			age = 7
			NOT = { age = 8 }
		}
	}

	immediate = {
		random_ward = {
			limit = {
				age = 7
				NOT = { age = 8 }
			}
			save_event_target_as = ward_cant_ride
		}
	}

	option = { # Teach them myself
		name = EVTOPTAhorse10
		event_target:ward_cant_ride = {
			opinion = { who = ROOT modifier = opinion_happy_with_guardian years = 5 }
		}
	}
	option = { # Let them flounder
		name = EVTOPTBhorse10
		event_target:ward_cant_ride = { prestige = -30 }
	}
}

# horse.20 The river crossing
# The spouse rides the warhorse into a flooded river and both are swept
# away: save the spouse (horse.21 or horse.22) or the horse.
# Scope: the rider; event_target:spouse_danger_nomad_river_crossing the
# spouse.
narrative_event = {
	id = horse.20
	title = EVTTITLEhorse20
	desc = EVTDESChorse20
	picture = GFX_evt_horse_friend
	border = GFX_event_narrative_frame_diplomacy

	is_triggered_only = yes # on_yearly_pulse

	has_character_flag = horse_owner
	capable_only = yes
	is_married = yes

	trigger = {
		is_nomadic = yes
		has_character_modifier = nomad_warhorse
		NOT = { has_character_flag = do_not_disturb }
		is_inaccessible_trigger = no
		any_spouse = { horse_river_spouse_trigger = yes }
	}

	immediate = {
		random_spouse = {
			limit = { horse_river_spouse_trigger = yes }
			save_event_target_as = spouse_danger_nomad_river_crossing
		}
	}

	option = { # Save the spouse
		name = EVTOPTAhorse20
		remove_character_modifier = nomad_warhorse
		hidden_tooltip = {
			random_list = {
				10 = { character_event = { id = horse.21 } } # Grateful
				10 = { character_event = { id = horse.22 } } # Angry
			}
		}
	}
	option = { # Save the horse
		name = EVTOPTBhorse20
		event_target:spouse_danger_nomad_river_crossing = {
			if = {
				limit = { is_invincible_trigger = yes }
				antideath_effect = yes
			}
			else = {
				death = { death_reason = death_accident_drowning }
			}
		}
	}
}

# horse.21 Pulled from the river
# The spouse is grateful (and may fall for the rescuer), or grateful but
# sorry for the horse.
# Scope: the rider; event_target:spouse_danger_nomad_river_crossing the
# spouse.
character_event = {
	id = horse.21
	desc = EVTDESChorse21
	picture = GFX_evt_nomad_camp
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # horse.20

	option = { # Glad they are safe
		name = EVTOPTAhorse21
		event_target:spouse_danger_nomad_river_crossing = {
			if = {
				limit = {
					is_lover = ROOT
				}
				opinion = { who = ROOT modifier = opinion_grateful years = 5 }
			}
			else_if = {
				limit = {
					is_no_females_race = no
					ROOT = { is_no_females_race = no }
				}
				add_lover = ROOT
			}
			else = {
				opinion = { who = ROOT modifier = opinion_grateful years = 5 }
			}
		}
		hidden_tooltip = {
			set_character_flag = nomad_warhorse_ignore_grief
			character_event = { id = horse.4 }
		}
	}
	option = { # But the horse...
		name = EVTOPTBhorse21
		event_target:spouse_danger_nomad_river_crossing = {
			opinion = { who = ROOT modifier = opinion_grateful years = 5 }
			reverse_opinion = { who = ROOT modifier = horse_regret years = 5 }
		}
		hidden_tooltip = {
			set_character_flag = nomad_warhorse_ignore_grief
			character_event = { id = horse.4 }
		}
	}
}

# horse.40 Who owns the biter?
# Hidden. Picks an adult vassal (or, failing that, a courtier) whose horse
# bites the warhorse at the watering hole (horse.41 bounces it so the owner
# is FROM).
# Scope: the rider.
character_event = {
	id = horse.40
	hide_window = yes
	is_triggered_only = yes # on_yearly_pulse

	only_playable = yes
	has_character_flag = horse_owner

	trigger = {
		has_character_modifier = nomad_warhorse
		NOT = { trait = incapable }
		is_inaccessible_trigger = no
		OR = {
			any_vassal = { horse_biter_owner_trigger = yes }
			any_courtier = { horse_biter_owner_trigger = yes }
		}
	}

	immediate = {
		if = {
			limit = { any_vassal = { horse_biter_owner_trigger = yes } }
			random_vassal = {
				limit = { horse_biter_owner_trigger = yes }
				character_event = { id = horse.41 }
			}
		}
		else = {
			random_courtier = {
				limit = { horse_biter_owner_trigger = yes }
				character_event = { id = horse.41 }
			}
		}
	}
}

# horse.41 Bounce
# Hidden. Sends horse.42 back to the rider with the biter's owner as FROM.
# Scope: the biter's owner; FROM the rider.
character_event = {
	id = horse.41
	hide_window = yes
	is_triggered_only = yes # horse.40

	immediate = {
		FROM = { character_event = { id = horse.42 } }
	}
}

# horse.42 Bitten
# Have the biter put down (the owner never forgives it) or lead the warhorse
# to another spot.
# Scope: the rider; FROM the biter's owner.
character_event = {
	id = horse.42
	desc = EVTDESChorse42
	picture = GFX_evt_horse_friend
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # horse.41

	option = { # Kill the beast
		name = EVTOPTAhorse42
		reverse_opinion = { who = FROM modifier = opinion_killed_horse years = 15 }
		ai_chance = {
			factor = 1
			modifier = { factor = 3 trait = wroth }
			modifier = { factor = 2 trait = cruel }
		}
	}
	option = { # Another watering spot
		name = EVTOPTBhorse42
		prestige = -30
		ai_chance = {
			factor = 1
			modifier = { factor = 3 trait = patient }
			modifier = { factor = 2 trait = kind }
		}
	}
}

# horse.50 A painter at the camp
# A painter from the settled lands offers a portrait, once a life: of the
# ruler, the spouse, a consort or the warhorse (the best for prestige).
# Scope: the ruler.
character_event = {
	id = horse.50
	desc = EVTDESChorse50
	picture = GFX_evt_market_altaic
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # on_yearly_pulse

	has_character_flag = horse_owner

	trigger = {
		prestige = 1500
		has_character_modifier = nomad_warhorse
		NOT = { trait = incapable }
		is_inaccessible_trigger = no
		NOT = { has_character_flag = nomad_painter }
	}

	immediate = {
		set_character_flag = nomad_painter
	}

	option = { # Myself
		name = EVTOPTAhorse50
		prestige = 50
		wealth = -50
	}
	option = { # My spouse
		name = EVTOPTBhorse50
		trigger = {
			spouse = { always = yes }
		}
		prestige = 25
		wealth = -50
		spouse = {
			prestige = 50
			opinion = { who = ROOT modifier = opinion_happy years = 5 }
		}
	}
	option = { # My consort
		name = EVTOPTChorse50
		trigger = {
			NOT = { spouse = { always = yes } }
			any_courtier = { is_consort = ROOT }
		}
		prestige = 25
		wealth = -50
		random_courtier = {
			limit = { is_consort = ROOT }
			prestige = 50
			opinion = { who = ROOT modifier = opinion_delighted years = 5 }
		}
	}
	option = { # My horse
		name = EVTOPTDhorse50
		prestige = 100
		wealth = -50
	}
	option = { # Too dear
		name = EVTOPTEhorse50
		prestige = -25
		random = {
			chance = 40
			add_trait = shy
			hidden_tooltip = { character_event = { id = 38263 } } # Notify Shy
		}
	}
}
