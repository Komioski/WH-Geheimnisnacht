# frostwitch.1 The Great Warding
# A Grand Ice Witch goes to the Lynsk and pours her own life into the ice.
# For ten years her lord's provinces in Kislev are warded (no Spring Driving
# burns them so easily, and they are calmer), and doom falls back. The rite
# costs her health for the rest of her life. Once per witch.
# Scope: ROOT is the witch.
character_event = {
	id = frostwitch.1
	desc = EVTDESCfrostwitch1
	picture = "GFX_evt_kislev_ice_witch_casting"
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes # decision frost_great_warding

	option = {
		name = EVTOPTAfrostwitch1
		add_character_modifier = { name = frost_warding_toll duration = -1 }
		reduce_doom_progress_minor_effect = yes
		prestige = 200
		change_society_currency = 200
		top_liege = {
			any_realm_province = {
				limit = { region = world_old_world_kislev }
				add_province_modifier = { name = frost_great_warding_mod years = 10 }
			}
			opinion = { who = ROOT modifier = opinion_frost_warded years = 20 }
		}
	}
}

# frostwitch.10 The sorcerer's trail
# A Chaos sorcerer is hiding in the oblast: dead cattle with their eyes
# burned out, a village that dreams the same dream. Follow the trail to him
# (frostwitch.11), or send the Chekist and the lord's men instead.
# Scope: ROOT is the witch.
character_event = {
	id = frostwitch.10
	desc = EVTDESCfrostwitch10
	picture = GFX_evt_chaos_sorcerer
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes # decision frost_hunt_sorcerer

	# Follow the trail
	option = {
		name = EVTOPTAfrostwitch10
		character_event = { id = frostwitch.11 days = 30 }
		ai_chance = { factor = 80 }
	}

	# Send the lord's men

		name = EVTOPTBfrostwitch10

			}
		}

# frostwitch.40 Two Grand Witches
# Two Grand Ice Witches at the same court fall out over a novice or a seat
# at the lord's side. Settle it with a duel of ice (learning; the loser is
# wounded), make peace (friends), or yield the seat.
# Scope: ROOT is a Grand Witch; event_target:frost_rival the other.
character_event = {
	id = frostwitch.40
	desc = EVTDESCfrostwitch40
	picture = "GFX_evt_kislev_ice_witch"
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes # on_five_year_pulse frostwitch_life
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		frost_grand_witch_trigger = yes
		host = {
			any_courtier = {
				frost_grand_witch_trigger = yes
				NOT = { character = ROOT }
			}
		}
	}

	immediate = {
		host = {
			random_courtier = {
				limit = {
					frost_grand_witch_trigger = yes
					NOT = { character = ROOT }
				}
				save_event_target_as = frost_rival
			}
		}
	}

	# A duel of ice
	option = {
		name = EVTOPTAfrostwitch40
		random_list = {
			50 = {
				modifier = { factor = 1.5 learning = 18 }
				prestige = 50
				change_society_currency = 100
				event_target:frost_rival = { add_trait_wounded_effect = yes }
			}
			50 = {
				modifier = { factor = 1.5 event_target:frost_rival = { learning = 18 } }
				add_trait_wounded_effect = yes
				event_target:frost_rival = { change_society_currency = 100 }
			}
		}
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = wroth }
			modifier = { factor = 2 trait = proud }
		}
	}

	# Make peace
	option = {
		name = EVTOPTBfrostwitch40
		trigger = { NOT = { is_rival = event_target:frost_rival } }
		add_friend = event_target:frost_rival
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = kind }
		}
	}

	# Yield the seat
	option = {
		name = EVTOPTCfrostwitch40
		change_society_currency = -100
		prestige = -25
		event_target:frost_rival = { change_society_currency = 100 }
		ai_chance = { factor = 20 }
	}
}

# frostwitch.50 The hag of the village
# An Ice Witch meets an Ungol hag who calls the Sisterhood children playing
# with snow. Scorn her, learn the older magic from her, or, when Chaos is in
# the land, swear a pact with her against it.
# Scope: ROOT is the witch; event_target:frost_hag the hag.
character_event = {
	id = frostwitch.50
	desc = EVTDESCfrostwitch50
	picture = GFX_evt_witch
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_five_year_pulse frostwitch_life
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		society_member_of = ice_witch_sisterhood
		society_rank >= 2
		NOT = { has_character_flag = frost_met_hag }
		top_liege = {
			any_realm_character = { frost_hag_trigger = yes }
		}
	}

	immediate = {
		set_character_flag = frost_met_hag
		top_liege = {
			random_realm_character = {
				limit = { frost_hag_trigger = yes }
				save_event_target_as = frost_hag
			}
		}
	}

	# Scorn her
	option = {
		name = EVTOPTAfrostwitch50
		change_society_currency = 50
		event_target:frost_hag = {
			opinion = { who = ROOT modifier = opinion_frost_scorned years = 20 }
		}
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = proud }
		}
	}

	# Learn from her
	option = {
		name = EVTOPTBfrostwitch50
		change_society_currency = -50
		add_character_modifier = { name = frost_hag_lore years = 20 }
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = humble }
		}
	}

	# A pact against Chaos
	option = {
		name = EVTOPTCfrostwitch50
		trigger = {
			top_liege = { bulwark_defending_vs_chaos_trigger = yes }
		}
		add_character_modifier = { name = frost_hag_pact years = 5 }
		event_target:frost_hag = {
			add_character_modifier = { name = frost_hag_pact years = 5 }
			opinion = { who = ROOT modifier = opinion_frost_pact years = 20 }
		}
		ai_chance = { factor = 60 }
	}
}

	# Knock and go in
	option = {
		name = EVTOPTAfrostwitch60
		character_event = { id = frostwitch.61 days = 1 }
		ai_chance = { factor = 50 }
	}

	# Call her by the old name
	option = {
		name = EVTOPTBfrostwitch60
		set_character_flag = frost_called_old_name
		character_event = { id = frostwitch.61 days = 1 }
		ai_chance = {
			factor = 40
			modifier = { factor = 2 frost_hag_trigger = yes }
		}
	}

	# Turn back
	option = {
		name = EVTOPTCfrostwitch60
		prestige = -25
		ai_chance = { factor = 10 }
	}
}

# frostwitch.61 The Hag-Mother
# Inside, the Hag-Mother sets her questions. A pilgrim who answers well
# comes home with her distaff (a great artifact, one in the world while it
# lasts); most come home wiser; some never come home. A hag, and one who
# called her by the old name, does better.
# Scope: ROOT is the pilgrim.
character_event = {
	id = frostwitch.61
	desc = EVTDESCfrostwitch61
	picture = GFX_evt_witch
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes # frostwitch.60

	option = {
		name = EVTOPTAfrostwitch61
		random_list = {
			20 = {
				modifier = { factor = 1.5 learning = 14 }
				modifier = { factor = 1.5 frost_hag_trigger = yes }
				modifier = { factor = 1.5 has_character_flag = frost_called_old_name }
				modifier = { factor = 0 has_global_flag = frost_distaff_given }
				set_global_flag = frost_distaff_given
				add_artifact = frost_hag_distaff
				custom_tooltip = { text = frost_hag_mother_gift_tt }
			}
			55 = {
				modifier = { factor = 1.2 learning = 10 }
				add_character_modifier = { name = frost_hag_mother_wisdom years = 20 }
				custom_tooltip = { text = frost_hag_mother_wisdom_tt }
			}
			25 = {
				modifier = { factor = 0.5 frost_hag_trigger = yes }
				modifier = { factor = 0.5 has_character_flag = frost_called_old_name }
				modifier = { factor = 0.7 learning = 14 }
				death = { death_reason = death_murder_unknown }
				custom_tooltip = { text = frost_hag_mother_lost_tt }
			}
		}
		clr_character_flag = frost_called_old_name
	}
}
