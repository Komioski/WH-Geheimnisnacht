###############################################################################
# Daemonic incursions and the warp tides
# Namespace: incursion (was daemonic_incursion in daemonic_incursion_events.txt)
# Faction: Chaos
# Serves: the Chaos realms of the warp (offmap powers khorne_warp,
#   nurgle_warp, slaanesh_warp, tzeentch_warp, undivided_warp and the
#   Brayherd, brayherd_warp), and every mortal realm they prey on
#
# While the winds of magic blow strong, the Chaos realms of the warp can
# tear open a rift inside a mortal realm and pour daemons into realspace.
# How often follows the warp tides: each realm can rise to ascendancy, be
# invaded by a rival power or, at the end of days, unite under Undivided.
# The Brayherd follows Undivided and raises warherds out of the deep
# forests. Built on the Jade Dragon invasion tombola (cathayinv.1).
#
# Rifts in the veil                    incursion.1 - incursion.4
# Warp tides                           incursion.10 - incursion.12
# Beastmen incursions                  incursion.30
#
# Callers: incursion.1, 10 and 30 from on_offmap_yearly_pulse, incursion.11
#   from on_offmap_status_changed (00_on_actions.txt). The war uses the
#   daemonic_incursion CB (daemonic_incursion_cb_types.txt), whose end
#   blocks banish the host and move the doom counter. Game rule
#   daemonic_incursions turns the rifts off.
# Saved: character flag daemonic_incursion_leader and modifier
#   daemonic_incursion_timer (25 years, on the victim); offmap flags
#   warp_tide_check (tmp), warp_tide_turned and no_status_news. Effects in
#   wh_incursion_effects.txt, triggers in wh_event_chain_triggers.txt and
#   the warp_ triggers. The doom counter shows the tides
#   (09_customizable_localisation_doom_counter.txt).
# Loc: localisation/00_WHG_incursion_events.csv
###############################################################################

namespace = incursion

# incursion.1 The veil thins
# Hidden. Once a year each Chaos realm of the warp may open a rift: about
# one time in sixteen at medium doom, more at high doom, and more again
# when the realm is ascendant, the powers stand united or the realm's own
# policy looks to the mortal world. A realm at war in the warp sends fewer.
# Scope: ROOT is the warp realm's governor. FROM: the warp realm. Fired by
#   on_offmap_yearly_pulse (00_on_actions.txt).
character_event = {
	id = incursion.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		FROM = {
			OR = {
				is_offmap_tag = khorne_warp
				is_offmap_tag = nurgle_warp
				is_offmap_tag = slaanesh_warp
				is_offmap_tag = tzeentch_warp
				is_offmap_tag = undivided_warp
			}
		}
		war = no
		NOT = { has_character_flag = daemonic_incursion_leader }
		NOT = {
			has_game_rule = {
				name = daemonic_incursions
				value = none
			}
		}
		OR = {
			is_doom_medium_trigger = yes
			is_doom_high_trigger = yes
			is_doom_very_high_trigger = yes
		}
	}

	immediate = {
		random_list = {
			6 = { # Incursion
				modifier = {
					factor = 2
					is_doom_high_trigger = yes
				}
				modifier = {
					factor = 4
					is_doom_very_high_trigger = yes
				}
				modifier = {
					factor = 2
					FROM = { has_policy = warp_focused_mortal }
				}
				modifier = {
					factor = 3
					FROM = { has_policy = warp_endtimes }
				}
				modifier = {
					factor = 0.5
					FROM = { has_policy = warp_focused_warp }
				}
				# Warp tides (see incursion.10)
				modifier = { # This realm is ascendant
					factor = 3
					FROM = { has_status = warp_ascendant }
				}
				modifier = { # The Ruinous Powers stand united
					factor = 2
					FROM = { has_status = warp_united }
				}
				modifier = { # Busy defending its own realm in the warp
					factor = 0.25
					FROM = {
						OR = {
						has_status = warp_khorne_invasion
						has_status = warp_nurgle_invasion
						has_status = warp_slaanesh_invasion
						has_status = warp_tzeentch_invasion
						}
					}
				}
				modifier = { # Busy invading another realm in the warp
					factor = 0.5
					FROM = { is_offmap_tag = khorne_warp }
					OR = {
						nurgle_warp = { has_status = warp_khorne_invasion }
						slaanesh_warp = { has_status = warp_khorne_invasion }
						tzeentch_warp = { has_status = warp_khorne_invasion }
						undivided_warp = { has_status = warp_khorne_invasion }
					}
				}
				modifier = { # Busy invading another realm in the warp
					factor = 0.5
					FROM = { is_offmap_tag = nurgle_warp }
					OR = {
						khorne_warp = { has_status = warp_nurgle_invasion }
						slaanesh_warp = { has_status = warp_nurgle_invasion }
						tzeentch_warp = { has_status = warp_nurgle_invasion }
						undivided_warp = { has_status = warp_nurgle_invasion }
					}
				}
				modifier = { # Busy invading another realm in the warp
					factor = 0.5
					FROM = { is_offmap_tag = slaanesh_warp }
					OR = {
						khorne_warp = { has_status = warp_slaanesh_invasion }
						nurgle_warp = { has_status = warp_slaanesh_invasion }
						tzeentch_warp = { has_status = warp_slaanesh_invasion }
						undivided_warp = { has_status = warp_slaanesh_invasion }
					}
				}
				modifier = { # Busy invading another realm in the warp
					factor = 0.5
					FROM = { is_offmap_tag = tzeentch_warp }
					OR = {
						khorne_warp = { has_status = warp_tzeentch_invasion }
						nurgle_warp = { has_status = warp_tzeentch_invasion }
						slaanesh_warp = { has_status = warp_tzeentch_invasion }
						undivided_warp = { has_status = warp_tzeentch_invasion }
					}
				}
				character_event = { id = incursion.2 }
			}
			90 = { # The veil holds this year
			}
		}
	}
}

# incursion.2 A rift opens
# Hidden. Picks a mortal realm of five counties or more (Araby, and the
# lands of the undead and greenskins, are spared most often), opens a rift
# in one of its provinces, spawns a host that grows with the doom and
# declares the daemonic_incursion war. Players nearby hear of it.
# Scope: ROOT is the warp realm's governor (incursion.1).
character_event = {
	id = incursion.2
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		# Araby lies furthest from the polar gates and is the safest of all.
		# Daemons also care less for the lands of the undead and greenskins.
		random_list = {
			75 = { set_character_flag = incursion_wants_usual_prey }
			20 = { set_character_flag = incursion_spares_araby }
			5 = { }
		}
		random_independent_ruler = {
			limit = {
				is_playable = yes
				num_of_count_titles_in_realm = 5
				NOR = {
					religion_group = chaos_gods_group
					is_offmap_governor = khorne_warp
					is_offmap_governor = nurgle_warp
					is_offmap_governor = slaanesh_warp
					is_offmap_governor = tzeentch_warp
					is_offmap_governor = undivided_warp
					has_character_modifier = daemonic_incursion_timer
				}
				any_realm_province = { is_land = yes }
			}
			preferred_limit = {
				OR = {
					AND = {
						ROOT = { has_character_flag = incursion_wants_usual_prey }
						daemonic_incursion_araby_trigger = no
						daemonic_incursion_unloved_prey_trigger = no
					}
					AND = {
						ROOT = { has_character_flag = incursion_spares_araby }
						daemonic_incursion_araby_trigger = no
					}
					ROOT = {
						NOR = {
							has_character_flag = incursion_wants_usual_prey
							has_character_flag = incursion_spares_araby
						}
					}
				}
			}
			save_event_target_as = incursion_target
		}
		clr_character_flag = incursion_wants_usual_prey
		clr_character_flag = incursion_spares_araby
		if = { # A victim was found
			limit = { event_target:incursion_target = { always = yes } }

			event_target:incursion_target = {
				add_character_modifier = {
					name = daemonic_incursion_timer
					years = 25
					hidden = yes
				}
				random_realm_province = {
					limit = { is_land = yes }
					save_event_target_as = incursion_rift
				}
			}

			set_character_flag = daemonic_incursion_leader
			save_event_target_as = incursion_leader

			# The host grows with the winds of magic
			if = {
				limit = { is_doom_very_high_trigger = yes }
				spawn_unit = {
					province = event_target:incursion_rift
					owner = ROOT
					earmark = daemonic_incursion_host
					cannot_inherit = yes
					attrition = 0
					reinforces = no
					troops = {
						daemonic_host = { 6000 6000 }
						daemonic_host = { 4000 4000 }
						daemonic_host = { 4000 4000 }
					}
				}
			}
			else_if = {
				limit = { is_doom_high_trigger = yes }
				spawn_unit = {
					province = event_target:incursion_rift
					owner = ROOT
					earmark = daemonic_incursion_host
					cannot_inherit = yes
					attrition = 0
					reinforces = no
					troops = {
						daemonic_host = { 4500 4500 }
						daemonic_host = { 3000 3000 }
						daemonic_host = { 3000 3000 }
					}
				}
			}
			else = {
				spawn_unit = {
					province = event_target:incursion_rift
					owner = ROOT
					earmark = daemonic_incursion_host
					cannot_inherit = yes
					attrition = 0
					reinforces = no
					troops = {
						daemonic_host = { 2500 2500 }
						daemonic_host = { 1500 1500 }
						daemonic_host = { 1500 1500 }
					}
				}
			}

			war = {
				target = event_target:incursion_target
				casus_belli = daemonic_incursion
			}
			add_doom_progress_minor_effect = yes

			event_target:incursion_target = { character_event = { id = incursion.3 } }
			any_player = {
				limit = {
					NOT = { character = event_target:incursion_target }
					is_within_diplo_range = event_target:incursion_target
				}
				narrative_event = { id = incursion.4 days = 1 }
			}
		}
	}
}

# incursion.10 The warp tides turn
# Hidden. Each realm is read every five years (the Brayherd every year).
# At low and medium doom every realm settles back to stable. At high doom
# and above a stable realm may rise to ascendancy (one at a time) or be
# invaded by a rival power, and ascendant or invaded realms settle back
# after a while; at very high doom Undivided may unite all four powers.
# The Brayherd, the beastman branch of Undivided, gathers while Undivided
# is ascendant or united and scatters otherwise.

				is_offmap_tag = undivided_warp
				is_offmap_tag = brayherd_warp
			}
			OR = { # The tides are read every 5 years (the Brayherd follows Undivided)
				is_offmap_tag = brayherd_warp
				NOT = { has_offmap_tmp_flag = warp_tide_check }
				had_offmap_tmp_flag = { flag = warp_tide_check years = 5 }
			}
		}
	}

	immediate = {
		FROM = {
			clr_offmap_tmp_flag = warp_tide_check

					limit = { NOT = { has_status = warp_stable } }
					warp_tide_turned_effect = yes
					set_status = warp_stable
				}
			}
			else = {
					if = {
						limit = {
							is_offmap_tag = khorne_warp
							has_status = warp_stable
							OR = {
								NOT = { has_offmap_flag = warp_tide_turned }
								had_offmap_tmp_flag = { flag = warp_tide_turned years = 3 }
							}
						}
						random_list = {
							5 = { # Rise to ascendancy
								modifier = {
									factor = 0
									OR = {
										warp_any_realm_ascendant_trigger = yes
										warp_powers_united_trigger = yes

								}

# Hidden. Tells the players who would notice: the Chaos faithful and the
# beastmen always, everyone when a realm rises or the powers unite. A quiet

# Scope: ROOT is the warp realm's governor. FROM: the warp realm. Fired by
#   on_offmap_status_changed (00_on_actions.txt).

	id = incursion.11
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		FROM = {
			OR = {
				is_offmap_tag = khorne_warp
				is_offmap_tag = nurgle_warp
				is_offmap_tag = slaanesh_warp
				is_offmap_tag = tzeentch_warp
				is_offmap_tag = undivided_warp
				is_offmap_tag = brayherd_warp
			}
		}
	}

	immediate = {
		if = { # A quiet reset at low doom: no news
			limit = { FROM = { has_offmap_flag = no_status_news } }
			FROM = { clr_offmap_flag = no_status_news }
		}
		else = {
			any_player = {
				limit = {
					OR = {
						religion_group = chaos_gods_group
						trait = creature_beastman
						FROM = {
							OR = {
								has_status = warp_ascendant
								has_status = warp_united
							}
						}
					}
				}
				narrative_event = { id = incursion.12 }
			}
		}
	}
}
