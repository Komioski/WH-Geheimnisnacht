# Way of Life: the intrigue focus
# Namespace: intrigue (was WoL.2000 - WoL.2305 in wol_intrigue_events.txt)
# Faction: Generic
#
# A ruler with the intrigue focus lives among whispers. The Spy On decision
# sets snoops on a rival: they may uncover a plot, dig up dirt (slander,
# bastardy, heresy, a foreign faith, deviancy, treason, a cult or demon
# worship) to denounce, blackmail or keep, make up vicious rumours, hire a
# killer or have the rival snatched on the road. The Liberate decision
# springs a prisoner or a forced concubine from a foreign court through a
# bribed, seduced or sympathetic agent. Once in a while the ruler catches
# their own spymaster making chalk marks on the wall.
#
# The spy's target is event_target:target_spy_on; the liberator's captive
# is event_target:target_captive and the agent event_target:target_agent.
# The suspect spymaster is event_target:intrigue_spymaster.
#
# The schemer's road                   intrigue.1 - intrigue.4
# Spy On: start, router, endings       intrigue.10 - intrigue.31
# Spy On: dirt and blackmail           intrigue.40 - intrigue.51
# Spy On: rumours, murder, abduction   intrigue.60 - intrigue.83
# Spy On: a cheating spouse            intrigue.100 - intrigue.101
# Liberate                             intrigue.110 - intrigue.142
# The spymaster plots                  intrigue.150 - intrigue.155
#
# Fired from: the spy_on, stop_spy and break_free decisions in
#   way_of_life_decisions.txt, on_focus_pulse in WHG_pulse_on_actions.txt
#   (intrigue.150) and the lover events (intrigue.1). Shared effects are in
#   WHG_character_effects.txt.
#
# Also in this file (with its own header and namespace):
#   Part 2: In hiding (namespace hiding, was CM.6000 - CM.6400 in
#   cm_murder_plot_events.txt)
###############################################################################

namespace = intrigue
# intrigue.1 Schemer level up chance
character_event = {
	id = intrigue.1

	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_dlc = "Way of Life"
	}

	immediate = {
		# Chance of trait adjustments
		if = {
			limit = {
				NOT = { trait = honest }
				NOT = { trait = deceitful }
				random = 85 # 15% chance
			}
			add_trait = deceitful
			character_event = { id = traitnote.255 } # Deceitful notification
		}

		if = {
			limit = { trait = honest random = 50 }
			remove_trait = honest
			character_event = { id = traitnote.311 } # Honest loss notification
		}

		# Chance of various Intrigue boosters
		character_event = { id = intrigue.2 days = 7 }

		# Chance of leveling up Schemer skills
		if = {
			limit = { has_character_modifier = wol_schemer_2 }
			random = {
				chance = 10
				character_event = { id = intrigue.4 days = 4 } # Gain the Schemer or Impaler trait
			}
		}
		else_if = {
			limit = { has_character_modifier = wol_schemer_1 }
			random = {
				chance = 20
				remove_character_modifier = wol_schemer_1
				add_character_modifier = { modifier = wol_schemer_2 duration = -1 }
			}
		}

		else_if = {
			limit = { NOR = { trait = schemer trait = impaler } }
			random = {
				chance = 50
				add_character_modifier = { modifier = wol_schemer_1 duration = -1 }
			}
		}
	}
}

# Part 2: In hiding (namespace hiding)
# Faction: Generic
# Serves: every ruler, consort and child who goes into hiding from a murder
#   plot, and the plotters who hunt them there
# (was CM.6000 - CM.6400 in the vanilla cm_murder_plot_events.txt)
#
# A ruler who fears for their life may hide, or hide a consort or a child
# (dynasty_decisions.txt). Kin, rivals and plotters hear of it. Years in a
# cellar change a person: they may turn paranoid, pious, fat, drunk, poetic
# or mad. Rumours of plots set the ruler thinking of hiding at all. A
# plotter whose target has gone to ground may still find a way in: hired
# killers (Clan Eshin for the Skaven, the Khainite assassins for the
# Druchii), a bribed cook or guard, a snake charmer, a disloyal chamber maid
# for the bedridden, or a witch's poisoned sweets for a child.
#
# The vanilla file carried each of the six murder methods as ten copied
# events; they are one chain here (hiding.50 - 60), and the method is a
# flag on the plotter (hiding_murder_<method>).
#
# News of hiding                      hiding.1 - hiding.7
# Life in hiding                      hiding.10 - hiding.28
# Plots broken off and cleared        hiding.30 - hiding.31
# Rumours of murder                   hiding.40 - hiding.43
# Murder in hiding                    hiding.50 - hiding.60
#
# Callers: hiding.1 - 6 and hiding.30 from the hiding decisions
#   (dynasty_decisions.txt); hiding.7 from every event that brings a hidden
#   character out (feasts, tournaments, holy orders, witch hunters and
#   others); hiding.10 - 28 from on_yearly_pulse and hiding.31 from on_death
#   (WHG_pulse_on_actions); hiding.40, 41 and 50 from the yearly lists
#   hiding_rumours and hiding_murder_chances (WHG_pulse_on_actions).
# Saved: event targets hiding_plotter, hiding_target; flags
#   hiding_murder_in_motion, hiding_murder_extra (plotter), hiding_murder_
#   assassin / poison / guard / snake / maid / candy (plotter, the method,
#   overwritten by the next attempt); variable lunatic_counter; modifier
#   suspected_murder_plot. Loc names come from GetHidingKiller,
#   GetHidingDeath and GetHidingGuild
#   (wh_hiding_customizable_localisation.txt).
###############################################################################

# hiding.50 A way in
# An adult target faces hired killers (Clan Eshin for the Skaven, the
# Khainite assassins for the Druchii), a bribed cook, a bribed guard or a
# snake charmer; a bedridden one a disloyal chamber maid; a child a witch's
# poisoned sweets.
# Scope: a murder plotter whose target is in hiding. Fired by the yearly
#   hiding_murder_chances list (was CM.6300 - 6350, MTTH 1000 months each).
character_event = {
	id = hiding.50
	picture = GFX_evt_shadow
	border = GFX_event_normal_frame_intrigue
	is_triggered_only = yes
	min_age = 16
	prisoner = no

	trigger = {
		NOT = { trait = incapable }
		hiding_murder_plotter_trigger = yes
	}

	weight_multiplier = {
		factor = 1
		modifier = {
			factor = 4 # Four ways in for an adult on their feet
			plot_target_char = {
				is_adult = yes
				NOT = { trait = incapable }
			}
		}
		modifier = { factor = 1.4 plot_power = 1.5 }
		modifier = { factor = 1.4 plot_power = 2.0 }
		modifier = { factor = 1.4 plot_power = 3.0 }
		modifier = { factor = 1.4 plot_power = 4.0 }
	}

	immediate = {
		save_event_target_as = hiding_plotter
		plot_target_char = { save_event_target_as = hiding_target }
		clr_character_flag = hiding_murder_assassin
		clr_character_flag = hiding_murder_poison
		clr_character_flag = hiding_murder_guard
		clr_character_flag = hiding_murder_snake
		clr_character_flag = hiding_murder_maid
		clr_character_flag = hiding_murder_candy
		if = {
			limit = { event_target:hiding_target = { trait = incapable } }
			set_character_flag = hiding_murder_maid
		}
		else_if = {
			limit = { event_target:hiding_target = { is_adult = no } }
			set_character_flag = hiding_murder_candy
		}
		else = {
			random_list = {
				25 = { set_character_flag = hiding_murder_assassin }
				25 = { set_character_flag = hiding_murder_poison }
				25 = { set_character_flag = hiding_murder_guard }
				25 = { set_character_flag = hiding_murder_snake }
			}
			if = { # The dead do not eat: their cook becomes a guard
				limit = {
					has_character_flag = hiding_murder_poison
					event_target:hiding_target = { is_undead_trigger = yes }
				}
				clr_character_flag = hiding_murder_poison
				set_character_flag = hiding_murder_guard
			}
		}
	}

	desc = { trigger = { has_character_flag = hiding_murder_assassin } text = EVTDESChiding50_assassin }
	desc = { trigger = { has_character_flag = hiding_murder_poison } text = EVTDESC_CM_6310 }
	desc = { trigger = { has_character_flag = hiding_murder_guard } text = EVTDESC_CM_6320 }
	desc = { trigger = { has_character_flag = hiding_murder_snake } text = EVTDESC_CM_6330 }
	desc = { trigger = { has_character_flag = hiding_murder_maid } text = EVTDESC_CM_6340 }
	desc = { trigger = { has_character_flag = hiding_murder_candy } text = EVTDESC_CM_6350 }

	option = {
		name = EVTOPTAhiding50 # Go
		ai_chance = { factor = 70 }
		scaled_wealth = -0.5
		set_character_flag = hiding_murder_in_motion
		hidden_tooltip = {
			any_plot_backer = { character_event = { id = hiding.51 } }
			event_target:hiding_target = { character_event = { id = hiding.52 days = 30 random = 12 } }
		}
	}
	option = {
		name = EVTOPTBhiding50 # Go, and pay for more
		ai_chance = {
			factor = 20
			modifier = { factor = 0 NOT = { scaled_wealth = 1.5 } }
		}
		scaled_wealth = -1.5
		set_character_flag = hiding_murder_in_motion
		set_character_flag = hiding_murder_extra
		hidden_tooltip = {
			any_plot_backer = { character_event = { id = hiding.51 } }
			event_target:hiding_target = { character_event = { id = hiding.52 days = 30 random = 12 } }
		}
	}
	option = {
		name = EVTOPTChiding50 # Wait
		ai_chance = { factor = 10 }
	}
}
