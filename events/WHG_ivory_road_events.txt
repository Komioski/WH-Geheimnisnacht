# WHG_ivory_road_events.txt - the Dragon Emperor's envoys and caravan raiders
#
# Namespace: ivory (was HL.5000-5062 in hl_silk_route_events).
#   1-13     An envoy of the Dragon Emperor of Grand Cathay demands tribute,
#            then serves as trade governor on a road to Cathay: traders,
#            gemstones, thieves, a pet snake, a bride, dice, resignation
#            (needs Horse Lords and Jade Dragon; five-year list
#            ivory_road_envoy)
#   20       The trade governor's province is checked yearly
#   30       The Dragon Emperor's wrath: an embargo
#   50-61    Raiders on a caravan road: share the loot, send the marshal or
#            ride out yourself (five-year list caravan_raiders)
# Faction: Cathay & the East
# Serves: rulers with trade posts on the roads to Grand Cathay (envoy) and
#   on any caravan road (raiders)
#
# Shared triggers and effects are in the Ivory Road section of
# WHG_horse_lords_triggers.txt and WHG_horse_lords_effects.txt.
###############################################################################

namespace = ivory

# ivory.1 The envoy demands tribute
# A ruler with a trade post on a road to Cathay is told that the Dragon
# Emperor expects tribute (five-year list ivory_road_envoy, was MTTH 100
# years).
# Scope: ROOT is the ruler.
character_event = {
	id = ivory.1
	desc = EVTDESCivory1
	picture = GFX_evt_chinese_emissary
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	only_playable = yes
	only_independent = yes
	only_capable = yes
	prisoner = no
	ai = no

	trigger = {
		has_dlc = "Horse Lords"
		has_dlc = "Jade Dragon"
		higher_tier_than = COUNT
		NOR = {
			has_character_flag = had_cathayan_envoy
			has_character_flag = do_not_disturb
			trait = infirm
		}
		NOT = { offmap_cathay = { governor = { character = ROOT } } }
		offmap_cathay = {
			NOR = {
				has_status = cathay_civil_war
				has_status = cathay_plague
			}
		}
		cathay_diplo_interaction_trigger = yes
		any_demesne_province = { ivory_road_cathay_province_trigger = yes }
	}

	immediate = {
		set_character_flag = had_cathayan_envoy
		random_demesne_province = {
			limit = { ivory_road_cathay_province_trigger = yes }
			save_event_target_as = trade_province
		}
	}

	# Pay
	option = {
		name = EVTOPTAivory1
		scaled_wealth = -1.5
		add_grace_minor_effect = yes
		hidden_tooltip = {
			character_event = { id = ivory.2 days = 90 random = 41 }
		}
	}
	# Refuse
	option = {
		name = EVTOPTBivory1
		hidden_tooltip = {
			letter_event = { id = ivory.30 days = 90 random = 41 }
		}
	}
}

# ivory.2 A gift from the Dragon Emperor

# ivory.3 The envoy at court
# Every three to six years something happens around the envoy; when he is
# gone or the province is lost, his service ends.
# Scope: ROOT is the host; hidden.
character_event = {
	id = ivory.3
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { ivory_envoy_serves_trigger = yes } }
			ivory_envoy_cleanup_effect = yes
		}
		else = {
			random_list = {
				30 = { # More traders
					modifier = {
						factor = 0
						event_target:trade_province = { has_province_modifier = surge_of_traders }
					}
					character_event = { id = ivory.4 }
				}
				30 = { # Gemstones
					modifier = { factor = 0 has_character_flag = cathayan_envoy_event_1 }
					character_event = { id = ivory.5 }
				}
				30 = { # Stolen goods
					modifier = { factor = 0 has_character_flag = cathayan_envoy_event_2 }
					character_event = { id = ivory.6 }
				}
				30 = { # The pet snake
					modifier = { factor = 0 has_character_flag = cathayan_envoy_event_3 }
					character_event = { id = ivory.8 }
				}
				30 = { # The snake strikes
					modifier = {
						factor = 0
						OR = {
							NOT = { has_character_flag = cathayan_envoy_event_3 }
							has_character_flag = cathayan_envoy_event_4
							has_character_flag = cathayan_envoy_snake_ban
						}
					}
					character_event = { id = ivory.9 }
				}
				30 = { # A bride
					modifier = {
						factor = 0
						OR = {
							has_character_flag = cathayan_envoy_event_5
							event_target:cathayan_envoy = { is_married = yes }
							NOT = { any_courtier = { ivory_envoy_bride_trigger = yes } }
						}
					}
					character_event = { id = ivory.10 }
				}
				30 = { # Dice
					modifier = { factor = 0 has_character_flag = cathayan_envoy_event_6 }
					character_event = { id = ivory.11 }
				}
				30 = { # Trade for a Cathayan treasure (cathayart.2)
					modifier = { factor = 0.1 any_artifact = { has_artifact_flag = cathayan } }
					modifier = { factor = 0.6 offmap_cathay = { has_status = cathay_unrest } }
					modifier = { factor = 0.3 offmap_cathay = { has_status = cathay_mongol_invasion } }
					modifier = { factor = 0.3 offmap_cathay = { has_status = cathay_jurchen_invasion } }
					modifier = { factor = 0.3 offmap_cathay = { has_status = cathay_famine } }
					modifier = { factor = 3 offmap_cathay = { has_status = cathay_golden_age } }
					modifier = { factor = 2 offmap_cathay = { has_policy = cathay_expansionist } }
					event_target:cathayan_envoy = { character_event = { id = cathayart.2 } }
				}
				10 = { # Resignation
					modifier = {
						factor = 0
						NOT = { had_character_flag = { flag = had_cathayan_envoy days = 500 } }
					}
					modifier = {
						factor = 0
						event_target:cathayan_envoy = {
							personal_opinion = { who = ROOT value = 20 }
						}
					}
					modifier = {
						factor = 4
						event_target:cathayan_envoy = {
							NOT = { personal_opinion = { who = ROOT value = 0 } }
						}
					}
					modifier = {
						factor = 4
						event_target:cathayan_envoy = {
							NOT = { personal_opinion = { who = ROOT value = -20 } }
						}
					}
					modifier = {
						factor = 4
						event_target:cathayan_envoy = {
							NOT = { personal_opinion = { who = ROOT value = -50 } }
						}
					}
					letter_event = { id = ivory.12 }
				}
			}
			repeat_event = { id = ivory.3 days = 1095 random = 1096 }
		}
	}
}

# ivory.9 The snake strikes
# The serpent swallows a hound: the ruler's own if there is one, else the
# hound of someone at court.
# Scope: ROOT is the host; event_target:ivory_hound_owner.
character_event = {
	id = ivory.9
	desc = {
		trigger = { event_target:ivory_hound_owner = { character = ROOT } }
		text = EVTDESCivory9self
	}
	desc = {
		trigger = { event_target:ivory_hound_owner = { NOT = { character = ROOT } } }
		text = EVTDESCivory9other
	}
	picture = GFX_evt_chinese_emissary
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	trigger = {
		OR = {
			has_character_modifier = pet_hunting_dog
			any_courtier_or_vassal = {
				is_adult = yes
				prisoner = no
			}
		}
	}

	immediate = {
		set_character_flag = cathayan_envoy_event_4
		if = {
			limit = { has_character_modifier = pet_hunting_dog }
			save_event_target_as = ivory_hound_owner
		}
		else_if = {
			limit = {
				any_courtier_or_vassal = {
					has_character_modifier = pet_hunting_dog
					prisoner = no
				}
			}
			random_courtier_or_vassal = {
				limit = {
					has_character_modifier = pet_hunting_dog
					prisoner = no
				}
				save_event_target_as = ivory_hound_owner
			}
		}
		else = {
			random_courtier_or_vassal = {
				limit = {
					is_adult = yes
					prisoner = no
					NOT = { has_character_flag = is_cathayan_envoy }
				}
				save_event_target_as = ivory_hound_owner
			}
		}
		event_target:ivory_hound_owner = {
			if = {
				limit = {
					NOT = { character = ROOT }
					has_character_modifier = pet_hunting_dog
				}
				ivory_snake_eats_hound_effect = yes
			}
		}
	}

	# Such is life
	option = {
		name = EVTOPTAivory9
		trigger = { event_target:ivory_hound_owner = { NOT = { character = ROOT } } }
		event_target:ivory_hound_owner = {
			opinion = {
				who = ROOT
				modifier = grievously_offended
				months = 60
			}
		}
		add_grace_trivial_effect = yes
	}
	# Slay the beast
	option = {
		name = EVTOPTBivory9
		event_target:cathayan_envoy = {
			opinion = {
				who = ROOT
				modifier = grievously_offended
				months = 60
			}
		}
		if = {
			limit = { event_target:ivory_hound_owner = { character = ROOT } }
			ivory_snake_eats_hound_effect = yes
		}
		detract_grace_trivial_effect = yes
	}
	# Mourn the hound
	option = {
		name = EVTOPTCivory9
		trigger = { event_target:ivory_hound_owner = { character = ROOT } }
		ivory_snake_eats_hound_effect = yes
		opinion = {
			who = event_target:cathayan_envoy
			modifier = grievously_offended
			months = 60
		}
		random = {
			chance = 33
			add_trait = depressed
			hidden_tooltip = { character_event = { id = 38288 } }
		}
	}
	# The envoy's head
	option = {
		name = EVTOPTDivory9
		trigger = { event_target:ivory_hound_owner = { character = ROOT } }
		ivory_snake_eats_hound_effect = yes
		event_target:cathayan_envoy = {
			if = {
				limit = { is_invincible_trigger = yes }
				antideath_effect = yes
			}
			else = {
				death = {
					death_reason = death_execution_beheading
					killer = ROOT
				}
			}
		}
		hidden_tooltip = { ivory_envoy_cleanup_effect = yes }
		detract_grace_major_effect = yes
	}
}
