# tsar.1 The palace holds
# Hidden. At the start of the game a ruler of Kislev who is no Ice Witch
# already pays the Sisterhood to keep the Ice Palace whole.
# Scope: ROOT holds e_kislev.
character_event = {
	id = tsar.1
	hide_window = yes

	is_triggered_only = yes # on_startup
	only_rulers = yes

	trigger = {
		has_landed_title = e_kislev
		NOT = { has_global_flag = tsar_crown_started }
	}

	immediate = {
		set_global_flag = tsar_crown_started
		coronation_regalia_effect = yes # The Frost Crown (wh_coronation_effects)
		if = {
			limit = { NOT = { trait = ice_witch } }
			add_character_modifier = { name = tsar_ice_palace_upkeep duration = -1 }
		}
	}
}

# tsar.2 A new ruler of Kislev
# Hidden. The new Tzarina is crowned in the Ice Palace (tsar.3), the new
# Tzar at Ursun's temple (tsar.4). A ruler who is no Ice Witch finds the
# palace dripping (tsar.10); an Ice Witch stops a palace that was left to
# melt.
# Scope: ROOT is the new holder, FROM the title.
character_event = {
	id = tsar.2
	hide_window = yes

	is_triggered_only = yes # on_new_holder, _inheritance, _usurpation
	only_rulers = yes

	trigger = {
		FROM = { title = e_kislev }
	}

	immediate = {
		if = {
			limit = { is_woman_trigger = yes }
			character_event = { id = tsar.3 days = 7 }
		}
		else = {
			character_event = { id = tsar.4 days = 7 }
		}
		if = {
			limit = { has_global_flag = kreml_guard_founded }
			character_event = { id = kislevnorth.20 days = 60 } # The Kreml Guard swears (WHG_bulwark_events)
		}
		if = {
			limit = { trait = ice_witch }
			clr_global_flag = ice_palace_melting
		}
		else_if = {
			limit = {
				NOT = { has_global_flag = ice_palace_melting }
				NOT = { has_global_flag = ice_palace_melted }
			}
			character_event = { id = tsar.10 days = 30 random = 30 }
		}
	}
}

# tsar.3 The Tzarina's crowning
# The new Tzarina is crowned in the Ice Palace with the frost crown, or in
# the Bokha Palace if the Ice Palace is gone.
# Scope: ROOT is the new Tzarina.
character_event = {
	id = tsar.3
	desc = {
		trigger = { has_global_flag = ice_palace_melted }
		text = EVTDESCtsar3b
	}
	desc = {
		trigger = { NOT = { has_global_flag = ice_palace_melted } }
		text = EVTDESCtsar3
	}
	picture = GFX_evt_coronation
	border = GFX_event_narrative_frame_diplomacy

	is_triggered_only = yes # tsar.2

	trigger = { has_landed_title = e_kislev }

	option = {
		name = EVTOPTAtsar3
		prestige = 100
		coronation_regalia_effect = yes # The Frost Crown, if it was never made
		if = {
			limit = { trait = ice_witch }
			piety = 50
		}
		any_vassal = {
			limit = { religion = kislev_gods }
			opinion = { who = ROOT modifier = opinion_tsar_crowned }
		}
	}
}

# tsar.4 The night in the bear cave
# The new Tzar spends a night in Ursun's cave before he is crowned at the
# temple. Go into the cave, or be crowned without it. Either way his
# reign begins with Ursun's Trial: he must win against Chaos, beasts or
# greenskins within five years (tsar.5, tsar.6).
# Scope: ROOT is the new Tzar.
character_event = {
	id = tsar.4
	desc = EVTDESCtsar4
	picture = GFX_evt_tamed_bear
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes # tsar.2

	trigger = { has_landed_title = e_kislev }

	immediate = {
		set_character_flag = tsar_ursun_trial
		coronation_regalia_effect = yes # The Frost Crown, if it was never made
		character_event = { id = tsar.6 days = 1825 }
	}

	# Into the cave
	option = {
		name = EVTOPTAtsar4
		random_list = {
			70 = {
				modifier = { factor = 1.5 trait = brave }
				modifier = { factor = 0.5 trait = craven }
				prestige = 150
				add_character_modifier = { name = tsar_ursun_blessed years = 10 }
			}
			20 = {
				prestige = 150
				add_trait = wounded
				add_character_modifier = { name = tsar_ursun_blessed years = 10 }
			}
			10 = {
				prestige = 50
				add_maimed_trait_effect = yes
			}
		}
		ai_chance = {
			factor = 80
			modifier = { factor = 0.2 trait = craven }
		}
	}

	# Crowned without the cave
	option = {
		name = EVTOPTBtsar4
		prestige = -100
		any_vassal = {
			limit = { religion = kislev_gods }
			opinion = { who = ROOT modifier = opinion_tsar_shunned_cave }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 3 trait = craven }
		}
	}
}

# Five years and no victory: the boyars say Ursun has turned from the Tzar,
# and a pretender rises.
# Scope: ROOT is the Tzar.
character_event = {
	id = tsar.6
	desc = EVTDESCtsar6
	picture = GFX_evt_kislev_boyars
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # tsar.4
	has_character_flag = tsar_ursun_trial

	trigger = {
		has_landed_title = e_kislev
	}

	immediate = {
		clr_character_flag = tsar_ursun_trial
		random_vassal = {
			limit = {
				higher_tier_than = COUNT
				culture_group = kislevite_group
				is_adult = yes
				prisoner = no
			}
			preferred_limit = { is_rival = ROOT }
			preferred_limit = { trait = ambitious }
			save_event_target_as = tsar_pretender
		}
	}

	option = {
		name = EVTOPTAtsar6
		add_character_modifier = { name = tsar_trial_failed years = 10 }
		any_vassal = {
			limit = { culture_group = kislevite_group }
			opinion = { who = ROOT modifier = opinion_tsar_trial_failed }
		}
		if = {
			limit = { event_target:tsar_pretender = { always = yes } }
			event_target:tsar_pretender = { add_weak_claim = e_kislev }
		}
	}
}

# tsar.10 The Thaw
# The new ruler is no Ice Witch, and the Ice Palace drips. Beg the
# Sisterhood to hold it (a cost every year), move the court to the Bokha
# Palace and leave the Ice Palace to melt slowly (tsar.13), or let it melt
# now: a new age, an angry Ice Court, and a flood (tsar.11).
# Scope: ROOT is the new ruler of Kislev.
character_event = {
	id = tsar.10
	desc = EVTDESCtsar10
	picture = GFX_evt_kislev_ice_witch
	border = GFX_event_narrative_frame_diplomacy

	is_triggered_only = yes # tsar.2

	trigger = {
		has_landed_title = e_kislev
		NOT = { trait = ice_witch }
		NOT = { has_global_flag = ice_palace_melted }
		NOT = { has_global_flag = ice_palace_melting }
	}

	# Beg the Sisterhood
	option = {
		name = EVTOPTAtsar10
		add_character_modifier = { name = tsar_ice_palace_upkeep duration = -1 }
		ai_chance = { factor = 60 }
	}

	# Move to the Bokha Palace
	option = {
		name = EVTOPTBtsar10
		add_character_modifier = { name = tsar_bokha_court duration = -1 }
		set_global_flag = ice_palace_melting
		character_event = { id = tsar.13 days = 1825 random = 1825 }
		ai_chance = { factor = 25 }
	}

	# Let it melt
	option = {
		name = EVTOPTCtsar10
		set_global_flag = ice_palace_melted
		add_character_modifier = { name = tsar_new_age years = 20 }
		any_realm_character = {
			limit = { trait = ice_witch }
			opinion = { who = ROOT modifier = opinion_tsar_palace_melted }
		}
		character_event = { id = tsar.11 days = 60 random = 60 }
		ai_chance = {
			factor = 15
			modifier = { factor = 3 trait = ambitious }
		}
	}
}

# tsar.15 The Ice Palace rises
# An Ice Witch on the throne calls the Ice Palace back out of the frost
# (decision tsar_raise_ice_palace). Great prestige, and once in her life a
# shard of the Frozen Throne.
# Scope: ROOT is the Tzarina.
character_event = {
	id = tsar.15
	desc = EVTDESCtsar15
	picture = GFX_evt_kislev_ice_witch_casting
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes # tsar_raise_ice_palace

	immediate = {
		clr_global_flag = ice_palace_melted
		clr_global_flag = ice_palace_melting
		remove_character_modifier = tsar_bokha_court
		remove_character_modifier = tsar_ice_palace_upkeep
	}

	option = {
		name = EVTOPTAtsar15
		prestige = 500
		if = {
			limit = {
				NOT = { has_character_flag = tsar_raised_palace }
				NOT = { has_artifact = tsar_frozen_throne_shard }
			}
			set_character_flag = tsar_raised_palace
			add_artifact = tsar_frozen_throne_shard
		}
		any_realm_character = {
			limit = { trait = ice_witch }
			remove_opinion = { who = ROOT modifier = opinion_tsar_palace_melted }
			opinion = { who = ROOT modifier = opinion_tsar_palace_raised }
		}
	}
}

# tsar.20 The Unwed Queen
# A great boyar presses his suit for the unwed Tzarina's hand. Marry him
# (he comes to her court, her house keeps the children), refuse him (he
# is shamed and may turn against her), or, as an Ice Witch, say she is wed
# to Kislev.
# Scope: ROOT is the Tzarina; event_target:tsar_suitor the boyar.
character_event = {
	id = tsar.20
	desc = EVTDESCtsar20
	picture = GFX_evt_kislev_boyars
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # on_five_year_pulse tsar_court
	only_rulers = yes
	only_women = yes

	trigger = {
		has_landed_title = e_kislev
		is_married = no
		is_adult = yes
		NOT = { age = 50 }
		any_vassal = {
			higher_tier_than = COUNT
			is_female = no
			is_adult = yes
			is_married = no
			prisoner = no
			culture_group = kislevite_group
		}
	}

	immediate = {
		random_vassal = {
			limit = {
				higher_tier_than = COUNT
				is_female = no
				is_adult = yes
				is_married = no
				prisoner = no
				culture_group = kislevite_group
			}
			preferred_limit = { trait = ambitious }
			save_event_target_as = tsar_suitor
		}
	}

	# Marry him
	option = {
		name = EVTOPTAtsar20
		add_spouse_matrilineal = event_target:tsar_suitor
		event_target:tsar_suitor = { opinion = { who = ROOT modifier = opinion_tsar_favoured } }
		ai_chance = {
			factor = 40
			modifier = { factor = 0.2 is_lover = event_target:tsar_suitor }
		}
	}

	# Refuse him
	option = {
		name = EVTOPTBtsar20
		add_character_modifier = { name = tsar_unwed_queen years = 5 }
		event_target:tsar_suitor = {
			opinion = { who = ROOT modifier = opinion_tsar_suit_refused }
			if = {
				limit = { trait = ambitious }
				join_faction = { faction = faction_independence faction_leader = THIS can_join_existing = yes }
			}
		}
		ai_chance = { factor = 40 }
	}

	# Wed to Kislev
	option = {
		name = EVTOPTCtsar20
		trigger = { trait = ice_witch }
		piety = 50
		event_target:tsar_suitor = { opinion = { who = ROOT modifier = opinion_tsar_suit_declined } }
		ai_chance = { factor = 40 }
	}
}

# tsar.25 The consort's plot
# Very rare. The Tzarina's husband has been gathering boyars who would
# rather see a man rule in her name. Throw him in the cells, send him away
# to the frontier, or give him a seat at court and keep him close.
# Scope: ROOT is the Tzarina; event_target:tsar_consort her husband.
character_event = {
	id = tsar.25
	desc = EVTDESCtsar25
	picture = GFX_evt_shadowy_cabal
	border = GFX_event_narrative_frame_intrigue

	is_triggered_only = yes # on_five_year_pulse tsar_court
	only_rulers = yes
	only_women = yes

	trigger = {
		has_landed_title = e_kislev
		any_spouse = {
			is_ruler = no
			prisoner = no
			host = { character = ROOT }
			OR = {
				trait = ambitious
				trait = deceitful
				intrigue = 12
			}
		}
	}

	immediate = {
		random_spouse = {
			limit = {
				is_ruler = no
				prisoner = no
				host = { character = ROOT }
				OR = {
					trait = ambitious
					trait = deceitful
					intrigue = 12
				}
			}
			save_event_target_as = tsar_consort
		}
	}

	# To the cells
	option = {
		name = EVTOPTAtsar25
		event_target:tsar_consort = {
			imprison = ROOT
			opinion = { who = ROOT modifier = opinion_tsar_disgraced }
		}
		prestige = 50
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = paranoid }
		}
	}

	# Send him to the frontier
	option = {
		name = EVTOPTBtsar25
		event_target:tsar_consort = {
			opinion = { who = ROOT modifier = opinion_tsar_suit_refused }
		}
		prestige = -25
		ai_chance = { factor = 40 }
	}

	# Keep him close
	option = {
		name = EVTOPTCtsar25
		event_target:tsar_consort = { opinion = { who = ROOT modifier = opinion_tsar_favoured } }
		prestige = -50
		ai_chance = {
			factor = 20
			modifier = { factor = 3 is_lover = event_target:tsar_consort }
		}
	}
}

# The crown calls the Great Pulk (decision tsar_call_great_pulk). Send your

		FROM = { war = yes }
		prisoner = no

	option = {
		name = EVTOPTAtsar41

		prestige = 25
		if = {
			limit = { society_member_of = boyar_duma }
			change_society_currency = 100

							owner = FROM

							}
							earmark = tsar_great_pulk
							disband_on_peace = yes
							maintenance_multiplier = 0.5
						}
					}
				}
			}
		}
		else = {
			FROM = {
				capital_scope = {
					FROM = {
						spawn_unit = {
							province = PREV
							owner = FROM
							troops = {
								light_cavalry = { 150 150 }
								archers = { 100 100 }
							}
							earmark = tsar_great_pulk
							disband_on_peace = yes

# chekist.12 The Chekist turns
# The Chekist with his own web moves against a ruler who is not feared. Strike
# first (one time in three he escapes to another court with his secrets),
# or buy him off.
# Scope: ROOT keeps the Chekist with his own web.
character_event = {
	id = chekist.12
	desc = EVTDESCchekist12
	picture = GFX_evt_witch_hunters
	border = GFX_event_narrative_frame_intrigue

	is_triggered_only = yes # chekist.11

	trigger = {
		is_alive = yes
		any_courtier = {
			has_minor_title = title_court_chekist
			has_character_flag = chekist_own_web
			prisoner = no
		}
	}

	immediate = {
		random_courtier = {
			limit = {
				has_minor_title = title_court_chekist
				has_character_flag = chekist_own_web
			}
			save_event_target_as = chekist_officer
			clr_character_flag = chekist_own_web
		}
	}

	# Strike first
	option = {
		name = EVTOPTAchekist12
		random_list = {
			67 = {
				modifier = { factor = 1.5 intrigue = 14 }
				event_target:chekist_officer = { imprison = ROOT }
				prestige = 50
			}
			33 = {
				event_target:chekist_officer = {
					remove_title = title_court_chekist
					opinion = { who = ROOT modifier = opinion_chekist_tortured_kin }
				}
				# He flees to a rival court, or any other court, with his secrets
				random_independent_ruler = {
					limit = {
						NOT = { character = ROOT }
						prisoner = no
					}
					preferred_limit = { is_rival = ROOT }
					preferred_limit = { opinion = { who = ROOT value < 0 } }
					event_target:chekist_officer = { move_character = PREV }
				}
				any_vassal = { opinion = { who = ROOT modifier = opinion_chekist_fear } }
				prestige = -50
			}
		}
		ai_chance = { factor = 60 }
	}

	# Buy him off
	option = {
		name = EVTOPTBchekist12
		trigger = { wealth = 100 }
		wealth = -100
		ai_chance = { factor = 40 }
	}
}

# chekist.20 A house for the Chekist
# Like the Order of Sigmar, the Chekist wants a castle of its own: a senior
# Chekist at court asks for a castle barony of the ruler's lands. Grant it
# (he becomes a vassal; Chekist standing), or refuse. Once in ten years.
# Scope: ROOT is a Kislevite ruler with a free castle barony;
# event_target:chekist_officer the Chekist.
character_event = {
	id = chekist.20
	desc = EVTDESCchekist20
	picture = GFX_evt_castle_construction
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # on_five_year_pulse chekist_excess

	only_rulers = yes

	trigger = {
		higher_tier_than = COUNT
		NOT = { has_character_modifier = chekist_house_cd }
		any_courtier = {
			society_member_of = kislev_chekist
			society_rank >= 2
			is_adult = yes
			prisoner = no
		}
		any_demesne_title = {
			tier = BARON
			holding_type = castle
			NOT = { is_capital = yes }
		}
	}

	immediate = {
		add_character_modifier = { name = chekist_house_cd years = 10 hidden = yes }
		random_courtier = {
			limit = {
				society_member_of = kislev_chekist
				society_rank >= 2
				is_adult = yes
				prisoner = no
			}
			preferred_limit = { has_minor_title = title_court_chekist }
			save_event_target_as = chekist_officer
		}
		random_demesne_title = {
			limit = {
				tier = BARON
				holding_type = castle
				NOT = { is_capital = yes }
			}
			save_event_target_as = chekist_house
		}
	}

	# Grant it
	option = {
		name = EVTOPTAchekist20
		event_target:chekist_officer = {
			grant_title = event_target:chekist_house
			change_society_currency = 200
			opinion = { who = ROOT modifier = opinion_chekist_house_granted }
		}
		ai_chance = { factor = 50 }
	}

	# Refuse
	option = {
		name = EVTOPTBchekist20
		event_target:chekist_officer = { opinion = { who = ROOT modifier = opinion_chekist_house_refused } }
		ai_chance = { factor = 50 }
	}
}
