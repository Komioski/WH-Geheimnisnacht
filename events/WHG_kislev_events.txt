# kislev.1 The Sisterhood's missions
# Every two years each sister without a quest may be given one by a Grand Ice
# Witch: a proper focus, the hunt for a man who meddles with magic (Ice
# Maidens), or a new sister to find. The AI sometimes skips the mission and
# just gains or loses standing.
# Scope: ROOT is a member of the Ice Witch Sisterhood.
character_event = {
	id = kislev.1
	hide_window = yes

	is_in_society = yes
	only_capable = yes

	is_triggered_only = yes # on_society_bi_yearly_pulse

	trigger = {
		society_member_of = ice_witch_sisterhood
		has_any_quest = no
		NOT = { has_character_modifier = quest_cooldown_timer }
		is_inaccessible_trigger = no
		any_society_member = {
			is_society_grandmaster = yes
			same_society_as = ROOT
			NOT = { has_character_flag = own_mission_tombola }
		}
	}
	immediate = {
		kislev_save_matriarch_effect = yes
		random_list = {
			20 = {
				trigger = {
					has_dlc = "Way of Life"
					is_landed = yes
					NOR = {
						has_focus = focus_scholarship
						has_focus = focus_rulership
						has_focus = focus_intrigue
						has_focus = focus_war
					}
				}
				event_target:lodge_hero = { character_event = { id = kislev.25 } }
			}
			25 = {
				trigger = {
					is_landed = yes
					society_rank = 2
					age = 16
					kislev_male_mage_exists_trigger = yes
				}
				modifier = { factor = 2 ai = yes society_rank == 2 }
				event_target:lodge_hero = { character_event = { id = kislev.5 } }
			}
			60 = {
				trigger = { is_landed = yes is_adult = yes kislev_recruit_exists_trigger = yes }
				modifier = { factor = 5 ice_witch_sisterhood = { num_of_society_members < 25 } }
				modifier = { factor = 5 has_landed_title = d_frosthome_lodge }
				modifier = { factor = 0.5 ice_witch_sisterhood = { num_of_society_members = 30 } }
				modifier = { factor = 0 ice_witch_sisterhood = { num_of_society_members = 45 } }
				modifier = { factor = 2.5 has_landed_title = e_kislev }
				modifier = { factor = 10 any_child = { kislev_witch_recruit_trigger = yes } }
				modifier = {
					factor = 25
					ai = yes
					any_player = { kislev_witch_recruit_trigger = yes }
				}
				event_target:lodge_hero = { character_event = { id = kislev.10 } }
			}
			75 = {
				trigger = { ai = yes }
				modifier = { factor = 0 ice_witch_sisterhood = { num_of_society_members < 15 } }
				random_list = {
					70 = {
						modifier = { factor = 0.9 society_influence >= 30 }
						modifier = { factor = 0.9 society_influence >= 40 }
						modifier = { factor = 0.8 society_influence >= 50 }
						modifier = { factor = 0.7 society_influence >= 70 }
						modifier = { factor = 0.6 society_influence >= 80 }
						modifier = { factor = 0.5 society_influence >= 90 }
						add_mission_succeed_influence_effect = yes
					}
					30 = { add_mission_fail_influence_effect = yes }
				}
			}
		}
	}
}

# kislev.13 The Sisterhood calls
# The girl may join and begin her studies (one to five years, quicker for the
# clever or those who already know Ice magic), or refuse if she is grown.
# Girls of Frosthome Lodge vassals move to the sister's court.
# Scope: ROOT is the recruit; FROM the sister.
character_event = {
	id = kislev.13
	desc = EVTDESCkislev13
	picture = "GFX_evt_kislev_ice_witch"
	portrait = event_target:recruit_to_be

	is_triggered_only = yes

	option = {
		name = EVTOPTAkislev13
		#Join
		if = {
			limit = {
				is_landed = no
				event_target:recruiter = {
					OR = {
						has_landed_title = d_frosthome_lodge
						liege = { has_landed_title = d_frosthome_lodge }
					}
				}
			}
			move_character = event_target:recruiter
			set_character_flag = ai_flag_refuse_concubinage
			set_character_flag = no_court_invites
		}
		add_trait = studying_magic
		join_society = ice_witch_sisterhood
		if = { limit = { trait = lore_ice } character_event = { id = kislev.20 years = 1 } }
		else = {
			random_list = {
				20 = {
					trigger = { learning = 15 }
					modifier = { factor = 2 learning = 20 }
					modifier = { factor = 5 trait = genius }
					character_event = { id = kislev.20 days = 696 }
				}
				20 = {
					modifier = { factor = 2 learning = 12 }
					modifier = { factor = 2 learning = 15 }
					modifier = { factor = 2 trait = genius }
					character_event = { id = kislev.20 years = 3 }
				}
				15 = { character_event = { id = kislev.20 years = 4 } }
				15 = {
					trigger = { NOT = { learning = 8 } }
					modifier = { factor = 6 NOT = { learning = 6 } }
					character_event = { id = kislev.20 years = 5 }
				}
			}
		}
		FROM = { character_event = { id = kislev.14 } }
		ai_chance = { factor = 25 }
	}
	option = {
		name = EVTOPTBkislev13
		#Refuse
		trigger = { age = 18 }
		set_character_flag = refused_magical_recruitment
		prestige = -250
		FROM = { character_event = { id = kislev.15 } }
		ai_chance = {
			factor = 5
			modifier = { factor = 0.5 diplomacy = 12 }
			modifier = { factor = 0.5 learning = 12 }
			modifier = { factor = 0.5 diplomacy = 15 }
			modifier = { factor = 0.5 learning = 15 }
			modifier = { factor = 0 OR = { trait = ambitious trait = craven } }
		}
	}
}

# kislev.17 A lesson in caution
# Ice bites the careless. A rash or slow pupil has a small chance to die of
# the lesson; otherwise she gains Learning and Diplomacy, more surely with a
# teacher who has magic herself.
# Scope: ROOT is the pupil; FROM the sister.
character_event = {
	id = kislev.17
	desc = EVTDESCkislev17
	picture = "GFX_evt_kislev_ice_witch"

	is_triggered_only = yes

	trigger = {
		NOT = { trait = ice_witch }
		FROM = { is_alive = yes }
	}
	fail_trigger_effect = { kislev_end_lessons_effect = yes }

	option = {
		name = EVTOPTAkislev17
		if = {
			limit = {
				OR = {
					AND = { is_adult = no NOT = { learning = 14 } }
					AND = { is_adult = yes NOT = { learning = 16 } }
					trait = wroth
				}
			}
			kislev_lesson_accident_effect = yes
		}
		hidden_tooltip = {
			FROM = { character_event = { id = kislev.16 days = 250 random = 150 } }
			random_list = {
				50 = { change_learning = 1 change_diplomacy = 1 }
				50 = { modifier = { factor = 1.45 FROM = { has_magic_power_trigger = no } } }
			}
		}
	}
}

# A Gospodar ruler who holds land in Kislev decides that Gospodar, Ungol and
# Ropsmann should become one people. The ruler takes up the new Kislevite
# culture and spreads it to Kislev city or one Gospodar province. Once per
# ruler; a player who refuses may be asked again. After that the culture
# spreads by kislev.151 - 153.
# Scope: ROOT is a playable Gospodar ruler.
character_event = {
	id = kislev.150
	desc = EVTDESCkislev150
	picture = "GFX_evt_kislev_town"

	only_playable = yes
	culture = gospodar

	is_triggered_only = yes # five-year list kislev_melting_pot

	trigger = {
		should_evolve_cultures_trigger = yes
		NOT = { has_character_flag = kislev_melting_pot_begun }
		OR = {
			AND = {
				any_demesne_province = { kislev_melting_pot_province_trigger = yes }
				202 = { culture = kislevite } # c_kislev
			}
			202 = { # c_kislev
				owner = { character = ROOT }
				NOT = { culture = kislevite }
			}
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = {
			factor = 10
			202 = { owner = { character = ROOT } NOT = { culture = kislevite } }
		}
		modifier = { factor = 2 202 = { owner = { character = ROOT } culture = kislevite } }
		modifier = {
			factor = 0.67
			202 = { owner = { NOT = { character = ROOT } } culture = kislevite }
		}
		modifier = { factor = 1.33 learning = 7 }
		modifier = { factor = 1.33 learning = 9 }
		modifier = { factor = 1.33 learning = 11 }
		modifier = { factor = 1.33 learning = 13 }
		modifier = { factor = 0.67 NOT = { learning = 5 } }
		modifier = { factor = 0.67 NOT = { learning = 3 } }
		modifier = { factor = 0.5 NOT = { learning = 1 } }
		modifier = { factor = 3.3 kislev_faster_melting_pots_trigger = yes }
	}

	option = {
		name = EVTOPTAkislev150
		set_character_flag = kislev_melting_pot_begun
		culture = kislevite
		if = {
			limit = { 202 = { owner = { character = ROOT } NOT = { culture = kislevite } } }
			202 = { culture = kislevite }
		}
		else_if = {
			limit = {
				202 = {
					owner = { character = ROOT }
					any_neighbor_province = {
						owner = { same_realm = ROOT }
						kislev_melting_pot_spread_trigger = yes
					}
				}
			}
			202 = {
				random_neighbor_province = {
					limit = {
						owner = { same_realm = ROOT }
						kislev_melting_pot_spread_trigger = yes
					}
					culture = kislevite
				}
			}
		}
		else = {
			random_demesne_province = {
				limit = { kislev_melting_pot_province_trigger = yes }
				culture = kislevite
			}
		}
	}
	option = {
		name = EVTOPTBkislev150
		ai_chance = { factor = 0 }
	}
}

# kislevsoc.2 Recruits for the Circle and the Duma
# Every five years an AI warrior of Ursun's faith may join the Circle of the
# Bear, and an AI Kislevite lord may take a seat in the Boyar Duma.
# Scope: ROOT is any character (five-year pulse).
character_event = {
	id = kislevsoc.2
	hide_window = yes

	is_triggered_only = yes # on_five_year_pulse
	ai = yes
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		is_in_society = no
		OR = {
			religion = kislev_gods
			culture_group = kislevite_group
		}
	}

	immediate = {
		if = {
			limit = { can_join_society = circle_of_the_bear }
			random = {
				chance = 30
				modifier = { factor = 2 martial = 15 }
				join_society = circle_of_the_bear
			}
		}
		if = {
			limit = {
				is_in_society = no
				is_ruler = yes
				can_join_society = boyar_duma
			}
			random = {
				chance = 30
				modifier = { factor = 2 higher_tier_than = COUNT }
				join_society = boyar_duma
			}
		}
	}
}

# kislevsoc.3 Society life
# Every two years a member may meet one of their society's flavour events,
# and a Grand Ice Witch is called to the Ice Palace.
# Scope: ROOT is a society member.
character_event = {
	id = kislevsoc.3
	hide_window = yes

	is_triggered_only = yes # on_society_bi_yearly_pulse

	trigger = {
		OR = {
			society_member_of = circle_of_the_bear
			society_member_of = boyar_duma
			AND = {
				society_member_of = ice_witch_sisterhood
				society_rank == 4
			}
		}
		prisoner = no
		is_adult = yes
	}

	immediate = {
		if = {
			limit = { society_member_of = circle_of_the_bear }
			random_list = {
				100 = { }
				10 = { character_event = { id = kislevsoc.40 days = 30 } }
				10 = { character_event = { id = kislevsoc.41 days = 30 } }
				10 = {
					trigger = { has_character_flag = kislevsoc_dawn }
					character_event = { id = kislevsoc.42 days = 30 }
				}
			}
		}
		else_if = {
			limit = { society_member_of = boyar_duma }
			random_list = {
				100 = { }
				10 = { character_event = { id = kislevsoc.95 days = 30 } }
				10 = {
					trigger = { is_ruler = yes }
					character_event = { id = kislevsoc.96 days = 30 }
				}
				10 = { character_event = { id = kislevsoc.97 days = 30 } }
			}
		}
		else = {
			random = {
				chance = 40
				character_event = { id = kislevsoc.50 days = 30 }
			}
		}
	}
}
