# ursun.1 The omens
# Once a game. The High Speaker of Ursun, or the First of the Circle of the
# Bear, sees the signs that the Father of Bears lies wounded in the east: a
# bear walks into the hall and lies down, the stars burn over the High Pass.
# Heed them and send a company east (ursun.2), or dismiss them.
# Scope: ROOT is the High Speaker or the First of the Circle.
character_event = {
	id = ursun.1
	desc = EVTDESCursun1
	picture = GFX_evt_tamed_bear
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes # on_decade_pulse ursun_omens
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		ursun_quest_leader_trigger = yes
		NOT = { has_global_flag = ursun_wounded_bear_begun }
		NOT = { has_global_flag = ursun_wounded_bear_done }
	}

	immediate = {
		set_global_flag = ursun_wounded_bear_begun
	}

	# Heed the omens
	option = {
		name = EVTOPTAursun1
		piety = 50
		character_event = { id = ursun.2 days = 30 }
		ai_chance = { factor = 90 }
	}

	# Dismiss them
	option = {
		name = EVTOPTBursun1
		piety = -50
		clr_global_flag = ursun_wounded_bear_begun
		ai_chance = { factor = 10 }
	}
}

# ursun.2 The company rides east
# Choose who leads the company: the best rider of the Circle at court (or a
# champion found for the task), or the speaker in person.
# Scope: ROOT is the quest giver.
character_event = {
	id = ursun.2
	desc = EVTDESCursun2
	picture = GFX_evt_kislev_bear_cavalry
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes # ursun.1

	immediate = {
		set_variable = { which = ursun_losses value = 0 }
	}

	# Send the best rider of the Circle
	option = {
		name = EVTOPTAursun2
		trigger = { wealth = 100 }
		wealth = -100
		if = {
			limit = {
				any_courtier = {
					society_member_of = circle_of_the_bear
					is_adult = yes
					prisoner = no
					is_incapable = no
				}
			}
			random_courtier = {
				limit = {
					society_member_of = circle_of_the_bear
					is_adult = yes
					prisoner = no
					is_incapable = no
				}
				preferred_limit = { society_rank >= 3 }
				save_event_target_as = ursun_leader
			}
		}
		else = {
			create_character = {
				random_traits = yes
				dynasty = random
				religion = ROOT
				culture = ROOT
				female = no
				age = 30
				attributes = { martial = 12 }
				trait = creature_human
				trait = brave
				trait = zealous
			}
			new_character = {
				save_event_target_as = ursun_leader
			}
		}
		custom_tooltip = { text = ursun_company_rides_tt }
		character_event = { id = ursun.3 months = 6 }
		ai_chance = { factor = 70 }
	}

	# Lead it yourself
	option = {
		name = EVTOPTBursun2
		trigger = { is_incapable = no }
		save_event_target_as = ursun_leader
		set_character_flag = ursun_quest_self
		custom_tooltip = { text = ursun_company_rides_tt }
		character_event = { id = ursun.3 months = 6 }
		ai_chance = {
			factor = 30
			modifier = { factor = 2 trait = brave }
			modifier = { factor = 0 trait = craven }
		}
	}
}

# ursun.3 The edge of the Wastes
# A Chaos warband crosses the company's trail. Fight (the leader's martial),
# or hide in the snow (the leader's intrigue). A loss counts against the
# quest; two losses or a dead leader end it in the Long Sleep (ursun.8).
# Scope: ROOT is the quest giver; event_target:ursun_leader the leader.
character_event = {
	id = ursun.3
	desc = EVTDESCursun3
	picture = GFX_evt_duel_against_chaos
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # ursun.2

	# Fight
	option = {
		name = EVTOPTAursun3
		random_list = {
			60 = {
				modifier = { factor = 1.5 event_target:ursun_leader = { martial = 14 } }
				modifier = { factor = 1.3 event_target:ursun_leader = { trait = brave } }
				prestige = 50
				custom_tooltip = { text = ursun_encounter_won_tt }
			}
			40 = {
				change_variable = { which = ursun_losses value = 1 }
				event_target:ursun_leader = {
					random_list = {
						70 = { add_trait_wounded_effect = yes }
						30 = { death = { death_reason = death_battle } }
					}
				}
				custom_tooltip = { text = ursun_encounter_lost_tt }
			}
		}
		ai_chance = { factor = 60 }
	}

	# Hide in the snow
	option = {
		name = EVTOPTBursun3
		random_list = {
			60 = {
				modifier = { factor = 1.5 event_target:ursun_leader = { intrigue = 12 } }
				custom_tooltip = { text = ursun_encounter_won_tt }
			}
			40 = {
				change_variable = { which = ursun_losses value = 1 }
				custom_tooltip = { text = ursun_encounter_lost_tt }
			}
		}
		ai_chance = { factor = 40 }
	}

	after = {
		# On to the next stage, or the Long Sleep if the company is broken
		if = {
			limit = {
				event_target:ursun_leader = { is_alive = yes }
				NOT = { check_variable = { which = ursun_losses value = 2 } }
			}
			character_event = { id = ursun.4 months = 6 }
		}
		else = {
			character_event = { id = ursun.8 days = 30 }
		}
	}
}

# ursun.6 The resting place
# In a cave under the mountains the company finds a bear the size of a
# hill, breathing slow, with a black wound in its side. Offer blood (the
# leader's), offer a life (a prisoner), or take a tooth for a relic: the
# Tooth of Ursun, at the risk of waking the god in rage. Success brings
# Ursun Wakes to every lord of Kislev's gods; the relic gamble can bring
# the Long Sleep (ursun.8).
# Scope: ROOT is the quest giver; event_target:ursun_leader the leader.
character_event = {
	id = ursun.6
	desc = EVTDESCursun6
	picture = GFX_evt_tamed_bear
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes # ursun.5

	# Offer blood
	option = {
		name = EVTOPTAursun6
		event_target:ursun_leader = {
			add_trait_wounded_effect = yes
			piety = 100
		}
		ursun_wakes_effect = yes
		ai_chance = { factor = 50 }
	}

	# Offer a life
	option = {
		name = EVTOPTBursun6
		trigger = { any_prisoner = { always = yes } }
		random_prisoner = {
			death = { death_reason = death_sacrificed }
		}
		ursun_wakes_effect = yes
		ai_chance = {
			factor = 30
			modifier = { factor = 0.2 trait = kind }
		}
	}

	# Take a tooth
	option = {
		name = EVTOPTCursun6
		random_list = {
			60 = {
				modifier = { factor = 1.3 event_target:ursun_leader = { trait = brave } }
				if = {
					limit = { NOT = { has_global_flag = ursun_tooth_taken } }
					set_global_flag = ursun_tooth_taken
					add_artifact = ursun_tooth
				}
				ursun_wakes_effect = yes
			}
			40 = {
				event_target:ursun_leader = { death = { death_reason = death_accident_bear } }
				character_event = { id = ursun.8 }
			}
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 2 trait = ambitious }
		}
	}

	after = {
		clr_character_flag = ursun_quest_self
		set_global_flag = ursun_wounded_bear_done
	}
}

# ursun.20 The drunk boyar
# At the Waking Day feast a boyar, deep in his cups, mocks the Father of
# Bears in front of the whole court. Punish him, forgive him, or let him
# wrestle a bear and see what Ursun thinks.
# Scope: ROOT is a lord of Kislev's gods; event_target:ursun_mocker the boyar.
character_event = {
	id = ursun.20
	desc = EVTDESCursun20
	picture = GFX_evt_feast
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_yearly_pulse ursun_waking
	only_rulers = yes
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		ursun_kislev_lord_trigger = yes
		NOT = { has_character_modifier = ursun_waking_cd }
		OR = {
			any_vassal = { ursun_mocker_trigger = yes }
			any_courtier = { ursun_mocker_trigger = yes }
		}
	}

	immediate = {
		add_character_modifier = { name = ursun_waking_cd years = 5 hidden = yes }
		if = {
			limit = { any_vassal = { ursun_mocker_trigger = yes } }
			random_vassal = {
				limit = { ursun_mocker_trigger = yes }
				preferred_limit = { trait = drunkard }
				save_event_target_as = ursun_mocker
			}
		}
		else = {
			random_courtier = {
				limit = { ursun_mocker_trigger = yes }
				preferred_limit = { trait = drunkard }
				save_event_target_as = ursun_mocker
			}
		}
	}

	# Punish him
	option = {
		name = EVTOPTAursun20
		piety = 50
		event_target:ursun_mocker = {
			opinion = { who = ROOT modifier = opinion_ursun_punished years = 5 }
		}
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = zealous }
		}
	}

	# Forgive him
	option = {
		name = EVTOPTBursun20
		piety = -25
		event_target:ursun_mocker = {
			opinion = { who = ROOT modifier = opinion_ursun_forgiven years = 5 }
		}
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = kind }
		}
	}

	# Let him wrestle a bear
	option = {
		name = EVTOPTCursun20
		prestige = 25
		event_target:ursun_mocker = {
			random_list = {
				40 = { death = { death_reason = death_accident_bear } }
				35 = { add_trait_wounded_effect = yes }
				25 = {
					modifier = { factor = 2 martial = 14 }
					prestige = 100
					opinion = { who = ROOT modifier = opinion_ursun_champion years = 10 }
				}
			}
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 2 trait = cruel }
		}
	}
}

# ursun.31 The flame goes out
# The capital has fallen to the enemy and the hearth of Dazh in the temple
# is cold. Until it is relit, the realm has Cold Hearths (fewer children,
# poorer health, no birth rites, no northern blood). Send a priest of the
# Rota of the Dawn to carry the flame back, make the pilgrimage to the
# temple in Kislev, or leave it for now (a decision relights it later).
# Scope: ROOT is the lord.
character_event = {
	id = ursun.31
	desc = EVTDESCursun31
	picture = GFX_evt_stone_church_kislev_gods
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # ursun.30

	immediate = {
		add_character_modifier = { name = ursun_cold_hearths duration = -1 }
	}

	# A priest of the Rota of the Dawn
	option = {
		name = EVTOPTAursun31
		trigger = {
			OR = {
				any_courtier = { kislevsoc_dawn_trigger = yes }
				any_vassal = { kislevsoc_dawn_trigger = yes }
				kislevsoc_dawn_trigger = yes
			}
		}
		piety = -50
		character_event = { id = ursun.32 days = 60 }
		ai_chance = { factor = 70 }
	}

	# The pilgrimage to Kislev
	option = {
		name = EVTOPTBursun31
		trigger = { wealth = 75 }
		wealth = -75
		character_event = { id = ursun.32 days = 180 }
		ai_chance = { factor = 25 }
	}

	# Not now
	option = {
		name = EVTOPTCursun31
		custom_tooltip = { text = ursun_relight_later_tt }
		ai_chance = { factor = 5 }
	}
}

# ursun.40 Hidden. A war begins: a lord of Kislev's gods on either side may
# swear on Tor's thunder (ursun.41).
# Scope: ROOT is the defender, FROM the attacker.
character_event = {
	id = ursun.40
	hide_window = yes

	is_triggered_only = yes # on_war_started

	immediate = {
		if = {
			limit = { ursun_tor_oath_candidate_trigger = yes }
			character_event = { id = ursun.41 }
		}
		FROM = {
			if = {
				limit = { ursun_tor_oath_candidate_trigger = yes }
				character_event = { id = ursun.41 }
			}
		}
	}
}

# ursun.41 Tor's Oath
# The warriors gather before the war and the priest of Tor asks whether the
# lord will swear on the thunder: no flight and no surrender. One who keeps
# the oath to the end of the war gains martial; one who loses the war is
# struck down by the thunder (rarely) or is Oathless for life.
# Scope: ROOT is the lord.
character_event = {
	id = ursun.41
	desc = EVTDESCursun41
	picture = GFX_evt_kislev_warriors
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # ursun.40

	# Swear on the thunder
	option = {
		name = EVTOPTAursun41
		set_character_flag = ursun_tor_sworn
		add_character_modifier = { name = ursun_tor_oath duration = -1 }
		custom_tooltip = { text = ursun_tor_oath_tt }
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = brave }
			modifier = { factor = 1.5 trait = zealous }
			modifier = { factor = 0.2 trait = craven }
		}
	}

	# Do not swear
	option = {
		name = EVTOPTBursun41
		ai_chance = { factor = 60 }
	}
}

	id = ursun.46

		clr_character_flag = ursun_tor_sworn

# ursun.50 The barrow of the Khan-Queen
# A Gospodar lord reaches the barrow of Miska on the steppe. Pray at the
# tomb (a chance of her vision: the Heir of Miska for a woman, Miska's
# favour for a man), or dig for the grave goods (gold and a curse). Grave

character_event = {
	id = ursun.50
	desc = EVTDESCursun50
	picture = GFX_evt_steppe_mercenaries

	# Pray at the tomb
	option = {
		name = EVTOPTAursun50
		random_list = {
			15 = {
				modifier = { factor = 2 trait = zealous }
				modifier = { factor = 1.5 is_female = yes }
				if = {
					limit = { is_female = yes NOT = { trait = heir_of_miska } }
					add_trait = heir_of_miska
				}
				else = {
					add_character_modifier = { name = ursun_miska_favour years = 20 }
				}
				custom_tooltip = { text = ursun_miska_vision_tt }
			}
			85 = {
				piety = 75
				prestige = 50
			}
		}
		random = {
			chance = 30
			character_event = { id = ursun.51 days = 3 }
		}
		ai_chance = { factor = 80 }
	}

	# Dig for the grave goods
	option = {
		name = EVTOPTBursun50
		wealth = 150
		add_character_modifier = { name = ursun_miska_curse years = 10 }
		ai_chance = {
			factor = 20
			modifier = { factor = 3 trait = greedy }

# ursun.70 The priests and the mutant
# A Kislevite of the court has mutated despite the northern blood, and the
# priests of Ursun demand they be cast out. Cast them out of the realm,
# throw them in the dungeon, or protect them and anger the priests.
# Scope: ROOT is a lord of Kislev's gods; event_target:ursun_tainted the
# mutant.
character_event = {
	id = ursun.70
	desc = EVTDESCursun70
	picture = GFX_evt_stone_church_kislev_gods
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_five_year_pulse ursun_tainted
	only_rulers = yes
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		ursun_kislev_lord_trigger = yes
		any_courtier = { ursun_tainted_trigger = yes }
	}

	immediate = {
		random_courtier = {
			limit = { ursun_tainted_trigger = yes }
			save_event_target_as = ursun_tainted
		}
	}

	# Cast them out
	option = {
		name = EVTOPTAursun70
		piety = 50
		random_independent_ruler = {
			limit = {
				NOT = { religion = kislev_gods }
				NOT = { religion_group = chaos_gods_group }
				is_adult = yes
			}
			event_target:ursun_tainted = { move_character = PREV }
		}
		ai_chance = { factor = 50 }
	}

	# The dungeon
	option = {
		name = EVTOPTBursun70
		piety = 25
		event_target:ursun_tainted = { imprison = ROOT }
		ai_chance = { factor = 30 }
	}

	# Protect them
	option = {
		name = EVTOPTCursun70
		piety = -75
		event_target:ursun_tainted = {
			opinion = { who = ROOT modifier = opinion_ursun_protected years = 20 }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 3 is_close_relative = event_target:ursun_tainted }
		}
	}
}
