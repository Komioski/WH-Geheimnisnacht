# kislevnorth.1 A bear of Urskin's blood
# The Tzar's bear-master brings out a great bear of the line of Urskin, the
# bear that Tzar Boris rode. One per Tzar: it serves its rider and dies with
# him, or goes back to the wilds (kislevnorth.2).
# Scope: ROOT is the Tzar.
character_event = {
	id = kislevnorth.1
	desc = EVTDESCkislevnorth1
	picture = GFX_evt_kislev_bear_cavalry
	border = GFX_event_narrative_frame_war

	is_triggered_only = yes # decision north_raise_urskin_bear

	option = {
		name = EVTOPTAkislevnorth1
		set_character_flag = has_mount
		add_artifact = north_urskin_bear
		prestige = 100
	}
}

# kislevnorth.2 Hidden. The rider of Urskin's bear dies: the bear lies down
# on his grave and dies too, or goes back into the forests around Kislev
# (a province modifier there for ten years). It never passes to an heir.
# Scope: ROOT is the dead rider.
character_event = {
	id = kislevnorth.2
	hide_window = yes

	is_triggered_only = yes # on_death

	has_character_flag = north_urskin_raised

	trigger = {
		has_artifact = north_urskin_bear
	}

	immediate = {
		destroy_artifact = north_urskin_bear
		random_list = {
			50 = { }
			50 = {
				202 = { add_province_modifier = { name = north_urskin_loose years = 10 } }
			}
		}
	}
}

# kislevnorth.10 The caravan from the east
# A caravan comes down the High Pass from Cathay with jade, silk, spices and
# ivory. Escort it (gold, a chance of raiders, kislevnorth.11), tax it hard
# (more gold, a chance that Cathay's merchants shun the road, kislevnorth.12),
# or let it pass freely (Praag's trade grows for five years).
# Scope: ROOT holds Praag.
character_event = {
	id = kislevnorth.10
	desc = EVTDESCkislevnorth10
	picture = GFX_evt_market_altaic
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # on_five_year_pulse north_life
	only_rulers = yes
	prisoner = no

	trigger = {
		has_landed_title = c_praag
		NOT = { 182 = { has_province_modifier = north_cathay_shuns } }
	}

	# Escort it
	option = {
		name = EVTOPTAkislevnorth10
		wealth = 100
		random = {
			chance = 25
			character_event = { id = kislevnorth.11 days = 10 }
		}
		ai_chance = { factor = 50 }
	}

	# Tax it hard
	option = {
		name = EVTOPTBkislevnorth10
		wealth = 200
		random = {
			chance = 30
			character_event = { id = kislevnorth.12 days = 60 }
		}
		ai_chance = {
			factor = 30
			modifier = { factor = 2 trait = greedy }
		}
	}

	# Let it pass freely
	option = {
		name = EVTOPTCkislevnorth10
		prestige = 25
		182 = { add_province_modifier = { name = north_high_pass_trade years = 5 } }
		ai_chance = {
			factor = 20
			modifier = { factor = 2 trait = charitable }
		}
	}
}

# kislevnorth.40 Longships on the coast
# Norscan longships come in to the coast to raid. Fight them on the shore
# (martial; a captured jarl, kislevnorth.41, or a burned coast), pay them
# off, or ambush them at sea with Erengrad's ships.
# Scope: ROOT is a Kislevite lord with a coastal province in Kislev.
character_event = {
	id = kislevnorth.40
	desc = EVTDESCkislevnorth40
	picture = GFX_evt_vikings_arriving_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # on_decade_pulse north_norscan
	only_rulers = yes
	min_age = 16
	prisoner = no

	trigger = {
		north_norscan_target_trigger = yes
	}

	immediate = {
		random_demesne_province = {
			limit = {
				region = world_old_world_kislev
				port = yes
			}
			save_event_target_as = north_raided_coast
		}
	}

	# Fight on the shore
	option = {
		name = EVTOPTAkislevnorth40
		random_list = {
			50 = {
				modifier = { factor = 1.5 martial = 14 }
				modifier = { factor = 1.3 trait = brave }
				prestige = 75
				character_event = { id = kislevnorth.41 days = 3 }
			}
			50 = {
				event_target:north_raided_coast = {
					add_province_modifier = { name = north_norscan_raided years = 5 }
				}
			}
		}
		ai_chance = { factor = 50 }
	}

	# Pay them off
	option = {
		name = EVTOPTBkislevnorth40
		trigger = { wealth = 100 }
		wealth = -100
		prestige = -25
		ai_chance = { factor = 25 }
	}

	# Ambush them at sea
	option = {
		name = EVTOPTCkislevnorth40
		trigger = {
			OR = {
				has_landed_title = c_erengrad
				any_vassal = { has_landed_title = c_erengrad }
			}
		}
		random_list = {
			70 = {
				modifier = { factor = 1.3 martial = 12 }
				prestige = 100
				character_event = { id = kislevnorth.41 days = 3 }
			}
			30 = {
				event_target:north_raided_coast = {
					add_province_modifier = { name = north_norscan_raided years = 5 }
				}
			}
		}
		ai_chance = { factor = 25 }
	}
}

# or keep him as a hostage (no Norscan raids on this coast for ten years).

character_event = {
	id = kislevnorth.41
	desc = EVTDESCkislevnorth41
	picture = GFX_evt_viking_throneroom_oldgods
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # kislevnorth.40

	# Ransom him
	option = {
		name = EVTOPTAkislevnorth41
		wealth = 150
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = greedy }
		}
	}

	# Bring him to Ursun
	option = {
		name = EVTOPTBkislevnorth41
		trigger = { religion = kislev_gods }
		piety = 50
		create_character = {
			random_traits = yes
			dynasty = random
			religion = kislev_gods
			culture = baersonling
			female = no
			age = 35
			attributes = { martial = 12 }
			trait = creature_human
			trait = brave
		}
		ai_chance = { factor = 30 }
	}

	# Keep him as a hostage
	option = {
		name = EVTOPTCkislevnorth41
		add_character_modifier = { name = north_norscan_truce years = 10 }
		ai_chance = { factor = 30 }
	}
}

# kislevnorth.50 Trolls in the thaw
# In the far north the trolls come down with the thaw. Hunt it (martial;
# a rare troll-hide cloak, once a lifetime; or a wound, and the troll heals
# and comes back, kislevnorth.51), pay troll-hunters, or bar the gates.
# Scope: ROOT is a lord with land in northern Kislev.
character_event = {
	id = kislevnorth.50
	desc = EVTDESCkislevnorth50
	picture = GFX_evt_troll_country
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # on_five_year_pulse north_life
	only_rulers = yes
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		north_troll_country_trigger = yes
	}

	immediate = {
		random_demesne_province = {
			limit = { region = world_old_world_kislev_northern }
			save_event_target_as = north_troll_province
		}
	}

	# Hunt it
	option = {
		name = EVTOPTAkislevnorth50
		random_list = {
			50 = {
				modifier = { factor = 1.5 martial = 14 }
				modifier = { factor = 1.3 combat_rating = 40 }
				modifier = { factor = 1.3 society_member_of = circle_of_the_bear }
				prestige = 75
				north_troll_hide_effect = yes
			}
			30 = {
				add_trait_wounded_effect = yes
			}
			20 = {
				custom_tooltip = { text = north_troll_heals_tt }
				character_event = { id = kislevnorth.51 years = 1 }
			}
		}
		ai_chance = {
			factor = 50
			modifier = { factor = 2 trait = brave }
			modifier = { factor = 0.2 trait = craven }
		}
	}

	# Pay troll-hunters
	option = {
		name = EVTOPTBkislevnorth50
		trigger = { wealth = 50 }
		wealth = -50
		ai_chance = { factor = 30 }
	}

	# Bar the gates
	option = {
		name = EVTOPTCkislevnorth50
		event_target:north_troll_province = {
			add_province_modifier = { name = north_troll_loose years = 2 }
		}
		ai_chance = { factor = 20 }
	}
}

# kislevnorth.60 Hidden. Kislev builds tall
# Once an AI Tzar(ina) holds all of Kislev's de jure lands, the crown turns
# from conquest to defence: a hidden trait lowers ambition and raises zeal,
# so the AI weakens Chaos and the beasts on its borders and builds at home
# rather than marching on its neighbours. Removed again if the realm loses
# de jure land.
# Scope: ROOT is an AI holder of e_kislev.
character_event = {
	id = kislevnorth.60
	hide_window = yes

	is_triggered_only = yes # on_five_year_pulse
	ai = yes
	only_rulers = yes

	trigger = {
		OR = {
			has_landed_title = e_kislev
			trait = kislev_tall_ai
		}
	}

	immediate = {
		if = {
			limit = {
				has_landed_title = e_kislev
				completely_controls = e_kislev
			}
			if = {
				limit = { NOT = { trait = kislev_tall_ai } }
				add_trait = kislev_tall_ai
			}
		}
		else = {
			remove_trait = kislev_tall_ai
		}
	}
}
