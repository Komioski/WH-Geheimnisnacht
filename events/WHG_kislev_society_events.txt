# kislevsoc.2 Recruits for the Circle and the Duma
# Every five years an AI warrior of Ursun's faith may join the Circle of the
# Bear, and an AI Kislevite lord may take a seat in the Boyar Duma.
# Scope: ROOT is any character (five-year pulse).
character_event = {
	id = kislevsoc.2
	hide_window = yes

	is_triggered_only = yes # on_five_year_pulse
	ai = yes
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		is_in_society = no
		OR = {
			religion = kislev_gods
			culture_group = kislevite_group
		}
	}

	immediate = {
		if = {
			limit = { can_join_society = circle_of_the_bear }
			random = {
				chance = 30
				modifier = { factor = 2 martial = 15 }
				join_society = circle_of_the_bear
			}
		}
		if = {
			limit = {
				is_in_society = no
				is_ruler = yes
				can_join_society = boyar_duma
			}
			random = {
				chance = 30
				modifier = { factor = 2 higher_tier_than = COUNT }
				join_society = boyar_duma
			}
		}
	}
}

# kislevsoc.3 Society life
# Every two years a member may meet one of their society's flavour events,
# and a Grand Ice Witch is called to the Ice Palace.
# Scope: ROOT is a society member.
character_event = {
	id = kislevsoc.3
	hide_window = yes

	is_triggered_only = yes # on_society_bi_yearly_pulse

	trigger = {
		OR = {
			society_member_of = circle_of_the_bear
			society_member_of = boyar_duma
			AND = {
				society_member_of = ice_witch_sisterhood
				society_rank == 4
			}
		}
		prisoner = no
		is_adult = yes
	}

	immediate = {
		if = {
			limit = { society_member_of = circle_of_the_bear }
			random_list = {
				100 = { }
				10 = { character_event = { id = kislevsoc.40 days = 30 } }
				10 = { character_event = { id = kislevsoc.41 days = 30 } }
				10 = {
					trigger = { has_character_flag = kislevsoc_dawn }
					character_event = { id = kislevsoc.42 days = 30 }
				}
			}
		}
		else_if = {
			limit = { society_member_of = boyar_duma }
			random_list = {
				100 = { }
				10 = { character_event = { id = kislevsoc.95 days = 30 } }
				10 = {
					trigger = { is_ruler = yes }
					character_event = { id = kislevsoc.96 days = 30 }
				}
				10 = { character_event = { id = kislevsoc.97 days = 30 } }
			}
		}
		else = {
			random = {
				chance = 40
				character_event = { id = kislevsoc.50 days = 30 }
			}
		}
	}
}

# kislevsoc.50 The palace weeps
# A Grand Ice Witch is called to keep the Ice Palace whole for the crown.
# Do the work (it costs her standing, wins the crown's thanks), or leave it
# dripping. A Tzarina who is herself an Ice Witch needs no help; a ruler who
# moved to the Bokha Palace or let it melt does not ask (tsar.10).
# Scope: ROOT is a rank-4 Ice Witch.
character_event = {
	id = kislevsoc.50
	desc = EVTDESCkislevsoc50
	picture = GFX_evt_kislev_ice_witch_casting
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # kislevsoc.3

	trigger = {
		NOT = { has_landed_title = e_kislev }
		NOT = { has_global_flag = ice_palace_melted }
		e_kislev = {
			has_holder = yes
			holder_scope = {
				NOT = { trait = ice_witch }
				has_character_modifier = tsar_ice_palace_upkeep # tsar.10: the crown begged the Sisterhood
			}
		}
	}

	# Keep the palace whole
	option = {
		name = EVTOPTAkislevsoc50
		change_society_currency = -200
		add_character_modifier = { name = kislevsoc_palace_keeper years = 5 }
		e_kislev = {
			holder_scope = { opinion = { who = ROOT modifier = opinion_kislevsoc_palace_kept } }
		}
		ai_chance = { factor = 80 }
	}

	# Let it drip
	option = {
		name = EVTOPTBkislevsoc50
		change_society_currency = 50
		e_kislev = {
			holder_scope = { opinion = { who = ROOT modifier = opinion_kislevsoc_palace_neglected } }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 5 e_kislev = { holder_scope = { is_rival = ROOT } } }
		}
	}
}
