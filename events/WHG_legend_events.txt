# Legends of the Old World: the Herrimault and the burning rift
# Namespace: legend (was 10000 - 10109 in the vanilla legends_events.txt)
# Faction: Generic
# Serves: player rulers of the Old World faiths (Part 1), every player
#   ruler (Part 2)
#
# Part 1: the Herrimault               legend.1 - legend.30
#   An outlaw of the greenwood robs the ruler's tax carts and feeds the
#   poor. The ruler may trap him at an archery contest, go into the woods
#   in disguise, ride with his band, hang him or take him into service.
#   (was the Robin Hood chain, 10000 - 10042)
# Part 2: the burning rift            legend.100 - legend.109
#   A chasm opens in the ruler's lands and breathes fire and screams. It
#   can be ignored for a while, then must be closed by prayer, by feeding
#   it, by filling it with stones or by a sacrifice.
#   (was the Gates of Hell chain, 10100 - 10109)
#
# Fired from: the on_five_year_pulse random_events list legends in
#   WHG_pulse_on_actions.txt (1, 100). Everything else follows from those.
# Saved: event targets legend_outlaw, legend_sweetheart, legend_sacrifice;
#   character flags legend_outlaw_hunt (the chain runs), legend_outlaw_met,
#   legend_outlaw_dead, legend_outlaw_wed, legend_rift_<way>; global flag
#   robin_hood_existed (once per game).
# Scripted triggers and effects: WHG_legend_triggers.txt and
#   WHG_legend_effects.txt in common/scripted_triggers and scripted_effects.
# Loc: localisation/00_WHG_legend_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

# legend.1 The Herrimault
# An outlaw in the ruler's woods robs tax carts and gives the coin to the
# poor, and the peasants sing of him. Wish him well (prestige; a noble may
# come to complain, legend.2), swear to catch him (a chance of diligent or
# deceitful; a noble complains), or let it be (a chance of content). Either
# way the ruler's council brings it up within the year (legend.4).
# Once per game. Players of the Old World faiths with bandits or unrest in
# their lands.
# Scope: ROOT is the ruler.
narrative_event = {
	id = legend.1
	title = EVTNAMElegend1
	desc = EVTDESClegend1
	picture = GFX_evt_bandits

	is_triggered_only = yes # on_five_year_pulse random list legends
	ai = no
	religion_group = old_world_group
	min_age = 20
	prisoner = no
	capable_only = yes
	war = no

	trigger = {
		NOT = { has_global_flag = robin_hood_existed }
		in_command = no
		is_ill = no
		is_inaccessible_trigger = no
		any_demesne_province = {
			OR = { has_province_modifier = highway_robber_band revolt_risk = 0.05 }
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 0.8 trait = just }
		modifier = { factor = 5 trait = arbitrary }
		modifier = { factor = 0.8 trait = charismatic_negotiator }
		modifier = { factor = 3 trait = midas_touched }
	}

	immediate = {
		set_global_flag = robin_hood_existed
		set_character_flag = legend_outlaw_hunt
	}

	option = { # Wish him well
		name = EVTOPTAlegend1
		prestige = 10
		random = {
			chance = 60
			narrative_event = { id = legend.2 days = 5 random = 10 tooltip = EVTTOOLTIPlegend2 }
		}
		hidden_effect = { narrative_event = { id = legend.4 days = 300 random = 120 } }
	}
	option = { # I will catch him
		name = EVTOPTBlegend1
		if = {
			limit = { NOT = { trait = diligent } }
			random = { chance = 25 add_trait = diligent }
		}
		else_if = {
			limit = { NOT = { trait = deceitful } }
			random = { chance = 25 add_trait = deceitful }
		}
		random = {
			chance = 40
			narrative_event = { id = legend.2 days = 5 random = 10 tooltip = EVTTOOLTIPlegend2 }
		}
		hidden_effect = { narrative_event = { id = legend.4 days = 200 random = 120 } }
	}
	option = { # Let it be
		name = EVTOPTClegend1
		if = { limit = { NOT = { trait = content } } random = { chance = 25 add_trait = content } }
		hidden_effect = { narrative_event = { id = legend.4 days = 365 random = 180 } }
	}
}

# legend.2 A noble complains
# A noble of the court was robbed in the woods (a lady of her jewels or a
# lord of his purse and his dignity). Pay them off, or promise that
# something will be done (a chance of diligent or ambitious).
# Fixed: the old pair of events (one per sex) used break = yes, and the
# lord's version could never give ambitious after diligent.
# Scope: ROOT is the ruler, event_target:legend_complainer the noble.
narrative_event = {
	id = legend.2
	title = EVTNAMElegend2
	desc = {
		trigger = { event_target:legend_complainer = { is_woman_trigger = yes } }
		text = EVTDESClegend2
	}
	desc = {
		trigger = { event_target:legend_complainer = { is_woman_trigger = no } }
		text = EVTDESClegend2b
	}
	picture = GFX_evt_bandits

	is_triggered_only = yes # legend.1

	trigger = {
		any_courtier = { legend_complainer_trigger = yes }
	}

	immediate = {
		random_courtier = {
			limit = { legend_complainer_trigger = yes }
			save_event_target_as = legend_complainer
		}
	}

	option = { # Pay them off
		name = EVTOPTAlegend2
		scaled_wealth = -0.02
		set_character_flag = legend_outlaw_ignored
	}
	option = { # Something will be done
		name = EVTOPTBlegend2
		set_character_flag = legend_outlaw_angered
		if = {
			limit = { NOT = { trait = diligent } }
			random = { chance = 25 add_trait = diligent }
		}
		else_if = {
			limit = { NOT = { trait = ambitious } }
			random = { chance = 25 add_trait = ambitious }
		}
	}
}

# legend.4 A trap for the outlaw
# The council asks what to do about the Herrimault. Leave it (prestige, and
# a craven or slothful streak), hold an archery contest no outlaw could
# resist (legend.5), or go into the woods in disguise (legend.10).
# Was MTTH 12 months on a flag; now scheduled by legend.1.
# Scope: ROOT is the ruler.
narrative_event = {
	id = legend.4
	title = EVTNAMElegend4
	desc = { trigger = { has_character_flag = legend_outlaw_angered } text = EVTDESClegend4b }
	desc = {
		trigger = { NOT = { has_character_flag = legend_outlaw_angered } }
		text = EVTDESClegend4
	}
	picture = GFX_evt_council

	is_triggered_only = yes # legend.1
	has_character_flag = legend_outlaw_hunt

	trigger = {
		is_alive = yes
		prisoner = no
		is_incapable = no
	}

	option = { # Too risky
		name = EVTOPTAlegend4
		prestige = -10
		if = { limit = { NOR = { trait = craven trait = berserker } } add_trait = craven }
		else_if = { limit = { NOT = { trait = slothful } } add_trait = slothful }
		legend_outlaw_end_effect = yes
	}
	option = { # An archery contest
		name = EVTOPTBlegend4
		random_courtier = {
			limit = { is_adult = yes }
			opinion = { modifier = clever_deceive who = ROOT years = 5 }
		}
		narrative_event = { id = legend.5 days = 5 random = 10 tooltip = EVTTOOLTIPlegend5 }
	}
	option = { # Into the woods in disguise
		name = EVTOPTClegend4
		prestige = 10
		if = { limit = { NOT = { trait = deceitful } } add_trait = deceitful }
		else_if = { limit = { NOT = { trait = patient } } add_trait = patient }
		narrative_event = { id = legend.10 days = 1 random = 3 tooltip = EVTTOOLTIPlegend10 }
	}
}

# legend.100 A rift in the earth
# A chasm opens in the ruler's lands, and fiery stones fly from it. Ignore
# it (a chance of slothful, or content for the already slothful) or go and
# look (legend.101 soon). Ignored, it gets worse within a year or so.
# Was MTTH 10000 months on every player ruler.
# Scope: ROOT is the ruler.
narrative_event = {
	id = legend.100
	title = EVTNAMElegend100
	desc = EVTDESClegend100
	picture = GFX_evt_pagan

	is_triggered_only = yes # on_five_year_pulse random list legends
	only_playable = yes
	min_age = 20
	max_age = 100
	capable_only = yes
	prisoner = no
	war = no

		in_command = no
		primary_title = { temporary = no }
		is_inaccessible_trigger = no
		legend_rift_open_trigger = no
		NOR = {
			has_character_modifier = exorcism_gateshell
			has_character_modifier = cattle_gateshell
			has_character_modifier = stones_gateshell
			has_character_modifier = sacrifice_gateshell
