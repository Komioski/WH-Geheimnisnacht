###############################################################################
# Lifestyles: how a grown character finds a calling
# Namespace: lifestyle (was 5000 - 5082 in lifestyle_event.txt)
# Faction: Generic
#
# Short chains that end in a lifestyle trait: scholar or mystic, gardener or
# impaler, duelist or poet, hunter or falconer, poet or falconer, celibate or
# hedonist, hedonist or poet in Araby, and the Ormazdic Jurist (faqih).
#
# Vanilla switched these chains off when Way of Life was installed. They now
# run beside the focus events: a hidden selector on the five-year pulse
# (lifestyle.1) offers one chain to a character of 25 or more who has no
# lifestyle trait yet, and friends can talk a character into one
# (friend.50 - friend.55, traits_effects_events.txt). The flag LE_ST
# marks a chain in progress; every ending clears it.
#
# Warhammer rules: the Old World faiths argue restraint against pleasure
# (and the priests mutter about the Dark Prince); Araby argues pleasure
# against poetry, and may train Jurists of Ormazd; the faithful of Shallya do
# not hunt; the subjects of a Druchii, greenskin, Chaos, Skaven or undead
# impaler cheer the stakes instead of pelting their lord.
#
# Selector                          lifestyle.1
# Scholar or mystic                 lifestyle.10 - lifestyle.13
# Gardener or impaler               lifestyle.20 - lifestyle.25
# Duelist or poet (courtship)       lifestyle.30 - lifestyle.37
# Hunter or falconer                lifestyle.40 - lifestyle.46
# Poet or falconer                  lifestyle.50 - lifestyle.56
# Celibate or hedonist              lifestyle.60 - lifestyle.66
# Araby: hedonist or poet           lifestyle.70 - lifestyle.74
# Araby: the Ormazdic Jurist        lifestyle.80 - lifestyle.82
#
# Fired from: on_five_year_pulse, lifestyle_start list (00_on_actions.txt);
#   friend.50 - friend.55 and traits_effects_events.txt.
###############################################################################

namespace = lifestyle

### Selector ###

# lifestyle.1 A calling
# Picks one lifestyle chain that fits the character. The old MTTH modifiers
# are kept as weights.
# Scope: ROOT is the character.
character_event = {
	id = lifestyle.1
	hide_window = yes
	is_triggered_only = yes # on_five_year_pulse, lifestyle_start list (00_on_actions.txt)

	min_age = 25
	only_capable = yes
	prisoner = no

	trigger = {
		is_living_human_trigger = yes
		OR = {
			NOT = { has_character_flag = LE_ST }
			had_character_flag = { flag = LE_ST days = 3650 }
		}
		in_command = no
		is_inaccessible_trigger = no
	}

	immediate = {
		if = { # A chain that stalled ten years ago is dropped
			limit = { has_character_flag = LE_ST }
			lifestyle_end_effect = yes
			lifestyle_clear_chain_effect = yes
		}
		else = { lifestyle_pick_chain_effect = yes }
	}
}
