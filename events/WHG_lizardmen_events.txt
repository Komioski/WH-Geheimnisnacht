# lizardmen.1 Return of the Chameleon Skinks
# Decade pulse when doom is high: Oxyotl returns and the Chameleon Skinks
# order is restored.
# Scope: ROOT is an Old Ones ruler.
narrative_event = {
	id = lizardmen.1
	title = "EVTTITLElizardmen1"
	desc = "EVTDESClizardmen1"
	picture = "GFX_evt_crusaders"
	border = "GFX_event_narrative_frame_religion"
	major = yes

	religion = old_ones

	is_triggered_only = yes

	trigger = {
		year > 2055
		is_doom_high_trigger = yes
		is_heretic = no
		NOR = { is_title_active = d_chameleon_skinks has_global_flag = chameleon_skinks_return }
	}

	immediate = {
		activate_title = { title = d_chameleon_skinks status = yes }
		create_random_soldier = {
			dynasty = none
			random_traits = yes
			religion = ROOT
			name = "Oxyotl"
		}
		new_character = {
			add_trait = creature_skink
			add_trait = shrewd
			add_trait = brave
			add_trait = trickster
			d_chameleon_skinks = { grant_title = PREV }
			set_defacto_liege = THIS
			wealth = 600
			piety = 500
			prestige = 1000
		}
		religion_authority = { modifier = religious_order_formed years = 10 }
		set_global_flag = chameleon_skinks_return
	}

	option = {
		name = "GodsBePraised"
		trigger = { religion = ROOT }
	}
	option = {
		name = "OK"
		trigger = { NOT = { religion = ROOT } }
	}
}

# lizardmen.2 A blessed Skink
# Decade pulse: a Skink Priest is born from the realm's spawning pools (more
# likely when the court has few).
# Scope: ROOT is an Old Ones ruler with a spawning pool.
character_event = {
	id = lizardmen.2
	title = EVTTITLElizardmen2
	desc = EVTDESClizardmen2
	picture = GFX_evt_lustria_jungle

	religion = old_ones

	is_triggered_only = yes # on_decade_pulse (list lustria)

	trigger = {
		higher_tier_than = COUNT
		NOT = { any_courtier = { trait = skink_priest count = 6 } }
		any_realm_province = {
			any_province_holding = {
				OR = {
					has_building = ct_spawning_pool
					has_building = tp_spawning_pool
					has_building = ca_spawning_pool
				}
				count = 1
			}
		}
	}

	weight_multiplier = {
		factor = 1
		# Courts without priests need them most
		modifier = { factor = 2 NOT = { any_courtier = { trait = skink_priest } } }
		modifier = { factor = 0.5 any_courtier = { trait = skink_priest count = 3 } }
		# More spawning pools, more chances
		modifier = {
			factor = 1.5
			any_realm_province = {
				any_province_holding = {
					OR = {
						has_building = ct_spawning_pool
						has_building = tp_spawning_pool
						has_building = ca_spawning_pool
					}
					count = 2
				}
			}
		}
		modifier = {
			factor = 1.5
			any_realm_province = {
				any_province_holding = {
					OR = {
						has_building = ct_spawning_pool
						has_building = tp_spawning_pool
						has_building = ca_spawning_pool
					}
					count = 4
				}
			}
		}
		# A Skink mage on the throne tends the pools himself
		modifier = {
			factor = 1.5
			trait = creature_skink
			OR = { trait = lore_heavens trait = lore_beasts }
		}
	}

	option = {
		name = EVTOPTAlizardmen2
		piety = 50
		hidden_tooltip = {
			random_list = {
				40 = {
					create_character = {
						random_traits = yes
						attributes = { learning = 9 }
						dynasty = RANDOM
						religion = old_ones
						culture = lizardman
						female = no
						age = 16
					}
				}
				35 = {
					create_character = {
						random_traits = yes
						attributes = { learning = 11 }
						dynasty = RANDOM
						religion = old_ones
						culture = lizardman
						female = no
						age = 16
					}
				}
				15 = {
					create_character = {
						random_traits = yes
						attributes = { learning = 13 }
						dynasty = RANDOM
						religion = old_ones
						culture = lizardman
						female = no
						age = 16
					}
				}
				10 = {
					create_character = {
						random_traits = yes
						attributes = { learning = 15 }
						dynasty = RANDOM
						religion = old_ones
						culture = lizardman
						female = no
						age = 16
					}
				}
			}
			# Give Magic
			new_character = {
				piety = 20
				prestige = 50
				add_trait = creature_skink
				add_trait = skink_priest
				add_trait = zealous
				add_trait = bright
				set_character_flag = spawned_skink_priest
				random_list = {
					65 = { add_trait = lore_heavens }
					35 = { add_trait = lore_beasts }
				}
				magic_power_random_effect = yes
			}
		}
	}
}

# lizardmen.7 Skink succession (choose)
# Hidden. Picks the Skink heir: the title_skink_heir holder, or a Skink with
# magic.
# Scope: ROOT is the new holder.
character_event = {
	id = lizardmen.7
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		# A Skink with magic, if the court has one
		if = {
			limit = {
				OR = {
					any_vassal = {
						trait = creature_skink
						OR = {
							has_character_modifier = magic_power_1
							has_character_modifier = magic_power_2
							has_character_modifier = magic_power_3
							has_character_modifier = magic_power_4
						}
					}
					any_courtier = {
						trait = creature_skink
						OR = {
							has_character_modifier = magic_power_1
							has_character_modifier = magic_power_2
							has_character_modifier = magic_power_3
							has_character_modifier = magic_power_4
						}
					}
				}
			}
			set_character_flag = skink_mage_available
		}
		# The named heir inherits if gifted, or if no gifted Skink exists
		if = {
			limit = {
				OR = {
					any_vassal = { has_minor_title = title_skink_heir }
					any_courtier = { has_minor_title = title_skink_heir }
				}
				OR = {
					NOT = { has_character_flag = skink_mage_available }
					any_vassal = {
						has_minor_title = title_skink_heir
						OR = {
							has_character_modifier = magic_power_1
							has_character_modifier = magic_power_2
							has_character_modifier = magic_power_3
							has_character_modifier = magic_power_4
						}
					}
					any_courtier = {
						has_minor_title = title_skink_heir
						OR = {
							has_character_modifier = magic_power_1
							has_character_modifier = magic_power_2
							has_character_modifier = magic_power_3
							has_character_modifier = magic_power_4
						}
					}
				}
			}
			if = {
				limit = { any_vassal = { has_minor_title = title_skink_heir } }
				random_vassal = {
					limit = { has_minor_title = title_skink_heir }
					character_event = { id = lizardmen.8 }
				}
			}
			else = {
				random_courtier = {
					limit = { has_minor_title = title_skink_heir }
					character_event = { id = lizardmen.8 }
				}
			}
		}
		# Otherwise a gifted Skink takes over
		else_if = {
			limit = { has_character_flag = skink_mage_available }
			if = {
				limit = {
					any_vassal = {
						trait = creature_skink
						OR = {
							has_character_modifier = magic_power_1
							has_character_modifier = magic_power_2
							has_character_modifier = magic_power_3
							has_character_modifier = magic_power_4
						}
					}
				}
				random_vassal = {
					limit = {
						trait = creature_skink
						OR = {
							has_character_modifier = magic_power_1
							has_character_modifier = magic_power_2
							has_character_modifier = magic_power_3
							has_character_modifier = magic_power_4
						}
					}
					character_event = { id = lizardmen.8 }
				}
			}
			else = {
				random_courtier = {
					limit = {
						trait = creature_skink
						OR = {
							has_character_modifier = magic_power_1
							has_character_modifier = magic_power_2
							has_character_modifier = magic_power_3
							has_character_modifier = magic_power_4
						}
					}
					character_event = { id = lizardmen.8 }
				}
			}
		}
		clr_character_flag = skink_mage_available
	}
}

# lizardmen.11 The soothing Great Plan
# Good contemplation: less stress, healing, patience or diligence.
# Scope: ROOT is a Slann.
character_event = {
	id = lizardmen.11
	desc = EVTDESClizardmen11
	picture = "GFX_evt_lustria_slann"

	is_triggered_only = yes

	trigger = {
		trait = creature_slann
	}

	option = {
		name = EVTOPTAlizardmen11
		random_list = {
			# Lunacy
			15 = {
				modifier = { factor = 2 trait = stressed }
				modifier = { factor = 2 trait = depressed }
				modifier = { factor = 2 trait = lunatic_3 }
				modifier = {
					factor = 0
					NOR = { trait = stressed trait = depressed trait = lunatic_3 }
				}
				if = {
					limit = { trait = stressed trait = depressed trait = lunatic_3 }
					change_variable = { which = "lunatic_counter" value = -15 }
				}
				else_if = {
					limit = { trait = stressed trait = depressed NOT = { trait = lunatic_3 } }
					remove_trait = depressed
				}
				else_if = {
					limit = { trait = stressed NOR = { trait = depressed trait = lunatic_3 } }
					remove_trait = stressed
				}
				# Fallback
				else = { piety = -25 prestige = -25 }
			}
			# Healing
			15 = { modifier = { factor = 0 NOT = { is_ill = yes } } healing_effect = yes }
			# Slothful
			15 = { modifier = { factor = 0 NOT = { trait = slothful } } remove_trait = slothful }
			# Patience
			15 = { modifier = { factor = 0 trait = patient } add_trait = patient }
			# Diligence
			15 = { modifier = { factor = 0 trait = diligent } add_trait = diligent }
			# General boost
			25 = { piety = 50 prestige = 50 }
		}
	}
}

	desc = EVTDESClizardmen12
	picture = GFX_evt_lustria_slann

	is_triggered_only = yes

	trigger = {
		trait = creature_slann
	}

	option = {
		name = EVTOPTAlizardmen12
		random_list = {
			# Lunacy
			15 = {
				modifier = { factor = 2 NOT = { trait = stressed } }
				modifier = { factor = 2 NOT = { trait = depressed } }
				modifier = { factor = 2 NOT = { trait = lunatic_3 } }
				modifier = { factor = 0 trait = lunatic_3 }
				if = {
					limit = { trait = stressed trait = depressed NOT = { trait = lunatic_3 } }
					change_variable = { which = "lunatic_counter" value = 5 }
					hidden_tooltip = {
						# lunacy Ping
						character_event = { id = mutate.5 }
					}
				}
				else_if = {
					limit = { trait = stressed NOR = { trait = depressed trait = lunatic_3 } }
					add_trait = depressed
				}
				else = { add_trait = stressed }
			}
			# Illness
			15 = {
				modifier = { factor = 0 is_ill = yes }
				if = { limit = { NOT = { has_dlc = "Reapers" } } add_trait = ill }
				if = { limit = { has_dlc = "Reapers" } add_symptom_effect = yes }
			}
			# Slothful
			15 = { modifier = { factor = 0 trait = slothful } add_trait = slothful }
			# Patience
			15 = { modifier = { factor = 0 NOT = { trait = patient } } remove_trait = patient }
			# Diligence
			15 = { modifier = { factor = 0 NOT = { trait = diligent } } remove_trait = diligent }
			# General malus
			25 = {
				modifier = { factor = 0 NOR = { piety > 50 prestige > 50 } }
				piety = -50
				prestige = -50
			}
		}
	}
}

	desc = EVTDESClizardmen15

	}

	immediate = {
		FROM = {
			diplomatic_immunity = yes
			any_faction_backer = { faction = faction_misaligned_chief diplomatic_immunity = yes }
		}
	}
	# Submit
	option = {
		name = EVTOPTAlizardmen15
		ai_chance = {
			factor = 60
			modifier = {
				factor = 0.15
				NOT = {
					FROM = { faction_power = { faction = faction_misaligned_chief power = 0.5 } }

			}

			modifier = { factor = 1.5 trait = content }
			modifier = { factor = 1.25 trait = humble }
			modifier = { factor = 1.5 trait = diligent }
			modifier = { factor = 1.25 trait = charitable }
			modifier = { factor = 1.25 trait = patient }
			modifier = { factor = 1.25 trait = content }

			modifier = { factor = 0 trait = defective }
		}
		primary_title = { gain_title = { target = FROM type = faction_demand } }
		hidden_tooltip = {
			any_vassal = {
				limit = { ai = no NOT = { in_faction = faction_misaligned_chief } }
				# Inform other vassals - Realignment has been accepted
				character_event = { id = lizardmen.19 }
			}
			FROM = {
				# Demand accepted - Faction leader is installed
				letter_event = { id = lizardmen.17 }
				any_faction_backer = {
					faction = faction_misaligned_chief
					# Inform faction backers - Faction has won
					letter_event = { id = lizardmen.18 }
					diplomatic_immunity = no
				}
				diplomatic_immunity = no
			}
		}
	}
	# Resist
	option = {
		name = EVTOPTBlizardmen15
		ai_chance = {
			factor = 40
			modifier = { factor = 1.25 trait = envious }
			modifier = { factor = 1.5 trait = greedy }

			modifier = { factor = 1.25 trait = wroth }

				factor = 0.75
				FROM = { faction_power = { faction = faction_misaligned_chief power = 1.25 } }
			}
			modifier = {
				factor = 0.75
				FROM = { faction_power = { faction = faction_misaligned_chief power = 1.5 } }
			}
			modifier = {
				factor = 0.5
				FROM = { faction_power = { faction = faction_misaligned_chief power = 1.0 } }
			}
			modifier = {
				factor = 0.5
				FROM = { faction_power = { faction = faction_misaligned_chief power = 2.0 } }
			}
			modifier = {
				factor = 0.5
				FROM = { faction_power = { faction = faction_misaligned_chief power = 3.0 } }
			}
			modifier = {
				factor = 0.5
				FROM = { faction_power = { faction = faction_misaligned_chief power = 4.0 } }
			}
		}
		hidden_tooltip = {
			if = {
				limit = {
					FROM = { faction_power = { faction = faction_misaligned_chief power = 1.0 } }

	trigger = {
		FROM = { NOT = { location = { has_province_modifier = province_slave_raid } } }

		} #only against the living
	}

	option = { #Take all and burn their lands!
		name = EVTOPTAlizardmen25

			limit = { FROM = { owner = { top_liege = { NOT = { character = PREV } } } } }

		}
		FROM = { owner = { character_event = { id = lizardmen.26 } } }
		FROM = { set_province_flag = flag_province_scourged }
		if = {
			limit = {
				FROM = {
					OR = {
						any_neighbor_province = { is_tribal = yes }
						has_province_flag = flag_province_scourged
						NOT = { num_of_settlements = 2 }
					}
					NOT = { is_nomadic = yes }
				}
			}
			random = { chance = 1 FROM = { convert_to = TRIBAL } } #closesrandom
		} #closesif
		if = {
			limit = {
				FROM = { location = { NOT = { has_province_modifier = province_scourged } } }
			}
			FROM = {
				location = { add_province_modifier = { name = province_scourged duration = 900 } }
			}
		} #closesif

		wealth = 150
		prestige = 150
		piety = 250
		decadence = 1
		change_variable = { which = "sacrifical_victims" value = 2 }
		if = { limit = { ai = yes } change_variable = { which = "sacrifical_victims" value = 2 } }
		ai_chance = {
			factor = 25
			modifier = { factor = 1.75 ROOT = { OR = { trait = lunatic_3 trait = lunatic_2 } } }
			modifier = { factor = 2 trait = cruel }
		}
	}
	option = { #Just take them all!
		name = EVTOPTBlizardmen25
		if = {
			limit = { FROM = { owner = { top_liege = { NOT = { character = PREV } } } } }
			FROM = { owner = { top_liege = { character_event = { id = lizardmen.28 } } } }
		}
		if = {
			limit = {
				FROM = { location = { NOT = { has_province_modifier = province_slave_raid } } }
			}
			FROM = {
				location = {
					add_province_modifier = { name = province_slave_raid duration = 1000 }
				}
			}
		} #closesif
		FROM = { owner = { character_event = { id = lizardmen.28 } } }
		wealth = 25
		prestige = 10
		piety = 10
		change_variable = { which = "sacrifical_victims" value = 6 }
		if = { limit = { ai = yes } change_variable = { which = "sacrifical_victims" value = 5 } }
		ai_chance = {
			factor = 75
			modifier = {
				factor = 1.5
				ROOT = { NOR = { trait = possessed trait = lunatic_3 trait = lunatic_2 } }
			}
			modifier = { factor = 1.7 trait = zealous }
		}
	}
	option = { #I don't care about this province
		name = EVTOPTClizardmen25
		piety = -25
		prestige = -25
		if = { limit = { trait = cruel } random = { chance = 1 remove_trait = cruel } }

			factor = 1
			modifier = {
				factor = 1.2
				ROOT = { NOR = { trait = possessed trait = lunatic_3 trait = lunatic_2 } }
			}
			modifier = { factor = 1.2 trait = kind }
		}
	}

# The jungle
# The Lustrian jungle has four densities (province modifiers lustria_light_jungle

# lustria.1 Jungle setup
# On startup, gives every Lustrian demesne province (outside Quetza) the
# jungle modifiers.
# Scope: ROOT is a ruler with a capital in Lustria.
character_event = {
	id = lustria.1
	hide_window = yes

	only_rulers = yes

	is_triggered_only = yes

	trigger = {
		is_save_game = no
		capital_scope = { location = { region = world_new_world_lustria } }
	}

	immediate = {
		ROOT = {
			any_demesne_province = {
				limit = {
					location = {
						region = world_new_world_lustria
						NOT = { duchy = { title = d_quetza } }
					}
					NOT = { has_province_modifier = lustria_jungle }
				}
				# Manually add the modifier on startup
				province_event = { id = lustria.2 }
			}
		}
	}

	option = {
		name = OK
	}
}

# lustria.2 Jungle setup for one province
# Adds the jungle modifier and a random starting density (lighter near rivers,
# coasts and big capitals).
# Scope: ROOT is the province.
province_event = {
	id = lustria.2
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		add_province_modifier = { name = lustria_jungle duration = -1 }
		# Random starting density
		if = {
			limit = {
				region = world_new_world_lustria
				NOR = {
					has_province_modifier = lustria_light_jungle
					has_province_modifier = lustria_medium_jungle
					has_province_modifier = lustria_dense_jungle
					has_province_modifier = lustria_oppressive_jungle
				}
			}
			# Make it so the initial jungle level is random
			random_list = {
				# Light Jungle
				25 = {
					# Kingdom level tier is more likely to have light jungle
					modifier = {
						factor = 2.0
						is_capital = yes
						holder_scope = { higher_real_tier_than = DUKE }
					}
					# Duke level is a bit more likely to have light jungle
					modifier = { factor = 1.5 is_capital = yes holder_scope = { real_tier = DUKE } }
					# Count level is a bit more likely to have light jungle
					modifier = {
						factor = 1.25
						is_capital = yes
						holder_scope = { real_tier = COUNT }
					}
					# Ports and rivers more likely to have light jungle
					modifier = { factor = 3 OR = { borders_major_river = yes port = yes } }
					add_province_modifier = { name = lustria_light_jungle duration = -1 }
				}
				# Medium Jungle
				25 = {
					# Ports and rivers more likely to have light jungle
					modifier = { factor = 2 OR = { borders_major_river = yes port = yes } }
					add_province_modifier = { name = lustria_medium_jungle duration = -1 }
				}
				# Dense Jungle
				25 = { add_province_modifier = { name = lustria_dense_jungle duration = -1 } }
				# Oppressive Jungle
				25 = {
					# Ports and rivers more likely to have light jungle
					modifier = { factor = 0 OR = { borders_major_river = yes port = yes } }
					add_province_modifier = { name = lustria_oppressive_jungle duration = -1 }
				}
			}
		}
	}

	option = {
		name = OK
	}
}

# lustria.3 The jungle grows
# Five-year pulse. The jungle in Lustrian demesne provinces grows a step. AI
# rulers may instead clear a step (lizardmen and Amazons rarely do).
# Scope: ROOT is a ruler holding Lustrian jungle.
character_event = {
	id = lustria.3
	desc = EVTDESClustria3
	picture = "GFX_evt_lustria_jungle"

	only_rulers = yes

	is_triggered_only = yes # on_five_year_pulse (list lustria_jungle)

	trigger = {
		any_demesne_province = {
			region = world_new_world_lustria
			terrain = jungle
			has_province_modifier = lustria_jungle
			OR = {
				NOT = { has_province_modifier = lustria_oppressive_jungle }
				ROOT = { ai = yes } # the AI also clears its jungle with this event
			}
		}
	}

	weight_multiplier = {
		factor = 1
		# The children of the Old Ones live with the jungle
		modifier = { factor = 0.4 religion_group = old_ones_group }
		# Feudal realms clear it
		modifier = { factor = 0.65 is_tribal = no is_nomadic = no }
		# Tribes use it more than they fight it
		modifier = { factor = 0.85 is_tribal = yes }
		# Nomads pass through and leave it be
		modifier = { factor = 2.0 is_nomadic = yes }
		# Good stewardship keeps the jungle at bay
		modifier = { factor = 0.9 realm_stewardship > 10 }
		modifier = { factor = 0.8 realm_stewardship > 15 }
		modifier = { factor = 0.7 realm_stewardship > 20 }
		modifier = { factor = 0.6 realm_stewardship > 25 }
	}

	# The jungle grows
	option = {
		name = OK
		trigger = { ai = no }
		any_demesne_province = {
			limit = {
				region = world_new_world_lustria
				terrain = jungle
				has_province_modifier = lustria_jungle
			}
			lustria_jungle_grow_effect = yes
		}
	}
	# AI: the jungle grows, or the ruler clears a step of it
	option = {
		name = OK
		trigger = { ai = yes }
		random_list = {
			# Growth, a coin toss for each province
			50 = {
				modifier = { factor = 1.2 realm_stewardship < 15 }
				any_demesne_province = {
					limit = {
						region = world_new_world_lustria
						terrain = jungle
						has_province_modifier = lustria_jungle
					}
					random = {
						chance = 50
						# Slower near rivers and coasts (the wild heart of Lustria is inland)
						modifier = { factor = 0.8 OR = { borders_major_river = yes port = yes } }
						modifier = { factor = 0.8 has_province_modifier = lustria_dense_jungle }
						modifier = { factor = 0.75 has_province_modifier = lustria_light_jungle }
						modifier = { factor = 0.9 ROOT = { is_tribal = no is_nomadic = no } }
						lustria_jungle_grow_effect = yes
					}
				}
			}
			# Clearing
			50 = {
				modifier = {
					factor = 0.1
					capital_scope = { has_province_modifier = lustria_light_jungle }
				}
				modifier = { factor = 0.1 religion_group = old_ones_group }
				modifier = { factor = 0.5 OR = { religion = amazon religion = kukaili } }
				modifier = { factor = 1.15 realm_stewardship > 10 }
				modifier = { factor = 1.15 realm_stewardship > 15 }
				modifier = { factor = 1.15 realm_stewardship > 20 }
				modifier = { factor = 1.15 realm_stewardship > 25 }
				modifier = { factor = 1.3 independent = no top_liege = { realm_stewardship > 20 } }
				any_demesne_province = {
					limit = {
						region = world_new_world_lustria
						terrain = jungle
						has_province_modifier = lustria_jungle
					}
					if = {
						limit = { has_province_modifier = lustria_light_jungle }
						random = { chance = 20 remove_province_modifier = lustria_light_jungle }
					}
					else = { lustria_jungle_clear_effect = yes }
				}
			}
		}
	}
}
