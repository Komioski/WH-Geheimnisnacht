# Lovers: affairs, first loves, jealous spouses and happy marriages
# Namespace: lover (was 64000 - 64140 in the vanilla lovers_events.txt)
# Faction: Generic
#
# The old lover events of the base game. Most are the fallback for players
# without Way of Life (lacks_dlc), whose lover events (WHG_seduction_events)
# replace them; the wounded lover, the happy marriage and the endings that
# other events call run for everyone.
#
# Confrontations                       lover.1 - lover.6
# A sick or wounded lover              lover.10 - lover.11
# First love                           lover.20 - lover.28
# Taking a mistress                    lover.30 - lover.35
# Love and its pleasures               lover.40 - lover.44
# The end of an affair                 lover.50 - lover.52
# The strain of an affair              lover.60 - lover.82
#
# Fired from: the on_yearly_pulse random_events of 00_on_actions.txt
#   (1, 5, 10, 11, 20, 30, 40, 41, 42), the on_bi_yearly_pulse lists
#   lovers_* in wh_on_actions.txt (the old MTTH events), MTTH (82, while in
#   command), and other events: the Grand Judge (WHG_court_office_events,
#   51), traits_effects_events (40, 51, 75) and rumours_events (68).
# Saved: event targets lover_target, lover_teen, lover_crush,
#   lover_sweetheart; character flags teenage_infatuation, lover_reply_good,
#   lover_reply_bad, lover_wounded_by_rival, lover_made_infirm,
#   never_lover, poet_spouse_muse, lost_loving_feeling,
#   adulterous_affair_public.
# Scripted triggers and effects: WHG_lover_triggers.txt and
#   WHG_lover_effects.txt in common/scripted_triggers and scripted_effects.
# Loc: localisation/00_WHG_lover_events.csv
# Full list with callers: docs/events/event_database.csv

# lover.1 My spouse has a lover
# Without Way of Life. ROOT learns their spouse has a lover and confronts
# the spouse (lover.2) or the lover (lover.3). Not in Araby, where the
# faiths of the court keep many wives.
# Scope: ROOT is the wronged spouse.
character_event = {
	id = lover.1
	desc = EVTDESClover1
	picture = GFX_evt_quarrel

	is_triggered_only = yes # on_yearly_pulse random_events, 00_on_actions.txt
	lacks_dlc = "Way of Life"
	capable_only = yes
	prisoner = no

	trigger = {
		cant_love_trigger = yes
		NOR = { religion = ormazd religion = kitabid }
		spouse = {
			NOR = { religion = ormazd religion = kitabid }
			in_command = no
			war = no
			is_abroad = no
			is_alive = yes
			has_lover = yes
			prisoner = no
			NOT = { is_lover = ROOT }
			any_lover = { NOT = { character = ROOT } }
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 2 trait = envious }
	}

	immediate = {
		spouse = {
			random_lover = {
				limit = { NOT = { character = ROOT } }
				save_event_target_as = lover_target
			}
		}
	}

	option = { # Confront my spouse
		name = EVTOPTAlover1
		spouse = { character_event = { id = lover.2 tooltip = EVTTOOLTIPlover2 } }
	}
	option = { # Confront the lover
		name = EVTOPTBlover1
		event_target:lover_target = {
			character_event = { id = lover.3 tooltip = EVTTOOLTIPlover3 }
		}
	}
}

# lover.2 Confronted by my spouse
# The cheating spouse refuses to give up the lover (the other is angry) or
# ends it (the other is merely irritated).
# Scope: ROOT is the cheating spouse, FROM the wronged one.
character_event = {
	id = lover.2
	desc = EVTDESClover2
	picture = GFX_evt_quarrel

	is_triggered_only = yes # lover.1, lover.5

	option = { # I will not give them up
		name = EVTOPTAlover2
		FROM = { reverse_opinion = { modifier = opinion_angry who = ROOT years = 2 } }
	}
	option = { # It is over
		name = EVTOPTBlover2
		random_lover = { limit = { NOT = { character = FROM } } lover_end_affair_effect = yes }
		FROM = { reverse_opinion = { modifier = opinion_irritated who = ROOT years = 2 } }
	}
}

# lover.3 Confronted by the spouse
# The lover is told to leave someone's husband or wife alone. Refuse, end
# it, or threaten: a strong lover may beat the spouse (lover.4, 1 in 5), a
# clever one arrange an accident or a poisoning (1 in 10 each).
# Scope: ROOT is the lover, FROM the wronged spouse.
character_event = {
	id = lover.3
	desc = EVTDESClover3
	picture = GFX_evt_quarrel

	is_triggered_only = yes # lover.1

	option = { # Refuse
		name = EVTOPTAlover3
		FROM = { reverse_opinion = { modifier = opinion_angry who = ROOT years = 2 } }
	}
	option = { # End it
		name = EVTOPTBlover3
		FROM = {
			spouse = { lover_end_affair_effect = yes }
			reverse_opinion = { modifier = opinion_irritated who = ROOT years = 2 }
		}
	}
	option = { # I will claw your eyes out
		name = EVTOPTClover3
		tooltip_info = martial
		trigger = { martial = 10 FROM = { NOT = { trait = wounded } } }
		FROM = {
			random = {
				chance = 20
				set_character_flag = lover_wounded_by_rival
				character_event = { id = lover.4 tooltip = EVTTOOLTIPlover4 }
			}
			reverse_opinion = { modifier = opinion_love_rival who = ROOT years = 5 }
		}
	}
	option = { # Your meddling will cost you
		name = EVTOPTDlover3
		tooltip_info = intrigue
		trigger = { intrigue = 10 FROM = { NOT = { trait = wounded } } }
		FROM = {
			random_list = {
				10 = {
					set_character_flag = lover_wounded_by_rival
					character_event = { id = lover.4 days = 14 tooltip = EVTTOOLTIPlover4 }
				}
				10 = {
					set_character_flag = lover_made_infirm
					character_event = { id = lover.4 days = 7 tooltip = EVTTOOLTIPlover4b }
				}
				80 = { }
			}
			reverse_opinion = { modifier = opinion_love_rival who = ROOT years = 5 }
		}
	}
}

# lover.10 A sick lover
# Without Way of Life or Reaper's Due. A lover is ill or pneumonic. Nurse
# them (a chance to recover; they are glad), pay for a physician (a better
# chance) or end the affair.
# Scope: ROOT is the ruler, event_target:lover_target the lover.
character_event = {
	id = lover.10
	desc = EVTDESClover10
	picture = GFX_evt_illness

	is_triggered_only = yes # on_yearly_pulse random_events, 00_on_actions.txt
	lacks_dlc = "Way of Life"
	capable_only = yes
	prisoner = no

	trigger = {
		NOT = { has_dlc = "Reapers" }
		cant_love_trigger = yes
		NOT = { trait = cruel }
		any_lover = {
			cant_love_trigger = yes
			is_abroad = no
			OR = { is_ill = yes trait = pneumonic }
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 0.5 is_ill = yes }
		modifier = { factor = 2 trait = kind }
	}

	immediate = {
		random_lover = {
			limit = { is_abroad = no OR = { is_ill = yes trait = pneumonic } }
			save_event_target_as = lover_target
		}
	}

	option = { # Nurse them
		name = EVTOPTAlover10
		ai_chance = { factor = 50 }
		event_target:lover_target = {
			lover_sickbed_effect = yes
			opinion = { modifier = opinion_glad who = ROOT years = 1 }
		}
	}
	option = { # Send for a physician
		name = EVTOPTBlover10
		ai_chance = { factor = 50 }
		scaled_wealth = -0.2
		event_target:lover_target = { lover_sickbed_effect = yes lover_sickbed_effect = yes }
	}
	option = { # End it
		name = EVTOPTClover10
		ai_chance = { factor = 0 }
		event_target:lover_target = { lover_end_affair_effect = yes }
	}
}

# lover.20 A first love
# Without Way of Life. An unmarried youth of 12 to 17 falls for a youth of
# the other sex at court and tells them (4 in 5 the feeling is returned,
# lover.21), or decides they were not so pretty after all (shy or craven).
# Fixed: the old event wrote to every youth at court who fit, each with its
# own answer.
# Scope: ROOT is the youth, event_target:lover_crush the one they love.
character_event = {
	id = lover.20
	desc = EVTDESClover20
	picture = GFX_evt_lovers

	is_triggered_only = yes # on_yearly_pulse random_events, 00_on_actions.txt
	lacks_dlc = "Way of Life"
	prisoner = no
	min_age = 12
	max_age = 17
	is_married = no

	trigger = {
		cant_love_trigger = yes
		is_no_females_race = no
		has_epidemic = no
		is_ill = no
		is_pregnant = no
		NOR = { trait = homosexual trait = celibate trait = incapable trait = eunuch }
		any_courtier = { lover_crush_target_trigger = yes }
	}

	immediate = {
		random_courtier = {
			limit = { lover_crush_target_trigger = yes }
			score_value = { value = 10 interesting_courtier_selection_score = yes }
			save_event_target_as = lover_crush
		}
	}

	option = { # Tell them
		name = EVTOPTAlover20
		ai_chance = { factor = 95 }
		event_target:lover_crush = {
			random_list = {
				80 = {
					set_character_flag = lover_reply_good
					character_event = { id = lover.21 tooltip = EVTTOOLTIPlover21 }
				}
				20 = {
					set_character_flag = lover_reply_bad
					character_event = { id = lover.21 tooltip = EVTTOOLTIPlover21b }
				}
			}
		}
	}
	option = { # Not that pretty anyway
		name = EVTOPTBlover20
		ai_chance = {
			factor = 5
			modifier = { factor = 15 trait = craven }
			modifier = { factor = 15 trait = shy }
		}
		if = { limit = { NOT = { trait = shy } } add_trait = shy }
		else_if = { limit = { NOR = { trait = craven trait = berserker } } add_trait = craven }
	}
}

# lover.30 A wife grown cold
# Without Way of Life. A married ruler whose wife is sick, past 45 or
# barren thinks of a lady of the court. Take her openly (lustful; the wife
# may hear, lover.31), discreetly (deceitful; less likely to be found out),
# or refuse the sin (piety; chaste, zealous or, rarely, celibate in the Old
# World faiths). The lady answers in lover.32.
# Scope: ROOT is the ruler, event_target:lover_target the lady.
character_event = {
	id = lover.30
	desc = EVTDESClover30

	only_rulers = yes
	capable_only = yes

		fertility = 0.5

		has_lover = no

		NOT = { has_character_flag = never_lover }

# lover.82 Far from my love
# A commander at war with one lover misses them. Stay faithful (a chance of
# honest), pray (a chance of zealous, or chaste; piety in Araby), or find
# comfort elsewhere (deceitful; 1 in 4 the old love fades, lover.51).
# The prayer options for the Old World and Araby are one.
# Scope: ROOT is the commander.
character_event = {
	id = lover.82
	desc = EVTDESClover82
	picture = GFX_evt_bad_news

	lacks_dlc = "Way of Life"
	only_men = yes
	min_age = 16
	capable_only = yes
	prisoner = no
	war = yes

	trigger = {
		cant_love_trigger = yes
		in_command = yes
		has_lover = yes
		NOT = { num_of_lovers = 2 }
		NOR = { trait = honest trait = chaste trait = deceitful }
		NOT = { num_of_traits = 7 }
	}

	mean_time_to_happen = {
		months = 1200
		modifier = { factor = 0.5 trait = stressed }
		modifier = { factor = 0.5 trait = depressed }
	}

	option = { # Stay faithful
		name = EVTOPTAlover82
		random = { chance = 25 add_trait = honest }
	}
	option = { # Pray
		name = EVTOPTBlover82
		random_list = {
			50 = { }
			25 = { add_trait = zealous }
			25 = {
				if = { limit = { religion_group = ormazd_gods_group } piety = 5 }
				else = { add_trait = chaste }
			}
		}
	}
	option = { # Comfort elsewhere
		name = EVTOPTClover82
		add_trait = deceitful
		random = { chance = 25 character_event = { id = lover.51 tooltip = EVTTOOLTIPlover51 } }
	}
}
