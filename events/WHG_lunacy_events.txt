###############################################################################
# Lunacy events
# Namespace: lunacy (was "slunacy" before the 2026 events refactor)
# Faction: Generic
#
# A lunatic (lunatic_1 twitchy, lunatic_2 maniac, lunatic_3 raving) or the
# wearer of a madness artifact slowly loses their mind. Maniacs and raving
# lunatics catch one of nine madnesses (character modifiers), each with its own
# follow-up events. Old vampires go mad with age and catch vampire madnesses.
# Each madness event feeds the lunatic_counter variable through mutate.5.
#
# Umbrella and twitchy events   lunacy.1 - lunacy.9
# Madness onsets                lunacy.10 - lunacy.15
# Bloodthirsty                  lunacy.16 - lunacy.21
# Blind Virtue                  lunacy.22 - lunacy.24
# Fake Mage                     lunacy.25 - lunacy.26
# Despairing Laughter           lunacy.27 - lunacy.29
# Unfeeling                     lunacy.30 - lunacy.32, lunacy.35 - lunacy.42
# God Delusion                  lunacy.33 - lunacy.34
# Vampire ageing and umbrella   lunacy.50 - lunacy.51
# Vampire madness onsets        lunacy.52 - lunacy.54
# Everchild Syndrome            lunacy.60 - lunacy.67
# Exquisite Agonies             lunacy.70 - lunacy.82
# Heart of Melancholy           lunacy.85 - lunacy.88
# Recovery                      lunacy.90
#
# Fired from common/on_actions/wh_on_actions.txt: the bi-yearly list lunacy
# (lunacy.1) and lunacy.51 every two years, the five-year lists
# lunacy_<madness> and the decade lists lunacy_vampire_aging, lunacy_everchild
# and lunacy_hibernation.
# Scripted triggers: common/scripted_triggers/wh_lunacy_triggers.txt
# Scripted effects: common/scripted_effects/wh_lunacy_effects.txt
# Artifacts: common/artifacts/22_artifacts_lunacy.txt
# Loc: localisation/00_WHG_lunacy_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = lunacy

# lunacy.1 Lunacy umbrella
# Every two years a mortal lunatic or madness artifact wearer may suffer a
# lunacy event: a twitchy quirk, or for maniacs and raving lunatics without a
# madness the onset of one. Doom, Morrslieb, mutations, Chaos cults and cursed
# lands make it more likely; a madness artifact alone makes it less likely.
# Scope: ROOT is the lunatic.
character_event = {
	id = lunacy.1
	hide_window = yes
	is_triggered_only = yes # on_bi_yearly_pulse, random list lunacy

	only_playable = yes

	trigger = {
		lunacy_can_go_mad_trigger = yes
		is_vampire_trigger = no
		OR = {
			trait = lunatic_1
			trait = lunatic_2
			trait = lunatic_3
			lunacy_madness_artifact_trigger = yes
		}
	}

	weight_multiplier = {
		factor = 1
		# Morrslieb and doom
		modifier = { factor = 1.25 has_global_flag = morrslieb_full }
		modifier = { factor = 0.5 is_doom_very_low_trigger = yes }
		modifier = { factor = 1.2 is_doom_low_trigger = yes }
		modifier = { factor = 1.5 is_doom_medium_trigger = yes }
		modifier = { factor = 1.75 is_doom_high_trigger = yes }
		modifier = { factor = 2 is_doom_very_high_trigger = yes }
		# A madness artifact alone is not true lunacy, and a healthy body resists it
		modifier = { factor = 0.6 NOR = { trait = lunatic_1 trait = lunatic_2 trait = lunatic_3 } }
		modifier = { factor = 0.9 lunacy_madness_artifact_trigger = yes health = 6 }
		modifier = { factor = 0.9 lunacy_madness_artifact_trigger = yes health = 8 }
		modifier = { factor = 0.9 lunacy_madness_artifact_trigger = yes health = 10 }
		# Race and body
		modifier = { factor = 0.75 trait = creature_dwarf }
		modifier = {
			factor = 0.75
			OR = {
				trait = creature_orc
				trait = creature_snotling
				trait = creature_goblin
				trait = creature_hobgoblin
			}
		}
		modifier = { factor = 2 trait = creature_fimir }
		modifier = { factor = 2 trait = creature_chaos_dwarf }
		modifier = { factor = 0.5 trait = creature_halfling }
		modifier = { factor = 1.5 has_minor_mutation_trigger = yes }
		modifier = { factor = 2 has_major_mutation_trigger = yes }
		modifier = { factor = 3 has_massive_mutation_trigger = yes }
		modifier = { factor = 1.25 trait = addict }
		modifier = { factor = 1.5 trait = lore_tzeentch }
		# The unschooled mind has no defence against the Winds of Magic
		modifier = {
			factor = 1.25
			NOR = {
				trait = magic_potential_1
				trait = magic_potential_2
				trait = magic_potential_3
				trait = magic_potential_4
			}
		}
		modifier = { factor = 1.25 has_magic_power_trigger = no }
		modifier = { factor = 2 has_character_flag = forsaken_flag }
		# Service of the Dark Gods
		modifier = { factor = 1.5 religion_group = chaos_gods_group }
		modifier = { factor = 1.25 society_member_of = cult_of_tzeentch }
		modifier = { factor = 1.5 society_member_of = cult_of_tzeentch2 }
		modifier = { factor = 1.75 society_member_of = cult_of_tzeentch3 }
		modifier = {
			factor = 1.25
			OR = {
				society_member_of = cult_of_khorne
				society_member_of = cult_of_slaanesh
				society_member_of = cult_of_nurgle
				society_member_of = cult_of_undivided
			}
		}
		modifier = {
			factor = 1.25
			OR = {
				society_member_of = cult_of_khorne2
				society_member_of = cult_of_slaanesh2
				society_member_of = cult_of_nurgle2
				society_member_of = cult_of_undivided2
			}
		}
		modifier = {
			factor = 1.25
			OR = {
				society_member_of = cult_of_khorne3
				society_member_of = cult_of_slaanesh3
				society_member_of = cult_of_nurgle3
				society_member_of = cult_of_undivided3
			}
		}
		# Cursed lands
		modifier = { factor = 1.5 location = { empire = { title = e_chaos_wastes } } }
		modifier = { factor = 1.25 location = { kingdom = { title = k_sylvania } } }
		modifier = { factor = 1.25 location = { kingdom = { title = k_mousillon } } }
		modifier = { factor = 1.25 location = { duchy = { title = d_geisterforst } } }
		modifier = { factor = 1.5 location = { has_province_modifier = warpdust_traces } }
		modifier = { factor = 0.5 ai = yes }
	}

	immediate = {
		random_list = {
			20 = { } # Nothing this time
			10 = { # Twitchy: whispers and shadows
				modifier = { factor = 0 lunacy_twitchy_trigger = no }
				modifier = { factor = 0 trait = paranoid }
				character_event = { id = lunacy.2 }
			}
			10 = { # Twitchy: unmoved by suffering
				modifier = { factor = 0 lunacy_twitchy_trigger = no }
				modifier = { factor = 0 trait = cruel }
				character_event = { id = lunacy.3 }
			}
			10 = { # Twitchy: too stressed
				modifier = { factor = 0 lunacy_twitchy_trigger = no }
				modifier = { factor = 0 NOT = { trait = stressed } }
				character_event = { id = lunacy.4 }
			}
			5 = { # Twitchy mage: dangerous experiment
				modifier = { factor = 0 lunacy_twitchy_trigger = no }
				modifier = { factor = 0 has_magic_power_trigger = no }
				character_event = { id = lunacy.5 }
			}
			5 = { # Twitchy: the drunkard feels poisoned
				modifier = { factor = 0 lunacy_twitchy_trigger = no }
				modifier = { factor = 0 NOT = { trait = drunkard } }
				character_event = { id = lunacy.9 }
			}
			10 = { # Maniac: Blind Virtue
				modifier = { factor = 0 NOT = { trait = lunatic_2 } }
				modifier = { factor = 0 lunacy_has_madness_trigger = yes }
				character_event = { id = lunacy.10 }
			}
			10 = { # Maniac: Fake Mage
				modifier = { factor = 0 NOT = { trait = lunatic_2 } }
				modifier = { factor = 0 lunacy_has_madness_trigger = yes }
				modifier = { factor = 0 has_magic_power_trigger = yes }
				character_event = { id = lunacy.11 }
			}
			10 = { # Maniac: Despairing Laughter
				modifier = { factor = 0 NOT = { trait = lunatic_2 } }
				modifier = { factor = 0 lunacy_has_madness_trigger = yes }
				character_event = { id = lunacy.12 }
			}
			10 = { # Maniac: Unfeeling
				modifier = { factor = 0 NOT = { trait = lunatic_2 } }
				modifier = { factor = 0 lunacy_has_madness_trigger = yes }
				character_event = { id = lunacy.13 }
			}
			10 = { # Raving: Bloodthirsty
				modifier = { factor = 0 NOT = { trait = lunatic_3 } }
				modifier = { factor = 0 lunacy_has_madness_trigger = yes }
				character_event = { id = lunacy.14 }
			}
			10 = { # Raving: God Delusion
				modifier = { factor = 0 NOT = { trait = lunatic_3 } }
				modifier = { factor = 0 lunacy_has_madness_trigger = yes }
				modifier = { factor = 0 is_ruler = no }
				character_event = { id = lunacy.15 }
			}
		}
	}
}

# lunacy.17 The victim dies
# The murder victim dies unless they cannot die. Their close kin, spouse and
# liege learn who did it (lunacy.18) and hate the murderer.
# Scope: ROOT is the victim, FROM the murderer.
character_event = {
	id = lunacy.17
	hide_window = yes
	is_triggered_only = yes # lunacy.16

	immediate = {
		if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
		else = {
			any_dynasty_member = {
				limit = { is_close_relative = ROOT NOT = { character = FROM } }
				opinion = { modifier = opinion_murderous_lunatic who = FROM years = 100 }
				character_event = { id = lunacy.18 }
			}
			any_spouse = {
				limit = { NOT = { character = FROM } }
				opinion = { modifier = opinion_murderous_lunatic who = FROM years = 100 }
				character_event = { id = lunacy.18 }
			}
			liege = {
				if = {
					limit = { NOT = { character = FROM } }
					opinion = { modifier = opinion_murderous_lunatic who = FROM years = 10 }
					character_event = { id = lunacy.18 }
				}
			}
			death = { death_reason = death_murder killer = FROM }
		}
	}
}

			factor = 1.4
			OR = { trait = vampire_strigoi_visible trait = vampire_strigoi_hidden }
		}

			factor = 1.25
			OR = { trait = vampire_necrarch_visible trait = vampire_necrarch_hidden }

	}

		random_list = {
			30 = { # The mind holds
				modifier = { factor = 1.5 NOR = { trait = creature_human trait = creature_amazon } }

# A vampire with Heart of Melancholy withdraws into hibernation: gains the

# (lunacy.88). About once in 30 years.

	id = lunacy.87
	desc = EVTDESClunacy87
	picture = GFX_evt_feral_vampire

	capable_only = yes

		has_character_modifier = heart_of_melancholy
		lunacy_can_act_trigger = yes
	}

	option = {
		name = EVTOPTAlunacy87
		hidden_tooltip = { character_event = { id = mutate.5 } } # Lunacy counter
		add_trait = resting
		if = {
			limit = { is_ruler = yes }
			capital_scope = {
				add_province_modifier = { name = hidden_hibernated_strigoi duration = -1 }

	}

# lunacy.90 The mind mends
# A sane character slowly sheds the strain that builds toward lunacy: the
# lunatic counter drops by 1. Without this the counter only ever grew, so a
# dwarf or elf living for centuries was all but sure to go mad. Long-lived and
# hardy races (lizardmen, greenskins) mend faster.
# Scope: ROOT is the character.
character_event = {
	id = lunacy.90
	hide_window = yes
	is_triggered_only = yes # on_five_year_pulse, random list lunacy_recovery

	trigger = {
		is_lunatic_trigger = no
		check_variable = { which = "lunatic_counter" value = 1 }
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 2 lifespan_dwarf_trigger = yes }
		modifier = { factor = 2 lifespan_elf_trigger = yes }
		modifier = { factor = 2 lifespan_other_long_lived_trigger = yes }
		modifier = { factor = 3 lifespan_hardy_race_trigger = yes }
	}

	immediate = {
		change_variable = { which = "lunatic_counter" value = -1 }
	}
}
