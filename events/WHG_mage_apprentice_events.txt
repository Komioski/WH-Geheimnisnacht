# Mage apprentice events
# Namespace: apprentice (was MageAp in S_mage_learning_events.txt)
# Faction: Generic
#
# How a character with the gift becomes a mage outside the colleges' own
# testing: as a court mage's apprentice, as a Chaos sorcerer's pupil, by
# teaching himself (hedge wizards, and college members who study alone), by
# teaching himself necromancy, or as a necromancer's apprentice. Wandering
# mages also offer their services to rulers whose courts have none.
#
# Becoming an apprentice                 apprentice.1 - apprentice.4
# A master dies                          apprentice.8
# The master's lessons                   apprentice.10 - apprentice.14
# The test and the news                  apprentice.15 - apprentice.20
# A Chaos sorcerer's pupil               apprentice.30 - apprentice.40
# Teaching yourself                      apprentice.50 - apprentice.57
# Teaching yourself necromancy           apprentice.60 - apprentice.81
# A necromancer's apprentice             apprentice.85 - apprentice.95
# Wandering mages                        apprentice.100 - apprentice.104
#
# Callers:
#   common/on_actions/wh_on_actions.txt: apprentice.1 in the
#     mage_apprentice_wants lists (on_adolescence_pulse, on_childhood_pulse),
#     apprentice.100 on on_five_year_pulse.
#   common/on_actions/wh_magic_on_actions.txt: apprentice.8 on on_death.
#   common/minor_titles/wh_minor_titles.txt: a new court mage fires
#     apprentice.1 for former apprentices at court.
#   decisions/wh_magic_decisions.txt: the hedge wizard decision fires
#     apprentice.50, the necromancy decision apprentice.60, ward_apprentice
#     apprentice.10 or apprentice.30, necromancer_ward_apprentice
#     apprentice.85. decisions/wh_vampire_decisions.txt: apprentice.85.
#   events/WHG_college_of_magic_events.txt: college.24 fires apprentice.10
#     (a college teacher) or apprentice.51 (studying alone).
# Scripted triggers: common/scripted_triggers/wh_apprentice_triggers.txt
# Scripted effects: common/scripted_effects/wh_apprentice_effects.txt
# Loc: localisation/00_WHG_mage_apprentice_events.csv
###############################################################################

# apprentice.1 A child wants to learn magic
# A child whose guardian is a mage, or whose court has a court mage, asks
# to be taught. Bright, ambitious children ask more often.
# Scope: ROOT is the child.
character_event = {
	id = apprentice.1
	desc = EVTDESCapprentice1
	picture = GFX_evt_apprentice

	min_age = 6

	is_triggered_only = yes # mage_apprentice_wants, title_teacher_mage

	trigger = {
		gender_is_allowed_magic_trigger = yes
		OR = {
			AND = { learning = 2 age = 6 }
			AND = { learning = 4 age = 11 }
			AND = { trait = creature_elf learning = 14 age = 20 }
		}
		OR = {
			age < 26
			check_variable = { which = "mage_teaching" value = 1 }
			trait = creature_elf
		}
		OR = {
			guardian = { is_trained_mage_trigger = yes }
			AND = {
				is_landed = yes
				any_courtier_or_vassal = { has_minor_title = title_teacher_mage }
			}
			AND = {
				is_landed = no
				liege = { any_courtier_or_vassal = { has_minor_title = title_teacher_mage } }
			}
		}
		NOR = {
			trait = hedge_wizard
			trait = mage_apprentice
			has_character_flag = failed_mage
			has_character_flag = priority_mage_recruit
			has_character_modifier = apprenticeship_refused
			has_epidemic = yes
			is_ill = yes
			is_incapable = yes
			is_trained_mage_trigger = yes
		}
	}

	weight_multiplier = {
		days = 1
		modifier = { factor = 4 trait = genius }
		modifier = { factor = 2 trait = brilliant }
		modifier = { factor = 1.5 trait = bright }
		modifier = { factor = 4 trait = ambitious }
		modifier = { factor = 1.25 trait = quick }
		modifier = { factor = 1.25 trait = smart }
		modifier = { factor = 1.25 trait = shrewd }
		modifier = {
			factor = 10
			culture_group = kislevite_group
			is_female = yes
			any_courtier_or_vassal = { has_minor_title = title_teacher_mage trait = ice_witch }
		}
		modifier = {
			factor = 10
			trait = creature_elf
			top_liege = { trait = malekith }
			is_female = yes
			any_courtier_or_vassal = { has_minor_title = title_teacher_mage }
		}
		modifier = {
			factor = 10
			trait = creature_elf
			NOT = { top_liege = { trait = malekith } }
			any_courtier_or_vassal = { has_minor_title = title_teacher_mage }
		}
		modifier = {
			factor = 5
			learning = 6
			is_female = no
			guardian = { is_trained_mage_trigger = yes trait = arabyan_scholar }
		}
		modifier = {
			factor = 1.25
			is_ruler = yes
			any_courtier_or_vassal = { has_minor_title = title_teacher_mage }
		}
		modifier = {
			factor = 1.25
			is_ruler = no
			liege = { any_courtier_or_vassal = { has_minor_title = title_teacher_mage } }
		}
		modifier = { factor = 0 OR = { trait = imbecile trait = hollowed } }
		modifier = { factor = 0.2 OR = { trait = slothful trait = idiot } }
	}

	option = { # Can you teach me?
		name = EVTOPTAapprentice1
		hidden_effect = {
			if = {
				limit = { guardian = { is_trained_mage_trigger = yes } }
				guardian = { character_event = { id = apprentice.2 } }
			}
			else_if = {
				limit = { is_landed = yes }
				random_courtier_or_vassal = {
					limit = { has_minor_title = title_teacher_mage is_trained_mage_trigger = yes }
					character_event = { id = apprentice.2 }
				}
			}
			else = {
				liege = {
					random_courtier_or_vassal = {
						limit = {
							has_minor_title = title_teacher_mage
							is_trained_mage_trigger = yes
						}
						character_event = { id = apprentice.2 }
					}
				}
			}
		}
	}
}

			}
		}
	}

# apprentice.8 A master dies
# Each unfinished apprentice of a dead mage goes straight to the test
# (apprentice.15, or apprentice.91 for a necromancer's apprentice) instead
# of waiting forever for the next lesson. College students of a Magister
# are left to the college, which finds them a new teacher (college.26).
# Scope: ROOT is the dead character.
character_event = {
	id = apprentice.8
	hide_window = yes
	is_triggered_only = yes # on_death

	trigger = {
		is_trained_mage_trigger = yes
		any_friend = { trait = mage_apprentice has_character_flag = mage_master@ROOT }
	}

	immediate = {
		save_event_target_as = mage_tutor
		any_friend = {
			limit = {
				trait = mage_apprentice
				has_character_flag = mage_master@ROOT
				NAND = {
					has_character_flag = learning_magic_in_college
					ROOT = { is_college_of_magic_member = yes society_rank = 3 }
				}
			}
			clr_character_flag = mage_master@ROOT
			if = {
				limit = { ROOT = { trait = lore_necromancy } }
				character_event = { id = apprentice.91 days = 5 }
			}
			else = { character_event = { id = apprentice.15 days = 5 } }
		}
	}
}

# apprentice.10 What to teach
# Every year or so the master chooses the next lesson. After fourteen
# lessons, or when the years of training are over, the apprentice can be
# tested; a pupil under sixteen gets a little more practice first. An adult
# apprentice of a college Magister joins his master's college.
# Scope: ROOT is the master, FROM is the apprentice.
character_event = {
	id = apprentice.10
	desc = EVTDESCapprentice10
	picture = GFX_evt_apprentice

	is_triggered_only = yes # apprentice.4, apprentice.11-14, ward_apprentice, college.24

	trigger = {
		FROM = {
			is_alive = yes
			trait = mage_apprentice
			NOT = { is_trained_mage_trigger = yes }
			NOT = { has_character_flag = failed_mage }
		}
	}

	immediate = {
		FROM = { apprentice_join_master_college_effect = yes }
	}

	option = { # Caution
		name = EVTOPTAapprentice10
		trigger = { FROM = { apprentice_lessons_left_trigger = yes } }
		hidden_effect = {
			FROM = {
				change_variable = { which = "mage_teaching" value = 1 }
				character_event = { id = apprentice.11 }
			}
		}
	}
	option = { # Experiment
		name = EVTOPTBapprentice10
		trigger = { FROM = { apprentice_lessons_left_trigger = yes } }
		hidden_effect = {
			FROM = {
				change_variable = { which = "mage_teaching" value = 1 }
				character_event = { id = apprentice.12 }
			}
		}
	}
	option = { # Focus
		name = EVTOPTCapprentice10
		trigger = { FROM = { apprentice_lessons_left_trigger = yes } }
		hidden_effect = {
			FROM = {
				change_variable = { which = "mage_teaching" value = 1 }
				character_event = { id = apprentice.13 }
			}
		}
	}
	option = { # Test
		name = EVTOPTDapprentice10
		trigger = {
			FROM = {
				age = 16
				OR = {
					check_variable = { which = "mage_teaching" value = 14 }
					magic_training_years_over = yes
				}
			}
		}
		ai_chance = { factor = 3 }
		hidden_effect = {
			FROM = {
				set_variable = { which = "mage_teaching" value = 0 }
				character_event = { id = apprentice.15 }
			}
		}
	}
	option = { # Almost there
		name = EVTOPTEapprentice10
		trigger = {
			FROM = { check_variable = { which = "mage_teaching" value = 14 } NOT = { age = 16 } }
		}
		hidden_effect = { FROM = { character_event = { id = apprentice.14 } } }
	}
}

# apprentice.15 The test
# The apprentice tries to call the winds on his own
# (apprentice_master_test_trigger). On success he takes his master's lore;
# on failure he will never be a mage. Master and parents hear the news.
# Scope: ROOT is the apprentice, event_target:mage_tutor the master (alive
# or dead).
character_event = {
	id = apprentice.15
	desc = {
		text = EVTDESCapprentice15
		trigger = {
			event_target:mage_tutor = { is_alive = yes }
			apprentice_master_test_trigger = yes
		}
	}
	desc = {
		text = EVTDESCapprentice15b
		trigger = {
			event_target:mage_tutor = { is_alive = yes }
			NOT = { apprentice_master_test_trigger = yes }
		}
	}
	desc = { text = EVTDESCapprentice15c trigger = { event_target:mage_tutor = { is_alive = no } } }
	picture = GFX_evt_apprentice

	is_triggered_only = yes # apprentice.8, apprentice.10

	trigger = {
		trait = mage_apprentice
	}

	option = { # This is just the beginning
		name = EVTOPTAapprentice15
		trigger = { apprentice_master_test_trigger = yes }
		remove_trait = mage_apprentice
		prestige = 150
		apprentice_tutor_lore_effect = yes
		apprentice_new_mage_effect = yes
		hidden_effect = {
			event_target:mage_tutor = {
				if = { limit = { is_alive = yes } character_event = { id = apprentice.17 } }
			}
			father = {
				if = {
					limit = { is_alive = yes NOT = { character = event_target:mage_tutor } }
					character_event = { id = apprentice.18 days = 1 }
				}
			}
			mother = {
				if = {
					limit = { is_alive = yes NOT = { character = event_target:mage_tutor } }
					character_event = { id = apprentice.18 days = 1 }
				}
			}
		}
	}
	option = { # Why can't I do it?
		name = EVTOPTBapprentice15
		trigger = { NOT = { apprentice_master_test_trigger = yes } }
		set_character_flag = failed_mage
		remove_trait = mage_apprentice
		prestige = -150
		hidden_effect = {
			clr_learning_flags_effect = yes
			clr_character_flag = learning_magic_in_college
			event_target:mage_tutor = {
				if = { limit = { is_alive = yes } character_event = { id = apprentice.19 } }
			}
			father = {
				if = {
					limit = { is_alive = yes NOT = { character = event_target:mage_tutor } }
					character_event = { id = apprentice.20 days = 1 }
				}
			}
			mother = {
				if = {
					limit = { is_alive = yes NOT = { character = event_target:mage_tutor } }
					character_event = { id = apprentice.20 days = 1 }
				}
			}
		}
	}
}

# A Chaos sorcerer's pupil
# The Ruinous Powers grant magic to their chosen. A sorcerer of Tzeentch,
# Nurgle or Slaanesh preaches his god to his pupil, teaches its tenets and a
# little incantation, and then the god itself judges the pupil. An
# unbeliever is ignored; a believer without the gift, or out of favour, is
# spawned, killed or marred. This chain was written but never reachable;
# apprentice.4 and ward_apprentice now start it.

# apprentice.30 The Ruinous Powers favour only their chosen
# The sorcerer chooses the next lesson: preach the god to an unbeliever,
# teach the tenets to a believer, or teach incantation. After fourteen
# lessons the god tests the pupil.
# Scope: ROOT is the sorcerer, FROM is the pupil.
character_event = {
	id = apprentice.30
	desc = EVTDESCapprentice30
	picture = GFX_evt_apprentice

	is_triggered_only = yes # apprentice.4, apprentice.31-35, ward_apprentice

	trigger = {
		FROM = { is_alive = yes trait = mage_apprentice NOT = { is_trained_mage_trigger = yes } }
	}

	option = { # Speak of the god
		name = EVTOPTAapprentice30
		trigger = {
			FROM = {
				NOT = { check_variable = { which = "mage_teaching" value = 14 } }
				NOT = { religion = ROOT }
			}
		}
		hidden_effect = {
			FROM = {
				change_variable = { which = "mage_teaching" value = 1 }
				character_event = { id = apprentice.31 }
			}
		}
	}
	option = { # Teach the tenets
		name = EVTOPTBapprentice30
		trigger = {
			FROM = {
				NOT = { check_variable = { which = "mage_teaching" value = 14 } }
				religion = ROOT
			}
		}
		hidden_effect = {
			FROM = {
				change_variable = { which = "mage_teaching" value = 1 }
				character_event = { id = apprentice.34 }
			}
		}
	}
	option = { # Teach incantation
		name = EVTOPTCapprentice30
		trigger = { FROM = { NOT = { check_variable = { which = "mage_teaching" value = 14 } } } }

# The pupil may forsake his faith for his master's god. Each god tempts

	id = apprentice.31
	desc = { text = EVTDESCapprentice31 trigger = { FROM = { religion = tzeentch } } }

	desc = { text = EVTDESCapprentice31c trigger = { FROM = { religion = slaanesh } } }

	is_triggered_only = yes # apprentice.30

	option = { # A good bargain
		name = EVTOPTAapprentice31
		religion = FROM
		if = { limit = { religion = khorne } piety = -500 }
