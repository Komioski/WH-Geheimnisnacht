# Magic events: mage apprentices, the gift, magic laws, spells and battle magic
#
# Part 1: Mage apprentice events
# Namespace: apprentice (was MageAp in S_mage_learning_events.txt)
# Faction: Generic
#
# How a character with the gift becomes a mage outside the colleges' own
# testing: as a court mage's apprentice, as a Chaos sorcerer's pupil, by
# teaching himself (hedge wizards, and college members who study alone), by
# teaching himself necromancy, or as a necromancer's apprentice. Wandering
# mages also offer their services to rulers whose courts have none.
#
# Becoming an apprentice                 apprentice.1 - apprentice.4
# A master dies                          apprentice.8
# The master's lessons                   apprentice.10 - apprentice.14
# The test and the news                  apprentice.15 - apprentice.20
# A Chaos sorcerer's pupil               apprentice.30 - apprentice.40
# Teaching yourself                      apprentice.50 - apprentice.57
# Teaching yourself necromancy           apprentice.60 - apprentice.81
# A necromancer's apprentice             apprentice.85 - apprentice.95
# Wandering mages                        apprentice.100 - apprentice.104
#
# Callers:
#   common/on_actions/wh_on_actions.txt: apprentice.1 in the
#     mage_apprentice_wants lists (on_adolescence_pulse, on_childhood_pulse),
#     apprentice.100 on on_five_year_pulse.
#   common/on_actions/wh_magic_on_actions.txt: apprentice.8 on on_death.
#   common/minor_titles/wh_minor_titles.txt: a new court mage fires
#     apprentice.1 for former apprentices at court.
#   decisions/wh_magic_decisions.txt: the hedge wizard decision fires
#     apprentice.50, the necromancy decision apprentice.60, ward_apprentice
#     apprentice.10 or apprentice.30, necromancer_ward_apprentice
#     apprentice.85. decisions/wh_vampire_decisions.txt: apprentice.85.
#   events/WHG_college_of_magic_events.txt: college.24 fires apprentice.10
#     (a college teacher) or apprentice.51 (studying alone).
# Scripted triggers: common/scripted_triggers/wh_apprentice_triggers.txt
# Scripted effects: common/scripted_effects/wh_apprentice_effects.txt
# Loc: localisation/00_WHG_magic_events.csv
#
# Also in this file (each part keeps its own header and namespace):
#   Magic gift events (namespace gift, was WHG_magic_gift_events)
#   Magic law events (namespace magiclaw, was WHG_magic_law_events)
#   Spell events (namespace spell, was WHG_spell_events)
#   Battle magic events (namespace battlemagic, was WHG_battle_magic_events)
#   Magic commissions (namespace commission, was commissioned_crafting in
#     JW_commisioned_crafting.txt)
#   Enchanting and alchemy (namespace crafting, was WH_crafting_events.txt)

# apprentice.1 A child wants to learn magic
# A child whose guardian is a mage, or whose court has a court mage, asks
# to be taught. Bright, ambitious children ask more often.
# Scope: ROOT is the child.
character_event = {
	id = apprentice.1
	desc = EVTDESCapprentice1
	picture = GFX_evt_apprentice

	min_age = 6

	is_triggered_only = yes # mage_apprentice_wants, title_teacher_mage

	trigger = {
		gender_is_allowed_magic_trigger = yes
		OR = {
			AND = { learning = 2 age = 6 }
			AND = { learning = 4 age = 11 }
			AND = { trait = creature_elf learning = 14 age = 20 }
		}
		OR = {
			age < 26
			check_variable = { which = "mage_teaching" value = 1 }
			trait = creature_elf
		}
		OR = {
			guardian = { is_trained_mage_trigger = yes }
			AND = {
				is_landed = yes
				any_courtier_or_vassal = { has_minor_title = title_teacher_mage }
			}
			AND = {
				is_landed = no
				liege = { any_courtier_or_vassal = { has_minor_title = title_teacher_mage } }
			}
		}
		NOR = {
			trait = hedge_wizard
			trait = mage_apprentice
			has_character_flag = failed_mage
			has_character_flag = priority_mage_recruit
			has_character_modifier = apprenticeship_refused
			has_epidemic = yes
			is_ill = yes
			is_incapable = yes
			is_trained_mage_trigger = yes
		}
	}

	weight_multiplier = {
		days = 1
		modifier = { factor = 4 trait = genius }
		modifier = { factor = 2 trait = brilliant }
		modifier = { factor = 1.5 trait = bright }
		modifier = { factor = 4 trait = ambitious }
		modifier = { factor = 1.25 trait = quick }
		modifier = { factor = 1.25 trait = smart }
		modifier = { factor = 1.25 trait = shrewd }
		modifier = {
			factor = 10
			culture_group = kislevite_group
			is_woman_trigger = yes
			any_courtier_or_vassal = { has_minor_title = title_teacher_mage trait = ice_witch }
		}
		modifier = {
			factor = 10
			trait = creature_elf
			top_liege = { trait = malekith }
			is_woman_trigger = yes
			any_courtier_or_vassal = { has_minor_title = title_teacher_mage }
		}
		modifier = {
			factor = 10
			trait = creature_elf
			NOT = { top_liege = { trait = malekith } }
			any_courtier_or_vassal = { has_minor_title = title_teacher_mage }
		}
		modifier = {
			factor = 5
			learning = 6
			is_female = no
			guardian = { is_trained_mage_trigger = yes trait = arabyan_scholar }
		}
		modifier = {
			factor = 1.25
			is_ruler = yes
			any_courtier_or_vassal = { has_minor_title = title_teacher_mage }
		}
		modifier = {
			factor = 1.25
			is_ruler = no
			liege = { any_courtier_or_vassal = { has_minor_title = title_teacher_mage } }
		}
		modifier = { factor = 0 OR = { trait = imbecile trait = hollowed } }
		modifier = { factor = 0.2 OR = { trait = slothful trait = idiot } }
	}

	option = { # Can you teach me?
		name = EVTOPTAapprentice1
		hidden_effect = {
			if = {
				limit = { guardian = { is_trained_mage_trigger = yes } }
				guardian = { character_event = { id = apprentice.2 } }
			}
			else_if = {
				limit = { is_landed = yes }
				random_courtier_or_vassal = {
					limit = { has_minor_title = title_teacher_mage is_trained_mage_trigger = yes }
					character_event = { id = apprentice.2 }
				}
			}
			else = {
				liege = {
					random_courtier_or_vassal = {
						limit = {
							has_minor_title = title_teacher_mage
							is_trained_mage_trigger = yes
						}
						character_event = { id = apprentice.2 }
					}
				}
			}
		}
	}
}

# apprentice.8 A master dies
# Each unfinished apprentice of a dead mage goes straight to the test
# (apprentice.15, or apprentice.91 for a necromancer's apprentice) instead
# of waiting forever for the next lesson. College students of a Magister
# are left to the college, which finds them a new teacher (college.26).
# Scope: ROOT is the dead character.
character_event = {
	id = apprentice.8
	hide_window = yes
	is_triggered_only = yes # on_death

	trigger = {
		is_trained_mage_trigger = yes
		any_friend = { trait = mage_apprentice has_character_flag = mage_master@ROOT }
	}

	immediate = {
		save_event_target_as = mage_tutor
		any_friend = {
			limit = {
				trait = mage_apprentice
				has_character_flag = mage_master@ROOT
				NAND = {
					has_character_flag = learning_magic_in_college
					ROOT = { is_college_of_magic_member = yes society_rank = 3 }
				}
			}
			clr_character_flag = mage_master@ROOT
			if = {
				limit = { ROOT = { trait = lore_necromancy } }
				character_event = { id = apprentice.91 days = 5 }
			}
			else = { character_event = { id = apprentice.15 days = 5 } }
		}
	}
}

# Every year or so the master chooses the next lesson. After fourteen
# lessons, or when the years of training are over, the apprentice can be
# tested; a pupil under sixteen gets a little more practice first. An adult
# apprentice of a college Magister joins his master's college.

	id = apprentice.10
	desc = EVTDESCapprentice10
	picture = GFX_evt_apprentice

	is_triggered_only = yes # apprentice.4, apprentice.11-14, ward_apprentice, college.24

	trigger = {
		FROM = {
			is_alive = yes
			trait = mage_apprentice
			NOT = { is_trained_mage_trigger = yes }
			NOT = { has_character_flag = failed_mage }
		}
	}

	immediate = {
		FROM = { apprentice_join_master_college_effect = yes }
	}

	option = { # Caution
		name = EVTOPTAapprentice10
		trigger = { FROM = { apprentice_lessons_left_trigger = yes } }
		hidden_effect = {
			FROM = {
				change_variable = { which = "mage_teaching" value = 1 }
				character_event = { id = apprentice.11 }
			}
		}
	}
	option = { # Experiment
		name = EVTOPTBapprentice10
		trigger = { FROM = { apprentice_lessons_left_trigger = yes } }
		hidden_effect = {
			FROM = {
				change_variable = { which = "mage_teaching" value = 1 }
				character_event = { id = apprentice.12 }
			}
		}
	}
	option = { # Focus
		name = EVTOPTCapprentice10
		trigger = { FROM = { apprentice_lessons_left_trigger = yes } }
		hidden_effect = {
			FROM = {
				change_variable = { which = "mage_teaching" value = 1 }
				character_event = { id = apprentice.13 }
			}
		}
	}
	option = { # Test
		name = EVTOPTDapprentice10
		trigger = {
			FROM = {
				age = 16
				OR = {
					check_variable = { which = "mage_teaching" value = 14 }
					magic_training_years_over = yes
				}
			}
		}
		ai_chance = { factor = 3 }
		hidden_effect = {
			FROM = {
				set_variable = { which = "mage_teaching" value = 0 }
				character_event = { id = apprentice.15 }
			}
		}
	}
	option = { # Almost there
		name = EVTOPTEapprentice10
		trigger = {
			FROM = { check_variable = { which = "mage_teaching" value = 14 } NOT = { age = 16 } }
		}
		hidden_effect = { FROM = { character_event = { id = apprentice.14 } } }
	}
}

# apprentice.15 The test
# The apprentice tries to call the winds on his own
# (apprentice_master_test_trigger). On success he takes his master's lore;
# on failure he will never be a mage. Master and parents hear the news.
# Scope: ROOT is the apprentice, event_target:mage_tutor the master (alive
# or dead).
character_event = {
	id = apprentice.15
	desc = {
		text = EVTDESCapprentice15
		trigger = {
			event_target:mage_tutor = { is_alive = yes }
			apprentice_master_test_trigger = yes
		}
	}
	desc = {
		text = EVTDESCapprentice15b
		trigger = {
			event_target:mage_tutor = { is_alive = yes }
			NOT = { apprentice_master_test_trigger = yes }
		}
	}
	desc = { text = EVTDESCapprentice15c trigger = { event_target:mage_tutor = { is_alive = no } } }
	picture = GFX_evt_apprentice

	is_triggered_only = yes # apprentice.8, apprentice.10

	trigger = {
		trait = mage_apprentice
	}

	option = { # This is just the beginning
		name = EVTOPTAapprentice15
		trigger = { apprentice_master_test_trigger = yes }
		remove_trait = mage_apprentice
		prestige = 150
		apprentice_tutor_lore_effect = yes
		apprentice_new_mage_effect = yes
		hidden_effect = {
			event_target:mage_tutor = {
				if = { limit = { is_alive = yes } character_event = { id = apprentice.17 } }
			}
			father = {
				if = {
					limit = { is_alive = yes NOT = { character = event_target:mage_tutor } }
					character_event = { id = apprentice.18 days = 1 }
				}
			}
			mother = {
				if = {
					limit = { is_alive = yes NOT = { character = event_target:mage_tutor } }
					character_event = { id = apprentice.18 days = 1 }
				}
			}
		}
	}
	option = { # Why can't I do it?
		name = EVTOPTBapprentice15
		trigger = { NOT = { apprentice_master_test_trigger = yes } }
		set_character_flag = failed_mage
		remove_trait = mage_apprentice
		prestige = -150
		hidden_effect = {
			clr_learning_flags_effect = yes
			clr_character_flag = learning_magic_in_college
			event_target:mage_tutor = {
				if = { limit = { is_alive = yes } character_event = { id = apprentice.19 } }
			}
			father = {
				if = {
					limit = { is_alive = yes NOT = { character = event_target:mage_tutor } }
					character_event = { id = apprentice.20 days = 1 }
				}
			}
			mother = {
				if = {
					limit = { is_alive = yes NOT = { character = event_target:mage_tutor } }
					character_event = { id = apprentice.20 days = 1 }
				}
			}
		}
	}
}

# A Chaos sorcerer's pupil
# The Ruinous Powers grant magic to their chosen. A sorcerer of Tzeentch,
# Nurgle or Slaanesh preaches his god to his pupil, teaches its tenets and a
# little incantation, and then the god itself judges the pupil. An
# unbeliever is ignored; a believer without the gift, or out of favour, is
# spawned, killed or marred. This chain was written but never reachable;

# apprentice.30 The Ruinous Powers favour only their chosen
# The sorcerer chooses the next lesson: preach the god to an unbeliever,
# teach the tenets to a believer, or teach incantation. After fourteen
# lessons the god tests the pupil.

		FROM = { is_alive = yes trait = mage_apprentice NOT = { is_trained_mage_trigger = yes } }
	}

	option = { # Speak of the god
		name = EVTOPTAapprentice30
		trigger = {
			FROM = {
				NOT = { check_variable = { which = "mage_teaching" value = 14 } }
				NOT = { religion = ROOT }
			}
		}
		hidden_effect = {
			FROM = {
				change_variable = { which = "mage_teaching" value = 1 }
				character_event = { id = apprentice.31 }
			}
		}
	}
	option = { # Teach the tenets
		name = EVTOPTBapprentice30
		trigger = {
			FROM = {
				NOT = { check_variable = { which = "mage_teaching" value = 14 } }
				religion = ROOT
			}
		}
		hidden_effect = {
			FROM = {
				change_variable = { which = "mage_teaching" value = 1 }

		}

					religion = grail
					religion = sigmarite
				}
			}
		}
		NOT = { any_courtier = { has_magic_power_trigger = yes } }
	}

	immediate = {
		if = {
			limit = { religion_group = chaos_gods_group }
			character_event = { id = apprentice.101 }
		}
		else_if = { limit = { religion = northern_gods } character_event = { id = apprentice.102 } }

# Battle magic events
# Namespace: battlemagic (was magicengine in wh_new_battle_magic.txt)
# Faction: Generic
#
# The automatic magic of mages, Khorne's daemon summoners, the Red Duke and
# battle priests who lead troops. In battle, every five days each such
# commander spends a point of his magic pool to conjure spell units (and
# daemons or the dead) for his side; the units vanish when the battle ends.
# In a siege, from the second month on, each one wears down the defences
# with a province modifier based on his magic power, at the cost of a point
# of pool for three months.
#
# Battle                                 battlemagic.1 - battlemagic.10
# Siege                                  battlemagic.20 - battlemagic.25
#
# Callers (common/on_actions/wh_magic_on_actions.txt):
#   battlemagic.1 on on_combat_pulse; battlemagic.6 on on_startup,
#   on_yearly_pulse and on_chronicle_start; battlemagic.10 on the battle won
#   and lost on_actions and on_new_holder_inheritance; battlemagic.20 on
#   on_unit_entering_province, on_command_unit and on_command_sub_unit;
#   battlemagic.21 on on_unit_entering_province.
# Support: battle_magic_casting_effect and calc_magic_pool_effect in
#   common/scripted_effects/wh_magic_scripted_effects.txt, triggers in
#   common/scripted_triggers/wh_battle_magic_triggers.txt.
# Loc: localisation/00_WHG_magic_events.csv
###############################################################################

namespace = battlemagic

# battlemagic.1 The battle begins (hidden)
# The commander, and every casting flank commander of his realm in the same
# province, fills his magic pool and starts casting.
# Scope: ROOT is a commander in battle (on_combat_pulse).
character_event = {
	id = battlemagic.1
	hide_window = yes

	is_triggered_only = yes # on_combat_pulse

	trigger = {
		is_alive = yes
		NOT = { has_character_flag = started_battle_casting }
	}

	immediate = {
		if = {
			limit = { has_character_flag = battle_magic_system_flag in_command = yes }
			set_character_flag = started_battle_casting
			calc_magic_pool_effect = yes
			character_event = { id = battlemagic.2 }
			character_event = { id = battlemagic.3 }
			character_event = { id = battlemagic.4 }
		}
		location = {
			any_province_character = {
				limit = {
					has_character_flag = battle_magic_system_flag
					NOT = { has_character_flag = started_battle_casting }
					in_command = yes
					in_battle = yes
					same_realm = ROOT
				}
				set_character_flag = started_battle_casting
				calc_magic_pool_effect = yes
				character_event = { id = battlemagic.2 }
				character_event = { id = battlemagic.3 }
				character_event = { id = battlemagic.4 }
			}
		}
	}
}

# battlemagic.2 Casting (hidden)
# Every five days of battle the caster spends a point of his pool on spell
# units. The first time a player's mage casts, the player is told how
# battle magic works (battlemagic.5).
# Scope: ROOT is the caster (battlemagic.1, then itself).
character_event = {
	id = battlemagic.2
	hide_window = yes

	is_triggered_only = yes # battlemagic.1
	has_character_flag = battle_magic_system_flag

	trigger = {
		is_alive = yes
		in_battle = yes
		combat = { is_retreating = no }
	}

	immediate = {
		if = {
			limit = { ai = no NOT = { has_character_flag = battle_magic_explained } }
			set_character_flag = battle_magic_explained
			narrative_event = { id = battlemagic.5 }
		}
		else_if = {
			limit = {
				ai = yes
				liege = { ai = no NOT = { has_character_flag = battle_magic_explained } }
			}
			liege = {
				set_character_flag = battle_magic_explained
				narrative_event = { id = battlemagic.5 }
			}
		}
		battle_magic_casting_effect = yes
		# on_combat_pulse is too slow, so the chain times itself
		repeat_event = { id = battlemagic.2 days = 5 }
	}
}

# battlemagic.23 Siege magic (hidden)
# Each month the caster stays in a hostile province, from the second month
# on, he wears down its defences: a siege modifier as strong as his magic
# power (Khorne's summoners and the Red Duke count as power 2), at the cost
# of a point of his battle magic pool for three months.
# Scope: ROOT is the caster (battlemagic.22, then itself).
character_event = {
	id = battlemagic.23
	hide_window = yes

	is_triggered_only = yes # battlemagic.22
	has_character_flag = battle_magic_system_flag

	trigger = {
		at_location = event_target:siege_location
		location = {
			NOT = { owner = { same_realm = ROOT } }
			# At war with the province's realm, or raiding it
			OR = {
				owner = { any_liege = { war_with = ROOT } }
				owner = { war_with = ROOT }
				owner = { ROOT = { any_liege = { war_with = PREVPREV } } }
				owner = { any_liege = { ROOT = { any_liege = { war_with = PREVPREV } } } }
			}
		}
	}

	immediate = {
		# Variables named local_ belong to this event chain only
		change_variable = { which = local_siegemonths value = 1 }
		if = {
			limit = {
				check_variable = { which = local_siegemonths value = 2 }
				battle_magic_system_trigger = yes
			}
			if = {
				limit = { NOT = { has_character_modifier = magic_siege_pool_reduction_timer } }
				character_event = { id = battlemagic.24 }
			}
			magic_xp_effect = yes
			add_character_modifier = {
				name = magic_siege_pool_reduction_timer
				hidden = yes
				duration = 90
			}
			calc_magic_pool_effect = yes
			set_variable = { which = local_siegepower which = cl }
			if = {
				limit = {
					OR = { khorne_summoner_trigger = yes summons_undead_in_battle_trigger = yes }
					NOT = { check_variable = { which = local_siegepower value = 2 } }
				}
				set_variable = { which = local_siegepower value = 2 }
			}
			if = {
				limit = { check_variable = { which = local_siegepower value = 4 } }
				location = {
					add_province_modifier = { name = magic_siege4 duration = 150 stacking = yes }
				}
			}
			else_if = {
				limit = { check_variable = { which = local_siegepower value = 3 } }
				location = {
					add_province_modifier = { name = magic_siege3 duration = 120 stacking = yes }
				}
			}
			else_if = {
				limit = { check_variable = { which = local_siegepower value = 2 } }
				location = {
					add_province_modifier = { name = magic_siege2 duration = 90 stacking = yes }
				}
			}
			else_if = {
				limit = { check_variable = { which = local_siegepower value = 1 } }
				location = {
					add_province_modifier = { name = magic_siege1 duration = 60 stacking = yes }
				}
			}
			if = {
				limit = { ai = no NOT = { has_character_flag = siege_magic_explained } }
				set_character_flag = siege_magic_explained
				narrative_event = { id = battlemagic.25 }
			}
			else_if = {
				limit = {
					ai = yes
					liege = { ai = no NOT = { has_character_flag = siege_magic_explained } }
				}
				liege = {
					set_character_flag = siege_magic_explained
					narrative_event = { id = battlemagic.25 }
				}
			}
		}
		repeat_event = { id = battlemagic.23 days = 30 }
	}
}

# Magic commissions
# Namespace: commission (was commissioned_crafting in
#   JW_commisioned_crafting.txt)
# Faction: Generic
#
# A ruler hires a mage of his court (or a Necrarch, for the undead) to make
# a magic item: a blade, an amulet of protection, a potion, a ring or one of
# the four elixirs. The buyer picks from what he can pay for and what the
# mage is strong enough to make; the mage takes the work or turns it down,
# and the item arrives when the work is done, one to ten years later. A
# buyer waits for one commission at a time, and a mage works on one at a
# time (the crafting_magic_item trait).
#
# Menus                                  commission.1 - commission.15
# Turned down                            commission.18 - commission.19
# The mage's answer                      commission.20
# The item is done                       commission.30
#
# Callers: the commission_crafting decision (decisions/wh_magic_decisions.txt)
#   and the Necrarch's answer vamp.116 (WHG_vampire_events) fire
#   commission.1 for the buyer, with FROM the mage.
# Scripted effects: commission_take_effect, commission_deliver_effect and
#   commission_clear_effect in common/scripted_effects/wh_magic_commission_effects.txt
# Loc: localisation/00_WHG_magic_commission_events.csv
###############################################################################

namespace = commission

# commission.1 A commission is asked for (hidden)
# Saves the buyer and the mage. A buyer still waiting for another item is
# told so (commission.18); a mage already at work turns it down
# (commission.19).
# Scope: ROOT is the buyer; FROM is the mage.
character_event = {
	id = commission.1
	hide_window = yes
	is_triggered_only = yes # commission_crafting decision, vamp.116

	immediate = {
		FROM = { save_event_target_as = wizard }
		save_event_target_as = buyer
		if = {
			limit = { has_character_flag = commission_waiting }
			character_event = { id = commission.18 }
		}
		else_if = {
			limit = { FROM = { trait = crafting_magic_item } }
			character_event = { id = commission.19 }
		}
		else = { character_event = { id = commission.2 } }
	}
}

# Enchanting and alchemy (namespace crafting, was WH_crafting_events.txt;
#   ids kept)
# Faction: Generic
# Serves: every ruler with the magic offmap who is a mage
#
# A mage crafts their own magic items from the Winds of Magic screen:
# amulets of protection, rings, enchanted swords, potions and elixirs. Each
# menu offers what the mage's power and treasury allow; the work takes one
# to five years, then the item is done. (Magic commissions, where a ruler
# hires a court mage, are the commission part above.)
#
#   Amulets                                crafting.0 - crafting.1
#   Rings                                  crafting.2 - crafting.3
#   Swords                                 crafting.4 - crafting.5
#   Potions                                crafting.6 - crafting.7
#   Elixirs                                crafting.8 - crafting.9
#
# Fixes: the Ring of Invisibility charged 200 gold on top of its price;
#   the Ring of Speed and Ring of Command (3 and 5 years by their labels)
#   and the +3 amulet (3 years) were done after one or two; the +5 amulet
#   cost a little more than the treasury check. The potion decision showed
#   the Strength price twice instead of Strength and Speed.
# Fired from: the craft_amulet, craft_ring, craft_weapon, craft_potion and
#   craft_elixir offmap decisions (decisions/wh_magic_decisions.txt).
# Loc: localisation/00_magic.csv
###############################################################################
