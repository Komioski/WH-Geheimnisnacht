###############################################################################
# Magic law events
# Namespace: magiclaw (was wh_magic_law in wh_magic_law.txt)
# Faction: Generic
#
# The realm's magic law (magic_law_0 - 3: burn the gifted, spare the
# gifted but ban learning, allow magic, chartered colleges) and the strife
# it causes. Changing the law between 0, 1 and 2 by vote risks a great
# revolt: a powerful vassal who hates the change raises a faction and goes
# to war. If none rises within 100 days the ruler has won. Factions can also
# demand a change by ultimatum. Fifty years after magic first became lawful
# in the Old World, people grow used to it. A mage who leads a peasant
# revolt to victory makes his new realm a haven for magic.
#
# The great revolt                       magiclaw.1 - magiclaw.9, magiclaw.15
# Ultimatums                             magiclaw.10 - magiclaw.14
# Fifty years of lawful magic            magiclaw.20 - magiclaw.21
# A mage's peasant realm                 magiclaw.22 - magiclaw.23
# Upkeep                                 magiclaw.30 - magiclaw.34
#
# Callers:
#   common/scripted_effects/wh_magic_scripted_effects.txt:
#     magic_law_pass_effect fires magiclaw.1.
#   common/cb_types/wh_anti_magic_cb_types.txt: the revolt CBs fire
#     magiclaw.7 when the ruler wins.
#   decisions/wh_magic_decisions.txt: the faction ultimatums fire
#     magiclaw.10.
#   common/cb_types/00_cb_types.txt: a winning peasant revolt fires
#     magiclaw.22.
#   common/on_actions/wh_magic_on_actions.txt: magiclaw.31 on on_startup,
#     magiclaw.34 on on_yearly_pulse, magiclaw.30 on on_bi_yearly_pulse,
#     magiclaw.20 on on_five_year_pulse.
#   common/scripted_effects/wh_assign_racial_trait.txt: magiclaw.32.
#   common/minor_titles/wh_minor_titles.txt: the court mage title fires
#     magiclaw.33 when it is lost.
# Loc: localisation/00_WHG_magic_law_events.csv
###############################################################################

namespace = magiclaw

# magiclaw.1 The law has changed (hidden)
# Notes whether magic gained or lost rights, and tells the ruler and the
# players among his lords (magiclaw.2).
# Scope: ROOT is the ruler.
character_event = {
	id = magiclaw.1
	hide_window = yes
	is_triggered_only = yes # magic_law_pass_effect

	immediate = {
		magic_law_up_or_down_effect = yes
		save_event_target_as = scoped_magic_ruler
		narrative_event = { id = magiclaw.2 days = 1 }
		any_realm_lord = { limit = { ai = no } narrative_event = { id = magiclaw.2 days = 1 } }
	}
}

# magiclaw.2 The Magic Menace
# The ruler's own copy sets up the great revolt: in 12 days a powerful
# vassal who disagrees is asked to rise (magiclaw.3); after 100 days with no
# revolt the ruler has won (magiclaw.8).
# Scope: ROOT is the ruler or one of his lords, FROM the ruler.
narrative_event = {
	id = magiclaw.2
	title = EVTTITLEmagiclaw2
	desc = {
		text = EVTDESCmagiclaw2
		trigger = {
			event_target:scoped_magic_ruler = { has_character_flag = GivingMoreMagicRights }
		}
	}
	desc = {
		text = EVTDESCmagiclaw2b
		trigger = {
			event_target:scoped_magic_ruler = { has_character_flag = TakingAwayMagicRights }
		}
	}
	picture = GFX_evt_witch_hunters
	border = GFX_event_narrative_frame_religion
	sound = pagan_reformation

	is_triggered_only = yes # magiclaw.1

	option = { # A necessary step
		name = EVTOPTAmagiclaw2
		trigger = { character = FROM }
		hidden_effect = {
			set_character_flag = in_great_magic_revolt
			set_character_flag = waiting_for_rebels
			character_event = { id = magiclaw.8 days = 100 }
			random_vassal = {
				limit = {
					is_landed = yes
					prisoner = no
					mercenary = no
					OR = { is_feudal = yes is_merchant_republic = yes }
				}
				preferred_limit = {
					is_powerful_vassal = yes
					ai = no
					OR = {
						AND = {
							event_target:scoped_magic_ruler = {
								has_character_flag = GivingMoreMagicRights
							}
							hates_legal_magic_trigger = yes
						}
						AND = {
							event_target:scoped_magic_ruler = {
								has_character_flag = TakingAwayMagicRights
							}
							supports_legal_magic_trigger = yes
						}
					}
				}
				preferred_limit = { is_powerful_vassal = yes ai = no }
				preferred_limit = {
					is_powerful_vassal = yes
					OR = {
						AND = {
							event_target:scoped_magic_ruler = {
								has_character_flag = GivingMoreMagicRights
							}
							hates_legal_magic_trigger = yes
						}
						AND = {
							event_target:scoped_magic_ruler = {
								has_character_flag = TakingAwayMagicRights
							}
							supports_legal_magic_trigger = yes
						}
					}
				}
				preferred_limit = { is_powerful_vassal = yes }
				character_event = { id = magiclaw.3 days = 12 }
			}
		}
	}
	option = { # A good choice
		name = EVTOPTBmagiclaw2
		trigger = { NOT = { character = FROM } }
	}
	option = { # The end times are near!
		name = EVTOPTCmagiclaw2
		trigger = { NOT = { character = FROM } }
	}
}

# magiclaw.3 Rise against the ruler?
# A vassal who disagrees with the change may raise a faction against the
# ruler, call the other lords to it (magiclaw.4) and go to war
# (magiclaw.5). If he does not disagree, another great vassal is asked.
# Scope: ROOT is the vassal.
character_event = {
	id = magiclaw.3
	desc = {
		text = EVTDESCmagiclaw3
		trigger = {
			event_target:scoped_magic_ruler = { has_character_flag = GivingMoreMagicRights }
		}
	}
	desc = {
		text = EVTDESCmagiclaw3b
		trigger = {
			event_target:scoped_magic_ruler = { has_character_flag = TakingAwayMagicRights }
		}
	}
	picture = GFX_evt_bad_news
	border = GFX_event_normal_frame_religion
	hide_new = yes

	is_triggered_only = yes # magiclaw.2

	trigger = {
		liege = { character = event_target:scoped_magic_ruler }
		OR = {
			AND = {
				event_target:scoped_magic_ruler = { has_character_flag = GivingMoreMagicRights }
				NOT = { supports_legal_magic_trigger = yes }
			}
			AND = {
				event_target:scoped_magic_ruler = { has_character_flag = TakingAwayMagicRights }
				supports_legal_magic_trigger = yes
			}
		}
	}

	fail_trigger_effect = {
		event_target:scoped_magic_ruler = {
			random_vassal = {
				limit = {
					is_landed = yes
					higher_tier_than = COUNT
					OR = { is_feudal = yes is_merchant_republic = yes }
					OR = {
						AND = {
							event_target:scoped_magic_ruler = {
								has_character_flag = GivingMoreMagicRights
							}
							NOT = { supports_legal_magic_trigger = yes }
						}
						AND = {
							event_target:scoped_magic_ruler = {
								has_character_flag = TakingAwayMagicRights
							}
							supports_legal_magic_trigger = yes
						}
					}
				}
				character_event = { id = magiclaw.3 }
			}
		}
	}

	option = { # We must rise up
		name = EVTOPTAmagiclaw3
		if = {
			limit = {
				event_target:scoped_magic_ruler = { has_character_flag = GivingMoreMagicRights }
			}
			set_character_flag = flag_antimagic_revolter
		}
		else = { set_character_flag = flag_promagic_revolter }
		custom_tooltip = { text = magiclaw_call_to_arms_tt }
		hidden_effect = {
			if = {
				limit = { has_character_flag = flag_antimagic_revolter }
				start_faction = { faction = faction_antimagic_revolters }
			}
			else = { start_faction = { faction = faction_promagic_revolters } }
			event_target:scoped_magic_ruler = {
				any_vassal = {
					limit = { prisoner = no NOT = { character = ROOT } higher_tier_than = BARON }
					letter_event = { id = magiclaw.4 }
				}
			}
			character_event = { id = magiclaw.5 days = 4 }
		}
		ai_chance = { factor = 100 }
	}
	option = { # There is nothing we can do
		name = EVTOPTBmagiclaw3
		trigger = { ai = no }
	}
}

# magiclaw.5 To war!
# The revolt's leader breaks from the ruler and declares war; the ruler and
# the players among his vassals hear of it (magiclaw.6). The ruler is
# pinged so that rebel troops can join (magiclaw.14).
# Scope: ROOT is the revolt's leader.
character_event = {
	id = magiclaw.5
	desc = EVTDESCmagiclaw5
	picture = GFX_evt_burning_stake
	border = GFX_event_normal_frame_war
	hide_new = yes

	is_triggered_only = yes # magiclaw.3

	trigger = {
		liege = { character = event_target:scoped_magic_ruler }
	}

	option = { # To war!
		name = EVTOPTAmagiclaw5
		hidden_effect = {
			set_character_flag = ai_flag_refuse_conversion
			event_target:scoped_magic_ruler = {
				clr_character_flag = waiting_for_rebels
				character_event = { id = magiclaw.6 }
				any_vassal = {
					limit = { is_playable = yes ai = no }
					character_event = { id = magiclaw.6 }
				}
			}
			top_liege = { ROOT = { set_defacto_liege = PREV } }
			event_target:scoped_magic_ruler = {
				if = {
					limit = { has_character_flag = GivingMoreMagicRights }
					any_vassal = {
						limit = { in_faction = faction_antimagic_revolters }
						event_target:scoped_magic_ruler = {
							opinion = { who = PREV modifier = opinion_charge_traitor months = 240 }
						}
					}
				}
				else = {
					any_vassal = {
						limit = { in_faction = faction_promagic_revolters }
						event_target:scoped_magic_ruler = {
							opinion = { who = PREV modifier = opinion_charge_traitor months = 240 }
						}
					}
				}
			}
		}
		if = {
			limit = {
				event_target:scoped_magic_ruler = { has_character_flag = GivingMoreMagicRights }
			}
			liege = {
				reverse_war = {
					target = ROOT
					casus_belli = antimagic_revolt
					faction = faction_antimagic_revolters
				}
			}
		}
		else = {
			liege = {
				reverse_war = {
					target = ROOT
					casus_belli = promagic_revolt
					faction = faction_promagic_revolters
				}
			}
		}
		prestige = 500
		piety = 200
		event_target:scoped_magic_ruler = { character_event = { id = magiclaw.14 } }
		ai_chance = { factor = 100 }
	}
}

# magiclaw.10 The faction's ultimatum
# A magic law faction demands one step its way: the anti-magic revolters
# want fewer rights, the pro-magic revolters more. Giving in changes the law
# and tells the faction (magiclaw.11 - 13); refusing means war, and a weak
# faction may draw rebel bands, a strong one loyalists for the ruler.
# Scope: ROOT is the ruler, FROM the faction leader.
letter_event = {
	id = magiclaw.10
	desc = EVTDESCmagiclaw10

	is_triggered_only = yes # faction ultimatum decisions

	trigger = {
		FROM = {
			OR = {
				in_faction = faction_antimagic_revolters
				in_faction = faction_promagic_revolters
			}
		}
	}

	immediate = {
		FROM = {
			diplomatic_immunity = yes
			if = {
				limit = { in_faction = faction_antimagic_revolters }
				any_faction_backer = {
					faction = faction_antimagic_revolters
					diplomatic_immunity = yes
				}
			}
			else = {
				any_faction_backer = {
					faction = faction_promagic_revolters
					diplomatic_immunity = yes
				}
			}
		}
	}

	option = { # Accept
		name = EVTOPTAmagiclaw10
		ai_chance = {
			factor = 20
			modifier = {
				factor = 0
				FROM = { in_faction = faction_antimagic_revolters }
				NOT = {
					FROM = { faction_power = { faction = faction_antimagic_revolters power = 0.5 } }
				}
			}
			modifier = {
				factor = 0
				FROM = { in_faction = faction_promagic_revolters }
				NOT = {
					FROM = { faction_power = { faction = faction_promagic_revolters power = 0.5 } }
				}
			}
			modifier = { factor = 1.25 is_weak_trigger = yes }
			modifier = { factor = 1.25 trait = craven }
			modifier = { factor = 1.25 trait = humble }
			modifier = { factor = 1.25 trait = kind }
			modifier = { factor = 1.25 trait = charitable }
			modifier = { factor = 1.25 trait = patient }
			modifier = { factor = 1.25 trait = content }
		}
		hidden_effect = {
			any_vassal = {
				limit = {
					ai = no
					NOR = {
						in_faction = faction_antimagic_revolters
						in_faction = faction_promagic_revolters
					}
				}
				character_event = { id = magiclaw.12 }
			}
			FROM = {
				letter_event = { id = magiclaw.13 }
				if = {
					limit = { in_faction = faction_antimagic_revolters }
					any_faction_backer = {
						faction = faction_antimagic_revolters
						letter_event = { id = magiclaw.11 }
						diplomatic_immunity = no
					}
				}
				else = {
					any_faction_backer = {
						faction = faction_promagic_revolters
						letter_event = { id = magiclaw.11 }
						diplomatic_immunity = no
					}
				}
				diplomatic_immunity = no
			}
		}
		FROM = {
			if = {
				limit = { in_faction = faction_antimagic_revolters }
				faction_antimagic_revolters = { # The faction's target title
					if = {
						limit = { has_law = magic_law_2 }
						add_law = { law = magic_law_1 cooldown = no opinion_effect = no }
					}
					else_if = {
						limit = { has_law = magic_law_1 }
						add_law = { law = magic_law_0 cooldown = no opinion_effect = no }
					}
				}
			}
			else = {
				faction_promagic_revolters = { # The faction's target title
					if = {
						limit = { has_law = magic_law_1 }
						add_law = { law = magic_law_2 cooldown = no opinion_effect = no }
					}
					else_if = {
						limit = { has_law = magic_law_0 }
						add_law = { law = magic_law_1 cooldown = no opinion_effect = no }
					}
				}
			}
		}
	}
	option = { # Refuse: war
		name = EVTOPTBmagiclaw10
		ai_chance = {
			factor = 80
			modifier = { factor = 1.25 is_strong_trigger = yes }
			modifier = { factor = 1.25 trait = brave }
			modifier = { factor = 1.25 trait = proud }
			modifier = { factor = 1.25 trait = envious }
			modifier = { factor = 1.25 trait = greedy }
			modifier = { factor = 1.25 trait = wroth }
			modifier = { factor = 1.25 trait = ambitious }
			modifier = {
				factor = 0.75
				FROM = {
					OR = {
						AND = {
							in_faction = faction_antimagic_revolters
							faction_power = { faction = faction_antimagic_revolters power = 1.5 }
						}
						AND = {
							in_faction = faction_promagic_revolters
							faction_power = { faction = faction_promagic_revolters power = 1.5 }
						}
					}
				}
			}
			modifier = {
				factor = 0.5

# magiclaw.30 A new realm without a magic law (hidden)
# The game gives no default law to a vassal who becomes independent. Where
# the law may be more than 0, he gets law 2 once magic has been lawful for
# fifty years, a random 0 - 2 while it is new, and law 0 otherwise.
# Scope: ROOT is an independent ruler.
character_event = {
	id = magiclaw.30
	hide_window = yes
	is_triggered_only = yes # on_bi_yearly_pulse
	only_rulers = yes
	only_independent = yes

	trigger = {
		magic_law_potential_trigger = yes
		NOT = { has_magic_law_any_trigger = yes }
	}

	immediate = {
		set_character_flag = magic_laws_forced
		if = {
			limit = {
				has_global_flag = magic_legal_old_world_50_years
				magic_law_1_and_above_allow_trigger = yes
			}
			primary_title = { add_law = { law = magic_law_2 cooldown = no opinion_effect = no } }
		}
		else_if = {
			limit = {
				has_global_flag = magic_legal_old_world
				magic_law_1_and_above_allow_trigger = yes
			}
			random_list = {
				20 = {
					primary_title = {
						add_law = { law = magic_law_0 cooldown = no opinion_effect = no }
					}
				}
				20 = {
					primary_title = {
						add_law = { law = magic_law_1 cooldown = no opinion_effect = no }
					}
				}
				20 = {
					primary_title = {
						add_law = { law = magic_law_2 cooldown = no opinion_effect = no }
					}
				}
			}
		}
		else = {
			primary_title = { add_law = { law = magic_law_0 cooldown = no opinion_effect = no } }
		}
		clr_character_flag = magic_laws_forced
	}
}
