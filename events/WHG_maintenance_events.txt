# aiclean.1 Nomad heir keeps the faith
# When a nomadic ruler dies, an heir of another religion group takes the
# ruler's faith. A government flip between chaos_horde_government and
# nomadic_government crashes the game, so the heir must match. A greenskin
# heir keeps the greenskin faith (they take a greenskin horde government).
# Scope: ROOT is the dying nomad.
character_event = {
	id = aiclean.1
	hide_window = yes

	only_rulers = yes

	is_triggered_only = yes

	trigger = {
		is_nomadic = yes
	}

	immediate = {
		# Greenskins never leave their gods; their heir takes a greenskin horde
		# government, so the chaos horde / nomad flip cannot happen
		current_heir = {
			if = {
				limit = {
					NOT = { religion_group = ROOT }
					NOT = { religion_group = greenskin_gods_group }
				}
				religion = ROOT
			}
		}
	}

	option = {
		name = OK
	}
}

# aiclean.7 Growing old
# Every five years a human, Skaven or halfling of 50, or a dwarf or Chaos
# Dwarf of 200 (lifespan_age_50_trigger), gets the old-age trait that matches
# their looks; women become infertile and men less fertile, once. Old dwarfs
# get the grey-bearded portraits. Elves, immortals, Chaos champions and ice
# witches do not age this way.
# Scope: ROOT is the character.
character_event = {
	id = aiclean.7
	hide_window = yes

	min_age = 50

	is_triggered_only = yes

	trigger = {
		OR = {
			trait = creature_human
			trait = creature_skaven
			trait = creature_halfling
			lifespan_dwarf_trigger = yes
		}
		lifespan_age_50_trigger = yes
		NOR = {
			is_immortal_trigger = yes
			is_chaos_champion_chosen_or_prince_trigger = yes
			trait = lore_ice
		}
	}

	immediate = {
		if = {
			limit = { NOR = { trait = old_1 trait = old_2 trait = old_3 trait = old_4 } }
			if = { limit = { trait = fair } add_trait = old_2 }
			else_if = { limit = { trait = pretty } add_trait = old_3 }
			else_if = { limit = { trait = beautiful } add_trait = old_4 }
			else = { add_trait = old_1 }
			if = { limit = { female = yes } add_trait = infertile }
			else = { fertility = -0.2 }
		}
		if = {
			limit = {
				trait = creature_chaos_dwarf
				NOR = {
					trait = transgender
					graphical_culture = olddwarfgfx
					graphical_culture = oldchaosdwarfgfx
					graphical_culture = slayerdwarfgfx
				}
			}
			set_graphical_culture = dwarf_chaos_old
		}
		if = {
			limit = {
				trait = creature_dwarf
				NOR = {
					trait = transgender
					graphical_culture = olddwarfgfx
					graphical_culture = oldchaosdwarfgfx
					graphical_culture = slayerdwarfgfx
				}
			}
			set_graphical_culture = dwarf_old
		}
		if = {
			limit = {
				trait = creature_chaos_dwarf
				trait = transgender
				NOR = {
					graphical_culture = olddwarfgfx
					graphical_culture = oldchaosdwarfgfx
					graphical_culture = slayerdwarfgfx
				}
			}
			set_graphical_culture = oldchaosdwarfrev
		}
		if = {
			limit = {
				trait = creature_dwarf
				trait = transgender
				NOR = {
					graphical_culture = olddwarfgfx
					graphical_culture = oldchaosdwarfgfx
					graphical_culture = slayerdwarfgfx
				}
			}
			set_graphical_culture = olddwarfrev
		}
	}
}

# aiclean.14 Dwarf ranks at game start
# Gives starting dwarf men the rank that fits their age and skills:
# beardling under 75, then thunderer, quarreller, hammerer or warrior, and
# longbeard from 250. Cowards are flagged and never become warriors.
# Scope: ROOT is the dwarf.
character_event = {
	id = aiclean.14
	hide_window = yes

	religion_group = dwarf_gods_group
	only_men = yes
	min_age = 16
	max_age = 300

	is_triggered_only = yes

	trigger = {
		is_save_game = no
		trait = creature_dwarf
		NOR = {
			trait = dwarf_slayer_king
			trait = dwarf_slayer
			trait = longbeard
			trait = ironbreaker
			has_character_flag = dwarf_did_not_become_warrior
			has_character_flag = becoming_thunderer
			has_character_flag = becoming_quarreller
			has_character_flag = becoming_dwarf_warrior
			AND = { NOT = { age = 75 } trait = beardling }
			AND = {
				NOT = { age = 250 }
				OR = { trait = dwarf_warrior trait = quarreller trait = hammerer trait = thunderer }
			}
		}
	}

	immediate = {
		remove_trait = beardling
		if = { limit = { NOT = { age = 75 } } add_trait = beardling }
		else_if = {
			limit = {
				age = 75
				diplomacy = 12
				NOR = {
					trait = quarreller
					trait = hammerer
					trait = dwarf_warrior
					trait = thunderer
				}
			}
			add_trait = thunderer
		}
		else_if = {
			limit = {
				age = 75
				stewardship = 12
				NOR = {
					trait = quarreller
					trait = hammerer
					trait = dwarf_warrior
					trait = thunderer
				}
			}
			add_trait = quarreller
		}
		else_if = {
			limit = {
				age = 75
				martial = 15
				NOR = {
					trait = quarreller
					trait = hammerer
					trait = dwarf_warrior
					trait = thunderer
				}
			}
			add_trait = hammerer
		}
		else_if = {
			limit = { age = 75 trait = craven }
			set_character_flag = dwarf_did_not_become_warrior
		}
		else_if = {
			limit = { age = 250 }
			remove_trait = quarreller
			remove_trait = hammerer
			remove_trait = dwarf_warrior
			remove_trait = thunderer
			add_trait = longbeard
		}
		else = { add_trait = dwarf_warrior }
	}

	option = {
		name = OK
	}
}

# spawns too many of them.
# Scope: ROOT is the ruler.
character_event = {
	id = aiclean.20
	hide_window = yes

	only_rulers = yes
	ai = yes

	is_triggered_only = yes

	trigger = {
		lower_tier_than = KING
		OR = {
			any_courtier = { trait = skaven_breeder count = 5 }
			any_courtier = { trait = khemri_soldier count = 10 }
		}
	}

	immediate = {
		any_courtier = {
			limit = { trait = skaven_breeder }
			death = { death_reason = death_missing }
		}
		any_courtier = {
			limit = { trait = khemri_soldier }
			death = { death_reason = death_missing }
		}
	}

	option = {
		name = OK
	}
}

# aiclean.30 Game start setup
# At the start of a new game: Skaven surface tunnels under the burrows, secret
# faiths for members of the Chaos and Atharti cults, and male holders of
# female-led titles vanish (aiclean.26 has already replaced them).
# Scope: ROOT is any character.
character_event = {
	id = aiclean.30
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		is_save_game = no
	}

	immediate = {
		# Set surface tunnel modifiers for Skaven
		any_province = {
			limit = {
				region = world_surface_burrows
				NOT = { has_province_modifier = skaven_surface_tunnels }
			}
			add_province_modifier = { name = skaven_surface_tunnels duration = -1 }
		}
		# Set Religions for secret cults
		trigger_switch = {
			on_trigger = society_member_of
			cult_of_khorne = { set_secret_religion = khorne }
			cult_of_nurgle = { set_secret_religion = nurgle }
			cult_of_tzeentch = { set_secret_religion = tzeentch }
			cult_of_slaanesh = { set_secret_religion = slaanesh }
			cult_of_undivided = { set_secret_religion = chaos }
			cult_of_pleasure = { set_secret_religion = atharti }
		}
		# Kills male rulers of female held titles
		if = {
			limit = { is_female = no is_female_led_random_ruler = yes }
			if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
			else = { death = { death_reason = death_missing } }
		}
	}

	option = {
		name = OK
	}
}

# aiclean.32 Wrong faith

	id = aiclean.32

		NOT = { has_character_flag = aborted_religion_fix }

			# Some events (an Amazon running away, for one) legitimately move a
			# character into another faith. Give up after 50 tries rather than
			# loop forever.
			limit = { check_variable = { which = "tried_to_correct_religion" value = 50 } }

# aiclean.44 The Red Duke's bloodline
# At game start removes the Red Duke's bloodline from anyone but him.
# Scope: ROOT is any character.
character_event = {
	id = aiclean.44
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		any_character = {
			limit = {
				NOT = { trait = special_char_red_duke }
				any_owned_bloodline = { has_bloodline_flag = warhammer_bloodline_red_duke }
			}
			any_owned_bloodline = {
				limit = { has_bloodline_flag = warhammer_bloodline_red_duke }
				remove_bloodline_member = PREV
			}
		}
	}
}

# aiclean.45 Burnt by the flames
# A failed walk through the Flames of Asuryan founds the burnt bloodline,
# unless the redeemed one is already active. Fired by asur.3 and asur.4.
# Scope: ROOT is the burnt elf.
character_event = {
	id = aiclean.45
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				NOT = {
					any_owned_bloodline = {
						has_bloodline_flag = burnt_by_flames_redeemed
						bloodline_is_active_for = PREV
					}
				}
			}
			if = { limit = { is_female = no } create_bloodline = { type = burnt_by_flames_male } }
			if = {
				limit = { is_female = yes }
				create_bloodline = { type = burnt_by_flames_female }
			}
		}
	}
}

# aiclean.46 Redeemed by the flames
# A descendant of the burnt passes through the flames unharmed and is crowned:
# 500 prestige, 1000 piety and the redeemed bloodline for the whole living
# dynasty. Fired by the Phoenix King crowning.
# Scope: ROOT is the new Phoenix King.
narrative_event = {
	id = aiclean.46
	title = EVTNAMEaiclean46
	desc = EVTDESCaiclean46
	picture = GFX_evt_phoenix_king
	sound = bloodline_added

	is_triggered_only = yes

	option = {
		name = EVTOPTAaiclean46
		prestige = 500
		piety = 1000
		if = {
			limit = { is_female = no }
			create_bloodline = { type = burnt_by_flames_redeemed_male }
			any_dynasty_member = {
				limit = { NOT = { trait = burnt_by_asuryan } }
				new_bloodline = { add_bloodline_member = PREV }
			}
		}
		if = {
			limit = { is_female = yes }
			create_bloodline = { type = burnt_by_flames_redeemed_female }
			any_dynasty_member = {
				limit = { NOT = { trait = burnt_by_asuryan } }
				new_bloodline = { add_bloodline_member = PREV }
			}
		}
		custom_tooltip = { text = aiclean_bloodline_dynasty_tt }
	}
}

# Faction: Generic

#
# offmapgen.1 empties the six Chaos warp currencies of a dead Chaos ruler,

# offmapgen.2 rolls the yearly Winds of Magic a ruler can draw on: a base
#   by magic potential (4, 6, 8 or 10), then a roll of +0 to +8, where the
#   step that matches the current doom is twice as likely. Rulers without
#   any magic potential get a flat 10.
#
# Fired from: common/on_actions/WHG_character_on_actions.txt: offmapgen.1 on on_death,
#   offmapgen.2 on on_startup and on_yearly_pulse.
###############################################################################

# offmapgen.1 A dead Chaos ruler's warp favour is emptied.
character_event = {
	id = offmapgen.1
	hide_window = yes

	religion_group = chaos_gods_group
	only_playable = yes

	is_triggered_only = yes # on_death (WHG_character_on_actions.txt)

	immediate = {
		set_offmap_currency = { offmap = undivided_warp value = 0 }
		set_offmap_currency = { offmap = slaanesh_warp value = 0 }
		set_offmap_currency = { offmap = nurgle_warp value = 0 }
		set_offmap_currency = { offmap = tzeentch_warp value = 0 }
		set_offmap_currency = { offmap = khorne_warp value = 0 }
		set_offmap_currency = { offmap = brayherd_warp value = 0 }
	}
}

# offmapgen.2 The Winds of Magic blow: set this year's magic for a ruler.
character_event = {
	id = offmapgen.2
	hide_window = yes

	only_playable = yes

	is_triggered_only = yes # on_startup, on_yearly_pulse (common/on_actions)

	trigger = {
		magic_offmap_enabled_trigger = yes
	}

	immediate = {
		if = {
			limit = { trait = magic_potential_1 }
			set_offmap_currency = { offmap = offmap_magic value = 4 }
		}
		else_if = {
			limit = { trait = magic_potential_2 }
			set_offmap_currency = { offmap = offmap_magic value = 6 }
		}
		else_if = {
			limit = { trait = magic_potential_3 }
			set_offmap_currency = { offmap = offmap_magic value = 8 }
		}
		else = {
			# magic_potential_4 starts at 10, and so does a ruler without the gift
			set_offmap_currency = { offmap = offmap_magic value = 10 }
		}
		if = {
			limit = {
				OR = {
					trait = magic_potential_1
					trait = magic_potential_2
					trait = magic_potential_3
					trait = magic_potential_4
				}
			}
			random_list = {
				20 = {
					modifier = { factor = 2 is_doom_very_low_trigger = yes }
				}
				20 = {
					modifier = { factor = 2 is_doom_low_trigger = yes }
					add_offmap_currency = { offmap = offmap_magic value = 2 }
				}
				20 = {
					modifier = { factor = 2 is_doom_medium_trigger = yes }
					add_offmap_currency = { offmap = offmap_magic value = 4 }
				}
				20 = {
					modifier = { factor = 2 is_doom_high_trigger = yes }
					add_offmap_currency = { offmap = offmap_magic value = 6 }
				}
				20 = {
					modifier = { factor = 2 is_doom_very_high_trigger = yes }
					add_offmap_currency = { offmap = offmap_magic value = 8 }
				}
			}
		}
	}
}

# Faction: Generic
#
# One-off set-up for the Shattered World and Random World starts, run by the
# first human player on a new game:
#   altstart.1  Great Conquerors: spawns 4, 8 or 12 conquerors in a Shattered
#               World, as the game rule asks
#   altstart.2  Shattered succession: gives every feudal and tribal realm the
#               succession law the game rule asks for
#   altstart.3  General clean-up: renamed mercenary bands and holy orders,
#               new coats of arms for renamed faiths' heads, a shipyard and
#               naval tech for every port so no one is stranded on an island,
#               and a chance of cognatic succession
#
# Fired from on_startup (common/on_actions/WHG_startup_on_actions.txt).
# Saved: global flags conquerors_setup, succession_setup,
#   general_maintenance_performed; event targets renaming_target, coa_target
#   (set_random_coa_effect).
# Scripted triggers: common/scripted_triggers/wh_altstart_triggers.txt
# Scripted effects: common/scripted_effects/wh_altstart_effects.txt and
#   02_scripted_effects_alternate_start.txt (vanilla copy)
# Loc: the vanilla ALT_* keys (HolyFury.csv), plus ALT_MERC_INQUISITION and
#   ALT_MERC_RAIDERS in localisation/00_WHG_character_events.csv
# Not carried over: HF.3 (Great Holy Wars and the Hashshashin for random
#   faith names; its caller was already switched off and d_hashshashin is
#   not a mod title). The unicorn prophecy HF.6 - HF.11 moved to
#   WHG_character_events.txt as charfl.70 - 75.
# Full list with callers: docs/events/event_database.csv
###############################################################################

			limit = { chaos = { is_name_randomized = yes } }
			k_the_brass_legion = { altstart_rename_holy_order_effect = yes }
		}
		if = {
			limit = { ormazd = { is_name_randomized = yes } }
			d_hashishin = { altstart_rename_holy_order_effect = yes }
		}
		if = {
			limit = { amazon = { is_name_randomized = yes } }
			d_kalim_devouts = { altstart_rename_holy_order_effect = yes }
		}
		if = {
			limit = { kislev_gods = { is_name_randomized = yes } }
			d_sons_ursa = { altstart_rename_holy_order_effect = yes }
		}
		# Being stranded on an island is no fun, whatever the setting
		any_independent_ruler = {
			any_realm_province = {
				limit = { port = yes }
				if = {
					limit = { NOT = { TECH_NAVAL = 1 } }
					change_tech = {
						technology = TECH_NAVAL
						value = 1
					}
				}
				any_province_holding = {
					altstart_port_shipyard_effect = yes
				}
			}
		}

		# A chance of cognatic succession, certain for a queen without sons
		any_playable_ruler = {
			limit = {
				is_landed = yes
				is_republic = no
				has_law = agnatic_succession
				NOT = { has_religion_feature = religion_patriarchal }
				any_demesne_title = { is_law_potential = cognatic_succession }
			}
			if = {
				limit = {
					is_woman_trigger = yes
					NOT = { any_child = { is_female = no } }
				}
				altstart_cognatic_effect = yes
			}
			else_if = {
				limit = {
					OR = {
						AND = {
							is_woman_trigger = yes
							any_child = { is_female = no }
						}
						AND = {
							is_female = no
							NOT = { any_child = { is_female = no } }
						}
					}
				}
				random = {
					chance = 90
					altstart_cognatic_effect = yes
				}
			}
			else_if = {
				limit = { is_woman_trigger = yes }
				random = {
					chance = 70
					altstart_cognatic_effect = yes
				}
			}
			else = {
				random = {
					chance = 50
					altstart_cognatic_effect = yes
				}
			}
			recalc_succession = yes
		}
	}
}

# gamerule.30 A family for the lonely
# on_startup, rule generate_families: landed rulers without a family get a
# spouse (except the peoples without women) and children, so the realms of
# the 2010 start have heirs.
# Scope: ROOT is the ruler.
character_event = {
	id = gamerule.30
	hide_window = yes

	is_triggered_only = yes # on_startup

	only_playable = yes
	min_age = 16

	trigger = {
		is_save_game = no
		is_landed = yes
		mercenary = no
		holy_order = no
		gamerule_family_rule_trigger = yes
		gamerule_family_people_trigger = yes
		OR = { is_married = no AND = { num_of_children < 1 spouse = { is_lowborn = yes } } }
		OR = { is_republic = yes is_feudal = yes is_nomadic = yes is_tribal = yes }
		NOR = {
			government = confucian_bureaucracy
			trait = eunuch
			trait = celibate
			has_nickname = nick_the_chaste
		}
		gamerule_family_lonely_trigger = yes
	}

	immediate = {
		# The parents of any new children
		if = {
			limit = { is_female = no }
			save_event_target_as = father_target
			spouse = { save_event_target_as = mother_target }
		}
		else = {
			save_event_target_as = mother_target
			spouse = { save_event_target_as = father_target }
		}
		# A spouse first
		if = {
			limit = { is_married = no can_marry = yes is_no_females_race = no }
			if = { limit = { is_female = no } gamerule_wife_effect = yes }
			else = { gamerule_husband_effect = yes }
		}
		# Then children
		if = { limit = { num_of_children < 1 } gamerule_children_effect = yes }
		# Historical children of a lord without a mother get the new wife
		if = {
			limit = { is_female = no is_married = yes }
			spouse = { save_event_target_as = new_mother }
			any_child = {
				limit = {
					NOR = { mother = { always = yes } trait = legit_bastard trait = bastard }
					event_target:new_mother = { could_be_parent_of = PREV }
				}
				set_mother = event_target:new_mother
			}
		}
		# Grown children who never got traits
		any_child = {
			limit = {
				age = 16
				NOR = {
					has_education_diplomacy_trigger = yes
					has_education_martial_trigger = yes
					has_education_learning_trigger = yes
					has_education_stewardship_trigger = yes
				}
			}
			add_random_education_trait = yes
			gamerule_copy_personality_effect = yes
			gamerule_copy_personality_effect = yes
			gamerule_copy_personality_effect = yes
		}
		any_child = {
			limit = { age = 10 NOT = { age = 16 } NOT = { personality_traits = 2 } }
			gamerule_copy_personality_effect = yes
			gamerule_copy_personality_effect = yes
		}
		any_child = {
			limit = { age = 6 NOT = { age = 10 } NOT = { personality_traits = 1 } }
			gamerule_copy_personality_effect = yes
		}
		recalc_succession = yes
	}
}
