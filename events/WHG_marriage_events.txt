###############################################################################
# Married life: quarrels, poison, gifts, advice and the marriage bed
# Namespace: marriage (was 30000 - 30110 in the vanilla married_life_events)
# Faction: Generic
#
# The everyday life of a married lord and lady: the cold bed and the
# divorce, the stepmother who poisons the heir, the paranoid husband, the
# spouse who prefers men, the spouse who tries to free a captive partner,
# the spouse who wants a councillor gone (or in the river), gifts, a
# clever spouse's advice, the lady who puts her favourites on the council,
# the child of another man, the bored young wife, the old husband worn out
# by love, and the barren marriage.
#
# Quarrels and divorce                 marriage.1 - marriage.4
# The poisoned heir                    marriage.10 - marriage.17
# Accusations of poison                marriage.20 - marriage.21
# A husband who prefers men            marriage.25 - marriage.29
# A captive spouse                     marriage.30 - marriage.37
# The spouse and the councillor        marriage.40 - marriage.47
# Gifts                                marriage.50 - marriage.52
# A clever spouse's advice             marriage.55 - marriage.59
# The lady's favourites                marriage.60 - marriage.62
# Another man's child                  marriage.65 - marriage.71
# The bored young wife                 marriage.75 - marriage.78
# The marriage bed                     marriage.80 - marriage.83
# The barren marriage                  marriage.85
#
# Fired from: the on_yearly_pulse random_events of 00_on_actions.txt (1, 2,
#   10, 20, 21, 25, 30, 40, 45, 50, 51, 52, 55 - 59, 60, 65, 75, 80, 85);
#   the rest from each other.
# Effects: common/scripted_effects/WHG_marriage_effects.txt
# Loc: localisation/00_WHG_marriage_events.csv
#
# Merged: the five "wife suggests a councillor" chains (30060 - 30074) are
#   one chain (60 - 62); the Ormazd learning advice (30110) is a second desc
#   of marriage.59; the two "loved him to death" endings are marriage.83.
# Fixed: unlanded wives looked for candidates among their own (empty)
#   courts; the lake and picnic outcomes of the bored wife were swapped and
#   the courtier's answer ran on flags it never had; the captive spouse's
#   would-be rescuer had to stand in the same castle as the captive, the
#   ransom paid the jailer from nowhere and the dungeon went to the wrong
#   spouse; advice could make an ambitious lord content; the old vanilla
#   `treasury` and `break` are gone; executions spare the invincible
#   everywhere, not only in some options; the love that kills an old
#   husband now follows how fast each race ages.
# Changed: advice and gifts go both ways, so a clever husband advises his
#   lady too and a lady may give gifts (races without women excluded).
###############################################################################

namespace = marriage

# marriage.1 Separate bedchambers
# A sick, mad or wretched lord is asked by a cold spouse to sleep apart.
# Scope: a married lord or lady.
character_event = {
	id = marriage.1
	desc = EVTDESCmarriage1
	picture = GFX_evt_quarrel
	is_triggered_only = yes # on_yearly_pulse
	min_age = 20
	capable_only = yes
	prisoner = no
	is_married = yes

	trigger = {
		is_pregnant = no
		spouse = {
			is_pregnant = no
			prisoner = no
			is_abroad = no
			at_location = ROOT
			NOT = { trait = incapable }
			NOT = { opinion = { who = ROOT value = -20 } }
		}
		OR = {
			trait = stressed
			trait = depressed
			trait = lunatic_1
			trait = lunatic_2
			trait = lunatic_3
			trait = possessed
			is_ill = yes
			is_maimed_trigger = yes
			trait = infirm
			trait = drunkard
		}
	}

	weight_multiplier = {
		days = 1
		modifier = {
			factor = 1.5
			OR = {
				trait = lunatic_1
				trait = lunatic_2
				trait = lunatic_3
				trait = possessed
				trait = has_typhus
				trait = has_bubonic_plague
			}
		}
		modifier = { factor = 2 OR = { trait = leper trait = infirm } }
		modifier = {
			factor = 1.25
			OR = { trait = pneumonic trait = syphilitic is_maimed_trigger = yes }
		}
	}

	option = { # gladly
		name = EVTOPTAmarriage1
		prestige = 2
		spouse = { opinion = { modifier = insulted who = ROOT } }
	}
	option = { # win the spouse back
		name = EVTOPTBmarriage1
		piety = 2
		spouse = { opinion = { modifier = opinion_rekindle_love who = ROOT years = 5 } }
	}
	option = { # as you wish
		name = EVTOPTCmarriage1
		prestige = -5
		piety = 5
		spouse = { opinion = { modifier = opinion_whipped who = ROOT years = 1 } }
	}
}

# marriage.2 Put her aside
# A Sigmarite or Myrmidian lord with a lawful heir and a childless wife who
# despises him may cast her out.
# Scope: a lord.
character_event = {
	id = marriage.2
	desc = EVTDESCmarriage2
	picture = GFX_evt_quarrel
	is_triggered_only = yes # on_yearly_pulse
	max_age = 60
	only_men = yes
	capable_only = yes
	prisoner = no
	is_married = yes

	trigger = {
		OR = { religion = sigmarite religion = myrmidian }
		any_child = { is_primary_heir = yes NOT = { trait = bastard } }
		spouse = {
			is_woman_trigger = yes
			prisoner = no
			is_abroad = no
			at_location = ROOT
			NOT = { trait = incapable }
			NOT = { num_of_children = 1 }
			NOT = { opinion = { who = ROOT value = -50 } }
		}
	}

	option = { # out of my sight
		name = EVTOPTAmarriage2
		prestige = -5
		spouse = {
			random = {
				chance = 25
				character_event = { id = marriage.3 tooltip = EVTTOOLTIPmarriage3 }
			}
		}
		remove_spouse = spouse
	}
	option = { # one more try
		name = EVTOPTBmarriage2
		piety = 5
		spouse = { reverse_opinion = { modifier = opinion_rekindle_love who = ROOT years = 2 } }
	}
}

# marriage.10 The stepmother's poison
# An ambitious, envious lady whose own child has a claim looks at her
# husband's heir by another woman. Poison makes him ill, crippled or dead,
# and someone may notice.
# Scope: a lady.
character_event = {
	id = marriage.10
	desc = EVTDESCmarriage10
	picture = GFX_evt_shadow
	is_triggered_only = yes # on_yearly_pulse
	only_women = yes
	capable_only = yes
	prisoner = no

	trigger = {
		is_woman_trigger = yes
		trait = ambitious
		trait = envious
		NOT = { any_child = { is_primary_heir = yes } }
		any_child = { is_pretender = yes }
		spouse = {
			is_female = no
			prisoner = no
			is_abroad = no
			at_location = ROOT
			NOT = { trait = incapable }
			NOT = { higher_tier_than = duke }
			any_child = { is_primary_heir = yes is_female = no }
		}
	}

	weight_multiplier = {
		days = 1
		modifier = { factor = 2 trait = wroth }
		modifier = { factor = 0.5 trait = kind }
	}

	immediate = {
		spouse = {
			random_child = {
				limit = { is_primary_heir = yes is_female = no }
				save_event_target_as = poisoned_heir
			}
		}
	}

	option = { # poison
		name = EVTOPTAmarriage10
		pacifists_lose_piety_effect = yes
		random_list = {
			33 = { # ill
				event_target:poisoned_heir = {
					if = {
						limit = { has_dlc = "Reapers" }
						set_character_flag = disease_not_serious
						add_trait = vomiting
						hidden_effect = {
							if = {
								limit = { is_disease_immune_race = yes }
								remove_trait = vomiting
							}
						}
						character_event = { id = treatment.6 days = 10 random = 30 }
					}
					else = {
						add_trait = ill
						hidden_effect = {
							if = { limit = { is_disease_immune_race = yes } remove_trait = ill }
						}
					}
				}
				random_list = {
					50 = { prestige = 15 }
					25 = {
						spouse = {
							character_event = {
								id = marriage.11
								days = 14
								tooltip = EVTTOOLTIPmarriage11
							}
						}
					}
					25 = {
						character_event = {
							id = marriage.12
							days = 7
							tooltip = EVTTOOLTIPmarriage12
						}
					}
				}
			}
			33 = { # crippled
				event_target:poisoned_heir = { add_trait = infirm }
				random_list = {
					20 = { prestige = 15 }
					40 = {
						spouse = {
							character_event = {
								id = marriage.11
								days = 14
								tooltip = EVTTOOLTIPmarriage11
							}
						}
					}
					40 = {
						character_event = {
							id = marriage.12
							days = 7
							tooltip = EVTTOOLTIPmarriage12
						}
					}
				}
			}
			33 = { # dead
				event_target:poisoned_heir = {
					if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
					else = { death = { death_reason = death_murder_poison killer = ROOT } }
				}
				random_list = {
					10 = { prestige = 15 }
					45 = {
						spouse = {
							character_event = {
								id = marriage.13
								days = 14
								tooltip = EVTTOOLTIPmarriage13
							}
						}
					}
					45 = {
						character_event = {
							id = marriage.14
							days = 7
							tooltip = EVTTOOLTIPmarriage14
						}
					}
				}
			}
		}
	}
	option = { # no
		name = EVTOPTBmarriage10
		piety = 20
	}
}
