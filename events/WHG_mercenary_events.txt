###############################################################################
# WHG_mercenary_events.txt - dynastic mercenary bands and the war chest
#
# Namespace: merc (was ZE.10000-10200 in ze_mercenary_events and
# ZE.11000-11050 in ze_mercenary_events_02).
#   1        Yearly tombola for a captain of a band raised by a ruler
#   10-26    The captain asks for more pay, pays the men less, breaks free
#   18-19    Hired and dismissed hooks (on_mercenary_hired/dismissed)
#   50       Ogre mercenaries take to man-eating (five-year pulse)
#   100-111  The captain asks to marry into the creator's family, wants
#            more men
#   120-150  The band is dissolved or the captain replaced (ze_decisions,
#            on_mercenary_captain_replacement)
#   200-206  Re-equip or reinforce a feudal band (ze_decisions, was
#            ze_extra_events)
#   250-258  The captain praises the band's founder to the hiring ruler,
#            who may feast with the founder and sign a pact
#
# Namespace: warchest (was ZE.21000-22200 in ze_war_chest_extortion and
# ze_war_chest_passive).
#   1-30     Extort subjects: taxes, a rich courtier, temple treasures
#   100-120  A vassal offers money for the war chest in return for a favour
###############################################################################

namespace = merc

# merc.1 Mercenary captain tombola
# Yearly pulse for captains of bands raised by a feudal or republican ruler:
# the captain's ambition may grow, their character shifts with the trade,
# and one of the demands below may follow.
# Scope: ROOT is the captain; hidden.
character_event = {
	id = merc.1
	hide_window = yes

	is_triggered_only = yes

	only_playable = yes

	trigger = {
		has_dlc = "Zeus"
		mercenary = yes
		primary_title = {
			has_title_flag = ze_dyn_merc
			band_creator = { OR = { is_feudal = yes is_republic = yes } }
		}
	}

	immediate = {
		if = {
			limit = { NOT = { has_character_flag = is_mercenary_leader } }
			set_character_flag = is_mercenary_leader
		}
		if = {
			limit = {
				NOR = {
					has_character_flag = mercenary_ambition_1
					has_character_flag = mercenary_ambition_2
					has_character_flag = mercenary_ambition_3
					has_character_flag = mercenary_ambition_4
					has_character_flag = mercenary_ambition_5
					has_character_flag = mercenary_ambition_6
					has_character_flag = mercenary_ambition_7
					has_character_flag = mercenary_ambition_8
					has_character_flag = mercenary_ambition_9
				}
			}
			set_character_flag = mercenary_ambition_1
		}
		# Ambition up chance
		random_list = {
			10 = {
				modifier = { factor = 4 has_character_flag = denied_last_request }
				modifier = { factor = 1.3 trait = cruel }
				modifier = { factor = 1.2 trait = wroth }
				modifier = { factor = 1.4 trait = deceitful }
				modifier = { factor = 1.6 trait = ambitious }
				modifier = { factor = 0.6 trait = just }
				modifier = { factor = 0.6 trait = kind }
				modifier = { factor = 0.4 trait = content }
				modifier = {
					factor = 1.5
					primary_title = {
						band_creator = { ROOT = { relative_power = { who = PREV power = 0.7 } } }
					}
				}
				modifier = {
					factor = 1.5
					primary_title = {
						band_creator = { ROOT = { relative_power = { who = PREV power = 1 } } }
					}
				}
				modifier = {
					factor = 1.5
					primary_title = {
						band_creator = { ROOT = { relative_power = { who = PREV power = 1.3 } } }
					}
				}

				increase_mercenary_ambition_level_effect = yes
			}
			60 = { }
		}
		clr_character_flag = denied_last_request
		# Random mercenary event
		random_list = {
			10 = { ### Improve on military education
				modifier = {
					factor = 0
					NOR = {
						trait = misguided_warrior
						trait = tough_soldier
						trait = skilled_tactician
					}
				}
				upgrade_education_effect = yes
			}
			10 = { ### Remove Craven
				modifier = { factor = 0 NOT = { trait = craven } }
				remove_trait = craven
			}
			10 = { ### add brave
				modifier = { factor = 0 NOR = { trait = craven trait = brave } }
				add_trait = brave
			}
			10 = { ### add diligent
				modifier = { factor = 0 NOR = { trait = diligent trait = slothful } }
				add_trait = diligent
			}
			10 = { ### Remove kind
				modifier = { factor = 0 NOT = { trait = kind } }
				remove_trait = kind
			}
			10 = { ### Remove humble
				modifier = { factor = 0 NOT = { trait = humble } }
				remove_trait = humble
			}
			10 = { ### add lustful
				modifier = { factor = 0 NOR = { trait = chaste trait = lustful } }
				add_trait = lustful
			}
			10 = { ### add wroth
				modifier = { factor = 0 NOR = { trait = wroth trait = patient } }
				add_trait = wroth
			}
			10 = { ### add cruel
				modifier = { factor = 0 NOR = { trait = cruel trait = kind } }
				add_trait = cruel
			}
			10 = { ### become wounded
				modifier = { factor = 0 NOR = { trait = wounded is_maimed_trigger = yes } }
				if = { limit = { NOT = { is_maimed_trigger = yes } } add_trait = wounded }
			}
			10 = { ### add maimed
				modifier = { factor = 0 NOT = { can_be_maimed_trigger = yes } }
				add_maimed_trait_effect = yes
			}
			10 = { ### add adventurer
				modifier = { factor = 0 NOT = { trait = adventurer } }
				add_trait = adventurer
			}
			10 = { ### Ask for more money
				modifier = {
					factor = 0
					NOT = {
						primary_title = {
							had_title_flag = { flag = mercenary_unemployment years = 3 }
						}
					}
				}
				modifier = {
					factor = 1.3
					primary_title = {
						had_title_flag = { flag = mercenary_unemployment years = 10 }
					}
				}
				modifier = {
					factor = 1.2
					OR = {
						has_character_flag = mercenary_ambition_4
						has_character_flag = mercenary_ambition_5
						has_character_flag = mercenary_ambition_6
						has_character_flag = mercenary_ambition_7
						has_character_flag = mercenary_ambition_8
						has_character_flag = mercenary_ambition_9
					}
				}
				modifier = {
					factor = 1.2
					OR = {
						has_character_flag = mercenary_ambition_5
						has_character_flag = mercenary_ambition_6
						has_character_flag = mercenary_ambition_7
						has_character_flag = mercenary_ambition_8
						has_character_flag = mercenary_ambition_9
					}
				}
				modifier = {
					factor = 1.2
					OR = {
						has_character_flag = mercenary_ambition_6
						has_character_flag = mercenary_ambition_7
						has_character_flag = mercenary_ambition_8
						has_character_flag = mercenary_ambition_9
					}
				}
				modifier = {
					factor = 1.3
					OR = {
						has_character_flag = mercenary_ambition_7
						has_character_flag = mercenary_ambition_8
						has_character_flag = mercenary_ambition_9
					}
				}
				modifier = {
					factor = 1.3
					OR = {
						has_character_flag = mercenary_ambition_8
						has_character_flag = mercenary_ambition_9
					}
				}
				character_event = { id = merc.10 }
			}
			10 = {
				modifier = {
					factor = 0
					NOR = {
						has_character_flag = mercenary_ambition_5
						has_character_flag = mercenary_ambition_6
						has_character_flag = mercenary_ambition_7
						has_character_flag = mercenary_ambition_8
						has_character_flag = mercenary_ambition_9
					}
				}
				modifier = {
					factor = 0
					NOT = { primary_title = { mercenary_siphon_factor = 0.05 } }
				}
				character_event = { id = merc.20 } # Start to pay less
			}
			20 = {
				character_event = { id = merc.110 } # Demand more soldiers
			}
			20 = {
				character_event = { id = merc.100 } # Ask to marry child of band creator
			}
			10 = {
				modifier = {
					factor = 0
					NOR = {
						has_character_flag = mercenary_ambition_6
						has_character_flag = mercenary_ambition_7
						has_character_flag = mercenary_ambition_8
						has_character_flag = mercenary_ambition_9
					}
				}
				modifier = { factor = 2 has_character_flag = mercenary_ambition_7 }
				modifier = { factor = 4 has_character_flag = mercenary_ambition_8 }
				modifier = { factor = 8 has_character_flag = mercenary_ambition_9 }
				character_event = { id = merc.25 } # Break Free
			}
			100 = { } # Nothing happened
		}
	}
}

# merc.11 Rations are running low
# Pay up, or let them starve; an ambitious captain may come back for more.
# Scope: ROOT is the creator, FROM the captain.
letter_event = {
	id = merc.11
	desc = EVTDESCmerc11

	is_triggered_only = yes

	option = {
		name = EVTOPTAmerc11
		transfer_scaled_wealth = { to = FROM value = 0.1 min = 5 }
		hidden_tooltip = {
			FROM = {
				primary_title = { clr_title_flag = mercenary_unemployment }
				if = {
					limit = {
						OR = {
							has_character_flag = mercenary_ambition_6
							has_character_flag = mercenary_ambition_7
							has_character_flag = mercenary_ambition_8
							has_character_flag = mercenary_ambition_9
						}
					}
					random_list = {
						30 = { # ask for even more money
							modifier = { factor = 1.3 has_character_flag = mercenary_ambition_7 }
							modifier = { factor = 1.7 has_character_flag = mercenary_ambition_8 }
							modifier = { factor = 2.5 has_character_flag = mercenary_ambition_9 }
							modifier = { factor = 1.5 trait = greedy }
							character_event = { id = merc.12 days = 90 }
						}
						60 = { }
					}
				}
			}
		}
	}

	option = {
		name = EVTOPTBmerc11
		random = {
			chance = 50
			add_trait = cruel
			hidden_tooltip = {
				character_event = {
					id = 38259 #Notify Cruel
				}
			}
		}
		FROM = {
			set_character_flag = denied_last_request
			opinion = { who = ROOT modifier = opinion_denied_request months = 60 }
		}
	}
}

				modifier = { factor = 2.0 trait = barbarian_3 }
				modifier = { factor = 3.0 trait = barbarian_4 }

# merc.100 Marrying into the creator's family
# The captain eyes an unmarried child of the creator.
# Scope: ROOT is the captain; hidden, fires merc.101 for the creator.
character_event = {
	id = merc.100

	hide_window = yes

	is_married = no

	is_triggered_only = yes

	trigger = {
		NOT = { has_character_flag = mercenary_pondered_marriage }
		is_lowborn = no
		can_marry = yes
		mercenary = yes
		primary_title = {
			band_creator = {
				any_child = {
					is_married = no
					is_betrothed = no
					can_marry = yes
					is_adult = yes
					ai = yes

					NOR = {
						has_ugly_trait_trigger = yes
						is_dumb_trigger = yes
						trait = dwarf
						same_sex = ROOT
						sibling = ROOT
					}
				}
			}
		}
	}

	immediate = {
		set_character_flag = mercenary_pondered_marriage
		primary_title = {
			band_creator = {
				random_child = {
					limit = {
						is_married = no
						is_betrothed = no
						can_marry = yes
						is_adult = yes
						ai = yes

						NOR = {
							has_ugly_trait_trigger = yes
							is_dumb_trigger = yes
							trait = dwarf
							same_sex = ROOT
							sibling = ROOT
						}
					}
					save_event_target_as = chosen_child
				}
				save_event_target_as = target_liege
				character_event = { id = merc.101 }
			}
		}
	}
}

# merc.121 The captain deserts
# The captain turns bandit: robber bands in up to three of the creator's
# provinces.
# Scope: ROOT is the creator, FROM the captain.

	id = merc.121
	picture = GFX_evt_large_army
	border = GFX_event_normal_frame_diplomacy
	desc = EVTDESCmerc121

	is_triggered_only = yes

	option = {
		name = EVTOPTAmerc121
		custom_tooltip = { text = EVTOPTAmerc121_tooltip }
		FROM = {
			show_scope_change = no
			opinion = { who = ROOT modifier = opinion_hate years = 100 }
			set_character_flag = flag_upset_with_band_creator
		}

			limit = { NOT = { has_province_modifier = highway_robber_band } }

				ROOT = {
					random_demesne_province = {
						limit = {
							NOT = { has_province_modifier = highway_robber_band }
							NOT = { province = PREVPREV }
						}
						add_province_modifier = { modifier = highway_robber_band duration = -1 }
						ROOT = {
							random_demesne_province = {
								limit = {
									NOT = { has_province_modifier = highway_robber_band }
									NOR = { province = PREVPREV province = PREVPREVPREV }
								}
								add_province_modifier = {
									modifier = highway_robber_band
									duration = -1
								}
							}
						}
					}
				}
			}
		}
		hidden_tooltip = {
			FROM = {
				character_event = { id = merc.130 } #Flag clearing event for old merc captain
			}
		}
	}
}
