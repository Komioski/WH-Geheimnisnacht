# The Monastic Orders
# Namespace: monastic (was MNM.3260 - 3270, MNM.4000 - 4026, MNM.4090 - 4091,
#   MNM.4200 - 4218, MNM.4300 - 4306, MNM.4399 - 4443, MNM.4450 - 4452,
#   MNM.4610 - 4626, MNM.4950 - 4991 in mnm_monastic_orders_events.txt, and
#   DR_societies.4023 - 4026 and 7118 - 7119 in DR_societies_mnm_events.txt,
#   whs_shal.1 in wh_cult_of_shallya_events.txt)
# Faction: Old World faiths
# Serves: the Order of the Bleeding Heart (cult_of_shallya) and the Amazon
#   Sisterhood (amazon_sisterhood), the two monastic societies
#
# The sisters and brothers of Shallya tend the sick, give to the poor and
# sit in penance for their sins; the Amazons keep their own sisterhood in
# Lustria. Their grandmaster hands out missions (penance, alms, theology, a
# pilgrimage, a new temple, for the Sisters of Shallya a hospital), their
# powers lay hands on the sick, send a nun or monk to court, swap
# celibacy and wash the sins from the faithful, and their life together
# brings friendships and more, poems, roses and snoring in the temple,
# heretical books to burn, wards to raise and, for the greedy, a holy relic
# of Shallya to steal from the motherhouse.
#
# Joining                              monastic.1 - monastic.3
# Missions                             monastic.10 - monastic.11
# Penance in seclusion                 monastic.20 - monastic.48
# The other missions                   monastic.50 - monastic.66
# Powers                               monastic.70 - monastic.84
# Friendship                           monastic.90 - monastic.98
# The relic of the motherhouse         monastic.100 - monastic.116
# The poet                             monastic.130 - monastic.140
# The rose garden                      monastic.150 - monastic.156
# Asleep in the temple                 monastic.160 - monastic.166
# Heretical books                      monastic.170 - monastic.171
# The ward                             monastic.180 - monastic.186
# The hospital, the laying on of hands monastic.190 - monastic.196
# The last mercy                       monastic.197 - monastic.198
#
# Callers: on_character_ask_to_join_society (monastic.1), the society pulse
#   (monastic.10, 90, 100, 130, 150, 160, 170), on_focus_selected through
#   common/objectives/00_focuses.txt (monastic.52), the pilgrimage in
#   events/WHG_saint_events.txt (monastic.56), on_settlement_construction_
#   completed (monastic.60), on_childhood_pulse (monastic.180),
#   on_adulthood (monastic.182), monastic_order_purity_effect in
#   00_scripted_effects.txt (monastic.83), society.2 (monastic.84), and the
#   decisions in mnm_society_decisions.txt (monastic.21, 65, 70, 71) and
#   mnm_monastic_order_decisions.txt (monastic.72, 75 - 80),
#   on_hospital_construction_completed in wh_on_actions.txt (monastic.191)
#   and the decision shallya_healing in wh_cult_societies_decisions.txt
#   (monastic.195) and shallyan_self_sacrifice there (monastic.197).
#   Triggers in
#   common/scripted_triggers/WHG_monastic_triggers.txt, effects in
#   common/scripted_effects/WHG_monastic_effects.txt, voices in
#   localisation/customizable_localisation/WHG_monastic_voices.txt, the
#   relics in common/artifacts/44_artifacts_monastic.txt.
# Saved: event targets monastic_leader, ascetic, interrupting_child,
#   other_parent, holy_seclusion_lover, book_target, potential_friend,
#   court_spiritual, monastic_befriender, monastic_friend,
#   monastic_artifact_stealer, monastic_artifact_lover,
#   monastic_artifact_witness, monastic_artifact_blackmailer,
#   monastic_hospital_province; variables
#   holy_seclusion_points, monastic_order_education_event; flags
#   monastic_orders_* (missions), holy_seclusion_*, success_seculison_desc_*,
#   purity_*, monastic_relic_*, monastic_stole_artifact,
#   flag_monastic_mid_theft, betrayed_cult_of_shallya,
#   monastic_friend_lover_@ROOT, 3264_*, 4202_*, 4212_*,
#   monastic_order_education_*, shallya_healing_* and the removed_* and
#   removing_* flags of the healing effects; global flags
#   monastic_relic_*_taken; the quest shallya_build_hospital_construction.
# Dropped: the dharmic, stoic and Christian orders and their pilgrimages
#   (MNM.8095 - 8099), the Benedictine gardening leftovers, and the Christian
#   and dharmic relics of the theft (now three relics of Shallya). The
#   hospital mission had no caller in the copy; it is now one of the

# A ruler asks to join a monastic order; the grandmaster answers.
# Scope: ROOT is the one asking, FROM the society. From
#   on_character_ask_to_join_society.

		FROM = { leader = { is_monastic_member_trigger = yes } }
	}

# monastic.10 The grandmaster's missions
# Picks the next mission of a member without one: penance in seclusion,
# alms, theology, a pilgrimage, a new temple or (for the Sisters of Shallya)
# a hospital. An AI member mostly settles
# it at once instead.
# Scope: ROOT is a member. From the society pulse.
character_event = {
	id = monastic.10
	hide_window = yes
	is_triggered_only = yes
	is_in_society = yes
	only_capable = yes

	trigger = {
		is_monastic_member_trigger = yes
		is_society_grandmaster = no
		has_any_quest = no
		is_inaccessible_trigger = no
		NOT = { has_character_modifier = quest_cooldown_timer }
	}

	immediate = {
		random_society_member = {
			limit = { is_society_grandmaster = yes same_society_as = ROOT }
			save_event_target_as = monastic_leader
		}
		random_list = {
			10 = { # Penance in seclusion (not for the Amazons, who serve in the field)
				modifier = { factor = 0 society_member_of = amazon_sisterhood }
				set_character_flag = monastic_orders_holy_seclusion
			}
			10 = { # Alms for the poor and the sick
				modifier = { factor = 0 is_playable = no }
				set_character_flag = monastic_orders_mission_charity
			}
			10 = { # Take up theology
				modifier = {
					factor = 0
					OR = {
						is_playable = no # Only playable characters get focuses
						NOT = { has_dlc = "Way of Life" }
						has_focus = focus_theology
						society_member_of = amazon_sisterhood
					}
				}
				set_character_flag = monastic_orders_mission_theology
			}
			10 = { # A pilgrimage to a holy site
				modifier = {
					factor = 0
					OR = {
						is_playable = no
						NOT = { religion_group = old_world_group }
						NOT = { has_dlc = "Sons of Abraham" }
						trait = pilgrim
						trait = on_pilgrimage
						has_alternate_start_parameter = { key = religion_names value = random }
					}
				}
				set_character_flag = monastic_orders_mission_pilgrimage
			}
			10 = { # Build a temple
				modifier = {
					factor = 0
					OR = {
						is_playable = no
						is_tribal = yes
						is_nomadic = yes
						NOT = { any_demesne_province = { num_of_empty_holdings = 1 } }
					}
				}
				set_character_flag = monastic_orders_mission_temple
			}
			10 = { # Build a hospital (the Sisters of Shallya only, with hospitals)
				modifier = {
					factor = 0
					OR = {
						is_playable = no
						NOT = { society_member_of = cult_of_shallya }
						NOT = { has_dlc = "Reapers" }
					}
				}
				set_character_flag = monastic_orders_mission_hospital
			}
			50 = { # The AI skips the mission and settles it at once
				modifier = { factor = 0 ai = no }
				random_list = {
					70 = {
						modifier = { factor = 0.9 society_influence >= 30 }
						modifier = { factor = 0.9 society_influence >= 40 }
						modifier = { factor = 0.8 society_influence >= 50 }
						modifier = { factor = 0.7 society_influence >= 70 }
						modifier = { factor = 0.6 society_influence >= 80 }
						modifier = { factor = 0.5 society_influence >= 90 }
						add_mission_succeed_influence_effect = yes
						monastic_order_purity_effect = yes
					}
					30 = { add_mission_fail_influence_effect = yes }
				}
			}
			85 = { # The AI does nothing
				modifier = { factor = 0 ai = no }
			}
		}
		event_target:monastic_leader = {
			character_event = { id = monastic.11 days = 7 random = 7 }
		}
	}
}

# monastic.21 The seclusion begins
# Starts the round of trials every 40 to 70 days (monastic.22).

	id = monastic.21
	picture = GFX_evt_secluded_in_prayer
	desc = EVTDESCmonastic21

	is_triggered_only = yes

		name = EVTOPTAmonastic21

# monastic.22 The trials of seclusion
# Each round brings one trial (a vice, loneliness, a hard passage, a child
# at the door, or a good day). At 4 points the penance succeeds, below -2
# it fails (monastic.45).
# Scope: ROOT is the member in seclusion.
character_event = {
	id = monastic.22
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				OR = {
					is_monastic_member_trigger = no
					NOT = { has_quest = quest_monastic_order_penance }
				}
			}
			character_event = { id = monastic.48 }
		}
		else_if = {
			limit = {
				monastic_in_seclusion_trigger = yes
				OR = {
					has_character_modifier = in_meditation
					has_character_modifier = doing_penance
				}
			}
			if = {
				limit = {
					check_variable = { which = holy_seclusion_points value = -2 }
					NOT = { check_variable = { which = holy_seclusion_points value = 4 } }
				}
				clear_event_target = interrupting_child
				random_list = {
					30 = { # Struggle with a vice
						modifier = {
							factor = 0
							trait = temperate
							trait = charitable
							trait = diligent
							NOT = { trait = lustful }
						}
						modifier = {
							factor = 0
							has_character_flag = holy_seclusion_struggled_with_vice
						}
						set_character_flag = holy_seclusion_struggled_with_vice
						random_list = {
							10 = {
								modifier = { factor = 0 NOT = { trait = lustful } }
								set_character_flag = holy_seclusion_lustful
							}
							10 = {
								modifier = { factor = 0 trait = temperate }
								set_character_flag = holy_seclusion_gluttonous
							}
							10 = {
								modifier = { factor = 0 trait = charitable }
								set_character_flag = holy_seclusion_greedy
							}
							10 = {
								modifier = { factor = 0 trait = diligent }
								set_character_flag = holy_seclusion_slothful
							}
						}
						character_event = { id = monastic.23 }
					}
					10 = { # Loneliness
						modifier = {
							factor = 0
							NAND = {
								OR = { trait = gregarious trait = socializer }
								any_courtier = {
									is_adult = yes
									prisoner = no
									NOR = {
										is_friend = ROOT
										is_lover = ROOT
										is_rival = ROOT
										any_spouse = { character = ROOT }
									}
								}
							}
						}
						modifier = {
							factor = 0
							has_character_flag = holy_seclusion_struggled_with_loneliness
						}
						set_character_flag = holy_seclusion_struggled_with_loneliness
						character_event = { id = monastic.28 }
					}
					10 = { # A hard passage of scripture
						modifier = {
							factor = 0
							OR = {
								NOT = { any_courtier = { has_job_title = job_spiritual } }
								has_character_flag = holy_seclusion_scripture_reading
							}
						}
						set_character_flag = holy_seclusion_scripture_reading
						character_event = { id = monastic.32 }
					}
					10 = { # A child at the door
						modifier = {
							factor = 0
							NOT = {
								any_child = {
									age = 4
									is_adult = no
									prisoner = no
									host = { character = ROOT }
									opinion = { who = ROOT value = 5 }
								}
							}
						}
						modifier = {
							factor = 0
							has_character_flag = holy_seclusion_interrupting_child
						}
						random_child = {
							limit = {
								age = 4
								is_adult = no
								prisoner = no
								host = { character = ROOT }
								opinion = { who = ROOT value = 5 }
							}
							preferred_limit = {
								is_primary_heir = ROOT
								NOR = {
									has_opinion_modifier = { modifier = kind_parent who = ROOT }
									has_opinion_modifier = { modifier = cruel_parent who = ROOT }
								}
							}
							save_event_target_as = interrupting_child
						}
						set_character_flag = holy_seclusion_interrupting_child
						event_target:interrupting_child = { character_event = { id = monastic.37 } }
					}
					10 = { # A good day
						modifier = { factor = 1.5 is_virtous = yes }
						modifier = { factor = 0.7 is_wicked = yes }
						modifier = { factor = 1.5 is_learned_trigger = yes }
						character_event = { id = monastic.44 }
					}
					10 = { # Every trial has been met: they come round again
						modifier = {
							factor = 0
							NAND = {
								has_character_flag = holy_seclusion_struggled_with_vice
								has_character_flag = holy_seclusion_struggled_with_loneliness
								has_character_flag = holy_seclusion_scripture_reading
								has_character_flag = holy_seclusion_interrupting_child
							}
						}
						clr_character_flag = holy_seclusion_struggled_with_vice
						clr_character_flag = holy_seclusion_struggled_with_loneliness
						clr_character_flag = holy_seclusion_scripture_reading
						clr_character_flag = holy_seclusion_interrupting_child
					}
				}
				character_event = { id = monastic.22 days = 40 random = 30 }
			}
			else = {
				random_society_member = {
					limit = { is_society_grandmaster = yes same_society_as = ROOT }
					character_event = { id = monastic.45 }
				}
			}
		}
	}
}
