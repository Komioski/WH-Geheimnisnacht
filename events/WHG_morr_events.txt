# Faction: Old World faiths
# Serves: the cult of Morr and its knightly order, the High and Chivalric
#   Order of Deserved Rest
#
# Joining, the AI's way into the order, and the two quests of the Raven
# Knights: the vampire hunt (a quest the monster hunt can close) and the
# hunt for the undead of a province where the living worship vampires as
# gods. In that province the knight may storm the town, sneak into the lair,
# turn the thralls against their masters, or purge undead and worshippers
# alike.
#
#   Joining                                cultofmorr.100 - cultofmorr.102
#   The AI joins and rises                 cultofmorr.110 - cultofmorr.111
#   The quest tombola                      cultofmorr.130
#   The vampire hunt                       cultofmorr.140 - cultofmorr.142
#   The undead hunt                        cultofmorr.150 - cultofmorr.161,
#                                          cultofmorr.200 - cultofmorr.201
#
# Fixes: the three ways to win the undead hunt cleared quest_vampire_hunt,
#   not the undead hunt they belong to, so the quest never ended; the
#   province that suffers the unrest was reached through FROM four steps
#   up and is now the saved event target morr_hunt_province. A failed hunt
#   killed the knight 30% of the time even if they could not die.
# Revived: the purge of undead and worshippers (cultofmorr.160 - 161, a
#   commented-out option), and the knight who fails but lives (200): a
#   failed hunt is 15% wounded, 15% dead now.
# Dropped: the rank-up events 120 - 122 (the order ranks up through the
#   generic society.1 chain of the mnm rank-up decision; nothing fired them).
# Added: a knight who wins the undead hunt may keep a Trophy of the Hunted
#   Monster (10%, once a lifetime, only if none held).
#
# Fired from: on_character_ask_to_join_society (100), on_five_year_pulse
#   (110), on_society_bi_yearly_pulse (130; random_events 140, 150) in
#   wh_on_actions.txt; province_morr_kill_decision (152,
#   wh_morr_decisions.txt); huntdown in WHG_hunting_events (142).
# Loc: localisation/00_WHG_morr_events.csv

# cultofmorr.110 An AI ruler of Morr's faith may join the order, or a member
# rise a rank (25%) (hidden).
character_event = {
	id = cultofmorr.110
	hide_window = yes

	only_rulers = yes

	is_triggered_only = yes # on_five_year_pulse (wh_on_actions.txt)

	trigger = {
		ai = yes
		prisoner = no
		NOT = { trait = incapable }
	}

	immediate = {
		if = {
			limit = { is_cult_of_morr_trigger = yes society_rank <= 3 society_can_rank_up = yes }
			random = { chance = 25 society_rank_up = 1 }
		}
		else_if = {
			limit = {
				is_adult = yes
				is_in_society = no
				controls_religion = no
				block_general_event_trigger = no
			}
			character_event = { id = cultofmorr.111 }
		}
	}
}

# cultofmorr.111 The AI joins the order 10% of the time (20% with 200
# prestige), and the first member becomes Grandmaster (hidden).
character_event = {
	id = cultofmorr.111
	hide_window = yes

	is_triggered_only = yes # cultofmorr.110

	immediate = {
		random_list = {
			90 = { }
			10 = {
				trigger = { can_join_society = cult_of_morr }
				modifier = { factor = 2 prestige = 200 }
				join_society = cult_of_morr
				random = {
					chance = 75
					if = {
						limit = { NOT = { society_influence >= 30 } }
						add_society_modifier = {
							modifier = cult_of_morr_successful_mission
							years = 10
						}
					}
				}
				if = {
					limit = {
						NOT = {
							society = { any_society_member = { is_society_grandmaster = yes } }
						}
					}
					set_society_grandmaster = yes
				}
			}
		}
	}
}

# cultofmorr.130 A brother of the order nearby offers a member a quest:
# a vampire hunt (60%) or an undead hunt (40%) (hidden).
character_event = {
	id = cultofmorr.130
	hide_window = yes

	is_in_society = yes
	capable_only = yes
	prisoner = no

	is_triggered_only = yes # on_society_bi_yearly_pulse (wh_on_actions.txt)

	trigger = {
		is_cult_of_morr_trigger = yes
		has_any_quest = no
		NOT = { has_character_modifier = quest_cooldown_timer }
		NOT = { is_inaccessible_trigger = yes }
		society = {
			any_society_member = {
				NOT = { character = ROOT }
				prisoner = no
				is_within_diplo_range = ROOT
				opinion = { who = ROOT value = 0 }
			}
		}
	}

	immediate = {
		society = {
			random_society_member = {
				limit = {
					NOT = { character = ROOT }
					prisoner = no
					is_within_diplo_range = ROOT
					opinion = { who = ROOT value = 0 }
				}
				preferred_limit = { same_realm = ROOT } # a brother of the same realm, if there is one
				save_event_target_as = event_questgiver
			}
		}
		event_target:event_questgiver = {
			random_list = {
				40 = { character_event = { id = cultofmorr.150 } } # undead hunt
				60 = { character_event = { id = cultofmorr.140 } } # vampire hunt
			}
		}
	}
}
