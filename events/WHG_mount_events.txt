###############################################################################
# Great mounts
# Namespace: mount (was mounts.6 - 22 in wh_mount_events)
# Faction: Generic
#
# A lord of the right people may spend years to win a great mount of
# legend: a Dragon Prince of Caledor wakes one of the sleeping dragons, a
# vampire stitches together an Abyssal Terror, a Blood Dragon rides out to
# find a Nightmare, a king of Nehekhara has a war chariot built, and a
# Norscan or a human servant of the Dark Gods takes a war mammoth. Each
# decision opens the offer; the mount comes when the work is done.
#
# The dragons of Caledor                  mount.1 - mount.3
# The Abyssal Terror                      mount.4 - mount.5
# The Nightmare                           mount.6 - mount.7
# The war chariot                         mount.8 - mount.9
# The war mammoth                         mount.10 - mount.11
#
# Callers: decisions/wh_mount_decisions.txt (1, 4, 6, 8, 10).
# Loc: localisation/00_WHG_mount_events.csv
#
# Revived: the mammoth (mounts.19 - 20) had no caller; it now has the
# trainwarmammoth decision (Komi).
# Fixed: the quest flags now clear when the mount comes, so a lord who
# loses it may try again; the dragon news named every tamer "he"; the
# piety costs of the Abyssal Terror and the Nightmare now match what the
# decisions ask for (they could leave a vampire deep in debt).
###############################################################################

namespace = mount

###############################################################################
# The dragons of Caledor
###############################################################################

# mount.1 A dragon mount
# The dragon is woken over ten years, for 1000 gold.
# Scope: a Caledorian lord.
character_event = {
	id = mount.1
	title = EVTNAMEmount1
	desc = EVTDESCmount1
	picture = GFX_evt_dragon
	is_triggered_only = yes # trainadragonmount decision

	option = {
		name = EVTOPTAmount1
		wealth = -1000
		set_character_flag = red_dragon_mount
		character_event = { id = mount.2 days = 3650 }
	}
	option = {
		name = EVTOPTBmount1
		trigger = { ai = no }
	}
}

# mount.2 The dragon wakes
# Players hear of it (mount.3).
# Scope: the tamer.
character_event = {
	id = mount.2
	title = EVTNAMEmount1
	desc = EVTDESCmount2
	picture = GFX_evt_dragon
	is_triggered_only = yes # mount.1

	option = {
		name = EVTOPTAmount2
		add_artifact = red_dragon_mount
		set_character_flag = has_mount
		clr_character_flag = red_dragon_mount
		prestige = 100
		hidden_effect = {
			any_player = {
				limit = { NOT = { character = ROOT } }
				character_event = { id = mount.3 days = 30 }
			}
		}
	}
}

# mount.3 News: a dragon of Caledor wakes
# Scope: a player; FROM is the tamer.
character_event = {
	id = mount.3
	title = EVTNAMEmount3
	desc = EVTDESCmount3
	picture = GFX_evt_dragon
	is_triggered_only = yes # mount.2

	option = {
		name = EVTOPTAmount3
	}
}

###############################################################################
# The Abyssal Terror
###############################################################################

# mount.4 An Abyssal Terror
# The vampire spends piety and much of the necromantic power gathered to
# stitch the beast together, which takes four and a half years.
# Scope: a vampire.
character_event = {
	id = mount.4
	title = EVTNAMEmount4
	desc = EVTDESCmount4
	picture = GFX_evt_vampire_battle
	is_triggered_only = yes # buildabyssalterror decision

	option = {
		name = EVTOPTAmount4
		piety = -800
		set_character_flag = abyssal_terror_mount
		if = {
			limit = { has_character_modifier = necromantic_power_5 }
			remove_character_modifier = necromantic_power_5
			add_character_modifier = { name = necromantic_power_2 duration = -1 }
		}
		else_if = {
			limit = { has_character_modifier = necromantic_power_4 }
			remove_character_modifier = necromantic_power_4
			add_character_modifier = { name = necromantic_power_1 duration = -1 }
		}
		else = {
			remove_character_modifier = necromantic_power_3
		}
		character_event = { id = mount.5 days = 1650 }
	}
	option = {
		name = EVTOPTBmount4
		trigger = { ai = no }
	}
}

# mount.5 The Abyssal Terror is ready
# Scope: the vampire.
character_event = {
	id = mount.5
	title = EVTNAMEmount4
	desc = EVTDESCmount5
	picture = GFX_evt_vampire_battle
	is_triggered_only = yes # mount.4

	option = {
		name = EVTOPTAmount5
		add_artifact = abyssal_terror_mount
		set_character_flag = has_mount
		clr_character_flag = abyssal_terror_mount
	}
}
