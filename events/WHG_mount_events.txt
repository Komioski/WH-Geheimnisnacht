# Faction: Generic
#
# A lord of the right people may spend years to win a great mount of
# legend: a Dragon Prince of Caledor wakes one of the sleeping dragons, a
# vampire stitches together an Abyssal Terror, a Blood Dragon rides out to
# find a Nightmare, a king of Nehekhara has a war chariot built, and a
# Norscan or a human servant of the Dark Gods takes a war mammoth. Each
# decision opens the offer; the mount comes when the work is done.
#
# The dragons of Caledor                  mount.1 - mount.3
# The Abyssal Terror                      mount.4 - mount.5
# The Nightmare                           mount.6 - mount.7
# The war chariot                         mount.8 - mount.9
# The war mammoth                         mount.10 - mount.11
#
# Part 2: stables and upkeep (was mountmaint.* in mount_maintenance_events)
# The breeding grounds                    mount.20 - mount.21
# The stables every two years             mount.22
# A mount dies, a foal grows              mount.23 - mount.24
# The fair mounts                         mount.25
# The AI buys a mount                     mount.30 - mount.32
#
# Callers: decisions/wh_mount_decisions.txt (1, 4, 6, 8, 10, 30, 31);
# on_startup (20); on_bi_yearly_pulse (22);
# common/scripted_effects/mounts_scripted_effects.txt (23, 24, 25).
# Triggers: common/scripted_triggers/WHG_mount_triggers.txt
# Effects: common/scripted_effects/WHG_mount_effects.txt
# Loc: localisation/00_WHG_mount_events.csv
#
# Dropped from part 2: mountmaint.15 (refund when a breeding ground runs
# dry) and mountmaint.19 (a looted mount goes home) had no callers and
# their targets were never set; the stock counters never go down, so a
# market is never empty.
#
# Revived: the mammoth (mounts.19 - 20) had no caller; it now has the
# trainwarmammoth decision (Komi).
# Fixed: the quest flags now clear when the mount comes, so a lord who
# loses it may try again; the dragon news named every tamer "he"; the
# piety costs of the Abyssal Terror and the Nightmare now match what the
# decisions ask for (they could leave a vampire deep in debt).

# mount.22 The stables, every two years
# Mounts age and tire, foals grow, the fair mounts are counted again, an AI
# lord with several mounts picks the best, and the court commander's mount
# is looked after (or taken back when he loses the post or gains land).
# Fixed: the "no change" half of the ageing roll was missing, so every
# mortal mount lost health every two years and died in twenty; the
# commanders are now found among the court and the vassals, not the whole
# realm.
# Scope: a playable lord with a mount.
character_event = {
	id = mount.22
	hide_window = yes
	is_triggered_only = yes # on_bi_yearly_pulse
	has_global_flag = mount_initialized
	only_playable = yes
	has_character_flag = has_mount

	immediate = {
		check_mount_flags_effect = yes
		if = {
			limit = { has_character_flag = has_mount }
			any_artifact = {
				limit = {
					has_artifact_flag = mount
					has_artifact_flag = alive
					NOR = { has_artifact_flag = immortal artifact_type = foal }
				}
				random_list = {
					40 = { } # no change
					60 = {
						modifier = { factor = 0.8 has_artifact_flag = robust }
						modifier = { factor = 0.7 has_artifact_flag = unique }
						decrease_mount_health_effect = yes
					}
				}
			}
			any_artifact = { limit = { artifact_type = foal } age_foal_effect = yes }
			if = {
				limit = { ai = yes war = no any_artifact = { count = 2 is_mount_trigger = yes } }
				set_character_flag = ai_recalc_equipped_mount
				character_event = { id = mount.32 days = 5 }
			}
		}
		any_courtier_or_vassal = {
			limit = { has_character_flag = has_mount_commander }
			if = { # the mount is dead
				limit = { NOT = { any_artifact = { has_artifact_flag = commander_mount } } }
				any_artifact = {
					limit = { has_artifact_flag = commander_addon }
					destroy_artifact = yes
				}
				clr_character_flag = has_mount_commander
				check_fair_mount_modifiers_effect = yes
			}
			else_if = { # no longer the commander, or landed now
				limit = { OR = { NOT = { has_minor_title = title_commander } is_ruler = yes } }
				any_artifact = {
					limit = {
						OR = {
							has_artifact_flag = commander_mount
							has_artifact_flag = commander_addon
						}
					}
					unsafe_destroy_artifact = yes
				}
				clr_character_flag = has_mount_commander
				check_fair_mount_modifiers_effect = yes
			}
			else = {
				any_artifact = {
					limit = { has_artifact_flag = commander_mount }
					random_list = {
						35 = { } # no change
						65 = {
							modifier = { factor = 0.8 has_artifact_flag = robust }
							decrease_mount_health_effect = yes
						}
					}
				}
			}
		}
		check_fair_mount_modifiers_effect = yes
	}
}

# mount.25 The fair mounts are counted
# The modifiers are cleared by check_fair_mount_modifiers_effect first.
# Scope: the owner.
character_event = {
	id = mount.25
	hide_window = yes
	is_triggered_only = yes # check_fair_mount_modifiers_effect

	immediate = {
		set_variable = { which = fair_mounts_counter value = 0 }
		any_artifact = {
			limit = { is_fair_mount_trigger = yes }
			ROOT = { change_variable = { which = fair_mounts_counter value = 1 } }
		}
		if = {
			limit = { check_variable = { which = fair_mounts_counter value = 5 } }
			add_character_modifier = { modifier = fair_mounts_5 duration = -1 }
		}
		else_if = {
			limit = { check_variable = { which = fair_mounts_counter value = 4 } }
			add_character_modifier = { modifier = fair_mounts_4 duration = -1 }
		}
		else_if = {
			limit = { check_variable = { which = fair_mounts_counter value = 3 } }
			add_character_modifier = { modifier = fair_mounts_3 duration = -1 }
		}
		else_if = {
			limit = { check_variable = { which = fair_mounts_counter value = 2 } }
			add_character_modifier = { modifier = fair_mounts_2 duration = -1 }
		}
		else_if = {
			limit = { check_variable = { which = fair_mounts_counter value = 1 } }
			add_character_modifier = { modifier = fair_mounts_1 duration = -1 }
		}
	}
}
