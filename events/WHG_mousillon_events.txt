# Cursed Mousillon
# The duchy is only truly cursed while one of the dead rules it. Under a
# living lord the curse sleeps: the sick stay dead and the taxes come in.
###############################################################################

# mousillon.1 The curse follows the ruler
# When a Mousillon county changes hands, every five years, and at startup:
# the holder gets or loses Cursed Mousillon on the Mousillon provinces they
# hold.
# Scope: ROOT is the holder of a Mousillon county.
character_event = {
	id = mousillon.1
	hide_window = yes

	is_triggered_only = yes # on_new_holder (all three), on_five_year_pulse, mousillon.2
	only_rulers = yes

	trigger = {
		mousillon_county_holder_trigger = yes
	}

	immediate = {
		mousillon_cursed_update_effect = yes
	}
}

# mousillon.2 Startup: set the curse
# On startup the host runs mousillon.1 for the holders of the four counties.
# Scope: ROOT is the host player.
character_event = {
	id = mousillon.2
	hide_window = yes

	is_triggered_only = yes # on_startup
	ai = no
	only_playable = yes

	trigger = {
		is_multiplayer_host_character = yes
	}

	immediate = {
		c_mousillon = { holder_scope = { character_event = { id = mousillon.1 } } }
		c_dras = { holder_scope = { character_event = { id = mousillon.1 } } }
		c_elne = { holder_scope = { character_event = { id = mousillon.1 } } }
		c_rachard = { holder_scope = { character_event = { id = mousillon.1 } } }
	}
}

###############################################################################
# The Red Pox
# In 1813 the Red Pox killed half of Mousillon. It never really left.
###############################################################################

# mousillon.10 The Red Pox returns
# Once a lifetime, a Mousillon lord finds the pox in the lower city. Shut
# the gates (mousillon.11 tells if it held), let it burn itself out
# (mousillon.12), or, for a master of the dead, let the dead rise.
# Scope: ROOT is the holder of a Mousillon county.
character_event = {
	id = mousillon.10
	desc = EVTDESCmousillon10
	picture = GFX_evt_plague_doctor
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # on_decade_pulse mousillon_tales
	only_rulers = yes

	trigger = {
		mousillon_county_holder_trigger = yes
		mousillon_tale_ready_trigger = yes
		NOT = { has_character_flag = mousillon_tale_pox }
	}

	immediate = {
		set_character_flag = mousillon_tale_pox
		mousillon_save_province_effect = yes
	}

	# Shut the gates
	option = {
		name = EVTOPTAmousillon10
		scaled_wealth = { value = -0.3 min = -100 }
		event_target:mousillon_province = {
			add_province_modifier = { name = mousillon_quarantine years = 3 }
		}
		character_event = { id = mousillon.11 days = 90 random = 60 }
		ai_chance = { factor = 50 }
	}

	# Let it burn out
	option = {
		name = EVTOPTBmousillon10
		event_target:mousillon_province = {
			add_province_modifier = { name = mousillon_red_pox years = 5 }
			spawn_disease = typhus
		}
		character_event = { id = mousillon.12 days = 60 random = 60 }
		ai_chance = {
			factor = 30
			modifier = { factor = 2 trait = greedy }
			modifier = { factor = 0.5 trait = kind }
		}
	}

	# Let the dead rise
	option = {
		name = EVTOPTCmousillon10
		trigger = { undead_master_trigger = yes }
		event_target:mousillon_province = {
			add_province_modifier = { name = mousillon_red_pox years = 5 }
		}
		add_character_modifier = { name = mousillon_pox_harvest years = 10 }
		piety = -25
		mousillon_survived_effect = yes
		ai_chance = { factor = 100 }
	}
}

# mousillon.11 Behind the gates
# Did the quarantine hold? Two times in three it did.
# Scope: ROOT is the lord who shut the gates. event_target:mousillon_province
#   (the quarantined province, saved in mousillon.10).
character_event = {
	id = mousillon.11
	desc = {
		trigger = { has_character_flag = mousillon_pox_held }
		text = EVTDESCmousillon11
	}
	desc = {
		trigger = { NOT = { has_character_flag = mousillon_pox_held } }
		text = EVTDESCmousillon11b
	}
	picture = GFX_evt_plague_doctor
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # mousillon.10

	immediate = {
		# event_target:mousillon_province carries over from mousillon.10
		random = {
			chance = 66
			set_character_flag = mousillon_pox_held
		}
	}

	option = {
		name = EVTOPTAmousillon11
		if = {
			limit = { has_character_flag = mousillon_pox_held }
			prestige = 50
		}
		else = {
			event_target:mousillon_province = {
				remove_province_modifier = mousillon_quarantine
				add_province_modifier = { name = mousillon_red_pox years = 5 }
				spawn_disease = typhus
			}
		}
		clr_character_flag = mousillon_pox_held
		mousillon_survived_effect = yes
	}
}

# mousillon.20 At Landuin's tomb
# A pilgrim (decision mousillon_landuin_pilgrimage) reaches the tomb. What
# answers depends on who kneels: a master of the dead faces the ghost
# (mousillon.23), the worthy are blessed (mousillon.21), the rest usually
# meet silence (mousillon.24), sometimes a curse (mousillon.22).
# Scope: ROOT is the pilgrim.
character_event = {
	id = mousillon.20
	desc = EVTDESCmousillon20
	picture = GFX_evt_stone_church
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # mousillon_landuin_pilgrimage

	immediate = {
		remove_character_modifier = mousillon_pilgrimage
	}

	option = {
		name = EVTOPTAmousillon20
		if = {
			limit = { undead_master_trigger = yes }
			character_event = { id = mousillon.23 }
		}
		else_if = {
			limit = { mousillon_landuin_worthy_trigger = yes }
			character_event = { id = mousillon.21 }
		}
		else = {
			random_list = {
				70 = { character_event = { id = mousillon.24 } }
				30 = { character_event = { id = mousillon.22 } }
			}
		}
	}
}

# mousillon.40 The last knight at the chapel
# Once a lifetime, the lord of Mousillon finds an old Grail Knight kneeling
# in the ruined Chapel of the Grail. Pay to reconsecrate it (mousillon.41
# tells how it went), let him die there with honour, send him away, or, for
# a vampire, feed.
# Scope: ROOT holds c_mousillon. event_target:mousillon_grail_knight is the
# old knight.
character_event = {
	id = mousillon.40
	desc = EVTDESCmousillon40
	picture = GFX_evt_knight_kneeling
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_decade_pulse mousillon_tales
	only_rulers = yes

	trigger = {
		has_landed_title = c_mousillon
		mousillon_tale_ready_trigger = yes
		NOT = { has_character_flag = mousillon_tale_chapel }
	}

	immediate = {
		set_character_flag = mousillon_tale_chapel
		create_character = {
			random_traits = no
			dynasty = random
			religion = grail
			culture = breton
			female = no
			age = 68
			attributes = { martial = 12 }
			trait = creature_human
			trait = bretonnian_knight
			trait = grail_knight
			trait = brave
			trait = just
			health = 2
		}
		new_character = {
			save_event_target_as = mousillon_grail_knight
		}
	}

	# Reconsecrate the chapel
	option = {
		name = EVTOPTAmousillon40
		trigger = { is_vampire_trigger = no }
		scaled_wealth = { value = -0.5 min = -150 }
		character_event = { id = mousillon.41 days = 365 random = 180 }
		ai_chance = {
			factor = 40
			modifier = { factor = 2 religion = grail }
			modifier = { factor = 0 NOT = { wealth = 100 } }
		}
	}

	# Let him die here
	option = {
		name = EVTOPTBmousillon40
		event_target:mousillon_grail_knight = { death = { death_reason = death_natural } }
		piety = 25
		prestige = 25
		mousillon_survived_effect = yes
		ai_chance = { factor = 40 }
	}

	# Send him away
	option = {
		name = EVTOPTCmousillon40
		event_target:mousillon_grail_knight = { death = { death_reason = death_missing } }
		mousillon_survived_effect = yes
		ai_chance = {
			factor = 20
			modifier = { factor = 0 is_vampire_trigger = yes }
		}
	}

	# Feed
	option = {
		name = EVTOPTDmousillon40
		trigger = { is_vampire_trigger = yes }
		event_target:mousillon_grail_knight = { death = { death_reason = death_murder killer = ROOT } }
		prestige = 50
		piety = -25
		mousillon_survived_effect = yes
		ai_chance = { factor = 100 }
	}
}

			chance = 40

		}
