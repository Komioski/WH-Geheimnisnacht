###############################################################################
# Mutation events
# Namespace: mutate
# Faction: Generic
#
# The touch of Chaos on mortal flesh and minds. Every five years a human,
# Amazon, Skaven or daemon-touched character may roll the mutation toughness
# test, which can twist them into a mutant, a Chaos Spawn or one forsaken by
# the gods. The same file runs the lunatic_counter variable that drives the
# lunacy traits, the opinion vassals hold of a mutant or mad liege, and how
# witch hunters and frightened courts deal with mutants.
#
# Mutation                      mutate.1 - mutate.4
# Lunacy counter                mutate.5 - mutate.9
# Vassal opinion                mutate.10 - mutate.11
# Witch hunters and mutants     mutate.12 - mutate.16
#
# Fired from common/on_actions/wh_on_actions.txt: the five-year list mutation
# (mutate.1), mutate.9 every five years, mutate.10-11 every two years, the
# yearly list whunter_events (mutate.12) and mutate.15 every year. mutate.5 is
# fired by lunacy, Slaanesh cult and red earring events whenever something
# strains the mind.
# Scripted triggers: common/scripted_triggers/wh_mutation_scripted_triggers.txt
# Scripted effects: common/scripted_effects/wh_mutation_scripted_effects.txt,
# common/scripted_effects/wh_lunacy_effects.txt
# Loc: localisation/00_WHG_mutation_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = mutate

# mutate.1 The touch of Chaos
# Every five years one character in five may face the winds of Chaos: one in
# five of those rolls the mutation toughness test, then may become a Chaos
# Spawn (mutate.2) or be forsaken by the gods (mutate.3). Higher Doom, the
# Chaos Wastes, warpstone, cursed Mousillon and the Geisterforst, magic and a
# Chaos faith make it likelier. Skaven, Kislevites, immortals and the Hollowed
# resist.
# Scope: ROOT is the character.
character_event = {
	id = mutate.1
	hide_window = yes
	is_triggered_only = yes # on_five_year_pulse, random list mutation

	trigger = {
		able_to_get_mutation_trigger = yes
	}

	weight_multiplier = {
		days = 1
		# Doom: the higher the Doom, the more the winds of Chaos twist flesh.
		# The Chaos Wastes and warpstone count as the highest step.
		modifier = {
			factor = 0.8
			is_doom_very_low_trigger = yes
			location = {
				NOR = {
					region = world_chaos_wastes
					has_province_modifier = warpdust_traces
					has_province_modifier = warpstone_meteor
					has_province_modifier = warpstone_vein
				}
			}
		}
		modifier = {
			factor = 0.9
			is_doom_low_trigger = yes
			location = {
				NOR = {
					region = world_chaos_wastes
					has_province_modifier = warpdust_traces
					has_province_modifier = warpstone_meteor
					has_province_modifier = warpstone_vein
				}
			}
		}
		modifier = {
			factor = 1.15
			OR = {
				is_doom_high_trigger = yes
				AND = { location = { region = world_norsca } is_doom_very_high_trigger = no }
				AND = { location = { region = world_troll_country } is_doom_very_high_trigger = no }
				location = { has_province_modifier = warpstone_meteor }
			}
			location = {
				NOR = { region = world_chaos_wastes has_province_modifier = warpstone_vein }
			}
		}
		modifier = {
			factor = 1.3
			OR = {
				is_doom_very_high_trigger = yes
				location = {
					OR = { region = world_chaos_wastes has_province_modifier = warpstone_vein }
				}
			}
		}
		# Cursed lands
		modifier = {
			factor = 1.4
			location = { duchy = { OR = { title = d_geisterforst title = d_mousillon } } }
		}
		# Magic Potential
		modifier = { factor = 1.1 trait = magic_potential_1 }
		modifier = { factor = 1.2 trait = magic_potential_2 }
		modifier = { factor = 1.3 trait = magic_potential_3 }
		modifier = { factor = 1.4 trait = magic_potential_4 }
		# The Hollowed have no connection to the winds
		modifier = { factor = 0 trait = hollowed }
		modifier = { factor = 1.1 trait = studying_magic }
		# Unsanctioned magic
		modifier = { factor = 1.2 OR = { trait = hedge_wizard trait = witch trait = warlock } }
		modifier = { factor = 1.2 true_religion_group = chaos_gods_group }
		# Skaven are steeped in warpstone from birth and change slowly
		modifier = { factor = 0.3 trait = creature_skaven }
		# Kislev's cold northern stock is hard to twist
		modifier = {
			factor = 0.5
			OR = { race = gospodar race = kislevite race = ropsmann race = ungol }
		}
		# Immortals have weathered the winds for centuries
		modifier = { factor = 0.5 lifespan_other_long_lived_trigger = yes }
	}

	immediate = {
		random = {
			chance = 20
			custom_tooltip = { text = mutations_test_tt mutation_toughness_test_effect = yes }
		}
		if = { limit = { become_chaos_spawn_trigger = yes } character_event = { id = mutate.2 } }
		else_if = { limit = { become_forsaken_trigger = yes } character_event = { id = mutate.3 } }
	}
}

# mutate.5 The mind strains
# Something has strained the character's mind: seven times in ten (fewer for
# the clever, the magically strong and long-lived races; far fewer for
# lizardmen and greenskins) the lunatic_counter climbs by one and the lunacy
# trait follows it. Mutations make the strain worse.
# Scope: ROOT is the character.
character_event = {
	id = mutate.5
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		lunacy_can_go_mad_trigger = yes
		NOR = { trait = avatar_kurnous has_character_flag = kurnous_frenzy }
	}

	immediate = {
		random_list = {
			30 = {
				# Nothing happens. Long-lived and hardy races shrug off what
				# would unhinge a human.
				modifier = { factor = 2 lifespan_dwarf_trigger = yes }
				modifier = { factor = 3 lifespan_elf_trigger = yes }
				modifier = { factor = 2 lifespan_other_long_lived_trigger = yes }
				modifier = { factor = 20 lifespan_hardy_race_trigger = yes }
				modifier = { factor = 1.5 NOR = { trait = creature_human trait = creature_amazon } }
				modifier = { factor = 0.7 trait = infirm }
				modifier = { factor = 0.7 trait = addict }
				modifier = { factor = 0.8 trait = depressed }
				modifier = { factor = 0.9 trait = stressed }
				modifier = { factor = 0.9 trait = drunkard }
				modifier = { factor = 0.5 trait = imbecile }
				modifier = { factor = 0.6 trait = idiot }
				modifier = { factor = 0.7 trait = slow }
				modifier = { factor = 0.8 trait = dumb }
				modifier = { factor = 0.9 trait = dim }
				modifier = { factor = 1.2 trait = bright }
				modifier = { factor = 1.5 trait = smart }
				modifier = { factor = 1.75 trait = quick }
				modifier = { factor = 2 trait = brilliant }
				modifier = { factor = 2.5 trait = genius }
				modifier = { factor = 1.25 has_character_modifier = magic_power_1 }
				modifier = { factor = 1.5 has_character_modifier = magic_power_2 }
				modifier = { factor = 1.75 has_character_modifier = magic_power_3 }
				modifier = { factor = 2 has_character_modifier = magic_power_4 }
				# Mutations: 1-9, 10-14, 15-19, 20 or more
				modifier = {
					factor = 0.75
					check_variable = { which = "mutation_counter" value = 1 }
					NOT = { check_variable = { which = "mutation_counter" value = 10 } }
				}
				modifier = {
					factor = 0.5
					check_variable = { which = "mutation_counter" value = 10 }
					NOT = { check_variable = { which = "mutation_counter" value = 15 } }
				}
				modifier = {
					factor = 0.25
					check_variable = { which = "mutation_counter" value = 15 }
					NOT = { check_variable = { which = "mutation_counter" value = 20 } }
				}
				modifier = {
					factor = 0.05
					check_variable = { which = "mutation_counter" value = 20 }
				}
			}
			70 = {
				change_variable = { which = "lunatic_counter" value = 1 }
				lunacy_counter_settle_effect = yes
			}
		}
	}
}

# mutate.9 Madness drifts
# Every five years a lunatic, or anyone whose counter has reached a stage,
# drifts: seven times in ten deeper into madness, three times in ten back
# towards sanity. Long-lived and hardy races mostly drift back, or they would
# all end up raving over the centuries. The counter is first synced to the
# trait so that treatments and vanilla events that add or cure lunacy are
# respected. Climbing a stage shows a notice; falling back is silent. Only
# characters whose counter has moved carry the lunacy_counter_active flag
# (set by lunacy_counter_check_effect); it is cleared once they are fully sane.
# Scope: ROOT is the character.
character_event = {
	id = mutate.9
	hide_window = yes
	is_triggered_only = yes # on_five_year_pulse

	has_character_flag = lunacy_counter_active

	trigger = {
		lunacy_can_go_mad_trigger = yes
		NOR = { trait = avatar_kurnous has_character_flag = kurnous_frenzy }
		OR = { is_lunatic_trigger = yes check_variable = { which = "lunatic_counter" value = 5 } }
	}

	immediate = {
		lunacy_counter_sync_effect = yes
		random_list = {
			70 = {
				modifier = { factor = 0.3 lifespan_dwarf_trigger = yes }
				modifier = { factor = 0.25 lifespan_elf_trigger = yes }
				modifier = { factor = 0.4 lifespan_other_long_lived_trigger = yes }
				modifier = { factor = 0.2 lifespan_hardy_race_trigger = yes }
				change_variable = { which = "lunatic_counter" value = 1 }
			}
			30 = { change_variable = { which = "lunatic_counter" value = -1 } }
		}
		lunacy_counter_settle_effect = yes
		# Fully sane again: drop out of the drift until the mind strains anew
		if = {
			limit = {
				is_lunatic_trigger = no
				NOT = { check_variable = { which = "lunatic_counter" value = 1 } }
			}
			clr_character_flag = lunacy_counter_active
		}
	}
}

# mutate.10 Serving a monster
# Every two years a vassal of a mutant liege, or of a maniac or raving liege,
# who is sane, unmutated and not of a Chaos or Skaven faith, comes to resent
# them. A mutant liege takes precedence over a mad one, and only faiths that
# frown on madness mind a mad liege.
# Scope: ROOT is the vassal.
character_event = {
	id = mutate.10
	hide_window = yes
	is_triggered_only = yes # on_bi_yearly_pulse

	only_rulers = yes

	trigger = {
		independent = no
		liege = {
			OR = {
				has_minor_mutation_trigger = yes
				has_major_mutation_trigger = yes
				has_massive_mutation_trigger = yes
				trait = lunatic_2
				trait = lunatic_3
			}
		}
		NOR = {
			is_mutated_trigger = yes
			has_minor_mutation_trigger = yes
			has_major_mutation_trigger = yes
			has_massive_mutation_trigger = yes
			is_lunatic_trigger = yes
			trait = incapable
			trait = trusting
			true_religion_group = chaos_gods_group
			true_religion_group = horned_rat_god_group
			has_any_opinion_modifier = opinion_mutant
			has_any_opinion_modifier = opinion_lunatic_w
		}
	}

	immediate = {
		if = {
			limit = {
				liege = {
					OR = {
						has_minor_mutation_trigger = yes
						has_major_mutation_trigger = yes
						has_massive_mutation_trigger = yes
					}
				}
			}
			liege = { ROOT = { opinion = { who = PREV modifier = opinion_mutant years = 100 } } }
		}
		else_if = {
			limit = { lunacy_sane_faith_trigger = yes }
			liege = { ROOT = { opinion = { who = PREV modifier = opinion_lunatic_w years = 100 } } }
		}
	}
}

# mutate.11 The resentment fades
# Every two years a vassal holding mutate.10's resentment drops it if the
# liege is no longer a mutant or lunatic, or the vassal has become one, has
# turned to Chaos or the Horned Rat, or is incapable or trusting.
# Scope: ROOT is the vassal.
character_event = {
	id = mutate.11
	hide_window = yes
	is_triggered_only = yes # on_bi_yearly_pulse

	only_rulers = yes

	trigger = {
		independent = no
		OR = {
			has_any_opinion_modifier = opinion_mutant
			has_any_opinion_modifier = opinion_lunatic_w
		}
		OR = {
			liege = {
				NOR = {
					has_minor_mutation_trigger = yes
					has_major_mutation_trigger = yes
					has_massive_mutation_trigger = yes
					is_lunatic_trigger = yes
				}
			}
			is_mutated_trigger = yes
			has_minor_mutation_trigger = yes
			has_major_mutation_trigger = yes
			has_massive_mutation_trigger = yes
			is_lunatic_trigger = yes
			trait = incapable
			trait = trusting
			true_religion_group = chaos_gods_group
			true_religion_group = horned_rat_god_group
		}
	}

	immediate = {
		liege = {
			ROOT = {
				remove_opinion = { who = PREV modifier = opinion_mutant }
				remove_opinion = { who = PREV modifier = opinion_lunatic_w }
			}
		}
	}
}

# mutate.12 A mutant at court
# A witch hunter on a court hunt spots a courtier of their liege whose
# mutation shows (mutate_exposable_mutant_trigger) and confronts them
# (mutate.13). Order mission hunts use whunter.10 instead.
# Scope: ROOT is the witch hunter.
character_event = {
	id = mutate.12
	title = WITCH_HUNT
	desc = EVTDESCmutate12
	picture = GFX_evt_witch_hunters
	is_triggered_only = yes # on_yearly_pulse, random list whunter_events

	has_character_flag = whunter_on_hunt
	religion_group = old_world_group

	trigger = {
		prisoner = no
		is_adult = yes
		trait = on_witch_hunt
		NOT = { trait = incapable }
		NOT = { has_character_flag = mission_hunt }
		top_liege = { NOT = { has_character_modifier = expelled_k_witch_hunters_order } }
		liege = {
			religion_group = old_world_group
			any_courtier = { NOT = { character = ROOT } mutate_exposable_mutant_trigger = yes }
		}
	}

	option = {
		name = EVTOPTAmutate12
		save_event_target_as = witch_hunter
		liege = {
			random_courtier = {
				limit = { NOT = { character = ROOT } mutate_exposable_mutant_trigger = yes }
				character_event = { id = mutate.13 }
			}
		}
	}
}

# mutate.13 Discovered
# The witch hunter has seen the courtier's mutation. The mutant can beg the
# liege for mercy (mutate.14) or flee the court before the fires are lit.
# Scope: ROOT is the mutant, FROM the witch hunter.
character_event = {
	id = mutate.13
	title = WITCH_HUNT
	desc = EVTDESCmutate13
	picture = GFX_evt_witch_hunters
	is_triggered_only = yes

	immediate = {
		add_character_modifier = { name = wh_accused duration = 730 hidden = yes }
	}

	# Appeal to the liege
	option = {
		name = EVTOPTAmutate13
		trigger = { independent = no }
		ai_chance = { factor = 3 }
		liege = { character_event = { id = mutate.14 } }
	}

	# Flee the court
	option = {
		name = EVTOPTBmutate13
		ai_chance = {
			factor = 1
			modifier = { factor = 2 trait = craven }
			modifier = { factor = 2 trait = paranoid }
			modifier = { factor = 0.5 trait = brave }
		}
		event_target:witch_hunter = { show_portrait = yes }
		prestige = -50
		banish = yes
	}
}

# mutate.14 The liege decides
# The mutant pleads innocence of Chaos. The liege can burn them, imprison
# them, or spare them (whunter.17-19). Burning or imprisoning upholds the
# hunter's accusation.
# Scope: ROOT is the liege, FROM the mutant, FROMFROM the witch hunter.
character_event = {
	id = mutate.14
	title = WITCH_HUNT
	desc = EVTDESCmutate14
	picture = GFX_evt_beheading
	is_triggered_only = yes

	# Burn the abomination
	option = {
		name = EVTOPTAmutate14
		trigger = {
			OR = {
				ai = no
				trait = zealous
				trait = paranoid
				trait = arbitrary
				trait = cruel
				NOT = { opinion = { who = FROM value = 1 } }
			}
		}
		event_target:witch_hunter = { show_portrait = yes }
		FROMFROM = { whunter_accusation_upheld_effect = yes }
		decrease_visibility_effect = yes
		FROM = { character_event = { id = whunter.17 } }
	}

	# Show mercy
	option = {
		name = EVTOPTBmutate14
		trigger = { OR = { ai = no opinion = { who = FROM value = 50 } } }
		event_target:witch_hunter = { show_portrait = yes }
		increase_visibility_minor_effect = yes
		FROM = { character_event = { id = whunter.19 } }
	}

	# Imprison them
	option = {
		name = EVTOPTCmutate14
		event_target:witch_hunter = { show_portrait = yes }
		FROMFROM = { whunter_accusation_upheld_effect = yes }
		decrease_visibility_effect = yes
		FROM = { character_event = { id = whunter.18 } }
	}
}

# mutate.15 Exposed
# Every year a follower of an Old World faith whose mutations can no longer be
# hidden (major or massive; minor for the AI), who is not hiding and lacks the
# cunning (intrigue 12) to keep them covered, is seen and cut down by their
# own court. Players in the realm hear of a landed victim (mutate.16).
# Scope: ROOT is the mutant.
character_event = {
	id = mutate.15
	title = Mutation
	desc = EVTDESCmutate15
	picture = GFX_evt_mutation
	is_triggered_only = yes # on_yearly_pulse

	religion_group = old_world_group

	trigger = {
		is_daemon_race_trigger = no
		OR = {
			AND = { ai = yes has_minor_mutation_trigger = yes }
			has_major_mutation_trigger = yes
			has_massive_mutation_trigger = yes
		}
		NOR = { intrigue = 12 trait = in_hiding trait = illusion_of_normality }
	}

	option = {
		name = EVTOPTAmutate15
		if = {
			limit = { higher_tier_than = BARON }
			any_realm_lord = {
				limit = {
					ai = no
					NOT = { character = ROOT }
					NOT = { has_character_flag = mutation_do_not_disturb }
				}
				character_event = { id = mutate.16 }
			}
		}
		if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }
		else = { death = { death_reason = death_execution } }
	}
}
