# Fired from common/on_actions/nb_on_actions.txt: naval.1

# (naval.100), the Erengrad Convoy's arrival (erengrad.13) and the voyage
# roll. A Convoy run's cargo can be spoiled by the sea
# (naval_spoil_convoy_cargo_effect); see events/WHG_erengrad_events.txt.

# naval.1 A unit moves
# Hidden. Whenever a commander's unit enters a province: the Convoy Master
# may have reached the port (erengrad.13); at sea, the naval battle check
# runs (naval.100) and, once a year at most, one time in thirty-three a
# voyage event comes up. The Convoy Master at sea meets them three times
# as often.
# Scope: ROOT is a commander whose unit entered a province.
character_event = {
	id = naval.1
	hide_window = yes

	is_triggered_only = yes # on_unit_entering_province
	min_age = 16
	only_capable = yes
	prisoner = no

	trigger = {
		in_command = yes
		NOT = { is_sub_unit_leader = yes }
	}

	immediate = {
		if = {
			limit = { erengrad_convoy_arrived_trigger = yes }
			liege = { character_event = { id = erengrad.13 } }
		}
		if = {
			limit = { is_at_sea = yes }
			character_event = { id = naval.100 }
			if = {
				limit = { NOT = { has_character_modifier = naval_voyage_cd } }
				random_list = {
					97 = {
						modifier = { factor = 0.33 has_minor_title = title_convoy_master }
					}
					3 = {
						add_character_modifier = { name = naval_voyage_cd years = 1 hidden = yes }
						random_list = {
							20 = { character_event = { id = naval.10 } }
							15 = {
								trigger = { NOT = { has_character_modifier = naval_white_star_escort } }
								modifier = { factor = 2 naval_sea_of_claws_trigger = yes }
								modifier = { factor = 0 culture_group = norscan_group }
								character_event = { id = naval.11 }
							}
							8 = {
								trigger = { naval_sea_of_claws_trigger = yes }
								character_event = { id = naval.12 }
							}
							8 = {
								trigger = {
									top_liege = {
										is_ruler = yes
										NOT = { has_character_flag = naval_pirate_amnesty }
									}
								}
								character_event = { id = naval.17 }
							}
							10 = {
								modifier = { factor = 0 culture_group = asur_group }
								character_event = { id = naval.19 }
							}
							10 = {
								trigger = {
									naval_sea_of_claws_trigger = yes
									NOT = { culture_group = norscan_group }
									NOT = { has_character_modifier = naval_white_star_escort }
								}
								character_event = { id = naval.20 }
							}
							8 = {
								trigger = {
									naval_sea_of_claws_trigger = yes
									NOT = { has_character_modifier = naval_white_star_escort }
									OR = {
										religion = kislev_gods
										religion = ulrican
										culture_group = kislevite_group
									}
								}
								character_event = { id = naval.21 }
							}
							10 = { character_event = { id = naval.22 } }
							10 = { character_event = { id = naval.23 } }
						}
					}
				}
			}
		}
	}
}

# Naval battles
# Moved from nautical_events.txt (namespace ships). An admiral at sea who
# meets a hostile fleet may engage; the outcome effects are the vanilla
# naval battle effects (assign_naval_outcome1_effect and the rest).
###############################################################################

# naval.100 Hostile fleet nearby
# Hidden. An admiral at sea who shares a sea zone with a hostile fleet saves
# both sides and asks the attacker whether to engage (naval.101). Sea
# raiders (naval_sea_raider_trigger) treat every foreign admiral as fair
# game; everyone else only fleets they are at war with. Runs on every move
# at sea, so it bails out early: no war and no raider means no search, and
# both admirals get naval_battle_cd (60 days) once they have met.
# Scope: ROOT is the admiral at sea.
character_event = {
	id = naval.100
	hide_window = yes

	is_triggered_only = yes # naval.1

	trigger = {
		is_at_sea = yes
		NOT = { has_character_modifier = naval_battle_cd }
		OR = {
			war = yes
			top_liege = { war = yes }
			naval_sea_raider_trigger = yes
		}
	}

	immediate = {
		random_character = {
			limit = {
				at_location = ROOT
				in_command = yes
				NOT = { has_character_modifier = naval_battle_cd }
				NOR = {
					same_realm = ROOT
					has_non_aggression_pact_with = ROOT
					mercenary = yes
					rebel = yes
				}
				OR = {
					war_with = ROOT
					ROOT = { naval_sea_raider_trigger = yes }
				}
			}
			save_event_target_as = nb_defending_admiral
			add_character_modifier = { name = naval_battle_cd days = 60 hidden = yes }
			# The defender's side: the top liege or liege at war with ROOT, else the admiral himself
			if = {
				limit = { top_liege = { any_current_enemy = { character = ROOT } } }
				top_liege = { save_event_target_as = nb_defending_liege }
			}
			else_if = {
				limit = { liege = { any_current_enemy = { character = ROOT } } }
				liege = { save_event_target_as = nb_defending_liege }
			}
			else = {
				save_event_target_as = nb_defending_liege
			}
		}
		if = {
			limit = { event_target:nb_defending_admiral = { always = yes } }
			save_event_target_as = nb_attacking_admiral
			add_character_modifier = { name = naval_battle_cd days = 60 hidden = yes }
			# The attacker's side, the same way round
			if = {
				limit = { top_liege = { any_current_enemy = { character = event_target:nb_defending_liege } } }
				top_liege = { save_event_target_as = nb_attacking_liege }
			}
			else_if = {
				limit = { liege = { any_current_enemy = { character = event_target:nb_defending_liege } } }
				liege = { save_event_target_as = nb_attacking_liege }
			}
			else = {
				save_event_target_as = nb_attacking_liege
			}
			character_event = { id = naval.101 }
		}
	}
}
