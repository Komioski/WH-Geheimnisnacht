###############################################################################
# Nehekhara: the Tomb Kings and their Liche Priests
# Namespace: tombking (Part 1, was khemri.1 and khemri.4 in
#   BR_khemri_events.txt)
# Faction: Tomb Kings
# Serves: the Tomb Kings of Nehekhara
#
# A player who wakes as a ruler of Nehekhara chooses to rule as a Tomb King
# or as a Liche Priest wielding the Lore of Necromancy. A Tomb King or Liche
# Priest can spend prestige to have the priests wake more of the legions
# sleeping beneath a demesne province, in three ever dearer stages.
#
# Part 1: the awakening
#   King or priest                 tombking.1
#   Awaken my servants             tombking.2
#
# Callers: wh_assign_racial_trait.txt (tombking.1, players only);
#   decisions/wh_tomb_kings_decisions.txt (tk_awaken_servants: 2).
# Saved: province modifiers tomb_kings_awakened_province_1 - 3
#   (wh_modifiers.txt); event target tk_tomb_province.
# Dropped: the commented-out khemri.0 - khemri.3 (old trait assignment,
#   now done by wh_assign_racial_trait.txt). tombking.1 no longer adds a
#   second 100 years of age on top of the racial effect.
# Loc: localisation/00_WHG_nehekhara_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = tombking

# tombking.1 King or priest
# A player who wakes in Nehekhara chooses to rule as a Tomb King or as a
# Liche Priest (the Lore of Necromancy and magic).
# Scope: ROOT is the player.
character_event = {
	id = tombking.1
	desc = EVTDESCtombking1
	picture = GFX_evt_large_army

	is_triggered_only = yes # wh_assign_racial_trait.txt

	option = { # A Tomb King
		name = EVTOPTAtombking1
		add_trait = khemri_tomb_king
	}
	option = { # A Liche Priest
		name = EVTOPTBtombking1
		add_trait = khemri_liche_priest
		add_trait = lore_necromancy
		hidden_effect = {
			enable_magic_xp_effect = yes
			magic_lvl_increase_effect = yes
		}
	}
}

# tombking.2 Awaken my servants
# The Liche Priests wake the next stage of the legions under a demesne
# province: more levies, a bigger garrison and more tax, for 250, 500 or
# 1000 prestige.
# Scope: ROOT is the Tomb King or Liche Priest, event_target:tk_tomb_province
#   the province.
character_event = {
	id = tombking.2
	desc = EVTDESCtombking2
	picture = GFX_evt_khemri

	is_triggered_only = yes # tk_awaken_servants decision

	immediate = {
		if = {
			limit = { capital_scope = { NOT = { has_province_modifier = tomb_kings_awakened_province_3 } } }
			capital_scope = { save_event_target_as = tk_tomb_province }
		}
		else = {
			random_demesne_province = {
				limit = { NOT = { has_province_modifier = tomb_kings_awakened_province_3 } }
				save_event_target_as = tk_tomb_province
			}
		}
	}

	option = { # First stage
		name = EVTOPTAtombking2
		trigger = {
			prestige = 250
			event_target:tk_tomb_province = {
				NOR = {
					has_province_modifier = tomb_kings_awakened_province_1
					has_province_modifier = tomb_kings_awakened_province_2
				}
			}
		}
		prestige = -250
		event_target:tk_tomb_province = {
			add_province_modifier = { name = tomb_kings_awakened_province_1 duration = -1 }
		}
	}
	option = { # Second stage
		name = EVTOPTAtombking2
		trigger = {
			prestige = 500
			event_target:tk_tomb_province = { has_province_modifier = tomb_kings_awakened_province_1 }
		}
		prestige = -500
		event_target:tk_tomb_province = {
			remove_province_modifier = tomb_kings_awakened_province_1
			add_province_modifier = { name = tomb_kings_awakened_province_2 duration = -1 }
		}
	}
	option = { # Third stage
		name = EVTOPTAtombking2
		trigger = {
			prestige = 1000
			event_target:tk_tomb_province = { has_province_modifier = tomb_kings_awakened_province_2 }
		}
		prestige = -1000
		event_target:tk_tomb_province = {
			remove_province_modifier = tomb_kings_awakened_province_2
			add_province_modifier = { name = tomb_kings_awakened_province_3 duration = -1 }
		}
	}
	option = { # Not now
		name = EVTOPTBtombking2
		ai_chance = { factor = 0 }
	}
}
