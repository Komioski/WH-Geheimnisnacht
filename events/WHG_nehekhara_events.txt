# Faction: Tomb Kings
# Serves: the Tomb Kings of Nehekhara
#
# A player who wakes as a ruler of Nehekhara chooses to rule as a Tomb King
# or as a Liche Priest wielding the Lore of Necromancy. A Tomb King or Liche
# Priest can spend prestige to have the priests wake more of the legions
# sleeping beneath a demesne province, in three ever dearer stages.
#
# Part 1: the awakening
#   King or priest                 tombking.1
#   Awaken my servants             tombking.2
#
# Callers: wh_assign_racial_trait.txt (tombking.1, players only);
#   decisions/wh_tomb_kings_decisions.txt (tk_awaken_servants: 2).
# Saved: province modifiers tomb_kings_awakened_province_1 - 3
#   (wh_modifiers.txt); event target tk_tomb_province.
# Dropped: the commented-out khemri.0 - khemri.3 (old trait assignment,
#   now done by wh_assign_racial_trait.txt). tombking.1 no longer adds a
#   second 100 years of age on top of the racial effect.
# Loc: localisation/00_WHG_nehekhara_events.csv
# Full list with callers: docs/events/event_database.csv

namespace = tombking

# tombking.1 King or priest
# A player who wakes in Nehekhara chooses to rule as a Tomb King or as a
# Liche Priest (the Lore of Necromancy and magic).
# Scope: ROOT is the player.
character_event = {
	id = tombking.1
	desc = EVTDESCtombking1
	picture = GFX_evt_large_army

	is_triggered_only = yes # wh_assign_racial_trait.txt

	option = { # A Tomb King
		name = EVTOPTAtombking1
		add_trait = khemri_tomb_king
	}
	option = { # A Liche Priest
		name = EVTOPTBtombking1
		add_trait = khemri_liche_priest
		add_trait = lore_necromancy
		hidden_effect = {
			enable_magic_xp_effect = yes
			magic_lvl_increase_effect = yes
		}
	}
}

# tombking.2 Awaken my servants
# The Liche Priests wake the next stage of the legions under a demesne
# province: more levies, a bigger garrison and more tax, for 250, 500 or
# 1000 prestige.
# Scope: ROOT is the Tomb King or Liche Priest, event_target:tk_tomb_province
#   the province.
character_event = {
	id = tombking.2
	desc = EVTDESCtombking2
	picture = GFX_evt_khemri

	is_triggered_only = yes # tk_awaken_servants decision

	immediate = {
		if = {
			limit = { capital_scope = { NOT = { has_province_modifier = tomb_kings_awakened_province_3 } } }
			capital_scope = { save_event_target_as = tk_tomb_province }
		}
		else = {
			random_demesne_province = {
				limit = { NOT = { has_province_modifier = tomb_kings_awakened_province_3 } }
				save_event_target_as = tk_tomb_province
			}
		}
	}

	option = { # First stage
		name = EVTOPTAtombking2
		trigger = {
			prestige = 250
			event_target:tk_tomb_province = {
				NOR = {
					has_province_modifier = tomb_kings_awakened_province_1
					has_province_modifier = tomb_kings_awakened_province_2
				}
			}
		}
		prestige = -250
		event_target:tk_tomb_province = {
			add_province_modifier = { name = tomb_kings_awakened_province_1 duration = -1 }
		}
	}
	option = { # Second stage
		name = EVTOPTAtombking2
		trigger = {
			prestige = 500
			event_target:tk_tomb_province = { has_province_modifier = tomb_kings_awakened_province_1 }
		}
		prestige = -500
		event_target:tk_tomb_province = {
			remove_province_modifier = tomb_kings_awakened_province_1
			add_province_modifier = { name = tomb_kings_awakened_province_2 duration = -1 }
		}
	}
	option = { # Third stage
		name = EVTOPTAtombking2
		trigger = {
			prestige = 1000
			event_target:tk_tomb_province = { has_province_modifier = tomb_kings_awakened_province_2 }
		}
		prestige = -1000
		event_target:tk_tomb_province = {
			remove_province_modifier = tomb_kings_awakened_province_2
			add_province_modifier = { name = tomb_kings_awakened_province_3 duration = -1 }
		}
	}
	option = { # Not now
		name = EVTOPTBtombking2
		ai_chance = { factor = 0 }
	}
}

# nehekhara.1 Struck down, not slain
# A Tomb King (or Malekith, Drachenfels, Arkhan, the Everqueen's heart, the
# Sisters of Twilight) is struck down and falls into the sleep of the dead
# until the magic binding the soul brings the body back.
# Scope: ROOT is the fallen ruler.
character_event = {
	id = nehekhara.1
	picture = GFX_evt_khemri_ruins
	desc = {
		trigger = { culture = nehekharan }
		text = EVTDESCAnehekhara1
		picture = GFX_evt_khemri_ruins
	}
	desc = {
		trigger = { NOT = { culture = nehekharan } }
		text = EVTDESCBnehekhara1
		picture = GFX_evt_realm_chaos
	}

	is_triggered_only = yes # antideath_effect (wh_scripted_effects.txt), WHG_amazon_events sacrifice

	option = {
		name = { trigger = { culture = nehekharan } text = EVTOPTAnehekhara1 }
		name = { trigger = { NOT = { culture = nehekharan } } text = EVTOPTBnehekhara1 }
		add_trait = incapacitated
		hidden_effect = {
			any_realm_province = {
				limit = {
					owner = { character = ROOT }
					OR = { culture = nehekharan ROOT = { trait = creature_nehekharan } }
				}
				add_province_modifier = { name = incapacitated_tomb_king duration = 9000 }
			}
			if = {
				limit = { OR = { trait = king_settra trait = drachenfels trait = black_arkhan } }
				character_event = { id = nehekhara.2 days = 730 random = 3650 }
			}
			else_if = {
				limit = { trait = malekith }
				character_event = { id = nehekhara.2 days = 365 random = 3650 }
			}
			else_if = {
				limit = {
					OR = {
						trait = fay_enchantress
						AND = {
							any_artifact = {
								artifact = heart_of_avelorn_everqueen
								is_artifact_active = yes
							}
							NOT = { has_character_flag = exhausted_heart }
						}
					}
				}
				character_event = { id = nehekhara.2 days = 365 }
			}
			else_if = {
				limit = { trait = sisters_of_twilight } # They are supposed to have near instantaneous revival
				if = {
					limit = {
						OR = {
							c_60018 = {
								is_alive = yes
								had_character_flag = { flag = protection_of_the_forest years <= 1 }
							}
							c_60019 = {
								is_alive = yes
								had_character_flag = { flag = protection_of_the_forest years <= 1 }
							}
						}
					}
					death = { death_reason = death_trait }
				}
				else = {
					health = -0.05
					clr_character_flag = protection_of_the_forest
					set_character_flag = protection_of_the_forest
					character_event = { id = nehekhara.2 days = 10 }
				}
			}
			else = { character_event = { id = nehekhara.2 days = 1095 random = 7300 } }
		}
	}
}

# nehekhara.3 Tomb robbers
# A desert expedition (or an adventurer on a loot-and-explore quest) robs
# the tombs of a Nehekharan holding: gold and prestige, a ruined building,
# and a 1% chance of tomb rot.
# Scope: ROOT is the robber, FROM the holding title.
character_event = {
	id = nehekhara.3
	desc = EVTDESCnehekhara3
	picture = GFX_evt_khemri_ruins
	only_playable = yes
	is_triggered_only = yes

	trigger = {
		NOT = { trait = creature_nehekharan }
		OR = {
			any_war = { attacker = { character = ROOT } using_cb = desert_expedition }
			has_quest = quest_adventurer_loot_and_explore
		}
		FROM = {
			OR = {
				location = { region = world_nehekhara culture = nehekharan }
				AND = { location = { culture = nehekharan } county = { is_quest_target_of = ROOT } }
			}
		}
	}

	option = {
		name = EVTOPTAnehekhara3
		if = {
			limit = { FROM = { holder_scope = { top_liege = { NOT = { character = PREV } } } } }
			FROM = { holder_scope = { top_liege = { character_event = { id = nehekhara.5 } } } }
		}
		FROM = { holder_scope = { character_event = { id = nehekhara.4 } } }
		if = {
			limit = {
				AND = {
					ROOT = { has_quest = quest_adventurer_loot_and_explore }
					FROM = { county = { is_quest_target_of = ROOT } }
				}
			}
			clr_quest = quest_adventurer_loot_and_explore
			sound_effect = hermetic_society_observe_planet
			add_mission_succeed_influence_effect = yes
			add_society_currency_medium_effect = yes
		}
		wealth = 75
		prestige = 50
		hidden_tooltip = {
			random = { chance = 25 FROM = { destroy_random_building = yes } }
			random = { chance = 1 ROOT = { character_event = { id = nehekhara.7 days = 1 } } }
		}
	}
}

# nehekhara.9 A landless Tomb King plans to take back his land
character_event = {
	id = nehekhara.9
	hide_window = yes

	min_age = 16
	capable_only = yes
	prisoner = no
	religion = nehekharan_gods

	trigger = {
		has_dlc = "The Old Gods"
		is_ruler = no
		trait = khemri_tomb_king
		any_claim = {
			has_holder = yes
			holder_scope = {
				independent = yes # Only do this vs independent rulers
				OR = {
					is_merchant_republic = no
					NOT = { capital_holding = { de_jure_liege_or_above = PREVPREV } }
				}
			}
			OR = { tier = KING holder_scope = { tier = PREV } }
		}

		NOT = { # Somewhat clunky way of checking that the character does not stand to inherit any titles
			any_heir_title = { always = yes }
		}

		diplomacy = 7

		health = 2
		NOT = { trait = content }
		NOT = { trait = craven }

		NOT = { has_character_modifier = planning_claimant_adventure }
	}

	mean_time_to_happen = {
		years = 50
		modifier = { factor = 0.2 has_character_flag = flag_denied_title }
		modifier = { factor = 0.5 trait = ambitious }
		modifier = { factor = 0.5 trait = envious }
		modifier = { factor = 0.75 diplomacy = 12 }
		modifier = { factor = 0.75 trait = brave }
		modifier = { factor = 0.75 trait = born_in_the_purple }
		modifier = { factor = 0.85 trait = proud }
		modifier = { factor = 1.5 trait = slothful }
		modifier = { factor = 1.5 trait = humble }
		modifier = { factor = 2.0 trait = kind }
	}

	immediate = {
		random_claim = {
			limit = {
				has_holder = yes
				holder_scope = {
					independent = yes # Only do this vs independent rulers
					OR = {
						# Don't target a liege unless I am ambitious
						NOT = { is_vassal_or_below = ROOT }
						ROOT = { trait = ambitious }
					}
					OR = {
						is_merchant_republic = no
						NOT = { capital_holding = { de_jure_liege_or_above = PREVPREV } }
					}
				}
			}

			holder_scope = {
				reverse_opinion = {
					who = ROOT
					years = 3
					modifier = opinion_claimant_adventure_target
				}
				character_event = {
					id = adventurer.51 # Warn the target
				}
			}
			ROOT = {
				add_character_modifier = {
					name = planning_claimant_adventure
					hidden = yes
					days = 730
				}
				character_event = {
					id = adventurer.52 # Fire the attack
					days = 730
				}
			}
		}
		if = {
			limit = { trait = in_hiding }
			remove_trait = in_hiding
			add_character_modifier = { name = went_out_of_hiding_timer duration = 180 hidden = yes }
			hidden_tooltip = { character_event = { id = hiding.7 } } # Notify plotters and family
		}
		set_character_flag = do_not_disturb
		character_event = { id = adventurer.59 days = 730 }
	}
}
