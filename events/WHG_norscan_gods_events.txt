# The gods of the north: patrons and the call of the Dark Gods
# Namespace: norgods (was ccgs.2, ccgs.4 - 5 and DR_startup.9 in
#   GS_decisionevents.txt)
# Faction: Norsca
# Serves: Norscan rulers and the Beastmen herds of Chaos Undivided
#
# The Norscans honour many gods: Kjorn the Lord of Battle, Loesh the Serpent,
# Nierg the Worm God and Tchar the Raven God, whom the southern lands call
# the Ruinous Powers, and the older gods of the north, Olric the Snow King
# and Orsen the Great Bear. Most Norscans give gifts to all of them, but a
# chieftain may raise one god above the others as a patron, and the realm's
# pantheon law decides which gods a chieftain may raise. A Norscan who
# tires of the old ways may give them up for the Dark Gods altogether.
#
# Turn to the Dark Gods                norgods.1
# Choose a patron god                  norgods.2
# Patrons at the start of the game     norgods.10
#
# Callers: norgods.1 from the convertchaosmenu decision and norgods.2 from
#   the patron_dark_god decision (decisions/wh_norscan_decisions.txt; the
#   AI picks with norscan_pick_patron_effect instead); norgods.10 from
#   on_startup (common/on_actions/00_on_actions_warbands.txt).
# Scripted triggers: common/scripted_triggers/WHG_norscan_gods_triggers.txt
# Scripted effects: common/scripted_effects/WHG_norscan_gods_effects.txt
# Localisation: localisation/00_WHG_norscan_gods_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = norgods

# norgods.1 Turn to the Dark Gods
# From the convertchaosmenu decision: a Norscan leaves the old ways for one of
# the Dark Gods or for Chaos Undivided. A Beastman of Chaos Undivided picks
# one god. A warband pays in favour, anyone else in piety.
# Scope: ROOT is the convert.
character_event = {
	id = norgods.1
	title = EVTNAMEnorgods1
	desc = { trigger = { religion = norscan_gods } text = EVTDESCnorgods1 }
	desc = { trigger = { NOT = { religion = norscan_gods } } text = EVTDESCnorgods1_undivided }
	picture = GFX_evt_khorne
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # convertchaosmenu

	option = { # Khorne
		name = EVTOPTAnorgods1
		trigger = { society_member_of = warband_of_khorne society_currency >= 1000 }
		norscan_dark_pact_cost_effect = yes
		religion = khorne
		norscan_convert_effect = yes
	}
	option = { # Slaanesh
		name = EVTOPTBnorgods1
		trigger = { society_member_of = warband_of_slaanesh society_currency >= 1000 }
		norscan_dark_pact_cost_effect = yes
		religion = slaanesh
		norscan_convert_effect = yes
	}
	option = { # Nurgle
		name = EVTOPTCnorgods1
		trigger = { society_member_of = warband_of_nurgle society_currency >= 1000 }
		norscan_dark_pact_cost_effect = yes
		religion = nurgle
		norscan_convert_effect = yes
	}
	option = { # Tzeentch
		name = EVTOPTDnorgods1
		trigger = { society_member_of = warband_of_tzeentch society_currency >= 1000 }
		norscan_dark_pact_cost_effect = yes
		religion = tzeentch
		norscan_convert_effect = yes
	}
	option = { # Chaos Undivided
		name = EVTOPTEnorgods1
		trigger = { NOT = { religion = chaos } }
		norscan_dark_pact_cost_effect = yes
		religion = chaos
		norscan_convert_effect = yes
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0.1
				is_member_of_any_warband_trigger = yes
				NOT = { society_member_of = warband_of_chaos }
			}
		}
	}
	option = { # not yet
		name = EVTOPTFnorgods1
		trigger = { ai = no }
	}
}

# norgods.2 Choose a patron god
# From the patron_dark_god decision (players; the AI uses
# norscan_pick_patron_effect). Close kin at court take up the same god.
# Scope: ROOT is the Norscan.
character_event = {
	id = norgods.2
	desc = EVTDESCnorgods2
	picture = GFX_evt_runestone_raising_oldgods
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # patron_dark_god

	option = { # Kjorn
		name = EVTOPTAnorgods2
		trigger = { norscan_dark_patron_allowed_trigger = yes }
		add_trait = norscan_patron_kharneth
		norscan_share_patron_effect = yes
	}
	option = { # Loesh
		name = EVTOPTBnorgods2
		trigger = { norscan_dark_patron_allowed_trigger = yes }
		add_trait = norscan_patron_shornaal
		norscan_share_patron_effect = yes
	}
	option = { # Nierg
		name = EVTOPTCnorgods2
		trigger = { norscan_dark_patron_allowed_trigger = yes }
		add_trait = norscan_patron_nurgleth
		norscan_share_patron_effect = yes
	}
	option = { # Tchar
		name = EVTOPTDnorgods2
		trigger = { norscan_dark_patron_allowed_trigger = yes }
		add_trait = norscan_patron_tchar
		norscan_share_patron_effect = yes
	}
	option = { # Olric
		name = EVTOPTEnorgods2
		trigger = { norscan_old_patron_allowed_trigger = yes }
		add_trait = norscan_patron_ulric
		norscan_share_patron_effect = yes
	}
	option = { # Orsen
		name = EVTOPTFnorgods2
		trigger = { norscan_old_patron_allowed_trigger = yes }
		add_trait = norscan_patron_ursun
		norscan_share_patron_effect = yes
	}
	option = { # not yet
		name = EVTOPTGnorgods2
	}
}

# norgods.10 Patrons at the start of the game
# on_startup: most AI Norscan chieftains already honour a patron god. About
# one in six follows all the gods alike.
# Scope: ROOT is the Norscan.
character_event = {
	id = norgods.10
	hide_window = yes

	is_triggered_only = yes # on_startup

	trigger = {
		is_save_game = no
		ai = yes
		is_adult = yes
		NOT = { dynasty = none }
		culture_group = norscan_group
		religion = norscan_gods
		has_norscan_patron_trigger = no
	}

	immediate = {
		random = { chance = 83 norscan_pick_patron_effect = yes }
	}
}
