# The gods of the north: patrons and the call of the Dark Gods
# Namespace: norgods (was ccgs.2, ccgs.4 - 5 and DR_startup.9 in
#   GS_decisionevents.txt; norsca.2 - 8 in wh_norsca_events.txt)
# Faction: Norsca
# Serves: Norscan rulers and the Beastmen herds of Chaos Undivided
#
# The Norscans honour many gods: Kjorn the Lord of Battle, Loesh the Serpent,
# Nierg the Worm God and Tchar the Raven God, whom the southern lands call
# the Ruinous Powers, and the older gods of the north, Olric the Snow King
# and Orsen the Great Bear. Most Norscans give gifts to all of them, but a
# chieftain may raise one god above the others as a patron, and the realm's
# pantheon law decides which gods a chieftain may raise. A Norscan who
# tires of the old ways may give them up for the Dark Gods altogether, and
# a small chieftain at war may hear one of them call. In the north only
# the strong may lead: an heir the clans' god does not favour loses the
# great vassals when the ruler dies.
#
# Turn to the Dark Gods                norgods.1
# Choose a patron god                  norgods.2
# Patrons at the start of the game     norgods.10
# The Dark Gods call                   norgods.20 - norgods.24
# Only the strong may lead             norgods.30 - norgods.32
#
# Callers: norgods.1 from the convertchaosmenu decision and norgods.2 from
#   the patron_dark_god decision (decisions/wh_norscan_decisions.txt; the
#   AI picks with norscan_pick_patron_effect instead); norgods.10 from
#   on_startup (common/on_actions); norgods.20 -
#   24 from the norsca list of on_decade_pulse, norgods.30 from on_death
#   and norgods.32 from on_bi_yearly_pulse (WHG_pulse_on_actions.txt).
# Scripted triggers: common/scripted_triggers/WHG_chaos_triggers.txt
# Scripted effects: common/scripted_effects/WHG_chaos_effects.txt
# Localisation: localisation/00_WHG_norscan_gods_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = norgods

# norgods.1 Turn to the Dark Gods
# From the convertchaosmenu decision: a Norscan leaves the old ways for one of
# the Dark Gods or for Chaos Undivided. A Beastman of Chaos Undivided picks
# one god. A warband pays in favour, anyone else in piety.
# Scope: ROOT is the convert.
character_event = {
	id = norgods.1
	title = EVTNAMEnorgods1
	desc = { trigger = { religion = norscan_gods } text = EVTDESCnorgods1 }
	desc = { trigger = { NOT = { religion = norscan_gods } } text = EVTDESCnorgods1_undivided }
	picture = GFX_evt_khorne
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # convertchaosmenu

	option = { # Khorne
		name = EVTOPTAnorgods1
		trigger = { society_member_of = warband_of_khorne society_currency >= 1000 }
		norscan_dark_pact_cost_effect = yes
		religion = khorne
		norscan_convert_effect = yes
	}
	option = { # Slaanesh
		name = EVTOPTBnorgods1
		trigger = { society_member_of = warband_of_slaanesh society_currency >= 1000 }
		norscan_dark_pact_cost_effect = yes
		religion = slaanesh
		norscan_convert_effect = yes
	}
	option = { # Nurgle
		name = EVTOPTCnorgods1
		trigger = { society_member_of = warband_of_nurgle society_currency >= 1000 }
		norscan_dark_pact_cost_effect = yes
		religion = nurgle
		norscan_convert_effect = yes
	}
	option = { # Tzeentch
		name = EVTOPTDnorgods1
		trigger = { society_member_of = warband_of_tzeentch society_currency >= 1000 }
		norscan_dark_pact_cost_effect = yes
		religion = tzeentch
		norscan_convert_effect = yes
	}
	option = { # Chaos Undivided
		name = EVTOPTEnorgods1
		trigger = { NOT = { religion = chaos } }
		norscan_dark_pact_cost_effect = yes
		religion = chaos
		norscan_convert_effect = yes
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0.1
				is_member_of_any_warband_trigger = yes
				NOT = { society_member_of = warband_of_chaos }
			}
		}
	}
	option = { # not yet
		name = EVTOPTFnorgods1
		trigger = { ai = no }
	}
}

# norgods.20 The call of Tchar (was norsca.2)
# A small Norscan chieftain in the thick of a war hears Tchar and may give up
# the many gods of the north for Tzeentch alone. Refusing sets
# refusedcallofchaos, and no Dark God calls again.
# Scope: ROOT is the chieftain.
character_event = {
	id = norgods.20
	title = EVTNAMEnorgods20
	desc = EVTDESCnorgods20
	picture = GFX_evt_tzeentch
	border = GFX_event_normal_frame_religion

	min_age = 16
	capable_only = yes
	only_rulers = yes
	religion = norscan_gods

	is_triggered_only = yes # on_decade_pulse (norsca list, WHG_pulse_on_actions.txt)

	trigger = {
		demesne_size <= 3
		norscan_dark_call_trigger = yes
		any_war = { days_since_last_hostile_action = { who = ROOT days = 180 } }
	}

	weight_multiplier = {
		factor = 1
		# Virtue dependent modifiers
		modifier = { factor = 1.3 trait = envious }
		modifier = { factor = 1.3 trait = patient }
		modifier = { factor = 1.3 trait = diligent }
		modifier = { factor = 0.3 trait = kind }
		# Personality dependent modifiers
		modifier = { factor = 1.1 trait = ambitious }
		modifier = { factor = 1.1 trait = deceitful }
		modifier = { factor = 0.3 trait = just }
		modifier = { factor = 1.5 trait = arbitrary }
		modifier = { factor = 1.5 trait = cynical }
		modifier = { factor = 0.4 trait = honest }
		modifier = { factor = 0.4 trait = brave }
		modifier = { factor = 1.1 trait = cruel }
		# mutations
		modifier = { factor = 1.5 is_mutated_trigger = yes }
		# Tzeenth favors magic power
		modifier = {
			factor = 1.5
			OR = {
				trait = magic_potential_1
				trait = magic_potential_2
				trait = magic_potential_3
				trait = magic_potential_4
			}
		}
		# And intellect
		modifier = { factor = 1.3 OR = { learning = 10 intrigue = 10 } }
		modifier = { factor = 1.3 OR = { learning = 15 intrigue = 15 } }
		#Tzeentch's Favoured WoL Focuses
		modifier = {
			factor = 1.1
			OR = { has_focus = focus_scholarship has_focus = focus_intrigue }
		}
		#Patron of Tchar
		modifier = { factor = 1.2 trait = norscan_patron_tchar }
	}

	option = { # Refuse
		name = EVTOPTAnorgods20
		ai_chance = {
			factor = 95
			modifier = { factor = 1.2 trait = brave }
			modifier = { factor = 1.4 trait = zealous }
			modifier = { factor = 1.2 is_strong_trigger = yes }
			modifier = { factor = 1.2 trait = just }
			modifier = { factor = 1.2 trait = content }
			modifier = { factor = 1.2 trait = diligent }
		}
		prestige = -50
		piety = 200
		set_character_flag = refusedcallofchaos
	}
	option = { # Answer the call
		name = EVTOPTBnorgods20
		ai_chance = {
			factor = 5
			modifier = { factor = 1.2 trait = ambitious }
			modifier = { factor = 1.2 trait = arbitrary }
			modifier = { factor = 1.2 trait = mystic }
			modifier = { factor = 1.2 trait = deceitful }
			modifier = { factor = 1.2 trait = cynical }
			modifier = { factor = 1.2 trait = lunatic_1 }
			modifier = { factor = 1.3 trait = lunatic_2 }
			modifier = { factor = 1.4 trait = lunatic_3 }
		}
		prestige = 200
		piety = -50
		norscan_convert_effect = yes # keep the marks of the old patron
		religion = tzeentch
		norscan_convert_effect = yes
	}
}

# norgods.30 The clans weigh the heir (was norsca.7)
# Hidden. When the ruler of a northern realm dies and the heir is too weak
# for the heir's own god, the great vassals break away: they go to the dead
# ruler's liege, or stand on their own in an independent realm. The heir
# keeps only the demesne and hears of it in norgods.31.
# Fixed: the news went to the dead ruler, not to the heir who reads it;
#   the liege is kept with an event target instead of an opinion marker.
# Scope: ROOT is the dead ruler.
character_event = {
	id = norgods.30
	hide_window = yes

	only_playable = yes

	is_triggered_only = yes # on_death (WHG_character_on_actions.txt)

	trigger = {
		is_nomadic = no
		demesne_size >= 3
		norscan_strength_realm_trigger = yes
		current_heir = { norscan_heir_unworthy_trigger = yes }
	}

	immediate = {
		current_heir = { save_event_target_as = norscan_weak_heir }
		if = {
			limit = { independent = no }
			liege = { save_event_target_as = norscan_overlord }
			any_vassal = {
				limit = { higher_tier_than = BARON }
				set_defacto_liege = event_target:norscan_overlord
			}
			if = { # Keep the dead ruler's own oath
				limit = { NOT = { liege = { character = event_target:norscan_overlord } } }
				set_defacto_liege = event_target:norscan_overlord
			}
		}
		else = { any_vassal = { limit = { higher_tier_than = BARON } set_defacto_liege = THIS } }
		event_target:norscan_weak_heir = {
			if = { limit = { ai = no } character_event = { id = norgods.31 days = 1 } }
		}
	}
}
