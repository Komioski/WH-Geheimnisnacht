###############################################################################
# Northern rites: the blot and the feasts of the north
# Merged in during the Nov 12 deep pass (each keeps its own header below):
#   Reavers' loot (namespace reaver, was WHG_reaver_events)
#   Berserkers (namespace berserk, was WHG_berserker_events)
#   Holmgang (namespace holmgang, was WHG_holmgang_events)
#   Runestones (namespace runestone, was WHG_runestone_events)
#   Prepared invasions (namespace invasion, was WHG_prepared_invasion_events)
#
# Namespace: rites (was TOG.600 - TOG.699 in oldgods_blot_events.txt and
#   TOG.800 - TOG.899 in oldgods_pagan_feasts.txt)
# Faction: Norsca
# Serves: the old Norscan gods, the reformed faith and Norscans of the Dark
#   Gods (the blot); the gods of Kislev (the festival of Dazh); both feasts
#
# The blot: once in nine winters a Norscan lord summons the vassals. Those of
# the faith come; those of other faiths pay to be excused, or refuse and are
# remembered for it. Beasts are slaughtered and their blood sprinkled on the
# idols, then the prisoners are hanged one by one from the blood-tree, up to
# four. Kin, the head of a faith, dwarfs, elves, greenskins, lizardmen and the
# faithful of other gods each meet it their own way; the host may spare any of
# them. With no prisoners, thralls are bought and hanged instead. A feast
# follows, and a lord who made a true offering may keep the blood bowl
# (once per life).
#
# The feast: up to four times in a feast one guest makes a scene: drunk,
# gluttonous, mad or lustful in full view of the hall, or two guests find a
# friend or a lover (and may be caught). The hall judges each by its own
# vices. Shared by the blot and the festival of Dazh.
#
# The feast also follows the beastmen's Great Hunt (WHG_great_hunt_events.txt,
# host flag holding_nerge), at the herdstone.
#
# The festival of Dazh: in spring a lord of Kislev invites the vassals to
# welcome the sun. One man of the faith rides as Dazh in a golden chariot with
# a torch for the sun and a crown of golden straw; then the feast.
#
# The blot                             rites.1 - rites.15
# The blot feast                       rites.20 - rites.21
# Feast scenes                         rites.50 - rites.59
# The festival of Dazh                 rites.100 - rites.108
# Housekeeping                         rites.90
#
# Fired from: the hold_blot decision (rites.1) and the hold_jarilo decision
#   (rites.100) in minor_decisions.txt, both with the safety rites.90. Shared triggers and effects are in
#   WHG_old_gods_triggers.txt and WHG_old_gods_effects.txt; the blood bowl is
#   in 41_artifacts_old_gods.txt.
# Saved: event targets blot_victim, rites_feast_host, rites_feast_culprit,
#   rites_feast_partner; flags holding_blot, rites_blot_guest, rites_blot_paid,
#   rites_blot_victim, rites_victim_* (host), rites_feast_* and rites_scene_*
#   (host and guests), holding_jarilo, rites_dazh_guest, took_blot_bowl (once
#   per life); variables
#   blot_offerings (hanged) and blot_fetched (brought out).
###############################################################################

# rites.1 The summons
# The faithful vassals are summoned (rites.2), the others asked to pay
# (rites.3). The rite begins in ten days (rites.5).
# Scope: ROOT is the host. Fired from the hold_blot decision.
character_event = {
	id = rites.1
	desc = EVTDESCrites1
	picture = GFX_evt_viking_throneroom_oldgods
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPTArites1
		any_vassal = {
			limit = {
				rites_summonable_vassal_trigger = yes
				rites_blot_faith_trigger = yes
			}
			character_event = { id = rites.2 tooltip = EVTTOOLTIPrites2 }
		}
		any_vassal = {
			limit = {
				rites_summonable_vassal_trigger = yes
				NOT = { rites_blot_faith_trigger = yes }
			}
			character_event = { id = rites.3 tooltip = EVTTOOLTIPrites3 }
		}
		hidden_effect = { character_event = { id = rites.5 days = 10 } }
	}
}

	border = GFX_event_normal_frame_religion

		name = EVTOPTArites5
		trigger = { has_character_flag = holding_blot }
		custom_tooltip = { text = blot_begins }
		set_variable = { which = blot_offerings value = 0 }
		set_variable = { which = blot_fetched value = 0 }
		hidden_effect = {
			any_vassal = {
				limit = { has_character_flag = rites_blot_guest }
				character_event = { id = rites.5 }
			}
			character_event = { id = rites.10 days = 3 }
		}
		if = {
			limit = { ai = no }
			chronicle = {
				entry = CHRONICLE_HELD_BLOT
				picture = GFX_evt_viking_throneroom_oldgods
			}
		}
	}
	option = { # Guest
		name = EVTOPTBrites5
		trigger = { NOT = { has_character_flag = holding_blot } }
	}
}

# rites.10 The next prisoner
# Fetches the next prisoner from the host's dungeon (up to four, counted in
# blot_fetched). With no prisoner at all (none hanged, none spared), thralls
# are bought (rites.15); otherwise the feast begins (rites.20).
# Scope: ROOT is the host.
character_event = {
	id = rites.10

	hide_window = yes
	is_triggered_only = yes

	trigger = { has_character_flag = holding_blot }

	immediate = {
		if = {
			limit = {
				NOT = { check_variable = { which = blot_fetched value = 4 } }
				any_courtier = { rites_blot_victim_trigger = yes }
			}
			change_variable = { which = blot_fetched value = 1 }
			random_courtier = {
				limit = { rites_blot_victim_trigger = yes }
				set_character_flag = rites_blot_victim
				character_event = { id = rites.11 days = 3 }
			}
		}
		else_if = {
			limit = {
				NOT = { check_variable = { which = blot_offerings value = 1 } }
				NOT = { has_character_flag = rites_blot_spared_one }
			}
			character_event = { id = rites.15 days = 3 }
		}
		else = {
			character_event = { id = rites.20 days = 5 }
		}
	}
}

# Namespace: berserk (was TOG.3000 - TOG.3009 in oldgods_various_events.txt)
# Faction: Norsca
# Serves: Norscans, Norse dwarfs, Ulricans, the men of Kislev and all who keep
#   the old Norscan gods
#
# In the press of battle a wrathful warrior may be taken by the rage: the
# wolf-spirit of Ulric, the bear of Ursun, the fury of Grimnir, the howling
# gods of the north. A berserker hews through scores of foes, takes a spear
# or an arrow without noticing, or is cut off and dragged down. At home the
# rage may rise against a tiresome courtier. One day the body can no longer
# bear the rage, and the berserker grows old.
#
# The rage awakens                     berserk.1
# In battle                            berserk.2 - berserk.5
# Too old for the rage                 berserk.6
# The rage at court                    berserk.10 - berserk.11
#
# Fired from: the battle tombola battle.1 in WHG_battle_events.txt (berserk.1
#   - berserk.5; event targets home_commander, enemy_commander, enemy_liege),
#   on_bi_yearly_pulse in 00_on_actions.txt (berserk.6, and berserk.10 in the
#   random list). Shared triggers are in WHG_old_gods_triggers.txt; the
#   berserker's spirit speaks in WHG_old_gods_voices.txt (GetBerserkerSpirit).
#   The berserkers of Khorne (khorne_berserker) share these events, with the
#   Blood God's own texts (salvaged from berserker.0 - 7 of m_Khorne.txt,
#   which were disabled); a berserker of Khorne never grows too old.
# Saved: event target berserk_courtier; flags berserk_rage_lost,
#   berserk_rage_held (cleared by berserk.11).
###############################################################################

namespace = berserk

# berserk.1 The rage awakens
# A wrathful or aggressive warrior of the berserker peoples, young enough for
# the rage, or any open follower of Khorne, who becomes a berserker of
# Khorne. A coward is cured of it.
# Scope: ROOT is home_commander.
character_event = {
	id = berserk.1
	desc = {
		trigger = { berserk_khorne_can_rage_trigger = no }
		text = EVTDESCberserk1
	}
	desc = {
		trigger = { berserk_khorne_can_rage_trigger = yes }
		text = EVTDESCberserk1_khorne
	}
	picture = GFX_evt_viking_battle_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	trigger = {
		OR = {
			berserk_can_rage_trigger = yes
			berserk_khorne_can_rage_trigger = yes
		}
		berserk_is_berserker_trigger = no
		OR = {
			trait = wroth
			trait = aggressive_leader
		}
	}

	option = {
		name = {
			text = EVTOPTAberserk1
			trigger = { berserk_khorne_can_rage_trigger = no }
		}
		name = {
			text = EVTOPTAberserk1_khorne
			trigger = { berserk_khorne_can_rage_trigger = yes }
		}
		if = {
			limit = { trait = craven }
			remove_trait = craven
		}
		if = {
			limit = { berserk_khorne_can_rage_trigger = yes }
			add_trait = khorne_berserker
		}
		else = {
			add_trait = berserker
		}
		allow_new_battle_events_effect = yes
	}
}

# berserk.4 Cut off and dragged down
# The berserker dies, slain by the enemy commander when there is one. The
# home liege (battle.11) and the enemy liege or commander (battle.14) hear of
# it, as for any fallen commander.
# Scope: ROOT is home_commander.
character_event = {
	id = berserk.4
	desc = {
		trigger = { NOT = { trait = khorne_berserker } }
		text = EVTDESCberserk4
	}
	desc = {
		trigger = { trait = khorne_berserker }
		text = EVTDESCberserk4_khorne
	}
	picture = GFX_evt_viking_battle_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	trigger = {
		berserk_is_berserker_trigger = yes
		is_alive = yes
		is_invincible_trigger = no
	}

	immediate = {
		hidden_effect = {
			liege = {
				if = {
					limit = { NOT = { character = ROOT } }
					save_event_target_as = home_liege
					character_event = { id = battle.11 }
				}
			}
			event_target:enemy_liege = {
				if = {
					limit = {
						NOR = {
							is_vassal_or_below = ROOT
							character = ROOT
						}
					}
					set_character_flag = temp_battle_kill_242
					character_event = { id = battle.14 }
				}
			}
			event_target:enemy_commander = {
				if = {
					limit = {
						NOR = {
							has_character_flag = temp_battle_kill_242
							is_vassal_or_below = ROOT
							character = ROOT
						}
					}
					character_event = { id = battle.14 }
				}
			}
			event_target:enemy_liege = { clr_character_flag = temp_battle_kill_242 }
		}
	}

	option = {
		name = {
			text = EVTOPTAberserk4
			trigger = { NOT = { trait = khorne_berserker } }
		}
		name = {
			text = EVTOPTAberserk4_khorne
			trigger = { trait = khorne_berserker }
		}
		if = {
			limit = {
				event_target:enemy_commander = {
					is_alive = yes
					NOT = { character = ROOT }
				}
			}
			death = {
				death_reason = death_battle
				killer = event_target:enemy_commander
			}
		}
		else = {
			death = { death_reason = death_battle }
		}
	}
}

# Holmgang
# Namespace: holmgang (was TOG.5000 - TOG.5029 and TOG.5999 in
#   oldgods_holmgang_events.txt)
# Faction: Norsca
# Serves: Norscans of every faith and all who keep the old Norscan gods;
#   their women fight only as shieldmaidens, berserkers or duellists
#
# By the old law of the north a grudge between free warriors may be carried to
# the holm, a bare skerry in the cold sea, and fought out there with shield
# and axe before witnesses. A lord who hates a man of his own realm (his
# vassal, a fellow vassal or his own liege) challenges him. Refusing shames
# the challenged; accepting sends both to the holm a week later.
#
# The fight is weighed on the duel engine's skill bands: the better fighter
# wins most of the time, never always. The loser yields, flees (only a craven)
# or dies, and the winner may carry a scar from it. A berserker or a man of
# the Blood God kills more often, a kind one spares more often. By holm law the
# loser pays the winner a weregild (the dead pay with their goods), and a
# grudge fought out on the holm is settled for some years. The winner may take
# the loser's blade as a trophy (once per life).
#
# The challenge                        holmgang.1 - holmgang.5
# On the holm                          holmgang.10 - holmgang.11
# The outcome                          holmgang.20 - holmgang.22
# Housekeeping                         holmgang.90
#
# Fired from: the holmgang list on on_five_year_pulse in wh_on_actions.txt
#   (holmgang.1, was three MTTH events of 15 years). Shared triggers and
#   effects are in WHG_old_gods_triggers.txt and WHG_old_gods_effects.txt.
# Saved: event targets holmgang_challenger, holmgang_defender, holmgang_winner,
#   holmgang_loser; flags holmgang_challenger, holmgang_defender and the
#   outcome flags (cleared by holmgang_clear_flags_effect); opinion
#   opinion_holmgang_settled.
###############################################################################

namespace = holmgang

###############################################################################
# The challenge
###############################################################################

# holmgang.1 A grudge
# A lord who hates ROOT (ROOT's liege, a fellow vassal or one of ROOT's own
# vassals) decides to settle it on the holm. Hot-headed, strong and warlike
# men draw challenges more often.
# Scope: ROOT is the hated man (holmgang_defender).
character_event = {
	id = holmgang.1

	hide_window = yes
	is_triggered_only = yes # on_five_year_pulse (holmgang)

	has_dlc = "The Old Gods"
	min_age = 16
	max_age = 49
	capable_only = yes
	prisoner = no

	trigger = {
		holmgang_fighter_trigger = yes
		OR = {
			liege = { holmgang_grudge_against_root_trigger = yes }
			liege = {
				any_vassal = { holmgang_grudge_against_root_trigger = yes }
			}
			any_vassal = { holmgang_grudge_against_root_trigger = yes }
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 2 trait = wroth }
		modifier = { factor = 2 trait = duelist }
		modifier = { factor = 2 trait = berserker }
		modifier = { factor = 1.5 combat_rating = 50 }
		modifier = { factor = 1.5 is_strong_trigger = yes }
		modifier = { factor = 1.5 trait = brave }
		modifier = { factor = 1.4 trait = viking }
		modifier = { factor = 1.5 has_focus = focus_war }
		modifier = { factor = 0.7 trait = slothful }
		modifier = { factor = 0.5 NOT = { combat_rating = 10 } }
		modifier = { factor = 0.5 trait = content }
		modifier = { factor = 0.5 trait = kind }
		modifier = { factor = 0.25 trait = craven }
	}

	immediate = {
		save_event_target_as = holmgang_defender
		random_list = {
			10 = { # The liege
				modifier = {
					factor = 0
					NOT = {
						liege = { holmgang_grudge_against_root_trigger = yes }
					}
				}
				liege = { save_event_target_as = holmgang_challenger }
			}
			10 = { # A fellow vassal
				modifier = {
					factor = 0
					NOT = {
						liege = {
							any_vassal = { holmgang_grudge_against_root_trigger = yes }
						}
					}
				}
				liege = {
					random_vassal = {
						limit = { holmgang_grudge_against_root_trigger = yes }
						save_event_target_as = holmgang_challenger
					}
				}
			}
			10 = { # A vassal of ROOT's
				modifier = {
					factor = 0
					NOT = {
						any_vassal = { holmgang_grudge_against_root_trigger = yes }
					}
				}
				random_vassal = {
					limit = { holmgang_grudge_against_root_trigger = yes }
					save_event_target_as = holmgang_challenger
				}
			}
		}
		event_target:holmgang_challenger = {
			set_character_flag = holmgang_challenger
			set_character_flag = do_not_disturb
			character_event = { id = holmgang.90 days = 50 } # Safety
			character_event = { id = holmgang.2 }
		}
		set_character_flag = holmgang_defender
		set_character_flag = do_not_disturb
		character_event = { id = holmgang.90 days = 50 } # Safety
	}
}

		name = EVTOPTAholmgang10

		}

# holmgang.11 The fight
# The better fighter wins most of the time: from about 95 in 100 against a
# far weaker foe to 40 in 100 against an equal, with draws most likely
# between equals. holmgang_outcome_effect then decides how it ends.
# Scope: ROOT is the challenger.
character_event = {
	id = holmgang.11

	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_alive = yes
		prisoner = no
		event_target:holmgang_defender = { is_alive = yes prisoner = no }
	}

	immediate = {
		random_list = {
			40 = { # The challenger wins
				modifier = { factor = 2.375 event_target:holmgang_defender = { is_the_worst_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 2.125 event_target:holmgang_defender = { is_terrible_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 1.75 event_target:holmgang_defender = { is_bad_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 1.375 event_target:holmgang_defender = { is_ok_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.75 event_target:holmgang_defender = { is_good_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.45 event_target:holmgang_defender = { is_great_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.2 event_target:holmgang_defender = { is_fantastic_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.05 event_target:holmgang_defender = { is_the_best_dueler_compared_to_prev_trigger = yes } }
				save_event_target_as = holmgang_winner
				event_target:holmgang_defender = { save_event_target_as = holmgang_loser }
			}
			20 = { # A draw
				modifier = { factor = 0.15 event_target:holmgang_defender = { is_the_worst_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.35 event_target:holmgang_defender = { is_terrible_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.6 event_target:holmgang_defender = { is_bad_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.75 event_target:holmgang_defender = { is_ok_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.75 event_target:holmgang_defender = { is_good_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.6 event_target:holmgang_defender = { is_great_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.35 event_target:holmgang_defender = { is_fantastic_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.15 event_target:holmgang_defender = { is_the_best_dueler_compared_to_prev_trigger = yes } }
				set_character_flag = holmgang_drew
				event_target:holmgang_defender = { set_character_flag = holmgang_drew }
			}
			40 = { # The challenged wins
				modifier = { factor = 0.05 event_target:holmgang_defender = { is_the_worst_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.2 event_target:holmgang_defender = { is_terrible_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.45 event_target:holmgang_defender = { is_bad_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 0.75 event_target:holmgang_defender = { is_ok_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 1.375 event_target:holmgang_defender = { is_good_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 1.75 event_target:holmgang_defender = { is_great_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 2.125 event_target:holmgang_defender = { is_fantastic_dueler_compared_to_prev_trigger = yes } }
				modifier = { factor = 2.375 event_target:holmgang_defender = { is_the_best_dueler_compared_to_prev_trigger = yes } }
				event_target:holmgang_defender = { save_event_target_as = holmgang_winner }
				save_event_target_as = holmgang_loser
			}
		}
		if = {
			limit = { has_character_flag = holmgang_drew }
			character_event = { id = holmgang.22 }
			event_target:holmgang_defender = { character_event = { id = holmgang.22 } }
		}
		else = {
			holmgang_outcome_effect = yes
			event_target:holmgang_winner = { character_event = { id = holmgang.20 } }
			event_target:holmgang_loser = { character_event = { id = holmgang.21 } }
		}
	}
}

# holmgang.20 Victory on the holm
# The loser yields, flees or dies; a scar for a hard fight. A man of the
# Blood God who kills is noticed by his god. The loser pays the weregild in
# holmgang.21.
# Scope: ROOT is the winner (holmgang_winner).
character_event = {
	id = holmgang.20
	desc = {
		trigger = { has_character_flag = holmgang_kill has_character_flag = holmgang_scarred }
		text = EVTDESCholmgang20_kill_scarred
	}
	desc = {
		trigger = { has_character_flag = holmgang_kill NOT = { has_character_flag = holmgang_scarred } }
		text = EVTDESCholmgang20_kill
	}
	desc = {
		trigger = { has_character_flag = holmgang_yield has_character_flag = holmgang_scarred }
		text = EVTDESCholmgang20_yield_scarred
	}
	desc = {
		trigger = { has_character_flag = holmgang_yield NOT = { has_character_flag = holmgang_scarred } }
		text = EVTDESCholmgang20_yield
	}
	desc = {
		trigger = { has_character_flag = holmgang_flight }
		text = EVTDESCholmgang20_flight
	}
	picture = GFX_evt_melee
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	option = {
		name = EVTOPTAholmgang20
		prestige = 200
		if = {
			limit = { has_character_flag = holmgang_scarred }
			add_scarred_tiered_effect = yes
		}
		if = {
			limit = {
				has_character_flag = holmgang_kill
				religion_khorne_worshipper_trigger = yes
			}
			piety = 50
		}
		if = {
			limit = { has_character_flag = holmgang_kill }
			custom_tooltip = { text = holmgang_dead_goods_tt }
		}
		else_if = {
			limit = { has_character_flag = holmgang_yield }
			custom_tooltip = { text = holmgang_weregild_tt }
			opinion = {
				modifier = opinion_holmgang_settled
				who = event_target:holmgang_loser
			}
		}
		if = {
			limit = { NOT = { has_character_flag = holmgang_flight } }
			event_target:holmgang_loser = { save_event_target_as = other_combatant }
			duel_trophy_effect = yes
		}
		holmgang_clear_flags_effect = yes
	}
}

# holmgang.21 Defeat on the holm
# Yield (wounded or maimed, pay a tenth of your wealth as weregild), flee
# (a craven; a deep shame) or die (the winner takes a fifth of your wealth).
# The invincible cheat death.
# Scope: ROOT is the loser (holmgang_loser).
character_event = {
	id = holmgang.21
	desc = {
		trigger = { has_character_flag = holmgang_yield NOT = { has_character_flag = holmgang_maimed } }
		text = EVTDESCholmgang21_yield
	}
	desc = {
		trigger = { has_character_flag = holmgang_yield has_character_flag = holmgang_maimed }
		text = EVTDESCholmgang21_maimed
	}
	desc = {
		trigger = { has_character_flag = holmgang_kill }
		text = EVTDESCholmgang21_kill
	}
	desc = {
		trigger = { has_character_flag = holmgang_flight }
		text = EVTDESCholmgang21_flight
	}
	picture = GFX_evt_bloody_man
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	option = { # Yield
		name = EVTOPTAholmgang21
		trigger = { has_character_flag = holmgang_yield }
		prestige = -200
		if = {
			limit = { has_character_flag = holmgang_maimed }
			add_maimed_trait_effect = yes
		}
		else = {
			add_trait = wounded
		}
		transfer_scaled_wealth = {
			to = event_target:holmgang_winner
			value = 0.1
		}
		opinion = {
			modifier = opinion_holmgang_settled
			who = event_target:holmgang_winner
		}
		holmgang_clear_flags_effect = yes
	}
	option = { # Death
		name = EVTOPTBholmgang21
		trigger = {
			has_character_flag = holmgang_kill
			NOT = { religion_khorne_worshipper_trigger = yes }
		}
		holmgang_death_effect = yes
	}
	option = { # Death, for the Blood God
		name = EVTOPTCholmgang21
		trigger = {
			has_character_flag = holmgang_kill
			religion_khorne_worshipper_trigger = yes
		}
		holmgang_death_effect = yes
	}
	option = { # Flight
		name = EVTOPTDholmgang21
		trigger = { has_character_flag = holmgang_flight }
		prestige = -400
		holmgang_clear_flags_effect = yes
	}
}

# Prepared invasions
# Namespace: invasion (was 62300 - 62322 in misc_old_gods_events.txt)
# Faction: Norsca
# Serves: lords of the old Norscan gods (the only faith that may prepare an
#   invasion)
#
# A Norscan lord who prepares an invasion gathers warriors for it month by
# month: bands of warriors from the fjords, a smaller crew or two, now and
# then a renowned champion with his own hearth-guard. The greater the lord's
# name and skill, and the weaker the target, the more come. If the invasion
# becomes impossible the host breaks up; if the lord waits too long, the
# warriors drift away and the lord is shamed before the faithful.
#
# The invasion fails                   invasion.1 - invasion.2
# Warriors flock to the banner         invasion.10
#
# Fired from: on_prepared_invasion_aborts (invasion.1),
#   on_prepared_invasion_expires (invasion.2) and on_prepared_invasion_monthly
#   (invasion.10) in 00_on_actions.txt.
# Saved: flags invasion_band_large, invasion_band_small, invasion_champion
#   (cleared by invasion.10).
###############################################################################

namespace = invasion

###############################################################################
# The invasion fails
###############################################################################

# invasion.1 The invasion is no longer possible
# The gathered host is disbanded.
# Scope: ROOT is the would-be invader, FROM the target.
character_event = {
	id = invasion.1
	desc = EVTDESCinvasion1
	picture = GFX_evt_viking_battle_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	immediate = {
		remove_character_modifier = launched_prepared_invasion
		remove_opinion = {
			who = FROM
			modifier = preparing_to_invade
		}
		FROM = {
			remove_opinion = {
				who = ROOT
				modifier = preparing_to_invade_me
			}
		}
		disband_event_forces = prepared_invasion
	}

	option = {
		name = EVTOPTAinvasion1
	}
}

# invasion.2 Waited too long
# Unless the war has begun, the host drifts away and the lord loses 200
# prestige and 50 piety, and the lords of the faith in the realm think less
# of him.
# Scope: ROOT is the would-be invader, FROM the target.
character_event = {
	id = invasion.2
	desc = EVTDESCinvasion2
	picture = GFX_evt_viking_battle_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	immediate = {
		remove_opinion = {
			who = FROM
			modifier = preparing_to_invade
		}
		FROM = {
			remove_opinion = {
				who = ROOT
				modifier = preparing_to_invade_me
			}
		}
		if = {
			limit = {
				NOT = {
					any_war = {
						any_attacker = { character = FROM }
					}
				}
			}
			disband_event_forces = prepared_invasion
		}
	}

	option = {
		name = EVTOPTAinvasion2
		if = {
			limit = {
				NOT = {
					any_war = {
						any_attacker = { character = FROM }
					}
				}
			}
			prestige = -200
			piety = -50
			any_realm_lord = {
				limit = { religion = ROOT }
				opinion = {
					who = ROOT
					modifier = opinion_failed_invasion
				}
			}
		}
	}
}
