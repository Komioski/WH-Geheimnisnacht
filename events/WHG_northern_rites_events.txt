###############################################################################
# Northern rites: the blot and the feasts of the north
# Namespace: rites (was TOG.600 - TOG.699 in oldgods_blot_events.txt and
#   TOG.800 - TOG.899 in oldgods_pagan_feasts.txt)
# Faction: Norsca
# Serves: the old Norscan gods, the reformed faith and Norscans of the Dark
#   Gods (the blot); the gods of Kislev (the festival of Dazh); both feasts
#
# The blot: once in nine winters a Norscan lord summons the vassals. Those of
# the faith come; those of other faiths pay to be excused, or refuse and are
# remembered for it. Beasts are slaughtered and their blood sprinkled on the
# idols, then the prisoners are hanged one by one from the blood-tree, up to
# four. Kin, the head of a faith, dwarfs, elves, greenskins, lizardmen and the
# faithful of other gods each meet it their own way; the host may spare any of
# them. With no prisoners, thralls are bought and hanged instead. A feast
# follows, and a lord who made a true offering may keep the blood bowl
# (once per life).
#
# The feast: up to four times in a feast one guest makes a scene: drunk,
# gluttonous, mad or lustful in full view of the hall, or two guests find a
# friend or a lover (and may be caught). The hall judges each by its own
# vices. Shared by the blot and the festival of Dazh.
#
# The feast also follows the beastmen's Great Hunt (WHG_great_hunt_events.txt,
# host flag holding_nerge), at the herdstone.
#
# The festival of Dazh: in spring a lord of Kislev invites the vassals to
# welcome the sun. One man of the faith rides as Dazh in a golden chariot with
# a torch for the sun and a crown of golden straw; then the feast.
#
# The blot                             rites.1 - rites.15
# The blot feast                       rites.20 - rites.21
# Feast scenes                         rites.50 - rites.59
# The festival of Dazh                 rites.100 - rites.108
# Housekeeping                         rites.90
#
# Fired from: the hold_blot decision (rites.1) and the hold_jarilo decision
#   (rites.100) in minor_decisions.txt, both with the safety rites.90. Shared triggers and effects are in
#   WHG_old_gods_triggers.txt and WHG_old_gods_effects.txt; the blood bowl is
#   in 41_artifacts_old_gods.txt.
# Saved: event targets blot_victim, rites_feast_host, rites_feast_culprit,
#   rites_feast_partner; flags holding_blot, rites_blot_guest, rites_blot_paid,
#   rites_blot_victim, rites_victim_* (host), rites_feast_* and rites_scene_*
#   (host and guests), holding_jarilo, rites_dazh_guest, took_blot_bowl (once
#   per life); variables
#   blot_offerings (hanged) and blot_fetched (brought out).
###############################################################################

# rites.1 The summons
# The faithful vassals are summoned (rites.2), the others asked to pay
# (rites.3). The rite begins in ten days (rites.5).
# Scope: ROOT is the host. Fired from the hold_blot decision.

	id = rites.1
	desc = EVTDESCrites1
	picture = GFX_evt_viking_throneroom_oldgods
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPTArites1
		any_vassal = {
			limit = {
				rites_summonable_vassal_trigger = yes
				rites_blot_faith_trigger = yes
			}
			character_event = { id = rites.2 tooltip = EVTTOOLTIPrites2 }
		}
		any_vassal = {
			limit = {
				rites_summonable_vassal_trigger = yes
				NOT = { rites_blot_faith_trigger = yes }
			}
			character_event = { id = rites.3 tooltip = EVTTOOLTIPrites3 }
		}
		hidden_effect = { character_event = { id = rites.5 days = 10 } }
	}
}

	}

	}

	border = GFX_event_normal_frame_religion

	option = { # Host
		name = EVTOPTArites5
		trigger = { has_character_flag = holding_blot }
		custom_tooltip = { text = blot_begins }
		set_variable = { which = blot_offerings value = 0 }
		set_variable = { which = blot_fetched value = 0 }
		hidden_effect = {
			any_vassal = {
				limit = { has_character_flag = rites_blot_guest }
				character_event = { id = rites.5 }
			}
			character_event = { id = rites.10 days = 3 }
		}
		if = {
			limit = { ai = no }
			chronicle = {
				entry = CHRONICLE_HELD_BLOT
				picture = GFX_evt_viking_throneroom_oldgods
			}
		}
	}
	option = { # Guest
		name = EVTOPTBrites5
		trigger = { NOT = { has_character_flag = holding_blot } }
	}
}

# rites.10 The next prisoner
# Fetches the next prisoner from the host's dungeon (up to four, counted in
# blot_fetched). With no prisoner at all (none hanged, none spared), thralls
# are bought (rites.15); otherwise the feast begins (rites.20).
# Scope: ROOT is the host.
character_event = {
	id = rites.10

	hide_window = yes
	is_triggered_only = yes

	trigger = { has_character_flag = holding_blot }

	immediate = {
		if = {
			limit = {
				NOT = { check_variable = { which = blot_fetched value = 4 } }
				any_courtier = { rites_blot_victim_trigger = yes }
			}
			change_variable = { which = blot_fetched value = 1 }
			random_courtier = {
				limit = { rites_blot_victim_trigger = yes }
				set_character_flag = rites_blot_victim
				character_event = { id = rites.11 days = 3 }
			}
		}
		else_if = {
			limit = {
				NOT = { check_variable = { which = blot_offerings value = 1 } }
				NOT = { has_character_flag = rites_blot_spared_one }
			}
			character_event = { id = rites.15 days = 3 }
		}
		else = {
			character_event = { id = rites.20 days = 5 }
		}
	}
}
