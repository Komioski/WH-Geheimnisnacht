# oblast.1 The Frost Fair
# In a hard winter the rivers of Kislev freeze and a fair grows on the ice.
# Walk the fair (one of oblast.2 - 5), or stay in the warm. The ice may
# crack under the crowd (oblast.6).
# Scope: ROOT is a Kislevite lord whose capital is in Kislev, in winter.
character_event = {
	id = oblast.1
	desc = EVTDESCoblast1
	picture = GFX_evt_market
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # on_yearly_pulse oblast_fair
	only_rulers = yes
	min_age = 16
	capable_only = yes
	prisoner = no
	war = no

	trigger = {
		oblast_kislev_lord_trigger = yes
		NOT = { has_character_modifier = oblast_fair_cd }
		capital_scope = {
			region = world_old_world_kislev
			is_winter = yes
		}
	}

	immediate = {
		add_character_modifier = { name = oblast_fair_cd years = 3 hidden = yes }
	}

	# Walk the fair
	option = {
		name = EVTOPTAoblast1
		random_list = {
			30 = { character_event = { id = oblast.2 days = 1 } }
			25 = { character_event = { id = oblast.3 days = 1 } }
			20 = {
				modifier = {
					factor = 0
					OR = {
						NOT = { wealth = 100 }
						has_character_flag = oblast_jeweller_bought
					}
				}
				character_event = { id = oblast.4 days = 1 }
			}
			25 = { character_event = { id = oblast.5 days = 1 } }
		}
		random = {
			chance = 15
			character_event = { id = oblast.6 days = 2 }
		}
		ai_chance = { factor = 80 }
	}

	# Stay in the warm
	option = {
		name = EVTOPTBoblast1
		ai_chance = { factor = 20 }
	}
}

	id = oblast.11
	desc = EVTDESCoblast11
	picture = GFX_evt_horsemanship
	border = GFX_event_normal_frame_war

	option = {
		name = EVTOPTAoblast11

			25 = {
				modifier = { factor = 1.5 martial = 14 }
				modifier = { factor = 1.3 health = 6 }

				modifier = { factor = 1.3 trait = strong }
				prestige = 150
				custom_tooltip = { text = oblast_long_ride_won_tt }

					limit = {
						NOT = { has_character_flag = oblast_steppe_horse_given }

						chance = 25

						set_character_flag = has_mount

					}
				}
				if = {
					limit = { culture = gospodar }
					any_vassal = {
						limit = { culture = ungol }

					}
				}
			}
			55 = {
				prestige = 25

# oblast.30 The drinking contest
# A rival or a boyar challenges the lord to drink him under the table.
# Accept (a contest of health and martial, oblast.31), or decline and be
# mocked.
# Scope: ROOT is a Kislevite lord; event_target:oblast_drinker the challenger.
character_event = {
	id = oblast.30
	desc = EVTDESCoblast30
	picture = GFX_evt_feast
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # on_five_year_pulse oblast_life
	only_rulers = yes
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		oblast_kislev_lord_trigger = yes
		OR = {
			any_vassal = { oblast_drinker_trigger = yes }
			any_courtier = { oblast_drinker_trigger = yes }
		}
	}

	immediate = {
		if = {
			limit = { any_vassal = { oblast_drinker_trigger = yes } }
			random_vassal = {
				limit = { oblast_drinker_trigger = yes }
				preferred_limit = { is_rival = ROOT }
				save_event_target_as = oblast_drinker
			}
		}
		else = {
			random_courtier = {
				limit = { oblast_drinker_trigger = yes }
				preferred_limit = { is_rival = ROOT }
				save_event_target_as = oblast_drinker
			}
		}
	}

	# Accept
	option = {
		name = EVTOPTAoblast30
		character_event = { id = oblast.31 days = 1 }
		ai_chance = {
			factor = 70
			modifier = { factor = 2 trait = drunkard }
			modifier = { factor = 0.3 trait = temperate }
		}
	}

	# Decline
	option = {
		name = EVTOPTBoblast30
		prestige = -25
		ai_chance = { factor = 30 }
	}
}

# stanitsa.1 The ataman's choosing
# Once per ruler: an ageing Ungol lord's stanitsa wants to know which son
# will lead it. Choose by election, by duel, or by the bear-wrestling. The
# winner is the stanitsa's favoured heir; his brothers resent it. Rarely an
# Ungol horn bow for him.
# Scope: ROOT is an Ungol ruler over 50 with two adult sons;
# event_target:stanitsa_ataman the chosen son.
character_event = {
	id = stanitsa.1
	desc = EVTDESCstanitsa1
	picture = GFX_evt_steppe_mercenaries
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # on_five_year_pulse stanitsa_life

	only_rulers = yes
	min_age = 50
	culture = ungol

	trigger = {
		NOT = { has_character_flag = stanitsa_ataman_chosen }
		num_of_children = 2
		any_child = {
			count = 2
			is_adult = yes
			is_female = no
			is_alive = yes
			prisoner = no
		}
	}

	immediate = {
		set_character_flag = stanitsa_ataman_chosen
	}

	# Election
	option = {
		name = EVTOPTAstanitsa1
		random_child = {
			limit = { is_adult = yes is_female = no prisoner = no }
			preferred_limit = { diplomacy = 12 }
			preferred_limit = { diplomacy = 8 }
			save_event_target_as = stanitsa_ataman
		}
		ai_chance = { factor = 40 }
	}

	# Duel
	option = {
		name = EVTOPTBstanitsa1
		random_child = {
			limit = { is_adult = yes is_female = no prisoner = no }
			preferred_limit = { martial = 12 }
			preferred_limit = { martial = 8 }
			save_event_target_as = stanitsa_ataman
		}
		random_child = {
			limit = {
				is_adult = yes
				is_female = no
				NOT = { character = event_target:stanitsa_ataman }
			}
			random = { chance = 30 add_trait = wounded }
		}
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = wroth }
		}
	}

	# Bear-wrestling
	option = {
		name = EVTOPTCstanitsa1
		random_child = {
			limit = { is_adult = yes is_female = no prisoner = no }
			preferred_limit = { trait = strong }
			preferred_limit = { trait = robust }
			save_event_target_as = stanitsa_ataman
		}
		event_target:stanitsa_ataman = {
			random = { chance = 20 add_trait = wounded }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 2 religion = kislev_gods }
		}
	}

	after = {
		event_target:stanitsa_ataman = {
			prestige = 100
			add_character_modifier = { name = stanitsa_ataman_chosen duration = -1 }
			if = {
				limit = {
					NOT = { has_character_flag = stanitsa_bow_given }
					NOT = { has_artifact = stanitsa_ungol_horn_bow }
				}
				random = {
					chance = 10
					set_character_flag = stanitsa_bow_given
					add_artifact = stanitsa_ungol_horn_bow
				}
			}
		}
		any_child = {
			limit = {
				is_adult = yes
				is_female = no
				NOT = { character = event_target:stanitsa_ataman }
			}
			opinion = { who = event_target:stanitsa_ataman modifier = opinion_stanitsa_passed_over }
		}
	}
}
