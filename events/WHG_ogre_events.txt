###############################################################################
# Ogres: the hunger of the young and the girth of a chieftain
# Namespace: ogre (was srandoms.4 - 6 in S_minor_events.txt)
# Faction: Ogres
# Serves: the ogres of the Great Maw
#
# An ogre child feels the hunger of the Maw: eat, and grow big and strong,
# or refuse, and be beaten and fed by the guardian. At adulthood the size
# an ogre reached decides their standing in the tribe. A thin chieftain is
# mocked by the warriors.
#
#   ogre.1 - 4    The hunger of the young (once in a childhood, from age 8)
#   ogre.10       Grown up: big and strong, or weak and pushed around
#   ogre.20       A thin chieftain (MTTH 54 years, player ogre rulers of the
#                 Great Maw who are not fat)
#
# Fired from: WHG_character_on_actions.txt on_yearly_childhood_pulse (ogre.1) and
#   on_adulthood (ogre.10).
# Fixed: the hunger of the young came every year from 8 to 11, with 150
#   prestige each time; it comes once now. Its "diplomacy" outcomes were
#   triggers, not effects, and did nothing.
# Dropped: srandoms.7 (a jailer eats a prisoner), which nothing fired;
#   prison.90 in WHG_prison_events is that event.
# Saved: character flag ogre_hunger_tested.
# Loc: localisation/00_WHG_ogre_events.csv
###############################################################################

namespace = ogre

# ogre.1 The hunger of the young
# Scope: ROOT is an ogre child of the Great Maw.
character_event = {
	id = ogre.1
	desc = EVTDESCogre1
	picture = GFX_evt_beastmen_kids
	border = GFX_event_normal_frame_war

	min_age = 8
	max_age = 16
	religion = great_maw

	trigger = {
		trait = creature_ogre
		NOT = { has_character_flag = ogre_hunger_tested }
	}

	immediate = {
		set_character_flag = ogre_hunger_tested
	}

	is_triggered_only = yes

	option = {
		name = EVTOPTAogre1
		random = { chance = 30 add_trait = gluttonous }
		if = { limit = { NOT = { age = 10 } } change_martial = 1 }
		else_if = { limit = { NOT = { martial = 15 } } change_martial = 2 }
		else = { change_martial = 1 }
		if = { limit = { NOT = { health = 8 } } health = 1 }
		prestige = 150
		guardian = { character_event = { id = ogre.2 } }
		ai_chance = {
			factor = 80
			modifier = { factor = 2 trait = gluttonous }
			modifier = { factor = 1.5 trait = greedy }
		}
	}
	option = {
		name = EVTOPTBogre1
		prestige = -5
		piety = -25
		health = -0.5
		random_list = {
			15 = { change_stewardship = 1 }
			15 = { change_stewardship = 2 }
			15 = { change_intrigue = 1 }
			15 = { change_intrigue = 2 }
			15 = { change_diplomacy = 1 }
			15 = { change_diplomacy = 2 }
		}
		guardian = { character_event = { id = ogre.3 } }
		ai_chance = {
			factor = 20
			modifier = { factor = 2 trait = temperate }
			modifier = { factor = 1.5 trait = shy }
			modifier = { factor = 1.5 trait = weak }
			modifier = { factor = 1.5 trait = fragile }
		}
	}
}

# ogre.10 Grown up: the size of an ogre
# Scope: ROOT is the ogre coming of age.
character_event = {
	id = ogre.10
	desc = { text = EVTDESCAogre10 trigger = { trait = is_fat } }
	desc = { text = EVTDESCBogre10 trigger = { NOT = { trait = is_fat } trait = gluttonous } }
	desc = { text = EVTDESCCogre10 trigger = { NOR = { trait = is_fat trait = gluttonous } } }
	picture = GFX_evt_ogre_feast
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	trigger = {
		trait = creature_ogre
	}

	option = {
		name = EVTOPTAogre10
		trigger = { OR = { trait = is_fat trait = gluttonous } }
		piety = 150
		prestige = 150
		if = {
			limit = { trait = strong }
			piety = 250
			prestige = 250
			change_martial = 3
			health = 0.5
		}
		if = {
			limit = { OR = { trait = tough trait = hardy } }
			remove_trait = tough
			remove_trait = hardy
			add_trait = strong
		}
		if = {
			limit = { OR = { trait = weak trait = soft trait = fragile } }
			remove_trait = weak
			remove_trait = soft
			remove_trait = fragile
			add_trait = tough
		}
		if = {
			limit = {
				NOR = {
					trait = weak
					trait = soft
					trait = fragile
					trait = tough
					trait = hardy
					trait = strong
				}
			}
			add_trait = strong
		}
		if = {
			limit = { trait = gluttonous NOT = { trait = is_fat } }
			remove_trait = is_malnourished
			add_trait = is_fat
		}
	}
	option = {
		name = EVTOPTBogre10
		trigger = { NOR = { trait = is_fat trait = gluttonous } }
		piety = -150
		prestige = -150
		add_trait = is_malnourished
		health = -0.5
	}
}

# ogre.20 A thin chieftain
# Scope: ROOT is a player ogre ruler.
character_event = {
	id = ogre.20
	desc = EVTDESCogre20
	picture = GFX_evt_ogre_raiders

	ai = no
	min_age = 14
	max_age = 100
	capable_only = yes
	prisoner = no
	only_rulers = yes
	religion = great_maw

	trigger = {
		NOT = { trait = is_fat }
		trait = creature_ogre
	}

	mean_time_to_happen = {
		years = 54
	}

	option = {
		name = EVTOPTAogre20
		prestige = -100
		any_vassal = {
			limit = { religion = ROOT trait = creature_ogre }
			opinion = { modifier = opinion_too_thin who = ROOT }
		}
		any_courtier = {
			limit = { religion = ROOT is_ruler = no trait = creature_ogre }
			opinion = { modifier = opinion_too_thin who = ROOT }
		}
	}
}
