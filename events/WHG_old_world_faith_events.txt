###############################################################################
# Old World faith events
# Namespace: sunfest (was CM.4100 - CM.4199 in the vanilla
#   cm_religious_events, never copied into the mod)
# Faction: Old World faiths
# Serves: the faithful of Myrmidia in Estalia, Tilea and beyond
#
# The Feast of the Blazing Sun, the vanilla Zunist Festival of the Sun moved
# to Myrmidia, the Eagle of the South. A Myrmidian ruler holds a month-long
# summer feast (hold_blazing_sun_festival) and invites the vassals. The
# faithful march at noon under the full sun in the Spear Procession
# (sunfest.30), which may harden, inspire, madden or blind them
# (sunfest.31 - 37), and everyone hears the service in Myrmidia's temple
# (sunfest.40). If the Knights of the Blazing Sun are abroad, their Grand
# Master may lend the host a lance of knights, once per lifetime and rarely
# (sunfest.41).
#
# The rest of the vanilla file: the Knights of the Sun and the Haruriyyah
# holy order events are covered by the mod's own holy orders (holyorder in
# WHG_crusade_events) and the generic holy order events; the Byzantine
# iconoclasm event is not carried over.
#
# Invitations                      sunfest.1 - sunfest.3
# The feast begins                 sunfest.10 - sunfest.11
# The feast ends                   sunfest.20 - sunfest.23
# The Spear Procession             sunfest.30 - sunfest.37
# The temple service               sunfest.40 - sunfest.41
#
# Saved: character flags sunfest_begun, attending_sunfest,
#   sunfest_knights_lent (once per lifetime), do_not_disturb (cleared at
#   the end); character modifier holding_blazing_sun_festival; unit earmark
#   sunfest_blazing_sun_lance.
#
# Callers:
#   decisions/WHG_old_world_faith_decisions.txt (hold_blazing_sun_festival:
#     1).
# Loc: localisation/00_WHG_old_world_faith_events.csv
###############################################################################

namespace = sunfest

###############################################################################
# Invitations
###############################################################################

# sunfest.1 The invitations go out
# From hold_blazing_sun_festival.
# Scope: the host.
character_event = {
	id = sunfest.1
	desc = EVTDESCsunfest1
	picture = GFX_evt_sun_temple
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPTAsunfest1
		any_vassal = {
			limit = {
				prisoner = no
				war = no
				is_incapable = no
				age = 16
				NOT = { has_character_flag = do_not_disturb }
			}
			letter_event = { id = sunfest.2 tooltip = EVTTOOLTIPsunfest2 }
		}
		hidden_tooltip = { character_event = { id = sunfest.10 days = 12 } }
	}
}

# sunfest.2 An invitation to the Feast of the Blazing Sun
# From sunfest.1.
# Scope: a vassal; FROM is the host.
letter_event = {
	id = sunfest.2
	desc = EVTDESCsunfest2
	border = GFX_event_letter_frame_religion

	is_triggered_only = yes

	option = { # Travel to the feast
		name = EVTOPTAsunfest2
		ai_chance = {
			factor = 2
			modifier = {
				factor = 10
				has_focus = focus_theology
				religion = myrmidian
			}
			modifier = {
				factor = 0.1
				is_inaccessible_trigger = yes
			}
			modifier = {
				factor = 2
				opinion = { who = FROM value = -30 }
			}
			modifier = {
				factor = 5
				opinion = { who = FROM value = -15 }
			}
			modifier = {
				factor = 5
				opinion = { who = FROM value = 0 }
			}
		}
		set_character_flag = do_not_disturb
		set_character_flag = attending_sunfest
		end_inaccessibility_effect = yes
		hidden_tooltip = {
			character_event = { id = sunfest.11 days = 12 }
			character_event = { id = sunfest.23 days = 42 } # safety clean-up
		}
	}
	option = { # Stay away
		name = EVTOPTBsunfest2
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0
				opinion = { who = FROM value = 25 }
				is_inaccessible_trigger = no
			}
			modifier = {
				factor = 5
				NOT = { religion = myrmidian }
			}
		}
		FROM = {
			letter_event = { id = sunfest.3 days = 3 tooltip = EVTTOOLTIPsunfest3 }
		}
	}
}

# sunfest.10 The feast begins
# From sunfest.1. The host and the adult Myrmidian courtiers walk in the
# Spear Procession; everyone hears the temple service.
# Scope: the host.
character_event = {
	id = sunfest.10
	desc = EVTDESCsunfest10
	picture = GFX_evt_sun_temple
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPTAsunfest10
		set_character_flag = sunfest_begun
		hidden_tooltip = {
			character_event = { id = sunfest.20 days = 30 } # the feast ends
			character_event = { id = sunfest.23 days = 31 } # safety clean-up
			narrative_event = { id = sunfest.30 days = 10 }
			character_event = { id = sunfest.40 days = 20 }
			any_courtier = {
				limit = {
					is_adult = yes
					is_ruler = no
					is_incapable = no
					religion = myrmidian
				}
				set_character_flag = attending_sunfest
				narrative_event = { id = sunfest.30 days = 10 }
				character_event = { id = sunfest.23 days = 31 }
			}
		}
	}
}

# sunfest.20 The feast ends (hidden)
# From sunfest.10.
# Scope: the host.

	id = sunfest.20
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		character_event = { id = sunfest.21 }
		any_vassal = {
			limit = { has_character_flag = attending_sunfest }

		}
	}
}

# sunfest.30 The Spear Procession
# From sunfest.10 and sunfest.11. Rarely, the noon sun leaves its mark.
# Scope: a Myrmidian host, guest or courtier.
narrative_event = {
	id = sunfest.30
	title = EVTTITLEsunfest30
	desc = EVTDESCsunfest30
	picture = GFX_evt_sun_temple
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes

	trigger = {
		is_incapable = no
	}

	option = {
		name = EVTOPTAsunfest30
		piety = 30
		hidden_tooltip = {
			random_list = {
				5 = { character_event = { id = sunfest.31 } }  # sunstroke
				5 = { character_event = { id = sunfest.32 } }  # madness
				5 = { character_event = { id = sunfest.33 } }  # possession
				10 = { character_event = { id = sunfest.34 } } # zeal
				10 = { character_event = { id = sunfest.35 } } # strength
				10 = { character_event = { id = sunfest.36 } } # the goddess's cunning
				5 = { character_event = { id = sunfest.37 } }  # blinded
				300 = { }
			}
		}
	}
}

# sunfest.31 Sunstroke
# Scope: an old or sickly marcher.
character_event = {
	id = sunfest.31
	desc = EVTDESCsunfest31
	picture = GFX_evt_illness
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	trigger = {
		immortal = no
		OR = {
			age = 70
			health_traits = 1
		}
	}

	option = {
		name = EVTOPTAsunfest31
		if = {
			limit = { is_ill = yes }
			add_trait = incapable
		}
		else = {
			add_symptom_effect = yes
		}
	}
}

# sunfest.40 The service in Myrmidia's temple
# From sunfest.10 and sunfest.11. A Myrmidian host whom the Knights of the
# Blazing Sun have never helped may be offered a lance of knights
# (sunfest.41), one time in ten.
# Scope: the host or a guest.
character_event = {
	id = sunfest.40
	desc = EVTDESCsunfest40
	picture = GFX_evt_sun_temple
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPTAsunfest40
		piety = 20
		add_character_modifier = {
			name = large_feast_timer
			duration = 365
			hidden = yes
		}
		if = {
			limit = {
				has_character_flag = sunfest_begun
				religion = myrmidian
				war = no
				NOT = { has_character_flag = sunfest_knights_lent }
				d_knights_of_the_blazing_sun = { has_holder = yes }
			}
			random = {
				chance = 10
				hidden_tooltip = { character_event = { id = sunfest.41 days = 2 } }
			}
		}
	}
}

# sunfest.41 A lance of the Blazing Sun
# Once per lifetime: the Grand Master of the Knights of the Blazing Sun,
# moved by the feast, offers the host a lance of knights.
# Scope: the host.
character_event = {
	id = sunfest.41
	desc = EVTDESCsunfest41
	picture = GFX_evt_knight_kneeling
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	trigger = {
		NOT = { has_character_flag = sunfest_knights_lent }
	}

	immediate = {
		set_character_flag = sunfest_knights_lent
	}

	option = { # Welcome them
		name = EVTOPTAsunfest41
		custom_tooltip = {
			text = EVTOPTAsunfest41_TT
			capital_scope = {
				PREV = {
					spawn_unit = {
						province = PREV
						owner = PREV
						troops = { knights = { 40 40 } heavy_infantry = { 60 60 } }
						earmark = sunfest_blazing_sun_lance
						maintenance_multiplier = 0.5
					}
				}
			}
		}
	}
	option = { # Send them on to the wars
		name = EVTOPTBsunfest41
		piety = 50
	}
}
