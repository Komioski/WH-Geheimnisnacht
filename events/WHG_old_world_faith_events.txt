###############################################################################
# Old World faith events
# Namespace: sunfest (was CM.4100 - CM.4199 in the vanilla
#   cm_religious_events, never copied into the mod)
# Namespace: penance (Part 2, was HFP.23001 - HFP.23026 in
#   HFP_religious_events.txt: penance.N = HFP.230NN minus 23000)
# Faction: Old World faiths
# Serves: the faithful of Myrmidia in Estalia, Tilea and beyond; the faiths
#   whose head can excommunicate (Sigmar, Myrmidia, the Lady, Solkan, Morr)
#
# The Feast of the Blazing Sun, the vanilla Zunist Festival of the Sun moved
# to Myrmidia, the Eagle of the South. A Myrmidian ruler holds a month-long
# summer feast (hold_blazing_sun_festival) and invites the vassals. The
# faithful march at noon under the full sun in the Spear Procession
# (sunfest.30), which may harden, inspire, madden or blind them
# (sunfest.31 - 37), and everyone hears the service in Myrmidia's temple
# (sunfest.40). If the Knights of the Blazing Sun are abroad, their Grand
# Master may lend the host a lance of knights, once per lifetime and rarely
# (sunfest.41).
#
# The rest of the vanilla file: the Knights of the Sun and the Haruriyyah
# holy order events are covered by the mod's own holy orders (holyorder in
# WHG_crusade_events) and the generic holy order events; the Byzantine
# iconoclasm event is not carried over.
#
# Invitations                      sunfest.1 - sunfest.3
# The feast begins                 sunfest.10 - sunfest.11
# The feast ends                   sunfest.20 - sunfest.23
# The Spear Procession             sunfest.30 - sunfest.37
# The temple service               sunfest.40 - sunfest.41
# Part 2: penance and absolution    penance.1 - penance.26
#
# Saved: character flags sunfest_begun, attending_sunfest,
#   sunfest_knights_lent (once per lifetime), do_not_disturb (cleared at
#   the end); character modifier holding_blazing_sun_festival; unit earmark
#   sunfest_blazing_sun_lance.
#
# Callers:
#   decisions/WHG_old_world_faith_decisions.txt (hold_blazing_sun_festival:
#     1).
# Loc: localisation/00_WHG_old_world_faith_events.csv
###############################################################################

namespace = sunfest
namespace = penance

# sunfest.1 The invitations go out
# From hold_blazing_sun_festival.
# Scope: the host.
character_event = {
	id = sunfest.1
	desc = EVTDESCsunfest1
	picture = GFX_evt_sun_temple
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPTAsunfest1
		any_vassal = {
			limit = {
				prisoner = no
				war = no
				is_incapable = no
				age = 16
				NOT = { has_character_flag = do_not_disturb }
			}
			letter_event = { id = sunfest.2 tooltip = EVTTOOLTIPsunfest2 }
		}
		hidden_tooltip = { character_event = { id = sunfest.10 days = 12 } }
	}
}

# sunfest.2 An invitation to the Feast of the Blazing Sun
# From sunfest.1.
# Scope: a vassal; FROM is the host.
letter_event = {
	id = sunfest.2
	desc = EVTDESCsunfest2
	border = GFX_event_letter_frame_religion

	is_triggered_only = yes

	option = { # Travel to the feast
		name = EVTOPTAsunfest2
		ai_chance = {
			factor = 2
			modifier = { factor = 10 has_focus = focus_theology religion = myrmidian }
			modifier = { factor = 0.1 is_inaccessible_trigger = yes }
			modifier = { factor = 2 opinion = { who = FROM value = -30 } }
			modifier = { factor = 5 opinion = { who = FROM value = -15 } }
			modifier = { factor = 5 opinion = { who = FROM value = 0 } }
		}
		set_character_flag = do_not_disturb
		set_character_flag = attending_sunfest
		end_inaccessibility_effect = yes
		hidden_tooltip = {
			character_event = { id = sunfest.11 days = 12 }
			character_event = { id = sunfest.23 days = 42 } # safety clean-up
		}
	}
	option = { # Stay away
		name = EVTOPTBsunfest2
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0
				opinion = { who = FROM value = 25 }
				is_inaccessible_trigger = no
			}
			modifier = { factor = 5 NOT = { religion = myrmidian } }
		}
		FROM = { letter_event = { id = sunfest.3 days = 3 tooltip = EVTTOOLTIPsunfest3 } }
	}
}

# Penance and absolution
# Namespace: penance (was HFP.23001 - HFP.23026 in HFP_religious_events.txt:
#   penance.N = HFP.230NN minus 23000; HFP.23027, the African sympathy for
#   the faiths of the Book, is gone)
# Serves: the faiths whose head can excommunicate (Sigmar, Myrmidia, the
#   Lady, Solkan, Morr); the cloister, pilgrimage and stolen-spouse hooks
#   serve every faith
#
# An excommunicated lord may issue a declaration of repentance (restored:
# the vanilla decision was never copied into the mod, so the whole chain was
# dead). The head of the faith names his price; paying it lifts the
# excommunication, refusing is remembered and makes the same demand more
# likely next time.
#
# The cloistered child                    penance.1
# Excommunication asked                   penance.2
# The price of absolution                 penance.3 - penance.23
# Pilgrims and stolen spouses             penance.24 - penance.26
#
# Callers: decisions/WHG_old_world_faith_decisions.txt
#   (issue_declaration_of_repentance: 3), dynasty_decisions.txt (1),
#   00_on_actions.txt (on_excommunicate_interaction: 2, on_imprisoned: 24,
#   on_war_ended_victory: 25, on_war_ended_defeat: 26).
# Saved: character modifier asked_repentance_from_sins_cooldown (48 months);
#   opinion modifiers opinion_pope_ex_demands_* (the head remembers a
#   refusal for 100 years), opinion_repented_sins.
###############################################################################

# penance.2 Excommunication asked
# Hidden. The faith head remembers who asked for an excommunication, so the
# same lord cannot ask again for six months.
# Scope: ROOT is the excommunicated, FROM the asker, FROMFROM the faith head.
#   Fired from on_excommunicate_interaction.
character_event = {
	id = penance.2
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		FROMFROM = {
			opinion = { modifier = opinion_requested_excommunication who = ROOT_FROM months = 6 }
		}
	}
}

# penance.3 The faith head weighs the penance
# Hidden. The head of the faith picks the price of absolution: gold,
# prestige, a vow of celibacy, a child for the cloister, abdication, the
# right to invest priests, divided inheritance, temple lands, freedom from
# the penitent, or the penitent's fealty. The odds follow his opinion, the
# penitent's sins and rank, and what was refused before.
# Scope: ROOT is the faith head, FROM the penitent. Fired by the decision
#   issue_declaration_of_repentance (WHG_old_world_faith_decisions.txt).
character_event = {
	id = penance.3
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		random_list = {
			# Possible impositions:

			#Tiered Money
			#Tiered Prestige
			#Vow of Celibacy
			#Son/Daughter must take vows.
			#Abdication-> Become Monk/Nun
			#Enforce temple Succession (Kings/Emperors)
			#Change succession to Gavelkind (unless Elective or worse) (King/Emperors)
			#Give temple lands back (Emperors)
			#Give Independence to temple (Emperors)

			30 = {
				#Pay tiered money.
				modifier = { factor = 0.5 vassal_of = FROM }
				modifier = { #Supported faith head more likely to ask for little
					factor = 1.5
					has_opinion_modifier = { who = FROM modifier = opinion_supported_pope }
				}
				modifier = { factor = 0.25 higher_tier_than = DUKE NOT = { wealth = 50 } }
				modifier = { factor = 0.5 higher_tier_than = DUKE NOT = { wealth = 250 } }
				modifier = { factor = 0 NOT = { opinion = { who = FROM value = -100 } } }
				modifier = { factor = 0.2 NOT = { opinion = { who = FROM value = -80 } } }
				modifier = { factor = 0.3 NOT = { opinion = { who = FROM value = -60 } } }
				modifier = { factor = 0.4 NOT = { opinion = { who = FROM value = -40 } } }
				modifier = { factor = 0.5 NOT = { opinion = { who = FROM value = -20 } } }
				modifier = { factor = 0.6 NOT = { opinion = { who = FROM value = 0 } } }
				modifier = { factor = 1.25 opinion = { who = FROM value = 20 } }
				modifier = { factor = 1.5 opinion = { who = FROM value = 40 } }
				modifier = { factor = 1.5 opinion = { who = FROM value = 60 } }
				modifier = { factor = 1.5 opinion = { who = FROM value = 80 } }
				modifier = { factor = 1.5 opinion = { who = FROM value = 100 } }
				modifier = { factor = 1.5 trait = greedy }
				modifier = { factor = 0.5 trait = zealous }
				FROM = { letter_event = { id = penance.4 } } #Asks for Gold.
			}
			30 = {
				#Pay tiered Prestige.
				trigger = { FROM = { higher_tier_than = DUKE } }
				modifier = { factor = 0.5 vassal_of = FROM }
				modifier = { factor = 1.5 trait = proud }
				modifier = { factor = 0.5 trait = cynical }
				FROM = { letter_event = { id = penance.6 } } #Asks for Prestige.
			}

			20 = {
				#Impose celibacy
				trigger = {
					FROM = { NOR = { trait = celibate trait = eunuch trait = chaste } }
					FROM = { num_of_children = 1 }
				}
				modifier = { factor = 0.5 vassal_of = FROM }
				modifier = { factor = 1.5 trait = chaste }
				modifier = { factor = 1.5 trait = celibate }
				modifier = { factor = 0.5 trait = lustful }
				modifier = { factor = 0.5 FROM = { trait = humble } }
