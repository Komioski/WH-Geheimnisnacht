###############################################################################
# The Palio: horse races of the Tilean cities
# Namespace: palio (was LoR.400 - LoR.504 in the vanilla hippodrome_events.txt,
#   the Byzantine hippodrome)
# Faction: Tilea, Estalia & Sartosa
# Serves: player rulers whose capital lies in Tilea
#
# Part 1: Race day                                 palio.1 - palio.3
#   (the quarters of the city race through the piazza; the ruler may back
#   the Eagle or the Tower)
# Part 2: The fixed Palio                          palio.10 - palio.27
#   (once a game a schemer asks a Tilean king to fix the race: a cut girth,
#   a poisoned horse, then a blade in the crowd)
#
# Callers: palio.1 on the yearly list palio_races, palio.10 on the five-year
#   list palio_schemes (wh_on_actions).
# Scripted triggers: common/scripted_triggers/WHG_palio_triggers.txt
# Localisation: localisation/00_WHG_palio_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = palio

###############################################################################
# Part 1: Race day
###############################################################################

# palio.1 Race day
# Yearly list palio_races, weight 5 (was MTTH 230 months): the city's
# quarters race, and the ruler may back one.
# Scope: ROOT is a player ruler with a Tilean capital.
character_event = {
	id = palio.1
	desc = EVTDESCpalio1
	picture = GFX_evt_hippodrome_byzantine

	is_triggered_only = yes # on_yearly_pulse, list palio_races

	trigger = {
		palio_host_trigger = yes
		NOT = { has_global_flag = palio_scheme_ongoing }
	}

	option = { # Back the Eagle
		name = EVTOPTApalio1
		wealth = -25
		random_list = {
			50 = { character_event = { id = palio.2 days = 8 tooltip = EVTTOOLTIPpalio2 } }
			50 = { character_event = { id = palio.3 days = 8 tooltip = EVTTOOLTIPpalio3 } }
		}
	}
	option = { # Back the Tower
		name = EVTOPTBpalio1
		wealth = -25
		random_list = {
			50 = { character_event = { id = palio.2 days = 8 tooltip = EVTTOOLTIPpalio2 } }
			50 = { character_event = { id = palio.3 days = 8 tooltip = EVTTOOLTIPpalio3 } }
		}
	}
	option = { # Just watch
		name = EVTOPTCpalio1
		custom_tooltip = { text = palio_nobet }
	}
}

###############################################################################
# Part 2: The fixed Palio
# The ruler rides in the Palio. A schemer offers to fix it three times: a cut
# girth (palio.10 - 11), a poisoned horse (palio.12 - 13), then a blade in the
# crowd (palio.20 - 27). Each fix wins (palio.15), loses (palio.16) or starts
# a riot (palio.17, 25). The flag palio_poison_offered marks the second fix.
###############################################################################

# palio.10 The schemer's offer: the girth
# Five-year list palio_schemes, weight 8 (was MTTH 780 months); once a game.
# Scope: ROOT is a player king or higher with a Tilean capital.
narrative_event = {
	id = palio.10
	title = EVTNAMEpalio10
	desc = EVTDESCpalio10
	picture = GFX_evt_emissary_byzantine

	is_triggered_only = yes # on_five_year_pulse, list palio_schemes

	trigger = {
		palio_schemer_trigger = yes
		NOT = { has_global_flag = palio_scheme }
	}

	option = { # Agree
		name = EVTOPTApalio10
		set_global_flag = palio_scheme
		set_global_flag = palio_scheme_ongoing
		narrative_event = { id = palio.11 days = 32 tooltip = EVTTOOLTIPpalio11 }
	}
	option = { # Refuse
		name = EVTOPTBpalio10
		set_global_flag = palio_scheme
		custom_tooltip = { text = palio_refuse }
	}
}

# palio.11 The girth is cut
# Scope: ROOT is the ruler.
narrative_event = {
	id = palio.11
	title = EVTNAMEpalio11
	desc = EVTDESCpalio11
	picture = GFX_evt_shadowy_cabal
	hide_from = yes

	is_triggered_only = yes # palio.10

	option = {
		name = EVTOPTApalio11
		change_martial = 1
		custom_tooltip = {
			text = palio_race_tooltip
			hidden_tooltip = {
				random_list = {
					40 = { narrative_event = { id = palio.15 } }
					40 = { narrative_event = { id = palio.16 } }
					20 = { narrative_event = { id = palio.17 } }
				}
			}
		}
	}
}

		name = EVTOPTApalio25

			text = palio_riots

						random_traits = yes

							home = PREV

						}

				}
			}
		}
	}
