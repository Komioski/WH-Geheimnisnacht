###############################################################################
# Patron gods
# Namespace: patron (was TOG.7100 - TOG.7528 in
#   oldgods_ancestor_worship_events.txt)
# Faction: Old World faiths
# Serves: the dawi and Norse dawi (Grungni, Grimnir, Valaya, Thungni), the
#   Ungol faith of the northern gods (Ursun, Dazh, Kalita, Tor) and the
#   living Nehekharans (Ptra, Asaph, Phakth, Usirian)
#
# A lord picks one god of the pantheon as patron (the pick_main_deity
# decision), and the believers of the same faith at court pick theirs. Every
# few years the patron shows the lord a sign: a better hand at the god's
# skill, a fellow worshipper as a friend, the god's own trial (Tor's storms,
# Asaph's passion, the hearth blessings of Valaya, Ptra and Ursun,
# Usirian's questions about death, and new signs of Grimnir, Grungni,
# Thungni, Dazh, Kalita and Phakth), renown in the god's field, or once in a
# life a monument raised to the god or to the house.
#
# Picking a patron                     patron.1 - patron.11
# Skill and friendship                 patron.20 - patron.22
# The god's trial                      patron.23 - patron.31, patron.37
# Renown                               patron.32 - patron.36
# Monument                             patron.40 - patron.43
#
# Fired from: the pick_main_deity decision in minor_decisions.txt
#   (patron.10, which starts the sign loop patron.1 for a playable lord);
#   on_character_convert_religion in 00_on_actions.txt (patron.2). The Baltic
#   pantheon picker TOG.7400 - TOG.7404 and the Finnish hunting gods' TOG.7517
#   had no caller and are dropped.
# Saved: event targets patron_friend, patron_thunder_target, patron_lover,
#   patron_death_child, patron_plotter, patron_plot_target,
#   patron_monument_raiser; flags patron_signs_running, scared_of_thunder,
#   asked_about_death, the *_tale flags, pagan_monument, monument_for_deity,
#   monument_for_ancestors.
###############################################################################

namespace = patron

# patron.1 The patron's sign
# Every three or four years the patron shows a playable follower a sign.
# Ends when the follower no longer has a patron.
character_event = {
	id = patron.1

	is_triggered_only = yes
	hide_window = yes

	immediate = {
		if = {
			limit = { has_minor_deity_trigger = no }
			clr_character_flag = patron_signs_running
		}
		else = {
			random_list = {
				20 = { # A better hand at the god's skill
					modifier = { factor = 0 patron_skill_sign_trigger = no }
					character_event = { id = patron.20 }
				}
				15 = { # A fellow worshipper
					random_courtier_or_vassal = {
						limit = {
							ai = yes
							NOR = {
								is_friend = ROOT
								is_lover = ROOT
								is_rival = ROOT
							}
							patron_shares_root_god_trigger = yes
						}
						character_event = { id = patron.21 }
					}
				}
				15 = { # The god's trial
					patron_trial_effect = yes
				}
				20 = { # Renown in the god's field
					patron_renown_effect = yes
				}
				50 = { # A monument, once in a life
					modifier = {
						factor = 0
						has_character_flag = pagan_monument
					}
					job_treasurer = { character_event = { id = patron.40 } }
				}
				30 = { }
			}
			character_event = { id = patron.1 days = 1000 random = 400 }
		}
	}
}

# patron.2 A convert loses the patron of the old faith (but a dawi turning
# from the ancestor gods to the Norse ancestor gods or back keeps it).
character_event = {
	id = patron.2

	is_triggered_only = yes
	hide_window = yes

	trigger = {
		has_minor_deity_trigger = yes
		NOR = {
			AND = {
				religion = ancestor_gods
				FROM = { religion = norse_ancestor_gods }
			}
			AND = {
				religion = norse_ancestor_gods
				FROM = { religion = ancestor_gods }
			}
		}
	}

	immediate = {
		patron_remove_god_effect = yes
	}
}

	id = patron.24

	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	option = { # Run out into the storm
		name = EVTOPTApatron24
		custom_tooltip = { text = EVTOPTApatron24_tt }
		piety = 50
		random_list = {
			60 = { }
			25 = {
				if = {
					limit = { trait = brave }
					prestige = 100
				}
				else = {
					add_trait_brave_effect = yes
				}
				FROM = {
					if = {
						limit = {
							has_dlc = "Zeus"
							NOT = { age = 16 }
						}
						add_trait = willful
					}
					else = {
						remove_trait_craven_effect = yes
						add_trait_brave_effect = yes
					}
				}
			}
			15 = {

# patron.43 Once a year: the monument falls when its raiser no longer holds
# the province.
province_event = {
	id = patron.43

	is_triggered_only = yes
	hide_window = yes

	trigger = {
		OR = {
			has_province_modifier = holy_monument
			has_province_modifier = ancestral_monument
		}
	}

	immediate = {
		if = {
			limit = { NOT = { owner = { character = event_target:patron_monument_raiser } } }
			remove_province_modifier = holy_monument
			remove_province_modifier = ancestral_monument
		}
		else = {
			province_event = { id = patron.43 days = 365 }
		}
	}
}
