###############################################################################
# Patron gods
# Namespace: patron (was TOG.7100 - TOG.7528 in
#   oldgods_ancestor_worship_events.txt)
# Faction: Old World faiths
# Serves: the dawi and Norse dawi (Grungni, Grimnir, Valaya, Thungni), the
#   Ungol faith of the northern gods (Ursun, Dazh, Kalita, Tor) and the
#   living Nehekharans (Ptra, Asaph, Phakth, Usirian)
#
# A lord picks one god of the pantheon as patron (the pick_main_deity
# decision), and the believers of the same faith at court pick theirs. Every
# few years the patron shows the lord a sign: a better hand at the god's
# skill, a fellow worshipper as a friend, the god's own trial (Tor's storms,
# Asaph's passion, the hearth blessings of Valaya, Ptra and Ursun,
# Usirian's questions about death, and new signs of Grimnir, Grungni,
# Thungni, Dazh, Kalita and Phakth), renown in the god's field, or once in a
# life a monument raised to the god or to the house.
#
# Picking a patron                     patron.1 - patron.11
# Skill and friendship                 patron.20 - patron.22
# The god's trial                      patron.23 - patron.31, patron.37
# Renown                               patron.32 - patron.36
# Monument                             patron.40 - patron.43
#
# Fired from: the pick_main_deity decision in minor_decisions.txt
#   (patron.10, which starts the sign loop patron.1 for a playable lord);
#   on_character_convert_religion in 00_on_actions.txt (patron.2). The Baltic
#   pantheon picker TOG.7400 - TOG.7404 and the Finnish hunting gods' TOG.7517
#   had no caller and are dropped.
# Saved: event targets patron_friend, patron_thunder_target, patron_lover,
#   patron_death_child, patron_plotter, patron_plot_target,
#   patron_monument_raiser; flags patron_signs_running, scared_of_thunder,
#   asked_about_death, the *_tale flags, pagan_monument, monument_for_deity,
#   monument_for_ancestors.
###############################################################################

namespace = patron

# patron.1 The patron's sign
# Every three or four years the patron shows a playable follower a sign.
# Ends when the follower no longer has a patron.
character_event = {
	id = patron.1

	is_triggered_only = yes
	hide_window = yes

	immediate = {
		if = { limit = { has_minor_deity_trigger = no } clr_character_flag = patron_signs_running }
		else = {
			random_list = {
				20 = { # A better hand at the god's skill
					modifier = { factor = 0 patron_skill_sign_trigger = no }
					character_event = { id = patron.20 }
				}
				15 = { # A fellow worshipper
					random_courtier_or_vassal = {
						limit = {
							ai = yes
							NOR = { is_friend = ROOT is_lover = ROOT is_rival = ROOT }
							patron_shares_root_god_trigger = yes
						}
						character_event = { id = patron.21 }
					}
				}
				15 = { # The god's trial
					patron_trial_effect = yes
				}
				20 = { # Renown in the god's field
					patron_renown_effect = yes
				}
				50 = { # A monument, once in a life
					modifier = { factor = 0 has_character_flag = pagan_monument }
					job_treasurer = { character_event = { id = patron.40 } }
				}
				30 = { }
			}
			character_event = { id = patron.1 days = 1000 random = 400 }
		}
	}
}

# patron.2 A convert loses the patron of the old faith (but a dawi turning
# from the ancestor gods to the Norse ancestor gods or back keeps it).
character_event = {
	id = patron.2

	is_triggered_only = yes
	hide_window = yes

	trigger = {
		has_minor_deity_trigger = yes
		NOR = {
			AND = { religion = ancestor_gods FROM = { religion = norse_ancestor_gods } }
			AND = { religion = norse_ancestor_gods FROM = { religion = ancestor_gods } }
		}
	}

	immediate = {
		patron_remove_god_effect = yes
	}
}

# patron.24 A child, spouse or lover cowers from Tor's thunder.
character_event = {
	id = patron.24
	desc = EVTDESCpatron24
	picture = GFX_evt_comet
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes

	option = { # Run out into the storm
		name = EVTOPTApatron24
		custom_tooltip = { text = EVTOPTApatron24_tt }
		piety = 50
		random_list = {
			60 = { }
			25 = {
				if = { limit = { trait = brave } prestige = 100 }
				else = { add_trait_brave_effect = yes }
				FROM = {
					if = { limit = { has_dlc = "Zeus" NOT = { age = 16 } } add_trait = willful }
					else = { remove_trait_craven_effect = yes add_trait_brave_effect = yes }
				}
			}
			15 = { add_character_modifier = { name = struck_by_lightning duration = 120 } }
		}
	}
	option = { # Tell a story
		name = EVTOPTBpatron24
		custom_tooltip = { text = EVTOPTBpatron24_tt }
		reverse_opinion = { who = FROM modifier = comforted_opinion years = 5 }
	}
	option = { # Tough love
		name = EVTOPTCpatron24
		custom_tooltip = { text = EVTOPTCpatron24_tt }
		reverse_opinion = { who = FROM modifier = tough_love_opinion years = 5 }
		random = { chance = 25 add_trait_wroth_effect = yes }
		FROM = {
			random_list = {
				50 = {
					if = { limit = { has_dlc = "Zeus" NOT = { age = 16 } } add_trait = willful }
					else = { remove_trait_craven_effect = yes add_trait_brave_effect = yes }
				}
				50 = {
					if = {
						limit = { has_dlc = "Zeus" NOT = { age = 16 } }
						if = { limit = { NOT = { trait = timid } } add_trait = timid }
					}
					else = { add_trait_craven_effect = yes }
				}
			}
		}
	}
}

# Part 2: The patron gods of the Old World
# Namespace: patron_god (was patron_god.1 - 10 in old_world_patron_events)
# Faction: Old World faiths
# Serves: the followers of Sigmar, Ulric, Taal, Myrmidia, Shallya, Manann
#   and Morr; Verena and Ranald, who have no faith of their own on the map,
#   can be picked by any of them
#
# A lord of the seven faiths picks a patron with the choose_patron_god
# decision. Players turn three pages of gods; the AI picks at once,
# weighing its faith, father, spouse, traits and life focus, or on a whim
# led by its focus alone. Every four to six years the patron's favour
# shows: Sigmar's renown, Ulric's wolves, Taal's harvest, Myrmidia's
# eagle, Shallya's mercy, Manann's calm seas, Morr's dream, Verena's owl
# and Ranald's luck. A convert to a faith outside the Old World loses the
# patron.
#
# Picking a patron             patron_god.1 - patron_god.3 (players),
#                              patron_god.10 (AI)
# The patron's favour          patron_god.20, patron_god.22
# A convert loses the patron   patron_god.21
#
# Saved: character flags selecting_patron_deity (set by the decision),
#   patron_god_favour_running; character modifiers patron_god_sigmar,
#   _ulric, _taal, _myrmidia, _shallya, _manann, _morr, _verena, _ranald.
# Callers: decisions/minor_decisions.txt (choose_patron_god: 1 for players,
#   10 for the AI); common/on_actions/00_on_actions.txt
#   (on_character_convert_religion: 21).
# Support: common/scripted_effects/WHG_old_gods_effects.txt
#   (patron_god_take_<god>_effect, patron_god_start_favour_effect),
#   common/scripted_triggers/WHG_old_gods_triggers.txt
#   (has_old_world_patron_trigger). Loc: 00_WHG_patron_god_events.csv;
#   the modifiers and chronicle lines stay in 00_patron_gods.csv.
#
# Fixed from the old file: a convert to Chaos or any faith outside the Old
# World kept the patron for life (remove_patrons_effect had no caller);
# the patrons gave nothing after the pick; the player pages carried AI
# weights the AI never saw (it skips the pages).

# patron_god.20 The favour loop (hidden)
# Every four to six years a playable follower with an Old World patron
# feels the patron's favour (22), unless captive or incapable that time.
# Ends when the follower no longer has a patron.
character_event = {
	id = patron_god.20
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_old_world_patron_trigger = no }
			clr_character_flag = patron_god_favour_running
		}
		else = {
			if = {
				limit = { is_playable = yes prisoner = no NOT = { trait = incapable } }
				character_event = { id = patron_god.22 }
			}
			character_event = { id = patron_god.20 days = 1460 random = 730 }
		}
	}
}
