namespace = coldheart

###############################################################################
# Part 1: the cold heart (namespace coldheart)
# Faction: Generic
# Serves: the Old World faiths
#
# A youth of the Old World faiths, raised by a cruel, drunken or maddened
# parent, finds one day that the pain of others means nothing to them. It
# was an MTTH event of eight years; it now rolls on the bi-yearly pulse
# between 14 and 18, so a youth gets two or three rolls. The followers of
# Shallya and the Lady are spared.
#
#   coldheart.1   A servant begs leave to visit his dying mother
#
# Fired from: on_bi_yearly_pulse coldheart_youth list (wh_on_actions.txt).
# Loc: localisation/00_WHG_personality_events.csv
###############################################################################

# coldheart.1 A youth with a cruel upbringing refuses a servant's plea and
# feels nothing. Take the psychopath trait, or fight the coldness and maybe
# win (a lost fight also leaves them stressed).
character_event = {
	id = coldheart.1
	desc = EVTDESCcoldheart1
	picture = GFX_evt_shadow

	is_triggered_only = yes # on_bi_yearly_pulse coldheart_youth list (wh_on_actions.txt)

	min_age = 14
	max_age = 18
	religion_group = old_world_group

	trigger = {
		NOR = {
			religion = grail
			religion = shallya
			trait = kind
			trait = just
			trait = psychopath
			trait = temperate
			trait = patient
		}
		OR = {
			father = {
				NOR = { trait = kind trait = temperate trait = patient trait = just trait = humble }
				calc_true_if = {
					amount = 4
					trait = cruel
					trait = wroth
					trait = psychopath
					trait = envious
					trait = impaler
					trait = lunatic_1
					trait = lunatic_2
					trait = lunatic_3
					trait = possessed
					trait = drunkard
					trait = stressed
					trait = depressed
					trait = greedy
					trait = deceitful
					trait = kinslayer
					trait = gluttonous
				}
			}
			mother = {
				NOR = { trait = kind trait = temperate trait = patient trait = just trait = humble }
				calc_true_if = {
					amount = 5
					trait = cruel
					trait = wroth
					trait = psychopath
					trait = envious
					trait = impaler
					trait = lunatic_1
					trait = lunatic_2
					trait = lunatic_3
					trait = possessed
					trait = drunkard
					trait = stressed
					trait = depressed
					trait = greedy
					trait = deceitful
					trait = kinslayer
					trait = gluttonous
					trait = lustful
				}
			}
		}
		calc_true_if = {
			amount = 2
			NOT = { trait = honest }
			NOT = { trait = just }
			NOT = { trait = charitable }
			NOT = { trait = humble }
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 3 trait = cruel }
		modifier = { factor = 3 trait = wroth }
		modifier = { factor = 2 trait = rowdy }
		modifier = { factor = 2 trait = haughty }
	}

	option = { # Their pain is nothing to me
		name = EVTOPTAcoldheart1
		add_trait = psychopath
		ai_chance = { factor = 70 }
	}
	option = { # Fight the coldness
		name = EVTOPTBcoldheart1
		random_list = {
			50 = {
				add_trait = psychopath
				if = { limit = { NOT = { trait = stressed } } add_trait = stressed }
			}
			50 = { custom_tooltip = { text = TTcoldheart1_spared } }
		}
		ai_chance = { factor = 30 }
	}
}

###############################################################################
# Part 2: the shaping of a character (namespace persona)
# Namespace: persona (was 1000 - 1071 in personality_traits_events.txt:
#   persona.N = old id minus 999)
# Faction: Generic
#
# The vanilla adult personality events. Every five years a grown character
# may face a small choice (a feast, a beggar, a poacher, an envoy's
# question) that tips them towards one trait or its opposite, with odds set
# by the traits they already have. A character with six personality traits
# may lose one instead. The faithful of Ormazd get their own lust event,
# and may vow to learn the Scriptures of the Prophets by heart (hafiz).
#
#   persona.1 - 20      Gaining a trait (persona.3, the second gluttony
#                       event, was a copy of persona.2 and is dropped)
#   persona.29 - 59     Losing a trait
#   persona.61          Araby: the lust event
#   persona.71 - 72     Araby: the hafiz vow and its keeping (72 was MTTH,
#                       now a year or two after the vow)
#
# Fired from: the on_five_year_pulse random list (00_on_actions.txt), with
#   the other trait events; persona.32 also from WHG_health_events (the
#   physician's warning to a glutton); persona.72 from persona.71.
# Notifications: 38251 - 38277 in trait_notification.txt.
# Saved: character flags aspiring_hafiz_flag, had_event_1013; character
#   modifiers aspiring_hafiz, well_fed.
# Loc: localisation/00_WHG_personality_events.csv; persona.71 - 72 keep the
#   mod's Araby keys EVTDESC1070 - 1071 (00_vanilla_replace_soi.csv and
#   the other replacement files).
###############################################################################

namespace = persona

# persona.1 Gaining Lustful or Chaste - male: a choice shapes the character, with odds set by their other traits.
character_event = {
	id = persona.1
	desc = EVTDESCpersona1
	picture = "GFX_evt_lovers"

	min_age = 16
	only_men = yes

	is_triggered_only = yes # on_five_year_pulse random list (00_on_actions.txt)

	trigger = {
		NOT = { personality_traits = 5 }
		NOT = { religion = ormazd }
		NOT = { religion = kitabid }
		NOT = { trait = chaste }
		# sexless
		NOR = { culture_group = orc_group culture_group = goblin_group }
		NOT = { trait = lustful }
		NOT = { trait = celibate }
		NOT = { trait = homosexual }
	}

	weight_multiplier = {
		days = 1

		modifier = { factor = 1.7 trait = greedy }
		modifier = { factor = 1.7 trait = hedonist }
		modifier = { factor = 1.7 trait = gluttonous }
		modifier = { factor = 0.5 trait = temperate }
	}

	option = {
		name = EVTOPTApersona1
		if = {
			limit = { trait = honest NOT = { trait = slothful } }
			random = {
				chance = 70

				add_trait = lustful
				hidden_tooltip = {
					character_event = {
						id = traitnote.272 #Notify Lustful
					}
				}
			}
		}
		if = {
			limit = { trait = slothful NOT = { trait = honest } }
			random = {
				chance = 20

				add_trait = lustful
				hidden_tooltip = {
					character_event = {
						id = traitnote.272 #Notify Lustful
					}
				}
			}
		}
		if = {
			limit = { NOT = { trait = honest } NOT = { trait = slothful } }
			random = {
				chance = 45

				add_trait = lustful
				hidden_tooltip = {
					character_event = {
						id = traitnote.272 #Notify Lustful
					}
				}
			}
		}
		if = {
			limit = { trait = honest trait = slothful }
			random = {
				chance = 45

				add_trait = lustful
				hidden_tooltip = {
					character_event = {
						id = traitnote.272 #Notify Lustful
					}
				}
			}
		}
	}
	option = {
		name = EVTOPTBpersona1
		if = {
			limit = { trait = arbitrary NOT = { trait = proud } }
			random = {
				chance = 70

				add_trait = chaste
				hidden_tooltip = {
					character_event = {
						id = traitnote.274 #Notify Chaste
					}
				}
			}
		}
		if = {
			limit = { trait = proud NOT = { trait = arbitrary } }
			random = {
				chance = 20

				add_trait = chaste
				hidden_tooltip = {
					character_event = {
						id = traitnote.274 #Notify Chaste
					}
				}
			}
		}
		if = {
			limit = { NOT = { trait = proud } NOT = { trait = arbitrary } }
			random = {
				chance = 45

				add_trait = chaste
				hidden_tooltip = {
					character_event = {
						id = traitnote.274 #Notify Chaste
					}
				}
			}
		}
		if = {
			limit = { trait = proud trait = arbitrary }
			random = {
				chance = 45

				add_trait = chaste
				hidden_tooltip = {
					character_event = {
