###############################################################################
# Pets: the court cat, the owl and the hound
# Namespace: pet (was 77800 - 77862 in pet_events.txt)
# Faction: Generic
#
# A ruler's pet cat, owl or hunting dog lives, makes trouble and dies.
# Cats come from add_cat_effect (00_scripted_effects.txt) and from offmap and
# bloodline gifts; owls from the owl effect and gbc_events; hounds from the
# hunting events (WoL.5290 is the hound's death).
#
# Warhammer rules: an owl is Verena's own bird; in the Old World a witch
# hunter or a zealot at court may name the cat a witch's familiar (pet.30),
# and a wizard can answer that in kind; a cruel ruler of Araby, the
# Southlands or Lustria, or an Amazon, feeds a cat-hater to the great cats
# instead of impaling them.
#
# Death and funerals                  pet.1 - pet.4
# Cat life (yearly pulse tombola)     pet.10 - pet.21
# The witch's familiar                pet.30
#
# Fired from: on_yearly_pulse pet list in 00_on_actions.txt (pet.10, pet.19);
#   add_cat_effect and the owl effect in 00_scripted_effects.txt,
#   wh_offmap_effects.txt, 06_scripted_effects_legendary_bloodlines.txt and
#   gbc_events.txt (pet.1, pet.3, pet.20).
###############################################################################

namespace = pet

# pet.10 Cat tombola
# Picks one of the cat events for a ruler who owns a cat.
# Scope: ROOT is the owner.
character_event = {
	id = pet.10
	hide_window = yes
	is_triggered_only = yes # on_yearly_pulse, pet events list (00_on_actions.txt)

	only_playable = yes
	only_capable = yes
	prisoner = no

	trigger = {
		has_character_modifier = pet_cat
	}

	immediate = {
		random_list = {
			10 = {
				trigger = {
					OR = {
						any_spouse = { pet_cat_hater_trigger = yes }
						any_lover = { pet_cat_hater_trigger = yes }
						any_friend = { pet_cat_hater_trigger = yes }
					}
				}
				character_event = { id = pet.11 }
			}
			10 = { character_event = { id = pet.16 } }
			10 = {
				trigger = { any_courtier = { pet_rabbit_parent_trigger = yes } }
				character_event = { id = pet.17 }
			}
			10 = {
				trigger = {
					NOR = {
						has_character_modifier = crazy_cat_lady
						has_character_modifier = crazy_cat_lord
					}
					pet_cat_collector_trigger = yes
				}
				character_event = { id = pet.18 }
			}
			10 = {
				trigger = { pet_familiar_accuser_trigger = yes }
				character_event = { id = pet.30 }
			}
		}
	}
}

# pet.11 Someone hates the cat
# A spouse, lover or friend demands the cat goes. Give it away, keep it
# (a lover or friend then issues an ultimatum, pet.12), or kill the
# complainer: impaled, or fed to the great cats in Araby, the Southlands
# and Lustria, or by an Amazon.
# Scope: ROOT is the owner. event_target:cat_hater is the complainer.
character_event = {
	id = pet.11
	desc = EVTDESCpet11
	picture = GFX_evt_cat
	border = GFX_event_normal_frame_intrigue
	is_triggered_only = yes # pet.10

	trigger = {
		OR = {
			any_spouse = { pet_cat_hater_trigger = yes }
			any_lover = { pet_cat_hater_trigger = yes }
			any_friend = { pet_cat_hater_trigger = yes }
		}
	}

	immediate = {
		if = {
			limit = { any_spouse = { pet_cat_hater_trigger = yes } }
			random_spouse = {
				limit = { pet_cat_hater_trigger = yes }
				save_event_target_as = cat_hater
			}
		}
		else_if = {
			limit = { any_friend = { pet_cat_hater_trigger = yes } }
			random_friend = {
				limit = { pet_cat_hater_trigger = yes }
				save_event_target_as = cat_hater
			}
		}
		else = {
			random_lover = {
				limit = { pet_cat_hater_trigger = yes }
				save_event_target_as = cat_hater
			}
		}
	}

	option = { # Get rid of the cat
		name = EVTOPTApet11
		remove_character_modifier = pet_cat
		reverse_opinion = {
			who = event_target:cat_hater
			modifier = opinion_fulfilled_demands
			years = 10
		}
	}
	option = { # Give the cat to a friend
		name = EVTOPTBpet11
		trigger = { any_friend = { pet_new_cat_owner_trigger = yes } }
		remove_character_modifier = pet_cat
		reverse_opinion = {
			who = event_target:cat_hater
			modifier = opinion_fulfilled_demands
			years = 10
		}
		random_friend = { limit = { pet_new_cat_owner_trigger = yes } pet_give_cat_effect = yes }
	}
	option = { # Give the cat to a courtier
		name = EVTOPTBpet11
		trigger = {
			NOT = { any_friend = { pet_new_cat_owner_trigger = yes } }
			any_courtier = { pet_new_cat_owner_trigger = yes NOT = { is_rival = ROOT } }
		}
		remove_character_modifier = pet_cat
		reverse_opinion = {
			who = event_target:cat_hater
			modifier = opinion_fulfilled_demands
			years = 10
		}
		random_courtier = {
			limit = { pet_new_cat_owner_trigger = yes NOT = { is_rival = ROOT } }
			pet_give_cat_effect = yes
		}
	}
	option = { # Keep the cat
		name = EVTOPTCpet11
		reverse_opinion = {
			who = event_target:cat_hater
			modifier = opinion_ignored_demands
			years = 10
		}
		if = {
			limit = {
				OR = { is_friend = event_target:cat_hater is_lover = event_target:cat_hater }
			}
			hidden_effect = { character_event = { id = pet.12 days = 45 random = 30 } }
		}
	}
	option = { # Impale them
		name = EVTOPTDpet11
		trigger = {
			event_target:cat_hater = { liege = { character = ROOT } }
			OR = {
				trait = impaler
				AND = { pet_great_cat_keeper_trigger = no pet_cruel_cat_lover_trigger = yes }
			}
		}
		custom_tooltip = { text = EVTTOOLTIPpet11_kill }
		event_target:cat_hater = {
			death = { death_reason = death_execution_impaling killer = ROOT }
		}
	}
	option = { # Feed them to the great cats
		name = EVTOPTEpet11
		trigger = {
			event_target:cat_hater = { liege = { character = ROOT } }
			pet_great_cat_keeper_trigger = yes
			pet_cruel_cat_lover_trigger = yes
		}
		custom_tooltip = { text = EVTTOOLTIPpet11_kill }
		if = {
			limit = { capital_scope = { region = world_new_world_lustria } }
			event_target:cat_hater = {
				death = { death_reason = death_execution_tiger killer = ROOT }
			}
		}
		else = {
			event_target:cat_hater = {
				death = { death_reason = death_execution_lion killer = ROOT }
			}
		}
	}
}

# pet.12 The ultimatum
# The lover or friend who hates the cat makes the owner choose.
# Scope: ROOT is the owner. event_target:cat_hater is the complainer.
character_event = {
	id = pet.12
	desc = EVTDESCpet12
	picture = GFX_evt_cat
	border = GFX_event_normal_frame_intrigue
	is_triggered_only = yes # pet.11

	trigger = {
		has_character_modifier = pet_cat
		OR = { is_lover = event_target:cat_hater is_friend = event_target:cat_hater }
	}

	option = { # Choose them
		name = EVTOPTApet12
		custom_tooltip = { text = EVTTOOLTIPpet12_a }
		reverse_remove_opinion = { who = event_target:cat_hater modifier = opinion_ignored_demands }
		remove_character_modifier = pet_cat
	}
	option = { # Choose the cat
		name = EVTOPTBpet12
		custom_tooltip = { text = EVTTOOLTIPpet12_b }
		if = {
			limit = { is_friend = event_target:cat_hater }
			remove_friend = event_target:cat_hater
		}
		if = { limit = { is_lover = event_target:cat_hater } remove_lover = event_target:cat_hater }
	}
}

# pet.13 How is the new owner doing?

# befriends the old one (pet.14) and a cruel one gets rid of the pet (pet.15).

	hide_window = yes

					trait = celibate
				}

			FROM = { character_event = { id = pet.14 } }
		}
		else_if = {
			limit = {
				OR = {
					trait = cruel

				}
			}
			FROM = { character_event = { id = pet.15 } }
		}
	}
}

### The witch's familiar ###

# pet.30 The witch's familiar
# In the Old World a witch hunter or a zealot at court names the ruler's cat
# a witch's familiar and wants it burned. Burn it, refuse, answer as a wizard,
# or burn the accuser instead. Once per life.
# Scope: ROOT is the owner. event_target:pet_accuser is the accuser.
character_event = {
	id = pet.30
	desc = EVTDESCpet30
	picture = GFX_evt_witch_hunter_pyre
	border = GFX_event_normal_frame_religion
	is_triggered_only = yes # pet.10

	trigger = {
		pet_familiar_accuser_trigger = yes
	}

	immediate = {
		set_character_flag = pet_familiar_accused
		if = {
			limit = {
				any_courtier = {
					pet_familiar_accuser_candidate_trigger = yes
					is_witch_hunter_trigger = yes
				}
			}
			random_courtier = {
				limit = {
					pet_familiar_accuser_candidate_trigger = yes
					is_witch_hunter_trigger = yes
				}
				save_event_target_as = pet_accuser
			}
		}
		else = {
			random_courtier = {
				limit = { pet_familiar_accuser_candidate_trigger = yes }
				save_event_target_as = pet_accuser
			}
		}
	}

	option = { # Burn it
		name = EVTOPTApet30
		custom_tooltip = { text = EVTTOOLTIPpet30_a }
		remove_character_modifier = pet_cat
		piety = 15
		reverse_opinion = {
			who = event_target:pet_accuser
			modifier = opinion_fulfilled_demands
			years = 5
		}
		if = { limit = { trait = kind } random = { chance = 50 add_trait_depressed_effect = yes } }
		ai_chance = {
			factor = 30
			modifier = { factor = 3 trait = zealous }
			modifier = { factor = 0.2 trait = kind }
			modifier = { factor = 0 has_magic_power_trigger = yes }
		}
	}
	option = { # It is a cat
		name = EVTOPTBpet30
		piety = -10
		reverse_opinion = {
			who = event_target:pet_accuser
			modifier = opinion_ignored_demands
			years = 5
		}
		ai_chance = { factor = 50 }
	}
	option = { # Familiar? Perhaps. Would you like to meet its master?
		name = EVTOPTCpet30
		trigger = { has_magic_power_trigger = yes }
		custom_tooltip = { text = EVTTOOLTIPpet30_c }
		prestige = 25
		add_rival = event_target:pet_accuser
		ai_chance = { factor = 40 }
	}
	option = { # Burn the accuser
		name = EVTOPTDpet30
		trigger = {
			event_target:pet_accuser = { liege = { character = ROOT } }
			pet_cruel_cat_lover_trigger = yes
		}
		custom_tooltip = { text = EVTTOOLTIPpet30_d }
		if = {
			limit = { event_target:pet_accuser = { is_witch_hunter_trigger = yes } }
			piety = -50
		}
		else = { piety = -25 }
		event_target:pet_accuser = {
			death = { death_reason = death_execution_burning killer = ROOT }
		}
		ai_chance = { factor = 5 modifier = { factor = 0 trait = zealous } }
	}
}
