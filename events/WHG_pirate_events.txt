# Pirates of the Old World's seas
# Namespace: pirate (was DR_pirates in DR_pirate_events.txt)
# Faction: Tilea, Estalia & Sartosa
# Serves: Sartosan pirate lords of the pirate government and every faithful
#   of Stromfels, the Shark God, wherever they sail
#
# A raider who loots a coast becomes a byword along it, with a pirate's
# name to match. Pirate lords snatch noble passengers off the ships of
# their neighbours and hold them for ransom, and their kin may pay to have
# them back. A pirate's lost eye, hand or leg only makes the crew fear and
# admire him more. When a pirate lord dies, his First Mate takes the ships
# and the lands, and the blood heir is left the name. A Pirate Legend of
# the Brethren may fit out a plunder fleet under a trusted courtier, who
# sails to raid in the Legend's name and sends half the booty home each
# year, until the fleet is sunk, its captain comes home rich, or the captain
# keeps the ships and turns free raider.
#
# The terror of the seas               pirate.1
# Taken for ransom                     pirate.10 - pirate.13
# Scars of the trade                   pirate.20
# The First Mate                       pirate.30 - pirate.31
# The plunder fleet                    pirate.39 - pirate.48
#
# Also in this file (each part keeps its own header and namespace):
#   Part 2: Sacking a holding (namespace pillage)
#
# Callers: pirate.1 from on_loot_settlement, pirate.10 from the five-year
#   pirate_raids list (was MTTH 100 years), pirate.20 from on_bi_yearly_pulse
#   (was DR_pirates.106 - 107), pirate.30 from on_new_holder_inheritance (all
#   WHG_pulse_on_actions.txt). The First Mate is the minor title
#   title_designatedheir (00_armoury_minor_titles.txt). pirate.39 from the
#   pirate_plunder_fleet decision (wh_pirate_decisions.txt), the Plunder
#   Fleet power of the adventurers_pirates society (was DR_pirates.101 -
#   103, commented out with their decision); then a yearly chain.
# Saved: event targets former_captain and designated_heir (pirate.30).
#   Persistent event target plunder_patron on the captain; flags
#   plunder_fleet_captain and plunder_fleet_patron; tributary type
#   plunder_fleet (00_tributary_types.txt).
#   Character modifiers one_eyed_pirate, one_handed_pirate and
#   one_legged_pirate (WHG_sea_event_modifiers.txt). Triggers in WHG_sea_triggers.txt.
# Loc: localisation/00_WHG_pirate_events.csv

# pirate.10 A prize at sea
# Hidden. A pirate lord's ships take a noble passenger off a neighbour's
# coast: a ruler, a rich courtier, the liege's close kin or a councillor of
# a great house. Any wards travelling with them are taken too (pirate.11).
# Scope: ROOT is a playable pirate lord or Stromfels faithful with a port
#   capital. Fired by the five-year pirate_raids list (WHG_pulse_on_actions; was
#   MTTH 100 years).
character_event = {
	id = pirate.10
	hide_window = yes
	is_triggered_only = yes

	only_playable = yes
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		OR = {
			religion = stromfels
			government = pirate_government
		}
		capital_scope = { port = yes }
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 2 tier = DUKE }
		modifier = { factor = 4 higher_tier_than = DUKE }
		modifier = { factor = 3 in_command = yes is_at_sea = yes }
		modifier = { factor = 1.5 government = pirate_government }
		modifier = { factor = 0.8 NOT = { martial = 2 } }
		modifier = { factor = 0.8 NOT = { martial = 4 } }
		modifier = { factor = 0.8 NOT = { martial = 6 } }
		modifier = { factor = 1.1 martial = 10 }
		modifier = { factor = 1.1 martial = 12 }
		modifier = { factor = 1.1 martial = 14 }
		modifier = { factor = 1.1 martial = 16 }
		modifier = { factor = 1.1 martial = 18 }
		modifier = { factor = 1.1 martial = 20 }
	}

	immediate = {
		random_playable_ruler = {
			limit = {
				capital_scope = { port = yes }
				demesne_size = 1
				NOT = { same_realm = ROOT }
				is_within_diplo_range = ROOT
				pirate_ransom_victim_trigger = yes
			}
			random_courtier = {
				limit = {
					age = 15
					prisoner = no
					NOT = { trait = in_hiding }
					NOT = { trait = incapable }
					has_epidemic = no
					in_command = no
					location = { port = yes }
					is_within_diplo_range = ROOT
					OR = { # Worth a ransom
						is_ruler = yes
						wealth = 100
						liege = { is_close_relative = PREV }
						AND = {
							NOT = { dynasty = 0 }
							is_councillor = yes
						}
					}
					pirate_ransom_victim_trigger = yes
				}
				character_event = { id = pirate.11 }
				any_ward = {
					limit = {
						prisoner = no
						NOT = { trait = in_hiding }
						NOT = { trait = incapable }
						in_command = no
						at_location = PREV
					}
					character_event = { id = pirate.11 }
				}
			}
		}
	}
}

# pirate.20 Scars of the trade
# Hidden upkeep. A pirate lord's lost eye, hand or leg makes the crew fear
# and admire him (a modifier per wound). The modifier goes when the wound
# heals or the lord is no longer a pirate.
# Scope: any character. Fired from on_bi_yearly_pulse (WHG_pulse_on_actions; was
#   DR_pirates.106 - 107).
character_event = {
	id = pirate.20
	hide_window = yes
	is_triggered_only = yes
	only_rulers = yes

	trigger = {
		OR = {
			government = pirate_government
			has_character_modifier = one_eyed_pirate
			has_character_modifier = one_handed_pirate
			has_character_modifier = one_legged_pirate
		}
	}

	immediate = {
		if = {
			limit = {
				trait = one_eyed
				government = pirate_government
			}
			if = {
				limit = { NOT = { has_character_modifier = one_eyed_pirate } }
				add_character_modifier = { name = one_eyed_pirate duration = -1 }
			}
		}
		else_if = {
			limit = { has_character_modifier = one_eyed_pirate }
			remove_character_modifier = one_eyed_pirate
		}
		if = {
			limit = {
				trait = one_handed
				government = pirate_government
			}
			if = {
				limit = { NOT = { has_character_modifier = one_handed_pirate } }
				add_character_modifier = { name = one_handed_pirate duration = -1 }
			}
		}
		else_if = {
			limit = { has_character_modifier = one_handed_pirate }
			remove_character_modifier = one_handed_pirate
		}
		if = {
			limit = {
				trait = one_legged
				government = pirate_government
			}
			if = {
				limit = { NOT = { has_character_modifier = one_legged_pirate } }
				add_character_modifier = { name = one_legged_pirate duration = -1 }
			}
		}
		else_if = {
			limit = { has_character_modifier = one_legged_pirate }
			remove_character_modifier = one_legged_pirate
		}
	}
}

# pirate.30 The First Mate takes the ships
# Hidden. When a pirate lord dies and his heir inherits, the First Mate of
# his house takes every county and every lesser title but the primary one,
# and is paid for it. The heir is told (pirate.31).
# Scope: ROOT is the heir. FROM: the inherited title. FROMFROM: the dead
#   pirate lord. Fired from on_new_holder_inheritance (WHG_realm_on_actions).
character_event = {
	id = pirate.30
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		government = pirate_government
		FROMFROM = {
			government = pirate_government
			primary_title = { title = FROM }
		}
		any_dynasty_member = {
			has_minor_title = title_designatedheir
			NOT = { character = ROOT }
		}
		any_demesne_title = {
			tier = COUNT
			can_be_given_away = yes
		}
	}

	immediate = {
		FROMFROM = { save_event_target_as = former_captain }
		random_dynasty_member = {
			limit = {
				has_minor_title = title_designatedheir
				NOT = { character = ROOT }
			}
			save_event_target_as = designated_heir
		}
		any_demesne_title = {
			limit = {
				tier = COUNT
				can_be_given_away = yes
			}
			usurp_title = { target = event_target:designated_heir type = invasion }
		}
		any_demesne_title = {
			limit = {
				higher_tier_than = COUNT
				NOT = { title = FROM }
			}
			usurp_title = { target = event_target:designated_heir type = invasion }
		}
		event_target:designated_heir = {
			wealth = 80
			prestige = 250
		}
		narrative_event = { id = pirate.31 days = 1 }
	}
}

character_event = {

# pirate.40 The fleet sails
# Hidden. A Pirate Legend gives a courtier ships and a crew of cutthroats
# and sends them out to plunder in the Legend's name (the plunder fleet
# decision in wh_pirate_decisions.txt, via pirate.39). The captain leaves
# the court with a landless title, a raiding host, a fleet and two mates,
# and pays tribute to the Legend (with Jade Dragon, so the fleet will not
# raid its own patron and can be called to the Legend's wars). The booty is
# shared once a year (pirate.41).
# Scope: ROOT is the new captain. FROM: the Pirate Legend.
character_event = {
	id = pirate.40
	hide_window = yes

	is_triggered_only = yes # pirate.39

	immediate = {
		end_inaccessibility_effect = yes
		if = {
			limit = { NOT = { trait = adventurer } }
			add_trait = adventurer
		}
		save_persistent_event_target = { name = plunder_patron scope = FROM }
		set_character_flag = plunder_fleet_captain
		FROM = {
			set_character_flag = plunder_fleet_patron
			capital_scope = {
				save_event_target_as = target_province
				ROOT = {
					set_defacto_liege = ROOT
					create_title = {
						tier = DUKE
						landless = yes
						temporary = yes
						adventurer = yes
						culture = ROOT
						name = "DYNAMIC_PIRATE_PLUNDER_FLEET"
						holder = ROOT
						base_title = THIS
					}
					adventurer_raid_host_effect = yes
					create_random_soldier = {
						random_traits = yes
						dynasty = none
						religion = ROOT
						culture = ROOT
						age = 30
					}
					new_character = {
						character_event = { id = aiclean.31 } # Racial trait
					}
					create_random_soldier = {
						random_traits = yes
						dynasty = none
						religion = ROOT
						culture = ROOT
						age = 25
					}
					new_character = {
						character_event = { id = aiclean.31 } # Racial trait
					}
					spawn_fleet = {
						province = closest
						owner = ROOT
						troops = { galleys = { 110 110 } }
					}
				}
			}
			if = {
				limit = { has_dlc = "Jade Dragon" }
				make_tributary = { who = ROOT tributary_type = plunder_fleet }
			}
		}
		character_event = { id = pirate.41 days = 365 }
	}
}

# pirate.41 Weighing the plunder
# Hidden. Once a year the captain takes stock. A fleet that is sunk or
# scattered, a captain in chains or a captain who has won land of their own
# is finished (pirate.44). A captain whose patron is dead keeps the ships
# and turns free raider (pirate.43). Otherwise the patron's share goes home
# (pirate.42), the captain may turn on the patron (pirate.43; likelier for
# the ambitious, the greedy and the disloyal), and after five years at sea
# the captain may come home with the last of the hoard (pirate.45).
# Scope: ROOT is the captain.
character_event = {
	id = pirate.41
	hide_window = yes

	is_triggered_only = yes # pirate.40, itself

	has_character_flag = plunder_fleet_captain

	immediate = {
		if = {
			limit = {
				OR = {
					prisoner = yes
					has_regiments = no
					NOT = { primary_title = { temporary = yes adventurer = yes } }
				}
			}
			character_event = { id = pirate.44 }
		}
		else_if = {
			limit = { persistent_event_target:plunder_patron = { is_alive = no } }
			character_event = { id = pirate.43 }
		}
		else = {
			random_list = {
				80 = { # Plunder on, and pay the share
					persistent_event_target:plunder_patron = { letter_event = { id = pirate.42 } }
					repeat_event = { id = pirate.41 days = 365 }
				}
				5 = { # Turn on the patron
					modifier = { factor = 3 trait = ambitious }
					modifier = { factor = 2 trait = greedy }
					modifier = { factor = 2 trait = deceitful }
					modifier = { factor = 2 trait = arbitrary }
					modifier = { factor = 3 NOT = { opinion = { who = persistent_event_target:plunder_patron value = 0 } } }
					modifier = { factor = 0.2 opinion = { who = persistent_event_target:plunder_patron value = 50 } }
					modifier = { factor = 0.2 is_friend = persistent_event_target:plunder_patron }
					modifier = { factor = 0.1 is_close_relative = persistent_event_target:plunder_patron }
					modifier = { factor = 0.5 trait = honest }
					modifier = { factor = 0.5 trait = content }
					character_event = { id = pirate.43 }
				}
				10 = { # Home with the hoard
					modifier = { factor = 0 NOT = { had_character_flag = { flag = plunder_fleet_captain days = 1825 } } }
					modifier = { factor = 2 had_character_flag = { flag = plunder_fleet_captain days = 2920 } }
					modifier = { factor = 3 had_character_flag = { flag = plunder_fleet_captain days = 4380 } }
					modifier = { factor = 2 wealth = 500 }
					modifier = { factor = 0.5 trait = ambitious }
					character_event = { id = pirate.45 }
				}
			}
		}
	}
}

letter_event = {

		}

# pirate.43 The fleet turns rogue
# Hidden. The captain keeps the ships and the plunder and sails for
# themself, and from now on is a free raider like any other (the raiding
# adventurers of WHG_adventurer_events.txt, adventurer.10): one day they
# will want land of their own. A living patron is told (pirate.46) and

###############################################################################
# Part 2: Sacking a holding
# Namespace: pillage (was DoWA.1 - 3 in DoW_Adventurers.txt)
# Faction: Tilea, Estalia & Sartosa
# Serves: pirate lords, the Druchii, the Ogres and the Norscans of the old
#   gods
#
# A raider who storms a rich holding (four buildings or more) asks
# whether his men may plunder it: the spoils depend on the holding, and
# the temple-cities of the Lizardmen hold the Old Ones' knowledge. Or he
# takes hostages for ransom instead. Where the slavers have already carried
# off the people (slaves_taken, slave.5), raiders come back to put what is
# left to the torch.
#
# Plunder or hostages                  pillage.1
# Put to the torch                     pillage.2 - pillage.3
#
# Callers: the adventurers_events list of on_siege_over_winner
#   (WHG_war_on_actions.txt). Slaves are taken by slave.4 - 5
#   (WHG_slavery_events.txt), not here.
# Saved: province modifiers slaves_taken (read) and
#   recently_burnt_the_land_not_owner; holding modifier sacked.
#   Triggers pillage_raider_trigger and pillage_temple_city_trigger in
#   WHG_sea_triggers.txt.
# Loc: localisation/00_WHG_pirate_events.csv
###############################################################################

namespace = pillage

# pillage.1 Sack the holding
# The raider's men want to plunder a stormed holding. Plunder it: gold by
# the holding (a city 50 and some learning, a temple 40 and piety, a castle
# 30 and prestige, a tribe 15), a Lizardmen temple-city more learning and
# gold; a building burns and the holder and his liege are furious. Or take
# the holder's people as hostages: courtiers in the holding are taken or
# killed, the holder taken half the time.
# Scope: ROOT is the winner. FROM is the holding title.
character_event = {
	id = pillage.1
	desc = {
		text = EVTDESCpillage1
		trigger = { FROM = { NOT = { culture_group = lizardman_group } } }
	}
	desc = {
		text = EVTDESCpillage1_lizardmen
		trigger = { FROM = { culture_group = lizardman_group } }
	}
	picture = {
		picture = GFX_evt_lizardmen_temple_fire
		trigger = { FROM = { culture_group = lizardman_group } }
	}
	picture = {
		picture = GFX_evt_norscan_raid
		trigger = {
			FROM = { NOT = { culture_group = lizardman_group } }
			culture_group = norscan_group
		}
	}
	picture = {
		picture = GFX_evt_ogre_raiders
		trigger = {
			FROM = { NOT = { culture_group = lizardman_group } }
			culture_group = ogre_group
		}
	}
	picture = {
		picture = GFX_evt_questing_ocean
		trigger = {
			FROM = { NOT = { culture_group = lizardman_group } }
			culture_group = druchii_group
		}
	}
	picture = {
		picture = GFX_evt_pirate_crew
		trigger = {
			FROM = { NOT = { culture_group = lizardman_group } }
			NOR = {
				culture_group = norscan_group
				culture_group = ogre_group
				culture_group = druchii_group
			}
		}
	}
	border = GFX_event_normal_frame_war
	portrait = event_target:pillage_top_liege

	is_triggered_only = yes # on_siege_over_winner, adventurers_events

	only_playable = yes
	prisoner = no

	trigger = {
		pillage_raider_trigger = yes
		FROM = {
			OR = {
				holding_type = city
				holding_type = temple
				holding_type = castle
				holding_type = tribal
			}
			num_of_buildings = 4
			location = { NOT = { has_province_modifier = slaves_taken } }
			holder_scope = { NOT = { character = ROOT } }
		}
	}

	immediate = {
		FROM = {
			location = { save_event_target_as = pillage_province }
			holder_scope = {
				save_event_target_as = pillage_holder
				top_liege = { save_event_target_as = pillage_top_liege }
			}
		}
	}

	# Plunder it
	option = {
		name = EVTOPTApillage1
		show_portrait = event_target:pillage_holder
		if = {
			limit = { FROM = { holding_type = city } }
			wealth = 50
			custom_tooltip = {
				text = pillage_learning_tt
				hidden_effect = {
					random_list = {
						10 = { military_techpoints = 5 }
						10 = { economy_techpoints = 5 }
						10 = { culture_techpoints = 5 }
					}
				}
			}
		}
		else_if = {
			limit = { FROM = { holding_type = temple } }
			wealth = 40
			piety = 25
		}
		else_if = {
			limit = { FROM = { holding_type = castle } }
			wealth = 30
			prestige = 15
		}
		else = {
			wealth = 15
		}
		if = {
			limit = {
				FROM = { culture_group = lizardman_group }
				event_target:pillage_province = { pillage_temple_city_trigger = yes }
			}
			wealth = 25
			custom_tooltip = {
				text = pillage_learning_tt
				hidden_effect = {
					random_list = {
						10 = { military_techpoints = 10 }
						10 = { economy_techpoints = 10 }
						10 = { culture_techpoints = 10 }
					}
				}
			}
		}
		FROM = { destroy_random_building = yes }
		event_target:pillage_holder = {
			opinion = { modifier = opinion_furious who = ROOT years = 3 }
		}
		event_target:pillage_top_liege = {
			if = {
				limit = { NOT = { character = event_target:pillage_holder } }
				opinion = { modifier = opinion_furious who = ROOT years = 3 }
			}
		}
		ai_chance = { factor = 90 }
	}
	# Take hostages
	option = {
		name = EVTOPTBpillage1
		show_portrait = event_target:pillage_holder
		event_target:pillage_holder = {
			random_courtier = {
				limit = { at_location = event_target:pillage_province }
				random_list = {
					35 = { imprison = ROOT }
					25 = {
						if = {
							limit = { is_invincible_trigger = yes }
							antideath_effect = yes
						}
						else = {
							death = { death_reason = death_murder killer = ROOT }
						}
					}
					40 = { }
				}
			}
			if = {
				limit = { at_location = event_target:pillage_province }
				random = {
					chance = 50
					imprison = ROOT
				}
			}
		}
		ai_chance = {
			factor = 10
			modifier = { factor = 3 trait = greedy }
			modifier = { factor = 2 trait = cruel }
		}
	}
}

# pillage.2 Put to the torch
# Hidden. Raiders storm a holding whose people the slavers already carried
# off, and burn what is left: a building, the holding sacked for a year, the
# land burnt for ten (the Norscans and Druchii burn it again), and the doom
# of the world rises a little. The raider is told (pillage.3).
# Scope: ROOT is the winner. FROM is the holding title.
character_event = {
	id = pillage.2
	hide_window = yes

	is_triggered_only = yes # on_siege_over_winner, adventurers_events

	only_playable = yes

	trigger = {
		pillage_raider_trigger = yes
		FROM = {
			OR = {
				holding_type = city
				holding_type = temple
				holding_type = castle
				holding_type = tribal
			}
			has_any_building = yes
			NOT = { has_holding_modifier = sacked }
			NOT = { has_holding_modifier = burned }
			location = {
				has_province_modifier = slaves_taken
				OR = {
					NOT = { has_province_modifier = recently_burnt_the_land_not_owner }
					ROOT = {
						OR = {
							culture_group = norscan_group
							culture_group = druchii_group
						}
					}
				}
			}
		}
	}

	immediate = {
		FROM = {
			destroy_random_building = yes
			add_holding_modifier = { modifier = sacked months = 12 }
			location = {
				add_province_modifier = { name = recently_burnt_the_land_not_owner years = 10 }
			}
		}
		add_doom_progress_tiny_effect = yes
		if = {
			limit = { ai = no }
			character_event = { id = pillage.3 }
		}
	}
}
