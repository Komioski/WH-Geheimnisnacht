###############################################################################
# The harem: wives of the southern courts
# Namespace: polygamy (was 89000 - 89035 in vanilla polygamy_events.txt)
# Faction: Araby & Southlands
# Serves: the many-wived households of the southern faiths (Ormazd, the
#   Kitabids, the Atruhayid and the Akghan)
#
# Two wives fall in love and their husband must decide what to do about it.
# A lesser wife schemes to become first wife, by poison or by charm, and the
# first wife she pushes aside may strike back. A first wife whose son is not
# the heir begs a county for him. A wife poisons the heir, or another wife's
# son, to make room for her own.
#
# Part 1: two wives in love             polygamy.1 - polygamy.4
# Part 2: who is first wife             polygamy.10 - polygamy.17
# Part 3: a county for the son          polygamy.20 - polygamy.23
# Part 4: poison for the heir           polygamy.30 - polygamy.33
# Part 5: poison for a rival's son      polygamy.40 - polygamy.43
#
# Fired from: on_yearly_pulse (list polygamy_lovers, was MTTH 3 months) and
#   on_five_year_pulse (list polygamy_harem, was MTTH 320 - 820 months) in
#   wh_on_actions.txt. The rest is chained.
# Saved: event targets polygamy_wife, polygamy_lover, polygamy_schemer,
#   polygamy_first_wife, polygamy_son and polygamy_victim; once-per-life
#   flags polygamy_wife_lover, polygamy_schemed, envious_mother,
#   first_wife_son and polygamy_murderer.
# Dropped: 89021, 89023, 89031 and 89033, which went to a child who is
#   always AI (and in two cases already dead).
###############################################################################

namespace = polygamy

# Part 1: two wives in love

# polygamy.1 Two wives in love
# A wife who loves women and another wife of the same husband who does too,
# living under one roof, become lovers. Once per life for both. The husband
# hears of it in two weeks (polygamy.2).
# Scope: ROOT is the first wife of the pair.
character_event = {
	id = polygamy.1
	desc = EVTDESCpolygamy1
	picture = GFX_evt_lovers

	is_triggered_only = yes # on_yearly_pulse, list polygamy_lovers

	only_women = yes
	min_age = 16
	only_capable = yes
	prisoner = no
	religion_group = ormazd_gods_group
	is_married = yes

	trigger = {
		is_woman_trigger = yes
		trait = homosexual
		NOT = { has_character_flag = polygamy_wife_lover }
		spouse = {
			any_spouse = {
				NOT = { character = ROOT }
				is_woman_trigger = yes
				trait = homosexual
				at_location = ROOT
				NOT = { has_character_flag = polygamy_wife_lover }
			}
		}
	}

	immediate = {
		save_event_target_as = polygamy_wife
		spouse = {
			random_spouse = {
				limit = {
					NOT = { character = ROOT }
					is_woman_trigger = yes
					trait = homosexual
					at_location = ROOT
					NOT = { has_character_flag = polygamy_wife_lover }
				}
				save_event_target_as = polygamy_lover
			}
		}
	}

	option = {
		name = EVTOPTApolygamy1
		set_character_flag = polygamy_wife_lover
		event_target:polygamy_lover = {
			set_character_flag = polygamy_wife_lover
			if = {
				limit = {
					is_no_females_race = no
					ROOT = { is_no_females_race = no }
				}
				add_lover = ROOT
			}
		}
		hidden_effect = {
			spouse = { character_event = { id = polygamy.2 days = 14 } }
		}
	}
}

# polygamy.33); otherwise the cup is caught (polygamy.31). Once per life.
# (Vanilla's NOT around two conditions let her plot again.)

character_event = {
	id = polygamy.30
	desc = EVTDESCpolygamy30
	picture = GFX_evt_quarrel

	is_triggered_only = yes # on_five_year_pulse, list polygamy_harem

	only_women = yes
	min_age = 16
	only_capable = yes
	prisoner = no
	religion_group = ormazd_gods_group
	is_married = yes

	trigger = {
		is_woman_trigger = yes
		is_pregnant = no
		is_inaccessible_trigger = no
		NOR = {
			has_character_flag = polygamy_murderer
			any_child = { is_primary_heir = yes }
		}
		any_child = {
			is_female = no
			is_alive = yes
			is_primary_heir = no
			is_pretender = yes
		}
		spouse = {
			is_ruler = yes
			any_child = {
				is_primary_heir = yes
				ai = yes
				NOT = { age = 16 }
				NOT = { mother = { character = ROOT } }
				at_location = ROOT
			}
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = {
			factor = 4
			trait = envious
		}
		modifier = {
			factor = 2
			trait = ambitious
		}
		modifier = {
			factor = 0.5
			trait = slothful
		}
		modifier = {
			factor = 0.25
			trait = content
		}
	}

	immediate = {
		spouse = {
			random_child = {
				limit = { is_primary_heir = yes }
				save_event_target_as = polygamy_victim
			}
		}
	}

	option = {
		name = EVTOPTApolygamy30
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0.1
				trait = kind
			}

				factor = 0.25
				trait = just
			}
			modifier = {
				factor = 0.5
				trait = honest
			}
			modifier = {
				factor = 2
				trait = cruel
			}
			modifier = {
				factor = 2
				trait = ambitious
			}
		}
		set_character_flag = polygamy_murderer
		custom_tooltip = { text = EVTTOOLTIPpolygamy30 }
		hidden_effect = {
			random_list = {
				35 = {
					event_target:polygamy_victim = {
						death = {
							death_reason = death_murder_unknown_poison
							killer = ROOT
						}
					}
					spouse = {
						random_list = {
							50 = { character_event = { id = polygamy.32 } }
							50 = { character_event = { id = polygamy.33 } }
						}
					}
				}
				65 = {
					spouse = { character_event = { id = polygamy.31 } }
				}
			}
		}
	}
	option = {
		name = EVTOPTBpolygamy30
		ai_chance = { factor = 1 }
	}
