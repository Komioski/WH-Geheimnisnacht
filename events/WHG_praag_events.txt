# praag.1 Praag falls
# Hidden. A servant of Chaos wins a siege of a Praag holding: the city has
# fallen. If the curse was eased, it comes back at once.
# Scope: ROOT owns the winning army; FROM is the taken holding.
character_event = {
	id = praag.1
	hide_window = yes

	is_triggered_only = yes # on_siege_over_winner

	trigger = {
		FROM = { location = { province_id = 182 } }
		praag_chaos_trigger = yes
	}

	immediate = {
		set_global_flag = praag_fell_to_chaos
		if = {
			limit = { has_global_flag = praag_cursed }
			182 = {
				remove_province_modifier = praag_curse_eased
				if = {
					limit = { NOT = { has_province_modifier = praag_curse } }
					add_province_modifier = { name = praag_curse duration = -1 }
				}
			}
		}
	}
}

# praag.2 Praag changes hands
# Hidden. A servant of Chaos taking the county counts as the fall. A free
# ruler taking it after the fall brings down the curse (praag.3).
# Scope: ROOT is the new holder; FROM the title.
character_event = {
	id = praag.2
	hide_window = yes

	is_triggered_only = yes # on_new_holder, on_new_holder_inheritance, on_new_holder_usurpation
	only_rulers = yes

	trigger = {
		FROM = { title = c_praag }
	}

	immediate = {
		if = {
			limit = { praag_chaos_trigger = yes }
			set_global_flag = praag_fell_to_chaos
		}
		else_if = {
			limit = {
				has_global_flag = praag_fell_to_chaos
				NOT = { has_global_flag = praag_cursed }
			}
			set_global_flag = praag_cursed
			182 = { add_province_modifier = { name = praag_curse duration = -1 } }
			character_event = { id = praag.3 days = 7 }
		}
	}
}

# praag.70 The Organ of Praag
# Each year the organist hears the organ and sinks one step deeper into
# madness (lunatic_1, then 2, then 3, with the lunatic_counter of
# WHG_mutation_events.txt), and grows lustful. They may also lay down the
# post instead. Stops once the organist is raving mad.
# Scope: ROOT is the organist. From on_yearly_pulse in wh_on_actions.txt;
#   the flag organist_of_praag comes from the title's gain_effect.
character_event = {
	id = praag.70
	desc = {
		text = EVTDESCpraag70_first
		trigger = {
			NOR = {
				trait = lunatic_1
				trait = lunatic_2
			}
		}
	}
	desc = {
		text = EVTDESCpraag70_deeper
		trigger = {
			OR = {
				trait = lunatic_1
				trait = lunatic_2
			}
		}
	}
	picture = GFX_evt_mad_spirit
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes # on_yearly_pulse
	has_character_flag = organist_of_praag # set by the minor title

	trigger = {
		has_minor_title = title_organist_of_praag
		NOT = { trait = lunatic_3 }
	}

	# Play on
	option = {
		name = EVTOPTApraag70
		if = {
			limit = { NOT = { trait = lustful } }
			add_trait = lustful
		}
		if = {
			limit = {
				NOR = {
					trait = lunatic_1
					trait = lunatic_2
				}
			}
			add_trait = lunatic_1
			set_variable = { which = "lunatic_counter" value = 5 }
		}
		else_if = {
			limit = { trait = lunatic_1 }
			remove_trait = lunatic_1
			add_trait = lunatic_2
			set_variable = { which = "lunatic_counter" value = 10 }
		}
		else = {
			remove_trait = lunatic_2
			add_trait = lunatic_3
			set_variable = { which = "lunatic_counter" value = 20 }
		}
		prestige = 25
		ai_chance = {
			factor = 90
			modifier = { factor = 0.5 trait = lunatic_2 }
		}
	}

	# Lay down the post
	option = {
		name = EVTOPTBpraag70
		remove_title = title_organist_of_praag
		ai_chance = {
			factor = 10
			modifier = { factor = 2 trait = content }
			modifier = { factor = 0 trait = lunatic_2 }
		}
	}
}
