###############################################################################
# Prepared invasions
# Namespace: invasion (was 62300 - 62322 in misc_old_gods_events.txt)
# Faction: Norsca
# Serves: lords of the old Norscan gods (the only faith that may prepare an
#   invasion)
#
# A Norscan lord who prepares an invasion gathers warriors for it month by
# month: bands of warriors from the fjords, a smaller crew or two, now and
# then a renowned champion with his own hearth-guard. The greater the lord's
# name and skill, and the weaker the target, the more come. If the invasion
# becomes impossible the host breaks up; if the lord waits too long, the
# warriors drift away and the lord is shamed before the faithful.
#
# The invasion fails                   invasion.1 - invasion.2
# Warriors flock to the banner         invasion.10
#
# Fired from: on_prepared_invasion_aborts (invasion.1),
#   on_prepared_invasion_expires (invasion.2) and on_prepared_invasion_monthly
#   (invasion.10) in 00_on_actions.txt.
# Saved: flags invasion_band_large, invasion_band_small, invasion_champion
#   (cleared by invasion.10).
###############################################################################

namespace = invasion

###############################################################################
# The invasion fails
###############################################################################

# invasion.1 The invasion is no longer possible
# The gathered host is disbanded.
# Scope: ROOT is the would-be invader, FROM the target.
character_event = {
	id = invasion.1
	desc = EVTDESCinvasion1
	picture = GFX_evt_viking_battle_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	immediate = {
		remove_character_modifier = launched_prepared_invasion
		remove_opinion = {
			who = FROM
			modifier = preparing_to_invade
		}
		FROM = {
			remove_opinion = {
				who = ROOT
				modifier = preparing_to_invade_me
			}
		}
		disband_event_forces = prepared_invasion
	}

	option = {
		name = EVTOPTAinvasion1
	}
}

# invasion.2 Waited too long
# Unless the war has begun, the host drifts away and the lord loses 200
# prestige and 50 piety, and the lords of the faith in the realm think less
# of him.
# Scope: ROOT is the would-be invader, FROM the target.
character_event = {
	id = invasion.2
	desc = EVTDESCinvasion2
	picture = GFX_evt_viking_battle_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	immediate = {
		remove_opinion = {
			who = FROM
			modifier = preparing_to_invade
		}
		FROM = {
			remove_opinion = {
				who = ROOT
				modifier = preparing_to_invade_me
			}
		}
		if = {
			limit = {
				NOT = {
					any_war = {
						any_attacker = { character = FROM }
					}
				}
			}
			disband_event_forces = prepared_invasion
		}
	}

	option = {
		name = EVTOPTAinvasion2
		if = {
			limit = {
				NOT = {
					any_war = {
						any_attacker = { character = FROM }
					}
				}
			}
			prestige = -200
			piety = -50
			any_realm_lord = {
				limit = { religion = ROOT }
				opinion = {
					who = ROOT
					modifier = opinion_failed_invasion
				}
			}
		}
	}
}

###############################################################################
# Warriors flock to the banner
###############################################################################

# invasion.10 Warriors arrive
# A large band (a tenth of the lord's strength, with ships), a small one
# (three in forty), or rarely a champion with a host of his own (three in
# twenty, his band stays after the invasion). More often for a famous,
# skilled lord; never against a target four times stronger.
# Scope: ROOT is the invader, FROM the target. Was three events (62320 -
#   62322).
character_event = {
	id = invasion.10
	desc = {
		trigger = { has_character_flag = invasion_band_large }
		text = EVTDESCinvasion10_large
	}
	desc = {
		trigger = { has_character_flag = invasion_band_small }
		text = EVTDESCinvasion10_small
	}
	desc = {
		trigger = { has_character_flag = invasion_champion }
		text = EVTDESCinvasion10_champion
	}
	picture = GFX_evt_vikings_arriving_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	# On an on_action, so a greater weight means more often
	weight_multiplier = {
		days = 1
		modifier = {
			factor = 1.25
			martial = 15
		}
		modifier = {
			factor = 1.25
			martial = 20
		}
		modifier = {
			factor = 1.25
			diplomacy = 15
		}
		modifier = {
			factor = 1.25
			diplomacy = 20
		}
		modifier = {
			factor = 1.25
			prestige = 1000
		}
		modifier = {
			factor = 0.75
			NOT = { martial = 10 }
		}
		modifier = {
			factor = 0.5
			NOT = { martial = 7 }
		}
		modifier = {
			factor = 0.75
			NOT = { diplomacy = 10 }
		}
		modifier = {
			factor = 0.5
			NOT = { diplomacy = 7 }
		}
		modifier = {
			factor = 0.5
			NOT = { prestige = 100 }
		}
		modifier = {
			factor = 0.75
			lower_tier_than = DUKE
		}
		modifier = {
			factor = 0
			relative_power = {
				who = FROM
				power = 4.0
			}
		}
		modifier = {
			factor = 0.5
			relative_power = {
				who = FROM
				power = 2.0
			}
		}
		modifier = {
			factor = 0.25
			relative_power = {
				who = FROM
				power = 3.0
			}
		}
	}

	immediate = {
		random_list = {
			15 = { set_character_flag = invasion_band_large }
			15 = { set_character_flag = invasion_band_small }
			2 = { set_character_flag = invasion_champion }
		}
	}

	option = {
		name = {
			trigger = { NOT = { has_character_flag = invasion_champion } }
			text = EVTOPTAinvasion10
		}
		name = {
			trigger = { has_character_flag = invasion_champion }
			text = EVTOPTAinvasion10_champion
		}
		if = {
			limit = { has_character_flag = invasion_band_large }
			capital_scope = {
				ROOT = {
					spawn_unit = {
						province = PREV
						owner = ROOT
						match_character = ROOT
						match_mult = 0.1
						earmark = prepared_invasion
						attrition = 1.0
					}
					spawn_fleet = {
						province = closest_to_capital
						owner = ROOT
						match_character = ROOT
						match_mult = 0.1
						match_min = 7
						earmark = prepared_invasion
					}
				}
			}
		}
		else_if = {
			limit = { has_character_flag = invasion_band_small }
			capital_scope = {
				ROOT = {
					spawn_unit = {
						province = PREV
						owner = ROOT
						match_character = ROOT
						match_mult = 0.075
						earmark = prepared_invasion
						attrition = 1.0
					}
					spawn_fleet = {
						province = closest_to_capital
						owner = ROOT
						match_character = ROOT
						match_mult = 0.075
						match_min = 5
						earmark = prepared_invasion
					}
				}
			}
		}
		else = {
			capital_scope = {
				create_character = {
					random_traits = yes
					dynasty = none
					religion = ROOT
					culture = ROOT
					female = no
					age = 30
					attributes = {
						martial = 8
					}
					trait = brave
					trait = zealous
					trait = brilliant_strategist
					trait = holy_warrior
				}
				new_character = {
					character_event = { id = aiclean.31 } # Racial trait
					spawn_unit = {
						province = PREV
						owner = ROOT
						match_character = ROOT
						match_mult = 0.15
						attrition = 1.0
					}
				}
			}
			spawn_fleet = {
				province = closest_to_capital
				owner = ROOT
				match_character = ROOT
				match_mult = 0.15
				match_min = 10
				earmark = prepared_invasion
			}
		}
	}

	after = {
		clr_character_flag = invasion_band_large
		clr_character_flag = invasion_band_small
		clr_character_flag = invasion_champion
	}
}
