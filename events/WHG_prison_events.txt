###############################################################################
# Prison events: torture, humiliation and turncoat prisoners
#
# Namespace: prison
# Faction: Generic
# Serves: any ruler with prisoners to torture, mutilate or humiliate, and
# their prisoners.
#
# Each punishment is a pair: the ruler's side, then the prisoner's.
#   prison.1-8       Humiliation: tar and feathers, the stocks, the walk of
#                    shame, paraded naked (nomads)
#   prison.11-22     Torture: the rack, the whip, the rat bucket, the feather
#                    (mad rulers), heat and cold
#   prison.23-32     Mutilation: a leg, a hand, an eye, the face, castration
#   prison.33-38     Poetry readings
#   prison.50        A kinsman in prison defects to the jailer
#   prison.60-64     Blinding and castration notices (was LoR.830-834)
#   prison.70-74     A prisoner asks for better quarters
#   prison.75-80     The urge to torture a prisoner
#   prison.81-82     A captured enemy commander, kept or sacrificed
#   prison.85-86     A vassal petitions the liege to free a friend
#   prison.90        An ogre or a huge orc eats a prisoner (was ogre.1 in
#                    ogre_events.txt)
#   prison.91 - 92   A prisoner given to the Dark Gods or to Khaine: the
#                    notice to the victim's liege and the sacrificer's
#                    reward (was SSI.60 and SSI.72 in sunset_invasion.txt)
#
# Fired by decisions/rip_torture_decisions.txt, the recruit_prisoner
# decision in rip_various_decisions.txt and the vassal_blinding_decision and
# vassal_castrate_decision in vassal_decisions.txt (60-64). prison.70, 75
# and 85 are on yearly lists in wh_on_actions; prison.81 is fired by the
# battle events (WHG_battle_events). prison.90 is fired by the
# ogre_eat_prisoner decision in wh_magic_decisions.txt, prison.91 - 92 by
# the vassal_sacrifice_decision in decisions/sunset_invasion.txt.
# Was events/rip_torture_events.txt (RIP.15001-15008 = prison.1-8,
# RIP.20001-20028 = prison.11-38), RIP.29700 in rip_flavor_events.txt and
# 50005 - 50026 in the vanilla imprisoned_events.txt (prison.70-86).
# Scripted: WHG_prison_triggers.txt, WHG_prison_effects.txt.
# Loc: localisation/00_WHG_prison_events.csv
###############################################################################

namespace = prison

	option = {
		name = EVTOPTprison22_A

				}
			}
			30 = {
				modifier = { factor = 0 is_disease_immune_race = yes }
				if = {
					limit = {
						NOR = {
							num_of_symptoms = 1
							is_ill = yes
							has_character_flag = developing_illness
							has_character_flag = disease_not_serious
						}
					}

					tooltip = { add_trait = cough }

					hidden_tooltip = {
						random_list = {
							80 = {
								set_character_flag = disease_not_serious
								character_event = { id = treatment.3 days = 30 random = 90 }
							}
							20 = {
								set_character_flag = got_pneumonia
								set_character_flag = developing_illness
							}
						}
						character_event = { id = symptom.1 }
					}
				}
			}
			5 = {
				if = {
					limit = { NOR = { trait = lunatic_1 trait = lunatic_2 trait = lunatic_3 } }

				}

# prison.90 The prisoner is eaten
# An ogre or a huge orc eats a prisoner of rank in front of the court.
# Scope: ROOT is the eater, FROM the prisoner. Fired by the
# ogre_eat_prisoner decision (wh_magic_decisions) just before the prisoner
# dies. The eater may grow fat (three in ten), boasts of it (prestige),
# and the court reacts: ogres, greenskins, the brave and the cruel cheer,
# the timid and the kind are scared.
# Fixed from ogre.1: the fat check looked at the prisoner instead of the
# eater, and the opinion went to everyone who was not clever or brave,
# as if dullness made people like a cannibal feast.
character_event = {
	id = prison.90
	desc = { trigger = { sway_voice_ogre_trigger = yes } text = EVTDESCprison90_ogre }
	desc = { trigger = { sway_voice_ogre_trigger = no } text = EVTDESCprison90_orc }
	picture = GFX_evt_beastmen_charge
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	option = {
		name = EVTOPTAprison90
		prestige = 25
		if = { limit = { NOT = { trait = is_fat } } random = { chance = 30 add_trait = is_fat } }
		any_courtier = {
			limit = {
				at_location = ROOT
				OR = {
					sway_voice_ogre_trigger = yes
					sway_voice_greenskin_trigger = yes
					trait = brave
					trait = cruel
				}
			}
			opinion = { who = ROOT modifier = gluttonous_pos }
		}
		any_courtier = {
			limit = {
				at_location = ROOT
				sway_voice_ogre_trigger = no
				sway_voice_greenskin_trigger = no
				NOR = { trait = brave trait = cruel }
				OR = { trait = craven trait = kind trait = shy }
			}
			opinion = { who = ROOT modifier = opinion_terrified }
		}
	}
}
