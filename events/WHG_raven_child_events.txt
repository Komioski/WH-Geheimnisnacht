###############################################################################
# The Raven's Child
# Namespace: ravenchild (was HF.25800 - HF.25828 in HF_childrens_crusade_events)
# Faction: Chaos
#   (a Tzeentchian child of the Chaos Wastes, the march on Kislev)
#
# Once in a game, a child at the court of a nomad lord of the Wastes dreams of
# the Raven God and sets out to lead a ragged Black Crusade of children and
# wretches south to Kislev. The march crosses the Wastes in stages (Far
# Wastes, Broken Lands, Eastern Steppes, Zanbaijin); in each, rulers of the
# Dark Gods are asked for food, men and faith, and their own children may run
# off to follow. One may become the child's sponsor. At Zanbaijin the march
# becomes a war for Kislev (CB childrens_crusade).
#
# The prophet                       ravenchild.1 - ravenchild.8
# The march                         ravenchild.9 - ravenchild.15
# News from the Wastes              ravenchild.16 - ravenchild.20
# The war for Kislev                ravenchild.21 - ravenchild.29
#
# Saved: event_target childrens_crusade_leader (the child), persistent target
#   childrens_crusade_sponsor_per (the one sponsor; never picked by the AI).
# Variables on the child: children_crusade_forces (1 = about 5,000 light
#   troops), children_crusade_heavy_troops (1 = about 1,000 heavy),
#   children_crusade_morale (morale modifier at the start of the war),
#   children_crusade_progress (pulses spent in the current land; at 3 the
#   march moves on).
#
# Callers:
#   common/on_actions/00_on_actions.txt: on_decade_pulse random list (1),
#     on_become_imprisoned_any_reason (4), on_death (2).
#   common/cb_types/01_cb_types.txt: childrens_crusade (24, 25, 26, 27, 28, 29).
#   common/scripted_effects/wh_ravenchild_effects.txt (16 - 21).
# Support: common/scripted_effects/wh_ravenchild_effects.txt,
#   common/scripted_triggers/wh_ravenchild_triggers.txt
# Loc: localisation/00_WHG_raven_child_events.csv
###############################################################################

# ravenchild.9 The march goes on (hidden)
# Every two months or so the child grows, may gain followers and captains,
# one ruler of the land being crossed hears of it (ravenchild.10), and
# after three stops the march moves on (ravenchild_move_effect).
# Scope: ROOT is the child (ravenchild.1, itself, ravenchild_move_effect).
narrative_event = {
	id = ravenchild.9
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		# The child grows

		# A martial focus
		if = {
			limit = { NOT = { age = 12 } NOT = { has_focus = focus_ch_struggle } }
			set_focus = focus_ch_struggle
		}
		if = {
			limit = { age = 12 NOT = { age = 16 } NOT = { has_focus = focus_ed_martial } }
			set_focus = focus_ed_martial
		}
		if = { limit = { age = 16 NOT = { has_focus = focus_war } } set_focus = focus_war }

		# Keeps the faith
		if = { limit = { trait = cynical } remove_trait = cynical }
		if = {
			limit = { age = 14 NOT = { trait = zealous } }
			random = { chance = 25 add_trait = zealous }
		}
		if = { limit = { NOT = { religion = tzeentch } } religion = tzeentch }
		if = {
			limit = { NOT = { has_character_modifier = voice_of_tzeentch } }
			add_character_modifier = { name = voice_of_tzeentch duration = -1 }
		}
		remove_disease_trait_effect = yes # Tzeentch heals

		# Learns to fight on the road
		if = { limit = { NOT = { martial = 18 } } random = { chance = 10 change_martial = 1 } }
		if = {
			limit = { NOT = { combat_rating = 50 } }
			random = { chance = 10 add_one_duel_experience_modifier = yes }
		}

		# Prophets trust too easily
		if = {
			limit = { NOT = { trait = trusting } }
			random = { chance = 10 add_trait = trusting }
		}

		random = { # More followers
			chance = 15
			modifier = { factor = 1.5 martial = 6 }
			modifier = { factor = 1.5 martial = 8 }
			modifier = { factor = 1.5 martial = 10 }
			modifier = { factor = 1.5 martial = 12 }
			modifier = { factor = 1.75 piety = 150 }
			modifier = { factor = 1.75 piety = 250 }
			modifier = { factor = 1.75 piety = 450 }
			modifier = { factor = 1.75 piety = 650 }
			modifier = { factor = 1.75 wealth = 150 }
			modifier = { factor = 1.75 wealth = 350 }
			modifier = { factor = 1.75 wealth = 550 }
			modifier = { factor = 1.75 prestige = 250 }
			modifier = { factor = 1.75 prestige = 500 }
			modifier = { factor = 1.75 prestige = 750 }
			modifier = { factor = 1.75 prestige = 1000 }
			modifier = {
				factor = 1.5
				persistent_event_target:childrens_crusade_sponsor_per = {
					is_alive = yes
					real_tier = DUKE
				}
			}
			modifier = {
				factor = 1.75
				persistent_event_target:childrens_crusade_sponsor_per = {
					is_alive = yes
					real_tier = KING
				}
			}
			modifier = {
				factor = 2
				persistent_event_target:childrens_crusade_sponsor_per = {
					is_alive = yes
					real_tier = EMPEROR
				}
			}
			change_variable = { which = children_crusade_forces value = 2 }
		}

		# A captain may join
		random_list = {
			2 = { spawn_fantastic_commander_effect = yes clear_event_target = invited_character }
			15 = { spawn_good_commander_effect = yes clear_event_target = invited_character }
			80 = {
				modifier = { factor = 0.85 martial = 6 }
				modifier = { factor = 0.85 martial = 8 }
				modifier = { factor = 0.85 martial = 10 }

# ravenchild.21 The war begins (hidden)
# The gathered forces become armies, morale becomes a modifier, and the
# child declares war on the holder of Kislev.
# Scope: ROOT is the child (ravenchild_move_effect).
narrative_event = {
	id = ravenchild.21
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		character_event = { id = ravenchild.22 }
		character_event = { id = ravenchild.23 }

		# Morale
		if = {
			limit = { NOT = { check_variable = { which = children_crusade_morale value = -15 } } }
			add_character_modifier = { name = children_crusade_morale_0 duration = 1000 }
		}
		if = {
			limit = {
				check_variable = { which = children_crusade_morale value = -15 }
				NOT = { check_variable = { which = children_crusade_morale value = -7 } }
			}
			add_character_modifier = { name = children_crusade_morale_1 duration = 1000 }
		}
		if = {
			limit = {
				check_variable = { which = children_crusade_morale value = -7 }
				NOT = { check_variable = { which = children_crusade_morale value = -3 } }
			}
			add_character_modifier = { name = children_crusade_morale_2 duration = 1000 }
		}
		if = {
			limit = {
				check_variable = { which = children_crusade_morale value = -3 }
				NOT = { check_variable = { which = children_crusade_morale value = 2 } }
			}
			add_character_modifier = { name = children_crusade_morale_3 duration = 1000 }
		}
		if = {
			limit = { check_variable = { which = children_crusade_morale value = 2 } }
			add_character_modifier = { name = children_crusade_morale_4 duration = 1000 }
		}
		set_variable = { which = children_crusade_progress value = 0 }
		set_variable = { which = children_crusade_forces value = 0 }
		set_variable = { which = children_crusade_heavy_troops value = 0 }
		set_variable = { which = children_crusade_morale value = 0 }

		c_kislev = {
			holder_scope = {
				event_target:childrens_crusade_leader = {
					unsafe_war = {
						target = PREV
						casus_belli = childrens_crusade
						thirdparty_title = PREVPREV
						tier = KING
					}
				}
			}
		}

		if = {
			limit = {
				persistent_event_target:childrens_crusade_sponsor_per = {
					has_character_flag = flag_childrens_crusade_sponsor_joins_war
				}
			}
			persistent_event_target:childrens_crusade_sponsor_per = {
				join_attacker_wars = ROOT
				join_defender_wars = ROOT
			}
		}
	}
}
