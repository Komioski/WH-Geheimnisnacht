###############################################################################
# Realm events
# Namespace: realm (was HFP.40000 - HFP.40038, HFP.24120, HFP.99999 in
#   HFP_realm_events)
# Faction: the cult of Sigmar, the merchant republics, the Border Princes
#
# The vanilla realm-forming news, moved to the mod's own analogues:
# - the Papal States lost become the Grand Theogonist cast out: when he
#   loses the last of the temple's lands in a war, every Sigmarite ruler
#   hears of it (at most once a decade);
# - the merchant kingdoms of Venice, Genoa, Pisa and Amalfi become any
#   merchant republic that takes a crown (a Tilean city, Marienburg);
# - the Swiss Confederation becomes the Eagle and Northern Border
#   Confederacies, and Antioch and Cyprus the principalities of Khypria
#   and Myrmidens: a Border Prince may refound any of them that has fallen
#   empty (decisions/WHG_realm_decisions.txt).
#
# The Grand Theogonist cast out   realm.1 - realm.2
# Merchant crowns                 realm.10
# The Border Princes refounded    realm.20 - realm.21
#
# Saved: title flag gt_cast_out on k_cult_of_sigmar; character flags
#   realm_refound_<eagle|northern|khypria|myrmidens>; event target
#   grand_theogonist.
#
# Callers:
#   common/scripted_effects/00_scripted_effects.txt (landless_pope_effect:
#     1), run by the CBs when a war is lost;
#   common/on_actions/00_on_actions.txt: on_create_title (10);
#   decisions/WHG_realm_decisions.txt (realm_refound_*: 20).
#
# Already covered elsewhere: the Hansa is the Marienburg League (REP.50-61),
# the HRE electors are the Elector Counts. Removed for want of an analogue:
# Austria, the Eastern Marches, Byzantium, the Spanish and crusader orders,
# Romagna, the HRE stem duchies and the Slavic union.
###############################################################################

namespace = realm

###############################################################################
# The Grand Theogonist cast out
###############################################################################

# realm.1 The temple's last lands are lost (hidden)
# Scope: the Grand Theogonist, who holds no land and whose realm holds none.
character_event = {
	id = realm.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_landed_title = k_cult_of_sigmar
		k_cult_of_sigmar = { NOT = { had_title_flag = { flag = gt_cast_out days = 3650 } } }
	}

	immediate = {
		k_cult_of_sigmar = { clr_title_flag = gt_cast_out set_title_flag = gt_cast_out }
		save_event_target_as = grand_theogonist
		narrative_event = { id = realm.2 }
		any_playable_ruler = {
			limit = { religion = sigmarite NOT = { character = ROOT } }
			narrative_event = { id = realm.2 }
		}
	}
}

# realm.2 The Grand Theogonist cast out (news)
# Scope: a Sigmarite ruler, or the Grand Theogonist.
narrative_event = {
	id = realm.2
	title = EVTTITLErealm2
	desc = { trigger = { character = event_target:grand_theogonist } text = EVTDESCrealm2_gt }
	desc = {
		trigger = { NOT = { character = event_target:grand_theogonist } }
		text = EVTDESCrealm2
	}
	picture = GFX_evt_cathedral
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes

	option = { #The Grand Theogonist
		name = EVTOPTArealm2
		trigger = { character = event_target:grand_theogonist }
		piety = -100
	}
	option = { #A zealot
		name = EVTOPTBrealm2
		trigger = { NOT = { character = event_target:grand_theogonist } trait = zealous }
		piety = 25
	}
	option = { #Anyone else
		name = EVTOPTCrealm2
		trigger = { NOT = { character = event_target:grand_theogonist } NOT = { trait = zealous } }
	}
}

###############################################################################
# Merchant crowns
###############################################################################

# realm.10 A merchant republic takes a crown
# From on_create_title: a merchant republic creates a kingdom (a Tilean
# city, Marienburg, Sartosa). Scope: the doge; FROM is the new title.
narrative_event = {
	id = realm.10
	title = EVTTITLErealm10
	desc = EVTDESCrealm10
	picture = GFX_evt_busy_trading_dock_republic
	border = GFX_event_narrative_frame_economy

	is_triggered_only = yes

	trigger = {
		is_merchant_republic = yes
		FROM = { tier = KING }
	}

	option = {
		name = EVTOPTArealm10
		prestige = 250
	}
}
