###############################################################################
# Realm events
# Namespace: realm (was HFP.40000 - HFP.40038, HFP.24120, HFP.99999 in
#   HFP_realm_events)
# Faction: the cult of Sigmar
#
# When the Grand Theogonist loses the last of the temple's lands in a war,
# every Sigmarite ruler hears of it (at most once a decade).
#
# The Grand Theogonist cast out   realm.1 - realm.2
#
# Saved: title flag gt_cast_out on k_cult_of_sigmar; event target
#   grand_theogonist.
#
# Callers:
#   common/scripted_effects/00_scripted_effects.txt (landless_pope_effect:
#     1), run by the CBs when a war is lost.
#
# Removed: the vanilla title-formation news (Hansa, the Swiss, Antioch,
# Cyprus, Austria, the Eastern Marches, Byzantium, the Spanish and
# crusader orders, the Papal States and Romagna, the HRE stem duchies, the
# Slavic union) and the HRE elector setup. None of these titles exist in
# the mod; the Marienburg League (REP.50), the Elector Counts and the holy
# orders cover them.
###############################################################################

namespace = realm

###############################################################################
# The Grand Theogonist cast out
###############################################################################

# realm.1 The temple's last lands are lost (hidden)
# Scope: the Grand Theogonist, who holds no land and whose realm holds none.
character_event = {
	id = realm.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_landed_title = k_cult_of_sigmar
		k_cult_of_sigmar = { NOT = { had_title_flag = { flag = gt_cast_out days = 3650 } } }
	}

	immediate = {
		k_cult_of_sigmar = { clr_title_flag = gt_cast_out set_title_flag = gt_cast_out }
		save_event_target_as = grand_theogonist
		narrative_event = { id = realm.2 }
		any_playable_ruler = {
			limit = { religion = sigmarite NOT = { character = ROOT } }
			narrative_event = { id = realm.2 }
		}
	}
}

# realm.2 The Grand Theogonist cast out (news)
# Scope: a Sigmarite ruler, or the Grand Theogonist.
narrative_event = {
	id = realm.2
	title = EVTTITLErealm2
	desc = { trigger = { character = event_target:grand_theogonist } text = EVTDESCrealm2_gt }
	desc = {
		trigger = { NOT = { character = event_target:grand_theogonist } }
		text = EVTDESCrealm2
	}
	picture = GFX_evt_cathedral
	border = GFX_event_narrative_frame_religion

	is_triggered_only = yes

	option = { #The Grand Theogonist
		name = EVTOPTArealm2
		trigger = { character = event_target:grand_theogonist }
		piety = -100
	}
	option = { #A zealot
		name = EVTOPTBrealm2
		trigger = { NOT = { character = event_target:grand_theogonist } trait = zealous }
		piety = 25
	}
	option = { #Anyone else
		name = EVTOPTCrealm2
		trigger = { NOT = { character = event_target:grand_theogonist } NOT = { trait = zealous } }
	}
}
