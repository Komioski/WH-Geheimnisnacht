###############################################################################
# Reavers' loot
# Namespace: reaver (was TOG.3300 - TOG.3404 in oldgods_various_events.txt)
# Faction: Norsca
# Serves: Norscan raiders of the old gods and the Dark Gods, and anyone of a
#   reaver bloodline
#
# A Norscan who loots a holding earns a name: a Reaver, then after fifty
# raids a Ravager, after a hundred a Norscan Warlord. One of another people
# with a reaver bloodline becomes a buccaneer instead. A looter's men may
# hand over a strange purple axe, or a chest of unknown metal that no one
# can open, until one day the key turns up in another holding: inside are
# the coins of the old priest-kings of Nehekhara, crumbling scrolls, junk, or
# a curse. A fleet home with loot fills the capital's markets.
#
# Loot                                 reaver.1 - reaver.4
# The reaver's name                    reaver.10 - reaver.13
# The fleet comes home                 reaver.20
#
# Fired from: on_loot_settlement (reaver.1 - reaver.3 and reaver.10 in the
#   random list, reaver.11 - reaver.13 always) and on_navy_returns_with_loot
#   (reaver.20) in 00_on_actions.txt.
# Saved: flags found_chest (the finder, for good), reaver_chest_* (cleared
#   by reaver.4), acquired_pirate_trait_by_bloodline; variable viking_raids
#   (raids counted, shared with WHG_doctrine_events.txt).
###############################################################################

namespace = reaver

###############################################################################
# Loot
###############################################################################

# reaver.1 The purple axe
# Scope: ROOT is the looter (players only).
character_event = {
	id = reaver.1
	desc = EVTDESCreaver1
	picture = GFX_evt_vikings_arriving_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes
	ai = no

	trigger = {
		NOT = { has_artifact = loot_axe }
	}

	option = {
		name = EVTOPTAreaver1
		add_artifact = loot_axe
		new_artifact = {
			set_creation_date = 1.1.1
		}
	}
	option = {
		name = EVTOPTBreaver1
		prestige = 10
	}
}

# reaver.2 The strange chest
# Scope: ROOT is the looter (players only).
character_event = {
	id = reaver.2
	desc = EVTDESCreaver2
	picture = GFX_evt_vikings_arriving_oldgods
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes
	ai = no

	trigger = {
		NOT = { has_artifact = strange_chest }
		NOT = { has_character_flag = found_chest }
	}

	option = {
		name = EVTOPTAreaver2
		custom_tooltip = { text = found_chest }
		add_artifact = strange_chest
		set_character_flag = found_chest
	}
}

# reaver.3 The key
# For the one who found the chest, or an heir who inherited it once its
# finder is gone. What is inside is reaver.4.
# Scope: ROOT is the looter (players only), holding the chest.
character_event = {
	id = reaver.3
	desc = {
		trigger = { has_character_flag = found_chest }
		text = EVTDESCreaver3_finder
	}
	desc = {
		trigger = { NOT = { has_character_flag = found_chest } }
		text = EVTDESCreaver3_heir
	}
	picture = GFX_evt_vikings_arriving_oldgods
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes
	ai = no

	trigger = {
		has_artifact = strange_chest
		OR = {
			has_character_flag = found_chest
			NOT = { any_dynasty_member = { has_character_flag = found_chest } }
		}
	}

	option = {
		name = EVTOPTAreaver3
		custom_tooltip = {
			text = opens_chest
			hidden_effect = {
				random_list = {
					40 = { set_character_flag = reaver_chest_gold }
					40 = { set_character_flag = reaver_chest_scrolls }
					40 = { set_character_flag = reaver_chest_junk }
					5 = { set_character_flag = reaver_chest_curse }
				}
				character_event = { id = reaver.4 }
			}
		}
	}
}

# reaver.4 The chest is opened
# Scope: ROOT is the looter (flag reaver_chest_*).
character_event = {
	id = reaver.4
	desc = {
		trigger = { has_character_flag = reaver_chest_gold }
		text = EVTDESCreaver4_gold
	}
	desc = {
		trigger = { has_character_flag = reaver_chest_scrolls }
		text = EVTDESCreaver4_scrolls
	}
	desc = {
		trigger = { has_character_flag = reaver_chest_junk }
		text = EVTDESCreaver4_junk
	}
	desc = {
		trigger = { has_character_flag = reaver_chest_curse }
		text = EVTDESCreaver4_curse
	}
	picture = {
		trigger = { NOT = { has_character_flag = reaver_chest_curse } }
		picture = GFX_evt_viking_throneroom_oldgods
	}
	picture = {
		trigger = { has_character_flag = reaver_chest_curse }
		picture = GFX_evt_shadowy_cabal
	}
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes

	option = {
		name = EVTOPTAreaver4_gold
		trigger = { has_character_flag = reaver_chest_gold }
		wealth = 200
		destroy_artifact = strange_chest
	}
	option = {
		name = EVTOPTAreaver4_scrolls
		trigger = { has_character_flag = reaver_chest_scrolls }
		custom_tooltip = { text = chest_empty }
		destroy_artifact = strange_chest
	}
	option = {
		name = EVTOPTAreaver4_junk
		trigger = { has_character_flag = reaver_chest_junk }
		wealth = 10
		destroy_artifact = strange_chest
	}
	option = {
		name = EVTOPTAreaver4_curse
		trigger = { has_character_flag = reaver_chest_curse }
		custom_tooltip = { text = seraphims }
		destroy_artifact = strange_chest
	}

	after = {
		clr_character_flag = reaver_chest_gold
		clr_character_flag = reaver_chest_scrolls
		clr_character_flag = reaver_chest_junk
		clr_character_flag = reaver_chest_curse
	}
}

# reaver.10 A Reaver
# A Norscan of the old gods or the Dark Gods becomes a Reaver; anyone else
# with a reaver bloodline a buccaneer. Raids are counted from here.
# Scope: ROOT is the looter.
character_event = {
	id = reaver.10
	desc = EVTDESCreaver10
	picture = GFX_evt_vikings_arriving_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes
	hide_from = yes
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		NOR = {
			trait = viking
			trait = ravager
			trait = seaking
			trait = sea_queen
			trait = norscan_warlord
			trait = buccaneer
		}
		OR = {
			reaver_norscan_raider_trigger = yes
			any_owned_bloodline = { has_bloodline_flag = bloodline_viking }
		}
	}

	option = {
		name = EVTOPTAreaver10
		if = {
			limit = { reaver_norscan_raider_trigger = yes }
			add_trait = viking
		}
		else = {
			set_character_flag = acquired_pirate_trait_by_bloodline
			add_trait = buccaneer
		}
		hidden_effect = { set_variable = { which = viking_raids value = 0 } }
	}
}

# reaver.11 The Ravager
# After fifty raids.
# Scope: ROOT is the looter.
character_event = {
	id = reaver.11
	desc = EVTDESCreaver11
	picture = GFX_evt_vikings_arriving_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes
	hide_from = yes
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		trait = viking
		reaver_norscan_raider_trigger = yes
		NOR = {
			trait = ravager
			trait = seaking
			trait = sea_queen
			trait = norscan_warlord
		}
		check_variable = { which = viking_raids value = 50 }
	}

	option = {
		name = EVTOPTAreaver11
		remove_trait = viking
		add_trait = ravager
	}
}

# reaver.13 Another raid
# Hidden. Counts a raid.
# Scope: ROOT is the looter.

	id = reaver.13

	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			trait = viking

		}

# reaver.20 Loot in the markets
# Scope: ROOT is the navy's owner, FROM the province. From
#   on_navy_returns_with_loot.
character_event = {
	id = reaver.20
	desc = EVTDESCreaver20
	picture = GFX_evt_vikings_arriving_oldgods
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	trigger = {
		NOT = {
			capital_scope = { has_province_modifier = loot_prosperity_boost }
		}
	}

	option = {
		name = EXCELLENT
		capital_scope = {
			add_province_modifier = {
				name = loot_prosperity_boost
				duration = 1825
			}
			hidden_effect = {
				change_variable = { which = prosperity_value value = 12 }
			}
		}
	}
}
