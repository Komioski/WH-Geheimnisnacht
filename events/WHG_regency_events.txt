# Regency: the regent and the ruler (namespace regency)
# Namespace: regency (was 61000 - 61303 in regency_events.txt; regency.N =
#   old id minus 61000, and the embezzlement 61000 - 61004 is regency.1 - 5)
# Faction: Generic
# Serves: every realm with a child, incapable or captive ruler
#
# A regent rules for a ruler who cannot: a child, a madman, an invalid. Once
# a year (the regency_events list in on_yearly_pulse) a regent may embezzle,
# grab a holding, give the ruler's gold to charity, humiliate the ruler,
# have the ruler declared mad and locked in a tower, or try to have a young
# ruler quietly die on a roof, in a forest, in a carriage, at a guard's hands
# or under a dagger. A child ruler may push the regent down the stairs, a
# just vassal may come to the ruler's aid, a courtier may plead or duel for
# the regency, and a queen or queen mother may take it over.
#
#   regency.1 - 5      The regent embezzles
#   regency.100 - 110  A holding, the charity box, a better steward
#   regency.120 - 135  The humiliated ruler and the valiant vassal; the
#                      child ruler pushes the regent (regency.135)
#   regency.140 - 170  The ruler declared mad, locked away, freed, the
#                      throne stolen, the escape from the tower
#   regency.190 - 213  The regent tries to kill a young ruler: the roof,
#                      the forest, the carriage, the guard, the dagger
#   regency.215 - 236  A courtier pleads or duels for the regency (the
#                      duel ends in 01_scripted_effects_duels.txt,
#                      regency.229)
#   regency.237 - 240  The queen or the queen mother takes over (the queen
#                      mother, regency.238, was missing from the list)
#   regency.300 - 303  A titled regent lowers crown authority (only
#                      without Conclave; kept as a DLC fallback)
#
# Fired from: on_yearly_pulse, the regency_events list in 00_on_actions.txt
#   (weights 250 - 750 over a base of 10000; the events check their own
#   triggers, mostly has_minor_title = title_regent).
# Fixed: the `break = yes` after a saved invincible character (six deaths)
#   is an if/else now; the claimant, the queen and the queen mother are
#   checked with is_woman_trigger.
# Saved: character flags mean_regent, valiant_vassal, pushed_regent,
#   imprisoned_by_regent, declared_ruler_insane, regent_lowered_CA,
#   flag_courtier_challenging_regent and the duel flags.
# Loc: localisation/00_WHG_regency_events.csv (the vanilla texts).
###############################################################################

namespace = regency

# regency.1 Regent embezzles money
character_event = {
	id = regency.1
	desc = EVTDESCregency1

	picture = GFX_evt_council
	border = GFX_event_normal_frame_economy

	min_age = 16
	prisoner = no
	capable_only = yes
	is_triggered_only = yes

	trigger = {
		has_minor_title = title_regent
		independent = no
		liege = { wealth = 20 }
		NOT = { trait = charitable }
		NOT = { trait = honest }
	}

	weight_multiplier = {
		days = 1
		modifier = { factor = 4 trait = greedy }
		modifier = { factor = 4 trait = deceitful }
		modifier = { factor = 1.33 trait = envious }
		modifier = { factor = 1.33 trait = arbitrary }
		modifier = { factor = 0.5 trait = kind }
		modifier = { factor = 0.5 trait = just }
	}

	option = {
		name = EVTOPTAregency1 # Embezzle
		ai_chance = { factor = 100 }
		liege = { wealth = -20 }
		wealth = 20
		random_list = {
			50 = {
				# Got away with it!
			}
			40 = {
				liege = {
					character_event = { id = regency.2 days = 4 tooltip = EVTTOOLTIPregency2 }
				} # Liege suspects
				if = {
					limit = { NOT = { has_job_title = job_treasurer } }
					liege = {
						job_treasurer = {
							character_event = {
								id = regency.3
								days = 4
								tooltip = EVTTOOLTIPregency3
							}
						}
					} # The Steward suspects
				}
			}
			10 = {
				liege = {
					character_event = { id = regency.4 days = 4 tooltip = EVTTOOLTIPregency4 }
				} # Liege has proof
				if = {
					limit = { NOT = { has_job_title = job_treasurer } }
					liege = {
						job_treasurer = {
							character_event = {
								id = regency.5
								days = 4
								tooltip = EVTTOOLTIPregency5
							}
						}
					} # The Steward has proof
				}
			}
		}
	}

	option = {
		name = EVTOPTBregency1 # I dare not!
		ai_chance = { factor = 0 }
		piety = 1
	}
}

		modifier = { factor = 0.5 trait = arbitrary }
		modifier = { factor = 2.0 trait = charitable }

				if = { limit = { has_law = crown_authority_4 } add_law = crown_authority_3 }
				if = {

					add_law = imperial_crown_authority_0

					add_law = imperial_crown_authority_2
				}
				if = {
					limit = { has_law = imperial_crown_authority_4 }
					add_law = imperial_crown_authority_3
