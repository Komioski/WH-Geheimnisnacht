###############################################################################
# Part 1: Patrician feuds (namespace feud)
# Faction: Generic
# Serves: the patrician houses of the merchant republics (Marienburg,
#   Tilea, the Border Princes)
#
# An AI patrician house snubs a player house (a ball it is not invited to)
# and a feud begins: the_feuders on the AI dynasty, the_victims on the
# player's. From then on the AI houses keep needling the player house until
# the feud ends in a wedding, a double funeral or a mutiny at sea.
#
#   feud.100 - 105    The great ball, and the even greater ball in reply
#   feud.109          The feuding house hates the player house (opinions)
#   feud.150 - 155    The rival's new flagship, sabotaged or not; feud.152
#                     (the ship capsizes under the faith's head at the
#                     Marriage of the Sea) was never fired in vanilla and is
#                     now a rare outcome of the sabotage
#   feud.160 - 168    Star-crossed lovers: a wedding ends the feud, or both
#                     lovers take poison
#   feud.180 - 186    The smugglers, and a son held for ransom
#   feud.190 - 195    A battle of galleys, and a Sartosan pirate galley
#   feud.200 - 203    A feuding house rules the republic
#   feud.205 - 206    Riots in the streets
#
# feud_phase_two (set by the flagship and the smugglers) opens the lovers
# and the galley battle. The AI side runs on MTTH events behind the ai and
# is_patrician pre-triggers, so only AI patricians are checked.
# Fixed: the `break = yes` after a saved invincible character skipped the
#   rest of the option (the other lover's death, the tooltip); it is an
#   if/else now. The truce (feud.167) is fired straight at the dead lover's
#   father instead of a daily MTTH on every patrician, and its liege check
#   now looks at the liege. Lovers need a race with women.
# Saved: dynasty flags the_feuders, the_victims, feud_phase_two,
#   flagship_feud, romeo_juliet_feud, smugglers_feud, pirate_feud,
#   evil_ruler_feud, player_ruler_feud, feud_peace_1, feud_peace_2;
#   character flags the_grand_ball, shamed_at_ball, shamed_player_at_ball,
#   romeo, julia.
# Loc: localisation/00_WHG_republic_events.csv (the vanilla texts; the
#   Venetian bucentaur and the Pope moved into the setting). The tooltips
#   keep their vanilla keys.
###############################################################################

namespace = feud

# The Great Ball (AI)
character_event = {
	id = feud.100
	hide_window = yes

	min_age = 16
	ai = yes
	is_patrician = yes
	capable_only = yes
	prisoner = no

	trigger = {
		has_dlc = "The Republic"
		liege = { is_merchant_republic = yes }
		in_command = no
		OR = {
			liege = {
				is_patrician = yes
				ai = no
				age = 16
				in_command = no
				prisoner = no
				NOT = { trait = incapable }
				NOT = { reverse_opinion = { who = ROOT value = 0 } }
				NOT = { has_dynasty_flag = the_victims }
				NOT = { has_character_flag = the_grand_ball }
				NOT = { is_inaccessible_trigger = yes }
			}
			liege = {
				any_vassal = {
					is_patrician = yes
					ai = no
					age = 16
					in_command = no
					prisoner = no
					NOT = { reverse_opinion = { who = ROOT value = 0 } }
					NOT = { trait = incapable }
					NOT = { has_dynasty_flag = the_victims }
					NOT = { has_character_flag = the_grand_ball }
					NOT = { is_inaccessible_trigger = yes }
				}
			}
		}
		NOT = { has_dynasty_flag = the_feuders }
	}

	mean_time_to_happen = {
		months = 320
	}

	immediate = {
		if = {
			limit = {
				liege = {
					ai = no
					NOT = {
						reverse_opinion = { who = ROOT value = 0 }
						has_dynasty_flag = the_victims
						any_vassal = { ai = no }
					}
					NOT = { is_inaccessible_trigger = yes }
				}
			}
			liege = { narrative_event = { id = feud.101 } set_character_flag = the_grand_ball }
		}
		if = {
			limit = {
				liege = {
					ai = yes
					any_vassal = {
						is_patrician = yes
						ai = no
						NOT = { has_dynasty_flag = the_victims }
					}
				}
			}
			liege = {
				random_vassal = {
					limit = {
						is_patrician = yes
						ai = no
						age = 16
						in_command = no
						prisoner = no
						NOT = { reverse_opinion = { who = ROOT value = 0 } }
						NOT = { has_dynasty_flag = the_victims }
						NOT = { is_inaccessible_trigger = yes }
					}
					narrative_event = { id = feud.101 }
					set_character_flag = the_grand_ball
				}
			}
		}
	}
}

					reverse_has_opinion_modifier = { who = ROOT modifier = opinion_feuding_family }

			}
			liege = {
				any_vassal = {
					is_patrician = yes

		}
	}
