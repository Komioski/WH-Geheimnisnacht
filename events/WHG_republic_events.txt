###############################################################################
# Part 1: Patrician feuds (namespace feud)
# Faction: Generic
# Serves: the patrician houses of the merchant republics (Marienburg,
#   Tilea, the Border Princes)
#
# An AI patrician house snubs a player house (a ball it is not invited to)
# and a feud begins: the_feuders on the AI dynasty, the_victims on the
# player's. From then on the AI houses keep needling the player house until
# the feud ends in a wedding, a double funeral or a mutiny at sea.
#
#   feud.100 - 105    The great ball, and the even greater ball in reply
#   feud.109          The feuding house hates the player house (opinions)
#   feud.150 - 155    The rival's new flagship, sabotaged or not; feud.152
#                     (the ship capsizes under the faith's head at the
#                     Marriage of the Sea) was never fired in vanilla and is
#                     now a rare outcome of the sabotage
#   feud.160 - 168    Star-crossed lovers: a wedding ends the feud, or both
#                     lovers take poison
#   feud.180 - 186    The smugglers, and a son held for ransom
#   feud.190 - 195    A battle of galleys, and a Sartosan pirate galley
#   feud.200 - 203    A feuding house rules the republic
#   feud.205 - 206    Riots in the streets
#
# feud_phase_two (set by the flagship and the smugglers) opens the lovers
# and the galley battle. The AI side runs on MTTH events behind the ai and
# is_patrician pre-triggers, so only AI patricians are checked.
# Fixed: the `break = yes` after a saved invincible character skipped the
#   rest of the option (the other lover's death, the tooltip); it is an
#   if/else now. The truce (feud.167) is fired straight at the dead lover's
#   father instead of a daily MTTH on every patrician, and its liege check
#   now looks at the liege. Lovers need a race with women.
# Saved: dynasty flags the_feuders, the_victims, feud_phase_two,
#   flagship_feud, romeo_juliet_feud, smugglers_feud, pirate_feud,
#   evil_ruler_feud, player_ruler_feud, feud_peace_1, feud_peace_2;
#   character flags the_grand_ball, shamed_at_ball, shamed_player_at_ball,
#   romeo, julia.
# Loc: localisation/00_WHG_republic_events.csv (the vanilla texts; the
#   Venetian bucentaur and the Pope moved into the setting). The tooltips
#   keep their vanilla keys.
###############################################################################

namespace = feud

# The Great Ball (AI)
character_event = {
	id = feud.100
	hide_window = yes

	min_age = 16
	ai = yes
	is_patrician = yes
	capable_only = yes
	prisoner = no

	trigger = {
		has_dlc = "The Republic"
		liege = { is_merchant_republic = yes }
		in_command = no
		OR = {
			liege = {
				is_patrician = yes
				ai = no
				age = 16
				in_command = no
				prisoner = no
				NOT = { trait = incapable }
				NOT = { reverse_opinion = { who = ROOT value = 0 } }
				NOT = { has_dynasty_flag = the_victims }
				NOT = { has_character_flag = the_grand_ball }
				NOT = { is_inaccessible_trigger = yes }
			}
			liege = {
				any_vassal = {
					is_patrician = yes
					ai = no
					age = 16
					in_command = no
					prisoner = no
					NOT = { reverse_opinion = { who = ROOT value = 0 } }
					NOT = { trait = incapable }
					NOT = { has_dynasty_flag = the_victims }
					NOT = { has_character_flag = the_grand_ball }
					NOT = { is_inaccessible_trigger = yes }
				}
			}
		}
		NOT = { has_dynasty_flag = the_feuders }
	}

	mean_time_to_happen = {
		months = 320
	}

	immediate = {
		if = {
			limit = {
				liege = {
					ai = no
					NOT = {
						reverse_opinion = { who = ROOT value = 0 }
						has_dynasty_flag = the_victims
						any_vassal = { ai = no }
					}
					NOT = { is_inaccessible_trigger = yes }
				}
			}
			liege = { narrative_event = { id = feud.101 } set_character_flag = the_grand_ball }
		}
		if = {
			limit = {
				liege = {
					ai = yes
					any_vassal = {
						is_patrician = yes
						ai = no
						NOT = { has_dynasty_flag = the_victims }
					}
				}
			}
			liege = {
				random_vassal = {
					limit = {
						is_patrician = yes
						ai = no
						age = 16
						in_command = no
						prisoner = no
						NOT = { reverse_opinion = { who = ROOT value = 0 } }
						NOT = { has_dynasty_flag = the_victims }
						NOT = { is_inaccessible_trigger = yes }
					}
					narrative_event = { id = feud.101 }
					set_character_flag = the_grand_ball
				}
			}
		}
	}
}

			}
			liege = {
				any_vassal = {
					is_patrician = yes

	}

# Part 2: Trade of the great ports (namespace trade)
# Faction: Generic
# Serves: the patrician houses of the merchant republics (Marienburg,
#   Tilea, the Border Princes)
#
# The daily business of a patrician house: ships lost and found, rival
# houses fighting over trade posts, the head of the republic asked to share
# them out, foreign lords raising mobs against them, and the bachelors of
# the house marrying beneath (or above) them.
#
#   trade.1 - 20      Shipping news: storms, corsairs, spices from Araby,
#                     strikes and fires. The player's own patrician house
#                     gets MTTH events (ai = no, is_patrician); the hidden
#                     AI ones (1, 4, 13, 19) send a rival house's news.
#   trade.300 - 313   Seize a trade post (plot_seize_trade_post in
#                     plot_decisions.txt): bribes, counter-bribes, or a
#                     petition to the head of the republic
#   trade.399 - 413   A bachelor of the house marries: a burgher's daughter,
#                     a bar wench, a peasant beauty, a rich heiress, a young
#                     serving girl, an older widow, or a woman from Araby or
#                     the Southlands
#   trade.500 - 514   Trade posts handed from house to house by the head of
#                     the republic (realm_decisions.txt)
#   trade.599 - 608   A foreign lord raises a mob against a trade post in
#                     the realm, and the owning house bribes or talks it
#                     down
#
# Perf: the bachelors and the trade post mobs were MTTH events on every
#   unmarried AI man in the world and on every province with a trade post.
#   They now hang off two MTTH hubs behind the is_patrician pre-trigger
#   (trade.399 picks one bachelor, trade.599 one trade post), and the
#   shipping events check has_dlc as a pre-trigger.
# Fixed: the foreign bride came from the real-world Mali (culture manden);
#   she now comes from Araby (medean, ramaccan) or the Southlands (bayaji).
#   Bachelors need a race with women.
# New: the sunken treasure (trade.16) may, rarely, hold a pearl collar
#   (collar_of_pearls: 10%, once a lifetime, only without one).
# Callers: decisions/plot_decisions.txt (trade.300),
#   decisions/realm_decisions.txt (trade.500, trade.510).
# Shared triggers: common/scripted_triggers/WHG_republic_triggers.txt
# Saved: character flags getting_married, plot_seize_trade_post_decision_
#   taken, poor_patrician, target_patrician, target_patrician_doge,
#   trade_found_pearls.
# Loc: localisation/00_WHG_republic_events.csv (the vanilla texts, with
#   Sartosan corsairs, spices from Araby and two quotes from other worlds
#   taken out). The custom tooltips keep their vanilla keys.
###############################################################################

	id = trade.500

	is_triggered_only = yes

					is_patrician = yes
					NOT = { character = FROM }
					num_of_trade_post_diff = { character = FROM value = 10 }
				}
			}
			random_vassal = {
				limit = {
					is_patrician = yes
					NOT = { character = FROM }
					num_of_trade_post_diff = { character = FROM value = 10 }
				}
				set_character_flag = target_patrician
			}
		}
		if = {
			limit = {
				FROM = { NOT = { num_of_trade_posts = 1 } }
				any_vassal = { is_patrician = yes num_of_trade_posts = 10 }
			}
			random_vassal = {
				limit = { is_patrician = yes num_of_trade_posts = 10 }
				set_character_flag = target_patrician
			}
		}
		FROM = { set_character_flag = poor_patrician }
	}

	option = {
		name = EVTOPTAtrade500
		FROM = { letter_event = { id = trade.501 } clr_character_flag = poor_patrician }
		hidden_tooltip = {
			any_vassal = {
				limit = { has_character_flag = target_patrician }
				clr_character_flag = target_patrician
			}
		}
	}
	option = {
		name = EVTOPTBtrade500
		ai_chance = { factor = 100 }
		any_vassal = {
			limit = { has_character_flag = target_patrician }

# Pigbarter falls (hidden): the houses go home
# Fired from on_new_holder and on_new_holder_usurpation.
# ROOT = new holder, FROM = title, FROMFROM = old holder.
# The house members who aren't landed leave for their house's home:
# - House of Yabbas (Arabyan): Copher
# - Westerlander house: Marienburg (the League if it exists)
# - Cathayan house: back along the eastern trade roads to Cathay
# - Reman house: Remas
# - Strigany house: Sylvania
# If a home has no holder they go to another ruler of their people, and if
# there is none of those either, to another merchant republic.
character_event = {
	id = pigbarter.10
	hide_window = yes

	is_triggered_only = yes
	only_rulers = yes

	trigger = {
		FROM = { title = c_pigbarter }
		has_global_flag = pigbarter_compact
		NOT = { has_global_flag = pigbarter_fallen }
		NOR = {
			is_merchant_republic = yes
			is_patrician = yes
			any_liege = { is_merchant_republic = yes }
		}
	}

	immediate = {
		set_global_flag = pigbarter_fallen

		# House of Yabbas: Copher
		c_copher = { holder_scope = { save_event_target_as = pigbarter_home_araby } }
		if = {
			limit = { NOT = { event_target:pigbarter_home_araby = { always = yes } } }
			random_independent_ruler = {
				limit = { culture_group = arabyan_group }
				save_event_target_as = pigbarter_home_araby
			}
		}
		# Westerlander house: Marienburg
		if = {
			limit = { k_marienburg_league = { has_holder = yes } }
			k_marienburg_league = {
				holder_scope = { save_event_target_as = pigbarter_home_westerland }
			}
		}
		else = {
			c_marienburg = { holder_scope = { save_event_target_as = pigbarter_home_westerland } }
		}
		if = {
			limit = { NOT = { event_target:pigbarter_home_westerland = { always = yes } } }
			random_independent_ruler = {
				limit = { culture_group = imperial_group }
				save_event_target_as = pigbarter_home_westerland
			}
		}
		# Cathayan house: a Cathayan colony on the map (Naparaya first), so the
		# house stays in play; only if none is left does it go home to Cathay.
		if = {
			limit = { any_independent_ruler = { cathay_colony_ruler_trigger = yes } }
			random_independent_ruler = {
				limit = { cathay_colony_ruler_trigger = yes }
				preferred_limit = { has_landed_title = k_naparaya }
				save_event_target_as = pigbarter_home_cathay
			}
		}
		else_if = {
			limit = {
				any_independent_ruler = {
					culture_group = oriental_group
					NOT = { has_landed_title = e_cathay_west_governor }
				}
			}
			random_independent_ruler = {
				limit = {
					culture_group = oriental_group
					NOT = { has_landed_title = e_cathay_west_governor }
				}
				save_event_target_as = pigbarter_home_cathay
			}
		}
		else = { offmap_cathay = { governor = { save_event_target_as = pigbarter_home_cathay } } }
		# Reman house: Remas
		c_remas = { holder_scope = { save_event_target_as = pigbarter_home_tilea } }
		if = {
			limit = { NOT = { event_target:pigbarter_home_tilea = { always = yes } } }
			random_independent_ruler = {
				limit = { culture_group = tilean_group }
				save_event_target_as = pigbarter_home_tilea
			}
		}
		# Strigany house: Sylvania
		k_sylvania = { holder_scope = { save_event_target_as = pigbarter_home_strigany } }
		if = {
			limit = { NOT = { event_target:pigbarter_home_strigany = { always = yes } } }
			random_independent_ruler = {
				limit = {
					OR = { culture_group = strigany_group religion_group = necromantic_group }
				}
				save_event_target_as = pigbarter_home_strigany
			}
		}
		# Fallback: another merchant republic
		random_independent_ruler = {

# Part 5: Republic elections (namespace election)
# Faction: Generic
# Serves: the doge and patricians of a merchant republic whose title has an
#   election term law (law_election_cycles_0 - 5 in
#   common/laws/wh_republic_laws.txt)
#
# The term laws set how long a doge holds office: 4, 8 or 16 years for the
# short-lived races, 10, 15 or 20 for the immortal ones (Dictatorship and the
# Drachau of the Druchii hold no elections). When the term runs out the doge
# must stand again against the heads of the patrician houses. For three
# months every candidate buys, speaks and schemes for the guilds' favour.
# Each step of favour is 50 prestige, counted in the variable
# election_prestige so that it can be taken back once the vote is over.
#
# On the day of the count the doge's prestige (with the seat's patronage,
# 100 prestige) is set against every rival's. If a rival stands higher, the
# doge lays down the seal: he abdicates, and the patrician election of the
# republic chooses the successor as it does on a doge's death, which
# rewards the same prestige. When the count is over the bought favour fades.
# A doge left with negative prestige by then is impeached for buying the
# vote and must abdicate too.
#
#   election.1         The term runs out: the election is called (hidden)
#   election.10        Campaign funds
#   election.20        Speeches before the guilds
#   election.30 - 31   Smears and bribes, and the smeared rival hears of it
#   election.40        The bribes come to light
#   election.50 - 51   The count: bribes exposed, then the tally (hidden)
#   election.52        The doge loses and lays down the seal
#   election.55        The new doge takes office (hidden)
#   election.60        The result
#   election.65        The new doge is impeached
#
# Rebuilt from the unfinished campaign (REP.80 - 90 in
# wh_republic_events.txt), which was never switched on and leaned on a
# hold_election effect that CK2 does not have. Its funds, speech and smear
# events and the design notes in the term laws (temporary prestige, impeached
# for voting fraud with negative prestige afterwards) are kept.
# Callers: on_yearly_pulse (1) in common/on_actions/wh_on_actions.txt.
# Saved: character flags election_term_set, election_running,
#   election_candidate, election_bribed, election_smeared,
#   election_fraud_check; character modifier election_timer; variables
#   election_prestige, election_votes; event targets election_doge,
#   election_title, election_leader, election_rival, election_new_doge.
# Effects and triggers: common/scripted_effects/WHG_republic_effects.txt,
#   common/scripted_triggers/WHG_republic_triggers.txt.
# Loc: localisation/00_WHG_republic_events.csv; election_timer keeps its key.
###############################################################################

namespace = election

# election.1 The term runs out
# A doge whose republic holds elections starts his term the first time the
# pulse finds him (any leftovers of an election that died with his
# predecessor are cleared then). When the term's timer has run out, the
# election is called: the doge and every patrician house under him stand,
# and the campaign events follow over three months. With no patrician to
# stand against him, the doge is simply confirmed for another term.
# Scope: ROOT is the doge.
character_event = {
	id = election.1
	hide_window = yes

	is_triggered_only = yes # on_yearly_pulse

	only_rulers = yes
	prisoner = no

	trigger = {
		is_merchant_republic = yes
		republic_election_law_trigger = yes
		NOT = { has_character_modifier = election_timer }
		NOT = { has_character_flag = election_running }
	}

	immediate = {
		if = {
			limit = { NOT = { has_character_flag = election_term_set } }
			republic_election_clear_effect = yes
			any_vassal = {
				limit = { has_character_flag = election_candidate }
				republic_election_clear_effect = yes
			}
			republic_election_term_effect = yes
		}
		else_if = {
			limit = { NOT = { any_vassal = { republic_election_candidate_trigger = yes } } }
			republic_election_term_effect = yes
		}
		else = {
			set_character_flag = election_running
			save_event_target_as = election_doge
			primary_title = { save_event_target_as = election_title }
			republic_election_clear_effect = yes
			set_character_flag = election_candidate
			character_event = { id = election.10 days = 3 }
			character_event = { id = election.20 days = 30 random = 10 }
			character_event = { id = election.30 days = 55 random = 10 }
			character_event = { id = election.50 days = 90 }
			any_vassal = {
				limit = { republic_election_candidate_trigger = yes }
				republic_election_clear_effect = yes
				set_character_flag = election_candidate
				character_event = { id = election.10 days = 3 random = 5 }
				character_event = { id = election.20 days = 30 random = 10 }
				character_event = { id = election.30 days = 55 random = 10 }
			}
		}
	}
}

# election.50 The count
# Each candidate who bought guild votes is found out three times in ten
# (election.40); the tally follows a day later.
# Scope: ROOT is the doge.
character_event = {
	id = election.50
	hide_window = yes

	is_triggered_only = yes # election.1

	has_character_flag = election_running

	immediate = {
		if = {
			limit = { has_character_flag = election_bribed }
			random = { chance = 30 character_event = { id = election.40 } }
		}
		any_vassal = {
			limit = { has_character_flag = election_candidate has_character_flag = election_bribed }
			random = { chance = 30 character_event = { id = election.40 } }
		}
		character_event = { id = election.51 days = 1 }
	}
}

# election.51 The tally
# A day after the bribes are known, sets the doge's prestige (with the seat's
# 100) against each rival's and saves the strongest as election_leader. The
# doge keeps the seal unless a rival stands strictly higher: a re-elected
# doge starts a new term and everyone hears the result; a beaten doge gets
# election.52.
# Scope: ROOT is the doge.
character_event = {
	id = election.51
	hide_window = yes

	is_triggered_only = yes # election.50

	has_character_flag = election_running

	immediate = {
		save_event_target_as = election_leader
		export_to_variable = { which = election_votes value = prestige }
		change_variable = { which = election_votes value = 100 }
		any_vassal = {
			limit = { has_character_flag = election_candidate }
			export_to_variable = { which = election_votes value = prestige }
			if = {
				limit = {
					event_target:election_leader = {
						NOT = {
							check_variable = {
								which = election_votes
								which = election_votes
								which = PREV
							}
						}
					}
				}
				save_event_target_as = election_leader
			}
		}
		if = {
			limit = { event_target:election_leader = { character = ROOT } }
			save_event_target_as = election_new_doge
			republic_election_term_effect = yes
			republic_election_end_effect = yes
		}
		else = { character_event = { id = election.52 } }
	}
}
