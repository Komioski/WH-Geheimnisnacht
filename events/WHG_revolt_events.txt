###############################################################################
# Revolts
# Namespace: revolt (was TOG.1000 - TOG.1191 in rebel_events_the_old_gods.txt,
#   HL.2000 - HL.2011 in rebel_events_horse_lords.txt and 80000 - 80062 in
#   rebel_events.txt)
# Faction: Generic
# Serves: every realm
#
# With The Old Gods the counties of a realm rise on their own (the old rebel
# system is off): peasants under a former soldier, zealots of a heresy under a
# charismatic preacher, the landholders of a lost kingdom, or the faithful of
# a faith the top liege does not share under a militant priest. Each rising
# can be reinforced while it lasts. Rebels who take a county seat may turn the
# county to their faith or culture and leave unrest behind, and the mob may
# kill someone who lived there. The undead and greenskins never rise as
# peasants: instead a rival may claim the county, a tomb prince or an upstart
# boss (revolt.50). Lizardmen are built to obey; where beastmen or the jungle
# already trouble a county, that trouble comes instead of a revolt. Under a nomad lord a lesser clan or
# herd may rise to demand a place among the great ones, its chieftain often
# sworn to a Dark God. Without The Old Gods the old rebel system runs
# instead: rebels who hold a county burn it, fall on those who live there and
# may hand it to another ruler.
#
# Peasants                             revolt.1 - revolt.5
# Zealots of a heresy                  revolt.10 - revolt.14
# A lost kingdom                       revolt.20 - revolt.24
# The faithful of another faith        revolt.30 - revolt.34
# Minor clans                          revolt.40 - revolt.44
# A rival rises (undead, greenskins)   revolt.50 - revolt.51
# The mob                              revolt.90 - revolt.94
# Without The Old Gods                 revolt.100 - revolt.122
#
# Fired from: on_rebel_revolt (risings and reinforcements),
#   on_siege_over_winner (county seats taken) and on_siege_over_loc_chars
#   (revolt.90; revolt.92 from on_siege_over_winner) in WHG_war_on_actions.txt;
#   the old rebel system (revolt.100, revolt.110, revolt.120) runs on mean
#   time to happen, behind
#   lacks_dlc = "The Old Gods"; revolt.1 also from WHG_chaos_cult_events,
#   WHG_chaos_horde_events, WHG_vampire_events and WHG_skaven_events (flag
#   force_peasant_revolt). Shared triggers are in WHG_old_gods_triggers.txt.
#   The scripted effects that raise the armies (spawn_*_army_effect,
#   spawn_*_reinforcements_effect) are in 00_scripted_effects.txt; those of
#   the minor clans (revolt_clan_*_effect) in WHG_old_gods_effects.txt.
# Saved: flags heretic_revolter, religious_revolter (on the leaders, set by
#   the effects), force_peasant_revolt (province, set by the callers);
#   event targets minor_clan_revolt_leader, province_top_liege,
#   revolt_burned_lord, revolt_burned_holding, revolt_defected_from; flags
#   revolt_band_* (cleared by revolt_clan_bands_effect), revolt_caught_*
#   (cleared by revolt.94 and revolt.110); global flag
#   possible_province_defection (cleared by revolt.121).
###############################################################################

# revolt.1 The peasants rise
# A former soldier leads the peasants of the county against the top liege,
# under a temporary rebel title, with an army scaled by the revolt strength
# rule. Every liege of the owner hears of it (revolt.2). Heretic, religious
# and national revolts come first where they fit; undead and lizardmen never
# rise, and greenskin, beastman or jungle unrest takes the place of a revolt.
# Five years without another rising in the county.
# Scope: ROOT is the county (province). From on_rebel_revolt, and forced
#   (flag force_peasant_revolt) by the Chaos cults, the Chaos hordes, the
#   Skaven and the vampires.
province_event = {
	id = revolt.1
	desc = EVTDESCrevolt1
	picture = GFX_evt_peasants
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	trigger = {
		OR = {
			has_dlc = "The Old Gods"
			has_province_flag = force_peasant_revolt
		}
		revolt_county_can_rise_trigger = yes

		# Will get clan rebels instead
		NAND = {
			has_empty_holding = yes
			owner = {
				is_nomadic = yes
			}
		}

		OR={
			has_province_flag = force_peasant_revolt
			any_province_holding = {NOT = { holding_type = nomad }}
		}

		# Heretic revolts have prio
		OR = {
			is_heretic = no
			owner = {
				top_liege = {
					NOT = { is_parent_religion = ROOT }
				}
			}
			NOT = { has_province_modifier = heretic_stronghold }
			NOT = { has_province_modifier = fanatic_uprising }
			has_province_flag = force_peasant_revolt
		}

		# Religious revolts have prio
		OR = {
			owner = {
				top_liege = {
					religion_group = ROOT
				}
			}
			has_province_flag = force_peasant_revolt
		}

		# Nationalist revolts have prio
		OR = {
			owner = {
				top_liege = {
					culture = ROOT
				}
			}
			kingdom = {
				OR = {
					has_holder = yes
					NOT = { culture = ROOT }
					NOT = {
						# The de jure kingdom must have had an appropriate holder in the past
						any_previous_holder = {
							ROOT = {
								owner = {
									top_liege = {
										NOT = { dynasty = PREVPREVPREV }
										OR = {
											is_nomadic = no
											NOT = { culture_group = ROOT }
										}
									}
								}
							}
						}
					}
				}
			}
			has_province_flag = force_peasant_revolt
		}

		OR = {
			has_province_flag = force_peasant_revolt
			NOT = { # Not if there is already an ongoing peasant revolt for this province
				owner = {
					top_liege = {
						war = yes

						any_war = {
							defender = { character = PREV }
							OR = {
								AND = {
									using_cb = peasant_revolt
									war_title = ROOT # The county
								}
								AND = {
									using_cb = heretic_revolt
									attacker = { religion = ROOT }
								}
								AND = {
									using_cb = religious_revolt
									attacker = { religion = ROOT }
								}
								AND = {
									using_cb = liberation_revolt
									thirdparty_title_scope = {
										ROOT = {
											kingdom = {
												title = PREVPREV
											}
										}
									}
								}
							}
						}
					}
				}
			}
		}
	}

	immediate = {
		add_province_modifier = {
			name = recent_county_uprising
			duration = 1825 # Five years of -100% revolt risk in this county
		}
		remove_province_modifier = rumors_of_witchcraft

		create_character = {
			random_traits = yes
			dynasty = none
			religion = ROOT
			culture = ROOT
			female = no
			age = 25
			attributes = {
				martial = 5
			}
			trait = peasant_leader
			trait = tough_soldier
		}

		new_character = {
			# Set Racial Trait
			character_event = {
			  id = aiclean.31
			}
			create_title = {
				tier = DUKE
				landless = yes
				temporary = yes
				rebel = yes
				culture = ROOT
				name = "PEASANT_REVOLT"
				holder = THIS
			}
			set_amazon_male_to_female = yes
			wealth = 100

			spawn_peasant_army_effect = yes

			if = {
				limit = {
					has_game_rule = {
						name = provincial_revolt_strength
						value = powerful
					}
				}
				spawn_peasant_army_effect = yes
				wealth = 150
			}

			if = {
				limit = {
					has_game_rule = {
						name = provincial_revolt_strength
						value = very_powerful
					}
				}
				spawn_peasant_army_effect = yes
				spawn_peasant_army_effect = yes
				spawn_peasant_army_effect = yes
				wealth = 250
			}

			if = {
				limit = {
					has_game_rule = {
						name = provincial_revolt_strength
						value = extremely_powerful
					}
				}
				spawn_peasant_army_effect = yes
				spawn_peasant_army_effect = yes
				spawn_peasant_army_effect = yes
				spawn_peasant_army_effect = yes
				spawn_peasant_army_effect = yes
				wealth = 350
			}

			# DoW on the province top liege
			ROOT = {
				owner = {
					top_liege = {
						reverse_war = {
							target = PREVPREVPREV
							casus_belli = peasant_revolt
							thirdparty_title = ROOT # The county
						}
						reverse_opinion = {
							who = PREVPREVPREV
							modifier = opinion_evil_tyrant
						}
					}
				}
			}
		}

		owner = {
			any_liege = { # Inform the lieges
				character_event = {
					id = revolt.2
				}
			}
		}
	}

	option = {
		name = EVTOPTArevolt1
	}
}

		trigger = { has_character_flag = revolt_caught_wounded }
		text = EVTDESCrevolt90_wounded
	}
	desc = {
		trigger = { has_character_flag = revolt_caught_survived }
		text = EVTDESCrevolt90_survive
	}

	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	trigger = {
		new_character = {
			rebel = yes

			OR = {
				NOT = { religion = ROOT }
				AND = {
					NOT = { has_character_flag = heretic_revolter }
					NOT = { has_character_flag = religious_revolter }
					NOT = { has_character_flag = shepherds_crusade_leader }
				}
			}
		}
	}

		}

			limit = { is_invincible_trigger = yes }
			set_character_flag = revolt_caught_survived
		}
		else = {
			random_list = {
				50 = { set_character_flag = revolt_caught_killed }
				25 = {
					trigger = { is_maimed_trigger = no }
					set_character_flag = revolt_caught_maimed
				}
				25 = {
					trigger = { NOT = { trait = wounded } }
					set_character_flag = revolt_caught_wounded
				}

				character_event = { id = revolt.91 }
			}

			trigger = { has_character_flag = revolt_caught_killed }

		}

			limit = { has_character_flag = revolt_caught_killed }

		}
		else_if = {
			limit = { has_character_flag = revolt_caught_maimed }
			add_maimed_trait_effect = yes
		}
		else_if = {
			limit = { has_character_flag = revolt_caught_wounded }
			add_trait = wounded
