# Way of Life: the rulership focus
# Namespace: rulership (was WoL.12000 - WoL.12507 in wol_rulership_events.txt)
# Faction: Generic
#
# A ruler with the rulership focus sits in judgement: murders, witches,
# feuding vassals, burnt fields, drunken lords and robbed travellers, and
# perhaps a castle from a travelling architect. Each judgement may teach the
# ruler stewardship, justice, diligence or ambition (rulership.1). Courts of
# law are for peoples that keep them: greenskins, the Skaven and the ogre
# tribes do not hear peasants (rulership_court_of_law_trigger). Witch trials
# happen in the Old World, the Moot and Araby, never in Kislev, where the
# ice witches are honoured (rulership_witch_trial_faith_trigger).
#
# Lessons of the court                 rulership.1 - rulership.6
# Reputation                           rulership.10 - rulership.11
# The architect                        rulership.20 - rulership.29
# The murders                          rulership.30 - rulership.37
# The weight of the crown              rulership.40 - rulership.41
# Disputes between vassals             rulership.50 - rulership.55
# The witch                            rulership.60
#
# Fired from: on_focus_pulse in 00_on_actions.txt (rulership.20, 30, 40, 41,
#   50, 52, 53, 54, 60) and on_yearly_pulse (rulership.10, 11). Shared
#   triggers and effects are in WHG_focus_triggers.txt and
#   WHG_focus_effects.txt.
###############################################################################

# rulership.1 Lessons of the court
# Hidden. After every judgement each lesson has its own 85% chance: a step
# of stewardship education, a step towards the administrator trait, and
# more justice, diligence and ambition (rulership.2 - rulership.6).
# Scope: ROOT is the ruler.
character_event = {
	id = rulership.1
	hide_window = yes

	is_triggered_only = yes # every judgement in this file

	immediate = {
		if = {
			limit = {
				OR = {
					trait = indulgent_wastrel
					trait = thrifty_clerk
					trait = fortune_builder
				}
				random = 85
			}
			character_event = { id = rulership.2 }
		}
		if = {
			limit = {
				OR = {
					NOT = { trait = administrator }
					religion = shallyan
				}
				NOT = { has_character_modifier = wol_raja_dharma }
				random = 85
			}
			character_event = { id = rulership.3 }
		}
		if = {
			limit = {
				NOT = { trait = just }
				random = 85
			}
			character_event = { id = rulership.4 }
		}
		if = {
			limit = {
				NOT = { trait = diligent }
				random = 85
			}
			character_event = { id = rulership.5 }
		}
		if = {
			limit = {
				NOT = { trait = ambitious }
				random = 85
			}
			character_event = { id = rulership.6 }
		}
	}
}

# rulership.2 A better steward
# The stewardship education trait goes up a step; the top step gives the
# Lawgiver nickname.
# Scope: ROOT is the ruler.
character_event = {
	id = rulership.2
	desc = EVTDESCrulership2
	picture = GFX_evt_relaxed_ruler
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # rulership.1

	hide_from = yes

	option = { # Excellent
		name = EXCELLENT
		if = {
			limit = { trait = indulgent_wastrel }
			hidden_tooltip = { remove_trait = indulgent_wastrel }
			add_trait = thrifty_clerk
		}
		else_if = {
			limit = { trait = thrifty_clerk }
			hidden_tooltip = { remove_trait = thrifty_clerk }
			add_trait = fortune_builder
		}
		else_if = {
			limit = { trait = fortune_builder }
			hidden_tooltip = { remove_trait = fortune_builder }
			add_trait = midas_touched
			if = {
				limit = { has_nickname = no }
				give_nickname = nick_the_lawgiver
			}
		}
	}
}

# rulership.3 The administrator's road
# Amateur Steward, then Steward, then the administrator trait (if no other
# lifestyle trait is held). A Shallyan administrator goes on to become
# Shallya's Almoner.
# Scope: ROOT is the ruler.
character_event = {
	id = rulership.3
	desc = EVTDESCrulership3
	picture = GFX_evt_relaxed_ruler
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # rulership.1

	hide_from = yes
	hide_new = yes

	trigger = {
		NOR = {
			AND = {
				lifestyle_traits = 1
				has_character_modifier = wol_steward_2
				NOT = { religion = shallyan }
			}
			has_character_modifier = wol_raja_dharma
		}
	}

	option = { # Excellent
		name = EXCELLENT
		if = {
			limit = {
				NOR = {
					has_character_modifier = wol_steward_1
					has_character_modifier = wol_steward_2
					trait = administrator
				}
			}
			add_character_modifier = { modifier = wol_steward_1 duration = -1 }
		}
		else_if = {
			limit = { has_character_modifier = wol_steward_1 }
			remove_character_modifier = wol_steward_1
			add_character_modifier = { modifier = wol_steward_2 duration = -1 }
		}
		else_if = {
			limit = {
				has_character_modifier = wol_steward_2
				NOT = { lifestyle_traits = 1 }
			}
			remove_character_modifier = wol_steward_2
			add_trait = administrator
		}
		else_if = {
			limit = {
				trait = administrator
				religion = shallyan
			}
			add_character_modifier = { modifier = wol_raja_dharma duration = -1 }
		}
	}
}

# rulership.20 An architect arrives
# Hidden, players only, once per life. A rich settled ruler with an empty
# holding slot in the demesne is offered a castle there. A treasurer with
# stewardship 5 can tell whether the architect is honest (3 in 4) or a
# charlatan; otherwise the ruler must guess (rulership.22).
# Scope: ROOT is the ruler.
character_event = {
	id = rulership.20
	hide_window = yes

	is_triggered_only = yes # on_focus_pulse

	only_playable = yes
	ai = no

	trigger = {
		has_focus = focus_rulership
		is_tribal = no
		is_nomadic = no
		wealth = 400
		NOT = { has_character_flag = architect_offer }
		NOT = { has_character_flag = building_castle }
		NOT = { has_character_modifier = in_seclusion }
		any_demesne_province = { has_empty_holding = yes }
	}

	immediate = {
		set_character_flag = architect_offer
		random_demesne_province = {
			limit = { has_empty_holding = yes }
			save_event_target_as = castle_site
		}
		if = {
			limit = { job_treasurer = { stewardship = 5 } }
			random_list = {
				75 = { set_character_flag = rulership_architect_trusted }
				25 = { set_character_flag = rulership_architect_charlatan }
			}
		}
		character_event = { id = rulership.22 }
	}
}

# rulership.22 The architect's proposal
# Pay 400 gold for a castle at event_target:castle_site, or send him away.
# What happens in 260 days depends on what the treasurer said: an honest
# architect finishes cheap or on time; a charlatan runs off or overspends;
# an unknown one may do anything (rulership.29).
# Scope: ROOT is the ruler; event_target:castle_site the province.
character_event = {
	id = rulership.22
	desc = { trigger = { has_character_flag = rulership_architect_trusted } text = EVTDESCrulership22b }
	desc = { trigger = { has_character_flag = rulership_architect_charlatan } text = EVTDESCrulership22c }
	desc = {
		trigger = {
			NOR = {
				has_character_flag = rulership_architect_trusted
				has_character_flag = rulership_architect_charlatan
			}
		}
		text = EVTDESCrulership22
	}
	picture = GFX_evt_throne_room
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # rulership.20

	option = { # Build the castle
		name = EVTOPTArulership22
		wealth = -400
		custom_tooltip = { text = castle_construction_begins }
		set_character_flag = building_castle
		event_target:castle_site = { set_province_flag = castle_province }
		hidden_tooltip = {
			if = {
				limit = { has_character_flag = rulership_architect_trusted }
				random_list = {
					60 = { set_character_flag = castle_outcome_cheap }
					40 = { set_character_flag = castle_outcome_on_time }
				}
			}
			else_if = {
				limit = { has_character_flag = rulership_architect_charlatan }
				random_list = {
					40 = { set_character_flag = castle_outcome_thief }
					40 = { set_character_flag = castle_outcome_costly }
					20 = { set_character_flag = castle_outcome_on_time }
				}
			}
			else = {
				random_list = {
					50 = { set_character_flag = castle_outcome_on_time }
					25 = { set_character_flag = castle_outcome_cheap }
					13 = { set_character_flag = castle_outcome_thief }
					12 = { set_character_flag = castle_outcome_costly }
				}
			}
			character_event = { id = rulership.29 days = 260 }
		}
	}
	option = { # Send him away
		name = EVTOPTBrulership22
		custom_tooltip = { text = architect_refused }
		clr_character_flag = rulership_architect_trusted
		clr_character_flag = rulership_architect_charlatan
	}
}

# rulership.29 The building season ends
# Hidden. The architect runs off (rulership.23), or, if the site is still in
# the demesne with an empty slot, the work ends as the outcome flag says
# (rulership.24 - rulership.27). A lost or filled site refunds the gold
# (rulership.28). Replaces the old daily check of the site.
# Scope: ROOT is the ruler.
character_event = {
	id = rulership.29
	hide_window = yes

	is_triggered_only = yes # rulership.22, rulership.24

	trigger = {
		has_character_flag = building_castle
	}

	immediate = {
		random_demesne_province = {
			limit = { has_province_flag = castle_province }
			save_event_target_as = castle_site
		}
		if = {
			limit = { has_character_flag = castle_outcome_thief }
			character_event = { id = rulership.23 }
		}
		else_if = {
			limit = {
				NOT = {
					any_demesne_province = {
						has_province_flag = castle_province
						has_empty_holding = yes
					}
				}
			}
			character_event = { id = rulership.28 }
		}
		else_if = {
			limit = { has_character_flag = castle_outcome_costly }
			character_event = { id = rulership.24 }
		}
		else_if = {
			limit = { has_character_flag = castle_outcome_cheap }
			character_event = { id = rulership.25 }
		}
		else_if = {
			limit = { has_character_flag = castle_outcome_delayed }
			character_event = { id = rulership.27 }
		}
		else = {
			character_event = { id = rulership.26 }
		}
	}
}

		has_focus = focus_rulership

		name = EVTOPTArulership30
		custom_tooltip = { text = peasant_hanged }
		hidden_tooltip = {
			character_event = { id = rulership.1 days = 1 }
			random = {
				chance = 50
				character_event = { id = rulership.31 days = 1825 random = 3650 }
