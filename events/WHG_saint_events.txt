###############################################################################
# Venerated Souls (saints) events
# Namespace: saint (was HF in HF_sainthood_events.txt, HF_saint_flavour_events,
#   HF_saintly_bloodlines_events and HF_saintly_pilgrimage_events)
# Faction: Old World faiths
#   (Sigmar, Ulric, Taal, Shallya, Manann, Morr, Myrmidia, Solkan and the
#   old faith; Kislev and Bretonnia's Lady have no Venerated Souls)
#
# In the Old World, men and women of rare piety are raised after death to
# Venerated Souls: lesser holy figures beneath their god. A local priest
# names the dead "Pious" first; years later the head of the cult (for the
# great cults, the Grand Conclave that Magnus the Pious founded) may name
# them a Venerated Soul. Their tomb draws pilgrims, their kin may carry a
# holy bloodline, and a corrupt priest can make a farce of it all.
#
# The Pious                               saint.1 - saint.3
# Raised to a Venerated Soul              saint.10 - saint.15
# The unworthy named Pious                saint.20 - saint.24
# The Grand Theogonist's favourite        saint.30 - saint.36
# Tombs of the Venerated                  saint.40 - saint.46
# Holy bloodlines                         saint.50 - saint.55
# Pilgrimage to a tomb                    saint.60 - saint.78
#
# Callers:
#   common/on_actions/WHG_character_on_actions.txt: on_death (1, 30), on_death random
#     list (20), on_siege_over_winner random list (46), on_yearly_pulse
#     random list (50).
#   common/on_actions/WHG_pulse_on_actions.txt: five-year list saint_tombs (40, 41,
#     42), five-year list saint_bloodlines (51, 53, 55).
#   common/cb_types/01_cb_types.txt: depose_impious_pope (35, 36).
#   the pilgrim part of this file: pilgrim.2 (60).
# Support: common/scripted_effects/WHG_faith_effects.txt,
#   common/scripted_triggers/WHG_faith_triggers.txt
# Loc: localisation/00_WHG_saint_events.csv
#
# Also in this file (each part keeps its own header and namespace):
#   Honoured Ancestors (namespace ancestor, was WHG_ancestor_events)
#   Pilgrimages (namespace pilgrim, was SoA in pilgrim_events.txt)
###############################################################################

namespace = saint

###############################################################################
# The Pious
# A devout lord dies (saint.1). If a priest of the same faith holds land in
# the realm, he names the dead "Pious" half a year to a year and a half
# later (saint.2). Once the Grand Theogonist has forbidden local namings
# (saint.23), only the head of the cult may do it.
###############################################################################

# saint.1 A devout death (hidden)
# Scope: ROOT is the dead character (on_death).
character_event = {
	id = saint.1
	hide_window = yes

	is_triggered_only = yes # on_death
	min_age = 16
	only_playable = yes
	religion_group = old_world_group

	trigger = {
		saint_worthy_trigger = yes
		OR = {
			AND = { tier = EMPEROR piety >= 1250 }
			AND = { tier = KING piety >= 1000 }
			AND = { tier = DUKE piety >= 750 }
			AND = { tier = COUNT piety >= 500 }
			has_character_modifier = true_christian_knight
			has_any_opinion_modifier = opinion_baptism_sponsor_target
		}
	}

	immediate = {
		save_event_target_as = saint_person
		top_liege = { save_event_target_as = saint_top_liege }
		capital_scope = { save_event_target_as = saint_capital }
		random_demesne_province = {
			limit = { saint_no_tomb_trigger = yes }
			save_event_target_as = random_saint_demesne_province
		}
		random_realm_province = {
			limit = { saint_no_tomb_trigger = yes }
			save_event_target_as = random_saint_realm_province
		}
		if = {
			limit = { ai = no }
			add_trait = i_used_to_be_a_player # Players wait less for the Conclave
		}
		random = {
			chance = 100
			modifier = {
				factor = 0.25
				any_owned_bloodline = { has_bloodline_flag = saintly_bloodline }
			}
			modifier = { factor = 0.8 trait = cynical }
			if = {
				limit = { has_global_flag = venerated_by_conclave_only religion = sigmarite }
				rightful_religious_head_scope = {
					character_event = { id = saint.2 days = 180 random = 360 }
				}
				set_character_flag = beatified
			}
			else_if = {
				limit = { any_realm_character = { saint_local_priest_trigger = yes } }
				random_realm_character = {
					limit = { saint_local_priest_trigger = yes }
					character_event = { id = saint.2 days = 180 random = 360 }
				}
				set_character_flag = beatified
			}
		}
		saint_canonization_roll_effect = yes
	}
}

# saint.2 A priest names the dead Pious (hidden)
# The priest (or the head of the cult) writes to the realm and the dead
# one's kin (saint.3).
# Scope: ROOT is the priest or the head of the cult (saint.1).
character_event = {
	id = saint.2
	hide_window = yes

	is_triggered_only = yes # saint.1

	trigger = {
		religion = event_target:saint_person
		NOT = { event_target:saint_person = { trait = beatified } }
		NOT = { event_target:saint_person = { trait = saint } }
	}

	immediate = {
		save_event_target_as = saint_proclaimer
		event_target:saint_person = {
			add_trait = beatified
			if = { limit = { is_woman_trigger = yes } set_special_character_title = BLESSED_TITLE_FEMALE }
			else = { set_special_character_title = BLESSED_TITLE }
		}
		saint_news_pious_effect = yes
	}
}

# saint.3 Named Pious
# Kin of the dead gain piety; when the head of the cult named them, his
# priests think better of the family too.
# Scope: ROOT is a ruler of the faith; FROM is the proclaimer (saint.2).
narrative_event = {
	id = saint.3
	title = EVTNAMEsaint3
	desc = { trigger = { FROM = { controls_religion = no } } text = EVTDESCsaint3 }
	desc = { trigger = { FROM = { controls_religion = yes } } text = EVTDESCsaint3b }
	picture = GFX_evt_beatification
	border = GFX_event_narrative_frame_religion
	portrait = event_target:saint_person

	is_triggered_only = yes # saint.2

	option = {
		name = EVTOPTAsaint3
		show_portrait = FROM
		if = {
			limit = { event_target:saint_person = { family = ROOT } }
			if = {
				limit = { FROM = { controls_religion = yes } }
				piety = 150
				FROM = { opinion = { modifier = opinion_family_saint who = ROOT years = 10 } }
			}
			else = { piety = 50 }
		}
	}
}

###############################################################################
# Raised to a Venerated Soul
# At death a roll decides whether the dead may one day be raised
# (saint_canonization_roll_effect). Years later, three to thirty, the head
# of the cult is asked (saint.11, saint.12). If he agrees, the dead becomes
# a Venerated Soul, a tomb is raised, and a holy bloodline may begin.
###############################################################################

# saint.11 The case comes before the cult (hidden)
# The head of the cult where the dead ruled hears it, or the dead's own.
# Scope: ROOT is a province of the dead's realm
# (saint_canonization_roll_effect).
province_event = {
	id = saint.11
	hide_window = yes

	is_triggered_only = yes # saint_canonization_roll_effect

	trigger = {
		event_target:saint_person = {
			OR = { trait = beatified has_religion_feature = religion_beatification }
		}
	}

	immediate = {
		if = {
			limit = {
				event_target:saint_capital = { owner = { religion = event_target:saint_person } }
			}
			event_target:saint_capital = {
				owner = { rightful_religious_head_scope = { character_event = { id = saint.12 } } }
			}
		}
		else = {
			event_target:saint_person = {
				rightful_religious_head_scope = { character_event = { id = saint.12 } }
			}
		}
	}
}

# saint.12 A Venerated Soul?
# The head of the cult decides. If he agrees: a miracle is told, the dead
# becomes a Venerated Soul, a tomb is raised, and a holy bloodline may
# begin. The AI always agrees.
# Scope: ROOT is the head of the cult (saint.11).
character_event = {
	id = saint.12
	desc = { trigger = { saint_grand_conclave_trigger = yes } text = EVTDESCsaint12 }
	desc = { trigger = { saint_grand_conclave_trigger = no } text = EVTDESCsaint12b }
	picture = GFX_evt_council
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # saint.11

	immediate = {
		save_event_target_as = canonizer_person
	}

	option = { # Raise the soul
		name = EVTOPTAsaint12
		custom_tooltip = { text = saint_canonize_TT }
		hidden_effect = {
			saint_miracle_effect = yes
			event_target:saint_person = {
				remove_trait = beatified
				add_trait = saint
				if = {
					limit = { is_woman_trigger = yes }
					set_special_character_title = SAINT_TITLE_FEMALE
				}
				else = { set_special_character_title = SAINT_TITLE }
			}
			saint_news_venerated_effect = yes
			narrative_event = { id = saint.15 }
			set_global_flag = saintly_burial_sites_exist
			saint_raise_tomb_effect = yes
			saint_found_bloodline_effect = yes
		}
		ai_chance = { factor = 1 }
	}
	option = { # Not yet
		name = EVTOPTBsaint12
		ai_chance = { factor = 0 }
	}
}

		days = 1

		save_event_target_as = saint_person

			limit = { saint_local_priest_trigger = yes }

###############################################################################
# The Grand Theogonist's favourite
# A corrupt Grand Theogonist (bad_priest) may raise a dead friend who shared
# his vices to a Venerated Soul (saint.30, saint.31). Rulers of the faith
# are outraged or side with him (saint.32); a zealous king may march to
# depose him (depose_impious_pope), and the faithful choose sides
# (saint.33). The war ends with saint.35 or saint.36.
###############################################################################

# saint.30 The favourite dies (hidden)
# Scope: ROOT is the dead character (on_death).
character_event = {
	id = saint.30
	hide_window = yes

	is_triggered_only = yes # on_death
	min_age = 16
	religion = sigmarite

	trigger = {
		death_in_cathay_trigger = no
		death_no_trace_trigger = no
		NOT = { trait = excommunicated }
		k_cult_of_sigmar = {
			holder_scope = {
				trait = bad_priest
				independent = yes
				NOT = { has_character_flag = impious_canonization_done }
				opinion = { who = ROOT value = 50 }
				saint_gt_shares_vice_with_root_trigger = yes
			}
		}
	}

	immediate = {
		save_event_target_as = saint_person
		random = {
			chance = 20
			k_cult_of_sigmar = {
				holder_scope = { character_event = { id = saint.31 days = 180 random = 300 } }
			}
		}
	}
}

# saint.31 The Grand Theogonist raises his friend (hidden)
# Scope: ROOT is the Grand Theogonist (saint.30).
character_event = {
	id = saint.31
	hide_window = yes

	is_triggered_only = yes # saint.30

	trigger = {
		has_landed_title = k_cult_of_sigmar
		trait = bad_priest
		NOT = { has_character_flag = impious_canonization_done }
		saint_gt_shares_vice_with_saint_trigger = yes
	}

	immediate = {
		set_character_flag = impious_canonization_done
		set_global_flag = impious_canonization_war_brewing
		religion_authority = { modifier = impious_canonization }
		log = "Unworthy Venerated Soul named by [Root.GetTitledName]"
		event_target:saint_person = {
			remove_trait = beatified
			add_trait = saint
			if = { limit = { is_woman_trigger = yes } set_special_character_title = SAINT_TITLE_FEMALE }
			else = { set_special_character_title = SAINT_TITLE }
			create_bloodline = {
				type = saintly_bloodline_07
				religion = event_target:saint_person
				inheritance = patrilineal
			}
			trigger_switch = {
				on_trigger = trait
				impaler = {
					give_nickname = nick_gods_torturer
					new_bloodline = { set_description = saintly_impaler_description }
				}
				cannibal_trait = {
					give_nickname = nick_gods_cannibal
					new_bloodline = { set_description = saintly_cannibal_description }
				}
				hedonist = {
					give_nickname = nick_gods_hedonist
					new_bloodline = { set_description = saintly_hedonist_description }
				}
				drunkard = {
					give_nickname = nick_gods_drunkard
					new_bloodline = { set_description = saintly_drunkard_description }
				}
				lunatic_1 = {
					give_nickname = nick_the_fool_for_christ
					new_bloodline = { set_description = saintly_lunatic_description }
				}
				lunatic_2 = {
					give_nickname = nick_the_fool_for_christ
					new_bloodline = { set_description = saintly_lunatic_description }
				}
				lunatic_3 = {
					give_nickname = nick_the_fool_for_christ
					new_bloodline = { set_description = saintly_lunatic_description }
				}
				seducer = {
					give_nickname = nick_gods_seducer
					new_bloodline = { set_description = saintly_seducer_description }
				}
				seductress = {
					give_nickname = nick_gods_seducer
					new_bloodline = { set_description = saintly_seducer_description }
				}
				lustful = {
					give_nickname = nick_papal_lover
					new_bloodline = { set_description = saintly_lustful_description }
				}
			}
		}
		any_playable_ruler = { limit = { religion = ROOT } narrative_event = { id = saint.32 } }
		# A zealous king or emperor may take up arms against him
		if = {
			limit = {
				independent = yes
				any_playable_ruler = { saint_papal_attacker_trigger = yes }
			}
			random_playable_ruler = {
				limit = { saint_papal_attacker_trigger = yes }
				preferred_limit = { tier = EMPEROR }
				save_event_target_as = papal_attacker
				add_trait = zealous
				unsafe_war = { target = ROOT casus_belli = depose_impious_pope }
			}
			any_playable_ruler = {
				limit = { religion = ROOT NOT = { character = event_target:papal_attacker } }
				narrative_event = { id = saint.33 days = 15 }
			}
		}
	}
}

# Pilgrimages
# Namespace: pilgrim (was SoA.5000 - SoA.5099 in pilgrim_events.txt)
# Faction: Faiths & doctrines
# Serves: the Old World cults (Sigmar, Ulric, Myrmidia, Manann, Taal,
#   Shallya, Morr and the Lady), the Asur, the Asrai, the Druchii and the
#   Dawi of the Karaz Ankor and of Norsca
#
# The devout make the journey to a great shrine of their faith once in their
# lives (decision go_on_pilgrimage). The pilgrim picks a shrine that is open
# to them (owned by a lord of their own faith group who is not at war with
# them; pilgrim_shrine_open_trigger), walks the road, arrives a month later,
# spends a few days at the shrine and comes home with the Pilgrim trait.
# Old World faithful may walk to the tomb of a Venerated Soul instead
# (saint.60). Holy orders of the pilgrim's faith ride out to meet them, the
# Dawi pass the ruins of lost holds, elves face the sea, the forest or rival
# corsairs. Several shrines have their own trial: Asuryan's flame, Khaine's
# Cauldron of Blood, the riddle of the Oak of Ages, the Great Book of
# Grudges and the Shrine of Grimnir, where a shamed dawi may swear the
# Slayer oath. A good pilgrimage may, rarely, bring a token home.
#
# Shrines (flag pilgrim_to_<key>): averheim (Sigmar), middenheim (Ulric),
#   remas and magritta (Myrmidia), krugenheim (Manann), talabheim (Taal),
#   couronne (Shallya), bordeleaux (the Lady), luccini (Morr),
#   isle_of_flame and gaen_vale (women only) (Asur), oak_of_ages (Asrai),
#   har_ganeth (the Druchii faiths), karaz_a_karak (Dawi), karak_kadrin
#   (Dawi and Norse Dawi), kraka_drak (Norse Dawi).
#
# Setting out                      pilgrim.1 - pilgrim.10
# On the road                      pilgrim.11 - pilgrim.29
# At the shrine                    pilgrim.30 - pilgrim.51
