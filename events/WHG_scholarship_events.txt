###############################################################################
# Way of Life: the scholarship focus, the observatory
# Namespace: scholarship (was WoL.6000 - WoL.6370 in wol_scholarship_events.txt)
# Faction: Generic
#
# A ruler with the scholarship focus may build an observatory (the
# build_observatory decision) and spend years of clear nights under the two
# moons. One road charts the Sun, Mannslieb and the wandering stars and ends
# in a heresy that most priests will not forgive. The other follows the
# stars that vanish, Morrslieb that keeps no table, and the Liber Caelestior
# of Necrodomo the Insane, and ends in knowledge no mortal should keep.
#
# Faiths that read the will of their gods in the sky (Tzeentch, the Old Ones,
# Cathay's dragons, Nehekhara and the Asur of Hoeth) see no heresy in either
# road (scholarship_star_faith_trigger). Tzeentch, the Slann and the Dawi
# have their own voices.
#
# The chain runs through scholarship.4, the next clear night: each step sets
# the variable stargazing_step to the event that comes next, and the router
# waits while the stargazer is jailed, ill, away from the scholarship focus
# or travelling, instead of dropping the chain.
#
# The observatory                     scholarship.1 - scholarship.5
# The road of the heavens             scholarship.10 - scholarship.17
# The road of the strange             scholarship.20 - scholarship.28
# The fate of the Liber Caelestior    scholarship.30 - scholarship.31
#
# Fired from: the build_observatory decision in way_of_life_decisions.txt
#   (scholarship.1) and on_death in 00_on_actions.txt (scholarship.30).
#   Shared triggers and effects are in WHG_focus_triggers.txt and
#   WHG_focus_effects.txt.
###############################################################################

# scholarship.4 The next clear night
# Hidden router for the whole chain. While the stargazer cannot study
# (scholarship_stargazer_busy_trigger, or away from court before the
# journey of scholarship.23) it looks again every 90 days. Otherwise it
# fires the step stored in stargazing_step. The worried kin
# (scholarship.12, scholarship.22) and the angry priest (scholarship.15)
# are picked here and answer through scholarship.5; without one the chain
# moves on.
# Abandoning the studies removes Studying the Stars and ends the chain.
# Scope: ROOT is the stargazer.
character_event = {
	id = scholarship.4
	hide_window = yes

	is_triggered_only = yes # scholarship.2 - scholarship.27

	trigger = {
		has_character_modifier = studying_the_stars
	}

	immediate = {
		if = {
			limit = {
				OR = {
					scholarship_stargazer_busy_trigger = yes
					AND = {
						is_variable_equal = { which = stargazing_step value = 23 }
						OR = {
							has_regent = yes
							in_command = yes
							has_character_flag = do_not_disturb
						}
					}
				}
			}
			repeat_event = { id = scholarship.4 days = 90 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 12 } }
			if = {
				limit = { any_courtier = { scholarship_worried_kin_trigger = yes } }
				random_courtier = {
					limit = { scholarship_worried_kin_trigger = yes }
					character_event = { id = scholarship.5 }
				}
			}
			else = {
				set_variable = { which = stargazing_step value = 13 }
				repeat_event = { id = scholarship.4 days = 180 }
			}
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 15 } }
			if = {
				limit = {
					scholarship_star_faith_trigger = no
					religion_head = {
						is_alive = yes
						ai = yes
						NOT = { character = ROOT }
					}
				}
				religion_head = {
					character_event = { id = scholarship.5 }
				}
			}
			else = {
				character_event = { id = scholarship.16 }
			}
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 22 } }
			if = {
				limit = { any_courtier = { scholarship_worried_kin_trigger = yes } }
				random_courtier = {
					limit = { scholarship_worried_kin_trigger = yes }
					character_event = { id = scholarship.5 }
				}
			}
			else = {
				set_variable = { which = stargazing_step value = 23 }
				repeat_event = { id = scholarship.4 days = 180 }
			}
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 3 } }
			character_event = { id = scholarship.3 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 10 } }
			character_event = { id = scholarship.10 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 11 } }
			character_event = { id = scholarship.11 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 13 } }
			character_event = { id = scholarship.13 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 14 } }
			character_event = { id = scholarship.14 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 16 } }
			character_event = { id = scholarship.16 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 17 } }
			character_event = { id = scholarship.17 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 20 } }
			character_event = { id = scholarship.20 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 23 } }
			character_event = { id = scholarship.23 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 25 } }
			character_event = { id = scholarship.25 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 26 } }
			character_event = { id = scholarship.26 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 27 } }
			character_event = { id = scholarship.27 }
		}
		else_if = {
			limit = { is_variable_equal = { which = stargazing_step value = 28 } }
			character_event = { id = scholarship.28 }
		}
	}
}

# scholarship.5 A visitor for the stargazer
# Hidden bounce, so the worried kin or the head of the faith is FROM in the
# event they bring: scholarship.12, scholarship.22 or the letter
# scholarship.15, by the stargazer's stargazing_step.
# Scope: ROOT is the kin or the head of the faith, FROM the stargazer.
character_event = {
	id = scholarship.5
	hide_window = yes

	is_triggered_only = yes # scholarship.4

	immediate = {
		FROM = {
			if = {
				limit = { is_variable_equal = { which = stargazing_step value = 12 } }
				character_event = { id = scholarship.12 }
			}
			else_if = {
				limit = { is_variable_equal = { which = stargazing_step value = 22 } }
				character_event = { id = scholarship.22 }
			}
			else_if = {
				limit = { is_variable_equal = { which = stargazing_step value = 15 } }
				letter_event = { id = scholarship.15 }
			}
		}
	}
}
