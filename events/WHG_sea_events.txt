###############################################################################
# The seas of the world: voyages of the pirate and raider societies
# Namespace: seas
# Faction: Generic
#
# The sea half of the naval game (the other half is colonies and
# exploration). Three societies sail under it: the Black Ark Corsairs
# (adventurers_druchii_corsairs), the Wolves of the Sea (adventurers_pirates:
# Sartosa, Araby, the Vampire Coast and the Orc freebootaz) and the coastal
# Norscans of the Freigattur (norscan_society). Every event reads the
# captain's voice and the sea they sail, so each of the eleven theatres
# (the Sea of Claws, the Great Western Ocean, the Tilean Sea, the waters of
# Araby, the Southern Sea, the Lustrian seas, the seas of Naggaroth,
# Ulthuan's Inner Sea, the Far East, the Sea of Chaos and the Skaven Inner
# Ocean) has its own cast, hazards and prizes.
#
# Rules kept everywhere: a faction never meets itself (the Vampire Coast's
# captains never meet the Vampire Coast's drowned fleet), and only sea lords
# alive in our 2010 history appear by name (Luthor Harkon, Lokhir
# Fellheart, Duriath Helbane, the jarl of Skeggi).
#
# Setup: sea lords, the Black Arks sail    seas.1-2
# Real sailing hook                        seas.5
# Raid or trade (Bjornlings)               seas.9
# Set Out: heading and quay omens          seas.10-16
# Encounter roller and homecoming          seas.18-19
# Encounters at sea                        seas.20-31
# Digging up buried treasure               seas.34
# Flagships and raiders                    seas.40-43
# The Black Arks                           seas.50-65
# Reputations                              seas.70-73
# The legend of the seas                   seas.100-116
# The Norscan raiders                      seas.121-160
#
# Fired from: seas.1 on_startup (wh_on_actions.txt); seas.5 from naval.1
# (an army at sea); seas.10 from the sea_set_out decision
# (decisions/WHG_sea_decisions.txt); seas.40 - 41 and seas.55 from
# on_five_year_pulse; seas.42 - 43, 50, 52 from the flagship, raider and ark
# decisions; seas.59 - 60 from on_war_started; seas.64 from on_yearly_pulse;
# seas.65 from on_death; seas.70 from the reputation decision, seas.71
# from on_war_started, seas.72 from on_battle_won_owner; seas.100 from
# on_society_progress_full, seas.110 from on_unit_entering_province (a
# player's legendary voyage, game rule warhammer_sea_legend), seas.116 from
# on_death; seas.9 and seas.10 from sea_go_a_viking; seas.130 from
# norse_demand_tribute; seas.140 from norse_champions_fleet; seas.155 from
# on_five_year_pulse; seas.160 from on_character_society_rank_up.
# Scripted triggers: common/scripted_triggers/WHG_sea_triggers.txt
# Scripted effects: common/scripted_effects/WHG_sea_effects.txt
# Voices: localisation/customizable_localisation/WHG_sea_voices.txt
# Loc: localisation/00_WHG_sea_events.csv
# Full list with callers: docs/events/event_database.csv

# seas.1 The sea lords of 2010 (hidden)
# Once a game: finds the sea lords our history has alive at the start
# (sea_find_named_lords_effect), so encounters can name them while they
# live.
# Scope: ROOT is a player at game start or load.
character_event = {
	id = seas.1
	hide_window = yes

	is_triggered_only = yes # on_startup

	trigger = {
		NOT = { has_global_flag = sea_lords_found }
	}

	immediate = {
		set_global_flag = sea_lords_found
		sea_find_named_lords_effect = yes
		character_event = { id = seas.2 days = 1 }
	}
}

# seas.5 A sail on the horizon (hidden)
# A captain of the sea societies who really sails, in command of a fleet,
# may meet what the sea there sends, once a year at most: the same
# encounters as a Set Out voyage, drawn for the sea they are in.
# Scope: ROOT is a commander at sea (naval.1).
character_event = {
	id = seas.5
	hide_window = yes

	is_triggered_only = yes # naval.1
	only_capable = yes

	trigger = {
		sea_captain_trigger = yes
		NOT = { has_character_flag = sea_on_voyage }
		NOT = { has_character_modifier = sea_encounter_cd }
	}

	immediate = {
		sea_set_theatre_from_location_effect = yes
		if = {
			limit = {
				OR = {
					has_character_flag = sea_in_claws
					has_character_flag = sea_in_western
					has_character_flag = sea_in_tilean
					has_character_flag = sea_in_araby
					has_character_flag = sea_in_southern
					has_character_flag = sea_in_lustria
					has_character_flag = sea_in_naggaroth
					has_character_flag = sea_in_ulthuan
					has_character_flag = sea_in_east
					has_character_flag = sea_in_chaos
					has_character_flag = sea_in_skaven
				}
			}
			add_character_modifier = { name = sea_encounter_cd years = 1 hidden = yes }
			character_event = { id = seas.18 }
		}
	}
}

# seas.11 Casting off (hidden)
# The voyage begins: three to six months at sea, two or three encounters,
# and an omen on the quay that sets the sea's mood (seas.12 - 16).
# Scope: ROOT is the captain.
character_event = {
	id = seas.11
	hide_window = yes

	is_triggered_only = yes # seas.10

	immediate = {
		set_character_flag = sea_on_voyage
		add_character_modifier = { name = sea_at_sea days = 150 }
		random_list = {
			50 = { set_variable = { which = sea_encounters_left value = 2 } }
			50 = { set_variable = { which = sea_encounters_left value = 3 } }
		}
		random_list = {
			20 = { character_event = { id = seas.12 } }
			20 = { character_event = { id = seas.13 } }
			20 = { character_event = { id = seas.14 } }
			15 = { character_event = { id = seas.15 } }
			25 = { character_event = { id = seas.16 } }
		}
		character_event = { id = seas.18 days = 20 random = 20 }
	}
}

# seas.18 What the sea sends (hidden)
# Draws the next encounter for the sea the captain sails, weighted by the
# sea and the sea's mood, then schedules the next one or the way home
# (seas.19). A captain taken or lost comes home early. On a real voyage
# (seas.5) only one encounter is drawn.
# Scope: ROOT is the captain.
character_event = {
	id = seas.18
	hide_window = yes

	is_triggered_only = yes # seas.5, seas.11, seas.18

	immediate = {
		if = {
			limit = {
				has_character_flag = sea_on_voyage
				OR = {
					prisoner = yes
					is_inaccessible_trigger = yes
					trait = incapable
					NOT = { check_variable = { which = sea_encounters_left value = 1 } }
				}
			}
			character_event = { id = seas.19 }
		}
		else = {
			random_list = {
				20 = { # A merchantman
					modifier = {
						factor = 1.5
						OR = {
							has_character_flag = sea_in_tilean
							has_character_flag = sea_in_araby
							has_character_flag = sea_in_claws
							has_character_flag = sea_in_western
						}
					}
					modifier = { factor = 0.3 has_character_flag = sea_in_skaven }
					modifier = { factor = 0 has_character_flag = sea_voyage_trade }
					character_event = { id = seas.20 }
				}
				10 = { # A storm
					modifier = { factor = 2 manann_mood_bad_trigger = yes }
					modifier = { factor = 0.5 manann_mood_good_trigger = yes }
					modifier = {
						factor = 1.5
						OR = {
							has_character_flag = sea_in_claws
							has_character_flag = sea_in_western
							has_character_flag = sea_in_southern
							has_character_flag = sea_in_chaos
						}
					}
					character_event = { id = seas.21 }
				}
				8 = { # A sea monster
					modifier = { factor = 1.5 manann_mood_bad_trigger = yes }
					modifier = {
						factor = 2
						OR = {
							has_character_flag = sea_in_claws
							has_character_flag = sea_in_lustria
							has_character_flag = sea_in_naggaroth
							has_character_flag = sea_in_east
						}
					}
					character_event = { id = seas.22 }

				6 = { # Mutiny (the dead do not mutiny)

				}

							has_character_flag = sea_in_lustria

# seas.160 The sea traits come with rank (hidden)
# Rank brings the sea traits of the north to a Freigattur raider (Viking at
# rank 2, Ravager at 3, Sea King or Sea Queen at 4) and the Pirate trait to
# a Wolf of the Sea or a Corsair at rank 2. A Norscan warlord of the land
# keeps their own trait.
# Scope: ROOT is the member; FROM the society.
character_event = {
	id = seas.160
	hide_window = yes
	is_triggered_only = yes # on_character_society_rank_up

	trigger = {
		NOT = { trait = norscan_warlord }
		OR = {
			sea_norse_raider_trigger = yes
			is_member_of_any_piratic_society_trigger = yes
		}
	}

	immediate = {
		if = {
			limit = { sea_norse_raider_trigger = yes }
			if = {
				limit = {
					society_rank == 4
					NOR = { trait = seaking trait = sea_queen }
				}
				remove_trait = viking
				remove_trait = ravager
				remove_trait = pirate
				if = {
					limit = { is_woman_trigger = yes }
					add_trait = sea_queen
				}
				else = {
					add_trait = seaking
				}
			}
			else_if = {
				limit = {
					society_rank == 3
					NOR = { trait = ravager trait = seaking trait = sea_queen }
				}
				remove_trait = viking
				remove_trait = pirate
				add_trait = ravager
			}
			else_if = {
				limit = {
					society_rank == 2
					NOR = { trait = viking trait = ravager trait = seaking trait = sea_queen trait = pirate }
				}
				add_trait = viking
			}
		}
		else_if = {
			limit = {
				society_rank >= 2
				NOR = { trait = pirate trait = viking trait = ravager trait = seaking trait = sea_queen }
			}
			add_trait = pirate
		}
	}
}
