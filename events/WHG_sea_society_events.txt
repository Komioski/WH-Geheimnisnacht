###############################################################################
# Missions of the sea societies
# Namespace: seaquest (was DoWA.200 - 211 and DoWA.250 - 258 in
#   DoW_Adventurers.txt)
# Faction: Generic
# Serves: the Wolves of the Sea (adventurers_pirates), the Black Ark
#   Corsairs (adventurers_druchii_corsairs) and the League of explorers
#   (adventurers_the_league)
#
# Every two years a captain of the pirate societies may set a member a
# mission: raid a rich coast, take up a warrior's focus, a duel, a war of
# conquest, bring a child into the trade, or two of their own. In the Game
# of Reavers the member races a rival pirate to plunder a rich coast first:
# the north of the Empire, Bretonnia, Estalia, Tilea or Ulthuan. In the
# hunt for the admiral, a navy sends a fleet against the member, and it must
# be beaten at sea within two years.
#
# The League's captains send members to map an unknown coast, to raid the
# tombs of Nehekhara, to win a colonial charter in Lustria or the
# Southlands, or on the same raids, focuses, wars and family business.
#
# The old missions had no caller and never fired; they now run on
# on_society_bi_yearly_pulse, as the warrior lodges' do (lodge.200).
#
# The pirates' missions                seaquest.1
# The League's missions                seaquest.2
# The Game of Reavers                  seaquest.10 - seaquest.17
# The hunt for the admiral             seaquest.20 - seaquest.24
# Mapping a coast                      seaquest.40 - seaquest.42
# Raiding the tombs                    seaquest.45 - seaquest.46
# A colonial charter                   seaquest.50 - seaquest.53
#
# Callers: seaquest.1 - 2 from on_society_bi_yearly_pulse
#   (00_on_actions.txt); seaquest.13 from on_loot_settlement
#   (wh_on_actions.txt); seaquest.22 from on_battle_won_owner, seaquest.23
#   from on_battle_lost_owner, seaquest.42 from on_unit_entering_province
#   and seaquest.52 from on_new_holder_usurpation (dow_on_actions.txt). The
#   shared missions (lodge.201, 204, 220, 225, 227, 235, 242) are in
#   WHG_warrior_society_events.txt. The tomb raid is won with the
#   exploration casus belli (cb_types_nehekharan.txt), which clears the
#   quest (nehekhara events).
# Saved: character flags game_of_reavers_flag and the region flags
#   ne/b/e/t/u_game_of_reavers_flag, dealwadmiralty_flag (the hunter),
#   dealwadmiral_target_flag (the admiral), seaquest_mapping_coast;
#   variable game_of_reavers_value; title flag temporary_admiralty_title.
#   Triggers in common/scripted_triggers/wh_sea_society_triggers.txt,
#   effects in common/scripted_effects/wh_sea_society_effects.txt.
# Loc: localisation/00_WHG_sea_society_events.csv (the societies' own loc
#   stays in dow_adventurers.csv)
# Full list with callers: docs/events/event_database.csv
###############################################################################

# seaquest.1 The pirates' missions (hidden)
# Every two years a captain of the Wolves of the Sea or the Black Ark
# Corsairs may set a member a mission: raid a rich coast (lodge.201), the
# Game of Reavers (seaquest.10, players), a warrior's focus (lodge.204), a
# friendly duel (lodge.220), a duel to the death (lodge.225, players), a
# war of conquest (lodge.227), the hunt for the admiral (seaquest.20,
# players) or a child for the trade (lodge.235). The AI mostly settles
# missions in the background.
# Scope: the member. event_target:lodge_hero is the captain who asks.
character_event = {
	id = seaquest.1
	hide_window = yes

	is_triggered_only = yes # on_society_bi_yearly_pulse
	is_in_society = yes
	only_capable = yes

	trigger = {
		is_member_of_any_piratic_society_trigger = yes
		seaquest_free_member_trigger = yes
		any_society_member = {
			NOT = { character = ROOT }
			same_society_as = ROOT
			society_rank = 3
		}
	}

	immediate = {
		random_society_member = {
			limit = {
				NOT = { character = ROOT }
				same_society_as = ROOT
				society_rank = 3
			}
			preferred_limit = { society_rank == 4 }
			save_event_target_as = lodge_hero
		}
		if = {
			limit = { is_landed = yes }
			capital_scope = { save_event_target_as = home_capital }
		}
		random_list = {
			60 = {
				trigger = { seaquest_pirate_can_raid_trigger = yes }
				event_target:lodge_hero = { character_event = { id = lodge.201 } }
			}
			20 = {
				trigger = {
					ai = no
					is_landed = yes
					society_rank = 2
					is_allowed_to_loot = yes
				}
				event_target:lodge_hero = { character_event = { id = seaquest.10 } }
			}
			20 = {
				trigger = {
					has_dlc = "Way of Life"
					is_landed = yes
					NOR = {
						has_focus = focus_war
						has_focus = focus_hunting
						has_focus = focus_rulership
						has_focus = focus_carousing
					}
				}
				event_target:lodge_hero = { character_event = { id = lodge.204 } }
			}
			20 = {
				trigger = { seaquest_duel_partner_trigger = yes }
				event_target:lodge_hero = { character_event = { id = lodge.220 } }
			}
			5 = {
				trigger = { seaquest_deadly_duel_trigger = yes }
				event_target:lodge_hero = { character_event = { id = lodge.225 } }
			}
			20 = {
				trigger = { seaquest_conquest_target_trigger = yes }
				event_target:lodge_hero = { character_event = { id = lodge.227 } }
			}
			30 = {
				trigger = {
					ai = no
					seaquest_admiral_reach_trigger = yes
				}
				event_target:lodge_hero = { character_event = { id = seaquest.20 } }
			}
			20 = {
				trigger = { seaquest_child_recruit_trigger = yes }
				event_target:lodge_hero = { character_event = { id = lodge.235 } }
			}
			75 = {
				trigger = { ai = yes }
				seaquest_ai_settles_mission_effect = yes
			}
			200 = {
				trigger = { ai = yes }
			}
		}
	}
}

# seaquest.11 The Game of Reavers
# The captain names a rival pirate (of another society one time in four, a
# player one time in two) and a rich coast both can reach: the north of the
# Empire, Bretonnia, Estalia, Tilea or Ulthuan. Whoever plunders 50 worth
# of its ports first wins (a tribe 1, a castle 2, a city or temple 4). A
# player rival is challenged too (seaquest.12). Five years at most
# (seaquest.17).
# Scope: the member. FROM is the captain.
society_quest_event = {
	id = seaquest.11
	desc = EVTDESCseaquest11
	picture = GFX_evt_pirate_crew
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # seaquest.10
	is_in_society = yes
	only_capable = yes

	trigger = {
		is_member_of_any_piratic_society_trigger = yes
		same_society_as = FROM
		seaquest_free_member_trigger = yes
		is_landed = yes
		is_allowed_to_loot = yes
		OR = {
			independent = yes
			top_liege = { is_allowed_to_loot = yes }
		}
	}

	immediate = {
		random = {
			chance = 50
			set_character_flag = game_of_reavers_player_target_flag
		}
		random = {
			chance = 25
			set_character_flag = game_of_reavers_rival_society_flag
		}
		random_playable_ruler = {
			limit = {
				NOT = { character = ROOT }
				is_member_of_any_piratic_society_trigger = yes
				NOT = { war_with = ROOT }
				NOR = {
					is_liege_of = ROOT
					has_non_aggression_pact_with = ROOT
					is_allied_with = ROOT
					AND = {
						is_tributary = yes
						suzerain = {
							OR = {
								character = ROOT
								any_liege = { character = ROOT }
							}
						}
					}
				}
				OR = {
					NOT = { distance_from_realm = { who = ROOT value = 800 } }
					ROOT = { has_character_flag = game_of_reavers_rival_society_flag }
				}
			}
			preferred_limit = {
				ROOT = { has_character_flag = game_of_reavers_rival_society_flag }
				NOT = { same_society_as = ROOT }
			}
			preferred_limit = {
				ROOT = { has_character_flag = game_of_reavers_player_target_flag }
				ai = no
			}
			preferred_limit = { war = no }
			preferred_limit = {
				NOR = {
					is_tribal = yes
					is_nomadic = yes
				}
			}
			save_event_target_as = pirate_rival
		}
		random_list = {
			20 = { # The north of the Empire
				trigger = {
					event_target:pirate_rival = { capital_scope = { can_naval_path_to = { target = c_marienburg distance < 800 } } }
					capital_scope = { can_naval_path_to = { target = c_marienburg distance < 800 } }
				}
				set_character_flag = ne_game_of_reavers_flag
			}
			15 = { # Bretonnia
				trigger = {
					event_target:pirate_rival = { capital_scope = { can_naval_path_to = { target = c_bordeleaux distance < 800 } } }
					capital_scope = { can_naval_path_to = { target = c_bordeleaux distance < 800 } }
				}
				set_character_flag = b_game_of_reavers_flag
			}
			30 = { # Estalia
				trigger = {
					event_target:pirate_rival = { capital_scope = { can_naval_path_to = { target = c_magritta distance < 800 } } }
					capital_scope = { can_naval_path_to = { target = c_magritta distance < 800 } }
				}
				set_character_flag = e_game_of_reavers_flag
			}
			20 = { # Tilea
				trigger = {
					event_target:pirate_rival = { capital_scope = { can_naval_path_to = { target = c_remas distance < 800 } } }
					capital_scope = { can_naval_path_to = { target = c_remas distance < 800 } }
				}
				set_character_flag = t_game_of_reavers_flag
			}
			15 = { # Ulthuan
				trigger = {
					event_target:pirate_rival = { capital_scope = { can_naval_path_to = { target = c_lothern distance < 800 } } }
					capital_scope = { can_naval_path_to = { target = c_lothern distance < 800 } }
				}
				set_character_flag = u_game_of_reavers_flag
			}
		}
		clr_character_flag = game_of_reavers_player_target_flag
		clr_character_flag = game_of_reavers_rival_society_flag
	}

	quest_target = event_target:pirate_rival

	option = {
		name = EVTOPTAseaquest11
		trigger = {
			event_target:pirate_rival = { always = yes }
			OR = {
				has_character_flag = ne_game_of_reavers_flag
				has_character_flag = b_game_of_reavers_flag
				has_character_flag = e_game_of_reavers_flag
				has_character_flag = t_game_of_reavers_flag
				has_character_flag = u_game_of_reavers_flag
			}
		}
		if = {
			limit = {
				event_target:pirate_rival = {
					ai = no
					has_any_quest = no
				}
			}
			event_target:pirate_rival = {
				seaquest_copy_reavers_region_effect = yes
				society_quest_event = { id = seaquest.12 days = 1 }
			}
		}
		seaquest_start_game_of_reavers_effect = yes
		character_event = { id = seaquest.17 days = 1825 }
	}
	option = {
		name = EVTOPTX_seaquest_11
		custom_tooltip = { text = decline_quest_tooltip }
		seaquest_clear_reavers_flags_effect = yes
		add_character_modifier = {
			name = quest_cooldown_timer
			hidden = yes
			days = 1000
		}
		ai_chance = { factor = 0 }
	}
}

# seaquest.13 A port plundered (hidden)
# A player in the Game of Reavers loots a port on the coast of the game: a
# tribe is worth 1, a castle 2, a city or temple 4. At 50 the player wins
# (seaquest.14) and a player rival still in the game loses (seaquest.15).
# If the rival is dead the game is over (seaquest.16).
# Scope: ROOT is the raider. FROM is the looted settlement.
character_event = {
	id = seaquest.13
	hide_window = yes

	is_triggered_only = yes # on_loot_settlement

	only_playable = yes
	ai = no
	has_character_flag = game_of_reavers_flag

	trigger = {
		FROM = { location = { port = yes } }
		OR = {
			AND = {
				has_quest = quest_game_of_reavers_pirate_ne
				FROM = { location = { region = world_old_world_empire_north } }
			}
			AND = {
				has_quest = quest_game_of_reavers_pirate_b
				FROM = { location = { region = world_old_world_bretonnia } }
			}
			AND = {
				has_quest = quest_game_of_reavers_pirate_e
				FROM = { location = { region = world_old_world_estalia } }
			}
			AND = {
				has_quest = quest_game_of_reavers_pirate_t
				FROM = { location = { region = world_old_world_tilea } }
			}
			AND = {
				has_quest = quest_game_of_reavers_pirate_u
				FROM = { location = { region = world_ulthuan } }
			}
		}
	}

	immediate = {
		quest_target = { save_event_target_as = pirate_rival }
		if = {
			limit = { event_target:pirate_rival = { is_alive = no } }
			character_event = { id = seaquest.16 }
		}
		else = {
			if = {
				limit = { FROM = { holding_type = tribal } }
				change_variable = { which = game_of_reavers_value value = 1 }
			}
			else_if = {
				limit = { FROM = { holding_type = castle } }
				change_variable = { which = game_of_reavers_value value = 2 }
			}
			else = {
				change_variable = { which = game_of_reavers_value value = 4 }
			}
			if = {
				limit = { check_variable = { which = game_of_reavers_value value = 50 } }
				character_event = { id = seaquest.14 }
				event_target:pirate_rival = {
					if = {
						limit = { has_character_flag = game_of_reavers_flag }
						character_event = { id = seaquest.15 }
					}
				}
			}
		}
	}
}
