###############################################################################
# Seeresses and the tainted child
# Namespace: seeress (was TOG.3200 - TOG.3260 in oldgods_various_events.txt)
# Faction: Kislev
# Serves: the Ungol and Kislevite faiths (the hags), the old Norscan gods
#   (the seeresses)
#
# A woman who sees what others cannot comes to a lord's hall and asks to
# stay: a hag of the stanitsa among the Ungol and Kislevites, a seeress
# among the Norscans. She may be young and fair, grown, or a crone.
#
# The tainted child: a hag may feel the touch of the Ruinous Powers on a
# child still in the womb. Born, the child may truly be tainted, or not.
# A tainted child does strange things at night, and courtiers die around it.
# The hag will ask for the child, to burn the taint out in the old way: the
# child may come back cleansed or hurt, or die, or the hag may be found torn
# apart. A tainted child who grows up hears the voices give their names, and
# turns to the Dark Gods.
#
# The foretold child: a Norscan seeress may see the gods' favour on a child
# in the womb. The child grows up bold and is spoken of.
#
# A seeress comes to court             seeress.1
# The womb                             seeress.10 - seeress.12
# The tainted child                    seeress.20 - seeress.26
# Grown                                seeress.30 - seeress.32
#
# Fired from: the five-year pulse list 'seeress' in WHG_character_on_actions.txt
#   (seeress.1), on_pregnancy (seeress.10), on_birth (seeress.20) and
#   on_adulthood (seeress.30) in WHG_character_on_actions.txt. Shared triggers are in
#   WHG_faith_triggers.txt.
# Saved: event targets seeress_seer, seeress_mother, seeress_child,
#   seeress_victim; flags court_seeress (on the seeress), seeress_young,
#   seeress_old (cleared by seeress.1), pregnant_with_demon and
#   seeress_foretold (mother), demon_child, demon_child_fake (shared with other
#   files), child_strange_1-3, child_kills_1-3, child_exorcism,
#   seeress_rite_* (cleared by seeress.26), seeress_foretold_child.
###############################################################################

namespace = seeress

# seeress.1 A woman who sees
# A hag for a lord of the Ungol and Kislevite faiths (she may take the post
# of wise woman), a seeress for one of the Norscan gods. Young and fair,
# grown, or old and ugly.
# Scope: ROOT is a player ruler without one. Was three MTTH events of 183
#   years each.
character_event = {
	id = seeress.1
	desc = {
		trigger = { seeress_hag_faith_trigger = yes has_character_flag = seeress_young }
		text = EVTDESCseeress1_hag_young
	}
	desc = {
		trigger = { seeress_hag_faith_trigger = yes NOT = { has_character_flag = seeress_young } NOT = { has_character_flag = seeress_old } }
		text = EVTDESCseeress1_hag
	}
	desc = {
		trigger = { seeress_hag_faith_trigger = yes has_character_flag = seeress_old }
		text = EVTDESCseeress1_hag_old
	}
	desc = {
		trigger = { seeress_hag_faith_trigger = no has_character_flag = seeress_young }
		text = EVTDESCseeress1_seer_young
	}
	desc = {
		trigger = { seeress_hag_faith_trigger = no NOT = { has_character_flag = seeress_young } NOT = { has_character_flag = seeress_old } }
		text = EVTDESCseeress1_seer
	}
	desc = {
		trigger = { seeress_hag_faith_trigger = no has_character_flag = seeress_old }
		text = EVTDESCseeress1_seer_old
	}
	picture = GFX_evt_viking_throneroom_oldgods
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	ai = no
	min_age = 16
	capable_only = yes
	only_playable = yes
	prisoner = no

	trigger = {
		in_command = no
		OR = {
			seeress_hag_faith_trigger = yes
			seeress_seer_faith_trigger = yes
		}
		seeress_seer_at_court_trigger = no
	}

	immediate = {
		random_list = {
			33 = { set_character_flag = seeress_young }
			34 = { }
			33 = { set_character_flag = seeress_old }
		}
	}

	option = {
		name = EVTOPTAseeress1
		if = {
			limit = { has_character_flag = seeress_young }
			create_character = {
				random_traits = no
				dynasty = none
				female = yes
				age = 22
				religion = ROOT
				culture = ROOT
				trait = mystic
				trait = scholarly_theologian
				trait = fair
			}
		}
		else_if = {
			limit = { has_character_flag = seeress_old }
			create_character = {
				random_traits = no
				dynasty = none
				female = yes
				age = 53
				religion = ROOT
				culture = ROOT
				trait = mystic
				trait = scholarly_theologian
				trait = ugly
			}
		}
		else = {
			create_character = {
				random_traits = no
				dynasty = none
				female = yes
				age = 35
				religion = ROOT
				culture = ROOT
				trait = mystic
				trait = scholarly_theologian
			}
		}
		new_character = {
			add_trait = creature_human
			set_character_flag = court_seeress
			random_list = {
				30 = { add_trait = magic_potential_1 }
				60 = { add_trait = magic_potential_2 }
			}
			if = {
				limit = { ROOT = { seeress_hag_faith_trigger = yes } }
				add_trait = hag_witch
				add_trait = lore_hag
				give_minor_title = title_wise_woman
			}
			else = {
				add_trait = witch
				random_list = {
					30 = { add_trait = lore_heavens }
					25 = { add_trait = lore_beasts }
					25 = { add_trait = lore_death }
					20 = { add_trait = lore_shadows }
				}
			}
			magic_power_random_effect = yes
		}
	}
	option = {
		name = EVTOPTBseeress1
		custom_tooltip = { text = seeress_sent_away_tt }
	}

	after = {
		clr_character_flag = seeress_young
		clr_character_flag = seeress_old
	}
}

# seeress.10 The seer feels the child
# Now and then (3%) a hag at the husband's court feels the Ruinous Powers on
# the unborn child (seeress.11); a Norscan seeress may see the gods' favour
# on it instead (seeress.12). Was an MTTH of 45 years on pregnant women.
# Scope: ROOT is the mother. From on_pregnancy.
character_event = {
	id = seeress.10

	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_pregnant = yes
		NOT = { has_character_flag = pregnant_with_demon }
		NOT = { has_character_flag = seeress_foretold }
		spouse = { seeress_seer_at_court_trigger = yes }
	}

	immediate = {
		random = {
			chance = 3
			spouse = {
				save_event_target_as = seeress_father
				random_courtier = {
					limit = { seeress_court_seer_trigger = yes }
					save_event_target_as = seeress_seer
				}
			}
			save_event_target_as = seeress_mother
			if = {
				limit = { event_target:seeress_seer = { seeress_court_hag_trigger = yes } }
				set_character_flag = pregnant_with_demon
				event_target:seeress_father = { character_event = { id = seeress.11 } }
			}
			else = {
				set_character_flag = seeress_foretold
				event_target:seeress_father = { character_event = { id = seeress.12 } }
			}
		}
	}
}

# seeress.20 Born
# A child the hag warned of is tainted, or only feared to be (50/50); its
# life is watched from the age of four (seeress.21). A child the seeress
# foretold is marked for it.
# Scope: ROOT is the newborn. From on_birth.
character_event = {
	id = seeress.20

	hide_window = yes
	is_triggered_only = yes

	trigger = {
		mother = {
			OR = {
				has_character_flag = pregnant_with_demon
				has_character_flag = seeress_foretold
			}
		}
	}

	immediate = {
		if = {
			limit = { mother = { has_character_flag = pregnant_with_demon } }
			random_list = {
				50 = { set_character_flag = demon_child }
				50 = { set_character_flag = demon_child_fake }
			}
			character_event = { id = seeress.21 days = 1460 random = 365 }
		}
		else = {
			set_character_flag = seeress_foretold_child
		}
		mother = {
			clr_character_flag = pregnant_with_demon
			clr_character_flag = seeress_foretold
		}
	}
}

# seeress.21 The child grows
# Every two to three years until grown: a tainted child does something
# strange (seeress.22, three times at most), or a courtier dies near it
# (seeress.23, after something strange, three times at most); a hag at the
# father's court asks for the child, tainted or not (seeress.24, once). Was
# seven MTTH events on every such child.
# Scope: ROOT is the child.
character_event = {
	id = seeress.21

	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			has_character_flag = demon_child
			has_character_flag = demon_child_fake
		}
		is_adult = no
		father = { is_alive = yes }
	}

	immediate = {
		random_list = {
			30 = { }
			25 = { # Strange
				modifier = { factor = 0 NOT = { has_character_flag = demon_child } }
				modifier = { factor = 0 has_character_flag = child_strange_3 }
				if = {
					limit = { has_character_flag = child_strange_2 }
					set_character_flag = child_strange_3
				}
				else_if = {
					limit = { has_character_flag = child_strange_1 }
					set_character_flag = child_strange_2
				}
				else = {
					set_character_flag = child_strange_1
				}
				father = { character_event = { id = seeress.22 } }
			}
			20 = { # A courtier dies
				modifier = { factor = 0 NOT = { has_character_flag = demon_child } }
				modifier = { factor = 0 NOT = { has_character_flag = child_strange_1 } }
				modifier = { factor = 0 has_character_flag = child_kills_3 }
				modifier = {
					factor = 0
					father = { NOT = { any_courtier = { seeress_child_victim_trigger = yes } } }
				}
				save_event_target_as = seeress_child
				father = {
					random_courtier = {
						limit = { seeress_child_victim_trigger = yes }
						save_event_target_as = seeress_victim
					}
				}
				if = {
					limit = { has_character_flag = child_kills_2 }
					set_character_flag = child_kills_3
				}
				else_if = {
					limit = { has_character_flag = child_kills_1 }
					set_character_flag = child_kills_2
				}
				else = {
					set_character_flag = child_kills_1
				}
				father = { character_event = { id = seeress.23 } }
			}
			15 = { # The hag asks for the child
				modifier = { factor = 0 has_character_flag = child_exorcism }
				modifier = {
					factor = 0
					father = {
						NOT = { any_courtier = { seeress_court_hag_trigger = yes } }
					}
				}
				set_character_flag = child_exorcism
				save_event_target_as = seeress_child
				father = {
					random_courtier = {
						limit = { seeress_court_hag_trigger = yes }
						save_event_target_as = seeress_seer
					}
					character_event = { id = seeress.24 }
				}
			}
		}
		character_event = { id = seeress.21 days = 730 random = 365 }
	}
}

# seeress.25 The rite
# The child is given a sleeping draught and taken to the hags' hut. Three
# nights later: the taint burned out (likeliest for a child who was never
# tainted), the child hurt, the child dead, or the hag torn apart (only a
# tainted child can do that).
# Scope: ROOT is the father; seeress_seer the hag, seeress_child the child.
character_event = {
	id = seeress.25
	desc = EVTDESCseeress25
	picture = GFX_evt_pagan
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	trigger = {
		event_target:seeress_child = { is_alive = yes }
		event_target:seeress_seer = { is_alive = yes }
	}

	option = {
		name = EVTOPTAseeress25
		custom_tooltip = { text = seeress_rite_begins_tt }
		random_list = {
			30 = {
				modifier = { factor = 2 event_target:seeress_child = { has_character_flag = demon_child_fake } }
				set_character_flag = seeress_rite_cleansed
			}
			20 = { set_character_flag = seeress_rite_hurt }
			20 = { set_character_flag = seeress_rite_child_dies }
			30 = {
				modifier = { factor = 0 event_target:seeress_child = { has_character_flag = demon_child_fake } }
				set_character_flag = seeress_rite_hag_dies
			}
		}
		hidden_effect = { character_event = { id = seeress.26 days = 3 } }
	}
}

# seeress.26 Three nights later
# Cleansed: the child comes home. Hurt: you pull the hag off the child and
# banish her or let her go. Dead: execute the hag, banish her or accept it.
# The hag torn apart: the taint stays, and no hag will try again.
# Scope: ROOT is the father (flag seeress_rite_*); seeress_seer the hag,
#   seeress_child the child.
character_event = {
	id = seeress.26
	desc = {
		trigger = { has_character_flag = seeress_rite_cleansed }
		text = EVTDESCseeress26_cleansed
	}
	desc = {
		trigger = { has_character_flag = seeress_rite_hurt }
		text = EVTDESCseeress26_hurt
	}
	desc = {
		trigger = { has_character_flag = seeress_rite_child_dies }
		text = EVTDESCseeress26_child_dies
	}
	desc = {
		trigger = { has_character_flag = seeress_rite_hag_dies }
		text = EVTDESCseeress26_hag_dies
	}
	picture = GFX_evt_pagan
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_character_flag = seeress_rite_child_dies }
			event_target:seeress_child = {
				if = {
					limit = { is_invincible_trigger = yes }
					antideath_effect = yes
				}
				else = {
					death = {
						death_reason = death_sacrificed_norse
						killer = event_target:seeress_seer
					}
				}
			}
		}
		else_if = {
			limit = { has_character_flag = seeress_rite_hag_dies }
			event_target:seeress_seer = {
				if = {
					limit = { is_invincible_trigger = yes }
					antideath_effect = yes
				}
				else = {
					death = {
						death_reason = death_murder_butchered
						killer = event_target:seeress_child
					}
				}
			}
		}
		else = {
			event_target:seeress_child = {
				seeress_cleanse_child_effect = yes
				if = {
					limit = { ROOT = { has_character_flag = seeress_rite_hurt } }
					add_trait = wounded
				}
			}
		}
	}

	option = { # Cleansed
		name = EVTOPTAseeress26_cleansed
		trigger = { has_character_flag = seeress_rite_cleansed }
		event_target:seeress_seer = {
			opinion = { modifier = opinion_grateful who = ROOT }
		}
	}
	option = { # Hurt: banish the hag
		name = EVTOPTAseeress26_hurt
		trigger = { has_character_flag = seeress_rite_hurt }
		event_target:seeress_seer = { banish = yes }
	}
	option = { # Hurt: let her go
		name = EVTOPTBseeress26_hurt
		trigger = { has_character_flag = seeress_rite_hurt }
	}
	option = { # Dead: execute the hag
		name = EVTOPTAseeress26_child_dies
		trigger = { has_character_flag = seeress_rite_child_dies }
		event_target:seeress_seer = {
			if = {
				limit = { is_invincible_trigger = yes }
				antideath_effect = yes
			}
			else = {
				death = {
					death_reason = death_execution
					killer = ROOT
				}
			}
		}
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = wroth }
		}
	}
	option = { # Dead: banish her
		name = EVTOPTBseeress26_child_dies
		trigger = { has_character_flag = seeress_rite_child_dies }
		event_target:seeress_seer = { banish = yes }
		ai_chance = { factor = 30 }
	}
	option = { # Dead: it had to be done
		name = EVTOPTCseeress26_child_dies
		trigger = { has_character_flag = seeress_rite_child_dies }
		piety = 25
		ai_chance = {
			factor = 30
			modifier = { factor = 2 trait = zealous }
		}

# seeress.30 Grown
# A tainted child who comes of age hears the voices give their names
# (seeress.31); one only feared to be tainted is free of it. A foretold
# child grows into the prophecy (seeress.32).
# Scope: ROOT is the new adult. From on_adulthood.
character_event = {
	id = seeress.30

	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			has_character_flag = demon_child
			has_character_flag = demon_child_fake
			has_character_flag = seeress_foretold_child
		}
	}

	immediate = {
		if = {
			limit = { has_character_flag = demon_child }
			character_event = { id = seeress.31 }
		}
		else_if = {
			limit = { has_character_flag = seeress_foretold_child }
			clr_character_flag = seeress_foretold_child
			character_event = { id = seeress.32 }
		}
		else = {
			seeress_cleanse_child_effect = yes
		}
	}
}
