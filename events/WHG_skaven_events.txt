# skaven.1 Breeding check
# A Skaven ruler (or a childless brother at court) sires a litter with one of
# the court's breeders. Breeders take turns: one who just gave birth rests for
# a while (stage_progressed modifier). Every five years the trait weights
# below (x100) are rolled against a no-litter weight of 25, so rulers at
# peace, in good health and with many breeders breed most.
# Scope: ROOT is the ruler.
character_event = {
	id = skaven.1
	hide_window = yes
	is_triggered_only = yes # on_five_year_pulse, random list skaven_breeding

	only_playable = yes
	religion_group = horned_rat_god_group
	only_men = yes
	min_age = 16
	capable_only = yes
	prisoner = no

	trigger = {
		trait = creature_skaven
		is_incapacitated_trigger = no
		NOR = {
			trait = eunuch
			trait = transgender
			trait = celibate
			trait = infertile
			trait = incapable
		}
		OR = {
			any_sibling = { skaven_breeding_brother_trigger = yes }
			AND = {
				NOT = { any_child = { count = 5 is_alive = yes } }
				OR = {
					NOT = {
						any_child = {
							count = 2
							is_alive = yes
							is_female = no
							NOT = { trait = white_skaven }
						}
					}
					NOT = { any_child = { count = 2 is_alive = yes is_female = yes } }
				}
			}
		}
		any_courtier = {
			skaven_ready_breeder_trigger = yes
			NOT = { has_character_modifier = stage_progressed }
		}
	}

	weight_multiplier = {
		days = 1
		# Circumstances
		modifier = { factor = 0.2 war = yes }
		modifier = { factor = 0.02 in_command = yes }
		modifier = { factor = 0.5 age = 30 }
		modifier = { factor = 1.2 any_sibling = { skaven_breeding_brother_trigger = yes } }
		modifier = { factor = 3.0 trait = retired_with_harem }
		modifier = { factor = 2.0 has_character_modifier = horny_harem_mod }
		# Personality
		modifier = { factor = 1.25 trait = lustful }
		modifier = { factor = 0.25 trait = shy }
		modifier = { factor = 1.15 trait = hedonist }
		modifier = { factor = 0.75 trait = kind }
		modifier = { factor = 0.67 trait = just }
		modifier = { factor = 1.05 trait = cruel }
		modifier = { factor = 1.05 trait = arbitrary }
		modifier = { factor = 0.05 trait = homosexual }
		modifier = { factor = 0.75 trait = temperate }
		modifier = { factor = 0.75 trait = charitable }
		modifier = { factor = 0.67 trait = slothful }
		modifier = { factor = 0.5 trait = diligent }
		modifier = { factor = 0.75 trait = patient }
		modifier = { factor = 0.8 trait = humble }
		modifier = { factor = 0.9 trait = craven }
		modifier = { factor = 1.1 trait = brave }
		modifier = { factor = 0.75 trait = paranoid }
		modifier = { factor = 1.2 trait = trusting }
		modifier = { factor = 1.1 trait = drunkard }
		modifier = { factor = 0.85 trait = addict }
		# Lifestyle
		modifier = { factor = 0.75 trait = scholar }
		modifier = { factor = 0.75 trait = gardener }
		modifier = { factor = 0.75 trait = hunter }
		modifier = { factor = 0.75 trait = duelist }
		modifier = { factor = 0.75 trait = falconer }
		modifier = { factor = 0.75 trait = mystic }
		# Body and health
		modifier = { factor = 0.25 trait = ill }
		modifier = { factor = 0.25 trait = stressed }
		modifier = { factor = 0.15 trait = depressed }
		modifier = { factor = 0.33 trait = hunchback }
		modifier = { factor = 0.67 trait = clubfooted }
		modifier = { factor = 0.25 trait = pneumonic }
		modifier = { factor = 0.25 trait = dwarf }
		modifier = { factor = 0.05 trait = inbred }
		modifier = { factor = 1.5 trait = strong }
		modifier = { factor = 0.75 trait = weak }
		modifier = { factor = 0 trait = incapable }
		modifier = { factor = 0.33 trait = disfigured }
		modifier = { factor = 0.33 trait = wounded }
		modifier = { factor = 0.33 trait = maimed }
		modifier = { factor = 0.5 trait = blinded }
		modifier = { factor = 0.2 trait = is_fat }
		modifier = { factor = 0.2 trait = is_malnourished }
		# Sickness barely slows Clan Pestilens
		modifier = { factor = 0.01 trait = infirm NOT = { culture = pestilens } }
		modifier = { factor = 0.01 trait = leper NOT = { culture = pestilens } }
		modifier = { factor = 0.7 trait = infirm culture = pestilens }
		modifier = { factor = 0.7 trait = leper culture = pestilens }
		# Children
		modifier = { factor = 1.2 any_child = { is_alive = no } }
		modifier = { factor = 0.8 any_child = { count = 2 is_alive = yes } }
		modifier = { factor = 0.25 any_child = { count = 4 is_alive = yes } }
		modifier = { factor = 0 any_child = { count = 8 is_alive = yes } }
		# More breeders, more litters
		modifier = { factor = 1.2 any_courtier = { count = 4 skaven_ready_breeder_trigger = yes } }
		modifier = { factor = 1.2 any_courtier = { count = 6 skaven_ready_breeder_trigger = yes } }
		modifier = { factor = 1.2 any_courtier = { count = 8 skaven_ready_breeder_trigger = yes } }
		modifier = { factor = 1.2 any_courtier = { count = 10 skaven_ready_breeder_trigger = yes } }
		modifier = { factor = 1.2 any_courtier = { count = 12 skaven_ready_breeder_trigger = yes } }
		modifier = { factor = 1.2 any_courtier = { count = 15 skaven_ready_breeder_trigger = yes } }
		# Education
		modifier = { factor = 0.75 trait = misguided_warrior }
		modifier = { factor = 0.75 trait = tough_soldier }
		modifier = { factor = 0.75 trait = skilled_tactician }
		modifier = { factor = 0.75 trait = brilliant_strategist }
		modifier = { factor = 0.75 trait = naive_appeaser }
		modifier = { factor = 0.75 trait = underhanded_rogue }
		modifier = { factor = 0.75 trait = charismatic_negotiator }
		modifier = { factor = 0.75 trait = grey_eminence }
	}

	immediate = {
		if = {
			limit = { any_sibling = { skaven_breeding_brother_trigger = yes } }
			random_sibling = {
				limit = { skaven_breeding_brother_trigger = yes }
				save_event_target_as = breeding_rat
			}
		}
		else = { save_event_target_as = breeding_rat }
		random_courtier = {
			limit = {
				skaven_ready_breeder_trigger = yes
				NOT = { has_character_modifier = stage_progressed }
			}
			character_event = { id = skaven.2 }
		}
	}
}

# from the parents. Black and white Skaven are announced to the ruler, and
# males go to the ruler's grooming choice (skaven.6).

	id = skaven.2
	hide_window = yes
	is_triggered_only = yes

	immediate = {

			}
		}
		new_character = {
			set_mother = ROOT
			set_father = event_target:breeding_rat

			# Temperament: all Skaven are born cowards, except Vulkan daredevils
			if = {
				limit = { culture = skaven_vulkan }
				random_list = {
					25 = { add_trait = craven change_intrigue = -1 change_martial = -1 }
					70 = { add_trait = stubborn change_martial = 1 change_learning = -1 }
					5 = {
						add_trait = brave
						add_trait = stubborn

						change_learning = -2
						change_intrigue = -1

				}
			}
			else = { add_trait = craven }

			add_trait = envious # and petty

			character_event = { id = HFP.20555 }
			character_event = { id = beauty.0 }
			character_event = { id = slaanesh.462 }
			if = { limit = { is_female = yes } add_trait = skaven_breeder }

			# Fur colour
			random_list = {
				70 = { set_graphical_culture = skaven_main }
				25 = {
					modifier = { factor = 1.25 ROOT = { trait = black_skaven } }
					modifier = { factor = 1.75 FROM = { trait = black_skaven } }
					modifier = { factor = 0.5 is_female = yes }
					set_graphical_culture = skaven_black
					add_trait = black_skaven
					if = {
						limit = { is_female = no }
						FROM = { character_event = { id = skaven.3 days = 1 } }
					}
				}
				5 = {
					modifier = { factor = 1.25 ROOT = { trait = white_skaven } }
					modifier = { factor = 0 is_female = yes }
					modifier = { factor = 1.75 FROM = { trait = white_skaven } }
					modifier = { factor = 10 FROM = { character = 776 } } # Straut's litters

			# Males are groomed for a career
			if = { limit = { is_female = no } character_event = { id = skaven.6 days = 1 } }

			# Personality inherited from the parents
			random_list = {
				25 = {

				25 = {
					modifier = { factor = 0.75 ROOT = { trait = deceitful } }

				55 = { add_trait = paranoid }
				45 = {

					modifier = { factor = 0.5 FROM = { trait = paranoid } }
				}
			}

			# Skills inherited from the ruler (FROM) and the breeder (ROOT)
			if = { limit = { FROM = { diplomacy = 5 } } change_diplomacy = 2 }

				limit = { FROM = { diplomacy = 15 } }
				random = { chance = 50 change_diplomacy = 2 }

			if = {
				limit = { FROM = { diplomacy = 20 } }
				random = { chance = 50 change_diplomacy = 4 }
			}
			if = { limit = { ROOT = { diplomacy = 5 } } change_diplomacy = 2 }
			if = { limit = { ROOT = { diplomacy = 10 } } change_diplomacy = 2 }
			if = {
				limit = { ROOT = { diplomacy = 15 } }
				random = { chance = 50 change_diplomacy = 2 }
			}
			if = {
				limit = { ROOT = { diplomacy = 20 } }
				random = { chance = 50 change_diplomacy = 4 }
			}

			if = { limit = { FROM = { martial = 5 } } change_martial = 2 }
			if = { limit = { FROM = { martial = 10 } } change_martial = 2 }
			if = { limit = { FROM = { martial = 15 } } random = { chance = 75 change_martial = 2 } }

			if = {
				limit = { ROOT = { martial = 15 } }
				change_martial = 2
				random = { chance = 75 change_martial = 2 }
			}

			if = {
				limit = { FROM = { stewardship = 15 } }
				change_stewardship = 2
				random = { chance = 75 change_stewardship = 2 }
			}
			if = {
				limit = { FROM = { stewardship = 20 } }
				random = { chance = 50 change_stewardship = 4 }
			}
			if = { limit = { ROOT = { stewardship = 5 } } change_stewardship = 2 }
			if = { limit = { ROOT = { stewardship = 10 } } change_stewardship = 2 }
			if = {
				limit = { ROOT = { stewardship = 15 } }
				change_stewardship = 2
				random = { chance = 75 change_stewardship = 2 }
			}
			if = {
				limit = { ROOT = { stewardship = 20 } }
				change_stewardship = 4

			}

			if = { limit = { FROM = { intrigue = 5 } } change_intrigue = 2 }
			if = { limit = { FROM = { intrigue = 10 } } change_intrigue = 2 }

				limit = { FROM = { intrigue = 15 } }
				change_intrigue = 2

				limit = { FROM = { intrigue = 20 } }
				random = { chance = 50 change_intrigue = 4 }

# skaven.5 A Grey Seer is anointed
# A gifted young Skaven becomes a Grey Seer: lore of magic by clan, and his
# theology education is raised one step.
# Scope: ROOT is the young Skaven.
character_event = {
	id = skaven.5
	desc = EVTDESCskaven5
	picture = GFX_evt_grey_seer
	is_triggered_only = yes

	option = {
		name = EVTOPTAskaven5
		prestige = 250
		piety = 250
		add_trait = grey_seer
		if = {
			limit = { culture = eshin }
			add_trait = lore_vermin
			add_trait = lore_shadows
			graphical_culture = skavenwgreyfx
		}
		else_if = { limit = { religion = vulkan_rat } add_trait = lore_fire }
		else = { add_trait = lore_vermin graphical_culture = skavenwgreyfx }
		random = { chance = 25 add_trait = proud }
		remove_education_effect = yes
		magic_power_random_effect = yes
		enable_magic_xp_effect = yes
		if = {
			limit = { trait = mastermind_theologian }
			change_learning = 2
			change_intrigue = 1
			change_martial = -2
		}
		else_if = {
			limit = { trait = scholarly_theologian }
			remove_trait = scholarly_theologian
			add_trait = mastermind_theologian
			change_martial = -3
		}
		else_if = {
			limit = { trait = martial_cleric }
			remove_trait = martial_cleric
			add_trait = scholarly_theologian
			change_martial = -3
		}
		else_if = {
			limit = { trait = detached_priest }
			remove_trait = detached_priest
			add_trait = martial_cleric
			change_martial = -2
		}
	}
}

# skaven.31 Choose the next Seerlord
# The new holder of the Horned Church hands it on to the designated Seer Heir
# (minor title), else to a Grey Seer among his vassals or courtiers, else to a
# newly created Grey Seer. A holder who is already a Grey Seer keeps it when no
# heir was designated.
# Scope: ROOT is the new holder of the Horned Church.
character_event = {
	id = skaven.31
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_landed_title = k_horned_church
	}

	immediate = {
		if = {
			limit = { any_vassal = { has_minor_title = title_seer_heir } }
			random_vassal = {
				limit = { has_minor_title = title_seer_heir }
				character_event = { id = skaven.32 }
			}
		}
		else_if = {
			limit = { any_courtier = { has_minor_title = title_seer_heir } }
			random_courtier = {
				limit = { has_minor_title = title_seer_heir }
				character_event = { id = skaven.32 }
			}
		}
		else_if = {
			limit = { trait = grey_seer }
			# Already a Grey Seer: he keeps the Horned Church
		}
		else_if = {
			limit = { any_vassal = { trait = grey_seer } }
			random_vassal = { limit = { trait = grey_seer } character_event = { id = skaven.32 } }
		}
		else_if = {
			limit = { any_courtier = { trait = grey_seer } }
			random_courtier = { limit = { trait = grey_seer } character_event = { id = skaven.32 } }
		}
		else = {
			create_random_priest = {
				random_traits = yes
				dynasty = random
				female = no
				culture = skaven_main
				religion = horned_rat
			}
			new_character = {
				set_graphical_culture = skaven_grey
				add_trait = creature_skaven
				add_trait = white_skaven
				add_trait = grey_seer
				add_trait = lore_vermin
				magic_power_random_effect = yes
				random_list = {
					90 = { add_trait = craven }
					10 = { add_trait = zealous }
				}
				character_event = { id = skaven.32 }
			}
		}
	}
}
