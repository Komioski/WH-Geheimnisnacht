###############################################################################
# Slavery
# Namespace: slave (was slave.1 - 4 in wh_slavery_system_events.txt)
# Faction: Generic
# Serves: the slaver faiths of slaver_faith_trigger (Druchii, pirates,
#   Ogres, greenskins, Chaos and the Chaos Dwarfs, Norsca, Amazons, Araby's
#   Atruhayid and others)
#
# The full round of the slave trade. Captives come from raids (slave.1),
# from the holdings a slaver storms (slave.4, slave.5), from prisoners put
# in the pens (dawizharr.36, decision sell_into_slavery) and from the
# market. They are kept in the slaver's pens, counted in lots in the
# character variable slaves (at most 10) and shown by the pens modifier:
# their labour pays, but the more there are, the more they are feared.
# They can be sold or bought at the market (slave.2), set to work in a
# province for 15 years (slave.3), given to the gods, driven into battle
# as a slave levy, or fed to the Chaos Dwarfs' forges (decisions in
# WHG_slavery_decisions.txt). Every year the pens may bring trouble
# (slave.10): an escape (slave.11), sickness (slave.12), a slave who will
# not break (slave.13, the pit slave.14), or a rising of the slaves worked
# in a province (slave.15, slave.16). When the master dies the pens pass to
# an heir who keeps slaves, or the slaves go free (slave.17). Lords of
# faiths that keep no slaves may free the slaves worked on their land
# (slave.19) and any they hold (decision free_slave_stock).
#
# Captives                              slave.1, slave.4 - slave.5
# The slave market                      slave.2
# Slaves set to work                    slave.3
# Trouble in the pens                   slave.10 - slave.14
# The slaves rise                       slave.15 - slave.16
# The pens pass on                      slave.17
# Freedom                               slave.19
#
# Callers: on_loot_province (slave.1, 5 in 100), on_siege_over_winner
#   (slave.4), on_yearly_pulse (slave.10) and on_death (slave.17) in
#   WHG_war_on_actions.txt; the decisions openslaverymenu (slave.2) and
#   deploy_slaves_province (slave.3) in decisions/wh_general_decisions.txt;
#   free_worked_slaves (slave.19) in decisions/WHG_slavery_decisions.txt.
# Shared code: WHG_slavery_effects.txt, WHG_slavery_triggers.txt,
#   WHG_slavery_modifiers.txt (the pens).
# Saved: variable slaves; flag slave_master; province modifiers
#   slaves_taken, deployed_slaves, slaves_freed.
# Loc: localisation/00_WHG_slavery_events.csv
###############################################################################

namespace = slave

# slave.1 Captives
# The raider rounds up the people of a looted province: one lot of slaves,
# and the province is stripped of hands for five years. Faiths of Chaos,
# Khaine and the Great Maw may give the captives to their gods instead.
# Scope: ROOT is the looter, FROM the province.
character_event = {
	id = slave.1
	desc = {
		trigger = { religion_group = cytharai_group }
		text = EVTDESCslave1_druchii
	}
	desc = {
		trigger = { NOT = { religion_group = cytharai_group } }
		text = EVTDESCslave1
	}
	picture = GFX_evt_slaves
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # on_loot_province
	only_playable = yes
	capable_only = yes
	prisoner = no

	trigger = {
		in_command = yes
		slave_keeper_trigger = yes
		FROM = { slave_stock_province_trigger = yes }
	}

	# Round them up
	option = {
		name = EVTOPTAslave1
		slave_gain_lot_effect = yes
		FROM = {
			add_province_modifier = { name = slaves_taken duration = 1835 }
		}
		ai_chance = { factor = 80 }
	}

	# Give them to the gods
	option = {
		name = EVTOPTBslave1
		trigger = { slave_sacrifice_faith_trigger = yes }
		piety = 30
		FROM = {
			add_province_modifier = { name = slaves_taken duration = 1835 }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 2 trait = zealous }
			modifier = { factor = 2 trait = cruel }
		}
	}

	# Let them go
	option = {
		name = EVTOPTCslave1
		ai_chance = {
			factor = 5
			modifier = { factor = 0 NOT = { trait = kind } }
		}
	}
}

# slave.2 The slave market
# The dealers come to haggle. The market may be glutted (35 gold a lot,
# buy at 60), as usual (50, buy at 80) or hungry (75, buy at 110). Sell one
# lot, three, or all, or buy one.
# Scope: ROOT is the slaver.
character_event = {
	id = slave.2
	desc = {
		trigger = { has_character_flag = slave_market_glut }
		text = EVTDESCslave2_glut
	}
	desc = {
		trigger = { has_character_flag = slave_market_hungry }
		text = EVTDESCslave2_hungry
	}
	desc = {
		trigger = {
			NOT = { has_character_flag = slave_market_glut }
			NOT = { has_character_flag = slave_market_hungry }
		}
		text = EVTDESCslave2
	}
	picture = GFX_evt_slaves

	is_triggered_only = yes # decision openslaverymenu

	immediate = {
		hidden_effect = {
			random_list = {
				30 = { set_character_flag = slave_market_glut }
				50 = { }
				20 = {
					modifier = { factor = 1.5 war = yes }
					set_character_flag = slave_market_hungry
				}
			}
		}
	}

	option = { # Sell one lot
		name = EVTOPTAslave2
		trigger = { check_variable = { which = slaves value = 1 } }
		slave_sell_lot_effect = yes
		ai_chance = {
			factor = 1
			modifier = { factor = 0.2 has_character_flag = slave_market_glut }
		}
	}
	option = { # Sell three lots
		name = EVTOPTBslave2
		trigger = { check_variable = { which = slaves value = 3 } }
		slave_sell_lot_effect = yes
		slave_sell_lot_effect = yes
		slave_sell_lot_effect = yes
		ai_chance = {
			factor = 2
			modifier = { factor = 0.1 has_character_flag = slave_market_glut }
		}
	}
	option = { # Sell them all
		name = EVTOPTCslave2
		trigger = { check_variable = { which = slaves value = 4 } }
		custom_tooltip = { text = EVTOPTCslave2_tt }
		hidden_effect = {
			while = {
				limit = { check_variable = { which = slaves value = 1 } }
				slave_sell_lot_effect = yes
			}
		}
		ai_chance = {
			factor = 3
			modifier = { factor = 3 has_character_flag = slave_market_hungry }
			modifier = { factor = 0 has_character_flag = slave_market_glut }
		}
	}
	option = { # Buy a lot
		name = EVTOPTEslave2
		trigger = { slave_can_buy_lot_trigger = yes }
		if = {
			limit = { has_character_flag = slave_market_glut }
			wealth = -60
		}
		else_if = {
			limit = { has_character_flag = slave_market_hungry }
			wealth = -110
		}
		else = { wealth = -80 }
		slave_gain_lot_effect = yes
		ai_chance = {
			factor = 0
			modifier = {
				factor = 2
				has_character_flag = slave_market_glut
				NOT = { check_variable = { which = slaves value = 3 } }
				wealth = 300
			}
		}
	}
	option = { # Not today
		name = EVTOPTDslave2
		ai_chance = {
			factor = 1
			modifier = { factor = 3 has_character_flag = slave_market_glut }
		}
	}

	after = {
		slave_market_close_effect = yes
	}
}

# slave.10 The pens (hidden)
# Once a year the pens may bring trouble, more often the fuller they are:
# an escape, sickness, or a slave who will not break (once a life), or a
# rising where slaves are worked (slave.15). A cruel or paranoid master
# keeps them cowed better; a kind one sees fewer sicken.
# Scope: the slave master. on_yearly_pulse.
character_event = {
	id = slave.10
	hide_window = yes

	is_triggered_only = yes # on_yearly_pulse
	only_rulers = yes
	has_character_flag = slave_master

	trigger = {
		check_variable = { which = slaves value = 1 }
	}

	immediate = {
		random_list = {
			76 = { }
			8 = {
				modifier = { factor = 1.5 has_character_modifier = slave_pens_2 }
				modifier = { factor = 2 has_character_modifier = slave_pens_3 }
				modifier = { factor = 0.6 trait = cruel }
				modifier = { factor = 0.6 trait = paranoid }
				character_event = { id = slave.11 }
			}
			6 = {
				modifier = { factor = 1.5 has_character_modifier = slave_pens_2 }
				modifier = { factor = 2 has_character_modifier = slave_pens_3 }
				modifier = { factor = 0.6 trait = kind }
				character_event = { id = slave.12 }
			}
			4 = {
				trigger = { NOT = { has_character_flag = slave_champion_met } }
				modifier = { factor = 1.5 has_character_modifier = slave_pens_3 }
				character_event = { id = slave.13 }
			}
			6 = {
				trigger = {
					any_demesne_province = { has_province_modifier = deployed_slaves }
				}
				modifier = { factor = 0.5 trait = cruel }
				modifier = { factor = 1.5 trait = kind }
				random_demesne_province = {
					limit = { has_province_modifier = deployed_slaves }
					province_event = { id = slave.15 }
				}
			}
		}
	}
}

# slave.15 The slaves rise
# The slaves worked in a province break their chains: the work there is
# over, a former slave leads them against the master (slave_revolt rebels,
# a peasant revolt), and the master hears of it (slave.16). Not while the
# province is cowed (slaves_cowed) or already risen.
# Scope: ROOT is the province, its owner the master.
province_event = {
	id = slave.15
	hide_window = yes

	is_triggered_only = yes # slave.10

	trigger = {
		has_province_modifier = deployed_slaves
		NOR = {
			has_province_modifier = slaves_cowed
			has_province_modifier = recent_county_uprising
		}
		owner = {
			top_liege = {
				NOT = {
					any_war = {
						defender = { character = PREV }
						using_cb = peasant_revolt
						war_title = ROOT
					}
				}
			}
		}
	}

	immediate = {
		remove_province_modifier = deployed_slaves
		add_province_modifier = { name = recent_county_uprising duration = 1825 }
		add_province_modifier = { name = slaves_taken duration = 1835 }
		create_character = {
			random_traits = yes
			dynasty = random
			culture = averlander
			religion = sigmarite
			female = no
			age = 30
			attributes = {
				martial = 8
			}
			trait = creature_human
			trait = tough_soldier
		}
		new_character = {
			random_list = {
				30 = { }
				20 = { culture = armoric religion = grail }
				15 = { culture = miraglian religion = myrmidian }
				15 = { culture = kislevite religion = kislev_gods }
				20 = { culture = cophic religion = atruhayid }
			}
			create_title = {
				tier = DUKE
				landless = yes
				temporary = yes
				rebel = yes
				culture = ROOT
				name = "SLAVE_REVOLT"
				holder = THIS
			}
			wealth = 100
			spawn_peasant_army_effect = yes
			ROOT = {
				owner = {
					add_rival = PREVPREV
					top_liege = {
						reverse_war = {
							target = PREVPREVPREV
							casus_belli = peasant_revolt
							thirdparty_title = ROOT
						}
					}
				}
			}
		}
		owner = { character_event = { id = slave.16 } }
	}
}
