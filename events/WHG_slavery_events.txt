###############################################################################
# Slavery
# Namespace: slave (was slave.1 - 4 in wh_slavery_system_events.txt)
# Faction: Generic
# Serves: the slaver faiths of slaver_faith_trigger (Druchii, pirates,
#   Ogres, greenskins, Chaos, Norsca, Amazons, Araby's Atruhayid and others)
#
# Raiders who loot a province may carry off its people (slave.1). The
# captives are counted in the character variable slaves: each lot can be
# sold (slave.2, from the decision openslaverymenu) or set to work in one
# of the raider's own provinces (slave.3, from the decision
# deploy_slaves_province), which gives deployed_slaves for 15 years. Druchii
# raids (druchii.67 and the like) add to the same stock.
#
# Raids                                 slave.1
# The slave market                      slave.2
# Slaves set to work                    slave.3
#
# Callers: on_loot_province in wh_on_actions.txt (slave.1, 5 in 100), the
#   decisions openslaverymenu (slave.2) and deploy_slaves_province
#   (slave.3) in decisions/wh_general_decisions.txt.
# Triggers in common/scripted_triggers/WHG_slavery_triggers.txt.
# Saved: variable slaves; province modifiers slaves_taken, deployed_slaves.
# Loc: localisation/00_WHG_slavery_events.csv
###############################################################################

namespace = slave

# slave.1 Captives
# The raider rounds up the people of a looted province: one lot of slaves,
# and the province is stripped of hands for five years. Faiths of Chaos,
# Khaine and the Great Maw may give the captives to their gods instead.
# Scope: ROOT is the looter, FROM the province.
character_event = {
	id = slave.1
	desc = {
		trigger = { religion_group = cytharai_group }
		text = EVTDESCslave1_druchii
	}
	desc = {
		trigger = { NOT = { religion_group = cytharai_group } }
		text = EVTDESCslave1
	}
	picture = GFX_evt_slaves
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # on_loot_province
	only_playable = yes
	capable_only = yes
	prisoner = no

	trigger = {
		in_command = yes
		is_nomadic = no
		slaver_faith_trigger = yes
		FROM = { slave_stock_province_trigger = yes }
	}

	# Round them up
	option = {
		name = EVTOPTAslave1
		change_variable = { which = slaves value = 1 }
		FROM = {
			add_province_modifier = { name = slaves_taken duration = 1835 }
		}
		ai_chance = { factor = 80 }
	}

	# Give them to the gods
	option = {
		name = EVTOPTBslave1
		trigger = { slave_sacrifice_faith_trigger = yes }
		piety = 30
		FROM = {
			add_province_modifier = { name = slaves_taken duration = 1835 }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 2 trait = zealous }
			modifier = { factor = 2 trait = cruel }
		}
	}

	# Let them go
	option = {
		name = EVTOPTCslave1
		ai_chance = {
			factor = 5
			modifier = { factor = 0 NOT = { trait = kind } }
		}
	}
}

# slave.2 The slave market
# The slaver sells one, three or five lots of slaves at 50 gold a lot.
# Scope: ROOT is the slaver.
character_event = {
	id = slave.2
	desc = EVTDESCslave2
	picture = GFX_evt_slaves

	is_triggered_only = yes # decision openslaverymenu

	option = {
		name = EVTOPTAslave2
		trigger = { check_variable = { which = slaves value = 1 } }
		change_variable = { which = slaves value = -1 }
		wealth = 50
		ai_chance = { factor = 1 }
	}
	option = {
		name = EVTOPTBslave2
		trigger = { check_variable = { which = slaves value = 3 } }
		change_variable = { which = slaves value = -3 }
		wealth = 150
		ai_chance = { factor = 2 }
	}
	option = {
		name = EVTOPTCslave2
		trigger = { check_variable = { which = slaves value = 5 } }
		change_variable = { which = slaves value = -5 }
		wealth = 250
		ai_chance = { factor = 4 }
	}

	# Not today
	option = {
		name = EVTOPTDslave2
		trigger = { ai = no }
	}
}

# slave.3 Slaves set to work
# One lot of slaves works a province of the slaver for 15 years: more
# taxes and cheaper building, but more unrest.
# Scope: ROOT is the slaver, FROM the settlement chosen.
character_event = {
	id = slave.3
	desc = EVTDESCslave3
	picture = GFX_evt_slaves

	is_triggered_only = yes # decision deploy_slaves_province

	option = {
		name = EVTOPTAslave3
		trigger = { check_variable = { which = slaves value = 1 } }
		change_variable = { which = slaves value = -1 }
		FROM = {
			location = {
				add_province_modifier = { name = deployed_slaves duration = 5500 }
			}
		}
	}

	# Not yet
	option = {
		name = EVTOPTBslave3
		trigger = { ai = no }
	}
}
