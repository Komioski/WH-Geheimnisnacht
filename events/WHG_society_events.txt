###############################################################################
# The running of the societies
# Namespace: society (was MNM.6661 - 6668, MNM.6991 - 6992 and
#   MNM.10000 - 10045 in mnm_misc_events.txt)
# Faction: Generic
# Serves: members of every society, and the AI lords who fill them
#
# The machinery every society shares: asking the grandmaster to rise a rank
# (with a rare relic for the faithful), sending recruiters to those who show
# interest in the assassins and the Chaos cults, the strain of a criminal
# secret, letters of expulsion, rivals and friends among the members, the
# whispers that grow as a member leans on his society's powers and the day
# the secret comes out, missions that end when their target dies, and taking
# up the banner of a society with no members left.
#
# Rank and recruiting                  society.1 - society.14
# Strain and expulsion                 society.20 - society.24
# Filling the societies (AI)           society.30
# Rivals and friends                   society.40 - society.45
# Whispers and exposure                society.60 - society.65
# Missions                             society.70 - society.81
# Joining and leaving                  society.90 - society.91
#
# Callers: the rank-up decision in decisions/mnm_society_decisions.txt
#   (society.1); on_five_year_pulse (society.30), on_death (society.80),
#   on_society_bi_yearly_pulse (society.20, .40, .43),
#   on_character_switch_society_interest (society.10, .91),
#   on_character_ask_to_join_society (society.91),
#   on_character_leave_society (society.90),
#   on_character_kicked_from_society (society.21),
#   on_character_stop_showing_interest (society.24), on_quest_success
#   (society.70) and on_quest_failure (society.71) in 00_on_actions.txt; the
#   visibility effects in 00_scripted_effects.txt (society.60 - .65).
#   Triggers in common/scripted_triggers/WHG_society_triggers.txt, effects in
#   common/scripted_effects/WHG_society_effects.txt.
# Saved: event targets rank_up_target (set by the decision),
#   assassins_recruiter, chaos_cult_recruiter, cult_of_pleasure_recruiter;
#   flags has_sent_request_to_rank_up, ongoing_recruitment,
#   <society>_attempted_recruitment, being_recruited_by_<god>,
#   being_recruited_by_the_cult_of_pleasure, ineligible_for_society_rivalry,
#   ineligible_for_society_friendship, secret_societies_interference (on the
#   liege), society_join_block.
# Dropped: MNM.10025, the save-game conversion of the vanilla secretly_pagan
#   trait (already switched off in on_startup); the hidden bounce events
#   MNM.10013 - 10017 became one (society.13); the AI join roll MNM.10031
#   became society_ai_join_effect, fired straight from society.30.
###############################################################################

namespace = society

# society.1 The request is sent
# Hidden. A member who asked to rise a rank sends word to the grandmaster
# (society.2). With no grandmaster the society's elders answer him at once.
# Scope: ROOT is the member. From the rank-up decision, 20 - 30 days later.
character_event = {
	id = society.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_character_flag = has_sent_request_to_rank_up
	}

	immediate = {
		if = {
			limit = { is_in_society = no }
			clr_character_flag = has_sent_request_to_rank_up
		}
		else_if = {
			limit = {
				society = {
					any_society_member = { is_society_grandmaster = yes }
				}
			}
			random_society_member = {
				limit = {
					is_society_grandmaster = yes
					same_society_as = ROOT
				}
				character_event = { id = society.2 }
			}
		}
		else = {
			letter_event = { id = society.3 days = 7 }
		}
	}
}

# society.2 The grandmaster reads the request
# Hidden. The grandmaster grants the rank (society.3). A member rising to the
# third rank of a monastic order, the Skaven warlords or a state witch hunter
# order may also be sent a relic, once.
# Scope: ROOT is the grandmaster, FROM the member.
character_event = {
	id = society.2
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		FROM = {
			letter_event = { id = society.3 }
			if = {
				limit = {
					society_rank == 2
					is_monastic_member_trigger = yes
					NOT = { has_artifact = fingerbone_of_saint }
				}
				random = {
					chance = 10
					modifier = {
						factor = 2
						ai = no
					}
					letter_event = { id = MNM.4431 days = 30 random = 15 } # The saint's fingerbone
				}
			}
			else_if = {
				limit = {
					society_rank == 2
					society_member_of = skaven_warlords_society
				}
				random = {
					chance = 10
					modifier = {
						factor = 2
						ai = no
					}
					letter_event = { id = warband.21 days = 30 random = 15 } # The armour-pile
				}
			}
			else_if = {
				limit = {
					society_rank == 2
					is_state_witch_hunter_trigger = yes
					NOT = { has_artifact = witch_hunter_rapier }
				}
				random = {
					chance = 10
					modifier = {
						factor = 2
						ai = no
					}
					letter_event = { id = whunter.168 days = 30 random = 15 } # The rapier
				}
			}
		}
	}
}

# society.3 Leave to rise
# The grandmaster (or the elders, when there is none) grants the member the
# next rank.

letter_event = {
	id = society.3
	desc = {
		trigger = { FROM = { NOT = { character = ROOT } } }
		text = EVTDESCsociety3
	}

		trigger = { FROM = { character = ROOT } }

	}

	is_triggered_only = yes

		society_rank < 4

	}

	option = {
		name = EVTOPTAsociety3
		society_rank_up = 1
		clr_character_flag = has_sent_request_to_rank_up
	}
}

# society.12 The assassins' recruiter
# Hidden. Starts the recruitment of the assassins (MNM.6020, or MNM.6022 for
# the faithful of Ormazd and for Clan Eshin).
# Scope: ROOT is the recruiter, FROM the recruit.
character_event = {
	id = society.12
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		FROM = {
			if = {
				limit = { NOT = { has_character_flag = ongoing_recruitment } }
				set_character_flag = assassins_attempted_recruitment
				set_character_flag = ongoing_recruitment
				if = {
					limit = {
						ROOT = { society_member_of = the_assassins }
						NOT = { religion = ormazd }
					}
					character_event = { id = MNM.6020 }
				}
				else = {
					character_event = { id = MNM.6022 }
				}
			}
		}
	}
}

# society.13 The cult's recruiter
# Hidden. Starts the recruitment of a Chaos cult (chaoscult.20) and marks
# which god is calling.
# Scope: ROOT is the recruiter, FROM the recruit.
character_event = {
	id = society.13
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		FROM = {
			if = {
				limit = { ROOT = { society_member_of = cult_of_khorne } }
				set_character_flag = khorne_attempted_recruitment
				set_character_flag = being_recruited_by_khorne
			}
			else_if = {
				limit = { ROOT = { society_member_of = cult_of_nurgle } }
				set_character_flag = nurgle_attempted_recruitment
				set_character_flag = being_recruited_by_nurgle
			}
			else_if = {
				limit = { ROOT = { society_member_of = cult_of_tzeentch } }
				set_character_flag = tzeentch_attempted_recruitment
				set_character_flag = being_recruited_by_tzeentch
			}
			else_if = {
				limit = { ROOT = { society_member_of = cult_of_slaanesh } }
				set_character_flag = slaanesh_attempted_recruitment
				set_character_flag = being_recruited_by_slaanesh
			}
			else = {
				set_character_flag = undivided_attempted_recruitment
				set_character_flag = being_recruited_by_undivided
			}
			character_event = { id = chaoscult.20 }
		}
	}
}

# society.30 The societies recruit
# Hidden. Every five years an AI lord may rise a rank in his society (those
# with their own rank-up events excepted), and he and the idle courtiers of
# every lord may be drawn into a society that suits them.
# Scope: ROOT is a ruler. From on_five_year_pulse.
character_event = {
	id = society.30
	hide_window = yes
	is_triggered_only = yes
	only_rulers = yes
	has_dlc = "Mystics"

	immediate = {
		if = {
			limit = {
				ai = yes
				prisoner = no
				NOT = { trait = incapable }
				is_in_society = yes
				NOR = { # These have their own rank-up events
					society_member_of = knights_of_bretonnia
					society_member_of = ice_witch_sisterhood
					is_cult_of_morr_trigger = yes
					is_college_of_magic_member = yes
				}
				society_rank < 3
				society_can_rank_up = yes
			}
			society_rank_up = 1
		}
		if = {
			limit = { society_ai_can_join_trigger = yes }
			random = {
				chance = 5
				modifier = {
					factor = 2
					higher_tier_than = BARON
				}
				modifier = {
					factor = 2
					is_witch_hunter_trigger = yes
				}
				modifier = {
					factor = 2
					trait = ice_witch
				}
				society_ai_join_effect = yes
			}
		}
		any_courtier = {
			limit = { society_ai_can_join_trigger = yes }
			random = {
				chance = 5
				modifier = {
					factor = 2
					is_witch_hunter_trigger = yes
				}
				modifier = {
					factor = 2
					trait = ice_witch
				}
				society_ai_join_effect = yes
			}
		}
	}
}
