###############################################################################
# Spell events
# Namespace: spell (was magic_learn_power, svampires and mageproc in
# wh_magic_system_power.txt)
# Faction: Generic
#
# A mage's power and what he does with it. Power grows with experience
# (spell.1) and the odd flash of insight (spell.2). The spells themselves are
# the magic decisions in decisions/wh_magic_decisions.txt: each one pays its
# cost there and fires one of these events in the caster's scope, which does
# the work and tells him what happened. Elixirs, potions, warpstone and
# mushrooms are drunk or eaten through the same events.
#
# Power                                  spell.1 - spell.2
# Raising the dead                       spell.10 - spell.12
# Foresight                              spell.20 - spell.21
# Elixirs, potions, warpstone, mushrooms spell.25 - spell.28
# The battlefield                        spell.30 - spell.33
# The land                               spell.40 - spell.47
# Transformation                         spell.50 - spell.55
# The spell fails                        spell.60 - spell.63
# Whispers                               spell.65
# Summoned allies                        spell.70 - spell.77
#
# Callers:
#   common/scripted_effects/wh_magic_scripted_effects.txt: the magic XP
#     effects fire spell.1.
#   common/on_actions/wh_magic_on_actions.txt: spell.2 on the five-year
#     list magic_breakthrough.
#   decisions/wh_magic_decisions.txt: every spell and consumable event.
#   common/scripted_effects/wh_offmap_effects.txt: the transformation
#     effects fire spell.51 - 53 and spell.60/62, the whisper effects
#     spell.60, spawn_magic_creature_effect spell.76.
# Support: common/scripted_effects/wh_spell_effects.txt
# Loc: localisation/00_WHG_spell_events.csv
###############################################################################

# spell.10 Raise the dead
# The necromancer spends necromantic power to raise a horde: each level
# spent calls up a greater host, and the dead may answer in greater numbers
# than he asked for.
# Scope: ROOT is the necromancer (undead_summoning decision).
character_event = {
	id = spell.10
	desc = EVTDESCspell10
	picture = GFX_evt_zombies

	is_triggered_only = yes # undead_summoning

	option = { # Endless horde: all five levels
		name = EVTOPTAspell10
		trigger = {
			has_character_modifier = necromantic_power_5
			has_character_modifier = magic_power_4
		}
		while = { count = 5 spell_necro_power_down_effect = yes }
		set_character_flag = spell_horde_5
		hidden_effect = { spell_horde_size_roll_effect = yes character_event = { id = spell.11 } }
	}
	option = { # Huge horde: four levels
		name = EVTOPTBspell10
		trigger = {
			OR = {
				has_character_modifier = necromantic_power_5
				has_character_modifier = necromantic_power_4
			}
			OR = { has_character_modifier = magic_power_3 has_character_modifier = magic_power_4 }
		}
		while = { count = 4 spell_necro_power_down_effect = yes }
		set_character_flag = spell_horde_4
		hidden_effect = { spell_horde_size_roll_effect = yes character_event = { id = spell.11 } }
	}
	option = { # Large horde: three levels
		name = EVTOPTCspell10
		trigger = {
			OR = {
				has_character_modifier = necromantic_power_5
				has_character_modifier = necromantic_power_4
				has_character_modifier = necromantic_power_3
			}
		}
		while = { count = 3 spell_necro_power_down_effect = yes }
		set_character_flag = spell_horde_3
		hidden_effect = { spell_horde_size_roll_effect = yes character_event = { id = spell.11 } }
	}
	option = { # Average horde: two levels
		name = EVTOPTDspell10
		trigger = {
			OR = {
				has_character_modifier = necromantic_power_5
				has_character_modifier = necromantic_power_4
				has_character_modifier = necromantic_power_3
				has_character_modifier = necromantic_power_2
			}
		}
		while = { count = 2 spell_necro_power_down_effect = yes }
		set_character_flag = spell_horde_2
		hidden_effect = { spell_horde_size_roll_effect = yes character_event = { id = spell.11 } }
	}
	option = { # Small horde: one level
		name = EVTOPTEspell10
		spell_necro_power_down_effect = yes
		set_character_flag = spell_horde_1
		hidden_effect = { spell_horde_size_roll_effect = yes character_event = { id = spell.11 } }
	}
}

# spell.26 Eat warpstone
# Warpstone feeds a mage's power but brings madness, mutation and
# addiction; a whole cluster may kill. Necromancers instead draw on the
# death in it: dust gives one level of necromantic power, a shard three, a
# cluster all five.
# Scope: ROOT is the eater (consume_warpstone decision).
character_event = {
	id = spell.26
	desc = EVTDESCspell26
	picture = GFX_evt_warpstone_vein

	is_triggered_only = yes # consume_warpstone

	option = { # Dust
		name = EVTOPTAspell26
		trigger = { has_artifact = warpstone_dust }
		unsafe_destroy_artifact = warpstone_dust
		add_offmap_currency = { offmap = offmap_magic value = 1 }
		hidden_effect = { channeling_effect = yes }
		if = { limit = { religion_group = necromantic_group } spell_necro_power_up_effect = yes }
		else = {
			add_character_modifier = { name = warpstone_1 duration = 365 stacking = yes }
			random = {
				chance = 5
				add_character_modifier = { name = warpstone_frenzy duration = 365 stacking = yes }
			}
			random = {
				chance = 5
				custom_tooltip = { text = mutations_test_tt mutation_toughness_test_effect = yes }
			}
			hidden_effect = {
				random = { chance = 15 add_trait = addict }
				character_event = { id = mutate.5 } # Lunacy counter
			}
		}
	}
	option = { # Shard
		name = EVTOPTBspell26
		trigger = { has_artifact = warpstone_shard }
		unsafe_destroy_artifact = warpstone_shard
		add_offmap_currency = { offmap = offmap_magic value = 3 }
		hidden_effect = { channeling_effect = yes }
		if = {
			limit = { religion_group = necromantic_group }
			while = { count = 3 spell_necro_power_up_effect = yes }
		}
		else = {
			add_character_modifier = { name = warpstone_3 duration = 365 stacking = yes }
			random = {
				chance = 50
				add_character_modifier = { name = warpstone_frenzy duration = 365 stacking = yes }
			}
			random = {
				chance = 15
				custom_tooltip = { text = mutations_test_tt mutation_toughness_test_effect = yes }
			}
			hidden_effect = {
				random = { chance = 50 add_trait = addict }
				character_event = { id = mutate.5 } # Lunacy counter
			}
		}
	}
	option = { # Cluster
		name = EVTOPTCspell26
		trigger = { has_artifact = warpstone_cluster }
		unsafe_destroy_artifact = warpstone_cluster
		add_offmap_currency = { offmap = offmap_magic value = 5 }
		hidden_effect = { channeling_effect = yes }
		if = {
			limit = { religion_group = necromantic_group }
			while = { count = 5 spell_necro_power_up_effect = yes }
		}
		else = {
			add_character_modifier = { name = warpstone_5 duration = 365 stacking = yes }
			add_character_modifier = { name = warpstone_frenzy duration = 365 stacking = yes }
			random = {
				chance = 30
				custom_tooltip = { text = mutations_test_tt mutation_toughness_test_effect = yes }
			}
			hidden_effect = {
				if = {
					limit = { is_invincible_trigger = yes }
					random = { chance = 5 add_trait = addict }
				}
				else = { add_trait = addict }
				character_event = { id = mutate.5 } # Lunacy counter
			}

				limit = { is_invincible_trigger = no }
				random = { chance = 25 death = { death_reason = death_suicide_poison } }
			}
		}

	}

# spell.31 Battle magic
# The mage strikes the enemy armies in his province: men die and the rest
# lose heart. The greatest spells (damage_special_location_troops) also
# break heavy foot, pikes and knights. The damage draws on his magic pool,
# refilled for the spell.
# Scope: ROOT is the caster (damage spell decisions).
character_event = {
	id = spell.31
	desc = EVTDESCspell31
	picture = GFX_evt_mage_battle_magic

	is_triggered_only = yes # damage spell decisions
	notification = yes

	immediate = {
		save_event_target_as = magic_caster
		calc_magic_pool_effect = yes
		location = { save_event_target_as = magic_loc }
		hidden_effect = {
			any_character = {
				limit = {
					war_with = ROOT
					any_army = { location = { province = event_target:magic_loc } }
				}
				any_army = {
					limit = { location = { province = event_target:magic_loc } }
