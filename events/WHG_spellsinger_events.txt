# Spellsinger events
# Namespace: spellsinger (was spellsinger in S_spellsinger_events.txt,
# renumbered)
# Faction: Wood Elves
#
# The mages of Athel Loren. A wandering spellsinger may come to an Asrai
# court. An Asrai child raised by a spellsinger (or a forest spirit) may beg
# to be taught; the guardian takes the child as an apprentice, teaches
# virtues and then the Weave, and the apprentice chooses the light path of
# the Highweavers or the dark path of the Darkweavers. Between twenty and
# thirty-five the guardian judges whether the apprentice is ready; one who
# never is gives up at thirty-five or later. Darkweavers who sink too deep
# into darkness need a Highweaver to balance their souls.
#
# A spellsinger comes to court            spellsinger.1
# Becoming an apprentice                  spellsinger.2 - spellsinger.6
# Lessons                                 spellsinger.10 - spellsinger.15
# The judgement                           spellsinger.20 - spellsinger.25
# The Darkweaver's burden                 spellsinger.30 - spellsinger.33
#
# Callers:
#   common/on_actions/wh_on_actions.txt: spellsinger.1 on the five-year
#     random list, spellsinger.30 on on_five_year_pulse, spellsinger.25 on
#     on_decade_pulse, spellsinger.2 and spellsinger.20 on the bi-yearly
#     lists spellsinger_pupils and spellsinger_judgement.
#   decisions/wh_asrai_decisions.txt: the guardian's "take an apprentice"
#     decision fires the relay spellsinger.6.
# Loc: localisation/00_WHG_spellsinger_events.csv
###############################################################################

# spellsinger.1 A wanderer from the Weave
# An old spellsinger wanders into the glade of an Asrai ruler of Athel
# Loren who has none. Taking them in costs piety; turning them away costs
# prestige.
# Scope: ROOT is an Asrai ruler (five-year random list).
character_event = {
	id = spellsinger.1
	desc = EVTDESCspellsinger1
	picture = GFX_evt_asrai_elves

	is_triggered_only = yes # on_five_year_pulse
	only_rulers = yes
	religion = asrai_cadai
	prisoner = no

	trigger = {
		OR = { trait = creature_elf trait = creature_treekin }
		NOT = { trait = asrai_spellsinger }
		NOT = {
			any_courtier = { OR = { trait = creature_forestspirit trait = asrai_spellsinger } }
		}
		capital_scope = { empire = { title = e_athel_loren } }
	}

	option = { # Welcome
		name = EVTOPTAspellsinger1
		piety = -100
		hidden_effect = {
			random_list = {
				25 = {
					create_character = {
						random_traits = yes
						dynasty = NONE
						religion = ROOT
						culture = ROOT
						female = yes
						age = 135
					}
				}
				25 = {
					create_character = {
						random_traits = yes
						dynasty = RANDOM
						religion = ROOT
						culture = ROOT
						female = yes
						age = 261
					}
				}
				25 = {
					create_character = {
						random_traits = yes
						dynasty = RANDOM
						religion = ROOT
						culture = ROOT
						female = no
						age = 143
					}
				}
				25 = {
					create_character = {
						random_traits = yes
						dynasty = NONE
						religion = ROOT
						culture = ROOT
						female = no
						age = 257
					}
				}
			}
			new_character = {
				piety = 200
				prestige = 150
				add_trait = creature_elf
				add_trait = asrai_spellsinger
				random_list = {
					30 = { add_trait = magic_potential_1 }
					60 = { add_trait = magic_potential_2 }
					10 = { add_trait = magic_potential_3 }
				}
				random_list = {
					50 = { add_trait = lore_high_magic }
					50 = { add_trait = lore_dark_magic }
				}
				magic_power_random_effect = yes
				enable_magic_xp_effect = yes
				random_list = { # The odd scar of a long road
					2 = { add_trait = blinded }
					1 = { add_trait = disfigured }
					1 = { add_trait = arbitrary }
					1 = { add_trait = cruel }
					1 = { add_trait = just }
					1 = { add_trait = zealous }
					2 = { add_trait = cynical }
					5 = { add_trait = scarred }
					1 = { add_trait = bright }
					1 = { add_trait = quick }
					48 = { }
				}
			}
		}
	}
	option = { # Turn them away
		name = EVTOPTBspellsinger1
		prestige = -250
	}
}

# spellsinger.2 The child asks
# A bright Asrai child whose guardian is a spellsinger or a forest spirit,
# and who has no apprentice yet, begs to be taught.
# Scope: ROOT is the child (bi-yearly list spellsinger_pupils).
character_event = {
	id = spellsinger.2
	desc = EVTDESCspellsinger2
	picture = GFX_evt_apprentice

	is_triggered_only = yes # on_bi_yearly_pulse, spellsinger_pupils
	min_age = 8
	max_age = 16
	religion = asrai_cadai
	prisoner = no
	only_capable = yes

	trigger = {
		trait = creature_elf
		learning = 4
		has_guardian = yes
		is_ill = no
		NOT = { trait = mage_apprentice }
		NOT = { trait = asrai_spellsinger }
		NOT = { has_character_modifier = apprenticeship_refused }
		guardian = {
			OR = { trait = asrai_spellsinger trait = creature_forestspirit }
			NOT = { any_friend = { trait = mage_apprentice } }
		}
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 4 trait = genius }
		modifier = { factor = 2 trait = brilliant }
		modifier = { factor = 1.33 trait = bright }
		modifier = { factor = 1.33 trait = quick }
		modifier = { factor = 0.1 OR = { trait = content trait = imbecile } }
		modifier = { factor = 0.2 OR = { trait = slothful trait = idiot } }
	}

	option = {
		name = EVTOPTAspellsinger2
		hidden_effect = { guardian = { character_event = { id = spellsinger.3 } } }
	}
}

# spellsinger.3 My ward wants to learn
# The guardian takes the child on (spellsinger.4) or refuses, which may
# leave the child shy, wrathful or humbled (vanilla childhood notices).
# Scope: ROOT is the guardian, FROM the child (spellsinger.2).
character_event = {
	id = spellsinger.3
	desc = EVTDESCspellsinger3
	picture = GFX_evt_apprentice

	is_triggered_only = yes # spellsinger.2

	option = { # Accept
		name = EVTOPTAspellsinger3
		ai_chance = { factor = 1.2 }
		if = { limit = { trait = proud } prestige = 10 }
		FROM = {
			opinion = { modifier = opinion_happy_with_guardian who = ROOT years = 2 }
			hidden_effect = { character_event = { id = spellsinger.4 } }
		}
	}
	option = { # Refuse
		name = EVTOPTBspellsinger3
		ai_chance = { factor = 0.1 }
		prestige = -10
		random_list = {
			20 = { }
			40 = {
				FROM = {
					if = {
						limit = { NOR = { trait = shy trait = gregarious } }
						add_trait = shy
						hidden_effect = {
							ROOT = { character_event = { id = 38163 } }
							character_event = { id = 38263 }
						}
					}
				}
			}
			40 = {
				FROM = {
					if = {
						limit = { NOR = { trait = wroth trait = patient } }
						add_trait = wroth
						hidden_effect = {
							ROOT = { character_event = { id = 38154 } }
							character_event = { id = 38254 }
						}
					}
				}
			}
			40 = {
				FROM = {
					if = {
						limit = { NOR = { trait = humble trait = proud } }
						add_trait = humble
						hidden_effect = {
							ROOT = { character_event = { id = 38160 } }
							character_event = { id = 38260 }
						}
					}
				}
			}
		}
		FROM = {
			add_character_modifier = { name = apprenticeship_refused duration = 3650 }
			hidden_effect = { character_event = { id = spellsinger.5 } }
		}
	}
}

# spellsinger.15 The path is chosen
# The guardian is pleased if the apprentice took his own path (a forest
# spirit has none and is always pleased), and less so if not.
# Scope: ROOT is the guardian, FROM the apprentice (spellsinger.14).
character_event = {
	id = spellsinger.15
	desc = { text = EVTDESCspellsinger15 trigger = { FROM = { trait = lore_high_magic } } }
	desc = { text = EVTDESCspellsinger15b trigger = { FROM = { trait = lore_dark_magic } } }
	picture = GFX_evt_apprentice

	is_triggered_only = yes # spellsinger.14

	option = { # Proud
		name = EVTOPTAspellsinger15
		trigger = {
			OR = {
				trait = creature_forestspirit
				AND = { trait = lore_high_magic FROM = { trait = lore_high_magic } }
				AND = { trait = lore_dark_magic FROM = { trait = lore_dark_magic } }
			}
		}
		piety = 100
	}
	option = { # Disappointed
		name = EVTOPTBspellsinger15
		trigger = {
			NOR = {
				trait = creature_forestspirit
				AND = { trait = lore_high_magic FROM = { trait = lore_high_magic } }
				AND = { trait = lore_dark_magic FROM = { trait = lore_dark_magic } }
			}
		}
		piety = -100
	}
}

	max_age = 35
	religion = asrai_cadai
	only_capable = yes

		trait = mage_apprentice

	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 0.005 NOT = { hidden_or_open_magic_potential_trigger = yes } }
		modifier = { factor = 0.8 trait = magic_potential_1 }
		modifier = { factor = 2 trait = magic_potential_3 }
		modifier = { factor = 4 trait = magic_potential_4 }
		modifier = { factor = 0.5 trait = infirm }
		modifier = {
			factor = 0.67
			OR = { trait = maimed trait = wounded trait = slow trait = dim trait = stutter }
		}
		modifier = { factor = 0.2 OR = { trait = idiot trait = imbecile } }
		modifier = {
			factor = 0.8
			OR = { trait = content trait = hedonist trait = slothful trait = lisp trait = ill }
		}
		modifier = { factor = 2 trait = diligent }
		modifier = {
			factor = 1.33
			OR = { trait = patient trait = proud trait = ambitious trait = scholar trait = mystic }

		modifier = { factor = 1.33 trait = bright }
		modifier = { factor = 2 trait = smart }
		modifier = { factor = 4 trait = quick }
		modifier = { factor = 5 trait = brilliant }
		modifier = { factor = 10 trait = genius }

		set_character_flag = spellsinger_judgement_due

			limit = { OR = { trait = asrai_spellsinger trait = creature_forestspirit } }

		}
	}
}
