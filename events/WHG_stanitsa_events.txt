# Faction: Kislev

# The older peoples of Kislev. The Ungols, horse-archers of the steppe, who
# take to Ursun, Dazh and Tor easily enough. The Ropsmann, the first men of
# Kislev: foot nomads of the northern woods and steppe who stand against
# horsemen with pike and heavy foot, keep their old tongue, and resist the
# new gods and the Kislevite melting pot. Komi's proposal doc, segment 3.

# Ungols: the ataman's choosing              stanitsa.1
# Ungols: the druzhina question              stanitsa.10
# Ungols and Ropsmann: the wise woman        stanitsa.11
# Ursun comes to the stanitsas               stanitsa.20 - 26
# Ropsmann: raiders, trade, the old ways     stanitsa.50 - 53
#
# Fired from common/on_actions/wh_on_actions.txt: the five-year list
# stanitsa_life (1, 10, 11, 20 - 22, 50 - 52) and the decade list
# stanitsa_keeper (53). decisions/WHG_stanitsa_decisions.txt sends the
# missions (25) and calls the Ungol rotas.

# stanitsa.1 The ataman's choosing
# Once per ruler: an ageing Ungol lord's stanitsa wants to know which son
# will lead it. Choose by election, by duel, or by the bear-wrestling. The
# winner is the stanitsa's favoured heir; his brothers resent it. Rarely an
# Ungol horn bow for him.
# Scope: ROOT is an Ungol ruler over 50 with two adult sons;
# event_target:stanitsa_ataman the chosen son.
character_event = {
	id = stanitsa.1
	desc = EVTDESCstanitsa1
	picture = GFX_evt_steppe_mercenaries
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # on_five_year_pulse stanitsa_life

	only_rulers = yes
	min_age = 50
	culture = ungol

	trigger = {
		NOT = { has_character_flag = stanitsa_ataman_chosen }
		num_of_children = 2
		any_child = {
			count = 2
			is_adult = yes
			is_female = no
			is_alive = yes
			prisoner = no
		}
	}

	immediate = {
		set_character_flag = stanitsa_ataman_chosen
	}

	# Election
	option = {
		name = EVTOPTAstanitsa1
		random_child = {
			limit = { is_adult = yes is_female = no prisoner = no }
			preferred_limit = { diplomacy = 12 }
			preferred_limit = { diplomacy = 8 }
			save_event_target_as = stanitsa_ataman
		}
		ai_chance = { factor = 40 }
	}

	# Duel
	option = {
		name = EVTOPTBstanitsa1
		random_child = {
			limit = { is_adult = yes is_female = no prisoner = no }
			preferred_limit = { martial = 12 }
			preferred_limit = { martial = 8 }
			save_event_target_as = stanitsa_ataman
		}
		random_child = {
			limit = {
				is_adult = yes
				is_female = no
				NOT = { character = event_target:stanitsa_ataman }
			}
			random = { chance = 30 add_trait = wounded }
		}
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = wroth }
		}
	}

	# Bear-wrestling
	option = {
		name = EVTOPTCstanitsa1
		random_child = {
			limit = { is_adult = yes is_female = no prisoner = no }
			preferred_limit = { trait = strong }
			preferred_limit = { trait = robust }
			save_event_target_as = stanitsa_ataman
		}
		event_target:stanitsa_ataman = {
			random = { chance = 20 add_trait = wounded }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 2 religion = kislev_gods }
		}
	}

	after = {
		event_target:stanitsa_ataman = {
			prestige = 100
			add_character_modifier = { name = stanitsa_ataman_chosen duration = -1 }
			if = {
				limit = {
					NOT = { has_character_flag = stanitsa_bow_given }
					NOT = { has_artifact = stanitsa_ungol_horn_bow }
				}
				random = {
					chance = 10
					set_character_flag = stanitsa_bow_given
					add_artifact = stanitsa_ungol_horn_bow
				}
			}
		}
		any_child = {
			limit = {
				is_adult = yes
				is_female = no
				NOT = { character = event_target:stanitsa_ataman }
			}
			opinion = { who = event_target:stanitsa_ataman modifier = opinion_stanitsa_passed_over }
		}
	}
}

# Ungols: the druzhina question
###############################################################################

# stanitsa.10 An Ungol asks for the druzhina
# An Ungol vassal asks a Gospodar or Kislevite lord to be raised to the
# druzhina and seated in the Boyar Duma. Raise him (the Gospodar boyars
# resent it), or refuse (the Ungols resent that).
# Scope: ROOT is a Gospodar or Kislevite ruler; event_target:stanitsa_ungol
# the vassal.
character_event = {
	id = stanitsa.10
	desc = EVTDESCstanitsa10
	picture = GFX_evt_kislev_druzhina
	border = GFX_event_normal_frame_diplomacy

	is_triggered_only = yes # on_five_year_pulse stanitsa_life

	only_rulers = yes

	trigger = {
		OR = {
			culture = gospodar
			culture = kislevite
		}
		any_vassal = {
			culture = ungol
			is_adult = yes
			prisoner = no
			NOT = { society_member_of = boyar_duma }
			NOT = { has_character_flag = stanitsa_druzhina_asked }
		}
	}

	immediate = {
		random_vassal = {
			limit = {
				culture = ungol
				is_adult = yes
				prisoner = no
				NOT = { society_member_of = boyar_duma }
				NOT = { has_character_flag = stanitsa_druzhina_asked }
			}
			set_character_flag = stanitsa_druzhina_asked
			save_event_target_as = stanitsa_ungol
		}
	}

	# Raise him
	option = {
		name = EVTOPTAstanitsa10
		event_target:stanitsa_ungol = {
			opinion = { who = ROOT modifier = opinion_stanitsa_druzhina_raised }
			if = {
				limit = { NOT = { society_member_of = boyar_duma } }
				join_society = boyar_duma
			}
		}
		any_vassal = {
			limit = {
				culture = gospodar
				society_member_of = boyar_duma
			}
			opinion = { who = ROOT modifier = opinion_kislevsoc_upstart_raised }
		}
		ai_chance = {
			factor = 50
			modifier = { factor = 2 culture = kislevite }
		}
	}

	# Refuse
	option = {
		name = EVTOPTBstanitsa10
		any_vassal = {
			limit = { culture = ungol }
			opinion = { who = ROOT modifier = opinion_stanitsa_druzhina_refused }
		}
		ai_chance = {
			factor = 50
			modifier = { factor = 2 trait = proud }
		}
	}
}

# stanitsa.11 The wise woman
# An old Ungol wise woman, a hag of the steppe, comes to court. Take her in
# (a hag-witch courtier), ask her to read the omens, or drive her off (she
# may lay the Hag's mark on the ruler).
# Scope: ROOT is an Ungol or Ropsmann ruler, or a Kislevite one with an
# Ungol or Ropsmann capital.
character_event = {
	id = stanitsa.11
	desc = EVTDESCstanitsa11
	picture = GFX_evt_witch
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_five_year_pulse stanitsa_life

	only_rulers = yes

	trigger = {
		culture_group = kislevite_group
		OR = {
			culture = ungol
			culture = ropsmann
			capital_scope = {
				OR = {
					culture = ungol
					culture = ropsmann
				}
			}
		}
		NOT = { any_courtier = { has_character_flag = stanitsa_wise_woman } }
	}

	# Take her in
	option = {
		name = EVTOPTAstanitsa11
		create_character = {
			random_traits = yes
			dynasty = none
			religion = ROOT
			culture = ungol
			female = yes
			age = 70
			attributes = { learning = 13 intrigue = 8 }
			trait = creature_human
			trait = lore_hag
			trait = magic_potential_2
		}
		new_character = { set_character_flag = stanitsa_wise_woman }
		ai_chance = { factor = 50 }
	}

	# Read the omens
	option = {
		name = EVTOPTBstanitsa11
		piety = 15
		ai_chance = { factor = 30 }
	}

	# Drive her off
	option = {
		name = EVTOPTCstanitsa11
		prestige = 5
		random = {
			chance = 30
			add_character_modifier = { name = kislev_hag_marked years = 10 }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 3 trait = zealous }
		}
	}
}
