# The stars and the doom
# Namespace: stars (was wh_star_events.txt)
# Faction: Old World faiths
#
# Every child of the Empire, Estalia and Tilea is born under one of the
# twenty signs of the Old World sky, and at nine or ten a priest of Morr
# speaks its doom: the manner of its death, in a single line of verse. At
# twelve the child burns its toys, and the children of Ulric and Taal go
# out alone into the wild for a year, the Quickening, to come home grown or
# not at all. A child whose horoscope was cast at birth by the Order of
# Mysteries (verena.130) and encouraged hears a doom that follows its
# stars, and the stars and the doom read together lend it a little more of
# the foretold gift. Lore from WFRP 2E Tome of Salvation (Morr's dooming)
# and The WFRP Companion (the star signs).
#
# Star signs                           stars.0, stars.21
# The burning of the toys              stars.1 - stars.4
# The Quickening (Ulric and Taal)      stars.5 - stars.18
# The dooming                          stars.19, stars.20, stars.22
#
# Callers:
#   stars.0 on_startup (rulers and their courtiers)
#   stars.21 on_birth and aiclean.31 (new_character_effect)
#   stars.1 on_adolescence
#   stars.19 on_startup and new_character_effect (nine or older)
#   stars.20 on_yearly_childhood_pulse (nine to eleven)
#   stars.22 from stars.19 and stars.20, to the horoscope maker
# Loc: localisation/00_WHG_star_events.csv
###############################################################################

namespace = stars

# stars.0 Star signs at the start
# Gives a ruler and their courtiers the sign they were born under.
# Scope: a ruler.
character_event = {
	id = stars.0
	hide_window = yes

	religion_group = old_world_group
	only_rulers = yes
	is_triggered_only = yes

	trigger = {
		NOR = {
			religion = kislev_gods
			religion = northern_gods
			religion = old_faith
		}
	}

	immediate = {
		set_star_sign_effect = yes
		any_courtier = {
			limit={
				religion_group = old_world_group
				NOR = {
					religion = kislev_gods
					religion = northern_gods
					religion = old_faith
				}
			}
			set_star_sign_effect = yes
		}
	}

	option = {
		name = OK
	}
}

# stars.1 Coming of age
# Hidden. At twelve a child of an Old World faith comes of age: a child of
# Ulric or Taal goes out on the Quickening (stars.5), any other burns its
# toys (stars.2).
# Scope: the child. Fired from on_adolescence.

	id = stars.1
	hide_window = yes

	religion_group = old_world_group

	is_triggered_only = yes

	trigger = {
		NOR = {
			religion = kislev_gods
			religion = northern_gods
			religion = old_faith
		}
	}

	immediate = {
		if = {
			limit = {
				wild_religion_empire_trigger = yes
			}
			# Begin Primary Ritual for old world group wild religions
			character_event = {
				id = stars.5
				days = 5
				random = 5

			# Begin Primary Ritual for old world group

				days = 5
				random = 5
			}
		}
	}

	option = {
		name = OK
	}
}

# stars.8 A trial in the wild
# Hidden. Once the child is thirteen it comes home (stars.7). Before that it
# meets travellers (stars.10), a lost hunter (stars.11) or a wounded child
# (stars.12), or faces hunger (stars.14) or a wound (stars.17); the last two
# come twice as often to the starving or feeble.
# Scope: the child (stars.5, four times over the year).
character_event = {
	id = stars.8
	hide_window = yes

	religion_group = old_world_group
	has_character_flag = quickening

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				age >= 13
			}
			# Return home safe and sound
			character_event = {
				id = stars.7
				random = 25
			}
		}
		else = {
			random_list = {
				# Traveler nearby - potential food source if you handle this right... or you could starve
				20 = {
					character_event = {
						id = stars.10
						days = 5
						random = 5
					}
				}
				# Meet a Lost hunter - potential food source if you handle this right... or you could starve
				20 = {
					character_event = {
						id = stars.11
						days = 5
						random = 5
					}
				}
				# A Dieing Kid - potential food source if you handle this right... or you could starve
				20 = {
					character_event = {
						id = stars.12
						days = 5
						random = 5
					}
				}
				# Need for Food - forage, hunt, or cry about it
				20 = {
					modifier = {
						factor = 2
						OR = {
							trait = is_malnourished
							trait = feeble
						}
					}
					character_event = {
						id = stars.14
						days = 5
						random = 5
					}
				}
				# Being Wounded
				20 = {
					modifier = {
						factor = 2
						OR = {
							trait = is_malnourished
							trait = feeble
						}
					}
					# Being Wounded
					character_event = {
						id = stars.17
						days = 5
						random = 5
					}
				}
			}
		}
	}

	option = {
		name = OK
	}
}
