# Steppe wargames, sky burials and wrestling
# Namespace: steppe (was HL.7000 - HL.7099 in hl_tournament_events.txt and
#   HLF.8000 - HLF.9004 in hl_extra_events.txt)
# Faction: Generic
# Serves: the horse peoples of the steppe (Kurgan, Hung, Ungol), Chaos
#   hordes, Kislev's tzars and the faithful of Ormazd
#
# Wargames: a lord invites the warriors of the realm to compete with horse,
# bow and sword; the best of them takes gold and glory.
#
# Sky burial: the heir of a dead lord of the steppe peoples, Chaos, Ormazd,
# Kislev or a sky-burial bloodline carries the body to the mountains for
# the birds. On the way: a storm, a hermit, and perhaps a bearded vulture
# that will not leave. A famous dead lord may instead get a golden tomb,
# and a madman may carry the body over the mountains to feed it to the
# Great Maw of the Ogre Kingdoms.
#
# Wrestling: a training bout among the warriors of a horde turns into a
# show, and the khan is called into the ring.
#
# Wargames                             steppe.1 - steppe.9
# Sky burial                           steppe.20 - steppe.32
# Wrestling                            steppe.40 - steppe.44
#
# Fired from: the hold_nomad_wargames decision in realm_decisions.txt
#   (steppe.1), the sky_burial_decision in minor_decisions.txt (steppe.22),
#   on_death in 00_on_actions.txt (steppe.20) and the steppe_wrestling list
#   on on_five_year_pulse in wh_on_actions.txt (steppe.40). Shared triggers
#   are in WHG_horse_lords_triggers.txt.
###############################################################################

# steppe.20 Who carries the body?
# Hidden. When a lord of a sky-burial people dies, the heir is told whose
# body to carry (sky_burial flags, cleared after sixty days by steppe.21) and
# may take the sky_burial_decision. The heir of a very famous lord may raise
# a golden tomb instead.
# Scope: the dead lord.
character_event = {
	id = steppe.20
	hide_window = yes
	is_triggered_only = yes # on_death

	only_playable = yes

	trigger = {
		OR = {
			has_dlc = "Horse Lords"
			any_owned_bloodline = { has_bloodline_flag = bloodline_sky_burial }
		}
		death_in_cathay_trigger = no
		death_no_trace_trigger = no
		steppe_sky_burial_people_trigger = yes
	}

	immediate = {
		current_heir = {
			if = {
				limit = { ROOT = { prestige = 20000 } }
				set_character_flag = sky_burial_tomb
			}
			if = {
				limit = { is_child_of = ROOT }
				if = {
					limit = { ROOT = { is_woman_trigger = yes } }
					set_character_flag = sky_burial_mother
				}
				else = {
					set_character_flag = sky_burial_father
				}
			}
			else_if = {
				limit = { sibling = ROOT }
				if = {
					limit = { ROOT = { is_woman_trigger = yes } }
					set_character_flag = sky_burial_sister
				}
				else = {
					set_character_flag = sky_burial_brother
				}
			}
			set_character_flag = sky_burial
			character_event = { id = steppe.21 days = 60 }
		}
	}
}

# steppe.42 The bout
# Hidden. Strength against strength: the stronger usually wins.
# Scope: the opponent; FROM the lord.
character_event = {
	id = steppe.42
	hide_window = yes
	is_triggered_only = yes # steppe.41

	immediate = {
		if = { # The lord is stronger
			limit = {
				OR = {
					AND = {
						FROM = { is_strong_trigger = yes }
						is_strong_trigger = no
					}
					AND = {
						FROM = { is_weak_trigger = no }
						is_weak_trigger = yes
					}
				}
			}
			random_list = {
				95 = { FROM = { long_character_event = { id = steppe.43 } } }
				5 = { FROM = { long_character_event = { id = steppe.44 } } }
			}
		}
		else_if = { # The opponent is stronger
			limit = {
				OR = {
					AND = {
						FROM = { is_weak_trigger = yes }
						is_weak_trigger = no
					}
					AND = {
						FROM = { is_strong_trigger = no }
						is_strong_trigger = yes
					}
				}
			}
			random_list = {
				5 = { FROM = { long_character_event = { id = steppe.43 } } }
				95 = { FROM = { long_character_event = { id = steppe.44 } } }
			}
		}
		else_if = { # Both weak
			limit = {
				FROM = { is_weak_trigger = yes }
				is_weak_trigger = yes
			}
			random_list = {
				50 = { FROM = { long_character_event = { id = steppe.43 } } }
				50 = { FROM = { long_character_event = { id = steppe.44 } } }
			}
		}
		else = { # Evenly matched
			random_list = {
				70 = { FROM = { long_character_event = { id = steppe.43 } } }
				30 = { FROM = { long_character_event = { id = steppe.44 } } }
			}
		}
	}
}
