# Way of Life: the theology focus
# Namespace: theology (was WoL.7000 - WoL.7092 in wol_theology_events.txt)
# Namespace: zeal (Part 2, was SoA.5400 - SoA.5425 in
#   soa_misc_religious_events.txt)
# Faction: Generic
#
# A ruler with the theology focus studies the holy books or the old tales of
# their faith, befriends its priests, visits its shrines, doubts, burns with
# zeal, and may end as a theologian. The faiths of the Old World, the Dawi,
# the elves, Araby, Nehekhara and Cathay study written scripture; Chaos, the
# Norse, the greenskins, the Horned Rat and the beast cults learn from tales
# (theology_scripture_faith_trigger in WHG_focus_triggers.txt). Mystic
# visions and the hermit's cave speak in the voice of the faith: Chaos,
# Araby, the Dawi and the greenskins have their own.
#
# Study                               theology.1 - theology.2
# A friend among the priests          theology.10 - theology.12
# The holy site                       theology.20 - theology.21
# Doubt, zeal and other gods          theology.30 - theology.50
# The theologian                      theology.70
# Visions and seclusion               theology.80 - theology.92
# Part 2: tolerance and zealotry      zeal.1 - zeal.10
#
# Fired from: on_focus_pulse in WHG_pulse_on_actions.txt (theology.1, 10, 20, 30,
#   40, 50, 70, 80, 90). Shared triggers and effects are in
#   WHG_focus_triggers.txt and WHG_focus_effects.txt.

# theology.1 Study the myths
# The ruler may spend the next months on the faith's holy books, or, in a
# faith without them, on the tales its priests tell. Studying leads to
# theology.2; refusing loses a little piety and any study begun.
# Scope: ROOT is the ruler.
character_event = {
	id = theology.1
	desc = { trigger = { theology_scripture_faith_trigger = yes } text = EVTDESCtheology1 }
	desc = { trigger = { theology_scripture_faith_trigger = no } text = EVTDESCtheology1b }
	picture = GFX_evt_bishop
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_focus_pulse

	only_playable = yes
	min_age = 16
	only_capable = yes
	prisoner = no

	trigger = {
		has_focus = focus_theology
		is_priest = no
		holy_order = no
		NOT = { is_inaccessible_trigger = yes }
		NOT = { has_character_flag = studied_religious_myths }
	}

	immediate = {
		set_character_flag = studied_religious_myths
	}

	option = { # Read the scriptures
		name = EVTOPTAtheology1
		trigger = { theology_scripture_faith_trigger = yes }
		piety = 20
		hidden_tooltip = { character_event = { id = theology.2 days = 90 } }
	}
	option = { # Summon the storytellers
		name = EVTOPTBtheology1
		trigger = { theology_scripture_faith_trigger = no }
		piety = 20
		hidden_tooltip = { character_event = { id = theology.2 days = 90 } }
	}
	option = { # No time
		name = EVTOPTCtheology1
		piety = -5
		if = {
			limit = { has_character_modifier = religious_studies }
			remove_character_modifier = religious_studies
		}
	}
}

# theology.20 A holy site in the realm
# One of the faith's holy sites lies in the ruler's realm, held by another.
# A visit costs some gold and is answered in theology.21.
# Scope: ROOT is the ruler.
character_event = {
	id = theology.20
	desc = EVTDESCtheology20
	picture = GFX_evt_cathedral
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_focus_pulse

	only_playable = yes
	min_age = 16
	only_capable = yes
	prisoner = no
	war = no

		has_focus = focus_theology
		in_command = no
		NOT = { is_inaccessible_trigger = yes }
		NOT = { has_character_flag = do_not_disturb }

		}

		set_character_flag = wol_theology_visited_holy_site

			limit = {
				is_holy_site = ROOT

			}

		}
	}

	option = { # Go
		name = EVTOPTAtheology20
		piety = 20
		scaled_wealth = -0.2
		set_character_flag = do_not_disturb
		hidden_tooltip = { character_event = { id = theology.21 days = 20 } }
	}
	option = { # Stay
		name = EVTOPTBtheology20
		piety = -5
	}

# theology.21 At the holy site
# The visit brings piety, perhaps zeal (or an end to cynicism), a step on
# the theologian's road, and rarely a shrine token to bring home
# (theology_shrine_token_effect).
# Scope: ROOT is the ruler.
character_event = {
	id = theology.21
	desc = EVTDESCtheology21
	picture = GFX_evt_cathedral
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # theology.20

	option = {
		name = EVTOPTAtheology21
		piety = 50
		if = {
			limit = { NOR = { trait = cynical trait = zealous } }
			random = {
				chance = 20
				add_trait = zealous
				character_event = { id = traitnote.269 } # Notify zealous
			}
		}
		else_if = {
			limit = { trait = cynical }
			random = {
				chance = 20
				remove_trait = cynical
				character_event = { id = traitnote.316 } # Notify cynical lost
			}
		}
		theology_study_step_effect = yes
		theology_shrine_token_effect = yes
		clr_character_flag = do_not_disturb
		clear_event_target = holy_site_visiting
	}
}

# theology.50 The silent gods
# A believer of a scripture faith no longer feels the gods. They may turn
# cynical (a zealot loses the zeal instead), hold to the faith at the risk of
# melancholy, or burn with new zeal (a cynic loses the cynicism instead).
# Scope: ROOT is the ruler.
character_event = {
	id = theology.50
	desc = EVTDESCtheology50
	picture = GFX_evt_religious_exultation
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # on_focus_pulse

	only_playable = yes
	min_age = 16
	only_capable = yes

	trigger = {
		has_focus = focus_theology
		theology_scripture_faith_trigger = yes
		NOT = { trait = cynical }
		NOT = { has_character_flag = wol_theology_question_gods }
	}

	immediate = {
		set_character_flag = wol_theology_question_gods
	}

	option = { # The gods are dead
		name = EVTOPTAtheology50
		ai_chance = { factor = 10 }
		if = {
			limit = { trait = zealous }
			remove_trait = zealous
		}
		else = { add_trait = cynical }
	}
	option = { # It will pass
		name = EVTOPTBtheology50
		ai_chance = { factor = 10 }
		piety = 100
		random = {
			chance = 30
			add_trait = depressed
			character_event = { id = traitnote.288 } # Notify depressed
		}
	}
	option = { # Perish the thought
		name = EVTOPTCtheology50
		ai_chance = { factor = 100 }
		trigger = { NOT = { trait = zealous } }
		add_trait = zealous
	}
}

# theology.70 A scholar of the faith

# Tolerance and zealotry
# Namespace: zeal (Part 2, was SoA.5400 - SoA.5425 in
#   soa_misc_religious_events.txt)
# Serves: every faith; Araby (zeal.3); holy warriors (zeal.10)
#
# A ruler who sympathises with another faith makes a friend among its
# followers at court; a zealous, wrathful and tactless lord insults every
# unbeliever in the realm; Arabyan lords, long used to Old World merchants
# in their ports, grow tolerant of the Old World faiths; and a holy warrior
# who storms a heretic city may let the soldiers kill them all. Vanilla
# keyed the first three to Christians, Muslims, Jews, pagans and
# Zoroastrians, so they never fired.
#
# Friends of other faiths                 zeal.1
# The tactless zealot                     zeal.2
# Arabyan tolerance                       zeal.3
# Kill them all                           zeal.10
#
# Callers: WHG_pulse_on_actions.txt (on_five_year_pulse zeal_tolerance list: 1 - 3),
#   WHG_war_on_actions.txt (on_siege_over_winner: 10). Triggers in
#   WHG_old_world_faith_triggers.txt.
# Saved: event target zeal_sacked_province (zeal.10).
###############################################################################

# zeal.1 A friend of another faith
# A ruler with sympathy for another faith befriends one of its followers at
# court. Was five MTTH events of 300 months, one per vanilla faith.
# Scope: ROOT is the ruler. From on_five_year_pulse (zeal_tolerance).
character_event = {
	id = zeal.1
	desc = EVTDESCzeal1
	picture = GFX_evt_courtiers_talking
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes
	only_playable = yes
	prisoner = no
	only_capable = yes
	min_age = 16

	trigger = {
		any_courtier = {
			zeal_possible_friend_trigger = yes
		}
	}

	immediate = {
		random_courtier = {
			limit = { zeal_possible_friend_trigger = yes }
			save_event_target_as = zeal_new_friend
		}
	}

	option = {
		name = EVTOPTAzeal1
		event_target:zeal_new_friend = {
			add_friend = ROOT
		}
	}
}
