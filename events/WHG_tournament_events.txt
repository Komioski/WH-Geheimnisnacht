###############################################################################
# The Bretonnian tourney
# Namespace: tourney (was tourney.1 - tourney.800 and 70001 - 70999 in the
#   mod's tournament_events.txt, itself the A Game of Thrones tourney adapted
#   by DarkReborn)
# Faction: Bretonnia
# Serves: lords of the Lady's faith, count tier and up; knights of Bretonnia
#   and the Lady's faithful ride, lords watch
#
# A lord of the Lady proclaims a tourney: a local one for the knights of the
# neighbourhood, a regional one for a Bretonnian dukedom, or a grand tourney
# for the whole realm. Knights ride in a knock-out of up to sixteen. Each
# tilt is decided by skill at arms, the Lady's blessing and the odd purse of
# gold; the loser is unhorsed, hurt, crippled or now and then killed. The
# champion, the runner-up and the two beaten in the semi-finals win medals,
# gold and fame, and the champion of a regional or grand tourney names the
# Queen of Love and Beauty and may carry off a charger or the Champion's
# Lance. Around the lists, guests find or lose their nerve, take to drink,
# learn or forget things, are knighted, quarrel, take squires and meet the
# Damsels of the Lady.
#
# The tourney is proclaimed            tourney.1 - tourney.3
# Opening, war, the host's death       tourney.10 - tourney.15
# The lists                            tourney.20 - tourney.29
# The champion                         tourney.30 - tourney.37
# The Queen of Love and Beauty         tourney.40 - tourney.46
# Around the lists                     tourney.50 - tourney.62
#
# Fired from: the decision hold_a_epic_tournament in realm_decisions.txt
#   (tourney.1); on_death in 00_on_actions.txt (tourney.12). Everything else
#   is chained. Triggers in common/scripted_triggers/WHG_tournament_triggers.txt,
#   effects in common/scripted_effects/WHG_tournament_effects.txt, the
#   Champion's Lance in common/artifacts/43_artifacts_tourney.txt.
# Saved: event targets tourney_host, tourney_foe, tourney_rider_a,
#   tourney_rider_b, tourney_tilt_winner, tourney_tilt_loser,
#   tourney_champion, tourney_queen, tourney_queen_1 - 4, tourney_squire,
#   tourney_boaster; host flags tourney_host, tourney_running, tourney_local,
#   tourney_regional, tourney_grand, tourney_semi_round, tourney_final_round,
#   epic_tournament (ten-year cooldown); guest flags tourney_rider,
#   tourney_watcher, tourney_out, tourney_paired, tourney_tilting,
#   tourney_bribed, tourney_blessed, tourney_first/second/third,
#   tourney_fate_*, tourney_lesson_*, tourney_no_msg_fallen; records
#   grand_tourny_1st and the rest, tourney_prize_won (once per life);
#   the medal modifiers grand_tourny_1st_1 - small_tourny_3rd_5; opinion
#   links opinion_attending_tournament and opinion_spectating_tournament
#   between each guest and the host; variable tourney_waits on the host.
# Dropped: the Fay Enchantress as dice roller (tourney.6/7, tourney.15: the
#   events went to the holder of k_cult_of_the_lady, so a player holding it
#   had to pick the winners); the AI-only shortcut tourney.550, which looked
#   for flags and opinions of the melee (attending_melee) that the joust never
#   set and so found no champion; the medal roll events 70078 - 70081 (no
#   caller); the dead AGOT leftovers (knight_character_effect, maesters).
#   Salvaged: the squire events (tourney.45 - 49) and the boasting rival
#   (70061 - 70073), both commented out, are tourney.57 - 61.
###############################################################################

# tourney.1 The heralds are ready
# The host chooses the size of the tourney or thinks better of it. A local
# tourney is open to counts and up, a regional one to dukes and up, a grand
# one to kings and up. The lists open in 30 days (tourney.10).
# Scope: ROOT is the host.
character_event = {
	id = tourney.1
	title = EVTNAMEtourney
	desc = EVTDESCtourney1
	picture = GFX_evt_courier

	is_triggered_only = yes # decision hold_a_epic_tournament

	immediate = {
		save_event_target_as = tourney_host
		set_character_flag = tourney_host
	}

	option = {
		name = EVTOPTAtourney1 # Local
		trigger = { treasury = 75 }
		ai_chance = {
			factor = 10
			modifier = {
				factor = 0.1
				higher_tier_than = DUKE
			}
		}
		treasury = -75
		prestige = 40
		set_character_flag = tourney_local
		set_character_flag = tourney_running
		custom_tooltip = { text = TOOLTIPtourney1_local }
		hidden_effect = {
			tourney_invite_effect = yes
			character_event = { id = tourney.10 days = 30 }
		}
	}
	option = {
		name = EVTOPTBtourney1 # Regional
		trigger = {
			higher_tier_than = COUNT
			treasury = 150
		}
		ai_chance = {
			factor = 20
			modifier = {
				factor = 0
				NOT = { treasury = 250 }
			}
		}
		treasury = -150
		prestige = 80
		set_character_flag = tourney_regional
		set_character_flag = tourney_running
		custom_tooltip = { text = TOOLTIPtourney1_regional }
		hidden_effect = {
			tourney_invite_effect = yes
			character_event = { id = tourney.10 days = 30 }
		}
	}
	option = {
		name = EVTOPTCtourney1 # Grand
		trigger = {
			higher_tier_than = DUKE
			treasury = 350
		}
		ai_chance = {
			factor = 40
			modifier = {
				factor = 0
				NOT = { treasury = 500 }
			}
			modifier = {
				factor = 2
				OR = {
					trait = ambitious
					trait = proud
				}
			}
		}
		treasury = -350
		prestige = 175
		set_character_flag = tourney_grand
		set_character_flag = tourney_running
		custom_tooltip = { text = TOOLTIPtourney1_grand }
		hidden_effect = {
			tourney_invite_effect = yes
			character_event = { id = tourney.10 days = 30 }
		}
	}
	option = {
		name = EVTOPTDtourney1 # Change my mind
		ai_chance = { factor = 0 }
		clr_character_flag = epic_tournament
		tourney_clear_host_effect = yes
	}
}

# tourney.2 An invitation to ride
# A rider is asked to the lists. Declining costs a knight more than a lord
# who outranks the host. A player may ask never to be bothered again with
# this size or with any tourney.
# Scope: ROOT is the rider, FROM the host.
letter_event = {
	id = tourney.2
	title = EVTNAMEtourney
	desc = {
		trigger = { FROM = { has_character_flag = tourney_local } }
		text = EVTDESCtourney2_local
	}
	desc = {
		trigger = { FROM = { has_character_flag = tourney_regional } }
		text = EVTDESCtourney2_regional
	}
	desc = {
		trigger = { FROM = { has_character_flag = tourney_grand } }
		text = EVTDESCtourney2_grand
	}

	is_triggered_only = yes # tourney_send_invitation_effect

	trigger = {
		FROM = { has_character_flag = tourney_running }
		prisoner = no
		war = no
		is_incapable = no
		NOR = {
			has_character_flag = tourney_rider
			has_character_flag = tourney_watcher
			has_character_flag = tourney_host
			has_character_flag = no_tournies_ever
			AND = {
				has_character_flag = no_small_tournies
				FROM = { has_character_flag = tourney_local }
			}
			AND = {
				has_character_flag = no_regional_tournies
				FROM = { has_character_flag = tourney_regional }
			}
		}
	}

	option = {
		name = EVTOPTAtourney2 # I will ride
		ai_chance = {
			factor = 60
			modifier = {
				factor = 2
				is_member_of_bretonnia_trigger = yes
			}
			modifier = {
				factor = 2
				combat_rating = 50
			}
			modifier = {
				factor = 1.5
				OR = {
					trait = brave
					trait = duelist
					trait = ambitious
					trait = proud
					trait = gregarious
				}
			}
			modifier = {
				factor = 2
				OR = {
					has_character_flag = grand_tourny_1st
					has_character_flag = normal_tourny_1st
					has_character_flag = small_tourny_1st
				}
			}
			modifier = {
				factor = 1.5
				opinion = { who = FROM value = 25 }
			}
			modifier = {
				factor = 0.5
				NOT = { opinion = { who = FROM value = -25 } }
			}
			modifier = {
				factor = 0.3
				OR = {
					trait = craven
					trait = shy
					trait = slothful
					trait = wounded
					is_ill = yes
				}
			}
		}
		if = {
			limit = { FROM = { has_character_flag = tourney_grand } }
			prestige = 20
		}
		else_if = {
			limit = { FROM = { has_character_flag = tourney_regional } }
			prestige = 10
		}
		else = {
			prestige = 5
		}
		tourney_join_lists_effect = yes
		if = {
			limit = { trait = in_hiding }
			remove_trait = in_hiding
			add_character_modifier = {
				name = went_out_of_hiding_timer
				duration = 180
				hidden = yes
			}
			hidden_effect = { character_event = { id = hiding.7 } } # Plotters and kin hear
		}
	}
	option = {
		name = EVTOPTBtourney2 # Regretfully, no
		trigger = { lower_tier_than = FROM }
		ai_chance = { factor = 30 }
		if = {
			limit = {
				is_member_of_bretonnia_trigger = yes
				is_ill = no
			}
			if = {
				limit = { FROM = { has_character_flag = tourney_grand } }
				prestige = -30
			}
			else_if = {
				limit = { FROM = { has_character_flag = tourney_regional } }
				prestige = -20
			}
			else = {
				prestige = -10
			}
		}
	}
	option = {
		name = EVTOPTCtourney2 # This is below me
		trigger = { NOT = { lower_tier_than = FROM } }
		ai_chance = { factor = 60 }
	}
	option = {
		name = EVTOPTDtourney2 # No more small tourneys
		trigger = {
			ai = no
			FROM = { has_character_flag = tourney_local }
		}
		set_character_flag = no_small_tournies
		prestige = -10
	}
	option = {
		name = EVTOPTEtourney2 # No more regional tourneys
		trigger = {
			ai = no
			FROM = { has_character_flag = tourney_regional }
		}
		set_character_flag = no_regional_tournies
		prestige = -20
	}
	option = {
		name = EVTOPTFtourney2 # No more tourneys at all
		trigger = { ai = no }
		set_character_flag = no_tournies_ever
		prestige = -20
	}
}

# tourney.10 The lists are raised
# Thirty days on, the guests have come. A host caught in a real war must
# call the tourney off and gets most of the money back; otherwise the lists
# open (tourney.20) and every guest may meet something around the lists
# (tourney.50).
# Scope: ROOT is the host.
character_event = {
	id = tourney.10
	title = EVTNAMEtourney
	desc = EVTDESCtourney10
	picture = GFX_evt_joust

	is_triggered_only = yes # tourney.1

	trigger = { has_character_flag = tourney_running }

	option = {
		name = EVTOPTAtourney10 # Let the lists open
		trigger = {
			OR = { # Minor wars do not stop a tourney
				war = no
				AND = {
					is_primary_war_attacker = no
					is_primary_war_defender = no
				}
				AND = {
					NOT = { any_current_enemy = { demesne_size = 1 } }
					NOT = { primary_title = { is_contested = yes } }
				}
			}
		}
		hidden_effect = {
			event_target:chronicle_player = {
				if = {
					limit = { character = ROOT }
					chronicle = {
						entry = CHRONICLE_GRAND_TOURNAMENT
						picture = GFX_evt_joust
					}
				}
			}
			character_event = { id = tourney.14 days = 150 } # Safety catch
			character_event = { id = tourney.20 days = 1 }
			any_opinion_modifier_target = {
				limit = {
					OR = {
						tourney_rider_of_root_trigger = yes
						tourney_watcher_of_root_trigger = yes
					}
				}
				character_event = { id = tourney.50 days = 2 random = 20 }
			}
		}
	}
	option = {
		name = EVTOPTBtourney10 # War calls me away
		trigger = {
			NOR = {
				war = no
				AND = {
					is_primary_war_attacker = no
					is_primary_war_defender = no
				}
				AND = {
					NOT = { any_current_enemy = { demesne_size = 1 } }
					NOT = { primary_title = { is_contested = yes } }
				}
			}
		}
		if = {
			limit = { has_character_flag = tourney_grand }
			prestige = -150
			treasury = 300
		}
		else_if = {
			limit = { has_character_flag = tourney_regional }
			prestige = -60
			treasury = 120
		}
		else = {
			prestige = -30
			treasury = 60
		}
		clr_character_flag = epic_tournament
		hidden_effect = {
			any_opinion_modifier_target = {
				limit = {
					OR = {
						tourney_rider_of_root_trigger = yes
						tourney_watcher_of_root_trigger = yes
					}
				}
				character_event = { id = tourney.11 }
			}
			tourney_clear_host_effect = yes
		}
	}
}

# tourney.14 Safety catch
# Clears whatever a broken chain left behind: a host whose tourney still
# runs 150 days after the opening, or a guest whose host is dead or no
# longer holds a tourney 200 days after accepting.
# Scope: ROOT is the host or a guest.
character_event = {
	id = tourney.14
	hide_window = yes

	is_triggered_only = yes # tourney.10, tourney_join_lists_effect, tourney.3

	immediate = {
		if = {
			limit = {
				character = event_target:tourney_host
				has_character_flag = tourney_running
			}
			any_opinion_modifier_target = {
				limit = {
					OR = {
						tourney_rider_of_root_trigger = yes
						tourney_watcher_of_root_trigger = yes
					}
				}
				tourney_clear_guest_effect = yes
			}
			tourney_clear_host_effect = yes
		}
		else_if = {
			limit = {
				NOT = { character = event_target:tourney_host }
				OR = {
					has_character_flag = tourney_rider
					has_character_flag = tourney_watcher
				}
				event_target:tourney_host = {
					OR = {
						is_alive = no
						NOT = { has_character_flag = tourney_running }
					}
				}
			}
			tourney_clear_guest_effect = yes
		}
	}
}

# tourney.20 The field
# Fewer than four riders: hedge knights ride in to make up the numbers. More
# than sixteen: the heralds' qualifying runs send the weakest AI riders home.
# The player riders are told (tourney.15) and the first round is called
# (tourney.21).
# Scope: ROOT is the host.
character_event = {
	id = tourney.20
	hide_window = yes

	is_triggered_only = yes # tourney.10

	trigger = { has_character_flag = tourney_running }

	immediate = {
		set_variable = { which = tourney_waits value = 0 }
		while = {
			limit = {
				NOT = { check_variable = { which = tourney_waits value = 4 } }
				NOT = {
					any_opinion_modifier_target = {
						count = 4
						tourney_in_lists_trigger = yes
					}
				}
			}
			tourney_hedge_knight_effect = yes
			change_variable = { which = tourney_waits value = 1 }
		}
		set_variable = { which = tourney_waits value = 0 }
		while = {
			limit = {
				NOT = { check_variable = { which = tourney_waits value = 64 } }
				any_opinion_modifier_target = {
					count = 17
					tourney_in_lists_trigger = yes
				}
				any_opinion_modifier_target = {
					tourney_in_lists_trigger = yes
					ai = yes
				}
			}
			random_opinion_modifier_target = {
				limit = {
					tourney_in_lists_trigger = yes
					ai = yes
				}
				preferred_limit = { NOT = { combat_rating = 20 } }
				preferred_limit = { NOT = { combat_rating = 35 } }
				preferred_limit = { NOT = { combat_rating = 50 } }
				preferred_limit = { NOT = { combat_rating = 75 } }
				set_character_flag = tourney_out
			}
			change_variable = { which = tourney_waits value = 1 }
		}
		set_variable = { which = tourney_waits value = 0 }
		any_opinion_modifier_target = {
			limit = {
				tourney_in_lists_trigger = yes
				ai = no
			}
			character_event = { id = tourney.15 }
		}
		character_event = { id = tourney.21 days = 1 }
	}
}

# tourney.21 A new round
# One rider left: the champion (tourney.30). Nobody left: the tourney simply
# ends (tourney.34). Otherwise marks the final (two riders) or the semi-finals
# (three or four) and starts the pairing (tourney.22).
# Scope: ROOT is the host.
character_event = {
	id = tourney.21
	hide_window = yes

	is_triggered_only = yes # tourney.20, tourney.24

	trigger = { has_character_flag = tourney_running }

	immediate = {
		set_variable = { which = tourney_waits value = 0 }
		any_opinion_modifier_target = {
			limit = { tourney_in_lists_trigger = yes }
			clr_character_flag = tourney_paired
			clr_character_flag = tourney_bribed
		}
		if = {
			limit = {
				NOT = { any_opinion_modifier_target = { tourney_in_lists_trigger = yes } }
			}
			character_event = { id = tourney.34 days = 1 }
		}
		else_if = {
			limit = {
				NOT = {
					any_opinion_modifier_target = {
						count = 2
						tourney_in_lists_trigger = yes
					}
				}
			}
			character_event = { id = tourney.30 days = 1 }
		}
		else = {
			clr_character_flag = tourney_semi_round
			clr_character_flag = tourney_final_round
			if = {
				limit = {
					NOT = {
						any_opinion_modifier_target = {
							count = 3
							tourney_in_lists_trigger = yes
						}
					}
				}
				set_character_flag = tourney_final_round
			}
			else_if = {
				limit = {
					NOT = {
