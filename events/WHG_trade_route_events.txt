###############################################################################
# Trade routes: the gold mines
# Namespace: goldmine (was HFP.80000 - HFP.80020 in vanilla
#   trans_saharan_trade_events.txt)
# Faction: Generic
# Serves: any player who owns a trade post with a gold mine (the mine
#   buildings tr_gold_mine_1 - 3 can be built at Montfort, Dassel,
#   Kreutzhofen, Munzig, Ilios and Nejaz)
#
# A gold mine on a trade post can strike a rich vein, run dry and, years
# later, come back to life. Too much gold drives up prices in its province.
#
# The gold rush                        goldmine.1 - goldmine.2
# The mine runs dry                    goldmine.10 - goldmine.12
# Gold drives up prices                goldmine.20
#
# Callers: on_yearly_pulse random list in 00_on_actions.txt (1, 10, 20).
# Saved: event target gold_rush_province; province flag recent_gold_rush (10 to 50 years); province
#   modifiers depleted_gold_mine, large_gold_inflation,
#   small_gold_inflation, soldiers_fixed_gold_inflation; event target
#   gold_inflation_province.
###############################################################################

namespace = goldmine

# goldmine.1 Gold rush
# A mine not tapped lately strikes a rich vein: three years of income
# (300 to 3000 gold; vanilla gave ten years, up to 10000). Not again in that
# province for 10 to 50 years.
# Scope: ROOT is the owner.
character_event = {
	id = goldmine.1
	title = EVTTITLEgoldmine1
	desc = {
		trigger = { NOT = { trait = creature_dwarf } }
		text = EVTDESCgoldmine1
	}
	desc = {
		trigger = { trait = creature_dwarf }
		text = EVTDESCgoldmine1_dwarf
	}
	picture = GFX_evt_treasure_chamber

	is_triggered_only = yes # on_yearly_pulse

	only_playable = yes

	trigger = {
		any_demesne_province = {
			has_trade_post = yes
			has_trans_saharan_gold_mine = yes
			NOT = { has_province_flag = recent_gold_rush }
			NOT = { has_province_modifier = depleted_gold_mine }
		}
	}

	immediate = {
		random_demesne_province = {
			limit = {
				has_trade_post = yes
				has_trans_saharan_gold_mine = yes
				NOT = { has_province_flag = recent_gold_rush }
				NOT = { has_province_modifier = depleted_gold_mine }
			}
			save_event_target_as = gold_rush_province
			set_province_flag = recent_gold_rush
			province_event = { id = goldmine.2 days = 3650 random = 14600 }
		}
	}

	option = {
		name = EVTOPTAgoldmine1
		scaled_wealth = { value = 3 min = 300 max = 3000 }
	}
}

# goldmine.2 The rush is over
# Scope: the mine's province.
province_event = {
	id = goldmine.2
	hide_window = yes

	is_triggered_only = yes # goldmine.1

	immediate = {
		clr_province_flag = recent_gold_rush
	}
}

# goldmine.10 The mine runs dry
# A working gold mine is exhausted. A new vein turns up in 10 to 60 years
# (goldmine.11).
# Scope: ROOT is the owner.
character_event = {
	id = goldmine.10
	desc = EVTDESCgoldmine10
	picture = GFX_evt_bad_news

	is_triggered_only = yes # on_yearly_pulse

	only_playable = yes

	trigger = {
		any_demesne_province = {
			has_trade_post = yes
			has_trans_saharan_gold_mine = yes
			NOT = { has_province_modifier = depleted_gold_mine }
		}
	}

	immediate = {
		random_demesne_province = {
			limit = {
				has_trade_post = yes
				has_trans_saharan_gold_mine = yes
				NOT = { has_province_modifier = depleted_gold_mine }
			}
			save_event_target_as = depleted_gold_mine_target
			province_event = { id = goldmine.11 days = 3650 random = 18250 }
		}
	}

	option = {
		name = EVTOPTAgoldmine10
		event_target:depleted_gold_mine_target = {
			add_province_modifier = {
				name = depleted_gold_mine
				duration = -1
			}
		}
	}
}

# goldmine.11 A new vein is found
# Tells whoever owns the province by then. (Vanilla defined this as a
# character event but fired it as a province event, so a dry mine never
# recovered.)
# Scope: the mine's province.
province_event = {
	id = goldmine.11
	hide_window = yes

	is_triggered_only = yes # goldmine.10

	trigger = {
		has_province_modifier = depleted_gold_mine
	}

	immediate = {
		if = {
			limit = { owner = { always = yes } }
			owner = { character_event = { id = goldmine.12 } }
		}
		else = {
			remove_province_modifier = depleted_gold_mine
		}
	}
}

# goldmine.12 The mine works again
# Scope: ROOT is the owner, FROM is the mine's province.
character_event = {
	id = goldmine.12
	desc = EVTDESCgoldmine12
	picture = GFX_evt_treasure_chamber

	is_triggered_only = yes # goldmine.11

	option = {
		name = EVTOPTAgoldmine12
		FROM = { remove_province_modifier = depleted_gold_mine }
	}
}

# goldmine.20 Gold drives up prices
# The flood of gold raises prices in the mine's province. Live with it for
# ten years, pay to soften it for four, or (cruel rulers) send soldiers to
# take the gold back. (Vanilla could pick any province of the demesne.)
# Scope: ROOT is the owner.
character_event = {
	id = goldmine.20
	desc = EVTDESCgoldmine20
	picture = GFX_evt_market

	is_triggered_only = yes # on_yearly_pulse

	only_playable = yes

	trigger = {
		any_demesne_province = {
			has_trade_post = yes
			has_trans_saharan_gold_mine = yes
			NOR = {
				has_province_modifier = depleted_gold_mine
				has_province_modifier = large_gold_inflation
				has_province_modifier = small_gold_inflation
				has_province_modifier = soldiers_fixed_gold_inflation
			}
		}
	}

	immediate = {
		random_demesne_province = {
			limit = {
				has_trade_post = yes
				has_trans_saharan_gold_mine = yes
				NOR = {
					has_province_modifier = depleted_gold_mine
					has_province_modifier = large_gold_inflation
					has_province_modifier = small_gold_inflation
					has_province_modifier = soldiers_fixed_gold_inflation
				}
			}
			save_event_target_as = gold_inflation_province
		}
	}

	option = {
		name = EVTOPTAgoldmine20
		event_target:gold_inflation_province = {
			add_province_modifier = {
				name = large_gold_inflation
				duration = 3650
			}
		}
	}
	option = {
		name = EVTOPTBgoldmine20
		scaled_wealth = { value = -0.2 min = -50 max = -1000 }
		event_target:gold_inflation_province = {
			add_province_modifier = {
				name = small_gold_inflation
				duration = 1460
			}
		}
	}
	option = {
		name = EVTOPTCgoldmine20
		trigger = { trait = cruel }
		tooltip_info = cruel
		event_target:gold_inflation_province = {
			add_province_modifier = {
				name = soldiers_fixed_gold_inflation
				duration = 1825
			}
		}
	}
}
