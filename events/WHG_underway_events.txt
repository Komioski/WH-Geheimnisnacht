###############################################################################
# Underway events
# Namespace: underway (was ungdrin)
# Faction: Dwarfs
#
# The Ungdrin Ankor, the Great Underway, was dug in the Golden Age to join
# every hold of the Karaz Ankor beneath the mountains. Earthquakes shattered
# it, burying whole stretches under rubble, and the Skaven and the Night
# Goblins took much of what was left. A hold that clears and shores up its
# stretch again opens it with a wonder upgrade (upgrade_underway_<hold>):
# its counties get the ungdrin_ankor movement modifier. A Train Station
# (upgrade_trains) lays rails for mining carriages through the hold's open
# stretch (ungdrin_trains). The stretch closes when the upgrade is destroyed
# or the wonder falls to non-Dawi, and opens again when the Dawi retake it.
# Holds on an open stretch also live with what comes up the tunnels.
# Lore: WFRP 4E Dwarf Player's Guide (the Underway built p.10, Ironbreakers
# p.70, the Underway shattered and Mining Carriages p.107).
#
# Opening and closing the Underway          underway.1 - underway.6
# Life on the Underway                      underway.20 - underway.24
#
# Callers:
#   on_actions (00_on_actions.txt): underway.1 on_wonder_upgrade_finish,
#     underway.4 on_wonder_upgrade_destroyed, underway.6
#     on_wonder_owner_change.
#   on_actions (wh_on_actions.txt): the underway_tunnels bi-yearly random
#     list (underway.20 - underway.24).
# Scripted triggers: common/scripted_triggers/wh_underway_triggers.txt
# Scripted effects: common/scripted_effects/wh_underway_effects.txt
# Loc: localisation/00_WHG_underway_events.csv
###############################################################################

namespace = underway

###############################################################################
# Opening and closing the Underway
# The segment of each upgrade, and which counties are lowland, are listed in
# wh_underway_effects.txt.
###############################################################################

# underway.1 An upgrade is finished
# An Underway upgrade opens its stretch (with rails, if the wonder has a
# Train Station); a Train Station lays rails through the open stretches.
# Scope: ROOT is the owner, FROM the upgrade, FROMFROM the wonder.
character_event = {
	id = underway.1
	hide_window = yes
	is_triggered_only = yes # on_wonder_upgrade_finish

	trigger = {
		FROM = { OR = { has_wonder_upgrade = upgrade_trains underway_segment_trigger = yes } }
	}

	immediate = {
		if = {
			limit = { FROM = { underway_segment_trigger = yes } }
			FROM = { underway_open_segment_effect = yes }
			if = {
				limit = { FROMFROM = { has_wonder_upgrade = upgrade_trains } }
				FROM = { underway_carriages_on_effect = yes }
			}
			character_event = { id = underway.2 }
		}
		else = {
			FROMFROM = { underway_carriages_on_effect = yes }
			if = {
				limit = { FROMFROM = { underway_segment_trigger = yes } }
				character_event = { id = underway.3 }
			}
		}
	}
}

# underway.4 An upgrade is destroyed
# A lost Underway upgrade closes its stretch; a lost Train Station takes up
# the rails.
# Scope: ROOT is the owner, FROM the upgrade, FROMFROM the wonder.
character_event = {
	id = underway.4
	hide_window = yes
	is_triggered_only = yes # on_wonder_upgrade_destroyed

	trigger = {
		FROM = { OR = { has_wonder_upgrade = upgrade_trains underway_segment_trigger = yes } }
	}

	immediate = {
		if = {
			limit = { FROM = { underway_segment_trigger = yes } }
			FROM = { underway_close_segment_effect = yes }
			character_event = { id = underway.5 }
		}
		else = { FROMFROM = { underway_carriages_off_effect = yes } }
	}
}

# underway.6 A hold changes hands
# The Underway runs only for the Dawi: it closes when a wonder with an
# Underway upgrade falls to anyone else, and opens again for a Dawi owner.
# Scope: ROOT is the new owner, FROM the wonder.
character_event = {
	id = underway.6
	hide_window = yes
	is_triggered_only = yes # on_wonder_owner_change

	trigger = {
		FROM = { underway_segment_trigger = yes }
	}

	immediate = {
		if = {
			limit = { culture_group = dwarf_group }
			FROM = { underway_open_segment_effect = yes }
			if = {
				limit = { FROM = { has_wonder_upgrade = upgrade_trains } }
				FROM = { underway_carriages_on_effect = yes }
			}
		}
		else = { FROM = { underway_close_segment_effect = yes } }
	}
}

###############################################################################
# Life on the Underway
# Bi-yearly random list underway_tunnels, for Dawi rulers whose capital lies
# on an open stretch (underway_hold_trigger). The Ironbreakers guard the
# deep ways; a hold that cannot hold a tunnel may seal it for a few years
# (underway_sealed). An Ironbreaker's lost gromril armour can turn up once
# in a lifetime (underway_found_gromril).
###############################################################################

# underway.20 Night Goblins in the deeps
# Scope: ROOT is a Dawi ruler on the Underway.
character_event = {
	id = underway.20
	desc = EVTDESCunderway20
	picture = GFX_evt_dwarf_fighter
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # on_bi_yearly_pulse, underway_tunnels
	only_rulers = yes
	religion_group = dwarf_gods_group

	trigger = {
		underway_hold_trigger = yes
		capital_scope = { NOT = { has_province_modifier = underway_sealed } }
	}

	immediate = {
		capital_scope = { save_event_target_as = underway_hold }
	}

	option = { # Send the Ironbreakers
		name = EVTOPTAunderway20
		random_list = {
			70 = {
				modifier = { factor = 1.5 any_courtier = { trait = ironbreaker } }
				modifier = { factor = 1.3 martial = 15 }
				prestige = 100
				if = {
					limit = {
						NOT = { has_character_flag = underway_found_gromril }
						NOT = { any_artifact = { has_artifact_flag = armor } }
					}
					random = {
						chance = 10
						set_character_flag = underway_found_gromril
						add_artifact = dawi_artifact_gromril_armour
						custom_tooltip = { text = underway_found_gromril_tt }
					}
				}
			}
			30 = { prestige = 25 wealth = -50 custom_tooltip = { text = underway_heavy_losses_tt } }
		}
		ai_chance = {
			factor = 70
			modifier = { factor = 2 trait = brave }
			modifier = { factor = 0.3 trait = craven }
		}
	}
	option = { # Seal the tunnel
		name = EVTOPTBunderway20
		prestige = -25
		event_target:underway_hold = {
			add_province_modifier = { name = underway_sealed duration = 1825 }
		}
		ai_chance = {
			factor = 30
			modifier = { factor = 3 trait = craven }
			modifier = { factor = 2 war = yes }
		}
	}
}
