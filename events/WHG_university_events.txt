###############################################################################
# University events
# Namespace: university (was DR_universities in DR_universities_events.txt)
# Faction: Empire
# Serves: Empire (and the other human realms of the Old World and Araby)
#
# The universities of men: Altdorf, Nuln, Streissen, Talabheim, Salzenmund
# and Marienburg in the Empire and the Wasteland, Praag in Kislev, Remas and
# Miragliano in Tilea, Magritta in Estalia, Martek and Kufra in Araby, and
# Sudenburg in the Southlands once it is founded. A young noble may ask to study, or a
# parent may send a child (send_son_to_university). The student lives at
# the court of the university's Rector for four years, writes home halfway,
# and comes home changed: learned, drunk, worn out, married,
# sick, bookish, debauched, or, in Araby, a sorcerer of the flame.
# A university is closed while its city is held by Chaos, the dead or the
# greenskins. Bretonnia and Mousillon never send their children, and the
# Border Princes rarely do. Araby's universities take no women.
#
# Asking and sending                     university.1 - university.5
# Graduation                             university.10 - university.18
# The letter home                        university.20 - university.21
# Called home                            university.25
# The Rectors                            university.30 - university.32
#
# Callers:
#   common/on_actions/wh_on_actions.txt: university.1 on the five-year pulse
#     list university_wish.
#   common/on_actions/wh_on_actions.txt: university.30 on on_startup,
#     university.31 on on_death, university.25 on on_new_holder and
#     on_new_holder_inheritance.
#   decisions/wh_empire_decisions.txt: send_son_to_university fires
#     university.4. decisions/WHG_university_decisions.txt: endow_university.
# Scripted effects: common/scripted_effects/wh_university_effects.txt
# Scripted triggers: common/scripted_triggers/wh_university_triggers.txt
# Event modifiers: common/event_modifiers/wh_university_modifiers.txt
# Artifacts: common/artifacts/31_artifacts_universities.txt
# Loc: localisation/00_WHG_university_events.csv
###############################################################################

# university.1 A young noble asks to study
# On the five-year pulse an AI noble may ask the parent he lives with to
# send him to the nearest open university. The bright, the eager and the
# diligent ask more often; heirs and daughters less often.
# Scope: ROOT is the young noble.
character_event = {
	id = university.1
	hide_window = yes
	is_triggered_only = yes # on_five_year_pulse, university_wish

	ai = yes
	min_age = 16
	max_age = 35
	capable_only = yes
	prisoner = no

	trigger = {
		university_candidate_trigger = yes
	}

	weight_multiplier = {
		factor = 1
		modifier = { factor = 2 trait = ambitious }
		modifier = { factor = 0.5 trait = content }
		modifier = { factor = 1.3 trait = quick }
		modifier = { factor = 0.7 trait = slow }
		modifier = { factor = 2 trait = genius }
		modifier = { factor = 0.5 trait = imbecile }
		modifier = { factor = 1.3 trait = gregarious }
		modifier = { factor = 0.7 trait = shy }
		modifier = { factor = 1.3 trait = diligent }
		modifier = { factor = 0.7 trait = slothful }
		modifier = { factor = 1.3 trait = scholar }
		modifier = { factor = 0.7 is_primary_heir = yes }
		modifier = { factor = 0.7 is_female = yes }
		# Few Border Princes can spare the coin
		modifier = { factor = 0.25 location = { region = world_old_world_border_princes } }
		modifier = { factor = 0.6 NOT = { learning = 4 } }
		modifier = { factor = 0.8 NOT = { learning = 8 } }
		modifier = { factor = 1.25 learning = 12 }
		modifier = { factor = 1.25 learning = 16 }
	}

	immediate = {
		save_event_target_as = university_student
		if = {
			limit = {
				father = {
					university_parent_trigger = yes
					ROOT = { host = { character = PREVPREV } }
				}
			}
			father = { save_event_target_as = university_parent }
		}
		else = { mother = { save_event_target_as = university_parent } }
		university_pick_effect = yes
		if = {
			limit = { event_target:university = { always = yes } }
			set_character_flag = university_wants
			event_target:university_parent = { character_event = { id = university.2 } }
		}
	}
}

# Graduation
# Four years after the student left, he comes home. One outcome is rolled
# and told to the parent, or to the student's lord if the parent has died,
# or to the student himself if he has no lord.
###############################################################################

# university.10 Graduation (roll)
# Scope: ROOT is the student. event_target:university_parent is the parent.
character_event = {
	id = university.10
	hide_window = yes
	is_triggered_only = yes # university_enrol_effect

	trigger = {
		has_character_flag = university_student
	}

	immediate = {
		clr_character_flag = university_student
		remove_character_modifier = university_studying
		save_event_target_as = university_student
		if = {
			limit = {
				event_target:university_parent = { is_alive = yes NOT = { character = ROOT } }
			}
			event_target:university_parent = { save_event_target_as = university_home }
		}
		else_if = {
			limit = {
				event_target:university_home_title = {
					has_holder = yes
					holder_scope = { NOT = { character = ROOT } }
				}
			}
			event_target:university_home_title = {
				holder_scope = { save_event_target_as = university_home }
			}
		}
		else = { save_event_target_as = university_home }
		random_list = {
			10 = { # Drunkard
				modifier = { factor = 0 trait = drunkard }
				modifier = { factor = 1.5 has_character_flag = university_funded }
				modifier = { factor = 1.33 trait = gluttonous }
				modifier = { factor = 0.8 trait = temperate }
				event_target:university_home = { character_event = { id = university.11 } }
			}
			10 = { # Better educated
				modifier = { factor = 0 university_can_improve_education_trigger = no }
				modifier = { factor = 1.5 has_character_flag = university_funded }
				modifier = {
					factor = 1.25
					location = { has_province_modifier = university_endowed }
				}
				modifier = { factor = 1.33 trait = diligent }
				modifier = { factor = 0.8 trait = slothful }
				event_target:university_home = { character_event = { id = university.12 } }
			}
			10 = { # Worn out
				modifier = { factor = 0 trait = depressed }
				modifier = { factor = 1.5 has_character_flag = university_frugal }
				modifier = { factor = 1.33 trait = kind }
				modifier = { factor = 0.8 trait = envious }
				event_target:university_home = { character_event = { id = university.13 } }
			}
			10 = { # Married
				modifier = { factor = 0 is_married = yes }
				modifier = { factor = 1.33 trait = lustful }
				modifier = { factor = 0.8 trait = chaste }
				event_target:university_home = { character_event = { id = university.14 } }
			}
			10 = { # Ill
				modifier = { factor = 0 trait = ill }
				modifier = { factor = 1.33 trait = weak }
				modifier = { factor = 0.8 is_strong_trigger = yes }
				event_target:university_home = { character_event = { id = university.15 } }
			}
			10 = { # Scholar
				modifier = { factor = 0 trait = scholar }
				modifier = { factor = 1.25 has_character_flag = university_frugal }
				modifier = {
					factor = 1.25
					location = { has_province_modifier = university_endowed }
				}
				modifier = { factor = 1.33 trait = shy }
				modifier = { factor = 0.8 trait = gregarious }
				event_target:university_home = { character_event = { id = university.16 } }
			}
			10 = { # Hedonist
				modifier = { factor = 0 trait = hedonist }
				modifier = { factor = 1.5 has_character_flag = university_funded }
				modifier = { factor = 1.33 trait = lustful }
				modifier = { factor = 0.8 trait = chaste }
				event_target:university_home = { character_event = { id = university.17 } }
			}
			15 = { # Araby's sorcery
				modifier = { factor = 0 university_araby_magic_trigger = no }
				modifier = { factor = 1.33 trait = diligent }
				modifier = { factor = 1.33 trait = magic_potential_3 }
				modifier = { factor = 1.33 trait = magic_potential_4 }
				modifier = { factor = 0.8 trait = slothful }
				event_target:university_home = { character_event = { id = university.18 } }
			}
			5 = { # Nothing worse for it: plain learning
				modifier = { factor = 0 trait = academic }
				event_target:university_home = { character_event = { id = university.16 } }
			}
		}
		clr_character_flag = university_funded
		clr_character_flag = university_frugal
		clr_character_flag = university_halfway
	}
}

				limit = { has_character_flag = magic_potential_flag_4 }
				add_trait = magic_potential_4
				clr_character_flag = magic_potential_flag_4
			}

				limit = { has_character_flag = magic_potential_flag_3 }
				add_trait = magic_potential_3
				clr_character_flag = magic_potential_flag_3
			}
			else_if = {
				limit = { has_character_flag = magic_potential_flag_2 }
				add_trait = magic_potential_2
				clr_character_flag = magic_potential_flag_2
			}
			else_if = {
				limit = { has_character_flag = magic_potential_flag_1 }
				add_trait = magic_potential_1
				clr_character_flag = magic_potential_flag_1
			}

			enable_magic_xp_effect = yes
		}
	}
}

# university.31 A Rector dies

	hide_window = yes

					religion = ROOT
					culture = ROOT
					female = no
					age = 52
					attributes = { learning = 12 }
				}
			}
			20 = {
				modifier = { factor = 0 religion = ormazd }
				create_character = {
					random_traits = yes
					dynasty = RANDOM
					religion = ROOT
					culture = ROOT
					female = yes
					age = 49
					attributes = { learning = 12 }
				}
			}
			10 = {
				modifier = { factor = 0 NOT = { religion = ormazd } }
				create_character = {
					random_traits = yes
					dynasty = RANDOM
					religion = ROOT
					culture = ROOT
					female = no
					age = 55
					attributes = { learning = 12 }

			character_event = { id = university.32 }
		}
	}
}
