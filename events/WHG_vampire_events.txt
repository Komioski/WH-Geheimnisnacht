# vamp.1 A sire's death
# When a vampire ruler dies, every vampire in the realm outside the dynasty
# (the thralls they made) gets a claim on the dead ruler's primary title
# (vamp.2).
# Scope: ROOT is the dead ruler.
character_event = {
	id = vamp.1
	hide_window = yes

	min_age = 16
	capable_only = yes
	prisoner = no

	is_triggered_only = yes # on_death

	trigger = {
		is_vampire_visible_trigger = yes
		any_realm_character = { is_vampire_visible_trigger = yes NOT = { dynasty = ROOT } }
	}
	immediate = {
		any_realm_character = {
			limit = { is_vampire_visible_trigger = yes NOT = { dynasty = ROOT } }
			character_event = { id = vamp.2 }
		}
	}
}

# vamp.2 Claim on the sire's title
# Hidden. Gives the thrall a claim on FROM's primary title.
# Scope: ROOT is the thrall, FROM the dead sire.
character_event = {
	id = vamp.2
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		FROM = { primary_title = { ROOT = { add_claim = PREV } } }
	}
}

# vamp.3 A thrall asks for a duchy
# A landless or baron vampire made by a liege of their dynasty who holds two
# or more duchies asks the sire for one of them (vamp.4).
# Scope: ROOT is the thrall.
character_event = {
	id = vamp.3
	desc = EVTDESCvamp3
	picture = "GFX_evt_vampire_politics"

	min_age = 16

	is_triggered_only = yes # on_bi_yearly_pulse, random list vamp_sire_demands

	trigger = {
		OR = { has_landed_title = no lower_tier_than = COUNT }
		is_vampire_visible_trigger = yes
		any_dynasty_member = {
			ROOT = { has_opinion_modifier = { modifier = opinion_sired_me who = PREV } }
			ROOT = { liege = { character = PREVPREV } }
			demesne_size = 2
			num_of_duke_titles = 2
			higher_tier_than = DUKE
		}
		NOT = { trait = incapable }
	}

	weight_multiplier = {
		days = 1
		modifier = { factor = 3 trait = ambitious }
		modifier = { factor = 2 trait = proud }
		modifier = { factor = 1.5 trait = greedy }
		modifier = { factor = 0.5 trait = content }
		modifier = { factor = 0.5 trait = humble }
	}

	option = {
		name = EVTOPTAvamp3
		random_dynasty_member = {
			limit = {
				ROOT = { has_opinion_modifier = { modifier = opinion_sired_me who = PREV } }
				ROOT = { liege = { character = PREVPREV } }
				demesne_size = 2
				num_of_duke_titles = 2
				higher_tier_than = DUKE
			}
			character_event = { id = vamp.4 days = 1 tooltip = EVTTOOLTIP66051 }
		}
	}
	option = {
		name = EVTOPTBvamp3
		trigger = { ai = no }
	}
}

			modifier = { factor = 2 from = { martial = 14 } }
			modifier = { factor = 2 from = { intrigue = 14 } }
			modifier = { factor = 1.5 opinion = { who = from value = 25 } }
			modifier = { factor = 2 opinion = { who = from value = 50 } }
			modifier = { factor = 3 opinion = { who = from value = 75 } }

		random_demesne_title = {
			limit = { tier = DUKE NOT = { is_primary_holder_title = yes } }
			grant_title = FROM
		}
	}

	option = {
		name = EVTOPTBvamp4
		FROM = { character_event = { id = vamp.6 } }
		ai_chance = { factor = 2 modifier = { factor = 0 opinion = { who = from value = 80 } } }

		FROM = { opinion = { modifier = dynastic_refused_title who = ROOT years = 100 } }
	}

# vamp.15 Jealous lover
# A vampire whose vampire lover keeps other lovers too demands to be the only
# one (vamp.16). The kind and charitable may choose to share.
# Scope: ROOT is the jealous vampire.
character_event = {
	id = vamp.15
	desc = EVTDESCvamp15
	picture = GFX_evt_vampire_politics
	is_triggered_only = yes # on_five_year_pulse, random list vamp_jealousy

	religion_group = necromantic_group
	capable_only = yes
	prisoner = no

	trigger = {
		is_vampire_trigger = yes
		cant_love_trigger = yes
		NOT = { trait = lahmian_queen }
		any_lover = { is_vampire_trigger = yes num_of_lovers = 2 cant_love_trigger = yes }
	}

	weight_multiplier = {
		days = 1
		modifier = { factor = 1.25 trait = wroth }
		modifier = { factor = 1.7 trait = vampire_lahmian_visible }
		modifier = { factor = 1.4 trait = vampire_carstein_visible }
		modifier = { factor = 1.25 trait = vampire_jade_visible }
		modifier = { factor = 2 trait = envious }
		modifier = { factor = 1.25 trait = cruel }
		modifier = { factor = 0.8 trait = charitable }
		modifier = { factor = 0.8 trait = kind }
		modifier = { factor = 0.8 trait = patient }
	}

	option = {
		name = EVTOPTAvamp15
		random_lover = {
			limit = { is_vampire_trigger = yes num_of_lovers = 2 cant_love_trigger = yes }
			character_event = { id = vamp.16 }
		}
	}
	option = {
		name = EVTOPTBvamp15
		trigger = { OR = { trait = charitable trait = kind } }
		ai_chance = { factor = 0 }
	}
}

# vamp.21 What to do with the ring?
# The holder of a stolen Carstein Ring may keep it (vampires only; gains
# claims on the old owner's titles if necromantic) or destroy it. Players
# hear about it (vamp.22 or carstein.8).
# Scope: ROOT is the new holder, FROM the ring's old owner.
character_event = {
	id = vamp.21
	desc = EVTDESCvamp21
	picture = "GFX_evt_carstein"
	border = GFX_event_normal_frame_war

	trigger = {
		OR = { has_character_flag = prisoner_claim_ring has_plot = plot_steal_carstein_ring }
	}

	is_triggered_only = yes

	option = {
		name = EVTOPTAvamp21
		#Steal for yourself
		trigger = { is_vampire_trigger = yes }
		FROM = { destroy_artifact = carstein_ring }
		add_artifact = carstein_ring
		if = {
			limit = { religion_group = necromantic_group }
			piety = 250
			prestige = 250
			FROM = { any_demesne_title = { ROOT = { add_claim = PREV } } }
		}
		ai_chance = { factor = 100 }
		hidden_tooltip = {
			any_playable_ruler = {
				limit = {
					OR = {
						culture_group = imperial_group
						culture_group = kislevite_group
						culture_group = estalian_group
						culture_group = tilean_group
						culture_group = bretonnian_group
						culture_group = dwarf_group
					}
					ai = no
				}
				character_event = { id = vamp.22 days = 7 }
			}
			any_playable_ruler = {
				limit = {
					NOT = { culture_group = imperial_group }
					NOT = { culture_group = kislevite_group }
					NOT = { culture_group = estalian_group }
					NOT = { culture_group = tilean_group }
					NOT = { culture_group = bretonnian_group }
					NOT = { culture_group = dwarf_group }
					ai = no
				}
				character_event = { id = vamp.22 days = 30 }
			}
		}
		clr_character_flag = prisoner_claim_ring
	}
	option = {
		name = EVTOPTBvamp21
		#Destroy it
		ai_chance = { factor = 100 modifier = { factor = 0 is_vampire_trigger = yes } }
		hidden_tooltip = {
			FROM = { destroy_artifact = carstein_ring }
			if = {
				limit = { NOT = { religion_group = necromantic_group } }
				piety = 1000
				prestige = 500
			}
			any_playable_ruler = {
				limit = {
					OR = {
						culture_group = imperial_group
						culture_group = kislevite_group
						culture_group = estalian_group
						culture_group = tilean_group
						culture_group = bretonnian_group
						culture_group = dwarf_group
					}
					ai = no
				}
				character_event = { id = carstein.8 days = 7 }
			}
			any_playable_ruler = {
				limit = {
					NOT = { culture_group = imperial_group }
					NOT = { culture_group = kislevite_group }
					NOT = { culture_group = estalian_group }
					NOT = { culture_group = tilean_group }
					NOT = { culture_group = bretonnian_group }
					NOT = { culture_group = dwarf_group }
					ai = no
				}
				character_event = { id = carstein.8 days = 30 }
			}
			clr_character_flag = prisoner_claim_ring
		}
	}
}

# vamp.100 Subjects for study
# A Necrarch on a thrall raid takes a capital: kill everyone (the owner and
# their top liege are told, vamp.101 and vamp.102), take them as test
# subjects (vamp.103) or leave them.
# Scope: ROOT is the Necrarch, FROM the province.
character_event = {
	id = vamp.100
	desc = EVTDESCvamp100
	picture = GFX_evt_slaves

	is_triggered_only = yes # on_siege_over_winner

	religion = nagashi

	trigger = {
		NOT = { location = { has_province_modifier = province_slave_raid } }
		NOT = { location = { has_province_modifier = province_scourged } }
		NOT = { location = { has_province_modifier = province_slave_raid_chaos_dwarf } }
		any_war = { attacker = { character = ROOT } using_cb = necrarch_thrall_raid }
		FROM = {
			is_capital = yes
			NOT = { culture = nehekharan }
			owner = { is_undead_trigger = no }
		} #only against the living
	}

	option = { #kill them all!
		name = EVTOPTAvamp100
		if = {
			limit = { FROM = { owner = { top_liege = { NOT = { character = PREV } } } } }
			FROM = { owner = { top_liege = { character_event = { id = vamp.102 } } } }
		}
		FROM = { owner = { character_event = { id = vamp.101 } } }
		FROM = { set_province_flag = flag_province_scourged }
		if = {
			limit = {
				FROM = {
					OR = {
						any_neighbor_province = { is_tribal = yes }
						has_province_flag = flag_province_scourged
						NOT = { num_of_settlements = 2 }
					}
					NOT = { is_nomadic = yes }
				}
			}
			random = { chance = 1 FROM = { convert_to = TRIBAL } }
		}
		if = {
			limit = {
				FROM = { location = { NOT = { has_province_modifier = province_scourged } } }
			}
			FROM = {
				location = { add_province_modifier = { name = province_scourged duration = 2400 } }
			}
		}

		wealth = 125
		prestige = 50
		piety = 250
		change_variable = { which = "sample_counter" value = 1 }
		ai_chance = {
			factor = 25
			modifier = { factor = 1.75 ROOT = { OR = { trait = lunatic_3 trait = lunatic_2 } } }
			modifier = { factor = 2 trait = mystic }
		}
	}
	option = { #take them all!
		name = EVTOPTBvamp100
		if = {
			limit = { FROM = { owner = { top_liege = { NOT = { character = PREV } } } } }
			FROM = { owner = { top_liege = { character_event = { id = vamp.103 } } } }
		}
		FROM = { owner = { character_event = { id = vamp.103 } } }
		wealth = 25
		prestige = 10
		piety = 10
		if = {
			limit = {
				FROM = { location = { NOT = { has_province_modifier = province_slave_raid } } }
			}
			FROM = {
				location = {
					add_province_modifier = { name = province_slave_raid duration = 2400 }
				}
			}
		}
		change_variable = { which = "sample_counter" value = 3 }
		ai_chance = {
			factor = 75
			modifier = {
				factor = 1.5
				ROOT = { NOT = { trait = possessed trait = lunatic_3 trait = lunatic_2 } }
			}
			modifier = { factor = 1.2 trait = kind }
		}
	}
	option = { #I don't care about this province
		name = EVTOPTCvamp100
		piety = -25
		prestige = -25
		if = { limit = { trait = cruel } random = { chance = 1 remove_trait = cruel } }
		ai_chance = {
			factor = 1
			modifier = {
				factor = 1.2
				ROOT = { NOR = { trait = possessed trait = lunatic_3 trait = lunatic_2 } }
			}
			modifier = { factor = 1.2 trait = kind }
		}
	}
}

	desc = EVTDESCvamp120
	picture = GFX_evt_vampire_jades

	is_triggered_only = yes # on_five_year_pulse

	min_age = 40

	trigger = {
		trait = vampire_jade_visible
	}

	option = { #Use the crystal
		name = EVTOPTAvamp120
		trigger = {
			is_incapacitated_trigger = no
			OR = {
				has_artifact = dark_crystal
				top_liege = {
					has_artifact = dark_crystal
					NOT = { has_character_flag = forbidden_jade_crystal }
				}
			}
		}
		hidden_tooltip = { character_event = { id = vamp.121 } }
		ai_chance = {
			factor = 90
			modifier = { factor = 0.5 trait = paranoid }
			modifier = { factor = 0.85 trait = proud }
		}
	}

	option = {
		name = EVTOPTBvamp120
		if = {
			limit = {
				top_liege = {
					NOT = { character = ROOT }
					has_character_flag = forbidden_jade_crystal
				}
			}
			top_liege = { ROOT = { opinion = { who = PREV modifier = opinion_forced_abstinence } } }
		}
		if = {
			limit = { trait = lore_heavens }
			remove_trait = lore_heavens
			remove_character_modifier = magic_power_1
			remove_character_modifier = magic_power_2
			remove_character_modifier = magic_power_3
			remove_character_modifier = magic_power_4
		}

		if = {
			limit = { has_character_modifier = jade_degradation_4 }
			change_variable = { which = "lunatic_counter" value = 3 }
			random = { chance = 20 change_diplomacy = -1 change_learning = -1 }

			limit = { has_character_modifier = jade_degradation_3 }
			change_variable = { which = "lunatic_counter" value = 1 }
			remove_character_modifier = jade_degradation_3
			add_character_modifier = { name = jade_degradation_4 duration = -1 }

			limit = { has_character_modifier = jade_degradation_2 }

			remove_character_modifier = jade_degradation_2
			add_character_modifier = { name = jade_degradation_3 duration = -1 }
		}

		if = {
			limit = { has_character_modifier = jade_degradation_1 }
			remove_character_modifier = jade_degradation_1
			add_character_modifier = { name = jade_degradation_2 duration = -1 }
		}

		if = {
			limit = {
				NOR = {
					has_character_modifier = jade_degradation_1
					has_character_modifier = jade_degradation_2
					has_character_modifier = jade_degradation_3
					has_character_modifier = jade_degradation_4
				}
			}
			add_character_modifier = { name = jade_degradation_1 duration = -1 }
		}

	}

character_event = {
	id = vamp.260
	desc = EVTDESCvamp260
	picture = GFX_evt_bad_news
	border = GFX_event_normal_frame_intrigue
	hide_from = yes

	is_triggered_only = yes

	option = {
		name = EVTOPTAvamp260
		if = { limit = { trait = incapable } vamp_curse_kill_effect = yes }
		else_if = { limit = { trait = maimed } add_trait = incapable }
		else_if = { limit = { trait = wounded } add_trait = maimed }
		else = { add_trait = wounded }
		hidden_tooltip = { FROM = { character_event = { id = vamp.261 days = 5 random = 14 } } }
	}
}

# vamp.270 Heart bound
# A lover, friend or spouse turns against the target.
# Scope: ROOT is the target, FROM the plotter.
character_event = {
	id = vamp.270
	desc = EVTDESCvamp270
	picture = GFX_evt_lunatic
	border = GFX_event_normal_frame_intrigue
	hide_from = yes

	is_triggered_only = yes

	option = {
		name = EVTOPTAvamp270
		hidden_tooltip = { FROM = { character_event = { id = vamp.271 days = 5 random = 14 } } }
		if = { limit = { has_lover = yes } random_lover = { remove_lover = ROOT add_rival = ROOT } }
		else_if = {
			limit = { any_friend = { is_alive = yes } }
			random_friend = { remove_friend = ROOT add_rival = ROOT }
		}
		else_if = { limit = { is_married = yes } random_spouse = { add_rival = ROOT } }
	}
}
