###############################################################################
# Vampire bloodline society events
# Namespace: vampsoc
# Faction: the seven vampire bloodlines
#
# One society per bloodline (common/societies/wh_vampire_bloodlines.txt).
# Every open vampire is locked into the society of their line; hidden
# Lahmians and von Carsteins are secret members. Living thralls of a
# necromantic faith serve at ranks 1-2. Each line has 4 ranks, a deed and a
# trial to climb, three rank powers, a head with an ambition, a mechanic of
# its own, flavour events, a rare artifact and a retinue.
#
# Startup sorting                           vampsoc.1
# Five-year sweep and flavour driver        vampsoc.2
# Rank-up notice                            vampsoc.5
# Feeding                                   vampsoc.10
# The head's aid                            vampsoc.11 - vampsoc.15
# Thrall service                            vampsoc.13 - vampsoc.14
#
# Fired from common/on_actions/wh_on_actions.txt: vampsoc.1 on_startup,
# vampsoc.2 on_five_year_pulse. The decisions in
# decisions/wh_vampire_society_decisions.txt fire the rest.
# Scripted triggers: common/scripted_triggers/wh_vamp_triggers.txt
# Scripted effects: common/scripted_effects/wh_vamp_effects.txt
# Event modifiers: common/event_modifiers/wh_vampire_modifiers.txt
# Loc: localisation/00_WHG_vampire_society_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = vampsoc

# vampsoc.1 The bloodlines gather
# Once per game the multiplayer host puts every vampire into the society of
# their bloodline, ranks them by standing (kings and dukes rank 3, other
# rulers rank 2) and makes the founders heads: Vlad von Carstein, Walach
# Harkon, Neferata, W'soran, Shdroi So-sou and Salimata. The Strigoi start
# without a head; reforming Strygos makes one.
# Scope: ROOT is the host player.
character_event = {
	id = vampsoc.1
	hide_window = yes

	ai = no
	only_playable = yes

	is_triggered_only = yes # on_startup

	trigger = {
		NOT = { has_global_flag = vampsoc_sorted }
		is_multiplayer_host_character = yes
	}

	immediate = {
		set_global_flag = vampsoc_sorted
		any_character = {
			limit = { vamp_needs_society_trigger = yes }
			vamp_join_bloodline_effect = yes
		}
		any_character = {
			limit = {
				vamp_society_member_trigger = yes
				is_vampire_trigger = yes
				is_ruler = yes
			}
			if = {
				limit = { society_rank == 1 }
				society_rank_up = 1
			}
			if = {
				limit = {
					higher_tier_than = COUNT
					society_rank == 2
					NOT = { trait = vampire_carstein_hidden }
				}
				society_rank_up = 1
			}
		}
		vampsoc_set_founder_heads_effect = yes
	}
}

# vampsoc.2 The bloodlines stir

# vampire by some path that missed the join) is put back, and an open
# vampire member may get one of their bloodline's flavour events.

character_event = {
	id = vampsoc.2
	hide_window = yes

	is_triggered_only = yes # on_five_year_pulse

	trigger = {
		is_vampire_trigger = yes
	}

	immediate = {
		if = {
			limit = { vamp_needs_society_trigger = yes }

			limit = {
				vamp_society_member_trigger = yes
				prisoner = no
				is_incapacitated_trigger = no
			}
			vampsoc_flavour_effect = yes
		}
	}
}

# vampsoc.10 The hunger
# The vampire feeds. A servant is safe. Worthy prey (a warrior of the realm)
# feeds deeper and counts as the Strigoi's deed, but may fight back. A
# Mahtmasi may instead answer a mortal who came asking, which is the
# Mahtmasi deed. The Blood Dragons lose Honour for feeding on the helpless.
# Scope: ROOT is a vampire member.
character_event = {
	id = vampsoc.10
	desc = {
		trigger = { society_member_of = ordo_draconis }
		text = EVTDESCvampsoc10_blood
	}
	desc = {
		trigger = { society_member_of = carrion_children }
		text = EVTDESCvampsoc10_mahtmasi
	}
	desc = {
		trigger = {
			NOT = { society_member_of = ordo_draconis }
			NOT = { society_member_of = carrion_children }
		}
		text = EVTDESCvampsoc10
	}
	picture = GFX_evt_vampire_kiss
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes # vampsoc_feed

	# A servant
	option = {
		name = EVTOPTAvampsoc10
		trigger = { NOT = { society_member_of = carrion_children } }
		add_character_modifier = { name = vampsoc_well_fed years = 2 }
		if = {
			limit = { society_member_of = ordo_draconis }
			custom_tooltip = { text = vampsoc_feed_dishonour_tt }
			vampsoc_honour_loss_small_effect = yes
		}
		else = {
			vamp_currency_small_effect = yes
		}
		ai_chance = {
			factor = 50
			modifier = { factor = 0.1 society_member_of = ordo_draconis }
		}
	}

	# Worthy prey
	option = {
		name = EVTOPTBvampsoc10
		trigger = { NOT = { society_member_of = carrion_children } }
		random_list = {
			75 = {
				add_character_modifier = { name = vampsoc_well_fed years = 4 }
				vamp_currency_medium_effect = yes
				prestige = 25
				if = {
					limit = { society_member_of = strigoi_brood }
					set_character_flag = vamp_deed_done
				}
			}
			25 = {
				modifier = { factor = 0.5 martial = 15 }
				modifier = { factor = 0.5 combat_rating = 40 }
				add_character_modifier = { name = vampsoc_well_fed years = 2 }
				vamp_currency_small_effect = yes
				custom_tooltip = { text = vampsoc_feed_fought_back_tt }
				random_list = {
					60 = { add_trait_wounded_effect = yes }
					40 = { prestige = -25 }
				}
			}
		}
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = brave }
			modifier = { factor = 3 society_member_of = strigoi_brood }
			modifier = { factor = 3 society_member_of = ordo_draconis }
			modifier = { factor = 0.3 trait = craven }
		}
	}

	# Mahtmasi: the one who asked
	option = {
		name = EVTOPTCvampsoc10
		trigger = { society_member_of = carrion_children }
		add_character_modifier = { name = vampsoc_well_fed years = 4 }
		vamp_currency_medium_effect = yes
		set_character_flag = vamp_deed_done
		piety = 10
	}

	# Mahtmasi: a stranger, unasked
	option = {
		name = EVTOPTDvampsoc10
		trigger = { society_member_of = carrion_children }
		add_character_modifier = { name = vampsoc_well_fed years = 2 }
		custom_tooltip = { text = vampsoc_feed_unasked_tt }
		vamp_currency_loss_small_effect = yes
		ai_chance = { factor = 10 }
	}
}

# vampsoc.11 A plea from the blood

	id = vampsoc.11

	# Gold
	option = {
		name = EVTOPTAvampsoc11

		wealth = -100
		FROM = { wealth = 100 }
		vamp_currency_small_effect = yes
		FROM = { character_event = { id = vampsoc.12 } }

			factor = 40
			modifier = { factor = 2 opinion = { who = FROM value = 25 } }
			modifier = { factor = 2 trait = charitable }
			modifier = { factor = 0.2 trait = greedy }
		}
	}

	# Lend the dead
	option = {
		name = EVTOPTBvampsoc11
		FROM = {
			add_character_modifier = { name = vampsoc_head_aid years = 3 }
			character_event = { id = vampsoc.12 }
		}
		vamp_currency_small_effect = yes
		ai_chance = {
			factor = 40
			modifier = { factor = 2 opinion = { who = FROM value = 25 } }
			modifier = { factor = 2 war = no }
		}
	}

	# Refuse
	option = {
		name = EVTOPTCvampsoc11
		FROM = { letter_event = { id = vampsoc.15 } }
		ai_chance = {
			factor = 20
			modifier = { factor = 0.2 opinion = { who = FROM value = 25 } }
			modifier = { factor = 3 NOT = { opinion = { who = FROM value = 0 } } }
			modifier = { factor = 2 trait = paranoid }
		}

# The Blood Dragon rides out to challenge a famed mortal champion. A win
# earns Honour (and for a squire, counts as the deed for rank 2); a loss

# Scope: ROOT is a member of the Ordo Draconis.

	id = vampsoc.150
	desc = EVTDESCvampsoc150
	picture = GFX_evt_joust
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # vampsoc_challenge_champion

	# Fight with honour
	option = {
		name = EVTOPTAvampsoc150
		vampsoc_fight_effect = yes

			limit = { has_character_flag = vampsoc_won }

				limit = { society_rank == 1 }
				set_character_flag = vamp_deed_done
			}

		}
		else = {
			custom_tooltip = { text = vampsoc_duel_lost_tt }
			add_trait_wounded_effect = yes
			vampsoc_honour_loss_small_effect = yes
		}

	}

	# Let the thirst fight for you
	option = {
		name = EVTOPTBvampsoc150
		add_character_modifier = { name = vampsoc_well_fed years = 2 }
		vampsoc_honour_loss_small_effect = yes
		prestige = 10
		ai_chance = {
			factor = 20
			modifier = { factor = 3 trait = wroth }
			modifier = { factor = 0.2 trait = honest }
		}
	}

# vampsoc.155 Challenge to the Grand Master
# A Blood Dragon Lord challenges the Grand Master before the knights of the
# Order. The victor leads the Ordo Draconis.
# Scope: ROOT is the challenger (rank 3); event_target:vampsoc_foe is the
# Grand Master, saved by vampsoc_challenge_grand_master.

	id = vampsoc.155
	desc = EVTDESCvampsoc155
	picture = GFX_evt_joust
	border = GFX_event_normal_frame_war

	is_triggered_only = yes # vampsoc_challenge_grand_master

	option = {
		name = EVTOPTAvampsoc155
		vampsoc_fight_effect = yes

			limit = { has_character_flag = vampsoc_won }

		}
	}
}
