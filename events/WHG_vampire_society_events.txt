###############################################################################
# Vampire bloodline society events
# Namespace: vampsoc
# Faction: the seven vampire bloodlines
#
# One society per bloodline (common/societies/wh_vampire_bloodlines.txt).
# Every open vampire is locked into the society of their line; hidden
# Lahmians and von Carsteins are secret members. Living thralls of a
# necromantic faith serve at ranks 1-2. Each line has 4 ranks, a deed and a
# trial to climb, three rank powers, a head with an ambition, a mechanic of
# its own, flavour events, a rare artifact and a retinue.
#
# Startup sorting                           vampsoc.1
# Five-year sweep and flavour driver        vampsoc.2
# Rank-up notice                            vampsoc.5
# Feeding                                   vampsoc.10
# The head's aid                            vampsoc.11 - vampsoc.15
# Thrall service                            vampsoc.13 - vampsoc.14
#
# Fired from common/on_actions/wh_on_actions.txt: vampsoc.1 on_startup,
# vampsoc.2 on_five_year_pulse. The decisions in
# decisions/wh_vampire_society_decisions.txt fire the rest.
# Scripted triggers: common/scripted_triggers/wh_vamp_triggers.txt
# Scripted effects: common/scripted_effects/wh_vamp_effects.txt
# Event modifiers: common/event_modifiers/wh_vampire_modifiers.txt
# Loc: localisation/00_WHG_vampire_society_events.csv
# Full list with callers: docs/events/event_database.csv
###############################################################################

namespace = vampsoc

# vampsoc.1 The bloodlines gather
# Once per game the multiplayer host puts every vampire into the society of
# their bloodline, ranks them by standing (kings and dukes rank 3, other
# rulers rank 2) and makes the founders heads: Vlad von Carstein, Walach
# Harkon, Neferata, W'soran, Shdroi So-sou and Salimata. The Strigoi start
# without a head; reforming Strygos makes one.
# Scope: ROOT is the host player.
character_event = {
	id = vampsoc.1
	hide_window = yes

	ai = no
	only_playable = yes

	is_triggered_only = yes # on_startup

	trigger = {
		NOT = { has_global_flag = vampsoc_sorted }
		is_multiplayer_host_character = yes
	}

	immediate = {
		set_global_flag = vampsoc_sorted
		any_character = {
			limit = { vamp_needs_society_trigger = yes }
			vamp_join_bloodline_effect = yes
		}
		any_character = {
			limit = {
				vamp_society_member_trigger = yes
				is_vampire_trigger = yes
				is_ruler = yes
			}
			if = {
				limit = { society_rank == 1 }
				society_rank_up = 1
			}
			if = {
				limit = {
					higher_tier_than = COUNT
					society_rank == 2
					NOT = { trait = vampire_carstein_hidden }
				}
				society_rank_up = 1
			}
		}
		vampsoc_set_founder_heads_effect = yes
	}
}

# vampsoc.2 The bloodlines stir
# Every five years: a vampire who has left their society (or was made a
# vampire by some path that missed the join) is put back, a member may get
# one of their bloodline's flavour events, and the head of the Court of
# Sylvania ends the Feud of the Counts once it is twenty years old.
# Scope: ROOT is any character on the five-year pulse.
character_event = {
	id = vampsoc.2
	hide_window = yes

	is_triggered_only = yes # on_five_year_pulse

	trigger = {
		is_vampire_trigger = yes
	}

	immediate = {
		if = {
			limit = { vamp_needs_society_trigger = yes }
			vamp_join_bloodline_effect = yes
		}
		if = {
			limit = {
				vamp_society_member_trigger = yes
				prisoner = no
				is_incapacitated_trigger = no
			}
			vampsoc_flavour_effect = yes
		}
		# The Feud of the Counts ends after twenty years
		if = {
			limit = {
				had_global_flag = { flag = vampsoc_feud_active days = 7300 }
				society_member_of = carstein_court
				society_rank == 4
			}
			clr_global_flag = vampsoc_feud_active
			character_event = { id = vampsoc.112 }
		}
	}
}

# The vampire feeds. A servant is safe. Worthy prey (a warrior of the realm)
# feeds deeper and counts as the Strigoi's deed, but may fight back. A
# Mahtmasi may instead answer a mortal who came asking, which is the
# Mahtmasi deed. The Blood Dragons lose Honour for feeding on the helpless.
# Scope: ROOT is a vampire member.
character_event = {
	id = vampsoc.10
	desc = {
		trigger = { society_member_of = ordo_draconis }
		text = EVTDESCvampsoc10_blood
	}
	desc = {
		trigger = { society_member_of = carrion_children }
		text = EVTDESCvampsoc10_mahtmasi
	}
	desc = {
		trigger = {
			NOT = { society_member_of = ordo_draconis }
			NOT = { society_member_of = carrion_children }
		}
		text = EVTDESCvampsoc10
	}
	picture = GFX_evt_vampire_kiss
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes # vampsoc_feed

	# A servant
	option = {
		name = EVTOPTAvampsoc10
		trigger = { NOT = { society_member_of = carrion_children } }
		add_character_modifier = { name = vampsoc_well_fed years = 2 }
		if = {
			limit = { society_member_of = ordo_draconis }
			custom_tooltip = { text = vampsoc_feed_dishonour_tt }
			vampsoc_honour_loss_small_effect = yes
		}
		else = {
			vamp_currency_small_effect = yes
		}
		ai_chance = {
			factor = 50
			modifier = { factor = 0.1 society_member_of = ordo_draconis }
		}
	}

	# Worthy prey
	option = {
		name = EVTOPTBvampsoc10
		trigger = { NOT = { society_member_of = carrion_children } }
		random_list = {
			75 = {
				add_character_modifier = { name = vampsoc_well_fed years = 4 }
				vamp_currency_medium_effect = yes
				prestige = 25
				if = {
					limit = { society_member_of = strigoi_brood }
					set_character_flag = vamp_deed_done
				}
			}
			25 = {
				modifier = { factor = 0.5 martial = 15 }
				modifier = { factor = 0.5 combat_rating = 40 }

					60 = { add_trait_wounded_effect = yes }
					40 = { prestige = -25 }
				}
			}
		}
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = brave }
			modifier = { factor = 3 society_member_of = strigoi_brood }
			modifier = { factor = 3 society_member_of = ordo_draconis }
			modifier = { factor = 0.3 trait = craven }
		}
	}

	# Mahtmasi: the one who asked
	option = {
		name = EVTOPTCvampsoc10
		trigger = { society_member_of = carrion_children }
		add_character_modifier = { name = vampsoc_well_fed years = 4 }
		vamp_currency_medium_effect = yes
		set_character_flag = vamp_deed_done

	}

		name = EVTOPTDvampsoc10

		custom_tooltip = { text = vampsoc_feed_unasked_tt }

	id = vampsoc.101
	desc = {
		trigger = { has_character_flag = vampsoc_trial_passed }

	}

		trigger = { NOT = { has_character_flag = vampsoc_trial_passed } }

	}

	border = GFX_event_normal_frame_religion

			60 = {
				modifier = { factor = 0.6 has_character_flag = vampsoc_night_bold }
				modifier = { factor = 1.3 martial = 12 }
				modifier = { factor = 1.3 intrigue = 12 }
				modifier = { factor = 1.3 trait = lore_necromancy }
				set_character_flag = vampsoc_trial_passed
			}
			40 = { }
		}
	}

	option = {
		name = {
			trigger = { has_character_flag = vampsoc_trial_passed }
			text = EVTOPTAvampsoc101
		}
		name = {
			trigger = { NOT = { has_character_flag = vampsoc_trial_passed } }
			text = EVTOPTAvampsoc101b
		}

		if = {
			limit = { has_character_flag = vampsoc_trial_passed }
			set_character_flag = vamp_trial_done
			custom_tooltip = { text = vampsoc_trial_passed_tt }

				limit = { has_character_flag = vampsoc_night_bold }

			}
			else = {

			}
		}

			add_character_modifier = { name = vampsoc_trial_cooldown years = 5 hidden = yes }

	}

# under her own banner. The mortal is caught in the web of the Sisterhood
# (the Tithe of the Houses counts them).
# Scope: ROOT is the Lahmian; event_target:vampsoc_mortal is the ruler.
character_event = {
	id = vampsoc.200
	desc = EVTDESCvampsoc200
	picture = GFX_evt_vampiress1
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes # vampsoc_assist_mortals

	# Gold
	option = {
		name = EVTOPTAvampsoc200
		trigger = { wealth = 100 }
		wealth = -100
		event_target:vampsoc_mortal = {
			wealth = 100
			opinion = { who = ROOT modifier = opinion_vampsoc_lahmian_aid years = 20 }
			add_character_modifier = { name = vampsoc_lahmian_web years = 20 hidden = yes }
		}
		vamp_currency_medium_effect = yes
		ai_chance = { factor = 50 }
	}

	# Sellswords
	option = {
		name = EVTOPTBvampsoc200
		trigger = { wealth = 150 }
		wealth = -150
		event_target:vampsoc_mortal = {
			opinion = { who = ROOT modifier = opinion_vampsoc_lahmian_aid years = 20 }
			add_character_modifier = { name = vampsoc_lahmian_web years = 20 hidden = yes }
			character_event = { id = vampsoc.201 }
		}
		vamp_currency_medium_effect = yes
		ai_chance = { factor = 50 }
	}

	# Think better of it
	option = {
		name = EVTOPTCvampsoc200
		ai_chance = { factor = 0 }
	}
}

# vampsoc.251 The work is done
# Learning decides the result. Success passes the trial with a reward by
# field; failure costs five years.
# Scope: ROOT is a Necrarch on trial.
character_event = {
	id = vampsoc.251
	desc = {
		trigger = { has_character_flag = vampsoc_trial_passed }
		text = EVTDESCvampsoc251
	}
	desc = {
		trigger = { NOT = { has_character_flag = vampsoc_trial_passed } }
		text = EVTDESCvampsoc251b
	}
	picture = GFX_evt_lab
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	trigger = { has_character_flag = vamp_on_trial }

	immediate = {
		random_list = {
			50 = {
				modifier = { factor = 1.5 learning = 15 }
				modifier = { factor = 1.5 learning = 20 }
				modifier = { factor = 1.5 trait = lore_necromancy }
				modifier = { factor = 1.3 trait = scholar }
				set_character_flag = vampsoc_trial_passed
			}
			50 = {
				modifier = { factor = 0.7 trait = patient }
			}
		}
	}

	option = {
		name = {
			trigger = { has_character_flag = vampsoc_trial_passed }
			text = EVTOPTAvampsoc251
		}
		name = {
			trigger = { NOT = { has_character_flag = vampsoc_trial_passed } }
			text = EVTOPTAvampsoc251b
		}
		clr_character_flag = vamp_on_trial
		if = {
			limit = { has_character_flag = vampsoc_trial_passed }
			set_character_flag = vamp_trial_done
			set_character_flag = vamp_magnum_opus_done
			custom_tooltip = { text = vampsoc_trial_passed_tt }
			vamp_currency_medium_effect = yes
			if = {
				limit = { has_character_flag = vampsoc_opus_anatomy }
				add_character_modifier = { name = vampsoc_opus_anatomy duration = -1 }
			}
			else_if = {
				limit = { has_character_flag = vampsoc_opus_spirit }
				add_character_modifier = { name = vampsoc_opus_spirit duration = -1 }
			}
			else = {
				add_character_modifier = { name = vampsoc_opus_stars duration = -1 }
			}
		}
		else = {
			custom_tooltip = { text = vampsoc_trial_failed_tt }
			add_character_modifier = { name = vampsoc_trial_cooldown years = 5 hidden = yes }
		}
		clr_character_flag = vampsoc_trial_passed
		clr_character_flag = vampsoc_opus_anatomy
		clr_character_flag = vampsoc_opus_spirit
		clr_character_flag = vampsoc_opus_stars
	}
}

# vampsoc.281 The recipe is complete
# The Lord has reconstructed the Recipe for Nagash. A permanent mastery of
# death, and every Necrarch gains Lore.
# Scope: ROOT is a Lord of the Dark Tower.
character_event = {
	id = vampsoc.281
	desc = EVTDESCvampsoc281
	picture = GFX_evt_lab
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	trigger = { has_character_flag = vampsoc_writing_recipe }

	option = {
		name = EVTOPTAvampsoc281
		clr_character_flag = vampsoc_writing_recipe
		set_character_flag = vampsoc_recipe_done
		add_character_modifier = { name = vampsoc_recipe_for_nagash duration = -1 }
		prestige = 500
		custom_tooltip = { text = vampsoc_recipe_order_tt }
		hidden_effect = {
			any_society_member = {
				limit = {
					society_member_of = necrarch_brotherhood
					NOT = { character = ROOT }
				}
				vamp_currency_medium_effect = yes
			}
		}
	}
}
