###############################################################################
# The Venturers: expeditions, charts and colonies
# Namespace: venture
# Faction: Generic
#
# The colonial half of the naval game (the other half is the pirate and
# raider societies of events/WHG_sea_*). The Venturers (adventurers_the_
# league) are the merchant adventurers, explorers and colonists of every
# people that sends ships past the edge of the charts: Marienburg's houses,
# Nordland's Second Fleet, the merchant princes of Tilea, the conquistadores
# of Magritta, Brionne and L'Anguille, Araby, the Sea Wardens of Cothique,
# the Dawi of Barak Varr, the East and the Norse who follow Losteriksson.
#
# An expedition is a risk and meant to be one. The Lustrian chapter of the
# Old World's own records puts the odds plainly: of every hundred ventures,
# more than half are never heard from again. Gold, guides, charts and a
# good captain shift the odds; the choices on the way shift them again.
#
# The expedition is planned                 venture.1 - 3
# The reckoning (hidden)                    venture.4
# On the way: what the jungle sends         venture.10 - 19
# Outcomes                                  venture.20 - 25
# Charting a coast                          venture.30 - 32
# Colonies                                  venture.40 - 41
# The colony's year                         venture.5, venture.50 - 65
# The Slann's answer                        venture.70 - 76
# The Great Voyage                          venture.80 - 89
# Reputations                               venture.90
# Venturer missions                         venture.100 - 106
# Tales: Swamp Town, the society's tales    venture.120 - 125
#
# Fired from: venture.1 from the decisions venture_sponsor_expedition and
# venture_lead_expedition (decisions/WHG_venture_decisions.txt); venture.30
# from venture_chart_coast; venture.40 from the Colonial Charter war
# (coastal_colonization, common/cb_types/wh_cb_types.txt); venture.5 from
# on_startup; venture.50 from on_yearly_pulse; venture.70 from
# on_five_year_pulse; venture.80 from venture_great_voyage and venture.90
# from venture_choose_reputation; venture.100 from seaquest.2
# (events/WHG_sea_society_events.txt); venture.120 from on_five_year_pulse,
# venture.125 from on_character_society_rank_up. The game rule
# warhammer_venture_lead decides whether players may lead in person (on
# by default).
# Scripted triggers: common/scripted_triggers/WHG_venture_triggers.txt
# Scripted effects: common/scripted_effects/WHG_venture_effects.txt
# Voices: localisation/customizable_localisation/WHG_venture_voices.txt
# Modifiers: common/event_modifiers/WHG_venture_modifiers.txt
# Colony buildings: common/buildings/WHG_venture_colony_buildings.txt
# Artifacts: common/artifacts/WHG_venture_artifacts.txt
# Bloodlines: common/bloodlines/WHG_venture_bloodlines.txt
# Saved: the expedition's flags and the variable venture_odds (see
#   WHG_venture_effects.txt), cleared by venture_clear_effect; the
#   variables venture_deeds (for the Great Voyage) and venture_slann_wrath
#   (temple-cities plundered); the province flag venture_surveyed and the
#   character flag venture_has_survey (a colony site for a charter).
# Loc: localisation/00_WHG_venture_events.csv
# Full list with callers: docs/events/event_database.csv

# captain, or two trials for a Venturer who leads in person. The reckoning
# (venture.4) comes some eight to ten months out.
# Scope: ROOT is the Venturer.

	id = venture.3
	hide_window = yes

	is_triggered_only = yes # venture.2

		set_character_flag = venture_expedition_out
		add_character_modifier = { name = venture_expedition_cd years = 5 }
		if = {
			limit = { venture_leading_trigger = no }
			venture_pick_captain_effect = yes
		}
		else = {
			add_character_modifier = { name = venture_on_expedition days = 330 }
		}
		venture_set_odds_effect = yes
		character_event = { id = venture.10 days = 45 random = 45 }
		if = {
			limit = { venture_leading_trigger = yes }
			character_event = { id = venture.10 days = 140 random = 40 }
		}
		character_event = { id = venture.4 days = 240 random = 60 }
	}
}

# venture.4 The reckoning (hidden)
# Rolls the outcome. Out of a hundred ventures with nothing to help them,
# about 55 are lost, 25 come home with a few survivors, 15 succeed and 5
# make their backers rich. Every point of venture_odds (stake, guides,
# charts, a seasoned leader, the choices on the way) shifts the roll; a
# temple-city is deadlier and richer, Naggaroth only deadlier. A sponsored
# venture whose captain is dead is lost.
# Scope: ROOT is the Venturer.
character_event = {
	id = venture.4
	hide_window = yes

	is_triggered_only = yes # venture.3

	trigger = {
		has_character_flag = venture_expedition_out
	}

	immediate = {
		if = {
			limit = {
				venture_leading_trigger = no
				event_target:venture_captain = { is_alive = no }
			}
			character_event = { id = venture.20 }
		}
		else = {
			random_list = {
				55 = { # Lost
					modifier = { factor = 0.8 check_variable = { which = venture_odds value = 2 } }
					modifier = { factor = 0.8 check_variable = { which = venture_odds value = 4 } }
					modifier = { factor = 0.8 check_variable = { which = venture_odds value = 6 } }
					modifier = { factor = 0.8 check_variable = { which = venture_odds value = 8 } }
					modifier = { factor = 1.3 NOT = { check_variable = { which = venture_odds value = 0 } } }
					modifier = { factor = 1.3 NOT = { check_variable = { which = venture_odds value = -2 } } }
					modifier = { factor = 1.3 has_character_flag = venture_dest_temple }
					modifier = { factor = 1.15 has_character_flag = venture_dest_naggaroth }
					if = {
						limit = { venture_leading_trigger = yes }
						random_list = {
							60 = { character_event = { id = venture.20 } }
							40 = { character_event = { id = venture.21 } }
						}
					}
					else = {
						character_event = { id = venture.20 }
					}
				}
				25 = { # A few survivors
					character_event = { id = venture.22 }
				}
				15 = { # Success
					modifier = { factor = 1.25 check_variable = { which = venture_odds value = 2 } }
					modifier = { factor = 1.25 check_variable = { which = venture_odds value = 4 } }
					modifier = { factor = 1.25 check_variable = { which = venture_odds value = 6 } }
					modifier = { factor = 1.25 check_variable = { which = venture_odds value = 8 } }
					modifier = { factor = 0.7 NOT = { check_variable = { which = venture_odds value = 0 } } }
					modifier = { factor = 1.2 venture_leading_trigger = yes }
					if = {
						limit = { has_character_flag = venture_dest_temple }
						character_event = { id = venture.25 }
					}
					else = {
						character_event = { id = venture.23 }
					}
				}
				5 = { # A fortune
					modifier = { factor = 1.4 check_variable = { which = venture_odds value = 2 } }
					modifier = { factor = 1.4 check_variable = { which = venture_odds value = 4 } }
					modifier = { factor = 1.4 check_variable = { which = venture_odds value = 6 } }
					modifier = { factor = 1.4 check_variable = { which = venture_odds value = 8 } }
					modifier = { factor = 0.5 NOT = { check_variable = { which = venture_odds value = 0 } } }
					modifier = { factor = 2 has_character_flag = venture_dest_temple }
					modifier = { factor = 1.3 venture_leading_trigger = yes }
					set_character_flag = venture_fortune
					if = {
						limit = { has_character_flag = venture_dest_temple }
						character_event = { id = venture.25 }
					}
					else = {
						character_event = { id = venture.24 }
					}
				}
			}
		}
	}
}

		NOT = { has_character_flag = venture_slann_answering }
	}

	immediate = {
		random_list = {
			80 = { # The Slann sleep on
				modifier = { factor = 0.75 check_variable = { which = venture_slann_wrath value = 2 } }

				trigger = { NOT = { has_character_flag = venture_slann_warned } }
				character_event = { id = venture.71 }
			}
			20 = {
				trigger = {
					has_character_flag = venture_slann_warned

# venture.120 Swamp Town (hidden)
# Once a game, from 2064 IC: a band of gamblers and charlatans run out of
# Port Reaver (in our lore, the Ranaldan Moritz Eulenspiegel and his
# friends) flee south past the ruins of Cadavo and found Swamp Town on a
# Lustrian harbour: the vilest and the safest place in Lustria to trade.
# The founder is a generated character; the county goes to them as a free
# colony, and the Venturers hear of it (venture.121).
# Scope: ROOT is any playable character (on_five_year_pulse).
character_event = {
	id = venture.120
	hide_window = yes

	is_triggered_only = yes # on_five_year_pulse

	only_playable = yes

	trigger = {
		NOT = { has_global_flag = venture_swamp_town }
		year = 2064
		any_province = { venture_swamp_town_site_trigger = yes }
	}

	immediate = {
		set_global_flag = venture_swamp_town
		random_province = {
			limit = { venture_swamp_town_site_trigger = yes }
			preferred_limit = {
				any_neighbor_province = {
					owner = {
						OR = {
							has_landed_title = d_port_reaver
							has_landed_title = d_cadavo
							any_liege = {
								OR = {
									has_landed_title = d_port_reaver
									has_landed_title = d_cadavo
								}
							}
						}
					}
				}
			}
			save_event_target_as = venture_colony_site
			county = { save_event_target_as = venture_colony_title }
		}
		random_list = {
			50 = {
				create_character = {
					random_traits = yes
					dynasty = random
					religion = sigmarite
					culture = neueslander
					female = no
					age = 38
					attributes = { intrigue = 12 diplomacy = 10 }
					trait = deceitful
					trait = gamer
				}
			}
			30 = {
				create_character = {
					random_traits = yes
					dynasty = random
					religion = myrmidian
					culture = oltremari
					female = no
					age = 38
					attributes = { intrigue = 12 diplomacy = 10 }
					trait = deceitful
					trait = gamer
				}
			}
			20 = {
				create_character = {
					random_traits = yes
					dynasty = random
					religion = myrmidian
					culture = oltremari
					female = yes
					age = 38
					attributes = { intrigue = 12 diplomacy = 10 }
					trait = deceitful
					trait = gamer
				}
			}
		}
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			save_event_target_as = venture_swamp_founder
			wealth = 300
		}
		event_target:venture_colony_title = {
			usurp_title_plus_barony_if_unlanded = { target = event_target:venture_swamp_founder type = invasion }
		}
		event_target:venture_swamp_founder = { set_defacto_liege = THIS }
		event_target:venture_colony_site = {
			culture = event_target:venture_swamp_founder
			set_province_flag = venture_colony
			set_province_flag = venture_swamp_town
			add_province_modifier = { name = venture_swamp_town duration = -1 }
			add_province_modifier = { name = venture_new_colony years = 10 }
		}
		any_player = {
			limit = {
				OR = {
					venture_member_trigger = yes
					any_realm_province = { region = world_new_world_lustria }
				}
			}
			character_event = { id = venture.121 days = 7 }
		}
	}
}
