# The Order of Mysteries
# Namespace: verena (was DR_societies.1000 - 1122, 1200 - 1299, 1404 - 1409
#   and 1417 - 1456 in DR_societies_mnm_events.txt, the Cult of Verena copy
#   of MNM.1000 - 1456 in mnm_hermetics_events.txt, and MNM.1200 - 1299 of
#   mnm_hermetics_events.txt)
# Faction: Old World faiths
# Serves: the Order of Mysteries (cult_of_verena), the Lorekeepers of
#   Verena who recover lost and forbidden knowledge
#
# Verena gave mankind the written word, and the Order of Mysteries goes
# after what was written and lost: in musty libraries, distant ruins and
# haunted tombs. Its brothers and sisters send each other on missions: to
# take on an apprentice, to build a library (from a reading room to a
# rival of the Grand Temple of Nuln, with copyists, lamps, bindings and
# books to choose, and fire, a Sigmarite mob or damp along the way), to
# recover a lost tome, scroll or tablet (once in a life, the Tablet of
# Lybaras from the sealed archives of Nehekhara), and to raid the hoard
# of a brother who locks his books away like the Scrollbearers. Its
# priests pray the Blind Maiden over still water to see what is to come,
# and its High Priests read the star signs of the Old World above a
# child's birth. Lore from WFRP 2E Tome of Salvation (the Cult of Verena,
# its orders, sects and miracles: the Blind Maiden, Owl's Wisdom) and The
# WFRP Companion (the star signs).
#
# Missions                             verena.1
# The library                          verena.10 - verena.22
# The apprentice                       verena.30 - verena.35
# The lost texts                       verena.40 - verena.49
# The Scrollbearer's hoard             verena.50 - verena.76
# The scrying                          verena.100 - verena.121
# The horoscope                        verena.130 - verena.141
#
# Callers: on_society_bi_yearly_pulse (verena.1), the decision
#   build_library in decisions/wh_cult_societies_decisions.txt
#   (verena.10), the gain_effect of title_hermetics_apprentice in
#   common/minor_titles/00_minor_titles.txt (verena.35), the decisions
#   hermetics_perform_scrying (verena.100) and hermetics_make_horoscope
#   (verena.130) in decisions/mnm_hermetics_decisions.txt, on_birth
#   (verena.105), on_death (106), on_adulthood (136), on_new_holder,
#   on_new_holder_inheritance and on_new_holder_usurpation (110, 111),
#   GRANT_LANDED_TITLE_INTERACTION_ACCEPT_EVENT (112), the combat pulse in
#   events/WHG_battle_events.txt (102, 103), the pet deaths in
#   common/scripted_effects/wh_pet_effects.txt, WHG_battle_events and
#   WHG_hunting_events (108), and the sickness and recovery effects in
#   common/scripted_effects/00_scripted_effects.txt (115 - 118). Triggers
#   in common/scripted_triggers/WHG_verena_triggers.txt, effects in
#   common/scripted_effects/WHG_verena_effects.txt, modifiers in
#   common/event_modifiers/WHG_verena_modifiers.txt.
# Saved: event targets hermetic_collaborator, apprentice_candidate,
#   sacrifice_for_science, ruin_site, temple_site, infiltration_target,
#   infiltration_companion, mnm_scrying_dead, new_holder_target,
#   old_holder_target, verena_omen_title, verena_omen_courtier,
#   horoscope_child, horoscope_maker and inspiring_adult; variables
#   laboratory_level, lab_ambition and horoscope_destiny_points; flags

#   verena_library_next_<fitting> (scribes, lamps, bindings, books),

#   infiltration_mutual_dislike, infiltration_rich, bribed_guards,

#   caught_destroying, great_theft, good_theft, ok_theft, bad_theft,

#   battle_focused_scrying_flag, relationship_focused_scrying_flag,

# verena.1 The missions of the Order
# Hidden. Every other year a member gets a mission from a brother or sister
# of the Order, a Lorekeeper of his realm or his own apprentice: find an
# apprentice, build a library, recover a lost text, raid the hoard of a
# Scrollbearer or watch the stars. The AI mostly settles its missions
# without the chain.
# Scope: a member of the Order of Mysteries (cult_of_verena).
character_event = {
	id = verena.1
	hide_window = yes
	is_triggered_only = yes # on_society_bi_yearly_pulse

	is_in_society = yes

	trigger = {
		society_member_of = cult_of_verena
		has_any_quest = no
		NOT = { has_character_modifier = quest_cooldown_timer }
		is_inaccessible_trigger = no
		prisoner = no
		society = {
			any_society_member = {
				NOT = { character = ROOT }
				prisoner = no
				is_within_diplo_range = ROOT
				opinion = { who = ROOT value = 0 }
			}
		}
	}

	immediate = {
		# Who gives the mission
		random_list = {
			20 = { # A member anywhere
				society = {
					random_society_member = {
						limit = {
							NOT = { character = ROOT }
							prisoner = no
							is_within_diplo_range = ROOT
							opinion = { who = ROOT value = 0 }
						}
						save_event_target_as = hermetic_collaborator
					}
				}
			}
			40 = { # A member of the realm
				modifier = {
					factor = 0
					society = {
						NOT = {
							any_society_member = {
								NOT = { character = ROOT }
								prisoner = no
								is_within_diplo_range = ROOT
								opinion = { who = ROOT value = 0 }
								same_realm = ROOT
							}
						}
					}
				}
				society = {
					random_society_member = {
						limit = {
							NOT = { character = ROOT }
							prisoner = no
							is_within_diplo_range = ROOT
							opinion = { who = ROOT value = 0 }
							same_realm = ROOT
						}
						save_event_target_as = hermetic_collaborator
					}
				}
			}
			40 = { # The apprentice
				modifier = {
					factor = 0
					NOT = { any_courtier_or_vassal = { has_minor_title = title_hermetics_apprentice prisoner = no } }
				}
				random_courtier_or_vassal = {
					limit = { has_minor_title = title_hermetics_apprentice prisoner = no }
					save_event_target_as = hermetic_collaborator
				}
			}
		}

		# The mission
		random_list = {
			20 = { # Find an apprentice
				modifier = {
					factor = 0
					OR = {
						ai = yes
						any_courtier_or_vassal = { has_minor_title = title_hermetics_apprentice }
					}
				}
				event_target:hermetic_collaborator = { character_event = { id = verena.30 days = 7 random = 7 } }
			}
			20 = { # Build a library
				modifier = {
					factor = 0
					OR = {
						has_character_flag = building_laboratory
						has_character_flag = built_laboratory
						is_landed = no
						AND = {
							ai = yes
							NOT = { society_rank == 4 }
						}
					}
				}
				event_target:hermetic_collaborator = { character_event = { id = verena.20 days = 7 random = 7 } }
			}
			15 = { # Recover a lost text
				modifier = {
					factor = 0
					is_landed = no
				}
				event_target:hermetic_collaborator = { character_event = { id = verena.40 days = 7 random = 7 } }
			}
			10 = { # Raid the hoard of a Scrollbearer
				modifier = {
					factor = 0
					society = {
						NOT = {
							any_society_member = {
								is_within_diplo_range = ROOT
								NOR = {
									character = ROOT
									character = event_target:hermetic_collaborator
									reverse_opinion = { who = event_target:hermetic_collaborator value = 20 }
									reverse_opinion = { who = ROOT value = 20 }
									has_character_flag = infiltration_rival
									has_character_flag = infiltration_mutual_dislike
									has_character_flag = infiltration_rich
								}
							}
						}
					}
				}
				event_target:hermetic_collaborator = { character_event = { id = verena.50 days = 7 random = 7 } }
			}
			10 = { # Watch the stars
				event_target:hermetic_collaborator = { character_event = { id = DR_societies.1828 days = 7 random = 7 } }
			}
			105 = { # The AI settles the mission by itself
				modifier = {
					factor = 0
					ai = no
				}
				random_list = {
					70 = {
						modifier = { factor = 0.9 society_influence >= 30 }
						modifier = { factor = 0.9 society_influence >= 40 }
						modifier = { factor = 0.8 society_influence >= 50 }
						modifier = { factor = 0.7 society_influence >= 70 }
						modifier = { factor = 0.6 society_influence >= 80 }
						modifier = { factor = 0.5 society_influence >= 90 }
						add_mission_succeed_influence_effect = yes
					}
					30 = {
						add_mission_fail_influence_effect = yes
					}
				}
			}
			175 = { # Nothing this time
				modifier = {
					factor = 0
					ai = no
				}
			}
		}
	}
}

# verena.14 The library is finished
# Hidden. A learned apprentice adds one level to the library. The level
# decides the library the member gets (verena.15 or verena.16).
# Scope: the member.
character_event = {
	id = verena.14
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		set_character_flag = built_laboratory
		clr_character_flag = building_laboratory
		verena_library_clear_effect = yes
		if = {
			limit = {
				any_courtier_or_vassal = {
					has_minor_title = title_hermetics_apprentice
					learning = 12
				}
			}
			change_variable = { which = laboratory_level value = 1 }
		}
		if = {
			limit = { check_variable = { which = laboratory_level value = 18 } }
			narrative_event = { id = verena.16 }
		}
		else = {
			character_event = { id = verena.15 }
		}
	}
}

# one, each with knowledge for the realm.
# Scope: the member.

	id = verena.15
	desc = {
		trigger = { NOT = { check_variable = { which = laboratory_level value = 6 } } }
		text = EVTDESCverena15_simple
	}
	desc = {
		trigger = {
			check_variable = { which = laboratory_level value = 6 }
			NOT = { check_variable = { which = laboratory_level value = 12 } }
		}
		text = EVTDESCverena15_normal
	}
	desc = {
		trigger = { check_variable = { which = laboratory_level value = 12 } }
		text = EVTDESCverena15_good
	}

	border = GFX_event_normal_frame_religion

	option = {
		name = {
			trigger = { NOT = { check_variable = { which = laboratory_level value = 6 } } }
			text = EVTOPTAverena15_simple
		}
		name = {
			trigger = { check_variable = { which = laboratory_level value = 6 } }
			text = EVTOPTAverena15
		}

			limit = { NOT = { check_variable = { which = laboratory_level value = 6 } } }

	}

# verena.140 The horoscope chain
# Hidden. Moves the chain on to its next stage: the reading (verena.131),
# the child hears of it at four (132), the first signs at six (133), the
# child's choice at thirteen (134, 135) and the yearly reckoning from then
# on (141). If the parent dies on the way, the child hears of it from
# others and the parent's stages are skipped.
# Scope: the child; event target horoscope_maker.
character_event = {
	id = verena.140
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_alive = yes
	}

	immediate = {
		if = {
			limit = { event_target:horoscope_maker = { is_alive = yes } }
			trigger_switch = {
				on_trigger = has_character_flag
				horoscope_destiny_revealed = {
					clr_character_flag = horoscope_destiny_revealed
					event_target:horoscope_maker = { character_event = { id = verena.131 } }
				}
				horoscope_child_receives_horoscope = {
					if = {
						limit = { age = 4 }
						clr_character_flag = horoscope_child_receives_horoscope
						set_character_flag = horoscope_maker_receives_update
						character_event = { id = verena.132 }
					}
					else = {
						character_event = { id = verena.140 days = 365 }
					}
				}
				horoscope_maker_receives_update = {
					if = {
						limit = { age = 6 }
						clr_character_flag = horoscope_maker_receives_update
						set_character_flag = horoscope_child_contemplates_their_destiny
						event_target:horoscope_maker = { character_event = { id = verena.133 } }
					}
					else = {
						character_event = { id = verena.140 days = 365 }
					}
				}
				horoscope_child_contemplates_their_destiny = {
					verena_horoscope_contemplate_effect = yes
				}
				horoscope_parent_is_informed = {
					clr_character_flag = horoscope_parent_is_informed
					set_character_flag = horoscope_child_evaluate
					event_target:horoscope_maker = { character_event = { id = verena.135 } }
				}
				horoscope_child_evaluate = {
					character_event = { id = verena.141 }
				}
			}
		}
		else = {
			set_character_flag = horoscope_maker_died
			trigger_switch = {
				on_trigger = has_character_flag
				horoscope_destiny_revealed = {
					clr_character_flag = horoscope_destiny_revealed
					clean_up_horoscope_effect = yes
				}
				horoscope_child_receives_horoscope = {
					clr_character_flag = horoscope_child_receives_horoscope
					character_event = { id = verena.132 }
				}
				horoscope_maker_receives_update = {
					clr_character_flag = horoscope_maker_receives_update
					set_character_flag = horoscope_child_contemplates_their_destiny
					character_event = { id = verena.140 days = 365 }
				}
				horoscope_child_contemplates_their_destiny = {
					verena_horoscope_contemplate_effect = yes
				}
				horoscope_parent_is_informed = {
					clr_character_flag = horoscope_parent_is_informed
					set_character_flag = horoscope_child_evaluate
					character_event = { id = verena.140 days = 365 }
				}
				horoscope_child_evaluate = {
					character_event = { id = verena.141 }
				}
			}
		}
	}
}

# verena.141 The yearly reckoning
# Hidden. From seventeen on, enough destiny points fulfil the destiny
# (verena.137) and few enough avoid it (138). Until then the points drift
# a little each year: up for a bright child, down for a dull one.
# Was MNM.1298 and 1299, merged.
# Scope: the child.
character_event = {
	id = verena.141
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				age = 17
				check_variable = { which = horoscope_destiny_points value = 10 }
			}
			character_event = { id = verena.137 }
		}
		else_if = {
			limit = {
				age = 17
				NOT = { check_variable = { which = horoscope_destiny_points value = -10 } }
			}
			character_event = { id = verena.138 }
		}
		else = {
			if = {
				limit = { is_dumb_trigger = no }
				change_variable = { which = horoscope_destiny_points value = 1 }
			}
			else = {
				subtract_variable = { which = horoscope_destiny_points value = 1 }
			}
			character_event = { id = verena.140 days = 365 random = 100 }
		}
	}
}
