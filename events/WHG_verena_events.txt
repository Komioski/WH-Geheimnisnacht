# The Order of Mysteries
# Namespace: verena (was DR_societies.1000 - 1122, 1404 - 1409 and
#   1417 - 1456 in DR_societies_mnm_events.txt, the Cult of Verena copy of
#   MNM.1000 - 1456 in mnm_hermetics_events.txt)
# Faction: Old World faiths
# Serves: the Order of Mysteries (cult_of_verena), the Lorekeepers of
#   Verena who recover lost and forbidden knowledge
#
# Verena gave mankind the written word, and the Order of Mysteries goes
# after what was written and lost: in musty libraries, distant ruins and
# haunted tombs. Its brothers and sisters send each other on missions: to
# take on an apprentice, to build a library (from a reading room to a
# rival of the Grand Temple of Nuln, with copyists, lamps, bindings and
# books to choose, and fire, a Sigmarite mob or damp along the way), to
# recover a lost tome, scroll or tablet (once in a life, the Tablet of
# Lybaras from the sealed archives of Nehekhara), and to raid the hoard
# of a brother who locks his books away like the Scrollbearers. Lore from
# WFRP 2E Tome of Salvation (the Cult of Verena, its orders and sects).
#
# Missions                             verena.1
# The library                          verena.10 - verena.22
# The apprentice                       verena.30 - verena.35
# The lost texts                       verena.40 - verena.49
# The Scrollbearer's hoard             verena.50 - verena.76
#
# Callers: on_society_bi_yearly_pulse (verena.1), the decision
#   build_library in decisions/wh_cult_societies_decisions.txt
#   (verena.10) and the gain_effect of title_hermetics_apprentice in
#   common/minor_titles/00_minor_titles.txt (verena.35). Triggers in
#   common/scripted_triggers/WHG_verena_triggers.txt, effects in
#   common/scripted_effects/WHG_verena_effects.txt.
# Saved: event targets hermetic_collaborator, apprentice_candidate,
#   sacrifice_for_science, ruin_site, temple_site, infiltration_target and
#   infiltration_companion; variables laboratory_level and lab_ambition;
#   flags building_laboratory, built_laboratory, hidden_lab,
#   very_hidden_lab, verena_library_<fitting> and
#   verena_library_next_<fitting> (scribes, lamps, bindings, books),
#   verena_lybaras_map, verena_lybaras_sought, do_not_disturb,
#   A/B/C/D_SUCCESS and _FAIL, infiltration_rival,
#   infiltration_mutual_dislike, infiltration_rich, bribed_guards,
#   guards_wake_up, distraction_failed, caught_stealing,
#   caught_destroying, great_theft, good_theft, ok_theft, bad_theft,
#   lab_destruction, lab_damage and lab_explosion_death.
# Dropped: the alchemist's laboratory of the vanilla Hermetic Society
#   (glass, crucibles and smiths' tools, now the fittings of a library),
#   the four fitting events DR_societies.1103 - 1106 (merged into
#   verena.13), the five finished-library events 1108 - 1112 (merged into
#   verena.15 and 16) and the hidden tier event 1107 (verena.14). The
#   Alexandria and Emerald Tablet option, commented out in the copy, is
#   back as the sealed archive of Lybaras; emerald_tablet is now the
#   Tablet of Lybaras. The raid wrecks the scriptorium but never burns a
#   book on purpose: the Order wept more for the books of Wolfenburg than
#   for its dead.

# verena.1 The missions of the Order
# Hidden. Every other year a member gets a mission from a brother or sister
# of the Order, a Lorekeeper of his realm or his own apprentice: find an
# apprentice, build a library, recover a lost text, raid the hoard of a
# Scrollbearer or watch the stars. The AI mostly settles its missions
# without the chain.
# Scope: a member of the Order of Mysteries (cult_of_verena).
character_event = {
	id = verena.1
	hide_window = yes
	is_triggered_only = yes # on_society_bi_yearly_pulse

	is_in_society = yes

	trigger = {
		society_member_of = cult_of_verena
		has_any_quest = no
		NOT = { has_character_modifier = quest_cooldown_timer }
		is_inaccessible_trigger = no
		prisoner = no
		society = {
			any_society_member = {
				NOT = { character = ROOT }
				prisoner = no
				is_within_diplo_range = ROOT
				opinion = { who = ROOT value = 0 }
			}
		}
	}

	immediate = {
		# Who gives the mission
		random_list = {
			20 = { # A member anywhere
				society = {
					random_society_member = {
						limit = {
							NOT = { character = ROOT }
							prisoner = no
							is_within_diplo_range = ROOT
							opinion = { who = ROOT value = 0 }
						}
						save_event_target_as = hermetic_collaborator
					}
				}
			}
			40 = { # A member of the realm
				modifier = {
					factor = 0
					society = {
						NOT = {
							any_society_member = {
								NOT = { character = ROOT }
								prisoner = no
								is_within_diplo_range = ROOT
								opinion = { who = ROOT value = 0 }
								same_realm = ROOT
							}
						}
					}
				}
				society = {
					random_society_member = {
						limit = {
							NOT = { character = ROOT }
							prisoner = no
							is_within_diplo_range = ROOT
							opinion = { who = ROOT value = 0 }
							same_realm = ROOT
						}
						save_event_target_as = hermetic_collaborator
					}
				}
			}
			40 = { # The apprentice
				modifier = {
					factor = 0
					NOT = { any_courtier_or_vassal = { has_minor_title = title_hermetics_apprentice prisoner = no } }
				}
				random_courtier_or_vassal = {
					limit = { has_minor_title = title_hermetics_apprentice prisoner = no }
					save_event_target_as = hermetic_collaborator
				}
			}
		}

		# The mission
		random_list = {
			20 = { # Find an apprentice
				modifier = {
					factor = 0
					OR = {
						ai = yes
						any_courtier_or_vassal = { has_minor_title = title_hermetics_apprentice }
					}
				}
				event_target:hermetic_collaborator = { character_event = { id = verena.30 days = 7 random = 7 } }
			}
			20 = { # Build a library
				modifier = {
					factor = 0
					OR = {
						has_character_flag = building_laboratory
						has_character_flag = built_laboratory
						is_landed = no
						AND = {
							ai = yes
							NOT = { society_rank == 4 }
						}
					}
				}
				event_target:hermetic_collaborator = { character_event = { id = verena.20 days = 7 random = 7 } }
			}
			15 = { # Recover a lost text
				modifier = {
					factor = 0
					is_landed = no
				}
				event_target:hermetic_collaborator = { character_event = { id = verena.40 days = 7 random = 7 } }
			}
			10 = { # Raid the hoard of a Scrollbearer
				modifier = {
					factor = 0
					society = {
						NOT = {
							any_society_member = {
								is_within_diplo_range = ROOT
								NOR = {
									character = ROOT
									character = event_target:hermetic_collaborator
									reverse_opinion = { who = event_target:hermetic_collaborator value = 20 }
									reverse_opinion = { who = ROOT value = 20 }
									has_character_flag = infiltration_rival
									has_character_flag = infiltration_mutual_dislike
									has_character_flag = infiltration_rich
								}
							}
						}
					}
				}
				event_target:hermetic_collaborator = { character_event = { id = verena.50 days = 7 random = 7 } }
			}
			10 = { # Watch the stars
				event_target:hermetic_collaborator = { character_event = { id = DR_societies.1828 days = 7 random = 7 } }
			}
			105 = { # The AI settles the mission by itself
				modifier = {
					factor = 0
					ai = no
				}
				random_list = {
					70 = {
						modifier = { factor = 0.9 society_influence >= 30 }
						modifier = { factor = 0.9 society_influence >= 40 }
						modifier = { factor = 0.8 society_influence >= 50 }
						modifier = { factor = 0.7 society_influence >= 70 }
						modifier = { factor = 0.6 society_influence >= 80 }
						modifier = { factor = 0.5 society_influence >= 90 }
						add_mission_succeed_influence_effect = yes
					}
					30 = {
						add_mission_fail_influence_effect = yes
					}
				}
			}
			175 = { # Nothing this time
				modifier = {
					factor = 0
					ai = no
				}
			}
		}
	}
}

# verena.14 The library is finished
# Hidden. A learned apprentice adds one level to the library. The level
# decides the library the member gets (verena.15 or verena.16).
# Scope: the member.
character_event = {
	id = verena.14
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		set_character_flag = built_laboratory
		clr_character_flag = building_laboratory
		verena_library_clear_effect = yes
		if = {
			limit = {
				any_courtier_or_vassal = {
					has_minor_title = title_hermetics_apprentice
					learning = 12
				}
			}
			change_variable = { which = laboratory_level value = 1 }
		}
		if = {
			limit = { check_variable = { which = laboratory_level value = 18 } }
			narrative_event = { id = verena.16 }
		}
		else = {
			character_event = { id = verena.15 }
		}
	}
}

# verena.15 A library of my own
# Level below 6: a simple library; below 12: a library; else an impressive
# one, each with knowledge for the realm.
# Scope: the member.
character_event = {
	id = verena.15
	desc = {
		trigger = { NOT = { check_variable = { which = laboratory_level value = 6 } } }
		text = EVTDESCverena15_simple
	}
	desc = {
		trigger = {
			check_variable = { which = laboratory_level value = 6 }
			NOT = { check_variable = { which = laboratory_level value = 12 } }
		}
		text = EVTDESCverena15_normal
	}
	desc = {
		trigger = { check_variable = { which = laboratory_level value = 12 } }
		text = EVTDESCverena15_good
	}
	picture = GFX_evt_library
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = {
			trigger = { NOT = { check_variable = { which = laboratory_level value = 6 } } }
			text = EVTOPTAverena15_simple
		}
		name = {
			trigger = { check_variable = { which = laboratory_level value = 6 } }
			text = EVTOPTAverena15
		}
		if = {
			limit = { NOT = { check_variable = { which = laboratory_level value = 6 } } }
			add_character_modifier = { name = cheap_laboratory duration = -1 }
		}
		else_if = {
			limit = { NOT = { check_variable = { which = laboratory_level value = 12 } } }
			economy_techpoints = 5
			add_character_modifier = { name = normal_laboratory duration = -1 }
		}
		else = {
			economy_techpoints = 15
			add_character_modifier = { name = good_laboratory duration = -1 }
		}
		verena_library_reward_effect = yes
	}
}

# verena.32 A servant with a quick mind
# Hidden. Still no apprentice: a young servant of the member's faith and
# culture turns up with a bright mind and a good trait (never a woman
# among races without women).
# Scope: the member.
character_event = {
	id = verena.32
	hide_window = yes
	is_triggered_only = yes

	trigger = { has_quest = quest_verena_find_apprentice }

	immediate = {
		random_list = {
			50 = { # A woman
				modifier = { factor = 0 is_no_females_race = yes }
				modifier = { factor = 1.5 is_female = no }
				create_random_priest = {
					random_traits = yes
					dynasty = none
					culture = ROOT
					religion = ROOT
					female = yes
					age = 24
				}
				new_character = {
					character_event = { id = aiclean.31 } # the race trait
					save_event_target_as = apprentice_candidate
				}
			}
			50 = { # A man
				modifier = { factor = 1.5 is_woman_trigger = yes }
				create_random_priest = {
					random_traits = yes
					dynasty = none
					culture = ROOT
					religion = ROOT
					female = no
					age = 24
				}
				new_character = {
					character_event = { id = aiclean.31 } # the race trait
					save_event_target_as = apprentice_candidate
				}
			}
		}
		event_target:apprentice_candidate = {
			remove_trait = slow
			remove_trait = imbecile
			remove_trait = inbred
			remove_trait = dull
			remove_trait = zealous
			remove_trait = cynical
			remove_trait = genius
			remove_trait = quick
			remove_trait = shrewd
			remove_trait = slothful
			random_list = { # A gift of the mind
				10 = { add_trait = genius }
				10 = { add_trait = quick }
				10 = { add_trait = shrewd }
				10 = { add_trait = scholar }
				10 = { add_trait = mystic }
				10 = { add_trait = gardener }
				10 = {
					modifier = { factor = 0 NOT = { has_dlc = "Reapers" } }
					add_trait = physician
				}
				40 = { }
			}
			random_list = { # A virtue of the scholar
				10 = { add_trait = erudite }
				10 = { add_trait = diligent }
				10 = {
					remove_trait = wroth
					add_trait = patient
				}
				10 = {
					remove_trait = lustful
					add_trait = chaste
				}
				30 = {
					remove_trait = content
					add_trait = ambitious
				}
			}
			while = {
				limit = { NOT = { learning = 10 } }
				change_learning = 1
			}
			character_event = { id = verena.33 }
		}
	}
}
