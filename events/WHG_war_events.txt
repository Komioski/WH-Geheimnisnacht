# Way of Life: the war focus
# Merged in during the Nov 12 deep pass (each keeps its own header below):
#   The sack of great cities (namespace greatsack, was WHG_great_sack_events)
#   Wars of hostages, faith and tribute (namespace warcb, was WHG_war_cb_events)
#
# Namespace: war (was WoL.11100 - WoL.11341 in wol_war_events.txt)
# Faction: Generic
#
# A ruler with the war focus trains body and mind for battle. Once in a
# while the focus pulse brings a morning run, a lecture for a lazy
# courtier, a spear-throwing contest, a sparring partner or a war game
# with wooden armies on the map table (war.10). Each success may harden
# the warrior (war.60): Warrior and Great Warrior, then the Duelist trait,
# or prestige, wrath, courage or strength. The pulse also brings insight:
# a better martial education, a new leader trait to choose (troops,
# terrain, the line of battle, the art of command, offence or defence),
# or a lesson to pass on to one of the ruler's commanders.
#
# The old WoL duel chain (WoL.11000 - 11014) is gone: duels are in
# WHG_duel_events.txt (duel namespace).
#
# The hub and training                 war.10 - war.22
# The war game                         war.25 - war.59
# Success in the focus                 war.60 - war.66
# Insight and leader traits            war.70 - war.75
# Teaching a commander                 war.80 - war.81, war.110 - war.111
#
# Fired from: on_focus_pulse in WHG_pulse_on_actions.txt (war.10, 70 - 75, 80)
#   and the feast events (war.60). Shared triggers and effects are in
#   WHG_focus_triggers.txt and WHG_focus_effects.txt.
###############################################################################

# war.10 The hub: one training event (on_focus_pulse).
character_event = {
	id = war.10
	hide_window = yes
	is_triggered_only = yes # on_focus_pulse, about one year in five (was MTTH 36 months)

	min_age = 16
	only_playable = yes
	only_capable = yes
	prisoner = no

	trigger = {
		has_focus = focus_war
		NOT = { trait = infirm }
		NOT = { has_character_flag = do_not_disturb }
		in_command = no
		prisoner = no
	}

	immediate = {
		random_list = {
			30 = {
				modifier = {
					OR = { has_character_modifier = run_training_timer trait = creature_ogre }
					factor = 0
				}
				character_event = { id = war.11 } # running
			}
			30 = {
				trigger = {
					NOT = { trait = slothful }
					any_courtier = { trait = slothful prisoner = no }
				}
				character_event = { id = war.12 } # slothful court
			}
			30 = {
				modifier = { has_character_modifier = spear_throwing_timer factor = 0 }
				character_event = { id = war.13 } # spear-throwing
			}
			30 = {
				modifier = { any_friend = { has_character_flag = sparring_partner } factor = 0 }
				character_event = { id = war.20 } # sparring-partner
			}
			30 = {
				modifier = {
					NOT = {
						any_friend = { has_character_flag = sparring_partner }
						has_character_flag = sparring_partner
					}
					factor = 0
				}
				random_friend = {
					limit = { has_character_flag = sparring_partner }
					save_event_target_as = sparring_partner
				}
				character_event = { id = war.21 } # sparring success
			}
			30 = {
				modifier = {
					NOT = {
						any_friend = { has_character_flag = sparring_partner }
						has_character_flag = sparring_partner
					}
					factor = 0
				}
				random_friend = {
					limit = { has_character_flag = sparring_partner }
					save_event_target_as = sparring_partner
				}
				character_event = { id = war.22 } # sparring accident
			}
			30 = {
				modifier = { has_character_modifier = wargame_timer factor = 0 }
				character_event = { id = war.25 }
			}
		}
	}
}

# war.25 Wargame start
character_event = {
	id = war.25
	desc = EVTDESCwar25
	picture = GFX_evt_war_planning
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	option = {
		name = EVTOPTAwar25
		hidden_tooltip = {
			# The invader: a vassal with a council seat, else a courtier
			# with one, else any able courtier.
			if = {
				limit = { any_vassal = { war_wargame_player_trigger = yes has_job_title = yes } }
				random_vassal = {
					limit = { war_wargame_player_trigger = yes has_job_title = yes }
					save_event_target_as = attacker
				}
			}
			else_if = {
				limit = { any_courtier = { war_wargame_player_trigger = yes has_job_title = yes } }
				random_courtier = {
					limit = { war_wargame_player_trigger = yes has_job_title = yes }
					save_event_target_as = attacker
				}
			}
			else_if = {
				limit = { any_courtier = { war_wargame_player_trigger = yes } }
				random_courtier = {
					limit = { war_wargame_player_trigger = yes }
					save_event_target_as = attacker
				}
			}
			else = {
				# No one fit to play: a child or a prisoner takes the table.
				random_courtier = {
					limit = { war_wargame_odd_player_trigger = yes }
					save_event_target_as = attacker
				}
			}
			# My fellow defender, picked the same way.
			if = {
				limit = {
					any_vassal = {
						war_wargame_player_trigger = yes
						has_job_title = yes
						NOT = { character = event_target:attacker }
					}
				}
				random_vassal = {
					limit = {
						war_wargame_player_trigger = yes
						has_job_title = yes
						NOT = { character = event_target:attacker }
					}
					save_event_target_as = defender
				}
			}
			else_if = {
				limit = {
					any_courtier = {
						war_wargame_player_trigger = yes
						has_job_title = yes
						NOT = { character = event_target:attacker }
					}
				}
				random_courtier = {
					limit = {
						war_wargame_player_trigger = yes
						has_job_title = yes
						NOT = { character = event_target:attacker }
					}
					save_event_target_as = defender
				}
			}
			else_if = {
				limit = {
					any_courtier = {
						war_wargame_player_trigger = yes
						NOT = { character = event_target:attacker }
					}
				}
				random_courtier = {
					limit = {
						war_wargame_player_trigger = yes
						NOT = { character = event_target:attacker }
					}
					save_event_target_as = defender
				}
			}
			else = {
				random_courtier = {
					limit = {
						war_wargame_odd_player_trigger = yes
						NOT = { character = event_target:attacker }
					}
					save_event_target_as = defender
				}
			}
			add_character_modifier = { modifier = wargame_timer days = 1095 hidden = yes }
			character_event = { id = war.26 }
		}
	}

	option = {
		name = EVTOPTBwar25
	}
}

# smoke: the city's lord, that lord's lieges, and every player ruler of the
# city's faiths hear that it has fallen. A sacked city pays little tax and
# raises few men for ten years, and cannot be sacked again for fifty. A
# horde spares its own kind (greatsack_victim_trigger), and the usual
# sacking events (chaoshorde.30 - chaoshorde.42, nomad.24, waaagh.30,

# greatsack.1 The city falls
# The horde's lord sacks the seat of a king or an emperor: plunder it
# (wealth) or raze it (prestige, more buildings burned, piety for a lord of
# the Dark Gods). Either way two buildings burn and the city is sacked for
# ten years. The city's lord, every liege above, and the player rulers of the
# city's faith group hear of it (greatsack.2).
# Scope: ROOT is the owner of the winning army, FROM the holding taken.
narrative_event = {
	id = greatsack.1
	title = EVTNAMEgreatsack1
	desc = {
		trigger = { greatsack_chaos_trigger = yes }
		text = EVTDESCgreatsack1_chaos
	}
	desc = {
		trigger = { greatsack_greenskin_trigger = yes }
		text = EVTDESCgreatsack1_greenskin
	}
	desc = {
		trigger = { greatsack_beastman_trigger = yes }
		text = EVTDESCgreatsack1_beastman
	}
	desc = {
		trigger = { greatsack_nomad_trigger = yes }
		text = EVTDESCgreatsack1_nomad
	}
	picture = GFX_evt_mongols_pillage_oldgods
	border = GFX_event_narrative_frame_war

	is_triggered_only = yes # on_siege_over_winner

	trigger = {
		greatsack_horde_trigger = yes
		FROM = {
			greatsack_great_city_trigger = yes
			greatsack_victim_trigger = yes
			location = {
				owner = {
					top_liege = {
						NOT = { character = ROOT }
						ROOT = { NOT = { is_vassal_or_below = PREV } }
					}
				}
			}
		}
	}

	immediate = {
		FROM = {
			save_event_target_as = great_sack_holding
			destroy_random_building = yes
			destroy_random_building = yes
			add_holding_modifier = { modifier = sacked months = 12 } # The usual sacking leaves it be for a year
			location = {
				save_event_target_as = great_sack_province
				set_province_flag = great_city_sacked
				add_province_modifier = {
					name = great_city_sacked
					duration = 3650 # Ten years
				}
				province_event = { id = greatsack.3 days = 18250 }
			}
		}
	}

	option = { # Plunder it
		name = EVTOPTAgreatsack1
		wealth = 150
		prestige = 50
		ai_chance = {
			factor = 1
			modifier = { factor = 2 trait = greedy }
			modifier = { factor = 0.5 greatsack_beastman_trigger = yes }
		}
	}
	option = { # Raze it
		name = EVTOPTBgreatsack1
		prestige = 150
		if = {
			limit = { religion_group = chaos_gods_group }
			piety = 50
		}
		event_target:great_sack_holding = {
			destroy_random_building = yes
		}
		ai_chance = {
			factor = 1
			modifier = { factor = 2 trait = wroth }
			modifier = { factor = 2 trait = cruel }
			modifier = { factor = 3 greatsack_beastman_trigger = yes }
		}
	}

	after = {
		event_target:great_sack_province = {
			owner = {
				character_event = { id = greatsack.2 }
				any_liege = {
					character_event = { id = greatsack.2 }
				}
			}
		}
		any_independent_ruler = {
			limit = {
				ai = no
				NOT = { character = ROOT }
				event_target:great_sack_province = {
					owner = {
						NOT = { top_liege = { character = PREVPREVPREV } } # Heard it already
						religion_group = PREVPREV
					}
				}
			}
			character_event = { id = greatsack.2 }
		}
	}
}
