###############################################################################
# Warbands
# Namespace: warband (was WB.25000 - WB.25037 in HF_warband_powers_events
#   and WB.1 - WB.10090 in hf_warband_events)
# Faction: Warrior societies
# Serves: the Chaos warbands (Undivided, Khorne, Nurgle, Tzeentch,
#   Slaanesh), the Skaven Warlords and the warbands of Gork and Mork
#
# The warband powers, each in the voice of its god, the Skaven clans or
# the greenskins.
# The Call to Glory (rank 4, not Gork and Mork, who have the Waaagh!): a
# champion at war calls on the warband, and fresh warriors join while the
# champion leads an army on land. The flow dries up after 20 bands unless
# the champion inspires or pays for more; rowdy bands demand plunder; the
# champion's own people flock to the capital. Rarely, once in a life, the
# Chosen (Blightkings, Stormvermin) come and stay for good.
# Summon Commander (rank 2): a champion joins the court and may take a
# commander's post.
# Summon Warriors (rank 3): a great host comes to the capital for one war;
# the power waits until peace.
#
# Greenskin and Skaven paths (rank 2): pick a way of war (the open field,
# the mountains, the caves or the forests; the tunnels, the Stormvermin or
# leading from the back). A new Skaven officer may raid the armour-pile.
# Chaos paths (rank 3): the duel, the art of conquest or the dungeon.
# Champion of the Gods (rank 4): the mark of the warband's god.
# Gifts: an Imp claimed by a Dark God (rank 2), Khorne's berserkers.
# Upkeep: AI recruits join every five years (Gork or Mork by temper),
# and a fallen warband rises again under a member of its faith.
#
# The Call to Glory            warband.1 - warband.7
# Summon Commander             warband.10
# Summon Warriors              warband.15 - warband.17
# Greenskin and Skaven paths   warband.20 - warband.21 (was WB.1 - WB.2)
# Chaos paths                  warband.30 - warband.38 (was WB.10020 - WB.10031)
# Champion of the Gods         warband.40 (was WB.10040)
# Imps and berserkers          warband.45 - warband.50 (was WB.10070 - WB.10080)
# Upkeep                       warband.60 - warband.63 (was WB.10050 - WB.10053)
#
# Saved: character modifiers call_to_glory, warband_summon_warriors_cooldown;
#   variable call_to_glory_variable; character flag warband_got_chosen;
#   earmarks call_to_glory_disciplined, call_to_glory_rowdy, warband_chosen,
#   warband_summoned_warriors; event target invited_warband_soldier.
#
# Callers:
#   decisions/hf_warrior_lodge_decisions.txt (warband_call_to_glory: 1,
#     warband_summon_warriors: 15),
#   decisions/employment_decisions.txt (warband_summon_commander: 10),
#   common/on_actions/00_on_actions.txt: on_war_ended_* (16).
#   decisions/hf_warrior_lodge_decisions.txt (warband_specialization: 20,
#     warband_choose_chaos_aspect: 30/31, warband_champion_training: 40,
#     warband_summon_familiar: 45, warband_khorne_berserker: 50),
#   decisions/wh_skaven_decisions.txt (Skaven path: 20),
#   events/mnm_misc_events.txt (Skaven promotion to rank 2: 21),
#   common/on_actions/00_on_actions_warbands.txt: on_five_year_pulse (60),
#     on_society_destroyed (63).
#   Call to Glory cleanup at peace: HF.25035/HF.25036, shared with the
#   warrior lodges.
# Support: common/scripted_triggers/wh_warband_triggers.txt,
#   common/scripted_effects/wh_warband_effects.txt
#
# Removed: a dead copy of the scarred-trait setup (WB.25020; HF.25020
# does it), dead copies of the lodge rank-up duel letters (WB.10060 -
# WB.10063; the shared HF.10060 - HF.10063 are used) and an uncalled
# Malice grant (WB.10090).
###############################################################################

# warband.2 The muster (hidden)
# Every month or so: a band arrives, the flow dries up, rowdy warriors
# grumble, the people flock to the capital or, once in a life, the Chosen
# come. Out of command it waits; at peace it ends. Scope: the caller.
character_event = {
	id = warband.2
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		random_list = {
			30 = { #A new band
				trigger = { warband_glory_ok_trigger = yes warband_glory_leading_trigger = yes }
				character_event = { id = warband.3 }
			}
			50 = { #Not leading an army: wait
				trigger = { warband_glory_ok_trigger = yes warband_glory_leading_trigger = no }
				character_event = { id = warband.2 days = 35 random = 5 }
			}
			100 = { #The call has lapsed
				trigger = { warband_glory_lapsed_trigger = yes }
				warband_glory_end_effect = yes
			}
			100 = { #The flow dries up
				trigger = {
					warband_glory_lapsed_trigger = no
					warband_glory_leading_trigger = yes
					check_variable = { which = "call_to_glory_variable" value = 20 }
				}
				character_event = { id = warband.4 }
			}
			5 = { #Rowdy warriors want more plunder
				trigger = {
					warband_glory_ok_trigger = yes
					warband_glory_leading_trigger = yes
					has_earmarked_regiments = call_to_glory_rowdy
					check_variable = { which = "call_to_glory_variable" value = 5 }
				}
				character_event = { id = warband.5 }
			}
			5 = { #The people flock to the capital
				trigger = {
					warband_glory_ok_trigger = yes
					warband_glory_leading_trigger = yes
					check_variable = { which = "call_to_glory_variable" value = 5 }
				}
				character_event = { id = warband.6 }
			}
			2 = { #The Chosen (rare, once in a life)
				trigger = {
					warband_glory_ok_trigger = yes
					warband_glory_leading_trigger = yes
					check_variable = { which = "call_to_glory_variable" value = 10 }
					NOT = { has_character_flag = warband_got_chosen }
				}
				character_event = { id = warband.7 }
			}
		}
	}
}

	picture = GFX_evt_duel_against_chaos

	immediate = {
		hidden_effect = {
			random_society_member = {
				limit = {
					same_society_as = ROOT
					ai = yes
					NOT = { character = ROOT }
					NOT = { block_general_event_trigger = yes }
					trait = duelist
				}
				save_event_target_as = mentor_target
			}
		}
	}

			limit = {
				OR = {
					reverse_has_opinion_modifier = {
						who = event_target:mentor_target
						modifier = opinion_budding_friendship
					}
					has_opinion_modifier = {
						who = event_target:mentor_target
						modifier = opinion_budding_friendship
					}
				}
			}
			tooltip = { add_friend = event_target:mentor_target }
		}
		else_if = {
			limit = {
				event_target:mentor_target = {
					potentially_interested_in_rivaling_root_trigger = no
				}
			}
			tooltip = {
				reverse_opinion = {
					who = event_target:mentor_target
					modifier = opinion_was_praised

				}
			}
		}
		else = {
			tooltip = {
				reverse_opinion = {
					who = event_target:mentor_target
					modifier = opinion_strangely_flattered
					years = 10
				}
			}
		}

# warband.60 Recruits for the warbands
# From on_five_year_pulse (common/on_actions/00_on_actions_warbands.txt).
# An independent AI ruler, two of the realm's lords and one other character
# may join a warband that is empty or short of members (warband.61).
# Scope: the ruler.
character_event = {
	id = warband.60
	hide_window = yes

	is_triggered_only = yes
	only_independent = yes

	immediate = {
		if = {
			limit = { ai = yes warband_recruit_trigger = yes }
			character_event = { id = warband.61 }
		}
		any_realm_lord = {
			limit = {
				ai = yes
				is_landed = yes
				NOT = { character = ROOT }
				warband_recruit_trigger = yes
			}
			score_value = { value = 10 warrior_recruit_score = yes }
			count = 2
			character_event = { id = warband.61 }
		}
		any_realm_character = {
			limit = { ai = yes warband_recruit_trigger = yes }
			score_value = { value = 10 warrior_recruit_score = yes }
			count = 1
			character_event = { id = warband.61 }
		}
	}
}

# warband.61 A recruit joins
# The warband of the recruit's god (Undivided worshippers, and Beastmen in
# particular, may choose one god), Gork or Mork by temper (brutal
# or kunnin' traits), or the Skaven Warlords. The first member
# becomes the leader. Rank-4 players hear of it (warband.62).
# Scope: the recruit.
character_event = {
	id = warband.61
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				religion_tzeentch_worshipper_trigger = yes
				can_join_society = warband_of_tzeentch
			}
			join_society = warband_of_tzeentch
		}
		else_if = {
			limit = {
				religion_nurgle_worshipper_trigger = yes
				can_join_society = warband_of_nurgle
			}
			join_society = warband_of_nurgle
		}
		else_if = {
			limit = {
				religion_slaanesh_worshipper_trigger = yes
				can_join_society = warband_of_slaanesh
			}
			join_society = warband_of_slaanesh
		}
		else_if = {
			limit = {
				religion_khorne_worshipper_trigger = yes
				can_join_society = warband_of_khorne
			}
			join_society = warband_of_khorne
		}
		else_if = {
			limit = { religion_chaos_worshipper_trigger = yes can_join_society = warband_of_chaos }
			random_list = {
				25 = { join_society = warband_of_chaos }
				5 = {
					trigger = { trait = creature_beastman can_join_society = warband_of_nurgle }
					join_society = warband_of_nurgle
				}
				3 = {
					trigger = { trait = creature_beastman can_join_society = warband_of_tzeentch }
					join_society = warband_of_tzeentch
				}
				5 = {
					trigger = { trait = creature_beastman can_join_society = warband_of_khorne }
					join_society = warband_of_khorne
				}
				5 = {
					trigger = { trait = creature_beastman can_join_society = warband_of_slaanesh }
					join_society = warband_of_slaanesh
				}
			}
		}
		else_if = {
			limit = {
				religion_group = greenskin_gods_group
				OR = { can_join_society = warband_gork can_join_society = warband_mork }
			}
			# Gork is brutal but kunnin', Mork is kunnin' but brutal: the
			# recruit's temper decides (warband_gork_weight_trigger etc.).
			random_list = {
				50 = {
					trigger = { can_join_society = warband_gork }
					modifier = { factor = 1.5 trait = wroth }
					modifier = { factor = 1.5 trait = brave }
					modifier = { factor = 1.5 trait = cruel }
					modifier = { factor = 1.3 trait = proud }
					modifier = { factor = 1.3 is_strong_trigger = yes }
					modifier = { factor = 1.3 martial = 14 }
					modifier = { factor = 1.2 culture_group = orc_group }
					modifier = { factor = 0.7 trait = craven }
					modifier = { factor = 0.7 trait = patient }
					join_society = warband_gork
				}
				50 = {
					trigger = { can_join_society = warband_mork }
					modifier = { factor = 1.5 trait = deceitful }
					modifier = { factor = 1.5 trait = schemer }
					modifier = { factor = 1.5 trait = paranoid }
					modifier = { factor = 1.3 trait = patient }
					modifier = { factor = 1.3 trait = ambitious }
					modifier = { factor = 1.3 intrigue = 14 }
					modifier = { factor = 1.2 culture_group = goblin_group }
					modifier = { factor = 1.3 has_magic_trigger = yes }
					modifier = { factor = 0.7 trait = honest }
					modifier = { factor = 0.7 trait = wroth }
					join_society = warband_mork
				}
			}
		}
		else_if = {
			limit = { culture_group = skaven_group can_join_society = skaven_warlords_society }
			join_society = skaven_warlords_society
		}

		if = {
			limit = { is_member_of_any_warband_trigger = yes }
			apply_random_minor_injury_effect = yes # The initiation was not gentle.
			if = {
				limit = {
					society = { NOT = { any_society_member = { is_society_grandmaster = yes } } }
				}
				set_society_grandmaster = yes
				add_society_currency_massive_effect = yes
			}
			society = {
				any_society_member = {
					limit = { ai = no society_rank = 4 }
					character_event = { id = warband.62 days = 1 random = 15 }
				}
			}
		}
	}
}
