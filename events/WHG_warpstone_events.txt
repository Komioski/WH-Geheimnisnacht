# Wyrdstone: the race for Mordheim, veins, meteors, mines and the shadow market
# Namespace: warpstone (was DoW_blackstone in dow_blackstone_events.txt)
# Faction: Generic
# Serves: every lord who covets wyrdstone: Imperial, Bretonnian and
#   Marienburger counts, the Skaven warlords, the Dark Lords of the undead,
#   and the dabblers in dark magic who trade in the shadow market
#
# In the year 1999 the twin-tailed comet struck Mordheim, the City of the
# Damned, and its ruins are thick with wyrdstone. Eleven years on, a lord
# who covets it may fund an expedition: a commander leads an army to seize a
# county of d_mordheim, or an agent slips into the ruins to bring stone
# home. Each warband follows its patron's people: Imperial state troops,
# Marienburgers, Bretonnian knights, Kislevites, Dawi treasure hunters,
# Skaven of the Under-Empire, the walking dead, the Possessed of the Pit,
# greenskins and beastmen, Druchii raiders or free companies. In the ruins
# the agent may find the stone, death, or stranger things: the Pit itself,
# a rival warband, the Sisters of Sigmar, the Possessed, or a relic of the
# old city. Once the mists lift, the lands of Mordheim are open to any war
# and the stone grows scarce. Elsewhere, wyrdstone is found in the mountains
# and the deep places, or falls from Morrslieb as a meteor. A mine yields
# stone each year until it is exhausted, but Skaven tunnel up from below
# and the miners sicken. The faithful may purge it instead. The shadow
# market buys and sells it, and the pious who deal there are talked about.
#
# The race for Mordheim             warpstone.1 - warpstone.19, 40 - 44
#   An expedition is funded           warpstone.1 - warpstone.6
#   The agent in the ruins            warpstone.10 - warpstone.13
#   Encounters in the ruins           warpstone.40 - warpstone.44
#   Wyrdstone from Mordheim           warpstone.15 - warpstone.16
#   The mists lift                    warpstone.18 - warpstone.19
# Wyrdstone in the earth            warpstone.20 - warpstone.28, 35 - 36
#   A vein is found                   warpstone.20 - warpstone.21
#   A meteor falls                    warpstone.22 - warpstone.24
#   The mines                         warpstone.25 - warpstone.28
#   The perils of the mine            warpstone.35 - warpstone.36
# The shadow market                 warpstone.30 - warpstone.33
#
# Callers: warpstone.1 from the fund_mordheim_expedition decision,
#   warpstone.30 from the warpstone_blackmarket decision
#   (wh_blackstone_decisions.txt); warpstone.15 and warpstone.25 every year
#   and the yearly mordheim_mist list (warpstone.18), the decade lists
#   warpstone_veins and warpstone_meteors (warpstone.20, 22), all in
#   wh_on_actions.txt. The war is the race_for_mordheim CB
#   (cb_types_dow_blackstone.txt), which clears maintaining_agents when it
#   ends. The mines are the blackstone_mine buildings (00_blackstone.txt).
# Saved: global flags clean_mordheim (the mists have lifted, read by many
#   CBs), blackstone_meteor_spotted (22-year cooldown) and warpstone_deposits
#   (a vein or meteor exists); character flags maintaining_agents,
#   mordheim_patron, mordheim_relic_found, martial_agent_mordheim,
#   intrigue_agent_mordheim, mercenary_agent_mordheim,
#   shadow_warpstone_dealings; province flags warpstone_vein,
#   warpstone_meteor, warpstone_ratmen_paid and warpstone_yield_1 - 3;
#   province variable blackstone_mining_progress; event targets
#   mordheim_target, mordheim_title, mordheim_beneficiary, mordheim_agent and
#   the_mine.
# Triggers in wh_warpstone_triggers.txt, effects in wh_warpstone_effects.txt.
#   Loc: localisation/00_WHG_warpstone_events.csv (decision, CB and tooltip
#   loc stay in 00_blackstone.csv).
###############################################################################

# warpstone.1 The holder of the target county
# Hidden. Saves the holder of the Mordheim county as the target and asks
# the patron to choose who leads the expedition (warpstone.2).
# Scope: ROOT is the county's holder. FROM: the county title. FROMFROM: the
#   patron who took the decision fund_mordheim_expedition.
character_event = {
	id = warpstone.1
	hide_window = yes
	is_triggered_only = yes # decision fund_mordheim_expedition

	immediate = {
		save_event_target_as = mordheim_target
		FROMFROM = { character_event = { id = warpstone.2 } }
	}
}

# warpstone.2 Who leads the expedition?
# The patron sends a skilled commander with an army, a skilled agent into
# the ruins, or, with no one fit at court, hires a captain for 75 gold.
# Scope: ROOT is the patron. FROM: the target's holder (warpstone.1).
#   FROMFROM: the county title.
character_event = {
	id = warpstone.2
	title = EVTNAMEwarpstone
	desc = EVTDESCwarpstone2
	picture = GFX_evt_mordheim

	is_triggered_only = yes # warpstone.1

	immediate = {
		FROMFROM = { save_event_target_as = mordheim_title }
		save_event_target_as = mordheim_beneficiary
	}

	option = { # A commander
		name = EVTOPTAwarpstone2
		trigger = {
			any_courtier = {
				warpstone_mordheim_agent_trigger = yes
				martial = 12
			}
		}
		random_courtier = {
			limit = {
				warpstone_mordheim_agent_trigger = yes
				martial = 12
			}
			set_character_flag = martial_agent_mordheim
			save_event_target_as = mordheim_agent
			character_event = { id = warpstone.3 days = 224 }
		}
		ai_chance = { factor = 50 }
	}
	option = { # An agent
		name = EVTOPTBwarpstone2
		trigger = {
			any_courtier = {
				warpstone_mordheim_agent_trigger = yes
				intrigue = 12
			}
		}
		random_courtier = {
			limit = {
				warpstone_mordheim_agent_trigger = yes
				intrigue = 12
			}
			set_character_flag = intrigue_agent_mordheim
			save_event_target_as = mordheim_agent
			character_event = { id = warpstone.3 days = 150 }
		}
		ai_chance = { factor = 50 }
	}
	option = { # Hire a captain
		name = EVTOPTCwarpstone2
		trigger = {
			wealth = 75
			NOT = {
				any_courtier = {
					warpstone_mordheim_agent_trigger = yes
					OR = {
						martial = 12
						intrigue = 12
					}
				}
			}
		}
		wealth = -75
		create_random_soldier = {
			female = no
			religion = ROOT
			culture = ROOT
			attributes = {
				martial = 8
				diplomacy = 4
				stewardship = 3
				intrigue = 12
				learning = 4
			}
		}
		new_character = {
			character_event = { id = aiclean.31 } # Racial trait
			set_character_flag = mercenary_agent_mordheim
			save_event_target_as = mordheim_agent
			character_event = { id = warpstone.3 days = 200 }
		}
		ai_chance = { factor = 50 }
	}
	option = { # Call it off
		name = EVTOPTDwarpstone2
		trigger = {
			NOT = { wealth = 75 }
			NOT = {
				any_courtier = {
					warpstone_mordheim_agent_trigger = yes
					OR = {
						martial = 12
						intrigue = 12
					}
				}
			}
		}
		clr_character_flag = maintaining_agents
	}
}

# warpstone.3 Into the City of the Damned
# The chosen one sets out. A commander marches on Mordheim (warpstone.4);
# an agent slips into the ruins (warpstone.10); a hired captain does
# either, as his skills lean. The patron checks a year on that the leader
# still lives (warpstone.6).
# Scope: ROOT is the leader. FROM: the patron (warpstone.2).
character_event = {
	id = warpstone.3
	title = EVTNAMEwarpstone
	desc = EVTDESCwarpstone3
	picture = GFX_evt_mordheim

	is_triggered_only = yes # warpstone.2

	option = {
		name = EVTOPTAwarpstone3
		add_trait = travelling
		if = {
			limit = { has_character_flag = martial_agent_mordheim }
			character_event = { id = warpstone.4 days = 364 }
		}
		else_if = {
			limit = { has_character_flag = intrigue_agent_mordheim }
			FROM = { character_event = { id = warpstone.10 days = 100 } }
		}
		else = {
			random_list = {
				50 = {
					modifier = { factor = 0.5 martial = 15 }
					modifier = { factor = 1.5 intrigue = 15 }
					FROM = { character_event = { id = warpstone.10 days = 225 } }
				}
				50 = {
					modifier = { factor = 1.5 martial = 15 }
					modifier = { factor = 0.5 intrigue = 15 }
					character_event = { id = warpstone.4 days = 225 }
				}
			}
		}
		FROM = { character_event = { id = warpstone.6 days = 400 } }
	}
}

# kind agent leaves stone for others. One search in five meets something in

# warpstone.26 A year in the mine
# Hidden. The mine yields one shard, two shards or a cluster (better from a
# meteor, worse from an improved mine on a thin vein), and the owner is
# told (warpstone.27). Once fifteen yields are dug, the vein is exhausted
# (warpstone.28). One year in ten Skaven tunnel in from below (warpstone.35,
# not under a Skaven owner or a mine that paid them off), and one in ten the
# miners sicken (warpstone.36).
# Scope: ROOT is the mine's province (warpstone.25).
province_event = {
	id = warpstone.26
	hide_window = yes
	is_triggered_only = yes # warpstone.25

	immediate = {
		save_event_target_as = the_mine
		if = {
			limit = { check_variable = { which = blackstone_mining_progress value = 15 } }
			province_event = { id = warpstone.28 days = 7 }
		}
		else = {
			random_list = {
				50 = {
					modifier = { factor = 0.8 warpstone_improved_mine_trigger = yes }
					modifier = { factor = 1.1 has_province_modifier = warpstone_meteor }
					change_variable = { which = blackstone_mining_progress value = 2 }
					set_province_flag = warpstone_yield_2
				}
				25 = {
					modifier = { factor = 1.1 warpstone_improved_mine_trigger = yes }
					modifier = { factor = 1.1 has_province_modifier = warpstone_meteor }
					change_variable = { which = blackstone_mining_progress value = 3 }
					set_province_flag = warpstone_yield_3
				}
				25 = {
					modifier = { factor = 0.8 warpstone_improved_mine_trigger = yes }
					modifier = { factor = 0.8 has_province_modifier = warpstone_meteor }
					change_variable = { which = blackstone_mining_progress value = 1 }
					set_province_flag = warpstone_yield_1
				}
			}
			owner = { character_event = { id = warpstone.27 days = 7 } }
			random_list = { # The perils of the mine
				80 = { }
				10 = { # Ratmen beneath
					modifier = { factor = 0 NOT = { warpstone_ratmen_target_trigger = yes } }
					modifier = { factor = 2 region = world_underempire }
					modifier = { factor = 1.5 region = world_karaz_ankor }
					owner = { character_event = { id = warpstone.35 days = 30 } }
				}
				10 = { # The miners sicken
					modifier = { factor = 0 has_province_modifier = warpdust_traces }
					modifier = { factor = 1.5 has_province_modifier = warpstone_meteor }
					owner = { character_event = { id = warpstone.36 days = 60 } }
				}
			}
		}
	}
}
