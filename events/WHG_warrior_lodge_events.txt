###############################################################################
# Warrior lodges
# Namespace: lodge (was HF.10003 - HF.10071 in hf_warrior_lodge_events)
# Faction: the warrior lodges (Warriors of Ulric, the Knightly Orders, the
#   Warriors Guild of the Dawi); the joining trial, the rank-up duel and the
#   AI rank-ups also serve the warbands (Chaos, Skaven, Gork and Mork)
#
# A character who asks to join a lodge or a warband is met by one of its
# champions and must win (or at least survive) a sparring duel against a
# member. The champion's voice is the society's voice: a Chosen of the Dark
# Gods, a Skaven Warlord, an Orc boss, a Wolf-priest's warrior, a knight of
# an Imperial order or a Dawi veteran. A parent in the society can push an
# adult child into the trial; the child's success or failure comes home to
# dinner. The powers of the lodges follow: the military path (duelist,
# strategist or hunter), warrior training, the toughness of greenskins, the
# berserk rage of Ulric's warriors and the Norscans (in the hall or in a
# cell) and the call to arms. Hidden upkeep fills the lodges with AI members,
# ranks them up and refounds a lodge that died out. A member at rank 3 can
# fight a Hero for a seat at the table; a Hero's household may follow the
# winner.
#
# The joining trial            lodge.1 - lodge.9
# A child joins the lodge      lodge.10 - lodge.15
# The military path            lodge.20 - lodge.28
# Warrior training             lodge.30
# Toughness                    lodge.35
# Berserk rage                 lodge.40 - lodge.44
# The call to arms             lodge.45
# Upkeep                       lodge.50 - lodge.54
# The seat at the table        lodge.60 - lodge.64
#
# Saved: character flags awaiting_initiation_trial, society_join_block,
#   they_accepted@, they_declined@, mock_duel, show_once_flag,
#   accepted_trial, refused_trial, parent_was_supportive, parent_was_mean,
#   choosing_military_aspect, choosing_warrior_training, opt_A/B/C,
#   no_harm, someone_is_hurt, hurt_self, may_join_society,
#   lodge_got_wolf_pelt, lodge_got_household (once per life); character
#   modifiers warband_failed_joining_cd (ten-year ban after a failed trial),
#   lodge_growth_cooldown, lost_rank_up_duel_cooldown; earmark
#   lodge_household; event targets martial_society_to_join, new_recruit,
#   champion_character, duel_target, combatant_1, combatant_2,
#   recruiting_parent, mentor_target, kinslayer_target.
#
# Callers: common/on_actions/00_on_actions.txt (on_character_ask_to_join_
#   society 1, on_five_year_pulse 50, on_society_bi_yearly_pulse 54,
#   on_society_destroyed 53); decisions/warrior_lodge_decisions.txt (20,
#   21, 30, 35, 40, 43, 45); decisions/hf_warrior_lodge_decisions.txt (10);
#   decisions/mnm_society_decisions.txt (60);
#   common/scripted_effects/01_scripted_effects_duels.txt (9, 61);
#   events/hfp_duel_events.txt (13).
# Support: common/scripted_triggers/wh_warband_triggers.txt (voices, lodge
#   recruits), common/scripted_effects/wh_warband_effects.txt
#   (lodge_household_spawn_effect), common/event_modifiers/
#   wh_warband_modifiers.txt (lodge_growth_cooldown).
#
# Fixed from vanilla: a captive leader made two champions answer the same
# recruit, and the gating event could send the verdict three times; the ten
# year ban after a failed trial was promised but never set; the hunter path
# showed Chaos torturer text to every lodge; "kill your son" killed no one
# and had no text; berserk rage looked for fellows of a lodge that does not
# exist in the mod and could hurt nobody while saying someone was hurt; the
# prison rage could pick no victim; the revival made every courtier join and
# paid the leader twice; AI recruits of the Old World faith picked the
# Knights or Ulric at fixed odds, now their temper decides; Kislev's
# warriors could never reach the Circle of the Bear; the growth notice used
# a cooldown modifier that did not exist; the seclusion clean-up HFP.19000
# moved to WHG_maintenance_events (aiclean.53).
###############################################################################

# lodge.1 A recruit knocks
# From on_character_ask_to_join_society. A free champion of the society
# (the leader if free, else the highest-ranked free member) is told about
# the recruit.
# Scope: ROOT is the recruit, FROM the society.
character_event = {
	id = lodge.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		NOT = { has_character_flag = is_being_recruited_to_warband_by_parent }
		NOT = { has_character_flag = awaiting_initiation_trial }
		FROM = {
			leader = {
				OR = {
					is_member_of_any_warrior_lodge_trigger = yes
					is_member_of_any_warband_trigger = yes
				}
			}
		}
	}

	immediate = {
		FROM = { save_event_target_as = martial_society_to_join }
		set_character_flag = society_join_block
		set_character_flag = awaiting_initiation_trial
		save_event_target_as = new_recruit
		# The leader if free, else the highest-ranked free member
		event_target:martial_society_to_join = {
			any_society_member = {
				limit = { prisoner = no }
				score_value = {
					value = 1
					additive_modifier = { value = 10000 is_society_grandmaster = yes }
					additive_modifier = { value = 1000 society_rank == 4 }
					additive_modifier = { value = 100 society_rank == 3 }
					additive_modifier = { value = 10 society_rank == 2 }
				}
				count = 1
				save_event_target_as = lodge_free_champion
			}
		}
		if = {
			limit = { event_target:lodge_free_champion = { always = yes } }
			event_target:lodge_free_champion = { character_event = { id = lodge.2 } }
		}
		else = {
			event_target:martial_society_to_join = {
				leader = { character_event = { id = lodge.2 } }
			}
		}
	}
}

# lodge.51 An AI recruit joins
# Ulric's faithful join the Warriors of Ulric (now and then the Knights);
# Kislevites join the Circle of the Bear; the other Old World faithful join
# the Knightly Orders or, if their temper is wild, the Warriors of Ulric;
# the Dawi join the Warriors Guild. The duel leaves a mark.
# Scope: ROOT is the recruit.
character_event = {
	id = lodge.51
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				OR = { religion = ulrican trait = norscan_patron_ulric }
				can_join_society = warriors_of_ulric
			}
			random_list = {
				25 = { join_society = warriors_of_ulric }
				5 = {
					trigger = { can_join_society = knights_of_old_world }
					modifier = { factor = 2 trait = honest }
					modifier = { factor = 2 trait = chaste }
					modifier = { factor = 0.5 trait = wroth }
					join_society = knights_of_old_world
				}
			}
		}
		else_if = {
			limit = { religion = kislev_gods can_join_society = circle_of_the_bear }
			join_society = circle_of_the_bear
		}
		else_if = {
			limit = {
				religion_group = old_world_group
				OR = {
					can_join_society = knights_of_old_world
					can_join_society = warriors_of_ulric
				}
			}
			# The Knightly Orders take the proud, the pious and the
			# well-mannered; Ulric's warriors take the wild ones.
			random_list = {
				20 = {
					trigger = { can_join_society = knights_of_old_world }
					modifier = { factor = 1.5 trait = honest }
					modifier = { factor = 1.5 trait = proud }
					modifier = { factor = 1.5 trait = zealous }
					modifier = { factor = 1.3 trait = chaste }
					modifier = { factor = 1.3 trait = duelist }
					modifier = { factor = 1.3 trait = strategist }
					modifier = { factor = 0.7 trait = wroth }
					modifier = { factor = 0.7 trait = gluttonous }
					join_society = knights_of_old_world
				}
				5 = {
					trigger = { can_join_society = warriors_of_ulric }
					modifier = { factor = 2 trait = wroth }
					modifier = { factor = 1.5 trait = brave }
					modifier = { factor = 1.5 trait = hunter }
					modifier = { factor = 1.5 trait = berserker }
					modifier = { factor = 1.3 trait = gluttonous }
					modifier = { factor = 1.3 trait = cynical }
					modifier = { factor = 0.5 trait = zealous }
					modifier = { factor = 0.5 trait = craven }
					join_society = warriors_of_ulric
				}
			}
		}
		else_if = {
			limit = { religion_group = dwarf_gods_group can_join_society = warriors_guild_dawi }
			join_society = warriors_guild_dawi
		}
		if = {
			limit = { is_in_society = yes }
			apply_random_minor_injury_effect = yes
			if = {
				limit = {
					society = { NOT = { any_society_member = { is_society_grandmaster = yes } } }
				}
				set_society_grandmaster = yes
				add_society_currency_massive_effect = yes
			}
			society = {
				any_society_member = {
					limit = { ai = no society_rank = 4 }
					character_event = { id = lodge.52 days = 1 random = 15 }
				}
			}
		}
	}
}
