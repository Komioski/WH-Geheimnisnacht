###############################################################################
# Warrior society events: warrior lodges, warbands and war aid
#
# Part 1: Warrior lodges
# Namespace: lodge (was HF.10003 - HF.10071 in hf_warrior_lodge_events)
# Faction: Warrior societies
# Serves: the warrior lodges (Warriors of Ulric, the Knightly Orders, the
#   Warriors Guild of the Dawi); the joining trial, the rank-up duel and the
#   AI rank-ups also serve the warbands (Chaos, Skaven, Gork and Mork)
#
# A character who asks to join a lodge or a warband is met by one of its
# champions and must win (or at least survive) a sparring duel against a
# member. The champion's voice is the society's voice: a Chosen of the Dark
# Gods, a Skaven Warlord, an Orc boss, a Wolf-priest's warrior, a knight of
# an Imperial order or a Dawi veteran. A parent in the society can push an
# adult child into the trial; the child's success or failure comes home to
# dinner. The powers of the lodges follow: the military path (duelist,
# strategist or hunter), warrior training, the toughness of greenskins, the
# berserk rage of Ulric's warriors and the Norscans (in the hall or in a
# cell) and the call to arms. Hidden upkeep fills the lodges with AI members,
# ranks them up and refounds a lodge that died out. A member at rank 3 can
# fight a Hero for a seat at the table; a Hero's household may follow the
# winner.
#
# The joining trial            lodge.1 - lodge.9
# A child joins the lodge      lodge.10 - lodge.15
# The military path            lodge.20 - lodge.28
# Warrior training             lodge.30
# Toughness                    lodge.35
# Berserk rage                 lodge.40 - lodge.44
# The call to arms             lodge.45
# Upkeep                       lodge.50 - lodge.54
# The seat at the table        lodge.60 - lodge.64
#
# Saved: character flags awaiting_initiation_trial, society_join_block,
#   they_accepted@, they_declined@, mock_duel, show_once_flag,
#   accepted_trial, refused_trial, parent_was_supportive, parent_was_mean,
#   choosing_military_aspect, choosing_warrior_training, opt_A/B/C,
#   no_harm, someone_is_hurt, hurt_self, may_join_society,
#   lodge_got_wolf_pelt, lodge_got_household (once per life); character
#   modifiers warband_failed_joining_cd (ten-year ban after a failed trial),
#   lodge_growth_cooldown, lost_rank_up_duel_cooldown; earmark
#   lodge_household; event targets martial_society_to_join, new_recruit,
#   lodge_hero_spoils,
#   champion_character, duel_target, combatant_1, combatant_2,
#   recruiting_parent, mentor_target, kinslayer_target.
#
# Callers: common/on_actions/00_on_actions.txt (on_character_ask_to_join_
#   society 1, on_five_year_pulse 50, on_society_bi_yearly_pulse 54,
#   on_society_destroyed 53); decisions/warrior_lodge_decisions.txt (20,
#   21, 30, 35, 40, 43, 45); decisions/hf_warrior_lodge_decisions.txt (10);
#   decisions/mnm_society_decisions.txt (60);
#   common/scripted_effects/01_scripted_effects_duels.txt (9, 61);
#   events/hfp_duel_events.txt (13).
# Support: common/scripted_triggers/wh_warband_triggers.txt (voices, lodge
#   recruits), common/scripted_effects/wh_warband_effects.txt
#   (lodge_household_spawn_effect), common/event_modifiers/
#   wh_warband_modifiers.txt (lodge_growth_cooldown).
#
# Fixed from vanilla: a captive leader made two champions answer the same
# recruit, and the gating event could send the verdict three times; the ten
# year ban after a failed trial was promised but never set; the hunter path
# showed Chaos torturer text to every lodge; "kill your son" killed no one
# and had no text; berserk rage looked for fellows of a lodge that does not
# exist in the mod and could hurt nobody while saying someone was hurt; the
# prison rage could pick no victim; the revival made every courtier join and
# paid the leader twice; AI recruits of the Old World faith picked the
# Knights or Ulric at fixed odds, now their temper decides; Kislev's
# warriors could never reach the Circle of the Bear; the growth notice used
# a cooldown modifier that did not exist; the seclusion clean-up HFP.19000
# moved to WHG_maintenance_events (aiclean.53).
#
# Also in this file (each part keeps its own header and namespace):
#   Warbands (namespace warband, was WHG_warband_events)
#   War aid (namespace waraid, was WHG_war_aid_events)
###############################################################################

# lodge.1 A recruit knocks
# From on_character_ask_to_join_society. A free champion of the society
# (the leader if free, else the highest-ranked free member) is told about
# the recruit.
# Scope: ROOT is the recruit, FROM the society.
character_event = {
	id = lodge.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		NOT = { has_character_flag = is_being_recruited_to_warband_by_parent }
		NOT = { has_character_flag = awaiting_initiation_trial }
		FROM = {
			leader = {
				OR = {
					is_member_of_any_warrior_lodge_trigger = yes
					is_member_of_any_warband_trigger = yes
				}
			}
		}
	}

	immediate = {
		FROM = { save_event_target_as = martial_society_to_join }
		set_character_flag = society_join_block
		set_character_flag = awaiting_initiation_trial
		save_event_target_as = new_recruit
		# The leader if free, else the highest-ranked free member
		event_target:martial_society_to_join = {
			any_society_member = {
				limit = { prisoner = no }
				score_value = {
					value = 1
					additive_modifier = { value = 10000 is_society_grandmaster = yes }
					additive_modifier = { value = 1000 society_rank == 4 }
					additive_modifier = { value = 100 society_rank == 3 }
					additive_modifier = { value = 10 society_rank == 2 }
				}
				count = 1
				save_event_target_as = lodge_free_champion
			}
		}
		if = {
			limit = { event_target:lodge_free_champion = { always = yes } }
			event_target:lodge_free_champion = { character_event = { id = lodge.2 } }
		}
		else = {
			event_target:martial_society_to_join = {
				leader = { character_event = { id = lodge.2 } }
			}
		}
	}
}

# the mountains, the caves or the forests; the tunnels, the Stormvermin or
# leading from the back). A new Skaven officer may raid the armour-pile.
# Chaos paths (rank 3): the duel, the art of conquest or the dungeon.
# Champion of the Gods (rank 4): the mark of the warband's god.
# Gifts: an Imp claimed by a Dark God (rank 2), Khorne's berserkers.
# Upkeep: AI recruits join every five years (Gork or Mork by temper),
# and a fallen warband rises again under a member of its faith.
#
# The Call to Glory            warband.1 - warband.7
# Summon Commander             warband.10
# Summon Warriors              warband.15 - warband.17
# Greenskin and Skaven paths   warband.20 - warband.21 (was WB.1 - WB.2)
# Chaos paths                  warband.30 - warband.38 (was WB.10020 - WB.10031)
# Champion of the Gods         warband.40 (was WB.10040)
# Imps and berserkers          warband.45 - warband.50 (was WB.10070 - WB.10080)
# Upkeep                       warband.60 - warband.63 (was WB.10050 - WB.10053)
#

#   variable call_to_glory_variable; character flag warband_got_chosen;
#   earmarks call_to_glory_disciplined, call_to_glory_rowdy, warband_chosen,
#   warband_summoned_warriors; event target invited_warband_soldier.

# warband.2 The muster (hidden)
# Every month or so: a band arrives, the flow dries up, rowdy warriors
# grumble, the people flock to the capital or, once in a life, the Chosen
# come. Out of command it waits; at peace it ends. Scope: the caller.
character_event = {
	id = warband.2
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		random_list = {
			30 = { #A new band
				trigger = { warband_glory_ok_trigger = yes warband_glory_leading_trigger = yes }
				character_event = { id = warband.3 }
			}
			50 = { #Not leading an army: wait
				trigger = { warband_glory_ok_trigger = yes warband_glory_leading_trigger = no }
				character_event = { id = warband.2 days = 35 random = 5 }
			}
			100 = { #The call has lapsed
				trigger = { warband_glory_lapsed_trigger = yes }
				warband_glory_end_effect = yes
			}
			100 = { #The flow dries up
				trigger = {
					warband_glory_lapsed_trigger = no
					warband_glory_leading_trigger = yes
					check_variable = { which = "call_to_glory_variable" value = 20 }
				}
				character_event = { id = warband.4 }
			}
			5 = { #Rowdy warriors want more plunder
				trigger = {
					warband_glory_ok_trigger = yes
					warband_glory_leading_trigger = yes
					has_earmarked_regiments = call_to_glory_rowdy
					check_variable = { which = "call_to_glory_variable" value = 5 }
				}
				character_event = { id = warband.5 }
			}
			5 = { #The people flock to the capital
				trigger = {
					warband_glory_ok_trigger = yes
					warband_glory_leading_trigger = yes
					check_variable = { which = "call_to_glory_variable" value = 5 }
				}
				character_event = { id = warband.6 }
			}
			2 = { #The Chosen (rare, once in a life)
				trigger = {
					warband_glory_ok_trigger = yes
					warband_glory_leading_trigger = yes
					check_variable = { which = "call_to_glory_variable" value = 10 }
					NOT = { has_character_flag = warband_got_chosen }
				}
				character_event = { id = warband.7 }
			}
		}
	}
}

###############################################################################
# War aid
# Namespace: waraid (was HF.50000 - HF.50107 in hf_warrior_lodge_war_aid_events)
# Faction: Warrior societies
# Serves: the warbands (Chaos, Skaven, Gork and Mork) and the warrior
#   lodges (Warriors of Ulric, the Knightly Orders, the Warriors Guild)
#
# When a war starts, a warband or lodge member on either side may write to
# a player of the same society and ask for help, in the voice of that
# society. If the player turns down the defender and the attacker shares
# the society too, the attacker makes a counter-offer. A player who joins
# becomes the ally's brother in arms (opinion modifier
# waraid_brother_in_arms, which also links the two at the end of the war).
# When the war ends, the ally writes again: glory and society currency for
# a win or a defence held, scorn for a defeat. After a win the ally may
# invite the player to celebrate (a pit fight, a feast, a hunt, a tourney,
# a drinking bout), and that may lead to friendship or love. Rarely, once
# in a life, the ally sends a company of veterans to serve the player.
#
# The call to war              waraid.1 - waraid.3
# The end of the war           waraid.10 - waraid.13
# Celebration and after        waraid.14 - waraid.16
#
# Saved: opinion modifier waraid_brother_in_arms; character flags
#   waraid_asked (5-year cooldown on the player), waraid_spurned_defender,
#   waraid_has_player_ally, waraid_outcome_<won|held|failed|lost>,
#   waraid_got_veterans; earmark waraid_veterans; event targets
#   waraid_defender, waraid_attacker, waraid_player, waraid_ally,
#   waraid_enemy.
#
# Callers: common/on_actions/00_on_actions.txt on_war_started (1),
#   on_war_ended_victory (10), on_war_ended_whitepeace (11),
#   on_war_ended_defeat (12).
# Support: common/scripted_triggers/wh_warband_triggers.txt,
#   common/scripted_effects/wh_warband_effects.txt,
#   common/opinion_modifiers/wh_war_aid_opinions.txt
#
# Fixed from vanilla: the attacker's letters joined the player to the
# defender's side, the attacker could only ask when the defender was a
# member too, the counter-offer could never fire, the letters named event
# targets that were never saved, and the follow-up letter (HF.50106) was
# never sent.
###############################################################################

namespace = waraid

###############################################################################

# waraid.1 A war starts

# player of the same society to ask for help (waraid.2 or waraid.3).
# Scope: ROOT is the defender, FROM the attacker.
character_event = {
	id = waraid.1
	hide_window = yes

# waraid.13 The ally writes after the war
# Won: glory, perhaps a celebration (waraid.14) or, rarely, veterans
# (waraid.16). Held: some glory. Failed or lost: scorn.
# Scope: the player.
letter_event = {
	id = waraid.13
	desc = { trigger = { has_character_flag = waraid_outcome_won } text = EVTDESCwaraid13_won }
	desc = { trigger = { has_character_flag = waraid_outcome_held } text = EVTDESCwaraid13_held }
	desc = {
		trigger = { has_character_flag = waraid_outcome_failed }
		text = EVTDESCwaraid13_failed
	}
	desc = { trigger = { has_character_flag = waraid_outcome_lost } text = EVTDESCwaraid13_lost }
	border = GFX_event_letter_frame_war

	is_triggered_only = yes

	# Victory! Let them tell the tale.
	option = {
		name = EVTOPTAwaraid13
		trigger = { has_character_flag = waraid_outcome_won }
		add_society_currency_major_effect = yes
		prestige = 100
		event_target:waraid_ally = {
			show_scope_change = no
			opinion = { who = ROOT modifier = opinion_grateful years = 5 }
		}
		if = {
			limit = { NOT = { has_character_flag = waraid_got_veterans } is_landed = yes }
			random = {
				chance = 10
				hidden_tooltip = { character_event = { id = waraid.16 days = 20 } }
			}
		}
		random = { chance = 50 hidden_tooltip = { character_event = { id = waraid.14 days = 30 } } }
	}

	# Is that all? I expected a share of the spoils.
	option = {
		name = EVTOPTBwaraid13
		trigger = { has_character_flag = waraid_outcome_won }
		add_society_currency_major_effect = yes
		if = {
			limit = { event_target:waraid_ally = { wealth = 50 } }
			wealth = 50
			event_target:waraid_ally = { show_scope_change = no wealth = -50 }
		}
		event_target:waraid_ally = {
			show_scope_change = no
			opinion = { who = ROOT modifier = opinion_disappointed years = 5 }
		}
	}

	# We held. That is enough.
	option = {
		name = EVTOPTCwaraid13
		trigger = { has_character_flag = waraid_outcome_held }
		add_society_currency_medium_effect = yes
		prestige = 50
		event_target:waraid_ally = {
			show_scope_change = no
			opinion = { who = ROOT modifier = opinion_grateful years = 5 }
		}
		random = { chance = 25 hidden_tooltip = { character_event = { id = waraid.14 days = 30 } } }
	}

	# The nerve of this brazen fool!
	option = {
		name = EVTOPTDwaraid13
		trigger = {
			OR = {
				has_character_flag = waraid_outcome_failed
				has_character_flag = waraid_outcome_lost
			}
		}
		opinion = { who = event_target:waraid_ally modifier = opinion_disappointed years = 5 }
	}

	# What should I have done differently?
	option = {
		name = EVTOPTEwaraid13
		trigger = {
			OR = {
				has_character_flag = waraid_outcome_failed
				has_character_flag = waraid_outcome_lost
			}
		}
		prestige = -25
		random = { chance = 25 change_martial = 1 }
	}

	# There is truth in these harsh words. I must abandon my craven ways.
	option = {
		name = EVTOPTFwaraid13
		trigger = { has_character_flag = waraid_outcome_lost trait = craven }
		remove_trait_craven_effect = yes
	}

	after = {
		clr_character_flag = waraid_outcome_won
		clr_character_flag = waraid_outcome_held
		clr_character_flag = waraid_outcome_failed
		clr_character_flag = waraid_outcome_lost
	}
