# whchapter.1 The tribunal sits
# An Inquisitor-General tries an accused prisoner: burn, force a recantation,
# or acquit.
# Scope: ROOT is the Inquisitor; event_target:tribunal_accused the prisoner.
# Fired by the inquisition_hold_tribunal decision.
character_event = {
	id = whchapter.1
	desc = EVTDESCwhchapter1
	picture = GFX_evt_witch_hunters
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPTAwhchapter1 # Burn
		event_target:tribunal_accused = { show_portrait = yes whunter_burn_at_stake_effect = yes }
		piety = 50
		change_society_currency = 75
		whunter_chapter_relic_effect = yes
		ai_chance = {
			factor = 50
			modifier = { factor = 2 trait = cruel }
			modifier = { factor = 2 trait = zealous }
			modifier = { factor = 0.3 trait = kind }
		}
	}

	option = {
		name = EVTOPTBwhchapter1 # Recant
		event_target:tribunal_accused = {
			if = { limit = { NOT = { religion = ROOT } } religion = ROOT }
			if = { limit = { is_cultist_trigger = yes } leave_society = yes }
			remove_character_modifier = under_suspicion
			remove_character_modifier = highly_suspect
			remove_character_modifier = known_satanist
			opinion = { who = ROOT modifier = opinion_tortured_me years = 10 }
		}
		piety = 20
		change_society_currency = 40
		ai_chance = {
			factor = 40
			modifier = { factor = 2 trait = just }
			modifier = { factor = 2 trait = patient }
		}
	}

	option = {
		name = EVTOPTCwhchapter1 # Acquit
		event_target:tribunal_accused = {
			remove_character_modifier = under_suspicion
			remove_character_modifier = highly_suspect
			opinion = { who = ROOT modifier = opinion_grateful years = 10 }
		}
		piety = -10
		prestige = 20
		ai_chance = {
			factor = 10
			modifier = { factor = 3 trait = kind }
			modifier = { factor = 0 trait = cruel }
		}
	}
}

# whchapter.4 Purged by fire
# A Vindicator of Solkan burns the taint out of a province. If a cultist hid at
# its lord's court, the flames find them.
# Scope: ROOT is the Vindicator; event_target:purge_cultist the cultist, if
# any. Fired by templar_purge_by_fire.
character_event = {
	id = whchapter.4
	desc = { trigger = { event_target:purge_cultist = { always = yes } } text = EVTDESCwhchapter4 }
	desc = {
		trigger = { NOT = { event_target:purge_cultist = { always = yes } } }
		text = EVTDESCwhchapter4b
	}
	picture = GFX_evt_burning_house
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPTAwhchapter4 # Burn the cultist
		trigger = { event_target:purge_cultist = { always = yes } }
		event_target:purge_cultist = { show_portrait = yes whunter_burn_at_stake_effect = yes }
		change_society_currency = 50
		whunter_chapter_relic_effect = yes
		event_target:purge_cultist = {
			host = {
				if = {
					limit = { NOT = { character = ROOT } }
					opinion = { who = ROOT modifier = opinion_purged_my_lands years = 5 }
				}
			}
		}
		ai_chance = { factor = 80 }
	}

	option = {
		name = EVTOPTBwhchapter4 # Leave them to their lord
		trigger = { event_target:purge_cultist = { always = yes } }
		event_target:purge_cultist = {
			show_portrait = yes
			add_character_modifier = { name = highly_suspect years = 5 }
		}
		piety = -10
		ai_chance = { factor = 20 modifier = { factor = 0 trait = wroth } }
	}

	option = {
		name = EVTOPTCwhchapter4 # Only ash
		trigger = { NOT = { event_target:purge_cultist = { always = yes } } }
	}
}

# whchapter.5 A chapter answers the Border Princes
# The chapter a lord of the Border Princes invited sends a hunter. The Templars
# of Solkan answer first; Myrmidian, Estalian and Tilean lords get the
# Inquisition, and a Sigmarite lord sometimes gets the Order of Sigmar.
# Scope: ROOT is the lord. Fired by border_princes_invite_chapter.
character_event = {
	id = whchapter.5
	desc = {
		trigger = { event_target:chapter_envoy = { society_member_of = templars_of_solkan } }
		text = EVTDESCwhchapter5
	}
	desc = {
		trigger = { event_target:chapter_envoy = { society_member_of = holy_inquisition_myrmidia } }
		text = EVTDESCwhchapter5b
	}
	desc = {
		trigger = {
			event_target:chapter_envoy = {
				NOR = {
					society_member_of = templars_of_solkan
					society_member_of = holy_inquisition_myrmidia
				}
			}
		}
		text = EVTDESCwhchapter5c
	}
	picture = GFX_evt_witch_hunters
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	immediate = {
		create_random_priest = {
			random_traits = yes
			dynasty = none
			female = 15
			age = 30
			culture = witchhunter_culture
			religion = sigmarite
		}
		new_character = {
			character_event = { id = aiclean.31 } # racial trait
			remove_education_effect = yes
			add_trait = witch_hunter
			set_character_flag = ai_flag_refuse_concubinage
			set_character_flag = ai_flag_refuse_marriage
			if = {
				limit = { ROOT = { whunter_inquisition_lands_trigger = yes } }
				whunter_local_chapter_origin_effect = yes
				whunter_join_local_chapter_effect = yes
			}
			else = {
				random_list = {
					70 = { culture = ROOT religion = solkanite join_society = templars_of_solkan }
					30 = {
						trigger = { ROOT = { religion = sigmarite } }
						join_society = sigmar_witch_hunters
					}
				}
			}
			if = {
				limit = { ROOT = { whunter_court_hunter_seat_empty_trigger = yes } }
				whunter_give_court_title_effect = yes
			}
			save_event_target_as = chapter_envoy
		}
	}

	option = {
		name = EVTOPTAwhchapter5
		show_portrait = event_target:chapter_envoy
		piety = 25
	}
}

# whchapter.10 A question of jurisdiction
# A hunter of another chapter at the same court claims the right to a case.
# Defer, stand firm, or put it to the lord.
# Scope: ROOT is the hunter; event_target:rival_hunter the other.
# Fired by on_five_year_pulse (random list whchapter_rivalry).
character_event = {
	id = whchapter.10
	desc = EVTDESCwhchapter10
	picture = GFX_evt_witch_hunters
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	trigger = {
		whunter_chapter_member_trigger = yes
		society_rank >= 2
		prisoner = no
		is_inaccessible_trigger = no
		liege = {
			OR = {
				whunter_rival_chapter_trigger = yes
				any_courtier = { whunter_rival_chapter_trigger = yes }
			}
		}
	}

	immediate = {
		liege = {
			if = {
				limit = { whunter_rival_chapter_trigger = yes }
				save_event_target_as = rival_hunter
			}
			else = {
				random_courtier = {
					limit = { whunter_rival_chapter_trigger = yes }
					save_event_target_as = rival_hunter
				}
			}
		}
	}

	option = {
		name = EVTOPTAwhchapter10 # Defer
		event_target:rival_hunter = {
			show_portrait = yes
			opinion = { who = ROOT modifier = opinion_chapter_deferred }
		}
		change_society_currency = -50
		prestige = -10
		ai_chance = {
			factor = 30
			modifier = { factor = 2 trait = humble }
			modifier = { factor = 0.3 trait = proud }
		}
	}

	option = {
		name = EVTOPTBwhchapter10 # Stand firm
		event_target:rival_hunter = {
			show_portrait = yes
			opinion = { who = ROOT modifier = opinion_chapter_jurisdiction }
		}
		change_society_currency = 50
		ai_chance = {
			factor = 50
			modifier = { factor = 2 trait = wroth }
			modifier = { factor = 2 trait = proud }
		}
	}

	option = {
		name = EVTOPTCwhchapter10 # Put it to the lord
		trigger = { liege = { NOT = { character = event_target:rival_hunter } } }
		random_list = {
			50 = {
				modifier = { factor = 1.5 liege = { opinion = { who = ROOT value = 25 } } }
				custom_tooltip = { text = whchapter10_lord_sides_with_you_tt }
				change_society_currency = 100
				event_target:rival_hunter = {
					opinion = { who = ROOT modifier = opinion_chapter_jurisdiction }
				}
			}
			50 = {
				custom_tooltip = { text = whchapter10_lord_sides_with_rival_tt }
				change_society_currency = -100
				prestige = -25
			}
		}
		ai_chance = { factor = 20 }
	}
}
