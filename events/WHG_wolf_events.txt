# wolf.1 A child is born
# The child of a carrier may inherit the blood (wolf.3 in 150 days); the
# first child of a wolf people may, rarely, show it from the cradle (wolf.2
# to the parents). Was HL.1000 and HL.1700.
# Scope: ROOT is the newborn; hidden.
character_event = {
	id = wolf.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_dlc = "Horse Lords"
	}

	immediate = {
		if = {
			limit = {
				OR = {
					father_even_if_dead = {
						has_character_modifier = wolf_blood
						is_father_real_father = yes
					}
					mother_even_if_dead = { has_character_modifier = wolf_blood }
				}
			}
			character_event = { id = wolf.3 days = 150 }
		}
		else_if = {
			limit = {
				wolf_blood_people_trigger = yes
				NOT = { wolf_blood_in_family_trigger = yes }
				OR = {
					father = { NOT = { num_of_children = 2 } }
					mother = { NOT = { num_of_children = 2 } }
				}
			}
			random = {
				chance = 5
				father = { narrative_event = { id = wolf.2 days = 300 } }
				mother = { narrative_event = { id = wolf.2 days = 300 } }
			}
		}
	}
}

# wolf.3 The blood runs true
# One child in eleven of a carrier inherits the blood, more for the
# steppe-born, the first generation's grandchildren, children of two carriers
# and families of a player. The parents hear of it.
# Scope: ROOT is the child; hidden.
character_event = {
	id = wolf.3
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		random_list = {
			10 = { # Inherits the wolf's blood
				modifier = { factor = 1.5 culture_group = hung_group }
				modifier = {
					factor = 2
					OR = {
						father_even_if_dead = { ai = no }
						mother_even_if_dead = { ai = no }
						father_even_if_dead = { father_even_if_dead = { ai = no } }
						father_even_if_dead = { mother_even_if_dead = { ai = no } }
						mother_even_if_dead = { father_even_if_dead = { ai = no } }
						mother_even_if_dead = { mother_even_if_dead = { ai = no } }
					}
				}
				modifier = {
					factor = 2
					OR = {
						father_even_if_dead = { has_character_flag = wolf_blood_first_generation }
						mother_even_if_dead = { has_character_flag = wolf_blood_first_generation }
					}
				}
				modifier = {
					factor = 2
					father_even_if_dead = { has_character_modifier = wolf_blood }
					mother_even_if_dead = { has_character_modifier = wolf_blood }
				}
				modifier = {
					factor = 1.5
					OR = {
						father_even_if_dead = { father_even_if_dead = { has_character_flag = wolf_blood_first_generation } }
						father_even_if_dead = { mother_even_if_dead = { has_character_flag = wolf_blood_first_generation } }
						mother_even_if_dead = { father_even_if_dead = { has_character_flag = wolf_blood_first_generation } }
						mother_even_if_dead = { mother_even_if_dead = { has_character_flag = wolf_blood_first_generation } }
					}
				}
				wolf_blood_gain_effect = yes
				set_character_flag = wolf_blood_inherited
				father = { narrative_event = { id = wolf.4 } }
				mother = { narrative_event = { id = wolf.4 } }
			}
			100 = { } # The blood sleeps
		}
	}
}

		name = EVTOPTBwolf30

			mother_even_if_dead = { always = yes }

	}
