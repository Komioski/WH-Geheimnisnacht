# wonder.1 A stage is finished
# Hidden. A wonder that finished a stage takes its first builder's
# description, unless it is a historical wonder, and gets a new name unless
# the player renamed it. The name may draw on its finest upgrade.
# Scope: ROOT is the owner, FROM the wonder, FROMFROM its province.
character_event = {
	id = wonder.1
	hide_window = yes

	is_triggered_only = yes # on_wonder_stage_finish

	immediate = {
		FROM = {
			save_event_target_as = target_wonder
			original_wonder_owner = { save_event_target_as = wonder_owner_original }
		}
		FROMFROM = { save_event_target_as = target_province }
		event_target:target_wonder = {
			any_upgrade_on_wonder = { # vanilla way: the score picks the upgrade saved
				limit = { is_active = yes NOT = { has_wonder_upgrade_flag = negative } }
				score_value = { # A fine upgrade first
					value = 10
					fancy_upgrade_selection_score = yes
				}
				save_event_target_as = upgrade_00
			}
		}
		wonder_stage_completion_sound_effect = yes
		event_target:target_wonder = {
			if = { # A hold or city with a lore name keeps it
				limit = { has_wonder_flag = wonder_lore_named }
				event_target:target_province = { wonder_naming_and_describing_effect = yes }
			}
			else_if = {
				limit = {
					NOR = {
						has_wonder_flag = wonder_desc_1
						has_wonder_flag = wonder_desc_2
						has_wonder_flag = wonder_desc_3
						has_wonder_flag = wonder_desc_4
						has_wonder_flag = wonder_desc_5
						has_wonder_flag = wonder_desc_6
						has_wonder_flag = wonder_desc_7
						has_wonder_flag = is_historical
					}
				}
				set_wonder_flag = wonder_desc_1
			}
			if = {
				limit = { NOT = { has_wonder_flag = wonder_lore_named } }
				set_description = wonder_desc_text_update
			}
			if = {
				limit = {
					has_custom_name = no
					NOT = { has_wonder_flag = is_historical }
					NOT = { has_wonder_flag = wonder_lore_named }
				}
				wonder_rename_effect = yes
			}
		}
	}
}

# wonder.2 An upgrade is finished
# Hidden. Once a wonder has kept a name or a description long enough (15 to
# 85 years by step), a new upgrade gives it the next one. The new name or
# text may name this upgrade and one other.
# Scope: ROOT is the owner, FROM the upgrade, FROMFROM the wonder,
# FROMFROMFROM its province.
character_event = {
	id = wonder.2
	hide_window = yes

	is_triggered_only = yes # on_wonder_upgrade_finish

	immediate = {
		FROM = {
			save_event_target_as = upgrade_00
			original_wonder_owner = { save_event_target_as = wonder_owner_original }
		}
		FROMFROM = { save_event_target_as = target_wonder }
		FROMFROMFROM = { save_event_target_as = target_province }
		event_target:target_wonder = {
			any_upgrade_on_wonder = { # vanilla way: the score picks the upgrade saved
				limit = {
					is_active = yes
					NOT = { wonder_upgrade = event_target:upgrade_00 }
					NOT = { has_wonder_upgrade_flag = negative }
				}
				score_value = { value = 10 fancy_upgrade_selection_score = yes }
				save_event_target_as = upgrade_01
			}
			if = {
				limit = {
					has_custom_name = no
					NOT = { has_wonder_flag = is_historical }
					NOT = { has_wonder_flag = wonder_lore_named }
					OR = {
						had_flag = { flag = wonder_name_1 years = 55 }
						had_flag = { flag = wonder_name_2 years = 65 }
						had_flag = { flag = wonder_name_3 years = 75 }
						had_flag = { flag = wonder_name_4 years = 85 }
						had_flag = { flag = wonder_name_5 years = 75 }
						had_flag = { flag = wonder_name_6 years = 75 }
					}
				}
				wonder_rename_effect = yes
			}
			if = {
				limit = {
					NOT = { has_wonder_flag = wonder_lore_named }
					OR = {
						had_flag = { flag = wonder_desc_1 years = 15 }
						had_flag = { flag = wonder_desc_2 years = 35 }
						had_flag = { flag = wonder_desc_3 years = 35 }
						had_flag = { flag = wonder_desc_4 years = 35 }
						had_flag = { flag = wonder_desc_5 years = 35 }
						had_flag = { flag = wonder_desc_6 years = 35 }
					}
				}
				wonder_redescribe_effect = yes
			}
			if = { # A hold or city with a lore name keeps it
				limit = { has_wonder_flag = wonder_lore_named }
				event_target:target_province = { wonder_naming_and_describing_effect = yes }
			}
		}
	}
}

# wonder.3 A stage is looted
# Hidden. A looted wonder becomes "Ruins of ..." if it had reached its third
# stage and was never renamed, or a "Destroyed ..." otherwise, with a ruined
# description. The stage flags come from wonder.200 when looting starts.
# Fixed: the second description sat in an else with a limit (never valid),
# and the stage flags were cleared on the owner instead of the wonder, so a
# restored and looted-again wonder kept both.
# Scope: ROOT is the owner, FROM the wonder, FROMFROM its province.
character_event = {
	id = wonder.3
	hide_window = yes

	is_triggered_only = yes # on_wonder_stage_loot_finish

	immediate = {
		FROM = {
			save_event_target_as = target_wonder
			original_wonder_owner = { save_event_target_as = wonder_owner_original }
		}
		FROMFROM = { save_event_target_as = target_province }
		event_target:target_wonder = {
			if = {
				limit = { has_wonder_flag = wonder_stage_3 }
				if = {
					limit = { NOT = { has_wonder_flag = was_renamed } }
					set_name = ruins_of_wonder_name
				}
				set_description = ruined_wonder_desc_1
			}
			else_if = {
				limit = { has_wonder_flag = wonder_stage_less_than_3 }
				set_name = destroyed_wonder_type
				set_description = ruined_wonder_desc_2
			}
			set_wonder_flag = was_renamed
			clr_wonder_flag = wonder_stage_3
			clr_wonder_flag = wonder_stage_less_than_3
		}
	}
}

# wonder.5 Lore names at game start
# Hidden. Once per game, every wonder in a province with a hand-written
# entry takes its lore name and description: this is how the one Great
# Dwarf hold wonder (wonder_dawi_hold) becomes Everpeak, Zhufbar, Barak
# Varr and the rest, and how Altdorf, Nuln and Talabheim get their names.
# Scope: a player.
character_event = {
	id = wonder.5
	hide_window = yes

	is_triggered_only = yes # on_startup
	ai = no

	trigger = {
		NOT = { has_global_flag = wonder_lore_names_set }
	}

	immediate = {
		set_global_flag = wonder_lore_names_set
		any_province = {
			limit = { has_wonder = yes }
			wonder_naming_and_describing_effect = yes
		}
	}
}

# wonder.50 A town springs up
# Fifty years after it was begun and with two upgrades, a wonder draws
# traders and pilgrims who settle around it. Let them stay (a new holding
# slot) or drive them off and keep what they carried. Once per province.
# The AI no longer needs 50 gold to welcome them (the town costs nothing).
# Scope: ROOT is the owner, FROM the upgrade, FROMFROM the wonder,
# FROMFROMFROM its province.
character_event = {
	id = wonder.50
	title = EVTNAMEwonder50
	desc = EVTDESCwonder50
	picture = GFX_evt_construction_site
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes # on_wonder_upgrade_finish, delayed list
	only_playable = yes
	war = no

	trigger = {
		is_inaccessible_trigger = no
		FROMFROM = { wonder_age = 50 num_of_wonder_upgrades = 2 }
		FROMFROMFROM = {
			NOR = { has_province_flag = had_event_LT_60001 num_of_max_settlements = 7 }
		}
	}

	immediate = {
		FROMFROM = { save_event_target_as = target_wonder }
		FROMFROMFROM = {
			set_province_flag = had_event_LT_60001
			save_event_target_as = target_province
		}
	}

	option = { # They may stay
		name = EVTOPTAwonder50
		event_target:target_province = { add_holding_slot = 1 }
		ai_chance = { factor = 70 }
	}
	option = { # Drive them off
		name = EVTOPTBwonder50
		wealth = 40
		ai_chance = {
			factor = 30
			modifier = { factor = 1.5 trait = greedy }
			modifier = { factor = 1.5 trait = cruel }
		}
	}
}

# wonder.54 The fountain of youth
# A ruler drinks from the fountain of the Magnificent Garden (decision,
# once in ten years, shared with the free drink at Ganaina, amazon.206) and
# grows two years younger, three for one who found the Pools of Lustria
# (lustria.93) and knows how to drink. A lunatic leaps in and swallows the
# water the wrong way, and is two years older for it.
# Scope: ROOT is the ruler.
character_event = {
	id = wonder.54
	title = EVTNAMEwonder54
	desc = { trigger = { is_lunatic_trigger = no NOT = { has_character_flag = found_the_pools } } text = EVTDESCwonder54 }
	desc = { trigger = { is_lunatic_trigger = no has_character_flag = found_the_pools } text = EVTDESCwonder54c }

	border = GFX_event_normal_frame_religion

	is_triggered_only = yes # decision in wonder_decisions.txt

	immediate = {
		random_realm_wonder = {
			limit = { has_wonder = wonder_garden }
			save_event_target_as = target_wonder
		}
	}

	option = { # Drink
		name = EVTOPTAwonder54
		trigger = { is_lunatic_trigger = no }
		if = {
			limit = { has_character_flag = found_the_pools }
			add_age = -3
		}
		else = { add_age = -2 }
	}
	option = { # Leap in
		name = EVTOPTBwonder54
		trigger = { is_lunatic_trigger = yes }
		add_age = 2
	}

# wonder.56 An evening at the tavern
# A stressed ruler with a tavern by one of the realm's wonders is asked out
# by a friend or a favourite of the court. Drinking the night away lifts the
# stress (and may start a habit, wonder.57); talking instead warms the
# friendship.
# Scope: ROOT is the ruler, event_target:drinking_buddy the companion.
character_event = {
	id = wonder.56
	title = EVTNAMEwonder56
	desc = EVTDESCwonder56
	picture = GFX_evt_feast
	border = GFX_event_normal_frame_diplomacy
	portrait = event_target:drinking_buddy

	is_triggered_only = yes # on_yearly_pulse random_events (WHG_pulse_on_actions)
	only_rulers = yes
	trait = stressed

	trigger = {
		is_inaccessible_trigger = no
		any_realm_wonder = { has_wonder_upgrade = upgrade_tavern }
		OR = { any_friend = { is_landed = yes } any_courtier_or_vassal = { always = yes } }
	}

	immediate = {
		save_drinking_character_effect = yes
		random_realm_wonder = {
			limit = { has_wonder_upgrade = upgrade_tavern }
			save_event_target_as = target_wonder
		}
	}

	option = { # Drink
		name = EVTOPTAwonder56
		remove_trait = stressed
		if = {
			limit = { NOT = { trait = drunkard } }
			custom_tooltip = {
				text = EVTTOOLTIPwonder56
				random = { chance = 5 character_event = { id = wonder.57 days = 15 random = 30 } }
			}
		}
	}
	option = { # Talk instead
		name = EVTOPTBwonder56
		event_target:drinking_buddy = {
			if = {
				limit = { NOT = { is_friend = ROOT } }
				opinion = { modifier = opinion_budding_friendship who = ROOT }
			}
			else = { opinion = { modifier = kind_pos who = ROOT } }
		}
	}
}

# wonder.58 The traps of the old owner
# Two weeks after a wonder with traps changes hands, two courtiers of the
# new owner wander into its depths and die there (the more beloved first).
# The old owner, if alive, hears of it (wonder.59). Lizardmen know the
# traps of the old temples and are spared; a Lizardman owner gets no event.
# Fixed: the old check looked for a Lizardman culture on the wonder itself.
# Scope: ROOT is the new owner, FROM the wonder, FROMFROM the old owner.
character_event = {
	id = wonder.58
	title = EVTNAMEwonder58
	desc = EVTDESCwonder58
	picture = GFX_evt_died_funeral
	border = GFX_event_normal_frame_intrigue

	is_triggered_only = yes # on_wonder_owner_change, delayed list

	trigger = {
		NOT = { culture_group = lizardman_group }
		FROM = { has_wonder_upgrade = upgrade_traps }
		any_courtier = { count = 2 wonder_trap_victim_trigger = yes }
	}

	immediate = {
		FROM = { save_event_target_as = target_wonder }
		FROMFROM = { save_event_target_as = old_owner }
		random_courtier = {
			limit = { wonder_trap_victim_trigger = yes }
			score_value = { value = 2 beloved_target_selection_score = yes }
			save_event_target_as = target_courtier
		}
		random_courtier = {
			limit = {
				wonder_trap_victim_trigger = yes
				NOT = { character = event_target:target_courtier }
			}
			save_event_target_as = target_courtier_2
		}
	}

	option = { # Curse them
		name = EVTOPTAwonder58
		event_target:target_courtier = { death = { death_reason = death_accident_wonder_traps } }
		event_target:target_courtier_2 = { death = { death_reason = death_accident_wonder_traps } }
		if = {
			limit = { event_target:old_owner = { is_alive = yes } }
			event_target:old_owner = { character_event = { id = wonder.59 } }
		}
	}
}
