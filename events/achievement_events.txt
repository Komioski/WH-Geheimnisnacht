character_event = {
	id = chronicled_achievements.1
	hide_window = yes
	is_triggered_only = yes
	ai = no
	trigger = {
		FROM = { e_the_empire }
		NOT = {
			has_global_flag = achievement_united_the_empire
		}
	}
	immediate = {
		set_global_flag = achievement_united_the_empire
		chronicle = {
			entry = CHRONICLE_ACHIEVEMENT_UNITED_THE_EMPIRE
			portrait = [Root.GetID]
			picture = GFX_evt_throne_room
			any_player = {
				limit = {
					NOT = {
						character = ROOT
					}
				}
				narrative_event = {
					id = chronicled_achievements.2
					days = 3
				}
			}
		}
	}
}

# Restored from vanilla 3.3.5.1 (these events are called by the mod but were missing)
# Casanova
character_event = {
	id = 718
	hide_window = yes
	
	is_triggered_only = yes
	
	ai = no
	
	trigger = {
		NOT = { has_character_flag = achievement_casanova }
	}
	
	immediate = {
		if = {
			limit = {
				NOT = { check_variable = { which = "casanova_check" value = 9.5 } }
			}
			change_variable = { which = "casanova_check" value = 1 }
		}
		if = {
			limit = {
				check_variable = { which = "casanova_check" value = 9.5 }
			}
			set_character_flag = achievement_casanova
		}
	}
}
