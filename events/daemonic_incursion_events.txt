# Built on the vanilla Jade Dragon invasion tombola (JD.10100). While the
# winds of magic blow strong, the Chaos realms of the warp can tear open a
# rift inside a mortal realm and pour daemons into realspace.
#
# daemonic_incursion.1 runs from on_offmap_yearly_pulse (ROOT = the realm's
# governor, FROM = the offmap power). The war uses the daemonic_incursion CB,
# whose end blocks banish the host and move the doom counter.

namespace = daemonic_incursion

# Incursion tombola (hidden)
character_event = {
	id = daemonic_incursion.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		FROM = {
			OR = {
				is_offmap_tag = khorne_warp
				is_offmap_tag = nurgle_warp
				is_offmap_tag = slaanesh_warp
				is_offmap_tag = tzeentch_warp
				is_offmap_tag = undivided_warp
			}
		}
		war = no
		NOT = { has_character_flag = daemonic_incursion_leader }
		NOT = {
			has_game_rule = {
				name = daemonic_incursions
				value = none
			}
		}
		OR = {
			is_doom_medium_trigger = yes
			is_doom_high_trigger = yes
			is_doom_very_high_trigger = yes
		}
	}

	immediate = {
		random_list = {
			10 = { # Incursion
				modifier = {
					factor = 2
					is_doom_high_trigger = yes
				}
				modifier = {
					factor = 4
					is_doom_very_high_trigger = yes
				}
				modifier = {
					factor = 2
					FROM = { has_policy = warp_focused_mortal }
				}
				modifier = {
					factor = 3
					FROM = { has_policy = warp_endtimes }
				}
				modifier = {
					factor = 0.25
					FROM = { has_policy = warp_focused_warp }
				}
				modifier = {
					factor = 1.5
					FROM = { has_status = warp_ascendant }
				}
				character_event = { id = daemonic_incursion.2 }
			}
			90 = { # The veil holds this year
			}
		}
	}
}

# A rift opens (hidden): pick a victim, open the rift, declare war
character_event = {
	id = daemonic_incursion.2
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		random_independent_ruler = {
			limit = {
				is_playable = yes
				num_of_count_titles_in_realm = 5
				NOR = {
					religion_group = chaos_gods_group
					is_offmap_governor = khorne_warp
					is_offmap_governor = nurgle_warp
					is_offmap_governor = slaanesh_warp
					is_offmap_governor = tzeentch_warp
					is_offmap_governor = undivided_warp
					has_character_modifier = daemonic_incursion_timer
				}
				any_realm_province = { is_land = yes }
			}
			save_event_target_as = incursion_target
		}
		if = {
			limit = { NOT = { event_target:incursion_target = { always = yes } } }
			break = yes
		}

		event_target:incursion_target = {
			add_character_modifier = {
				name = daemonic_incursion_timer
				years = 25
				hidden = yes
			}
			random_realm_province = {
				limit = { is_land = yes }
				save_event_target_as = incursion_rift
			}
		}

		set_character_flag = daemonic_incursion_leader
		save_event_target_as = incursion_leader

		# The host grows with the winds of magic
		if = {
			limit = { is_doom_very_high_trigger = yes }
			spawn_unit = {
				province = event_target:incursion_rift
				owner = ROOT
				earmark = daemonic_incursion_host
				cannot_inherit = yes
				attrition = 0
				reinforces = no
				troops = {
					daemonic_host = { 6000 6000 }
					daemonic_host = { 4000 4000 }
					daemonic_host = { 4000 4000 }
				}
			}
		}
		else_if = {
			limit = { is_doom_high_trigger = yes }
			spawn_unit = {
				province = event_target:incursion_rift
				owner = ROOT
				earmark = daemonic_incursion_host
				cannot_inherit = yes
				attrition = 0
				reinforces = no
				troops = {
					daemonic_host = { 4500 4500 }
					daemonic_host = { 3000 3000 }
					daemonic_host = { 3000 3000 }
				}
			}
		}
		else = {
			spawn_unit = {
				province = event_target:incursion_rift
				owner = ROOT
				earmark = daemonic_incursion_host
				cannot_inherit = yes
				attrition = 0
				reinforces = no
				troops = {
					daemonic_host = { 2500 2500 }
					daemonic_host = { 1500 1500 }
					daemonic_host = { 1500 1500 }
				}
			}
		}

		war = {
			target = event_target:incursion_target
			casus_belli = daemonic_incursion
		}
		add_doom_progress_minor_effect = yes

		event_target:incursion_target = { character_event = { id = daemonic_incursion.3 } }
		any_player = {
			limit = {
				NOT = { character = event_target:incursion_target }
				is_within_diplo_range = event_target:incursion_target
			}
			narrative_event = { id = daemonic_incursion.4 days = 1 }
		}
	}
}
