###############################################################################
# Brayherd events: the pilgrimage to the Heart of the Dark and the boons of
# the Brayherd offmap power
# Namespace: brayherd (was "JD_BRAY" before the 2026 events refactor)
# Faction: Beastmen
#
# The pilgrimage is built like the Jade Dragon kow-tow journey: a few random
# events on the road, a few at the Heart of the Dark, then the rite at the
# Great Herdstone, which pays out Brayherd favour.
#
# Setting out                   brayherd.1 - brayherd.4
# On the road                   brayherd.10 - brayherd.24
#   Rain and memories           brayherd.10
#   Letters home                brayherd.11 - brayherd.12
#   Travelling trader           brayherd.13
#   A warlord's hospitality     brayherd.15 - brayherd.18
#   The Jabberslythe hunt       brayherd.20 - brayherd.24
# At the Heart of the Dark      brayherd.30 - brayherd.33
# The rite at the Herdstone     brayherd.39
# Boons from the Brayherd       brayherd.50 - brayherd.56
# An exceptional bray           brayherd.60 - brayherd.64
#
# Fired from the Brayherd offmap decisions in decisions/wh_chaos_decisions.txt.
# Scripted triggers: common/scripted_triggers/wh_brayherd_triggers.txt
# Scripted effects: common/scripted_effects/wh_brayherd_effects.txt
# Localisation: localisation/00_brayherd_events.csv
# Full list with callers: docs/events/event_database.csv
#
# Journey state lives on the pilgrim: flag brayherd_on_pilgrimage while
# travelling, variables brayherd_road_events and brayherd_heart_events (events
# left in each stage) and brayherd_herdstone_favour (how well the rite goes).
# The vanilla flag went_on_kow_tow_pilgrimage is shared on purpose, so nobody
# can be on the Cathay and the Brayherd pilgrimage at once.
# brayherd_pilgrimage_end_effect clears it all and sets brayherd_pilgrimage_done.
###############################################################################

namespace = brayherd

# brayherd.1 The journey begins
# The pilgrim says farewell and sets off for the Heart of the Dark.
# Scope: ROOT is the pilgrim.
character_event = {
	id = brayherd.1
	title = brayherd_pilgrimage_title
	desc = EVTDESCbrayherd1
	picture = GFX_evt_beastman_chief
	border = GFX_event_normal_frame_diplomacy
	is_triggered_only = yes # decision brayherd_pilgrimage

	immediate = {
		set_character_flag = brayherd_on_pilgrimage
	}

		name = EVTOPTAbrayherd1
		add_trait = travelling
		hidden_effect = { character_event = { id = brayherd.2 } }
	}

# brayherd.2 Planning the route
# Sets how many road events the journey has: two from near the Drakwald,
# three from further away, four from the far corners of the world. Distance is
# measured from Marienburg (province 222), at the edge of the forest.
# Scope: ROOT is the pilgrim.
character_event = {
	id = brayherd.2
	hide_window = yes
	is_triggered_only = yes # brayherd.1

	has_character_flag = brayherd_on_pilgrimage

	immediate = {
		set_variable = { which = brayherd_herdstone_favour value = 0 }
		set_variable = { which = brayherd_road_events value = 4 }
		capital_scope = {
			if = {
				limit = { distance = { where = 222 value < 500 } }
				ROOT = { set_variable = { which = brayherd_road_events value = 2 } }
			}
			else_if = {
				limit = { distance = { where = 222 value < 900 } }
				ROOT = { set_variable = { which = brayherd_road_events value = 3 } }
			}
		}
		character_event = { id = brayherd.3 days = 90 random = 30 }
	}
}

# brayherd.3 Road event roll
# Picks one road event the pilgrim has not had yet. When the road events are
# used up, or none fits, the pilgrim reaches the Heart of the Dark (brayherd.4).
# Scope: ROOT is the pilgrim.
character_event = {
	id = brayherd.3
	hide_window = yes
	is_triggered_only = yes # brayherd.2, brayherd_return_to_road_effect

	has_character_flag = brayherd_on_pilgrimage

	immediate = {
		if = {
			limit = { NOT = { check_variable = { which = brayherd_road_events value = 1 } } }
			character_event = { id = brayherd.4 days = 50 random = 30 }
		}
		else = {
			random_list = {
				50 = { # Rain and memories of the dead
					trigger = {
						NOT = { has_character_flag = brayherd_had_rain }
						OR = {
							father_even_if_dead = { brayherd_mourned_relative_trigger = yes }
							mother_even_if_dead = { brayherd_mourned_relative_trigger = yes }
							any_sibling_even_if_dead = { brayherd_mourned_relative_trigger = yes }
						}
					}
					set_character_flag = brayherd_road_rolled
					character_event = { id = brayherd.10 }
				}
				50 = { # Jabberslythe hunt
					trigger = { NOT = { has_character_flag = brayherd_hunted } }
					set_character_flag = brayherd_road_rolled
					character_event = { id = brayherd.20 }
				}
				50 = { # Homesick
					trigger = {
						NOT = { has_character_flag = brayherd_wrote_home }
						any_courtier_or_vassal = { brayherd_missed_one_trigger = yes }
					}
					set_character_flag = brayherd_road_rolled
					character_event = { id = brayherd.11 }
				}
				50 = { # Travelling trader
					trigger = {
						NOT = { has_character_flag = brayherd_met_merchant }
						NOT = { any_artifact = { count = 6 has_artifact_flag = beastman } }
					}
					set_character_flag = brayherd_road_rolled
					character_event = { id = brayherd.13 }
				}
				50 = { # A warlord's invitation, only on longer journeys
					trigger = {
						NOT = { has_character_flag = brayherd_was_hosted }
						capital_scope = { distance = { where = 222 value >= 500 } }
						any_playable_ruler = { brayherd_host_trigger = yes }
					}
					set_character_flag = brayherd_road_rolled
					character_event = { id = brayherd.15 }
				}
			}
			if = {
				limit = { NOT = { has_character_flag = brayherd_road_rolled } }
				character_event = { id = brayherd.4 days = 150 random = 60 }
			}
			clr_character_flag = brayherd_road_rolled
		}
	}
}

# brayherd.12 A letter from the road
# The person at home reads the pilgrim's letter and warms to them: love
# letters for a lover or spouse, a delighted parent, or a pen pal.
# Scope: ROOT is the person at home, FROM the pilgrim.
character_event = {
	id = brayherd.12
	title = brayherd_pilgrimage_title
	desc = { trigger = { NOT = { is_child_of = FROM } } text = EVTDESCbrayherd12 }
	desc = { trigger = { is_child_of = FROM } text = EVTDESCbrayherd12b }
	picture = GFX_evt_recieve_letter
	border = GFX_event_normal_frame_diplomacy
	is_triggered_only = yes # brayherd.11

	option = { # Read it
		name = { text = EVTOPTAbrayherd12 trigger = { NOT = { is_child_of = FROM } } }
		name = { text = EVTOPTAbrayherd12b trigger = { is_child_of = FROM } }
		if = {
			limit = { OR = { is_lover = FROM any_spouse = { character = FROM } } }
			opinion = { who = FROM modifier = opinion_love_letters years = 15 }
		}
		else_if = {
			limit = { OR = { is_mother = FROM is_father = FROM } }
			opinion = { who = FROM modifier = opinion_delighted_parent years = 15 }
		}
		else = { opinion = { who = FROM modifier = opinion_pen_pal years = 15 } }
	}
}

# brayherd.13 The travelling trader
# A shifty trader offers a beastman artifact the pilgrim does not own yet. Buy
# it, hire the trader (kind pilgrims), rob them (deceitful ones) or move on.
# Scope: ROOT is the pilgrim.
character_event = {
	id = brayherd.13
	title = brayherd_pilgrimage_title
	desc = EVTDESCbrayherd13
	picture = GFX_evt_voodoo_shaman
	border = GFX_event_normal_frame_diplomacy
	portrait = event_target:brayherd_trader
	is_triggered_only = yes # brayherd.3

	has_character_flag = brayherd_on_pilgrimage

	immediate = {
		set_character_flag = brayherd_met_merchant
		hidden_effect = {
			brayherd_warp = {
				governor = {
					create_random_intriguer = {
						age = 25
						random_traits = yes
						female = 50
						dynasty = none
						attributes = { intrigue = 4 }
					}
					new_character = {
						character_event = { id = aiclean.31 } # Racial trait
						upgrade_education_effect = yes
						save_event_target_as = brayherd_trader
						brayherd_trader_stock_effect = yes
					}
				}
			}
		}
	}

	option = { # Buy it
		name = EVTOPTAbrayherd13
		custom_tooltip = { text = EVTOPTAbrayherd13_tt }
		scaled_wealth = { value = -0.4 min = -150 }

			event_target:brayherd_trader = {
				event_target:brayherd_trader_artifact = {
					transfer_artifact = { from = PREV to = ROOT }
				}
				brayherd_trader_leaves_effect = yes
			}
		}
	}

	option = { # Hire the trader
		name = EVTOPTBbrayherd13
		trigger = { trait = kind }

				event_target:brayherd_trader_artifact = {
					transfer_artifact = { from = PREV to = ROOT }
				}
			}
		}

			show_scope_change = no
			if = { limit = { is_invincible_trigger = yes } antideath_effect = yes }

		}
	}

	option = { # Move on
		name = EVTOPTDbrayherd13
		custom_tooltip = { text = EVTOPTDbrayherd13_tt }
		hidden_effect = { event_target:brayherd_trader = { brayherd_trader_leaves_effect = yes } }
	}

		brayherd_return_to_road_effect = yes
	}
}

# brayherd.39 Before the Great Herdstone
# Ends the pilgrimage. The rite's outcome is rolled from the favour gathered
# on the journey plus half the pilgrim's diplomacy, rank and the Brayherd's
# opinion: a bad rite (favour 100 for those who humble themselves, at a cost
# in prestige), an okay one (favour 150) or a great one (favour 300, piety,
# martial and health).
# Scope: ROOT is the pilgrim.
character_event = {
	id = brayherd.39
	title = brayherd_pilgrimage_title
	desc = EVTDESCbrayherd39
	picture = GFX_evt_beastmen_herdstone
	border = GFX_event_normal_frame_diplomacy
	is_triggered_only = yes # brayherd.30

	has_character_flag = brayherd_on_pilgrimage

	immediate = {
		export_to_variable = { which = brayherd_temp value = diplomacy }
		divide_variable = { which = brayherd_temp value = 2 }
		change_variable = { which = brayherd_herdstone_favour which = brayherd_temp }
		set_variable = { which = brayherd_temp value = 0 }
		if = {
			limit = { real_tier = KING }
			change_variable = { which = brayherd_herdstone_favour value = 5 }
		}
		else_if = {
			limit = { real_tier = EMPEROR }
			change_variable = { which = brayherd_herdstone_favour value = 7 }
		}
		if = {
			limit = { disliked_by_offmap = { type = brayherd_warp } }
			subtract_variable = { which = brayherd_herdstone_favour value = 5 }
		}
		else_if = {
			limit = { liked_by_offmap = { type = brayherd_warp } }
			change_variable = { which = brayherd_herdstone_favour value = 5 }
		}
		random_list = {
			45 = { # Bad rite
				trigger = {
					NOT = { check_variable = { which = brayherd_herdstone_favour value = 23 } }
				}
				modifier = {
					factor = 0.85
					check_variable = { which = brayherd_herdstone_favour value = 12 }
				}
				modifier = {
					factor = 0.85
					check_variable = { which = brayherd_herdstone_favour value = 13 }
				}
				modifier = {
					factor = 0.85
					check_variable = { which = brayherd_herdstone_favour value = 14 }
				}
				modifier = {
					factor = 0.85
					check_variable = { which = brayherd_herdstone_favour value = 15 }
				}
				modifier = {
					factor = 0.85
					check_variable = { which = brayherd_herdstone_favour value = 17 }
				}
				modifier = {
					factor = 0.85
					check_variable = { which = brayherd_herdstone_favour value = 19 }
				}
				modifier = {
					factor = 0.85
					check_variable = { which = brayherd_herdstone_favour value = 21 }
				}
				set_character_flag = brayherd_rite_bad
			}
			45 = { # Okay rite
				set_character_flag = brayherd_rite_okay
			}
			10 = { # Great rite
				trigger = { check_variable = { which = brayherd_herdstone_favour value = 10 } }
				modifier = {
					factor = 1.1
					check_variable = { which = brayherd_herdstone_favour value = 12 }
				}
				modifier = {
					factor = 1.1
					check_variable = { which = brayherd_herdstone_favour value = 12 }
				}
				modifier = {
					factor = 1.1
					check_variable = { which = brayherd_herdstone_favour value = 13 }
				}
				modifier = {
					factor = 1.1
					check_variable = { which = brayherd_herdstone_favour value = 14 }
				}
				modifier = {
					factor = 1.1
					check_variable = { which = brayherd_herdstone_favour value = 15 }
				}
				modifier = {
					factor = 1.1
					check_variable = { which = brayherd_herdstone_favour value = 17 }
				}
				modifier = {
					factor = 1.1
					check_variable = { which = brayherd_herdstone_favour value = 19 }
				}
				modifier = {
					factor = 1.1
					check_variable = { which = brayherd_herdstone_favour value = 21 }
				}
				modifier = {
					factor = 1.1
					check_variable = { which = brayherd_herdstone_favour value = 23 }
				}
				modifier = {
					factor = 1.1
					check_variable = { which = brayherd_herdstone_favour value = 25 }
				}
				modifier = {
					factor = 1.1
					check_variable = { which = brayherd_herdstone_favour value = 27 }
				}
				modifier = {
					factor = 1.1
					check_variable = { which = brayherd_herdstone_favour value = 29 }
				}
				set_character_flag = brayherd_rite_great
			}
		}
	}

	option = { # Bad rite: humble yourself
		name = EVTOPTAbrayherd39
		trigger = { has_character_flag = brayherd_rite_bad }
		# Kept from the Jade Dragon original (ask Komi before deleting):
		#add_trait = kow_tow_completed_tier_1
		add_offmap_currency = { offmap = brayherd_warp value = 100 }
		trigger_switch = {
			on_trigger = real_tier
			EMPEROR = { prestige = -1000 }
			KING = { prestige = -500 }
			fallback = { prestige = -100 }
		}
	}

	option = { # Bad rite: storm out
		name = EVTOPTBbrayherd39
		trigger = { has_character_flag = brayherd_rite_bad }
	}

	option = { # Okay rite
		name = EVTOPTCbrayherd39
		trigger = { has_character_flag = brayherd_rite_okay }
		add_offmap_currency = { offmap = brayherd_warp value = 150 }
		trigger_switch = {
			on_trigger = real_tier
			KING = { prestige = -400 piety = 250 change_martial = 1 health = 0.5 }
			EMPEROR = { prestige = -750 piety = 250 change_martial = 1 health = 0.5 }
		}
	}

	option = { # Great rite
		name = EVTOPTDbrayherd39
		trigger = { has_character_flag = brayherd_rite_great }
		# Kept from the Jade Dragon original (ask Komi before deleting):
		#add_trait = kow_tow_completed_tier_3
		add_offmap_currency = { offmap = brayherd_warp value = 300 }
		trigger_switch = {
			on_trigger = real_tier
			KING = { prestige = -500 piety = 300 change_martial = 2 health = 1 }
			EMPEROR = { prestige = -750 piety = 300 change_martial = 2 health = 1 }
		}
	}

	after = {
		brayherd_pilgrimage_end_effect = yes
	}
}
