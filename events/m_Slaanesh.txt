character_event = {
	id = slaaneshbless.4

	hide_window = yes

	only_capable = yes
	prisoner = no
	min_age = 16
	only_playable = yes

	trigger = {
		open_follower_of_slaanesh = yes
		has_ambition = obj_slaanesh_shed_a_virtue
		any_friend = {
			trait = slaanesh_daemonette
		}
		NOT = { has_character_flag = Pet_Daemonette_helped_ambition }
	}

	mean_time_to_happen = {
		years = 2
	}

	immediate = {
		set_character_flag = Pet_Daemonette_helped_ambition
		any_friend = {
			limit = { trait = slaanesh_daemonette }
			character_event = { id = slaaneshbless.5 }
		}
	}
}

# Pet_Daemonette bounce event
character_event = {
	id = slaaneshbless.5

	hide_window = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				OR = {
					FROM = { trait = zealous }
					FROM = { trait = ambitious }
					trait = genius
				}
			}
			random = {
				chance = 75
				FROM = { set_character_flag = slaanesh_obj_success }
			}
		}
		if = {
			limit = {
				FROM = { NOT = { trait = zealous } }
				FROM = { NOT = { trait = ambitious } }
				NOT = {	trait = genius }
			}
			random = {
				chance = 25
				FROM = { set_character_flag = slaanesh_obj_success }
			}
		}
		FROM = { character_event = { id = slaaneshbless.6 } }
	}
}

character_event = {
	id = slaaneshbless.6
	desc = slaaneshbless6
	picture = GFX_evt_bishop
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = slaaneshbless6A
		trigger = {
			has_character_flag = slaanesh_obj_success
		}
		if = {
			limit = { has_character_flag = amb_had_celibate trait = celibate }
			remove_trait = celibate
		}
		else_if = {
			limit = { has_character_flag = amb_had_chaste trait = chaste }
			remove_trait = chaste
		}
		else_if = {
			limit = { has_character_flag = amb_had_shy trait = shy }
			remove_trait = shy
		}
		else_if = {
			limit = { has_character_flag = amb_had_kind trait = kind }
			remove_trait = kind
		}
		else_if = {
			limit = { has_character_flag = amb_had_temperate trait = temperate }
			remove_trait = temperate
		}
		else_if = {
			limit = { has_character_flag = amb_had_charitable trait = charitable }
			remove_trait = charitable
		}
		else_if = {
			limit = { has_character_flag = amb_had_wroth trait = wroth }
			remove_trait = wroth
		}
		else_if = {
			limit = { has_character_flag = amb_had_humble trait = humble }
			remove_trait = humble
		}
		else_if = {
			limit = { has_character_flag = amb_had_ugly trait = disfigured }
			remove_trait = disfigured
		}
		else_if = {
			limit = { has_character_flag = amb_had_ugly trait = ugly }
			remove_trait = ugly
		}
		clr_character_flag = slaanesh_obj_success
	}
	option = {
		name = slaaneshbless6B
		trigger = {
			NOT = { has_character_flag = slaanesh_obj_success }
		}
		if = {
			limit = { NOT = { trait = diligent } }
			prestige = -10
			piety = -10
			random = {
				chance = 5
				add_trait = diligent
			}
		}
	}
