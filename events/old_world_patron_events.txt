# Choose a patron god of the Old World pantheon.
# Adapted from vanilla RoI.100/101 (Hindu patron deity).
# Called by choose_patron_god in decisions/minor_decisions.txt.
# Characters lean strongly towards the god of their own faith,
# then towards the gods of their father and spouse.

namespace = patron_god

# Page 1
character_event = {
	id = patron_god.1
	desc = EVTDESC_patron_god_1
	picture = GFX_evt_priest_blessing
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	option = {
		name = EVTOPT_patron_god_sigmar
		add_character_modifier = {
			name = patron_god_sigmar
			duration = -1
		}
		if = {
			limit = { religion = sigmarite }
			piety = 50
		}
		clr_character_flag = selecting_patron_deity
		if = {
			limit = { ai = no }
			chronicle = {
				entry = CHRONICLE_PATRON_GOD_SIGMAR
				picture = GFX_evt_priest_blessing
			}
		}
		ai_chance = {
			factor = 1
			modifier = {
				factor = 20
				religion = sigmarite
			}
			modifier = {
				factor = 6
				father = { religion = sigmarite }
			}
			modifier = {
				factor = 6
				any_spouse = { religion = sigmarite }
			}
		}
	}
	option = {
		name = EVTOPT_patron_god_ulric
		add_character_modifier = {
			name = patron_god_ulric
			duration = -1
		}
		if = {

				factor = 6

			}
		}
	}
	option = {
		name = EVTOPT_patron_god_more
		hidden_tooltip = { character_event = { id = patron_god.2 } }
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0
				OR = {
					religion = sigmarite
					religion = ulrican
					religion = taalite
					father = {
						OR = {
							religion = sigmarite
							religion = ulrican
							religion = taalite
						}
					}
					any_spouse = {
						OR = {
							religion = sigmarite
							religion = ulrican
							religion = taalite
						}
					}
				}
			}
			modifier = {
				factor = 40
				OR = {
					religion = myrmidian
					religion = shallyan
					religion = manannite
					religion = morrian
				}
			}
		}
	}
}
