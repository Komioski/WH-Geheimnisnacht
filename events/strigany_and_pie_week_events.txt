# Restored from vanilla soa_jewish_events.txt and merged with the mod's own
# versions of these ideas:
# - Expulsion (SoA.105, SoA.107) expels the Strigany, as in the epidemic
#   blame events (RIP.13402) and the peasant crusade (HFP.42322).
# - Passover (SoA.500-505, SoA.599) is the Halfling Pie Week
#   (celebrate_passover in minor_decisions.txt, loc in 00_imperial.csv).

namespace = SoA

# Strigany Expelled (vanilla SoA.105)
character_event = {
	id = SoA.105
	desc = EVTDESC_SoA_105
	picture = GFX_evt_wandering_strigany
	border = GFX_event_normal_frame_religion
	
	is_triggered_only = yes
	
	hide_from = yes
	
	immediate = {
		any_realm_lord = {
			limit = {
				religion = vampiric
				ai = yes
			}
			any_demesne_title = {
				grant_title_no_opinion = ROOT
			}
		}
	}

	option = {
		name = EVTOPTA_SoA_105
		hidden_tooltip = {
			any_realm_lord = {
				limit = { ai = no }
				character_event = { id = SoA.107 }
			}
		}
	}
}

### Halfling Pie Week (vanilla SoA.500-505)

# Invitation Sent
character_event = {
	id = SoA.500
	desc = EVTDESC_SoA_500
	picture = GFX_evt_halfling_chef
	border = GFX_event_normal_frame_religion
	
	is_triggered_only = yes
	
	hide_from = yes
	
	option = {
		name = EVTOPTA_SoA_500
		any_vassal = {
			limit = {
				religion_group = halfling_gods_group
				age = 16
				prisoner = no
				war = no
				NOT = { trait = incapable }
			}
			character_event = { id = SoA.501 days = 2 tooltip = EVTTOOLTIP_SoA_501 }
		}
		custom_tooltip = { text = passover_invitations_sent }
		hidden_tooltip = { character_event = { id = SoA.503 days = 10 } }
		hidden_tooltip = { character_event = { id = SoA.504 days = 12 } }
	}
}

# Invitation Received
character_event = {
	id = SoA.501
	desc = EVTDESC_SoA_501
	picture = GFX_evt_halfling_chef
	border = GFX_event_normal_frame_religion
	
	is_triggered_only = yes

	trigger = { NOT = { is_inaccessible_trigger = yes } }
	
	option = {
		name = EVTOPTA_SoA_501
		ai_chance = {
			factor = 80
			modifier = {
				factor = 0
				NOT = {	opinion = { who = ROOT value = -20 } }
			}
		}
		custom_tooltip = { text = passover_accepted }
		set_character_flag = attending_passover
		set_character_flag = do_not_disturb
		hidden_tooltip = { character_event = { id = SoA.503 days = 8 } }
		hidden_tooltip = { character_event = { id = SoA.599 days = 20 } } # Safety catch flag clearing
	}
	option = {
		name = EVTOPTB_SoA_501
		ai_chance = {
			factor = 20
		}
		FROM = {
			opinion = {
				modifier = opinion_refused_passover
				who = ROOT
			}
		}
		custom_tooltip = { text = passover_declined }
		hidden_tooltip = {
			FROM = { character_event = { id = SoA.502 } }
		}
	}
}

# Pie Week Jubilee Ends (Host)
character_event = {
	id = SoA.504
	desc = EVTDESC_SoA_504
	picture = GFX_evt_halfling_chef
	border = GFX_event_normal_frame_religion
	
	hide_from = yes
	
	is_triggered_only = yes
	
	option = {
		name = EVTOPTA_SoA_504
		any_vassal = {
			limit = { has_character_flag = attending_passover }
			opinion = {
				modifier = opinion_passover
				who = ROOT
			}
		}
		hidden_tooltip = {
			any_vassal = {
				limit = { has_character_flag = attending_passover }
				character_event = { id = SoA.505 }
			}
		}
		add_character_modifier = {
			name = held_passover_timer
			duration = 1825
			hidden = yes
		}
		prestige = 100
		piety = 100
		clr_character_flag = holding_passover
		clr_character_flag = do_not_disturb
	}
}
