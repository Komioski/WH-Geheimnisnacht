# Two halves:
# - cathay_expansion: what an Expansionist Cathay does in the west. It sends
#   armies to its colonies when they are attacked, sends settlers in peace,
#   and hunts the exiles who fled the Dragon Throne (rebel generals,
#   displaced princes, and anyone who sacked Cathay).
# - cathay_war: Cathay's own wars at home. Chaos assaults the Great Bastion
#   (End Times, needs doom), ogre tribes raid out of the Mountains of
#   Mourning, and the Skaven gnaw at the empire from below. Chaos alone can
#   take the Dragon Throne; the Dragon Emperor may come back for it.
#
# Yearly events fire from on_offmap_yearly_pulse: ROOT = governor, FROM = offmap.
# Loc: localisation/00_cathay_wars.csv

namespace = cathay_expansion
namespace = cathay_war

# Expansionist Cathay looks west (hidden)
character_event = {
	id = cathay_expansion.1
	hide_window = yes

	has_dlc = "Jade Dragon"

	is_triggered_only = yes

	trigger = {
		FROM = {
			is_offmap_tag = offmap_cathay
			has_policy = cathay_expansionist
			offmap_ruler = { NOT = { cathay_chaos_claimant_faith_trigger = yes } } # A Chaos usurper does not care for the colonies
			NOR = {
				has_status = cathay_famine
				has_status = cathay_plague
			}
			OR = {
				NOT = { has_offmap_tmp_flag = cathay_expansion_acted }
				had_offmap_tmp_flag = { flag = cathay_expansion_acted years = 2 }
			}
		}
	}

	immediate = {
		random_list = {
			40 = { # An army for a colony under attack
				trigger = {
					any_independent_ruler = {
						cathay_colony_ruler_trigger = yes
						war = yes
						is_primary_war_defender = yes
						NOT = { has_character_modifier = cathay_expedition_aid_cd }
					}
				}
				modifier = {
					factor = 0.25
					cathay_beset_trigger = yes
				}
				random_independent_ruler = {
					limit = {
						cathay_colony_ruler_trigger = yes
						war = yes
						is_primary_war_defender = yes
						NOT = { has_character_modifier = cathay_expedition_aid_cd }
					}
					character_event = { id = cathay_expansion.2 }
				}
				FROM = { set_offmap_tmp_flag = cathay_expansion_acted }
			}
			15 = { # Settlers for a colony at peace
				trigger = {
					any_independent_ruler = {
						cathay_colony_ruler_trigger = yes
						war = no
						NOT = { has_character_modifier = cathay_settlers_cd }
					}
				}
				random_independent_ruler = {
					limit = {
						cathay_colony_ruler_trigger = yes
						war = no
						NOT = { has_character_modifier = cathay_settlers_cd }
					}
					character_event = { id = cathay_expansion.4 }
				}
				FROM = { set_offmap_tmp_flag = cathay_expansion_acted }
			}
			25 = { # Hunt an exile
				trigger = {
					any_character = {
						cathay_exile_trigger = yes
						NOT = { has_character_modifier = cathay_hunted_cd }
					}
				}
				modifier = { # The Dragon Emperor never forgets
					factor = 2
					dragon_emperor_reigns_trigger = yes
				}
				random_character = {
					limit = {
						cathay_exile_trigger = yes
						NOT = { has_character_modifier = cathay_hunted_cd }
					}
					preferred_limit = { has_character_flag = cathay_sworn_vengeance }
					add_character_modifier = {
						name = cathay_hunted_cd
						years = 5
						hidden = yes
					}
					save_event_target_as = cathay_hunted_exile
					if = { # A guest at someone's court: demand them back
						limit = {
							is_ruler = no
							host = { is_ruler = yes }
						}
						host = { character_event = { id = cathay_expansion.6 } }
					}
					else = { # Send the jade assassins
						ROOT = { character_event = { id = cathay_expansion.5 } }
					}
				}
				FROM = { set_offmap_tmp_flag = cathay_expansion_acted }
			}
			20 = { }
		}
	}
}

# The Dragon sends aid. ROOT = colony ruler, FROM = governor
character_event = {
	id = cathay_expansion.2
	title = EVTNAME_cathay_expansion_2
	desc = EVTDESC_cathay_expansion_2
	picture = GFX_evt_china_invades
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	immediate = {
		add_character_modifier = {
			name = cathay_expedition_aid_cd
			years = 5
			hidden = yes
		}
		capital_scope = { save_event_target_as = cathay_aid_landing }
		spawn_unit = {
			province = event_target:cathay_aid_landing
			owner = ROOT
			troops = {
				light_infantry = { 1500 1500 }
				heavy_infantry = { 1000 1000 }
				archers = { 1500 1500 }
				light_cavalry = { 500 500 }
			}
			earmark = cathay_dragon_host
			disband_on_peace = yes
			attrition = 1.0
		}
		if = {
			limit = { offmap_cathay = { has_status = cathay_golden_age } }
			spawn_unit = {
				province = event_target:cathay_aid_landing
				owner = ROOT
				troops = {
					heavy_infantry = { 1500 1500 }
				}
				earmark = cathay_dragon_host
				disband_on_peace = yes
				attrition = 1.0
			}
		}
		any_player = {
			limit = {
				NOT = { character = ROOT }
				has_offmap_news_enabled = offmap_cathay
				is_within_diplo_range = ROOT
			}
			narrative_event = { id = cathay_expansion.3 }
		}
	}

	option = {
		name = EVTOPTA_cathay_expansion_2
		custom_tooltip = { text = cathay_dragon_host_tt }
		add_offmap_currency = {
			offmap = offmap_cathay
			value = 50
		}
	}
}

					}
					modifier = {
						factor = 1.5
						dragon_emperor_reigns_trigger = yes
					}
					any_player = {
						limit = {
							NOT = { character = PREV }
							OR = {
								AND = {
									has_offmap_news_enabled = offmap_cathay
									is_within_diplo_range = PREV
								}
								any_courtier = { character = PREVPREV }
							}
						}
						narrative_event = { id = cathay_expansion.7 }
					}
					death = {
						death_reason = death_murder
						killer = ROOT
					}
				}
				70 = { # Survived
					add_character_modifier = {
						name = hunted_by_cathay
						years = 5
					}
					if = {
						limit = { ai = no }
						character_event = { id = cathay_expansion.8 }
					}
				}
			}
		}
	}
}
