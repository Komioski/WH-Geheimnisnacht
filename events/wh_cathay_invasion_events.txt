# Non-Chaos realms invade Cathay for what it has, not for the Dragon Throne.
# Started by the cathay_launch_expedition decision; the war is
# invasion_of_cathay_cb and its on_success pays out the chosen goal.
# Loc: localisation/00_cathay_wars.csv

namespace = cathay_expedition

# Choose the goal of the expedition. ROOT = invader
character_event = {
	id = cathay_expedition.1
	title = EVTNAME_cathay_expedition_1
	desc = EVTDESC_cathay_expedition_1
	picture = GFX_evt_china_invades
	border = GFX_event_normal_frame_war

	is_triggered_only = yes

	option = { # Plunder
		name = EVTOPTA_cathay_expedition_1
		custom_tooltip = { text = cathay_expedition_goal_plunder_tt }
		cathay_expedition_clear_flags_effect = yes
		set_character_flag = cathay_expedition_goal_plunder
		cathay_expedition_launch_effect = yes
		ai_chance = {
			factor = 10
			modifier = {
				factor = 3
				OR = {
					culture_group = ogre_group
					culture_group = orc_group
					culture_group = goblin_group
					culture_group = hung_group
					culture_group = kurgan_group
				}
			}
		}
	}
	option = { # Slaves
		name = EVTOPTB_cathay_expedition_1
		custom_tooltip = { text = cathay_expedition_goal_slaves_tt }
		cathay_expedition_clear_flags_effect = yes
		set_character_flag = cathay_expedition_goal_slaves
		cathay_expedition_launch_effect = yes
		ai_chance = {
			factor = 3
			modifier = {
				factor = 5
				OR = {
					culture_group = druchii_group
					culture_group = dwarf_chaos_group
					culture_group = skaven_group
				}
			}
		}
	}
	option = { # Influence
		name = EVTOPTC_cathay_expedition_1
		custom_tooltip = { text = cathay_expedition_goal_influence_tt }
		cathay_expedition_clear_flags_effect = yes
		set_character_flag = cathay_expedition_goal_influence
		cathay_expedition_launch_effect = yes
		ai_chance = {
			factor = 5
			modifier = {
				factor = 2
				diplomacy = 15
			}
		}
	}
	option = { # Knowledge
		name = EVTOPTD_cathay_expedition_1
		custom_tooltip = { text = cathay_expedition_goal_knowledge_tt }
		cathay_expedition_clear_flags_effect = yes
		set_character_flag = cathay_expedition_goal_knowledge
		cathay_expedition_launch_effect = yes
		ai_chance = {
			factor = 5
			modifier = {
				factor = 3
				OR = {
					culture_group = asur_group
					culture_group = lizardman_group
				}
			}
			modifier = {
				factor = 2
				learning = 15
			}
		}
	}
	option = { # Not yet
		name = EVTOPTE_cathay_expedition_1
		ai_chance = { factor = 0 }
	}
}
