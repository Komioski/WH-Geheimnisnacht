# Cathay is ruled by the immortal Dragon Emperor (history/characters, 18064).
# Vanilla Jade Dragon only changes Cathay's policy when a new emperor takes
# the throne (JD.50000 from on_offmap_ruler_changed), which never happens
# here, so the policy would stay Isolationist forever and everything that
# needs an Open Cathay would never fire. These events let him change his
# mind instead. The policy news itself comes from on_offmap_policy_changed.
# He turns Expansionist to defend Cathay's colonies in the west, to hunt the
# exiles who fled him, or to punish those who sacked Cathay
# (events/wh_cathay_events.txt does the work).

namespace = dragon_emperor

# The Dragon Emperor reconsiders Cathay's policy (hidden)
# Fired from on_offmap_yearly_pulse. ROOT = governor, FROM = offmap.
# At most once every 15 years, then a small yearly chance.
character_event = {
	id = dragon_emperor.1
	hide_window = yes

	has_dlc = "Jade Dragon"

	is_triggered_only = yes

	trigger = {
		FROM = {
			is_offmap_tag = offmap_cathay
			days_since_policy_change = 5475
		}
		dragon_emperor_reigns_trigger = yes
	}

	immediate = {
		FROM = {
			if = {
				limit = { has_policy = cathay_isolationist }
				random_list = {
					8 = { # Open the borders to trade
						modifier = {
							factor = 0
							has_offmap_flag = cathay_isolationist_trend
						}
						modifier = {
							factor = 1.5
							has_status = cathay_stable
						}
						modifier = {
							factor = 2
							has_status = cathay_golden_age
						}
						modifier = {
							factor = 0.25
							cathay_is_suffering_trigger = yes
						}
						set_policy = cathay_open
					}
					2 = { # March west
						modifier = { # A colony in the west is under attack
							factor = 4
							cathay_colony_in_danger_trigger = yes
						}
						modifier = { # Exiles and despoilers still at large
							factor = 3
							cathay_exile_at_large_trigger = yes
						}
						modifier = {
							factor = 5
							has_offmap_flag = cathay_expansionist_trend
						}
						modifier = {
							factor = 2
							has_status = cathay_golden_age
						}
						modifier = {
							factor = 0.25
							cathay_is_suffering_trigger = yes
						}
						modifier = { # Busy with a war at home
							factor = 0
							cathay_beset_trigger = yes
						}
						set_policy = cathay_expansionist
						clr_offmap_flag = cathay_expansionist_trend
					}
					90 = { }
				}
			}
			else_if = {
				limit = { has_policy = cathay_open }
				random_list = {
					5 = { # Close the borders again
						modifier = {
							factor = 0
							has_offmap_tmp_flag = cathay_forced_open
							NOT = { had_offmap_tmp_flag = { flag = cathay_forced_open years = 50 } }
						}
						modifier = {
							factor = 5
							has_offmap_flag = cathay_isolationist_trend
						}
						modifier = {
							factor = 3
							cathay_is_suffering_trigger = yes
						}
						modifier = {
							factor = 0.5
							has_status = cathay_golden_age
						}
						set_policy = cathay_isolationist
						clr_offmap_flag = cathay_isolationist_trend
						clr_offmap_tmp_flag = cathay_forced_open
					}
					4 = { # March west
						modifier = { # A colony in the west is under attack
							factor = 4
							cathay_colony_in_danger_trigger = yes
						}
						modifier = { # Exiles and despoilers still at large
							factor = 3
							cathay_exile_at_large_trigger = yes
						}
						modifier = {
							factor = 5
							has_offmap_flag = cathay_expansionist_trend
						}
						modifier = {
							factor = 2
							has_status = cathay_golden_age
						}
						modifier = {
							factor = 0.25
							cathay_is_suffering_trigger = yes
						}
						modifier = { # Busy with a war at home
							factor = 0
							cathay_beset_trigger = yes
						}
						set_policy = cathay_expansionist
						clr_offmap_flag = cathay_expansionist_trend
					}
					91 = { }
				}
			}
			else_if = {
				limit = { has_policy = cathay_expansionist }
				random_list = {
					12 = { # Enough of the west; back to trade
						modifier = { # Nothing left to fight for
							factor = 3
							NOR = {
								cathay_colony_in_danger_trigger = yes
								cathay_exile_at_large_trigger = yes
							}
						}
						modifier = {
							factor = 2
							cathay_is_suffering_trigger = yes
						}
						set_policy = cathay_open
					}
					3 = { # Withdraw behind the Great Bastion
						modifier = {
							factor = 0
							has_offmap_tmp_flag = cathay_forced_open
							NOT = { had_offmap_tmp_flag = { flag = cathay_forced_open years = 50 } }
						}
						modifier = {
							factor = 4
							cathay_is_suffering_trigger = yes
						}
						set_policy = cathay_isolationist
						clr_offmap_tmp_flag = cathay_forced_open
					}
					85 = { }
				}
			}
		}
	}
}
