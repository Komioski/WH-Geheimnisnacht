# The jungle
# The Lustrian jungle has four densities (province modifiers lustria_light_jungle
# to lustria_oppressive_jungle). It grows every few years and brings sickness,
# beasts and lost villages to warm-blooded settlers.
###############################################################################

# lustria.1 Jungle setup
# On startup, gives every Lustrian demesne province (outside Quetza) the
# jungle modifiers.
# Scope: ROOT is a ruler with a capital in Lustria.
character_event = {
	id = lustria.1
	hide_window = yes

	only_rulers = yes

	is_triggered_only = yes

	trigger = {
		is_save_game = no
		capital_scope = { location = { region = world_new_world_lustria } }
	}

	immediate = {
		ROOT = {
			any_demesne_province = {
				limit = {
					location = {
						region = world_new_world_lustria
						NOT = { duchy = { title = d_quetza } }
					}
					NOT = { has_province_modifier = lustria_jungle }
				}
				# Manually add the modifier on startup
				province_event = { id = lustria.2 }
			}
		}
	}

	option = {
		name = OK
	}
}

# lustria.2 Jungle setup for one province
# Adds the jungle modifier and a random starting density (lighter near rivers,
# coasts and big capitals).
# Scope: ROOT is the province.
province_event = {
	id = lustria.2
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		add_province_modifier = { name = lustria_jungle duration = -1 }
		# Random starting density
		if = {
			limit = {
				region = world_new_world_lustria
				NOR = {
					has_province_modifier = lustria_light_jungle
					has_province_modifier = lustria_medium_jungle
					has_province_modifier = lustria_dense_jungle
					has_province_modifier = lustria_oppressive_jungle
				}
			}
			# Make it so the initial jungle level is random
			random_list = {
				# Light Jungle
				25 = {
					# Kingdom level tier is more likely to have light jungle
					modifier = {
						factor = 2.0
						is_capital = yes
						holder_scope = { higher_real_tier_than = DUKE }
					}
					# Duke level is a bit more likely to have light jungle
					modifier = { factor = 1.5 is_capital = yes holder_scope = { real_tier = DUKE } }
					# Count level is a bit more likely to have light jungle
					modifier = {
						factor = 1.25
						is_capital = yes
						holder_scope = { real_tier = COUNT }
					}
					# Ports and rivers more likely to have light jungle
					modifier = { factor = 3 OR = { borders_major_river = yes port = yes } }
					add_province_modifier = { name = lustria_light_jungle duration = -1 }
				}
				# Medium Jungle
				25 = {
					# Ports and rivers more likely to have light jungle
					modifier = { factor = 2 OR = { borders_major_river = yes port = yes } }
					add_province_modifier = { name = lustria_medium_jungle duration = -1 }
				}
				# Dense Jungle
				25 = { add_province_modifier = { name = lustria_dense_jungle duration = -1 } }
				# Oppressive Jungle
				25 = {
					# Ports and rivers more likely to have light jungle
					modifier = { factor = 0 OR = { borders_major_river = yes port = yes } }
					add_province_modifier = { name = lustria_oppressive_jungle duration = -1 }
				}
			}
		}
	}

	option = {
		name = OK
	}
}

# lustria.3 The jungle grows
# Five-year pulse. The jungle in Lustrian demesne provinces grows a step. AI
# rulers may instead clear a step (lizardmen and Amazons rarely do).
# Scope: ROOT is a ruler holding Lustrian jungle.
character_event = {
	id = lustria.3
	desc = EVTDESClustria3
	picture = "GFX_evt_lustria_jungle"

	only_rulers = yes

	is_triggered_only = yes # on_five_year_pulse (list lustria_jungle)

	trigger = {
		any_demesne_province = {
			region = world_new_world_lustria
			terrain = jungle
			has_province_modifier = lustria_jungle
			OR = {
				NOT = { has_province_modifier = lustria_oppressive_jungle }
				ROOT = { ai = yes } # the AI also clears its jungle with this event
			}
		}
	}

	weight_multiplier = {
		factor = 1
		# The children of the Old Ones live with the jungle
		modifier = { factor = 0.4 religion_group = old_ones_group }
		# Feudal realms clear it
		modifier = { factor = 0.65 is_tribal = no is_nomadic = no }
		# Tribes use it more than they fight it
		modifier = { factor = 0.85 is_tribal = yes }
		# Nomads pass through and leave it be
		modifier = { factor = 2.0 is_nomadic = yes }
		# Good stewardship keeps the jungle at bay
		modifier = { factor = 0.9 realm_stewardship > 10 }
		modifier = { factor = 0.8 realm_stewardship > 15 }
		modifier = { factor = 0.7 realm_stewardship > 20 }
		modifier = { factor = 0.6 realm_stewardship > 25 }
	}

	# The jungle grows
	option = {
		name = OK
		trigger = { ai = no }
		any_demesne_province = {
			limit = {
				region = world_new_world_lustria
				terrain = jungle
				has_province_modifier = lustria_jungle
			}
			lustria_jungle_grow_effect = yes
		}
	}
	# AI: the jungle grows, or the ruler clears a step of it
	option = {
		name = OK
		trigger = { ai = yes }
		random_list = {
			# Growth, a coin toss for each province
			50 = {
				modifier = { factor = 1.2 realm_stewardship < 15 }
				any_demesne_province = {
					limit = {
						region = world_new_world_lustria
						terrain = jungle
						has_province_modifier = lustria_jungle
					}
					random = {
						chance = 50
						# Slower near rivers and coasts (the wild heart of Lustria is inland)
						modifier = { factor = 0.8 OR = { borders_major_river = yes port = yes } }
						modifier = { factor = 0.8 has_province_modifier = lustria_dense_jungle }
						modifier = { factor = 0.75 has_province_modifier = lustria_light_jungle }
						modifier = { factor = 0.9 ROOT = { is_tribal = no is_nomadic = no } }
						lustria_jungle_grow_effect = yes
					}
				}
			}
			# Clearing
			50 = {
				modifier = {
					factor = 0.1
					capital_scope = { has_province_modifier = lustria_light_jungle }
				}
				modifier = { factor = 0.1 religion_group = old_ones_group }
				modifier = { factor = 0.5 OR = { religion = amazon religion = kukaili } }
				modifier = { factor = 1.15 realm_stewardship > 10 }
				modifier = { factor = 1.15 realm_stewardship > 15 }
				modifier = { factor = 1.15 realm_stewardship > 20 }
				modifier = { factor = 1.15 realm_stewardship > 25 }
				modifier = { factor = 1.3 independent = no top_liege = { realm_stewardship > 20 } }
				any_demesne_province = {
					limit = {
						region = world_new_world_lustria
						terrain = jungle
						has_province_modifier = lustria_jungle
					}
					if = {
						limit = { has_province_modifier = lustria_light_jungle }
						random = { chance = 20 remove_province_modifier = lustria_light_jungle }
					}
					else = { lustria_jungle_clear_effect = yes }
				}
			}
		}
	}
}

# lustria.4 Jungle fever
# Five-year pulse. The ruler or a courtier catches a sickness from the jungle
# (worse in dense jungle and for the sickly).
# Scope: ROOT is a non-lizardman ruler in Lustria; target_character is the
# sick one.
character_event = {
	id = lustria.4
	desc = {
		text = EVTDESClustria4
		trigger = { NOT = { event_target:target_character = { character = ROOT } } }
	}
	desc = {
		text = EVTDESClustria4b
		trigger = { event_target:target_character = { character = ROOT } }
	}
	picture = "GFX_evt_lustria_jungle"

	only_rulers = yes

	is_triggered_only = yes # on_five_year_pulse (list lustria)

	trigger = {
		NOT = { culture_group = lizardman_group }
		location = { region = world_new_world_lustria has_province_modifier = lustria_jungle }
	}

	weight_multiplier = {
		factor = 1
		# The deeper the jungle, the worse the fevers
		modifier = { factor = 1.2 location = { has_province_modifier = lustria_medium_jungle } }
		modifier = { factor = 1.6 location = { has_province_modifier = lustria_dense_jungle } }
		modifier = { factor = 2.2 location = { has_province_modifier = lustria_oppressive_jungle } }
		# The sickly fall ill more easily
		modifier = { factor = 1.5 health < 5 }
		# Amazons and Pygmies were born to it
		modifier = { factor = 0.5 OR = { religion = amazon religion = kukaili } }
	}

	immediate = {
		random_list = {
			50 = { save_event_target_as = target_character }
			50 = {
				trigger = {
					any_courtier = {
						location = {
							region = world_new_world_lustria
							has_province_modifier = lustria_jungle
						}
					}
				}
				random_courtier = {
					limit = {
						location = {
							region = world_new_world_lustria
							has_province_modifier = lustria_jungle
						}
					}
					save_event_target_as = target_character
				}
			}
		}
	}

	option = {
		name = OK
		event_target:target_character = { add_symptom_effect = yes }
	}
}

# lustria.7 Attacked by a jungle beast
# Decade pulse (and sometimes while clearing jungle). A monster of the jungle
# stalks the ruler and forces a duel.
# Scope: ROOT is a ruler in Lustria who commands, quests or hunts.
character_event = {
	id = lustria.7
	desc = { text = EVTDESClustria7 trigger = { NOT = { has_focus = focus_hunting } } }
	desc = { text = EVTDESClustria7b trigger = { has_focus = focus_hunting } }
	picture = "GFX_evt_lustria_dinosaur"

	only_rulers = yes

	is_triggered_only = yes

	trigger = {
		location = { region = world_new_world_lustria }
		OR = {
			in_command = yes
			trait = questing
			has_focus = focus_hunting
			has_character_flag = the_rush_of_hunt
		}
	}

	weight_multiplier = {
		factor = 1
		# The deeper the jungle, the bigger the beasts
		modifier = { factor = 1.25 location = { has_province_modifier = lustria_medium_jungle } }
		modifier = { factor = 1.5 location = { has_province_modifier = lustria_dense_jungle } }
		modifier = { factor = 2 location = { has_province_modifier = lustria_oppressive_jungle } }
	}

	option = {
		name = OK
		# Creature gets native name #Creatures: Razorbacks, Cold Ones, Terradons, Stegadon, Sauropod, Thunder Lizards, Carnosaur
		create_random_soldier = {
			random_traits = no
			dynasty = none
			religion = old_ones
			culture = lizardman
			name = " "
			female = no
			age = 30
			attributes = { martial = 5 diplomacy = 0 stewardship = 0 learning = 0 intrigue = 0 }
		}
		# scope to character we created and make him send us a duel invite event - so we can't run away!
		new_character = {
			set_graphical_culture = generic_monster
			add_trait = creature_monster
			add_trait = wroth
			add_trait = brave
			random_list = {
				10 = {
					give_nickname = nick_the_razorback
					change_martial = 5
					add_five_duel_experience_modifier = yes
				}
				10 = {
					give_nickname = nick_the_cold_one
					change_martial = 6
					add_ten_duel_experience_modifier = yes
					if = {
						limit = { ROOT = { combat_rating >= 100 } }
						add_ten_duel_experience_modifier = yes
						add_ten_duel_experience_modifier = yes
					}
					if = {
						limit = { ROOT = { combat_rating >= 100 has_focus = focus_hunting } }
						add_twenty_five_duel_experience_modifier = yes
						add_twenty_five_duel_experience_modifier = yes
					}
				}
				10 = { give_nickname = nick_the_terradon change_martial = 7 }
				10 = { give_nickname = nick_the_stegadon change_martial = 8 }
				10 = { give_nickname = nick_the_sauropod change_martial = 9 }
				10 = {
					give_nickname = nick_the_thunder_lizard
					change_martial = 10
					add_five_duel_experience_modifier = yes
				}
				10 = {
					give_nickname = nick_the_carnosaur
					change_martial = 15
					add_ten_duel_experience_modifier = yes
					add_ten_duel_experience_modifier = yes
					if = {
						limit = {
							ROOT = { combat_rating >= 100 }
							NOT = { has_focus = focus_hunting }
						}
						add_ten_duel_experience_modifier = yes
						add_ten_duel_experience_modifier = yes
						add_ten_duel_experience_modifier = yes
					}
					if = {
						limit = { ROOT = { combat_rating >= 100 has_focus = focus_hunting } }
						add_five_duel_experience_modifier = yes
						add_twenty_five_duel_experience_modifier = yes
						add_twenty_five_duel_experience_modifier = yes
						add_twenty_five_duel_experience_modifier = yes
						add_twenty_five_duel_experience_modifier = yes
					}
				}
			}
			character_event = { id = lustria.8 days = 1 }
		}
		if = { limit = { has_focus = focus_hunting } set_character_flag = the_rush_of_hunt }
	}
}

# lustria.15 Cutting back the jungle
# From the cutback_jungle_decision settlement decision: do it yourself, send
# the steward, or leave it.
# Scope: ROOT is the ruler, FROM the holding; cutback_province is its
# province.
character_event = {
	id = lustria.15
	desc = EVTDESClustria15
	picture = "GFX_evt_lustria_jungle"

	is_triggered_only = yes # cutback_jungle_decision

	immediate = {
		FROM = { location = { save_event_target_as = cutback_province } }
	}

	# Do it yourself
	option = {
		name = EVTOPTAlustria15
		trigger = { in_command = no is_incapacitated_trigger = no }
		if = { limit = { is_tribal = no } prestige = -50 }
		else = { prestige = -25 }
		random = {
			chance = 5
			set_character_flag = the_rush_of_hunt
			character_event = { id = lustria.7 days = 15 random = 12 }
		}
		character_event = { id = lustria.16 days = 20 random = 30 }
		ai_chance = { factor = 1 modifier = { factor = 2 stewardship = 12 } }
	}
	# Send the steward
	option = {
		name = EVTOPTBlustria15
		trigger = {
			job_treasurer = { is_adult = yes in_command = no is_incapacitated_trigger = no }
		}
		if = { limit = { is_tribal = no } wealth = -50 }
		else = { wealth = -25 }
		job_treasurer = { letter_event = { id = lustria.20 } }
		ai_chance = { factor = 2 }
	}
	# Never mind
	option = {
		name = EVTOPTClustria15
		ai_chance = { factor = 0.1 }
	}
}
