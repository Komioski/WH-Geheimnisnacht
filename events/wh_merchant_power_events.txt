# Two merchant powers that don't follow the Marienburg League's story
# (REP.40-70 in wh_republic_events.txt):
#
# - The Communard Concord. The Communard Cities of Estalia are an old,
#   established republic, not a rising one. Some decades into the game their
#   tribunes demand an answer to the newer merchants crowding their seas: buy
#   up trading posts on the southern coasts, arm the cities, or take on the
#   Marienburg League directly.
#
# - The Pigbarter Compact. Pigbarter is a trading town of five houses from
#   five corners of the world, perched among greenskins, ogres and Chaos
#   dwarfs. It starts with a compact for its defence, its neighbours keep
#   demanding tribute, and when the town falls its houses scatter home.

namespace = communard_concord

# The tribunes grow restless (hidden)
# Fired from on_decade_pulse.
character_event = {
	id = communard_concord.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_landed_title = e_communards
		independent = yes
		is_merchant_republic = yes
		year = 2030
		NOT = { has_global_flag = communard_concord_called }
	}

	immediate = {
		random_list = {
			60 = {
				set_global_flag = communard_concord_called
				character_event = { id = communard_concord.2 days = 30 random = 300 }
			}
			40 = { }
		}
	}
}

# The Tribunes' Assembly
character_event = {
	id = communard_concord.2
	desc = EVTDESC_communard_concord_2
	picture = GFX_evt_council_quarrelling
	border = GFX_event_normal_frame_economy

	is_triggered_only = yes

	trigger = {
		has_landed_title = e_communards
		independent = yes
		is_merchant_republic = yes
	}

	option = { # Buy up trading posts along the southern coasts
		name = EVTOPTA_communard_concord_2
		trigger = { wealth = 150 }
		ai_chance = { factor = 50 }
		wealth = -150
		add_character_modifier = {
			name = communard_free_ports
			years = 25
		}
		custom_tooltip = { text = communard_free_ports_tooltip }
		set_global_flag = communard_free_ports
		hidden_effect = {
			character_event = { id = communard_concord.3 days = 5 }
			narrative_event = { id = communard_concord.4 days = 2 }
		}
	}
	option = { # Arm the cities
		name = EVTOPTB_communard_concord_2
		ai_chance = { factor = 30 }
		add_character_modifier = {
			name = communard_city_militias
			years = 25
		}
		set_global_flag = communard_militias
		hidden_effect = {
			narrative_event = { id = communard_concord.4 days = 2 }
		}
	}
	option = { # Take on the Marienburg League
		name = EVTOPTC_communard_concord_2
		trigger = {
			k_marienburg_league = { has_holder = yes }
		}
		ai_chance = {
			factor = 20
			modifier = {
				factor = 2
				trait = ambitious
			}
		}
		add_character_modifier = {
			name = communard_league_rivalry
			years = 25
		}
		k_marienburg_league = {
			holder_scope = {
				save_event_target_as = league_holder
				opinion = {
					modifier = opinion_unhappy
					who = ROOT
					years = 20
				}
				reverse_opinion = {
					modifier = opinion_unhappy
					who = ROOT
					years = 20
				}
			}
		}
		custom_tooltip = { text = communard_league_rivalry_tooltip }
		set_global_flag = communard_league_rivalry
		hidden_effect = {
			random_province = {
				limit = {
					has_trade_post = yes
					trade_post_owner = {
						OR = {
							character = event_target:league_holder
							same_realm = event_target:league_holder
						}
					}
				}
				trade_post_owner = { character_event = { id = communard_concord.6 } }
				seize_trade_post = ROOT
			}
			narrative_event = { id = communard_concord.4 days = 2 }
		}
	}
}

		}
	}

	id = pigbarter_compact.4
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_landed_title = d_pigbarter
		NOT = { has_global_flag = pigbarter_fallen }
		1274 = { owner = { same_realm = ROOT } }
	}

	immediate = {
		if = {
			limit = {
				any_neighbor_independent_ruler = {
					NOT = { trait = creature_human }
					NOT = { is_merchant_republic = yes }
				}
			}
			character_event = { id = pigbarter_compact.5 }
		}
		character_event = { id = pigbarter_compact.4 days = 1095 random = 730 }
	}
}

# Pigbarter falls (hidden): the houses scatter home
# Fired from on_new_holder and on_new_holder_usurpation.
# ROOT = new holder, FROM = title, FROMFROM = old holder.
character_event = {
	id = pigbarter_compact.10
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		FROM = { title = c_pigbarter }
		has_global_flag = pigbarter_compact
		NOT = { has_global_flag = pigbarter_fallen }
		NOR = {
			is_merchant_republic = yes
			is_patrician = yes
			any_liege = { is_merchant_republic = yes }
		}
	}

	immediate = {
		set_global_flag = pigbarter_fallen
		save_event_target_as = pigbarter_conqueror

		# Where each house goes
		random_independent_ruler = { # House of Yabbas (Arabyan)
			limit = { culture_group = arabyan_group }
			preferred_limit = { is_merchant_republic = yes }
			save_event_target_as = pigbarter_refuge_araby
		}
		random_independent_ruler = { # Westerlander house
			limit = { culture_group = imperial_group }
			preferred_limit = {
				OR = {
					has_landed_title = k_marienburg_league
					culture = westerlander
				}
			}
			save_event_target_as = pigbarter_refuge_westerland
		}
		random_independent_ruler = { # Cathayan house
			limit = { culture_group = oriental_group }
			save_event_target_as = pigbarter_refuge_cathay
		}
		random_independent_ruler = { # Reman house
			limit = { culture_group = tilean_group }
			preferred_limit = { is_merchant_republic = yes }
			save_event_target_as = pigbarter_refuge_tilea
		}
		random_independent_ruler = { # Strigany house
			limit = {
				OR = {
					culture_group = strigany_group
					religion_group = necromantic_group
				}
			}
			save_event_target_as = pigbarter_refuge_strigany
		}
		random_independent_ruler = { # Anyone left over goes to another merchant republic
			limit = {
				is_merchant_republic = yes
				NOT = { character = ROOT }
			}
			save_event_target_as = pigbarter_refuge_any
		}

		any_character = {
			limit = {
				is_ruler = no
				prisoner = no
				OR = {
					dynasty = 48135
					dynasty = 13583
					dynasty = 9192
					dynasty = 43176
					dynasty = 87072
				}
			}
			if = {
				limit = { dynasty = 48135 event_target:pigbarter_refuge_araby = { always = yes } }
				move_character = event_target:pigbarter_refuge_araby
			}
			else_if = {
				limit = { dynasty = 13583 event_target:pigbarter_refuge_westerland = { always = yes } }
				move_character = event_target:pigbarter_refuge_westerland
			}
			else_if = {
				limit = { dynasty = 9192 event_target:pigbarter_refuge_cathay = { always = yes } }
				move_character = event_target:pigbarter_refuge_cathay
			}
			else_if = {
				limit = { dynasty = 43176 event_target:pigbarter_refuge_tilea = { always = yes } }
				move_character = event_target:pigbarter_refuge_tilea
			}
			else_if = {
				limit = { dynasty = 87072 event_target:pigbarter_refuge_strigany = { always = yes } }
				move_character = event_target:pigbarter_refuge_strigany
			}
			else_if = {
				limit = { event_target:pigbarter_refuge_any = { always = yes } }
				move_character = event_target:pigbarter_refuge_any
			}
		}

		any_player = {
			limit = {
				OR = {
					is_within_diplo_range = ROOT
					is_merchant_republic = yes
					is_patrician = yes
					dynasty = 48135
					dynasty = 13583
					dynasty = 9192
					dynasty = 43176
					dynasty = 87072
				}
			}
			narrative_event = { id = pigbarter_compact.11 }
		}
	}
}
