# Two merchant powers that don't follow the Marienburg League's story
# (REP.40-70 in wh_republic_events.txt):
#
# - The Communard Concord. The Communard Cities of Estalia are an old,
#   established republic, not a rising one. Some decades into the game their
#   tribunes demand an answer to the newer merchants crowding their seas: buy
#   up trading posts on the southern coasts, arm the cities, or take on the
#   Marienburg League directly.
#
# - The Pigbarter Compact. Pigbarter is a trading town of five houses from
#   five corners of the world, perched among greenskins, ogres and Chaos
#   dwarfs. It starts with a compact for its defence and a standing guard,
#   gets help from the houses' kin when attacked, sees an occasional tribute
#   demand, and when the town falls its houses go home.

namespace = communard_concord

# The tribunes grow restless (hidden)
# Fired from on_decade_pulse.
character_event = {
	id = communard_concord.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_landed_title = e_communards
		independent = yes
		is_merchant_republic = yes
		year = 2030
		NOT = { has_global_flag = communard_concord_called }
	}

	immediate = {
		random_list = {
			60 = {
				set_global_flag = communard_concord_called
				character_event = { id = communard_concord.2 days = 30 random = 300 }
			}
			40 = { }
		}
	}
}

			name = communard_city_militias

		}
		set_global_flag = communard_militias
		hidden_effect = {
			narrative_event = { id = communard_concord.4 days = 2 }
		}
	}
	option = { # Take on the Marienburg League
		name = EVTOPTC_communard_concord_2
		trigger = {
			k_marienburg_league = { has_holder = yes }
		}
		ai_chance = {
			factor = 20
			modifier = {
				factor = 2
				trait = ambitious
			}
		}
		add_character_modifier = {
			name = communard_league_rivalry
			years = 25
		}

			holder_scope = {
				save_event_target_as = league_holder

					modifier = opinion_unhappy
					who = ROOT
					years = 20
				}
				reverse_opinion = {
					modifier = opinion_unhappy
					who = ROOT
					years = 20
				}
			}
		}
		custom_tooltip = { text = communard_league_rivalry_tooltip }
		set_global_flag = communard_league_rivalry
		hidden_effect = {
			random_province = {
				limit = {
					has_trade_post = yes
					trade_post_owner = {
						OR = {
							character = event_target:league_holder
							same_realm = event_target:league_holder
						}
					}
				}
				trade_post_owner = { character_event = { id = communard_concord.6 } }
				seize_trade_post = ROOT

		}

	}

		set_global_flag = pigbarter_compact
		1274 = {
			add_province_modifier = {
				name = pigbarter_compact
				duration = -1
			}
		}
		spawn_unit = {
			province = 1274
			home = 1274
			owner = ROOT
			earmark = pigbarter_house_guard
			attrition = 0.5
			troops = {
				heavy_infantry = { 600 600 }
				archers = { 400 400 }
				ogre_maneater = { 150 150 }
			}
		}
	}

	option = { # Pay the ogre tribes to keep the peace
		name = EVTOPTA_pigbarter_compact_1
		ai_chance = { factor = 50 }
		wealth = -50

			add_province_modifier = {
				name = pigbarter_ogre_peace

		}

			add_province_modifier = {

# Yearly watch over Pigbarter (hidden)
# Sends help from the houses' kin when Pigbarter is attacked, and now and
# then a tribute demand. Demands are kept rare: none in the first five
# years, at least six years apart (eight after paying), only from strong
# non-human neighbours, and never from ogres while the ogre peace holds.
character_event = {
	id = pigbarter_compact.4
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		NOT = { has_global_flag = pigbarter_fallen }
	}

	immediate = {
		if = {
			limit = {
				has_landed_title = d_pigbarter
				1274 = { owner = { same_realm = ROOT } }
			}
			# Kin abroad send help when Pigbarter is attacked
			if = {
				limit = {
					any_war = { defender = { character = ROOT } }
					NOT = { has_character_modifier = pigbarter_aid_timer }
				}
				character_event = { id = pigbarter_compact.6 }
			}
			# Tribute demands
			else_if = {
				limit = {
					year = 2015
					war = no
					NOT = { has_character_modifier = pigbarter_tribute_timer }
					any_neighbor_independent_ruler = { pigbarter_tribute_demander_trigger = yes }
				}
				random_list = {
					20 = { character_event = { id = pigbarter_compact.5 } }
					80 = { }
				}
			}
		}
		character_event = { id = pigbarter_compact.4 days = 365 }
	}
}

# Pigbarter falls (hidden): the houses go home
# Fired from on_new_holder and on_new_holder_usurpation.
# ROOT = new holder, FROM = title, FROMFROM = old holder.
# The house members who aren't landed leave for their house's home:
# - House of Yabbas (Arabyan): Copher
# - Westerlander house: Marienburg (the League if it exists)
# - Cathayan house: back along the eastern trade roads to Cathay
# - Reman house: Remas
# - Strigany house: Sylvania
# If a home has no holder they go to another ruler of their people, and if
# there is none of those either, to another merchant republic.
character_event = {
	id = pigbarter_compact.10
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		FROM = { title = c_pigbarter }
		has_global_flag = pigbarter_compact
		NOT = { has_global_flag = pigbarter_fallen }
		NOR = {
			is_merchant_republic = yes
			is_patrician = yes
			any_liege = { is_merchant_republic = yes }
		}
	}

	immediate = {
		set_global_flag = pigbarter_fallen
		1274 = { remove_province_modifier = pigbarter_compact }

		# House of Yabbas: Copher
		c_copher = {
			holder_scope = { save_event_target_as = pigbarter_home_araby }
		}
		if = {
			limit = { NOT = { event_target:pigbarter_home_araby = { always = yes } } }
			random_independent_ruler = {
				limit = { culture_group = arabyan_group }
				save_event_target_as = pigbarter_home_araby
			}
		}
		# Westerlander house: Marienburg
		if = {
			limit = { k_marienburg_league = { has_holder = yes } }
			k_marienburg_league = { holder_scope = { save_event_target_as = pigbarter_home_westerland } }
		}
		else = {
			c_marienburg = { holder_scope = { save_event_target_as = pigbarter_home_westerland } }
		}
		if = {
			limit = { NOT = { event_target:pigbarter_home_westerland = { always = yes } } }
			random_independent_ruler = {
				limit = { culture_group = imperial_group }
				save_event_target_as = pigbarter_home_westerland
			}
		}
		# Cathayan house: Cathay
		offmap_cathay = { governor = { save_event_target_as = pigbarter_home_cathay } }
		if = {
			limit = { NOT = { event_target:pigbarter_home_cathay = { always = yes } } }
			random_independent_ruler = {
				limit = { culture_group = oriental_group }
				save_event_target_as = pigbarter_home_cathay
			}
		}
		# Reman house: Remas
		c_remas = { holder_scope = { save_event_target_as = pigbarter_home_tilea } }
		if = {
			limit = { NOT = { event_target:pigbarter_home_tilea = { always = yes } } }
			random_independent_ruler = {
				limit = { culture_group = tilean_group }
				save_event_target_as = pigbarter_home_tilea
			}
		}
		# Strigany house: Sylvania
		k_sylvania = { holder_scope = { save_event_target_as = pigbarter_home_strigany } }
		if = {
			limit = { NOT = { event_target:pigbarter_home_strigany = { always = yes } } }
			random_independent_ruler = {
				limit = {
					OR = {
						culture_group = strigany_group
						religion_group = necromantic_group
					}
				}
				save_event_target_as = pigbarter_home_strigany
			}
		}
		# Fallback: another merchant republic
		random_independent_ruler = {
			limit = {
				is_merchant_republic = yes
				NOT = { character = ROOT }
			}
			preferred_limit = { has_landed_title = e_communards }
			save_event_target_as = pigbarter_home_any
		}

		any_character = {
			limit = {
				is_ruler = no
				prisoner = no
				OR = {
					dynasty = 48135
					dynasty = 13583
					dynasty = 9192
					dynasty = 43176
					dynasty = 87072
				}
			}
			if = {
				limit = { dynasty = 48135 event_target:pigbarter_home_araby = { always = yes } }
				move_character = event_target:pigbarter_home_araby
			}
			else_if = {
				limit = { dynasty = 13583 event_target:pigbarter_home_westerland = { always = yes } }
				move_character = event_target:pigbarter_home_westerland
			}
			else_if = {
				limit = { dynasty = 9192 event_target:pigbarter_home_cathay = { always = yes } }
				move_character = event_target:pigbarter_home_cathay
			}
			else_if = {
				limit = { dynasty = 43176 event_target:pigbarter_home_tilea = { always = yes } }
				move_character = event_target:pigbarter_home_tilea
			}
			else_if = {
				limit = { dynasty = 87072 event_target:pigbarter_home_strigany = { always = yes } }
				move_character = event_target:pigbarter_home_strigany
			}
			else_if = {
				limit = { event_target:pigbarter_home_any = { always = yes } }
				move_character = event_target:pigbarter_home_any
			}
		}

		any_player = {
			limit = {
				OR = {
					is_within_diplo_range = ROOT
					is_merchant_republic = yes
					is_patrician = yes
					dynasty = 48135
					dynasty = 13583
					dynasty = 9192
					dynasty = 43176
					dynasty = 87072
				}
			}
			narrative_event = { id = pigbarter_compact.11 }
		}
	}
}
