### The Rise of the Marienburg League
# Adapted from vanilla "The Rise of the Hansa" (REP.40-77).
# The Westerland guilds ask their ruler to charter a merchant league in a
# northern port city. If permitted, a new merchant republic forms there,
# takes nearby trade posts and sets up its patrician houses.
# REP.40 is fired for the ruler of Westerland at game start (51000).

# Guilds seek a charter (setup)
character_event = {
	id = REP.40
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		OR = {
			has_landed_title = e_westerland
			has_landed_title = k_westerland
		}
		independent = yes
		religion_group = old_world_group
		NOT = { has_global_flag = hansa_formed }
		any_vassal = { is_merchant_republic = yes }
		check_variable = { which = global_marienburg_expansion which <= 10 }
	}

	immediate = {
		random_realm_title = {
			limit = {
				tier = baron
				holding_type = city
				location = {
					port = yes
					OR = {
						region = world_old_world_empire_north
						region = world_old_world_empire_wasteland
					}
				}
				owner = {
					NOR = {
						is_patrician = yes
						is_merchant_republic = yes
						any_liege = { is_merchant_republic = yes }
					}
				}
			}
			save_event_target_as = RepublicanCity
		}
		if = {
			limit = { event_target:RepublicanCity = { always = yes } }
			random_vassal = {
				limit = { is_merchant_republic = yes }
				save_event_target_as = RepublicanLeader
				character_event = { id = REP.41 }
			}
		}
	}
}

					}
				}
