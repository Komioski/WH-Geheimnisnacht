# Voices of the two murderous orders for events/WHG_assassins_events.txt:
# the Hashishin of Ormazd (the_assassins) and Clan Eshin of the Skaven
# (eshin_society). A character speaks for Clan Eshin if he is one of its
# members, or, outside the Hashishin, is joining it, owes it gold or trains
# for it (the flags assassins_joining_eshin, assassins_debt_eshin and
# assassins_protege_eshin); everyone else speaks for the Hashishin.
# GetAssnRank replaces GetSocietyRank, which another file overrides for the
# Norscans alone, and GetAssnOutcry the vanilla GetAssassinsCrimeOutcry.

defined_text = {
	name = GetAssnOrder
	text = {
		localisation_key = assn_order_eshin
		trigger = {
			OR = {
			society_member_of = eshin_society
			AND = {
				NOT = { society_member_of = the_assassins }
				OR = {
					has_character_flag = assassins_joining_eshin
					has_character_flag = assassins_debt_eshin
					has_character_flag = assassins_protege_eshin
				}
			}
		}
		}
	}
	fallback_text = {
		localisation_key = assn_order_hashishin
	}
}

# Scope: the victim; event_target:assassin is the killer. What the
# assassin shouts as he strikes, by why his order wants the victim dead.
defined_text = {
	name = GetAssnOutcry
	text = {
		localisation_key = assn_outcry_marked
		trigger = {
			OR = {
				has_character_flag = assassins_target_marked_for_death
				has_character_flag = eshin_target_marked_for_death
			}
		}
	}
	text = {
		localisation_key = assn_outcry_blood
		trigger = {
			NOR = {
				has_character_flag = assassins_target_marked_for_death
				has_character_flag = eshin_target_marked_for_death
			}
			OR = {
				has_character_flag = assassins_target_killed_member
				has_character_flag = assassins_target_assaulted_member
			}
		}
	}
	text = {
		localisation_key = assn_outcry_infidels
		trigger = {
			NOR = {
				has_character_flag = assassins_target_marked_for_death
				has_character_flag = eshin_target_marked_for_death
				has_character_flag = assassins_target_killed_member
				has_character_flag = assassins_target_assaulted_member
			}
			OR = {
				has_character_flag = assassins_target_war_attacker
				has_character_flag = assassins_target_war_attacker_vassal
				has_character_flag = assassins_target_war_attacker_commander
				has_character_flag = assassins_target_persecutor
			}
		}
	}
	text = {
		localisation_key = assn_outcry_council
		trigger = {
			NOR = {
				has_character_flag = assassins_target_marked_for_death
				has_character_flag = eshin_target_marked_for_death
				has_character_flag = assassins_target_killed_member
				has_character_flag = assassins_target_assaulted_member
				has_character_flag = assassins_target_war_attacker
				has_character_flag = assassins_target_war_attacker_vassal
				has_character_flag = assassins_target_war_attacker_commander
				has_character_flag = assassins_target_persecutor
			}
			has_character_flag = assassins_target_upstart
		}
	}
	text = {
		localisation_key = assn_outcry_chaos
		trigger = {
			NOR = {
				has_character_flag = assassins_target_marked_for_death
				has_character_flag = eshin_target_marked_for_death
				has_character_flag = assassins_target_killed_member
				has_character_flag = assassins_target_assaulted_member
				has_character_flag = assassins_target_war_attacker
				has_character_flag = assassins_target_war_attacker_vassal
				has_character_flag = assassins_target_war_attacker_commander
				has_character_flag = assassins_target_persecutor
				has_character_flag = assassins_target_upstart
			}
			has_character_flag = assassins_target_chaos_cultist
		}
	}
	text = {
		localisation_key = assn_outcry_eshin
		trigger = {
			NOR = {
				has_character_flag = assassins_target_marked_for_death
				has_character_flag = eshin_target_marked_for_death
				has_character_flag = assassins_target_killed_member
				has_character_flag = assassins_target_assaulted_member
				has_character_flag = assassins_target_war_attacker
				has_character_flag = assassins_target_war_attacker_vassal
				has_character_flag = assassins_target_war_attacker_commander
				has_character_flag = assassins_target_persecutor
				has_character_flag = assassins_target_upstart
				has_character_flag = assassins_target_chaos_cultist
			}
			event_target:assassin = { society_member_of = eshin_society }
		}
	}
	fallback_text = {
		localisation_key = assn_outcry_hashishin
	}
}
