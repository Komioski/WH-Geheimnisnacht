# How the raised stone is first seen: a Norscan stone on the hill, a dawi
# stone cut into the ancestor hall.
# Scope: the commissioner.
defined_text = {
	name = GetRunestoneOpening
	text = {
		trigger = { culture_group = dwarf_group }
		localisation_key = runestone_opening_dawi
	}
	text = {
		localisation_key = runestone_opening_norsca
	}
}

# How the commissioner leaves the stone.
# Scope: the commissioner.
defined_text = {
	name = GetRunestoneClosing
	text = {
		trigger = { culture_group = dwarf_group }
		localisation_key = runestone_closing_dawi
	}
	text = {
		localisation_key = runestone_closing_norsca
	}
}

# Whose child the commissioner was: both parents, one, or neither known.
# Scope: the commissioner.
defined_text = {
	name = GetRunestoneLineage
	text = {
		trigger = {
			father_even_if_dead = { always = yes }
			mother_even_if_dead = { always = yes }
		}
		localisation_key = runestone_lineage_both
	}
	text = {
		trigger = {
			father_even_if_dead = { always = yes }
			NOT = { mother_even_if_dead = { always = yes } }
		}
		localisation_key = runestone_lineage_father
	}
	text = {
		trigger = {
			mother_even_if_dead = { always = yes }
			NOT = { father_even_if_dead = { always = yes } }
		}
		localisation_key = runestone_lineage_mother
	}
	text = {
		localisation_key = runestone_lineage_wild
	}
}

# The dead parent, named as the commissioner's father or mother and, if
# known, as the child of their own father or mother.
# Scope: the commissioner; event target runestone_parent.
defined_text = {
	name = GetRunestoneParentLineage
	text = {
		trigger = {
			has_character_flag = runestone_for_father
			event_target:runestone_parent = { father_even_if_dead = { always = yes } }
		}
		localisation_key = runestone_parent_lineage_father_known
	}
	text = {
		trigger = { has_character_flag = runestone_for_father }
		localisation_key = runestone_parent_lineage_father
	}
	text = {
		trigger = {
			event_target:runestone_parent = { mother_even_if_dead = { always = yes } }
		}
		localisation_key = runestone_parent_lineage_mother_known
	}
	text = {
		localisation_key = runestone_parent_lineage_mother
	}
}
